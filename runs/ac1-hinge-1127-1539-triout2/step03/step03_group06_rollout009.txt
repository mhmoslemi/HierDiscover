"""
Gradient‐based softmax minimisation of the peak of a self‑convolution.
The algorithm directly minimises the maximal entry of a*a while
maintaining the ℓ₁–norm √(2 n).  It uses an exponential‑gradient update
with a soft‑max approximation of the max, a momentum buffer and occasional
jitter.  Several independent restarts over different lengths are run
until the 1000 s budget expires.
"""

import time, math, numpy as np
try: from scipy.optimize import linprog; SCIPY = True
except Exception: SCIPY = False

def _conv_max_fft(a):
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    idx = int(np.argmax(conv))
    return conv, float(conv[idx]), idx

def _softmax_grad(a, alpha):
    """
    Returns (gradient, current max value) where the gradient is the
    sub‑gradient of the soft‑max of the self‑convolution.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]

    max_val = float(conv.max())
    # soft‑max weights, numerically stable
    # w_k ∝ exp(alpha * (conv_k - max_val))
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient_i = 2 Σ_j a_j * w_{i+j}
    # compute via convolution of rev(a) with w
    rev = a[::-1]
    conv_len2 = n + conv_len - 1
    fft_len2 = 1 << (conv_len2 - 1).bit_length()
    rev_fft = np.fft.rfft(rev, n=fft_len2)
    w_fft = np.fft.rfft(w, n=fft_len2)
    grad_rev = np.fft.irfft(rev_fft * w_fft, n=fft_len2)[:conv_len2]
    start = n - 1
    grad = 2.0 * grad_rev[start:start + n]
    return grad, max_val

def _run_opt_softmax(a0, deadline,
                     eta0=0.7, beta=0.9, decay=0.9999,
                     alpha0=0.4, alpha_mul=1.015,
                     jitter_prob=0.015,
                     max_iters=200000):
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)
    # ensure the scale
    total = a.sum()
    if total == 0.0:
        a.fill(S / n)
    else:
        a *= (S / total)

    v = np.zeros_like(a)
    eta = eta0
    alpha = alpha0
    best_max = float('inf')
    best_seq = a.copy()
    it = 0
    last_imp = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max = _softmax_grad(a, alpha)

        # momentum
        v = beta * v + (1.0 - beta) * grad

        # exponentiated update
        a = a * np.exp(-eta * v)

        # renormalize
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # jitter
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # schedule updates
        eta *= decay
        alpha *= alpha_mul

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()
            last_imp = it

        if time.time() >= deadline - 0.1:
            break

    return best_seq, best_max

def _lp_refine(f, deadline, max_steps=4):
    if not SCIPY:
        return f
    n = f.shape[0]
    if n > 3000:
        return f
    # Build convolution matrix H such that (H @ g) = f * g
    H = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    S = math.sqrt(2.0 * n)
    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        c = -np.ones(n)  # maximise sum,gives minimise -sum
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=H, b_ub=np.full(2 * n - 1, M),
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:
            break
        if not res.success:
            break
        g0 = res.x
        s = g0.sum()
        if s == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s)

"""
Soft‑max based exponentiated‑gradient search for a step function that
minimises the peak of its self‑convolution.

The external score is    evaluate_sequence(a) = 2·n·max(a*a)/||a||₁²,
which for a vector normalised to ℓ₁‑norm √(2 n) equals the maximal
convolution entry.  The optimiser keeps the ℓ₁‑norm fixed, works with a
soft‑max approximation of the max (log‑sum‑exp), and updates the
coefficients with a multiplicative (exponentiated) rule plus momentum.
Occasional random jitter helps escape plateaus.  Several independent
restarts with different lengths are performed until the 1000 s wall‑clock
budget expires, and promising candidates are refined by a tiny LP
whenever SciPy is available.

The function `propose_candidate` is the entry point required by the
harness and returns the best sequence found (a plain list of non‑negative
floats).
"""

import time
import math
import numpy as np

# optional linear‑programming refinement
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    _SCIPY_AVAILABLE = False


def _conv_max_fft(a: np.ndarray):
    """Full self‑convolution of ``a`` using FFT; returns (conv, max, argmax)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    idx = int(np.argmax(conv))
    return conv, float(conv[idx]), idx


def _softmax_grad(a: np.ndarray, alpha: float):
    """
    Returns the gradient of the soft‑max of the self‑convolution and the
    true maximal convolution value.
    Gradient_i = 2·∑_j a_j·w_{i+j}, where w_k ∝ exp(alpha·(c_k−max_c)).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()

    # self‑convolution a*a
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]

    max_val = float(conv.max())
    # stable soft‑max weights
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2·(rev(a) * w) restricted to n entries
    rev = a[::-1]
    conv_len2 = n + conv_len - 1
    fft_len2 = 1 << (conv_len2 - 1).bit_length()
    rev_fft = np.fft.rfft(rev, n=fft_len2)
    w_fft = np.fft.rfft(w, n=fft_len2)
    grad_rev = np.fft.irfft(rev_fft * w_fft, n=fft_len2)[:conv_len2]

    start = n - 1                         # position of index 0 in grad_rev
    grad = 2.0 * grad_rev[start:start + n]
    return grad, max_val


def _run_opt_softmax(a0: np.ndarray,
                     deadline: float,
                     eta0: float = 0.7,
                     beta: float = 0.9,
                     decay: float = 0.9999,
                     alpha0: float = 0.4,
                     alpha_mul: float = 1.015,
                     max_alpha: float = 8.0,
                     jitter_prob: float = 0.015,
                     max_iters: int = 80000):
    """
    Exponentiated‑gradient descent on the soft‑max surrogate.
    Returns the best vector seen and its exact maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # normalise to the required ℓ₁‑norm
    tot = a.sum()
    a = a * (S / tot) if tot > 0 else np.full(n, S / n)

    v = np.zeros_like(a)                 # momentum buffer
    eta = eta0
    alpha = alpha0

    best_max = float('inf')
    best_seq = a.copy()
    last_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline - 0.1:
            break

        grad, cur_max = _softmax_grad(a, alpha)

        # momentum
        v = beta * v + (1.0 - beta) * grad

        # multiplicative (exponentiated) update – guarantees positivity
        a = a * np.exp(-eta * v)

        # renormalise
        tot = a.sum()
        a = a * (S / tot) if tot > 0 else np.full(n, S / n)

        # occasional jitter to escape tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a = a * (S / a.sum())

        # schedule updates
        eta *= decay
        alpha = min(alpha * alpha_mul, max_alpha)

        # record improvement (exact max, not the soft‑max)
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()
            last_improve = it

        # stop early if we have been idle for a while
        if it - last_improve > 3000:
            break

    return best_seq, best_max


def _lp_refine(f: np.ndarray, deadline: float, max_steps: int = 4):
    """
    Tiny linear‑programming refinement (Frank‑Wolfe style).  It solves
    maximise Σ g_i  subject to g_i≥0 and (f*g)_k ≤ M, where M is the
    current maximal convolution entry.  A few Frank‑Wolfe steps often
    lower the peak a little.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:                     # keep the dense LP tractable
        return f

    S = math.sqrt(2.0 * n)

    # build convolution matrix H  ( (H @ g)[k] = (f * g)[k] )
    H = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break
        conv = np.convolve(f, f)
        M = conv.max()

        c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
        bounds = [(0, None) for _ in range(n)]
        try:
            res = linprog(c, A_ub=H, b_ub=np.full(2 * n - 1, M),
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:                 # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s = g0.sum()
        if s == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s)

        # convex combination and re‑normalisation
        t = 0.2
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f
