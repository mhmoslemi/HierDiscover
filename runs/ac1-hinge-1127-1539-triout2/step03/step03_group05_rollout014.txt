"""
Propose a non‑negative step‑function (vector of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

by combining a diverse parametric sweep, a simulated‑annealing style
random‑mass‑transfer search and a short projected‑Adam refinement.
The search respects a ~1000 s wall‑clock budget and always returns the
best vector encountered.

The main ideas are:
* Try many analytic families (exponential, power‑law, linear, symmetric U‑shapes)
  and random Dirichlet draws to obtain a good starting point.
* Perform a massive Monte‑Carlo local search where a tiny amount of mass is
  moved from a source index to a destination index.  Moves that lower the
  objective are always accepted; uphill moves are accepted with probability
  exp(‑Δ/T) where the temperature T is slowly cooled.
* Occasionally inject a small Gaussian jitter and re‑project onto the ℓ₁‑simplex
  (sum = √(2 n)) to keep the mass constraint.
* After the stochastic phase run a few projected‑Adam steps on the
  log‑sum‑exp smoothed version of the max‑convolution to polish the
  result.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper functions – no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of a vector using FFT.
    Returns an array of length 2*len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ x_i = target }.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_exponential_decay(
    n: int, target_sum: float, rate: float
) -> np.ndarray:
    """
    a_i ∝ exp(-rate * i).  Result is projected onto the ℓ₁‑simplex.
    rate ≥ 0 (rate = 0 gives the uniform vector).
    """
    raw = np.exp(-rate * np.arange(n, dtype=np.float64))
    return project_onto_simplex(raw, target_sum)


def generate_power_decay(
    n: int, target_sum: float, power: float
) -> np.ndarray:
    """
    a_i ∝ (n-i) ** power (monotonically decreasing).
    power ≥ 0 (power = 0 gives the uniform vector).
    """
    raw = (np.arange(n, 0, -1, dtype=np.float64)) ** power
    return project_onto_simplex(raw, target_sum)


def generate_linear_U(
    n: int,
    target_sum: float,
    param: float = 0.0,
) -> np.ndarray:
    """
    Symmetric 'U' shape: 1 + param * |i-mid|/mid.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = 1.0 + param * d
    return project_onto_simplex(raw, target_sum)


def random_dirichlet(n: int, target_sum: float, alpha: float = 0.5) -> np.ndarray:
    """
    Sample a Dirichlet‑like vector (Gamma draws) and project onto the simplex.
    Small alpha concentrates mass on a few coordinates.
    """
    raw = np.random.gamma(shape=alpha, scale=1.0, size=n)
    return project_onto_simplex(raw, target_sum)


def transfer_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Move a fraction `scale` of the mass from a random source index to a
    random destination index.  Sum stays exactly the same.
    Returns a new array (original is untouched).
    """
    n = a.shape[0]
    # choose a source that actually holds some mass
    src_candidates = np.where(a > 1e-14)[0]
    if src_candidates.size == 0:
        return a.copy()
    i = np.random.choice(src_candidates)
    j = np.random.randint(n)
    if i == j:
        return a.copy()
    delta = scale * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    return out


def jitter_and_project(a: np.ndarray, target_sum: float, std_factor: float) -> np.ndarray:
    """
    Add small Gaussian noise (std = std_factor * mean(a)) and re‑project
    onto the simplex to keep feasibility.
    """
    noise = np.random.randn(a.size) * std_factor * np.mean(a)
    noisy = np.maximum(a + noise, 0.0)
    return project_onto_simplex(noisy, target_sum)


def smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed maximum of the self‑convolution.
    Returns (gradient, current max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)

    # stable log‑sum‑exp weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w * rev(a)) restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def adam_refine(a: np.ndarray, target_sum: float, iters: int = 300):
    """
    A short projected‑Adam optimisation on the smoothed objective.
    Starts from `a` (which should already satisfy the simplex constraint).
    """
    lr = 0.02 * target_sum / a.size
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, iters + 1):
        grad, cur_max = smoothed_gradient(a, tau_factor=0.001)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)
    return a


def objective_maxconv(a: np.ndarray) -> float:
    """Return max(conv(a,a)); the scale‑invariant objective under our normalisation."""
    return conv_self_fft(a).max()
