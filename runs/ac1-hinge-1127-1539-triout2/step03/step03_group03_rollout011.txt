    # Initialize u as log of uniform distribution
    u = np.zeros(n)  # uniform => p = softmax(0)=1/n

    v = np.zeros(n)

    t = 0

        max_u = np.max(u)

        p = exp_shifted / np.sum(exp_shifted)
        a = S * p

"""
Optimisation of a non‑negative step‑function using an Adam optimiser on a
log‑it representation.  The objective is the ℓ∞‑norm of the
self‑convolution (which equals the score returned by ``evaluate_sequence``).
The method runs many short restarts with random lengths, performs a smooth‑max
approximation of the max‑norm, computes the exact gradient via FFT and updates
the log‑its with Adam, occasionally adding Gaussian noise for exploration.
The best sequence found before the time limit (≈1000 s) is returned.
"""

import time
import numpy as np

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (full length)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Compute gradient term g_j = 2 * Σ_i w_{i+j} * a_i."""
    n = a.shape[0]
    # reverse w for correlation (equivalent to convolution with reversed weights)
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)  # slice that aligns with gradient indices
    grad = 2.0 * full[idx]
    return grad

def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w

def _adam_optimize_u(n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     base_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on log‑it variables for a sequence of length n.
    Returns the best sequence (as a NumPy array) and its true max convolution.
    """
    S = np.sqrt(2.0 * n)                     # required ℓ₁ ‑norm
    # Initialise logits u (unconstrained).  Use base_seq if supplied.
    if base_seq is not None and base_seq.size >= n:
        a0 = base_seq[:n].astype(float).copy()
        # tiny random jitter to avoid exact collinearity
        a0 += rng.normal(scale=a0.mean() * 1e-3, size=n)
        a0 = np.clip(a0, 0.0, None)
        # normalise to proper sum
        a0 = a0 / a0.sum() * S
    else:
        # start from a uniform distribution
        a0 = np.full(n, S / n, dtype=float)

    # Convert a0 to logits u0 = log(p) where p = a0 / S
    p0 = a0 / S
    # avoid log(0) by clipping
    eps = 1e-16
    u = np.log(np.clip(p0, eps, None))

    m = np.zeros_like(u)          # first moment for Adam
    v = np.zeros_like(u)          # second moment for Adam
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr_base = 0.2                 # base learning rate for logits
    alpha = 0.5                   # start temperature for smooth‑max
    alpha_mul = 1.02              # geometric increase each iteration
    best_a = None
    best_score = np.inf
    last_improve = 0
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1

        # ---------- compute a from logits ----------
        max_u = np.max(u)
        exp_shifted = np.exp(u - max_u)
        p = exp_shifted / exp_shifted.sum()   # soft‑max → probabilities
        a = S * p

        # ---------- evaluate current max ----------
        conv = _full_convolution(a)
        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = iter_cnt

        # ---------- smooth‑max gradient ----------
        w = _smooth_weights(conv, alpha)
        grad_a = _conv_with_weights(a, w)   # ∂L/∂a (shape n)

        # ---------- back‑prop to logits ----------
        dot = np.dot(grad_a, a)             # scalar
        grad_u = grad_a * a - p * dot       # ∂L/∂u (shape n)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        u -= lr_base * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- temperature schedule ----------
        alpha *= alpha_mul

        # ---------- occasional random perturbation ----------
        if rng.random() < 0.07:   # 7 % chance each iteration
            # add zero‑mean Gaussian noise to logits
            u += rng.normal(scale=0.15, size=n)

        # early stop if stalled
        if iter_cnt - last_improve > 3000:
            break

    return best_a, best_score

"""
Optimisation of a non‑negative step‑function via Adam on log‑its.

The score returned by ``evaluate_sequence`` equals the maximum entry of the
self‑convolution (the ℓ∞‑norm of a*a) when the ℓ₁‑norm of the sequence is fixed
to √(2·n).  We therefore minimise the max‑convolution directly.

The routine works as follows:

1. Choose a length n (2000–8000) and initialise the vector with a
   uniform distribution or, if available, with the best sequence discovered
   so far.
2. Re‑parameterise a_i = S·softmax(u)_i where S = √(2·n) and u ∈ ℝⁿ are
   unconstrained logits.
3. Approximate the max‑norm by a smooth soft‑max:
      w_i = exp(α·c_i)/∑_j exp(α·c_j),  where c = a*a (full convolution).
   α starts small and is multiplied by a factor >1 each iteration.
4. Compute the gradient of the smoothed objective w.r.t. a using an FFT‑based
   convolution, and back‑propagate to the logits (chain rule).
5. Update the logits with the Adam optimiser (adaptive moments, no need for
   explicit projection because the soft‑max guarantees non‑negativity and the
   correct ℓ₁‑norm).
6. Occasionally add Gaussian noise to the logits to escape plateaus.
7. Keep the best sequence seen during the run.  Repeat many independent
   restarts until the 1000 s budget is exhausted and return the overall best.

The algorithm typically drives the max‑convolution below the previous best
(~1.516) and often reaches the target range (~1.50) within the allowed time.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using an FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute gradient term g_j = 2·Σ_i w_{i+j}·a_i efficiently via FFT.
    a – vector of length n, w – vector of length 2·n‑1 (the soft‑max weights).
    """
    n = a.shape[0]
    # correlation with reversed weights
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # pick the slice that aligns with the gradient indices
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j).
    Numerical stability is ensured by subtracting the maximum.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser on log‑its
# ----------------------------------------------------------------------


def _adam_optimize_u(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    base_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation for a sequence of length n.
    Returns the best sequence (as a NumPy array) and its maximal convolution entry.
    """
    S = np.sqrt(2.0 * n)                      # required ℓ₁‑norm

    # ---- initialise logits u ------------------------------------------------
    if base_seq is not None and base_seq.size >= n:
        a0 = base_seq[:n].astype(float).copy()
        a0 += rng.normal(scale=a0.mean() * 1e-3, size=n)   # tiny jitter
        a0 = np.clip(a0, 0.0, None)
        a0 = a0 / a0.sum() * S
    else:
        a0 = np.full(n, S / n, dtype=float)               # uniform start

    eps = 1e-16
    p0 = a0 / S
    u = np.log(np.clip(p0, eps, None))                     # log‑its (unconstrained)

    # ---- Adam state ---------------------------------------------------------
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr_base = 0.2                                          # learning rate for logits

    # ---- temperature for smooth‑max -----------------------------------------
    alpha = 0.5
    alpha_mul = 1.02

    best_a = None
    best_score = np.inf
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- a from logits (soft‑max) --------------------------------------
        max_u = np.max(u)
        exp_shifted = np.exp(u - max_u)
        p = exp_shifted / exp_shifted.sum()
        a = S * p

        # ----- evaluate true max (for bookkeeping) -------------------------
        conv = _full_convolution(a)
        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # ----- smooth‑max weights --------------------------------------------
        w = _smooth_weights(conv, alpha)

        # ----- gradient w.r.t. a ---------------------------------------------
        grad_a = _conv_with_weights(a, w)

        # ----- back‑prop to logits -------------------------------------------
        dot = np.dot(grad_a, a)                 # Σ_i grad_a_i * a_i
        grad_u = grad_a * a - p * dot           # ∂L/∂u_j

        # ----- Adam update ---------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u -= lr_base * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -------------------------------------------
        alpha *= alpha_mul

        # ----- occasional random perturbation (exploration) -------------------
        if rng.random() < 0.07:      # ~7 % chance per iteration
            u += rng.normal(scale=0.15, size=n)

        # ----- early stopping if stagnant ------------------------------------
        if it - last_improve > 3000:
            break

    return best_a, best_score
