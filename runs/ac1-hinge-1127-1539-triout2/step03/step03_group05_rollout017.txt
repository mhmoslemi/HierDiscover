    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3
    # get base if any
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)):
        n = len(base)
        a = np.array(base, dtype=float)
    else:
        n = 8000  # choose
        a = np.random.exponential(scale=1.0, size=n)
    target = math.sqrt(2.0 * n)
    # project to simplex to match sum exactly
    a = project_onto_simplex(np.maximum(a, 0.0), target)

    # initial conv
    conv = conv_self_fft(a)
    cur_max = conv.max()
    max_idx = int(conv.argmax())
    best = a.copy()
    best_score = cur_max
    iter = 0
    # hyperparameters
    step_factor = 0.3
    temperature_start = 1e-2
    # loop
    while time.time() < deadline:
        iter += 1
        # dynamic step factor decay
        if iter % 5000 == 0:
            step_factor *= 0.9
        # temperature decays linearly
        elapsed = time.time() - start
        T = temperature_start * max(0.0, 1.0 - elapsed / budget_s)

        move_type = np.random.rand()
        # store revert info
        revert = None
        if move_type < 0.6:
            # active transfer
            low = max(0, max_idx - (n - 1))
            high = min(max_idx, n - 1)
            if low < high:
                i = np.random.randint(low, high + 1)
                if a[i] > 0:
                    delta = step_factor * a[i] * np.random.rand()
                    j = np.random.randint(n)
                    while j == i:
                        j = np.random.randint(n)
                    a[i] -= delta
                    a[j] += delta
                    revert = ('pair', i, j, delta)
        elif move_type < 0.9:
            # random transfer
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] > 0:
                delta = step_factor * a[i] * np.random.rand()
                a[i] -= delta
                a[j] += delta
                revert = ('pair', i, j, delta)
        else:
            # jitter
            a_old = a.copy()
            jitter = np.random.randn(n) * step_factor * target / n
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a, target)
            revert = ('jitter', a_old)

        # compute new max
        conv_new = conv_self_fft(a)
        new_max = conv_new.max()
        new_idx = int(conv_new.argmax())

"""
Propose a non‑negative step‑function (vector of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  The objective simplifies to
`max(conv)` when the ℓ₁‑norm of the vector is fixed to √(2 n).

The algorithm is a single‑threaded simulated‑annealing / hill‑climbing
search that works directly on the simplex:

1. **Initialisation** – start from the previously found best construction
   (if the global variable ``height_sequence_1`` exists) or from a random
   Dirichlet‑type vector normalised to the required sum.

2. **Local moves** – three kinds of mass‑preserving moves are used
   repeatedly (while the 1000 s budget is not exhausted):
        * *active transfer* – move a fraction of mass from an index that
          participates in the currently maximal convolution entry to a
          random other index;
        * *random transfer* – move a fraction of mass between two random
          indices;
        * *jitter* – add a tiny Gaussian noise to the whole vector and
          re‑project onto the ℓ₁‑simplex (occasionally, to escape plateaus).

   After each move the full convolution is recomputed (FFT based).  The new
   state is accepted if the maximal convolution does not increase; otherwise
   it may be accepted with a temperature‑dependent probability (simulated
   annealing).  The temperature decays linearly to zero.

3. **Best‑ever tracking** – every improvement updates the stored best
   sequence.  The search stops early if the bound 1.503 is reached.

All helper routines are top‑level, use only NumPy and the standard library,
and respect the 1000 s time limit.  The returned list is guaranteed to be
non‑negative and to sum to √(2 n), so the security checks in
``evaluate_sequence`` will never reject it.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a vector with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = target}.
    O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function that minimises the evaluation function."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a known construction if available
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        a = np.array(base, dtype=np.float64)
    else:
        # a comfortable size – large enough to expose many degrees of freedom
        n = 8000
        a = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)          # sum that normalises the objective
    a = project_onto_simplex(np.maximum(a, 0.0), target_sum)

    # initial convolution statistics
    conv = conv_self_fft(a)
    cur_max = float(conv.max())
    max_idx = int(conv.argmax())

    best = a.copy()
    best_score = cur_max
    it = 0

    # ------------------------------------------------------------------
    # Hyper‑parameters (they adapt slowly during the run)
    # ------------------------------------------------------------------
    step_factor = 0.30                     # maximal fraction of a source value moved
    temperature_start = 1e-2               # initial annealing temperature
    jitter_prob = 0.005                    # how often a global jitter is tried
    active_prob = 0.60                     # probability of an "active" move
    random_prob = 0.35                     # probability of a pure random transfer

    print(f"[{time.time():.2f}] start search – n={n}, initial max={cur_max:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # --- schedule adaptations -------------------------------------------------
        if it % 5000 == 0:
            step_factor = max(0.01, step_factor * 0.90)   # slowly shrink steps

        elapsed = time.time() - start
        # linear temperature decay to zero
        T = temperature_start * max(0.0, 1.0 - elapsed / budget_s)

        # --- decide which move to perform ----------------------------------------
        move_type = np.random.rand()
        revert = None            # information to roll back a rejected move

        if move_type < active_prob:
            # ---- ACTIVE TRANSFER (focus on indices that build the current max) ----
            low = max(0, max_idx - (n - 1))
            high = min(max_idx, n - 1)
            if low <= high:
                i = np.random.randint(low, high + 1)
                j = max_idx - i                     # the partner that appears in the max entry
                # pick the larger of the two as source
                src = i if a[i] >= a[j] else j
                if a[src] > 0:
                    delta = step_factor * a[src] * np.random.rand()
                    dst = np.random.randint(n)
                    while dst == src:
                        dst = np.random.randint(n)
                    a[src] -= delta
                    a[dst] += delta
                    revert = ('pair', src, dst, delta)
        elif move_type < active_prob + random_prob:
            # ---- RANDOM PAIR TRANSFER ------------------------------------------------
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] > 0:
                delta = step_factor * a[i] * np.random.rand()
                a[i] -= delta
                a[j] += delta
                revert = ('pair', i, j, delta)
        else:
            # ---- SMALL JITTER (global perturbation) ---------------------------------
            if np.random.rand() < jitter_prob:
                a_old = a.copy()
                jitter = np.random.randn(n) * step_factor * target_sum / n
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = project_onto_simplex(a, target_sum)
                revert = ('jitter', a_old)

        # -------------------------------------------------------------------------
        # Evaluate the new state
        # -------------------------------------------------------------------------
        new_conv = conv_self_fft(a)
        new_max = float(new_conv.max())
        new_idx = int(new_conv.argmax())

        # -------------------------------------------------------------------------
        # Acceptance test (simulated annealing / hill‑climbing)
        # -------------------------------------------------------------------------
        accept = False
        if new_max <= cur_max + 1e-14:                     # improvement or negligible rise
            accept = True
        else:
            if T > 0 and np.random.rand() < math.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            cur_max = new_max
            max_idx = new_idx
            if new_max < best_score - 1e-12:
                best_score = new_max
                best = a.copy()
                print(f"[{time.time():.2f}] iter {it:07d} – improved to {best_score:.6f}")
            if best_score < 1.503:       # reached the target bound – stop early
                print(f"[{time.time():.2f}] target bound reached – stopping")
                break
        else:
            # ---- REVERT REJECTED MOVE -------------------------------------------
            if revert is not None:
                kind = revert[0]
                if kind == 'pair':
                    _, src, dst, delta = revert
                    a[src] += delta
                    a[dst] -= delta
                elif kind == 'jitter':
                    _, a_old = revert
                    a = a_old
                # cur_max / max_idx stay unchanged

        # occasional clean‑up to avoid numerical drift
        if it % 10000 == 0:
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Return the best vector discovered
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] search finished – best max(conv) = {best_score:.6f} after {it} iters")
    return best.tolist()
