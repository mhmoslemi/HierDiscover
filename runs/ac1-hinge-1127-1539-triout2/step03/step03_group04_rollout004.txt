def upsample(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    # enforce symmetry: new_a = (new_a + new_a[::-1]) / 2
    new_a = (new_a + new_a[::-1]) / 2
    return new_a
```

def _adam_gradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0*n)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 512.0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = target_sum * 0.2  # base learning rate

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1*m + (1 - beta1) * g
        v = beta2*v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # enforce nonnegativity + simplex
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # Symmetrize
        a = (a + a[::-1]) * 0.5
        a = _project_to_simplex(a, target_sum)

        # Occasionally jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum*0.001/(beta+1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            a = (a + a[::-1]) * 0.5
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD step={it} beta={beta:.2f} score={best_val:.6f}")

        # increase beta
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta*1.5, max_beta)

        it += 1
        # break if improvements become negligible
        if it > 5000 and (time.time() - (deadline - (deadline - (deadline - time.time())))) < 0: pass # ignore.

    return best_a, best_val
```

"""
A multi‑stage optimizer for the max‑convolution / ℓ¹‑norm benchmark.

* Stage 0 – initialise from the previous best (if available) or from a
  random power‑law shape, optionally up‑sampled to the target length.
* Stage 1 – Adam‐based projected gradient descent on a soft‑max
  approximation of the ℓ∞‑norm of the self‑convolution.
  The temperature β is annealed, the iterates are kept on the simplex,
  and the vector is forced to be symmetric after each step.
* Stage 2 – deterministic guided hill‑climbing:
  locate the current maximal convolution entry, move a small fraction of
  mass from a high‑sensitivity index to a low‑sensitivity one, re‑project
  onto the simplex and keep only improving moves.  This quickly removes
  the remaining peak.
* The routine iterates over several lengths (growing geometrically) and
  keeps the overall best sequence found before the time budget expires.

The code respects all competition rules (no I/O, no closures, only NumPy
and the provided `evaluate_sequence` function).  It prints progress
messages and finally returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective (2·n·max(conv)/(sum(a)²))."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:  # should never happen with our projection
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where b = a*a (full convolution) and
    g = ∇ (log‑sum‑exp_{β}(b)) – the gradient of the soft‑max approximation
    of the ℓ∞‑norm of the convolution.
    """
    n = a.shape[0]

    # full convolution via FFT
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the convolution entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # gradient = 2·(a ⋆ p)  (convolution of a reversed with p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1           # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]

    return g, b


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear up‑sampling of a (assumed symmetric) to a new length.
    Afterwards the result is explicitly symmetrised again.
    """
    old_n = a.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = (new_a + new_a[::-1]) * 0.5
    return new_a

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Projected Adam descent on the soft‑max approximation of the max‑convolution.
    The temperature β is annealed; after each step we enforce the simplex
    constraint and symmetry.
    Returns the best vector seen together with its (exact) score.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 1.0            # soft‑max temperature
    max_beta = 512.0
    step_base = target_sum * 0.20   # base learning‑rate scale

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = step_base / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat

        # keep feasible
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # enforce symmetry (does not change the sum)
        a = (a + a[::-1]) * 0.5

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            a = (a + a[::-1]) * 0.5

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD it={it:04d} beta={beta:.2f} score={best_val:.6f}")

        # anneal the temperature
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _guided_hill_climb(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Deterministic hill‑climbing that moves mass from indices that
    contribute heavily to the current maximal convolution entry
    to indices that contribute little.
    Only strictly improving moves are kept.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.07   # at most 7 % of a donor’s mass is moved per step

    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # sensitivity vector: contribution of a[i] to conv[k_max]
        sens = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # indices lo … hi have a partner j = k_max – i
        sens[lo:hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        weight = sens * a                     # larger weight → better donor
        donor = int(np.argmax(weight))
        if weight[donor] <= 0.0:
            break  # nothing to improve

        # choose a receiver among the 10 % lowest weighted indices
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            candidates = np.where(np.arange(n) != donor)[0]
        receiver = int(rng.choice(candidates))

        if donor == receiver or a[donor] <= 1e-12:
            continue

        # move a fraction of donor's mass to the receiver
        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        # enforce feasibility (sum stays constant, but projection removes tiny numerical drift)
        a = _project_to_simplex(a, target_sum)
        a = (a + a[::-1]) * 0.5   # keep symmetry

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH improved={best_val:.6f}")
        else:
            # revert the move
            a[receiver] -= delta
            a[donor] += delta
            # optional tiny random perturbation to avoid stagnation
            if rng.random() < 0.01:
                a += rng.normal(scale=target_sum * 1e-4, size=n)
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, target_sum)
                a = (a + a[::-1]) * 0.5

    return best_a, best_val

            sub_deadline = time.time() + min(45.0, remaining * 0.5)

            # Stage 1 – Adam gradient descent
            a_opt, val = _adam_gradient_descent(a0, sub_deadline, rng)

            # Stage 2 – guided hill climbing (use the overall deadline)
            if time.time() < deadline:
                a_opt, val = _guided_hill_climb(a_opt, deadline, rng)

            # keep the globally best sequence
            if val < best_score:
                best_score = val
                best_seq = a_opt.copy()
                print(f"** GLOBAL BEST ** n={n}  score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)
        best_seq = a

    # Trim trailing zeros for a cleaner output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
