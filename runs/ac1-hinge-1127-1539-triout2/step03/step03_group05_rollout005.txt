def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, ∑x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index to another and project.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def biased_pair_transfer(a: np.ndarray, target_sum: float,
                         grad: np.ndarray, scale: float = 0.02) -> np.ndarray:
    """
    Move a small amount of mass from a high‑gradient index (large contribution
    to the current maximum) to a low‑gradient index.
    """
    n = a.shape[0]
    # pick from the top 20 % of gradient entries
    thresh_hi = np.percentile(grad, 80)
    hi = np.where(grad >= thresh_hi)[0]
    # pick from the bottom 20 %
    thresh_lo = np.percentile(grad, 20)
    lo = np.where(grad <= thresh_lo)[0]
    if len(hi) == 0 or len(lo) == 0:
        # fallback to plain random transfer
        return random_pair_transfer(a, target_sum, scale=scale)

    i = np.random.choice(hi)
    j = np.random.choice(lo)
    delta = scale * a[i] * np.random.rand()
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def smoothed_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the log‑sum‑exp smooth approximation of max(conv(a,a)).
    Returns (gradient, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # stable weights
    w = np.exp(conv / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    # gradient entries correspond to the central n elements of conv_full
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)).
    Returns (gradient, max_conv, idx_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (vector of non‑negative heights) that minimises
    2 n max(conv(a,a)) / (∑a)².  The algorithm combines Adam optimisation on a
    smoothed objective with a simulated‑annealing refinement.  It returns the
    best sequence found before the 1000 s deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Problem size and starting point
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        a = np.array(base, dtype=np.float64)
    else:
        # choose a comfortable size in the 8‑9k range
        n = np.random.randint(7500, 8800)
        a = np.random.dirichlet(alpha=np.full(n, 0.7))  # mildly spiky
        a = a * np.sqrt(2.0 * n)                       # normalise ℓ₁‑norm

    target_sum = math.sqrt(2.0 * n)        # after normalisation objective = max(conv)

    # ensure feasibility
    a = project_onto_simplex(a, target_sum)

    # keep the best solution seen so far
    best_seq = a.copy()
    best_score = conv_self_fft(a).max()
    print(f"[{time.time():.2f}] start – n={n}, initial max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on smoothed max
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # step size (initial learning rate)
    step_lr = 0.5 * (target_sum / n)      # comparable to typical entry magnitude

    # smoothing temperature schedule
    tau = max(0.05 * best_score, 1e-6)    # start fairly smooth
    tau_decay = 0.9995
    min_tau = 1e-9

    iter_cnt = 0
    no_improve = 0

    while time.time() < deadline:
        # compute gradient of the smoothed objective
        grad, cur_max = smoothed_gradient(a, tau)

        # Adam bias‑corrected updates
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        step = step_lr * m_hat / (np.sqrt(v_hat) + eps)

        # update and re‑project onto the ℓ₁‑simplex
        a_new = a - step
        a = project_onto_simplex(a_new, target_sum)

        # keep track of the best (real) max‑convolution value
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            print(f"Iter {iter_cnt:06d} – new best max={best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # anneal the smoothing temperature
        tau = max(tau * tau_decay, min_tau)

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.008:
            a = random_pair_transfer(a, target_sum, scale=0.015)

        # if we have stagnated for a long time, shrink the learning rate
        if no_improve >= 4000:
            step_lr *= 0.6
            no_improve = 0
            print(f"Iter {iter_cnt:06d} – learning‑rate reduced to {step_lr:.4e}")

        # early stop when we already beat the target bound
        if best_score < 1.503:
            print(f"Early stop – target already reached: {best_score:.6f}")
            break

    print(f"[{time.time():.2f}] Adam phase finished – best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing refinement
    # ------------------------------------------------------------------
    cur_a = best_seq.copy()
    cur_max = best_score
    temperature = best_score * 0.25          # initial temperature
    cooling = 0.9998
    sa_iter = 0

    while time.time() < deadline:
        sa_iter += 1
        # compute sub‑gradient of the current maximal entry (helps pick source index)
        grad_max, max_val, idx_max = subgradient_max(cur_a)

        # propose a biased transfer
        cand = biased_pair_transfer(cur_a.copy(), target_sum, grad_max, scale=0.02)

        # evaluate new max‑convolution
        new_max = conv_self_fft(cand).max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / temperature) if temperature > 1e-12 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = cand
            cur_max = new_max

            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_seq = cur_a.copy()
                print(f"SA iter {sa_iter:06d} – improved max={best_score:.6f}")

        # geometric cooling
        temperature = max(temperature * cooling, 1e-12)

        # stop early if we have reached the target
        if best_score < 1.503:
            print(f"Target reached during SA at iter {sa_iter}")
            break

    print(f"[{time.time():.2f}] Simulated‑annealing finished – best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final random polishing (tiny pair transfers)
    # ------------------------------------------------------------------
    for _ in range(80):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.005)
        cand_max = conv_self_fft(cand).max()
        if cand_max < best_score:
            best_score = cand_max
            best_seq = cand.copy()
            print(f"Polish improved max={best_score:.6f}")

    print(f"[{time.time():.2f}] Search completed – final max={best_score:.6f}")
    return best_seq.tolist()
```

This implementation respects the 1000 s time budget, stays within the allowed API, and uses a combination of adaptive Adam optimisation on a smooth surrogate of the objective together with a gradient‑guided simulated‑annealing refinement.  The returned list can be fed directly to `evaluate_sequence`.
