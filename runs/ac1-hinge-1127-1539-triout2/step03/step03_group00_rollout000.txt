"""search_step_function.py

Search for a non‑negative step function that minimizes the competition
objective (the maximum entry of its self‑convolution after ℓ¹‑normalisation).

Algorithm
---------

1. **Length schedule** – examine several lengths up to 20 000 (larger
   lengths give more flexibility for lowering the peak).

2. **Systematic family sweep** – for each length generate candidates from
   a few low‑dimensional families:
        • pure power law      a_i ∝ (i+1)^(-p)
        • pure exponential   a_i ∝ exp(-λ·i)
        • mixed power·exp    a_i ∝ (i+1)^(-p)·exp(-λ·i)
   The parameters `p` and `λ` are sampled on fine grids.  Every vector is
   projected onto the ℓ¹‑simplex {x≥0, Σx = √(2·n)} and its score
   (the maximal convolution entry) is recorded.

3. **Retention** – keep the best *K* candidates per length (K≈5).

4. **Refinement** – each retained candidate is refined by two stages
   (both respecting the simplex constraint):

   a) **Adam optimisation** on a smooth log‑sum‑exp approximation of the
      max‑convolution.  The temperature (β) is annealed from 0.5 up to
      4096, momentum is provided by the Adam moment estimates and the
      iterate is projected back onto the simplex after each step.

   b) **Biased simulated‑annealing** – the gradient of the (soft)
      max‑convolution yields a probability vector that highlights the
      indices participating most in the current peak.  Mass is moved
      preferentially from high‑impact indices to random destinations,
      accepting uphill moves with a slowly decreasing temperature.

5. **Upsampling** – once the best sequence has been found, it is linearly
   interpolated to a larger length (if time permits) and the refinement
   steps are repeated.  Interpolation preserves the shape while enlarging
   the support, which usually reduces the peak further.

6. The algorithm stops when the time budget (default 1000 s) is exhausted
   and returns the best sequence discovered (as a plain Python list).

All helper functions are top‑level, there are no closures or lambdas,
and only the standard library, ``numpy`` and ``scipy`` are used.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation: max entry of a*a."""
    # a is assumed to already satisfy Σa = √(2·n)
    return _max_convolution_fft(a)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto simplex {x_i≥0, Σx_i = s}.
    Duchi et al. (2008) implementation.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of smoothed max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
      = 2·(a ⋆ p) with p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # ---- convolution a*a ---------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))  # stabilised
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    # convolution of a (reversed) with p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on smooth max‑convolution
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    The iterate is projected onto the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 0.5            # initial smoothing
    beta_max = 4096.0
    it = 1

    while time.time() < deadline:
        # gradient of the smooth surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step size – scaled inversely with sqrt of the moving variance
        step = 0.5 * target_sum / (np.sqrt(v_hat) + eps)

        a -= step * m_hat  # descent

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # anneal β (geometric schedule)
        if beta < beta_max:
            beta = min(beta_max, beta * 1.45)

        it += 1

    return a


# ----------------------------------------------------------------------
# Biased simulated‑annealing (mass transfers guided by the gradient)
# ----------------------------------------------------------------------


def _biased_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated annealing that preferentially moves mass from indices
    contributing most to the current peak (as given by the soft‑max gradient).
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _score(a)

    # temperature initialisation (relative to current score)
    T = best_score * 0.02 + 1e-6
    cooling = 0.9997

    rng = np.random.default_rng()
    while time.time() < deadline:
        # obtain a soft‑max gradient to bias the choice of donor index
        g, b = _gradient_softmax(a, beta=2048.0)   # large β ≈ arg‑max
        # positive part of gradient indicates indices that increase the peak
        prob = np.maximum(g, 0.0)
        prob_sum = prob.sum()
        if prob_sum == 0.0:
            donor = rng.integers(0, n)
        else:
            prob = prob / prob_sum
            donor = rng.choice(n, p=prob)

        # choose a distinct recipient uniformly at random
        recip = rng.integers(0, n)
        while recip == donor:
            recip = rng.integers(0, n)

        # transfer a random fraction (up to 15 %) of the donor's mass
        frac = rng.random() * 0.15
        delta = a[donor] * frac
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[recip] += delta

        cur_score = _score(a)
        dE = cur_score - best_score

        # Metropolis acceptance
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))
        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert move
            a[recip] -= delta
            a[donor] += delta

        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Random / parametric initialisers
# ----------------------------------------------------------------------


def _initial_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=float) + 1.0
    a = idx ** (-p)
    return a


def _initial_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    return a


def _initial_mixed(n: int, p: float, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float) + 1.0
    a = (idx ** (-p)) * np.exp(-lam * (idx - 1.0))
    return a


def _normalise_to_simplex(v: np.ndarray, n: int) -> np.ndarray:
    target = np.sqrt(2.0 * n)
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        v = np.full_like(v, target / n)
    else:
        v = target * v / v.sum()
    return v


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample_sequence(a: np.ndarray, n_new: int) -> np.ndarray:
    """Linearly interpolate a to length n_new and re‑project onto simplex."""
    n_old = a.size
    if n_new == n_old:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_interp = np.interp(x_new, x_old, a)
    return _normalise_to_simplex(a_interp, n_new)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the competition objective.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Length schedule – up to 20 000 (larger supports tend to lower the peak)
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 5000, 7000, 10000, 15000, 20000]

    # ------------------------------------------------------------------
    # Parameter grids used for systematic exploration
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.05, 2.0, 39)               # power‑law exponents
    lam_grid = np.geomspace(5e-5, 0.02, 24)          # exponential rates

    # ------------------------------------------------------------------
    # Search loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Phase A – systematic sweep over low‑dimensional families
        # --------------------------------------------------------------
        top_candidates = []  # list of (score, vector)

        # Helper to record a candidate
        def maybe_keep(vec):
            nonlocal best_seq, best_score, top_candidates
            sc = _score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] New global best (grid) n={n} score={sc:.6f}")
            # keep up to 5 best for this length
            if len(top_candidates) < 5 or sc < top_candidates[-1][0]:
                top_candidates.append((sc, vec.copy()))
                top_candidates.sort(key=lambda x: x[0])
                top_candidates = top_candidates[:5]

        # ----- pure power‑law -------------------------------------------------
        for p in p_grid:
            a = _initial_power_law(n, p)
            a = _normalise_to_simplex(a, n)
            maybe_keep(a)

        # ----- pure exponential -----------------------------------------------
        for lam in lam_grid:
            a = _initial_exponential(n, lam)
            a = _normalise_to_simplex(a, n)
            maybe_keep(a)

        # ----- mixed families -------------------------------------------------
        # Sample a subset of the Cartesian product to keep runtime modest
        for p in p_grid[::5]:
            for lam in lam_grid[::4]:
                a = _initial_mixed(n, p, lam)
                a = _normalise_to_simplex(a, n)
                maybe_keep(a)

        # ----- random Dirichlet draws (extra diversity) ----------------------
        for _ in range(15):
            a = rng.dirichlet(alpha=np.ones(n)) * target_sum
            maybe_keep(a)

        # ------------------------------------------------------------------
        # Phase B – refinement of the retained candidates
        # ------------------------------------------------------------------
        for _, seed_vec in top_candidates:
            if time.time() >= deadline:
                break

            # allocate a modest slice of time for each refinement (≈30 s)
            sub_dead = min(deadline, time.time() + 30.0)

            # Adam optimisation on smoothed max‑convolution
            a_adam = _adam_opt(seed_vec, sub_dead)

            # Biased simulated annealing (uses whatever time is left)
            a_refined = _biased_simulated_annealing(a_adam, deadline)

            sc = _score(a_refined)
            if sc < best_score:
                best_score = sc
                best_seq = a_refined.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} score={sc:.6f}")

        # ------------------------------------------------------------------
        # Optional upsampling step – if time permits, enlarge the best vector
        # ------------------------------------------------------------------
        if best_seq is not None and n < lengths[-1]:
            # try to upsample to the next length in the schedule (if any)
            next_n = lengths[lengths.index(n) + 1] if n in lengths[:-1] else None
            if next_n is not None and time.time() < deadline:
                a_up = _upsample_sequence(best_seq, next_n)
                # quick local refinement on the upsampled vector
                sub_dead = min(deadline, time.time() + 20.0)
                a_up_ref = _adam_opt(a_up, sub_dead)
                a_up_ref = _biased_simulated_annealing(a_up_ref, deadline)
                sc_up = _score(a_up_ref)
                if sc_up < best_score:
                    best_score = sc_up
                    best_seq = a_up_ref.copy()
                    print(f"[{time.time():.1f}] Upsampled best to n={next_n} score={sc_up:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_score = _score(best_seq)

    # Trim trailing tiny entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} length = {best_seq.size}")
    return best_seq.tolist()
