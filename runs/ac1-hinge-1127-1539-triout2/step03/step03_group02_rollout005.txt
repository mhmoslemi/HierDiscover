"""
Propose a step‑function (non‑negative coefficient vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Parametric sweep** – generate a collection of symmetric “U‑shapes’’
   (linear, quadratic, exponential, power, inverse‑power, cosine) and
   keep the best candidate after projecting onto the ℓ₁‑simplex
   (∑a = √(2 n)).

2. **Adam‑based smoothed gradient descent** – minimise a smooth
   approximation of the max‑convolution using the log‑sum‑exp
   (temperature τ is gradually cooled).  After each step the vector is
   projected back onto the simplex and symmetrised, which respects the
   natural symmetry of the problem.

   Several restarts, jittering and occasional random pair‑transfers are
   employed to escape plateaus.

3. **Polishing** – a short burst of tiny random mass transfers keeps
   the best sequence seen before the 1000 s budget expires.

The code respects the time budget, prints progress and finally returns the
best sequence as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())       # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – fallback to uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny fraction of mass from one random index to another,
    then re‑project onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse, cosine}.
    param controls the shape (unused for cosine).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid               # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "cosine":
        raw = 0.5 * (1.0 + np.cos(np.pi * d))
    else:
        raise ValueError(f"unknown shape kind: {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution of a."""
    return float(conv_self_fft(a).max())


def smooth_max_and_gradient(a: np.ndarray, tau_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(conv(a, a)) and its gradient.
    The temperature τ = tau_factor * max_conv (lower bounded by 1e-12).
    Returns (max_conv, gradient) where gradient has shape a.shape.
    """
    n = a.shape[0]

    # full convolution a * a
    conv = conv_self_fft(a)
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)

    # log‑sum‑exp weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w  *  a_rev)  (central n entries)
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return max_conv, grad

        "linear":       np.linspace(0.0, 8.0, 41),   # step 0.2
        "quadratic":    np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power":        np.linspace(0.5, 5.0, 11),
        "inverse":      np.linspace(0.5, 5.0, 11),
        "cosine":       np.array([0.0]),            # no extra parameter
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, float(p))
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # Consider the supplied baseline if it exists
    if init_seq is not None:
        init_seq = project_onto_simplex(init_seq, target_sum)
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves initial score → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam‑based smoothed gradient descent with restarts
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step_counter = 0                       # global iteration counter
    t = 0                                   # Adam time‑step
    # Hyper‑parameters
    lr = 0.5 * target_sum / n                # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    tau_factor = 0.2                         # initial temperature factor
    min_tau_factor = 0.001

    # bookkeeping
    best_global_seq = best_seq.copy()
    best_global_score = best_score
    stagnation = 0
    STAG_LIMIT = 8000                         # when to jitter / restart

    print(f"[{time.time():.2f}] Entering Adam‑descent phase")
    while time.time() < deadline:
        # --- smooth max & gradient ---
        max_conv, grad = smooth_max_and_gradient(a, tau_factor)

        # --- Adam update ---
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # projection onto the simplex + enforce symmetry
        a_new = project_onto_simplex(a_new, target_sum)
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = project_onto_simplex(a_new, target_sum)

        a = a_new

        # evaluate the new max‑convolution
        cur_max = compute_max_conv(a)

        if cur_max + 1e-12 < best_global_score:
            best_global_score = cur_max
            best_global_seq = a.copy()
            stagnation = 0
            print(f"Iter {step_counter:07d} – new best {best_global_score:.6f}")
        else:
            stagnation += 1

        # cooling schedule for τ
        tau_factor = max(tau_factor * 0.993, min_tau_factor)

        # occasional stochastic mass transfer
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.006)

        # jitter if we have been stuck for a while
        if stagnation >= 3000:
            jitter = np.random.randn(n) * 0.0004
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            a = (a + a[::-1]) * 0.5
            a = project_onto_simplex(a, target_sum)
            stagnation = 0
            print(f"Iter {step_counter:07d} – jitter applied")

        # full restart if stagnation gets too large
        if stagnation >= STAG_LIMIT:
            kind = np.random.choice(shape_kinds)
            if kind in ("power", "inverse"):
                param = np.random.uniform(0.5, 5.0)
            else:
                param = np.random.uniform(0.0, 6.0)
            a = generate_shape(n, target_sum, kind, param)
            a = (a + a[::-1]) * 0.5
            a = project_onto_simplex(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            tau_factor = 0.2
            stagnation = 0
            print(f"Iter {step_counter:07d} – restart with {kind} shape")

        step_counter += 1
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – final polishing with tiny random transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Adam phase finished; best score {best_global_score:.6f}")
    for _ in range(120):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_global_seq.copy(), target_sum, scale=0.0025)
        cand_score = compute_max_conv(cand)
        if cand_score < best_global_score:
            best_global_score = cand_score
            best_global_seq = cand.copy()
            print(f"Polish improved to {best_global_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence with score {best_global_score:.6f}")
    return best_global_seq.tolist()
