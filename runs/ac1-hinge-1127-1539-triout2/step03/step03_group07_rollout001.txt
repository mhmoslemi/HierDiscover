def lp_step(a, M):
    n = a.shape[0]
    L = 2*n - 1
    # Build lists for COO: rows (0..L-1), cols (0..n-1), data
    rows = []
    cols = []
    data = []
    for k in range(L):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for i in [i_min, i_max], coefficient of b_i is a_{k-i}
        # that is a[k - i]
        for i in range(i_min, i_max+1):
            rows.append(k)
            cols.append(i)
            data.append(a[k - i])
    # Build csr_matrix
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(L, n))
    b_ub = np.full(L, M)

    # Solve LP: maximize sum(b) => minimize -sum(b)
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return original a
        return a
    b = res.x
    # compute sum(b)
    sum_b = b.sum()
    if sum_b <= 0:
        return a
    scale = np.sqrt(2.0 * n) / sum_b
    b_scaled = b * scale
    return b_scaled

"""
Candidate generator for the step‑function minimisation task.

Algorithm Overview
------------------
1. **Random Dirichlet starts** – generate several non‑negative vectors
   with L1‑norm √(2·n) for a chosen length `n`.

2. **Linear‑programming refinement** – for a current vector `a`
   compute `M = max(a*a)`.  Keeping `M` fixed, solve the LP  

        maximise   Σ b_i
        subject to  (a * b)[k] ≤ M   for k = 0,…,2n‑2
                    b_i ≥ 0

   The convolution `(a * b)` is linear in `b`; the constraints are therefore
   linear and can be handled by `scipy.optimize.linprog`.  The optimal
   `b` is scaled to have the required L1‑norm and a convex combination
   `(1‑t)·a + t·b` is taken (with a slowly decreasing step size `t`).
   This is exactly the iteration described in the problem statement and
   drives the maximum self‑convolution down.

3. **Local stochastic hill‑climbing** – after the LP loop a short
   simulated‑annealing style hill‑climber (the same routine that was
   used previously) explores the neighbourhood of the best vector
   found, helping to escape possible stagnation.

4. **Multiple lengths & restarts** – the whole procedure is repeated
   for several lengths (500, 1000, 2000, 4000) and for a small number of
   random restarts.  The best sequence over all runs is returned.

All operations respect the non‑negativity and L1‑normalisation that
`evaluate_sequence` implicitly assumes (the external score equals the
maximum entry of the self‑convolution when Σa = √(2·n)).  The code
maintains a hard deadline (default 1000 s) and logs progress with
`print`.

"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return float(np.max(conv))


# ----------------------------------------------------------------------
# Linear‑programming refinement (step 2 of the algorithm)
# ----------------------------------------------------------------------
def lp_step_fixed_M(a: np.ndarray, M: float) -> np.ndarray:
    """
    Given a vector a (length n) and a bound M,
    solve the LP that maximises Σ b_i subject to a * b ≤ M component‑wise,
    with b ≥ 0.  The returned vector is the optimal b, *scaled* to have
    L1‑norm √(2·n).  If the LP fails, the original a is returned unchanged.
    """
    n = a.shape[0]
    L = 2 * n - 1

    # Build sparse matrix A_ub where (A_ub @ b)[k] = (a * b)[k].
    rows = []
    cols = []
    data = []

    # For each output index k, the contributing b_i have i∈[max(0,k-n+1), min(n-1,k)].
    for k in range(L):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # coefficient of b_i is a_{k-i}
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            data.append(a[k - i])

    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(L, n))
    b_ub = np.full(L, M, dtype=np.float64)

    # Objective: maximise Σ b_i  → minimise -Σ b_i
    c = -np.ones(n, dtype=np.float64)

    # Solve with HiGHS, which handles large sparse LPs efficiently.
    res = scipy.optimize.linprog(
        c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method="highs", options={"presolve": True}
    )
    if not res.success:
        # LP could not improve – fall back to the unchanged vector.
        return a.copy()

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a.copy()

    scale = np.sqrt(2.0 * n) / sum_b
    b_scaled = b * scale
    return b_scaled


def lp_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a sequence of LP‑based refinements starting from a0.
    Returns the best vector seen and its maximal self‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = conv_max(a)

    # Convex‑combination step size t – start moderately large, then decay.
    t = 0.5
    it = 0
    while time.time() < deadline and it < 200:
        it += 1
        cur_M = conv_max(a)

        # LP step with the current bound.
        b = lp_step_fixed_M(a, cur_M)

        # Convex update.
        a = (1.0 - t) * a + t * b

        # Renormalise (numerical drift may occur).
        total = a.sum()
        if total > 0:
            a *= (np.sqrt(2.0 * n) / total)
        else:
            a.fill(np.sqrt(2.0 * n) / n)

        # Record improvement.
        cur_val = conv_max(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(
                f"[LP improve] time={time.time():.1f}s n={n} iter={it} max={cur_val:.6f}"
            )

        # Decay the mixing factor.
        t *= 0.995
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (step 3 of the algorithm)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climber that moves a random fraction of mass
    between two positions.  Returns the best vector and its maximal convolution.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_max = conv_max(a)
    best_a = a.copy()
    best_max = cur_max

    temperature = max(0.05 * cur_max, 1e-4)
    it = 0

    while time.time() < deadline and it < 25000:
        it += 1

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = conv_max(a_new)

        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(
                    f"[hill climb improve] time={time.time():.1f}s n={n} iter={it} max={new_max:.6f}"
                )
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        temperature *= 0.9995

        # occasional jitter to diversify
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()
            cur_max = conv_max(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(
                    f"[jitter] time={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}"
                )

    return best_a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low evaluation score.
    Returns a Python list of heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    best_seq = None
    best_score = float("inf")

    # If a previously discovered construction is present, record it.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(
                    f"[baseline] existing height_sequence_1 score = {sc:.6f} length = {cand.size}"
                )
        except Exception:
            pass

    # Lengths to explore – larger lengths give more flexibility but cost more.
    lengths = [500, 1000, 2000, 4000]  # 8000 would be too heavy for LP.
    run_id = 0

    while time.time() < deadline:
        # Choose a length (bias towards larger sizes for richer search space)
        n = int(rng.choice(lengths, p=np.array([0.15, 0.25, 0.35, 0.25])))

        # Random Dirichlet start – concentration controls sparsity.
        conc = rng.uniform(0.05, 1.0)
        a0 = rng.dirichlet(np.full(n, conc))
        a0 = a0 * np.sqrt(2.0 * n)  # enforce Σa = √(2·n)

        # Phase A – LP refinement.
        sub_deadline = min(deadline, time.time() + 0.8 * (deadline - time.time()))
        a_lp, max_conv_lp = lp_refine(a0, sub_deadline)

        # Phase B – optional local hill‑climbing (short, cheap).
        a_hc, max_conv_hc = hill_climb(a_lp, deadline, rng)

        # The evaluation function returns 2·n·max/(sum²); for our normalisation
        # the external score equals the maximal self‑convolution.
        score = max_conv_hc

        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_hc.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

        # Occasionally do a cheap exponential‑descent restart – it can
        # discover improvements in regions where the LP gets stuck.
        if rng.random() < 0.12 and time.time() < deadline:
            # A very short exponential‑descent episode (max 3000 iters)
            a_exp, max_conv_exp = _run_exponential_descent(
                a0,
                n,
                deadline,
                top_k=rng.integers(8, 13),
                eta0=rng.uniform(0.4, 0.7),
                beta=0.9,
                decay=0.9999,
                jitter_prob=0.02,
                max_iters=3000,
            )
            if max_conv_exp < best_score - 1e-12:
                best_score = max_conv_exp
                best_seq = a_exp.copy()
                print(f"[exp restart improve] run={run_id} n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing was found (highly unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # Final verification – the platform will re‑evaluate as well.
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")

    return best_seq.tolist()


# ----------------------------------------------------------------------
# Exponentiated gradient descent (used as a cheap secondary tactic)
# ----------------------------------------------------------------------
def _run_exponential_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    top_k: int = 10,
    eta0: float = 0.6,
    beta: float = 0.9,
    decay: float = 0.99993,
    jitter_prob: float = 0.015,
    max_iters: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent with momentum.
    Returns the best vector seen and its maximal self‑convolution.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)

    cur_best = conv_max(a)
    best_a = a.copy()

    v = np.zeros_like(a)  # momentum buffer
    eta = eta0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # Full self‑convolution via FFT
        conv_len = 2 * n - 1
        fft_len = 1 << (conv_len - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
        cur_max = float(np.max(conv_full))

        # Indices of the largest K entries (sub‑gradient approximation)
        if conv_full.size <= top_k:
            shifts = np.arange(conv_full.size)
        else:
            shifts = np.argpartition(conv_full, -top_k)[-top_k:]

        # Sub‑gradient from the selected shifts
        g = _gradient_from_shifts(a, shifts)

        # Momentum update
        v = beta * v + (1.0 - beta) * g

        # Exponential multiplicative update (preserves non‑negativity)
        a = a * np.exp(-eta * v)

        # Renormalise to maintain Σa = √(2·n)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # Occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # Learning‑rate decay
        eta *= decay

        if cur_max < cur_best - 1e-12:
            cur_best = cur_max
            best_a = a.copy()
            print(
                f"[exp descent improve] time={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}"
            )

    return best_a, cur_best


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (hard) max‑convolution w.r.t. a,
    summed over the supplied shift indices.
    For a shift k the contribution to g_j is 2·a_{k−j} whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g
