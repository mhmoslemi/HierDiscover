   (temperature α).  The gradient of the softmax‐objective can be
   written as a convolution, which we evaluate with FFTs.  Adam
   optimisation together with a slowly increasing α yields fast
   descent.

2. **Targeted mass‑swap moves.**  
   After each gradient step we locate the index `k*` where the
   convolution attains its maximum.  The contributing indices
   `i` satisfy `0 ≤ i < n` and `0 ≤ k*‑i < n`.  We move a small
   fraction of mass from the largest coefficient among the
   contributors to the smallest coefficient outside that set.
   This directly reduces the dominant convolution entry while keeping
   the L¹‑norm fixed.

3. **Multi‑restart random search.**  
   Many independent local optimisations are launched with random
   lengths `n ∈ [800, 5000]` and random seeds.  The best sequence found
   over the whole budget (≤ 1000 s) is returned.  If a previously
   discovered sequence `height_sequence_1` exists it is used as an
   additional seed.

All helper functions are top‑level (no closures) and only use NumPy.
The code respects the 1000 s time limit and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max objective:
        g_j = 2 * Σ_i w_{i+j} * a_i .
    Both a (length n) and w (length 2n‑1) are real.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max on the convolution
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)."""
    scaled = alpha * conv
    # numerical stability
    m = np.max(scaled)
    exp_shifted = np.exp(scaled - m)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser for the smooth surrogate
# ----------------------------------------------------------------------


def _adam_step(a: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float,
               beta1: float,
               beta2: float,
               eps: float,
               alpha: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam update on the smoothed maximum.
    Returns (a_new, m_new, v_new).
    """
    # 1. compute smooth objective gradient
    conv = _full_convolution(a)
    w = _softmax_weights(conv, alpha)
    grad = _conv_with_weights(a, w)

    # 2. Adam moments
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step

    return a_new, m, v


# ----------------------------------------------------------------------
# Targeted mass‑swap (helps to beat plateaus)
# ----------------------------------------------------------------------


def _targeted_swap(a: np.ndarray,
                   conv: np.ndarray,
                   swap_frac: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the current maximal convolution entry by moving a small
    fraction of mass from a heavy contributor inside the active window
    to a light contributor outside it.
    """
    n = a.shape[0]
    k_star = int(np.argmax(conv))

    # indices that actually participate in conv[k_star]
    i_min = max(0, k_star - (n - 1))
    i_max = min(n - 1, k_star)
    inside = np.arange(i_min, i_max + 1)

    # pick source: largest a_i inside the window
    src = inside[np.argmax(a[inside])]

    # pick destination: smallest a_j outside the window
    mask_out = np.ones(n, dtype=bool)
    mask_out[inside] = False
    if not np.any(mask_out):
        # window covers whole array – fall back to random swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        src, dst = i, j
    else:
        dst = np.argmin(a[mask_out])
        dst = np.arange(n)[mask_out][dst]

    if a[src] <= 0:
        return a  # nothing to move

    delta = swap_frac * a[src] * rng.random()
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    return a_new


# ----------------------------------------------------------------------
# Local optimiser – Adam + targeted swaps
# ----------------------------------------------------------------------


def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Optimise a single candidate using Adam on the smooth surrogate and
    occasional targeted swaps.  Returns the best vector found and its
    (true) maximal self‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.35
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for the soft‑max temperature
    alpha = 0.5
    alpha_mul = 1.0015

    # swap schedule
    swap_prob = 0.09
    swap_frac = 0.025

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    iter_cnt = 0
    last_improve = 0

    while time.time() < deadline:
        iter_cnt += 1

        # ----- Adam step -----
        a, m, v = _adam_step(a, m, v, iter_cnt, lr, beta1, beta2, eps, alpha)

        # ----- projection onto the simplex (keeps sum = S) -----
        a = _project_simplex(a, S)

        # ----- evaluate true max (no smoothing) -----
        conv = _full_convolution(a)
        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = iter_cnt

        # ----- occasional targeted mass swap -----
        if rng.random() < swap_prob:
            a_swapped = _targeted_swap(a, conv, swap_frac, rng)
            a_swapped = _project_simplex(a_swapped, S)
            conv_swapped = _full_convolution(a_swapped)
            max_swapped = np.max(conv_swapped)
            # accept if it improves or with a tiny metropolis probability
            if max_swapped < cur_max or rng.random() < np.exp(-(max_swapped - cur_max) / (0.02 * cur_max + 1e-12)):
                a = a_swapped
                cur_max = max_swapped
                if cur_max < best_score:
                    best_score = cur_max
                    best_a = a.copy()
                    last_improve = iter_cnt

        # ----- update schedules -----
        alpha *= alpha_mul

        # early stop if no progress for a while
        if iter_cnt - last_improve > 3000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with L¹‑norm √(2·n)."""
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # tiny jitter to escape exact replicas
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        # Dirichlet(1) = uniform over the simplex
        a = rng.random(n)

    # normalise to desired sum
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Run a budget‑aware multi‑restart optimiser and return the best step‑function.
    The returned list will be fed to `evaluate_sequence` by the harness.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # Try to re‑use a previously discovered construction if available
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < deadline:
        attempt += 1

        # Random problem size (the longer the sequence, the richer the search space)
        n = rng.integers(800, 5000)

        a0 = _initial_sequence(base_seq, n, rng)

        # Allocate a time slice (≈ 12 s or remaining time) for this local run
        slice_deadline = min(time.time() + 12.0, deadline)

        a_opt, score = _optimize_local(a0, slice_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} "
                  f"new best max‑conv={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – run the black‑box evaluator once
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a simple uniform sequence (never happens in practice)
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n)
        return a.tolist()

    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal max‑conv = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
