"""
Local hybrid optimizer using momentum, exponentiated gradient,
and simulated‑annealing mass swaps.  It explores a range of
sequence lengths and runs many restarts within the time budget,
keeping the best sequence (lowest max self‑convolution entry,
hence lowest `evaluate_sequence` score).
"""
import time
import numpy as np

def _project_to_simplex(v, S):
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all zero case – return uniform distribution
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a):
    """Full self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _hardmax_gradient(a, k):
    """Sub‑gradient of max entry (hard max) for index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)
    g[valid] = 2.0 * a[partner[valid]]
    # diagonal contribution appears only once
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]
    return g


def _gradient_and_max(a, tau):
    """Return gradient of the τ‑soft‑max approximation and current max."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau <= 1e-12:
        k = np.argmax(b)
        g = _hardmax_gradient(a, k)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
        g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


def _eg_update(a, g, eta, S):
    """Exponentiated‑gradient update on the simplex."""
    # avoid overflow/underflow by clipping the exponent
    exp_factor = np.exp(-eta * g)
    a_new = a * exp_factor
    # renormalise to the required ℓ₁‑norm S
    total = a_new.sum()
    if total == 0:
        # fallback to uniform distribution
        return np.full_like(a, S / a.shape[0])
    a_new = a_new * (S / total)
    return a_new


def _additive_update(a, g, eta, S):
    """Additive projected gradient step."""
    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)
    return a_proj


def _initial_sequence(base_seq, n, rng):
    """Create a feasible non‑negative vector of length n with ℓ₁‑norm = √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=max(a.mean(), 1e-9) * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)  # uniform Dirichlet(1)
    a = _project_to_simplex(a, S)
    return a

    best_max = cur_max

    eta = 0.78                # additive step size
    eta_decay = 0.9996
    eta_eg = 0.55             # EG step size
    eta_eg_decay = 0.9996
    mu = 0.92                 # momentum coefficient
    mu_decay = 0.9999

    epsilon_move = 0.05
    temp = cur_max * 0.15
    temp_decay = 0.9997
    jitter_prob = 0.012
    jitter_scale = 0.018
    tau = cur_max             # initial temperature for the soft‑max smooth

        # --------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    b_candidate = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_candidate)
                    accept = False

                    else:

                        if rng.random() < prob:

                    if accept:
                        a = a_candidate
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                    temp *= temp_decay
            continue

        # --------------------------------------------------------------
        # 2) Gradient based update (soft‑max for smoothing)
        # --------------------------------------------------------------
        g, max_b = _gradient_and_max(a, max(tau, 1e-12))
        cur_max = max_b

        # momentum update
        momentum = mu * momentum - eta * g

        # choose additive or EG update
        if rng.random() < 0.5:
            a_new = _additive_update(a + momentum, g=None, eta=0.0, S=S)  # projection only
            # Actually we already applied momentum, so just project:
            a_new = _project_to_simplex(a + momentum, S)
        else:

        # evaluate new candidate
        b_new = _full_convolution_fft(a_new)
        new_max = np.max(b_new)

        # accept if improvement, otherwise accept with SA probability
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
            a = a_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        # else keep old a (reject)

        # --------------------------------------------------------------
        # 3) Occasional jitter
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=max(a.mean(), 1e-9) * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _full_convolution_fft(a)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # decay schedules
        eta *= eta_decay
        eta_eg *= eta_eg_decay
        mu *= mu_decay
        tau = max(tau * tau_decay, 1e-12)
        # No strict need to decay temp here because we reuse temp from swaps

    return best_a, best_max

"""
Hybrid local optimiser for the max‑self‑convolution problem.

The optimiser mixes:
  • momentum‑based projected gradient descent on a τ‑soft‑max
    approximation of the ∞‑norm,
  • exponentiated‑gradient (multiplicative) updates on the simplex,
  • occasional simulated‑annealing mass swaps,
  • rare Gaussian jitter.

Many independent restarts with different lengths are performed within the
time budget.  The best sequence (lowest maximal convolution entry, i.e.
lowest `evaluate_sequence` score) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Projection onto the ℓ₁‑simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries would become zero – fall back to uniform distribution
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Full self‑convolution via a real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (full convolution) using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
#  Sub‑gradient of the hard‑max (τ → 0)
# ----------------------------------------------------------------------
def _hardmax_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the entry b_k = (a*a)[k] (hard max case)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)
    g[valid] = 2.0 * a[partner[valid]]
    # diagonal contribution appears only once
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]
    return g


# ----------------------------------------------------------------------
#  Gradient of the τ‑soft‑max approximation and current maximal entry
# ----------------------------------------------------------------------
def _gradient_and_max(a: np.ndarray, tau: float):
    """Return (gradient, max entry) for the τ‑soft‑max of a*a."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau <= 1e-12:                     # hard max – sub‑gradient
        k = np.argmax(b)
        g = _hardmax_gradient(a, k)
    else:                                 # smooth soft‑max
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # soft‑max weights
        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
        g = 2.0 * conv[n - 1:n - 1 + n]   # slice gives length n
    return g, max_b


# ----------------------------------------------------------------------
#  Exponentiated‑gradient update on the simplex
# ----------------------------------------------------------------------
def _eg_update(a: np.ndarray, g: np.ndarray, eta: float, S: float) -> np.ndarray:
    """Multiplicative EG step followed by ℓ₁‑normalisation."""
    # avoid overflow/underflow by clipping the exponent
    factor = np.exp(-eta * g)
    a_new = a * factor
    total = a_new.sum()
    if total == 0.0:
        # fall back to uniform distribution on the simplex
        return np.full_like(a, S / a.shape[0])
    a_new = a_new * (S / total)
    return a_new


# ----------------------------------------------------------------------
#  Initialise a feasible vector of length n (ℓ₁‑norm = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with sum √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=max(a.mean(), 1e-9) * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                     # Dirichlet(1) = uniform on simplex
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
#  Local optimiser (one restart)
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hybrid optimiser:
      – momentum‑based projected gradient on a soft‑max surrogate,
      – exponentiated‑gradient steps,
      – simulated‑annealing mass swaps,
      – occasional jitter.
    Returns the best vector seen and its maximal self‑convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best record
    cur_max = _full_convolution_fft(a).max()
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters (slow decay → stable convergence)
    eta = 0.78
    eta_decay = 0.9996
    eta_eg = 0.55
    eta_eg_decay = 0.9996
    mu = 0.92
    mu_decay = 0.9999
    move_prob = 0.13
    epsilon_move = 0.05
    temp = cur_max * 0.15
    temp_decay = 0.9997
    jitter_prob = 0.012
    jitter_scale = 0.018
    tau = cur_max                     # start temperature for soft‑max
    tau_decay = 09995

    momentum = np.zeros_like(a)

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) random mass swap (simulated annealing)
        # ------------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    cand_max = _full_convolution_fft(a_candidate).max()
                    accept = False
                    if cand_max <= cur_max:
                        accept = True
                    else:
                        prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                        if rng.random() < prob:
                            accept = True
                    if accept:
                        a = a_candidate
                        cur_max = cand_max
                        momentum[:] = 0.0            # reset stale momentum
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                    temp *= temp_decay
            continue

        # ------------------------------------------------------------------
        # 2) gradient‑based update (soft‑max surrogate)
        # ------------------------------------------------------------------
        g, max_b = _gradient_and_max(a, max(tau, 1e-12))
        cur_max = max_b

        # momentum update (classical heavy‑ball)
        momentum = mu * momentum - eta * g

        # choose additive (projected) or multiplicative (EG) step
        if rng.random() < 0.5:
            a_new = _project_to_simplex(a + momentum, S)
        else:
            a_new = _eg_update(a, g, eta_eg, S)

        new_max = _full_convolution_fft(a_new).max()

        # accept if improvement, otherwise with SA probability
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
            a = a_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        # else keep the old a (reject)

        # ------------------------------------------------------------------
        # 3) occasional jitter
        # ------------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=max(a.mean(), 1e-9) * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            cur_max = _full_convolution_fft(a).max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
            momentum[:] = 0.0

        # decay schedules
        eta *= eta_decay
        eta_eg *= eta_eg_decay
        mu *= mu_decay
        tau = max(tau * tau_decay, 1e-12)
        temp *= temp_decay

    return best_a, best_max
