The evaluation function  

\[
E(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2}
\]

is *scale‑invariant*: multiplying the whole vector by a constant does not change the score.  
Therefore we can fix the ℓ₁‑norm to  

\[
\sum_i a_i = \sqrt{2n},
\]

so that the score equals the maximal entry of the self‑convolution.  
The task becomes:

*find a non‑negative vector on the ℓ₁‑simplex that minimises the maximum of its convolution with itself.*

The code below uses a **soft‑max (exponential) re‑parameterisation** of the simplex, an **Adam optimiser** on a **log‑sum‑exp smoothed** version of the objective, and a small amount of **random jitter / pair‑transfer** to escape plateaus.  
After the smooth phase a few sub‑gradient steps on the exact max‑convolution are performed, followed by a short polishing pass.

The whole search respects the 1000 s budget and always returns the best vector found.

```python
"""
Propose a non‑negative step‑function (vector of heights) that minimises  

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1. *Initialisation* – generate a variety of simple shapes (uniform,
   random Dirichlet, linear / quadratic / exponential “U‑shapes”, β‑shapes)
   and keep the best one.
2. *Smoothed Adam optimisation* – minimise the log‑sum‑exp smoothed
   max‑convolution while staying on the ℓ₁‑simplex using an exponential
   (soft‑max) parametrisation and the Adam optimiser.  The smoothing
   temperature τ is decreased slowly; occasional random jitter breaks
   stagnation.
3. *Sub‑gradient refinement & polishing* – switch to a pure
   sub‑gradient step on the exact maximum entry, then a short random
   pair‑transfer polishing phase.

All steps are time‑aware; the function aborts gracefully when the time
budget is exhausted and returns the best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index i to a random index j,
    then project back onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Symmetric “U‑shaped’’ vector of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥0 controls how strong the edges are.
    The result is projected onto the ℓ₁‑simplex (sum = target_sum).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def beta_shape(
    n: int,
    target_sum: float,
    beta: float = 2.0,
) -> np.ndarray:
    """
    Shape proportional to (i+1)^β (n-i)^β, symmetric and bell‑shaped.
    The vector is projected onto the ℓ₁‑simplex.
    """
    i = np.arange(n, dtype=np.float64)
    raw = ((i + 1) ** beta) * ((n - i) ** beta)
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (grad_a, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)

    # stable log‑sum‑exp weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad_a = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad_a, max_conv


def max_convolution_subgradient(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a, a)) (the maximal entry of the self‑convolution).
    Returns (grad_a, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm respects a ~budget_s time limit and returns the best
    vector encountered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        print(f"[{time.time():.2f}] Using length from existing construction: n={n}")
    else:
        # A slightly larger size gives a richer geometry while staying fast.
        n = 12000
        print(f"[{time.time():.2f}] No base sequence – using n={n}")

    target_sum = np.sqrt(2.0 * n)          # ℓ₁‑normalisation makes the score = max(conv)

    # ------------------------------------------------------------------
    # Phase 0 – generate a pool of initial vectors
    # ------------------------------------------------------------------
    init_vectors = []

    # uniform
    init_vectors.append(np.full(n, target_sum / n))

    # a few Dirichlet draws (different concentration parameters)
    for alpha in (0.2, 1.0, 5.0):
        vec = np.random.dirichlet(np.full(n, alpha)) * target_sum
        init_vectors.append(vec)

    # parametric U‑shapes (coarse sweep)
    for kind in ("linear", "quadratic", "exponential"):
        for p in np.linspace(0.0, 3.0, 7):
            init_vectors.append(generate_U_shape(n, target_sum, kind=kind, param=p))

    # β‑shapes (bell‑like)
    for beta in (0.5, 1.0, 2.0, 4.0):
        init_vectors.append(beta_shape(n, target_sum, beta=beta))

    # If a previous construction is present, include it (projected to the new n if needed)
    if base is not None:
        base_arr = np.array(base, dtype=np.float64)
        if len(base_arr) != n:
            # simple linear interpolation to the new length, then re‑normalise
            x_old = np.linspace(0, 1, len(base_arr))
            x_new = np.linspace(0, 1, n)
            from numpy import interp
            interp_vals = np.interp(x_new, x_old, base_arr)
            base_arr = project_onto_simplex(interp_vals, target_sum)
        else:
            base_arr = project_onto_simplex(base_arr, target_sum)
        init_vectors.append(base_arr)

    # ------------------------------------------------------------------
    # Evaluate the pool, keep the best as a starting point
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for vec in init_vectors:
        cur = conv_self_fft(vec).max()
        if cur < best_score:
            best_score = cur
            best_seq = vec.copy()
    print(f"[{time.time():.2f}] Initial best score after pool = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed Adam optimisation on the soft‑max parametrisation
    # ------------------------------------------------------------------
    eps = 1e-12
    # initialise logits so that softmax reproduces best_seq
    z = np.log(best_seq + eps)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    alpha = 0.6                     # base learning rate for logits
    tau_factor = 0.12               # smoothing temperature multiplier
    iter_cnt = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Starting Adam optimisation (n={n})")

    while time.time() < deadline:
        iter_cnt += 1

        # ----- produce a from current logits (soft‑max on the simplex) -----
        exp_z = np.exp(z - np.max(z))
        a = exp_z / exp_z.sum() * target_sum

        # ----- smoothed gradient w.r.t. a -----
        grad_a, cur_max = smoothed_gradient(a, tau_factor)

        # ----- transform gradient to logits space (soft‑max Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot / target_sum)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        # limit the step size to stay in a reasonable neighbourhood
        max_step = target_sum * 0.2
        step = np.clip(step, -max_step, max_step)

        z -= step

        # ----- keep the best vector seen so far -----
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.2f}] Iter {iter_cnt:06d} – new best {best_score:.6f}")
        else:
            no_improve += 1

        # ----- schedule adjustments -----
        if iter_cnt % 800 == 0:
            alpha *= 0.90                # slowly decay learning rate
            tau_factor = max(0.001, tau_factor * 0.93)   # cooler temperature
        # occasional jitter to break plateaus
        if no_improve >= 4000:
            jitter = np.random.randn(n) * target_sum * 5e-4
            a_jit = np.clip(a + jitter, 0.0, None)
            a_jit = project_onto_simplex(a_jit, target_sum)
            z = np.log(a_jit + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{time.time():.2f}]   jitter applied at iter {iter_cnt}")

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            print(f"[{time.time():.2f}] Target bound reached ({best_score:.6f}) – terminating early")
            break

    # ------------------------------------------------------------------
    # Phase 2 – pure sub‑gradient refinement on the exact max‑convolution
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Adam optimisation finished – best score {best_score:.6f}")

    eta_sub = target_sum / (n * 2.0)     # modest step size
    sub_iter = 0
    while time.time() < deadline and sub_iter < 2500:
        sub_iter += 1
        grad, cur_max = max_convolution_subgradient(best_seq)
        best_seq = best_seq - eta_sub * grad
        best_seq = project_onto_simplex(np.maximum(best_seq, 0.0), target_sum)

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            print(f"[{time.time():.2f}] Sub‑grad iter {sub_iter:04d} – improved to {best_score:.6f}")
            # slowly increase step size when we make progress
            eta_sub = min(eta_sub * 1.04, target_sum / (n * 0.5))
        else:
            if sub_iter % 500 == 0:
                eta_sub *= 0.85   # decay if stagnant

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final random polishing
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting final polishing")
    for _ in range(150):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.005)
        cand_score = conv_self_fft(cand).max()
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}]   polishing improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished – final best score {best_score:.6f}")
    return best_seq.tolist()
```
