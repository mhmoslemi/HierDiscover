def smooth_gradient_step_adam(a, target_sum, m, v, t, lr, tau_factor, beta1, beta2, eps):
    conv = conv_self_fft(a)
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)
    # compute weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    w /= w_sum
    # gradient = 2 * conv(w, a[::-1]) central n
    grad_full = conv_full_fft(w, a[::-1])
    grad = 2.0 * grad_full[n-1:n-1+len(a)]

    # Adam update
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad**2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, m, v, max_conv

"""
Propose a step‑function (non‑negative coefficient vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.

Algorithm
---------
* Choose a problem size n (use the length of the supplied baseline if any,
  otherwise 16384 – a power of two that is fast with FFTs).
* Initialise several diverse candidates (uniform, several deterministic
  “U‑shapes”, and a few random vectors).
* For every candidate run a short Adam optimisation on a smooth
  approximation of the max‑convolution:
      – compute the self‑convolution,
      – build the soft‑max weights w = exp((conv‑max)/τ) / Σ,
      – gradient of the smooth objective = 2·conv(w, rev(a))[central n].
      – Adam update → projection onto the ℓ₁‑simplex (sum = √(2 n)).
  τ (smoothing) and the learning rate are slowly decayed.
* After the Adam phase perform a few sub‑gradient steps on the true
  max‑convolution (average sub‑gradient over all maximal indices).
* Finish with a polishing stage that repeatedly transfers a tiny amount of
  mass between two random positions and keeps any improvement.
* The best vector seen before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using an FFT of power‑of‑two length."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a with itself."""
    return conv_full_fft(a, a)


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = target}."""
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Move a tiny fraction of the mass from one random index to another
    and project back onto the simplex."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ vector of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    param controls the shape.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Optimisation primitives (smooth Adam step and true sub‑gradient step)
# ----------------------------------------------------------------------


def smooth_gradient_step_adam(a: np.ndarray,
                              target_sum: float,
                              m: np.ndarray,
                              v: np.ndarray,
                              t_iter: int,
                              lr: float,
                              tau_factor: float,
                              beta1: float,
                              beta2: float,
                              eps: float):
    """
    One Adam update on the smooth (log‑sum‑exp) surrogate of max(conv).
    Returns (new_a, new_m, new_v, max_conv_before_update).
    """
    n = a.shape[0]

    # self‑convolution and maximal entry
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # smoothing weight (log‑sum‑exp)
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient of the smooth surrogate:
    # grad = 2 * conv(w, a_rev) restricted to the central n entries
    a_rev = a[::-1]
    grad_full = conv_full_fft(w, a_rev)
    grad = 2.0 * grad_full[n - 1:n - 1 + n]

    # Adam moments
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t_iter)
    v_hat = v / (1.0 - beta2 ** t_iter)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, m, v, max_conv


def subgradient_step(a: np.ndarray, target_sum: float, step: float):
    """
    Sub‑gradient descent on the true max‑convolution.
    Returns (new_a, max_conv_before_step).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # mask of (approximately) maximal entries
    eps = max(1e-12, 1e-9 * max_conv)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    # gradient = average sub‑gradient over the active set
    a_rev = a[::-1]
    grad_full = conv_full_fft(mask, a_rev)
    grad = (2.0 / cnt) * grad_full[n - 1:n - 1 + n]

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, max_conv

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises

    and returns the best sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (use previous construction if available)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        baseline = np.array(base, dtype=np.float64)
    else:
        n = 16384                 # default power‑of‑two size
        baseline = None

    target_sum = math.sqrt(2.0 * n)   # normalisation: Σ a_i = √(2 n)

    # ------------------------------------------------------------------
    # Prepare a pool of initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform start
    init_candidates.append(np.full(n, target_sum / n))

    # deterministic U‑shapes (few representative parameters)
    shape_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
    param_samples = {
        "linear": [0.0, 2.0, 5.0, 8.0],
        "quadratic": [0.0, 2.0, 4.0, 6.0],
        "exponential": [0.0, 2.0, 4.0, 6.0],
        "power": [0.5, 1.5, 3.0, 5.0],
        "inverse": [0.5, 1.5, 3.0, 5.0],

    for kind in shape_kinds:

    # a handful of random Dirichlet‑like vectors (concentrated and diffuse)
    for _ in range(5):
        rnd = np.random.rand(n)

    # baseline if we have one
    if baseline is not None:
        init_candidates.append(baseline.copy())

    # ------------------------------------------------------------------
    # Global trackers
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Iterate over all seeds (while time permits)
    # ------------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:

        a = init.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t_iter = 1

        # Adam hyper‑parameters (chosen once, then decayed)
        lr = 0.5 * (target_sum / n)           # ≈ 0.5·average entry
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        tau_factor = 0.08

        # Phase 1 – smooth Adam optimisation
        # ------------------------------------------------------------------
        while time.time() < deadline:

            t_iter += 1

            # keep the best vector seen so far
            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best (init {idx}) → {best_score:.6f}")

            # Schedules
            if t_iter % 5000 == 0:
                lr *= 0.94
                tau_factor = max(0.001, tau_factor * 0.94)

            # Occasionally perform a small random mass transfer to escape plateaus
            if np.random.rand() < 0.012:
                a = random_pair_transfer(a, target_sum, scale=0.008)

            # Stop the Adam phase after a moderate number of iterations
            if t_iter > 50000:
                break

        # ------------------------------------------------------------------
        # Phase 2 – sub‑gradient refinement on the true max‑convolution
        # ------------------------------------------------------------------
        step_sub = 0.02 * (target_sum / n)
        sub_iter = 0
        while time.time() < deadline and sub_iter < 8000:
            a, cur_max = subgradient_step(a, target_sum, step_sub)
            sub_iter += 1

            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_seq = a.copy()

    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence with score {best_score:.6f}")
    return a_best.tolist()
