def _subgradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2*n)
    # use decreasing step size schedule
    step = target_sum * 0.1
    while time.time() < deadline:
        # compute convolution
        size = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        max_b = b.max()
        mask = (b == max_b)
        # compute p as mask normalized
        weight = mask.astype(float)
        weight_sum = weight.sum()
        if weight_sum == 0:
            break
        p = weight / weight_sum
        # gradient = 2 * (a * p) convolution (via FFT)
        # compute convolution of a_rev with p
        a_rev = a[::-1]
        conv_len = n + (2*n - 1) - 1  # 3n-2
        size2 = 1 << ((conv_len).bit_length())
        Ar_fft = np.fft.rfft(a_rev, n=size2)
        Pr_fft = np.fft.rfft(p, n=size2)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n-1 : n-1 + n]
        # step size diminish
        norm = np.linalg.norm(g)
        if norm > 0:
            lr = step / (norm + 1e-12)
            a = a - lr * g
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        # optionally adapt step size: if improvement not seen, reduce step.
        # but we can just slowly decrease step
        step *= 0.9995
    return a

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The algorithm combines

* a systematic random‑parameter search over several analytic families
  (pure exponential, power‑law, mixed) on increasingly larger vector
  lengths,
* a fast projected‑gradient descent on a log‑sum‑exp smooth approximation
  of the maximum self‑convolution (annealed β, momentum),
* a simulated‑annealing refinement that moves mass between indices while
  accepting occasional uphill moves.

The search is time‑bounded (default 1000 s).  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the linear convolution a*a using an FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation:
        score = max(a*a)

    ``a`` is assumed to already satisfy Σ a_i = sqrt(2·n) and a_i ≥ 0.
    """
    return _max_convolution_fft(a)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    (Duchi et al. 2008)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where

        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)

    The gradient is computed using FFTs.
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))               # numerical stability
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp smooth
    approximation of the maximum convolution.  Momentum and an adaptive
    step size are used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    max_beta = 1024.0
    step_base = target_sum * 0.18          # base magnitude for step
    momentum = np.zeros_like(a)
    rho = 0.9                               # momentum decay

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection back onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that moves a random fraction of mass between two indices.
    Accepts uphill moves with probability exp(-Δ/T).  Temperature decays
    multiplicatively.  Returns the best sequence seen.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    # initialise temperature as a modest fraction of current score
    T = cur_score * 0.05 + 1e-6
    cooling = 0.9994

    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick distinct indices
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction of mass from i to j
        frac = rng.random() * 0.30          # up to 30 % of a[i]
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        # geometric cooling
        T *= cooling

        # occasional global jitter (to escape deep minima)
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random / analytic family initialisers (used in the main loop)
# ----------------------------------------------------------------------


def _make_candidate(n: int, rng) -> np.ndarray:
    """Create a random analytic candidate on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)

    choice = rng.integers(0, 4)

    if choice == 0:                     # pure power‑law
        p = rng.uniform(0.05, 2.0)
        a_raw = (idx + 1.0) ** (-p)

    elif choice == 1:                   # exponential decay
        lam = rng.uniform(0.0005, 0.02)
        a_raw = np.exp(-lam * idx)

    elif choice == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.05, 2.0)
        lam = rng.uniform(0.0005, 0.02)
        a_raw = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    else:                               # mass concentrated at the end (reverse exp)
        lam = rng.uniform(0.0005, 0.02)
        a_raw = np.exp(-lam * (n - 1 - idx))

    # small Gaussian jitter before projection
    a_raw += rng.normal(scale=target_sum * 0.001, size=n)
    a = _project_to_simplex(a_raw, target_sum)
    return a

        print(f"[{time.time():.1f}] Final SA polish score={best_score:.6f}")

    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_score = _score(best_seq)

    # Trim trailing near‑zero values for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] FINAL BEST score = {best_score:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
