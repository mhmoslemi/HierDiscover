"""
Search for a step‑function that minimises the external ``evaluate_sequence``
score.

The algorithm performs many short simulated‑annealing runs on vectors of
different lengths.  Each vector is kept ℓ₁‑normalised to
``sqrt(2·n)`` – with this normalisation the external score equals the
maximum entry of the self‑convolution, which can be computed efficiently by
FFT.  A move consists of transferring a random fraction of the mass from one
index to another; moves are accepted with a Metropolis rule so the search can
escape local minima.  Very occasional linear‑programming refinements are
applied when SciPy is available.

The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence found.
"""

import time
import math
import numpy as np

# optional LP refinement – used only if SciPy is present
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:   # pragma: no cover
    _SCIPY_AVAILABLE = False


def _conv_max_fft(a: np.ndarray) -> float:
    """
    Return max entry of the linear self‑convolution of ``a``.
    Uses FFT for O(n log n) cost.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(Fa * Fa, fft_len)[:conv_len]
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct dense matrix H with (H @ g)[k] = (f * g)[k] for the
    linear‑programming refinement.
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, mix: float = 0.2,
               steps: int = 3) -> np.ndarray:
    """
    Simple LP refinement (the same idea as in the reference code).
    Finds a vector ``g`` with larger ℓ₁‑mass while respecting the current
    convolution bound, then mixes it back into ``f``.
    """
    if not _SCIPY_AVAILABLE or f.size > 3000:
        return f

    n = f.size
    S = math.sqrt(2.0 * n)

    for _ in range(steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 6})
        except Exception:                  # pragma: no cover
            break

        if not res.success:
            break

        g = res.x
        if g.sum() == 0.0:
            g = np.full(n, S / n)
        else:
            g = g * (S / g.sum())

        # convex combination and re‑normalisation
        f = (1.0 - mix) * f + mix * g
        f = f / f.sum() * S

    return f


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Performs many short simulated‑annealing runs on
    vectors of different lengths, keeps the best vector seen (according to
    the max‑convolution surrogate) and finally returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    best_seq = None
    best_score = float('inf')

    # lengths to explore – a mix of short and relatively long vectors
    lengths = [1024, 2048, 4096, 8192, 16384]

    # ------------------------------------------------------------------
    # 1)  Repeated SA runs with random initialisation
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = int(rng.choice(lengths))
        S = math.sqrt(2.0 * n)               # target ℓ₁‑norm

        # start from a random Dirichlet‑like vector (all entries >0)
        a = rng.random(n)
        a = a / a.sum() * S

        cur_val = _conv_max_fft(a)           # surrogate = external score
        # simulated‑annealing schedule
        T0 = 0.25
        max_iter = 4000
        for it in range(1, max_iter + 1):
            if time.time() >= deadline - 0.2:
                break

            # ----- propose a mass‑transfer move -----
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # move a random fraction (up to 50 %) of the mass at i to j
            frac = rng.random() * 0.5
            delta = a[i] * frac
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_val = _conv_max_fft(a_new)

            # ----- Metropolis acceptance -----
            T = T0 * (1.0 - it / max_iter)
            if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / (T + 1e-12)):
                a = a_new
                cur_val = new_val
                if cur_val < best_score:
                    best_score = cur_val
                    best_seq = a.copy()
                    print(f"[improve] n={n} score={best_score:.6f}")

        # ----- occasional LP refinement (cheap, cheap) -----
        if best_seq is not None and best_seq.size <= 2500 and time.time() < deadline - 2.0:
            refined = _lp_refine(best_seq, deadline)
            refined_val = _conv_max_fft(refined)
            if refined_val < best_score:
                best_score = refined_val
                best_seq = refined.copy()
                print(f"[LP] n={n} refined_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Final polishing of the global best (if we have one)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # a few more SA steps on the current best
        n = best_seq.size
        S = math.sqrt(2.0 * n)
        a = best_seq.copy()
        cur_val = _conv_max_fft(a)
        T0 = 0.15
        max_iter = 8000
        for it in range(1, max_iter + 1):
            if time.time() >= deadline - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.random() * 0.3
            delta = a[i] * frac
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            new_val = _conv_max_fft(a_new)
            T = T0 * (1.0 - it / max_iter)
            if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / (T + 1e-12)):
                a = a_new
                cur_val = new_val
                if cur_val < best_score:
                    best_score = cur_val
                    best_seq = a.copy()
        # final LP pass (if possible)
        if _SCIPY_AVAILABLE and best_seq.size <= 2500:
            best_seq = _lp_refine(best_seq, deadline)

    # ------------------------------------------------------------------
    # 3)  Fallback – uniform vector (never fails)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)

    # evaluate with the official function for reporting
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:   # pragma: no cover
        final_score = float('inf')
    print(f"[final] evaluated score = {final_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
