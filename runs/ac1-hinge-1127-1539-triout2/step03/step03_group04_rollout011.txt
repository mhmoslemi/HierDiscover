def _anneal_optim(a_init, deadline):
    a = a_init.copy()
    n = a.shape[0]
    S = a.sum()
    best_a, best_val = a.copy(), _score(a)
    cur_val = best_val
    # temperature schedule
    T0 = best_val * 0.1
    T = T0
    decay = 0.9995
    rng = np.random.default_rng()
    while time.time() < deadline:
        # propose move
        donor = rng.integers(0, n)
        # ensure donor has mass
        if a[donor] < 1e-12: continue
        receiver = rng.integers(0, n)
        if receiver == donor: continue
        # choose fraction
        frac = rng.random() * 0.3   # up to 30% of donor
        delta = a[donor] * frac
        # apply move
        a[donor] -= delta
        a[receiver] += delta
        # compute new score
        new_val = _score(a)
        # acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] Anneal improved {best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta
        # annealing step
        T = max(T * decay, 1e-8)
    return best_a, best_val
```

"""
Simulated‑annealing optimizer for the max‑convolution / ℓ¹‑norm benchmark.

The algorithm works in three phases under the global time budget:

1. **Fast candidate generation** – a small set of diverse shapes
   (power‑law, power‑law‑with‑exponential decay, sinusoidal, semi‑circular
   and random Dirichlet draws) are created for a handful of lengths.
   Each vector is projected onto the ℓ¹‑simplex Σa = √(2·n) and scored
   with a fast FFT based convolution.  The best few candidates are kept.

2. **Annealing refinement** – starting from each shortlisted vector a
   the algorithm performs a projected simulated‑annealing walk.
   In each step a random amount of mass is transferred from a donor
   index to a receiver index while keeping Σa fixed and all entries ≥0.
   Moves that improve the objective are always accepted; uphill moves are
   accepted with probability exp(−Δ/ T) where the temperature T is
   exponentially cooled.  Periodic jitter and simplex projection keep
   the walk numerically stable.

3. **Result selection** – the best sequence found before the deadline is
   returned (trailing zeros are trimmed).

The implementation deliberately avoids heavy external dependencies and
uses only NumPy (FFT for convolution) and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Compute the maximum entry of the linear convolution a*a using FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Exact objective used in the competition (identical to the provided
    evaluate_sequence after the normalisation Σa = √(2·n)).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:   # should never happen for our normalised vectors
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used in the coarse search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (pure power law)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = _project_to_simplex(a, target_sum)
    return a


def _exp_power_law_sequence(n: int, p: float, lam: float,
                            target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) * exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-lam * i)
    a = _project_to_simplex(a, target_sum)
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i = sin(π·(i+1)/(n+1)).  Positive on (0,π)."""
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
    a = _project_to_simplex(a, target_sum)
    return a


def _semicircle_sequence(n: int, target_sum: float) -> np.ndarray:
    """
    a_i proportional to a discrete semicircle:
    sqrt(1 - ((2*i+1 - n)/n)^2) for i within the interval,
    zero outside (which never happens for the discrete set).
    """
    i = np.arange(n, dtype=np.float64)
    x = (2 * i + 1 - n) / n
    a = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    a = _project_to_simplex(a, target_sum)
    return a


def _random_dirichlet_sequence(n: int, target_sum: float,
                               rng: np.random.Generator) -> np.ndarray:
    """Draw a random point from the simplex using a Dirichlet(1,…,1)."""
    a = rng.dirichlet(np.ones(n, dtype=np.float64))
    a *= target_sum
    return a


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a_init: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected simulated annealing.
    Moves a random fraction of mass from a donor index to a receiver index.
    Accepts uphill moves with probability exp(-Δ/T).  The temperature T
    decays exponentially.
    """
    a = a_init.copy()
    n = a.shape[0]
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule
    T = max(cur_val * 0.02, 1e-4)          # start a small fraction of the current value
    decay = 0.9997                         # slow exponential cooling
    min_T = 1e-9
    max_frac = 0.25                        # maximum fraction of donor's mass to move
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        # pick donor with probability proportional to its mass (helps picking non‑zero)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] < 1e-12:
            continue
        # pick a different receiver uniformly
        receiver = rng.integers(0, n - 1)
        if receiver >= donor:
            receiver += 1

        # amount to move
        frac = rng.random() * max_frac
        delta = a[donor] * frac

        # perform the move
        a[donor] -= delta
        a[receiver] += delta

        # re‑project very rarely to combat numerical drift
        if (it & 0x3FF) == 0:    # every 1024 iterations
            a = _project_to_simplex(a, a.sum())

        new_val = _score(a)
        improve = new_val < cur_val
        if improve or (rng.random() < np.exp(-(new_val - cur_val) / max(T, min_T))):
            # accept
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] Anneal improved: {best_val:.6f}")
        else:
            # reject – revert
            a[receiver] -= delta
            a[donor] += delta

        # temperature decay
        T = max(T * decay, min_T)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the benchmark.
    A fast multi‑stage method:
        * generate a diverse set of low‑dimensional candidates;
        * keep the top few according to the exact score;
        * refine each with simulated annealing;
    The best sequence encountered before the time budget expires is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Stage 1 – coarse candidate generation
    # ------------------------------------------------------------------
    # We consider a modest collection of lengths; larger n does not
    # dramatically change the objective after normalisation, but a few
    # sizes give the optimiser freedom to find a better shape.
    lengths = [1500, 2500, 3500]
    topk_global = 5            # total number of candidates kept for refinement
    candidates = []            # list of (score, array)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----- power‑law sweep ------------------------------------------------
        for p in np.linspace(0.30, 0.80, 26):      # 0.020 steps
            a = _power_law_sequence(n, p, target_sum)
            val = _score(a)
            candidates.append((val, a))

        # ----- power‑law + exponential sweep ----------------------------------
        for p in np.linspace(0.30, 0.80, 13):
            for lam in np.linspace(0.0, 0.004, 9):
                a = _exp_power_law_sequence(n, p, lam, target_sum)
                val = _score(a)
                candidates.append((val, a))

        # ----- deterministic shapes -------------------------------------------
        a = _sin_sequence(n, target_sum)
        candidates.append((_score(a), a))

        a = _semicircle_sequence(n, target_sum)
        candidates.append((_score(a), a))

        # ----- random Dirichlet draws -----------------------------------------
        for _ in range(5):
            a = _random_dirichlet_sequence(n, target_sum, rng)
            candidates.append((_score(a), a))

    # keep the globally best few candidates
    candidates.sort(key=lambda x: x[0])
    candidates = candidates[:topk_global]

    # ------------------------------------------------------------------
    # Stage 2 – annealing refinement
    # ------------------------------------------------------------------
    for base_val, a0 in candidates:
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget (roughly equal per candidate)
        remaining = deadline - time.time()
        sub_budget = min(30.0, remaining / max(1, len(candidates)))
        sub_deadline = time.time() + sub_budget

        a_opt, val = _anneal_optimize(a0, sub_deadline)

        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"** GLOBAL BEST **  n={a_opt.size}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – constant sequence (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # Trim trailing zeros (just in case)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
