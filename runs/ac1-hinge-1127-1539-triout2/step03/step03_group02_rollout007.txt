"""
Optimization of a non‑negative step‑function (height sequence) that
minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for a fixed ℓ₁‑norm.  The algorithm works as follows:

1. Choose the problem size n (reuse a previously discovered sequence if
   available, otherwise a power of two ≈ 8192 for fast FFTs).

2. Generate a small pool of diverse initial candidates (uniform,
   several simple “U‑shapes” and a few random simplex vectors).

3. For each candidate run a constrained Adam optimiser on a
   smooth log‑sum‑exp approximation of the max‑convolution.
   * Gradient of the smoothed objective is obtained via a single FFT‑
     based convolution.
   * After each update the vector is projected onto the simplex
     {x≥0, Σx = √(2 n)} and symmetrised (a_i = a_{n‑1‑i}) to keep the
     search space small.

4. The learning‑rate α and the smoothing factor τ are slowly
   annealed; occasional random pair‑transfers and small jitter steps
   help escape shallow local minima.

5. The best sequence seen before the time budget expires is returned.

The whole routine respects the 1000 s budget and never uses more than
one CPU core.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2 · len(a)‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())            # next power‑of‑two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Simple projection onto the ℓ₁‑simplex Σx = target,  x ≥ 0
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = target} (O(n) algorithm)."""
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all coordinates will be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small stochastic perturbations (pair transfers)
# ----------------------------------------------------------------------


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a small amount of mass from one index to another and re‑project."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Simple parametric “U‑shapes” for initialisation
# ----------------------------------------------------------------------


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped” sequence.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


# ----------------------------------------------------------------------
# Adam optimisation step on the smoothed max‑convolution objective
# ----------------------------------------------------------------------


def adam_step(a: np.ndarray,
              m: np.ndarray,
              v: np.ndarray,
              t_iter: int,
              target_sum: float,
              alpha: float,
              beta1: float,
              beta2: float,
              tau_factor: float):
    """
    Perform one Adam update on the log‑sum‑exp smoothed approximation of
    max(conv(a, a)).  Returns the updated a, first and second moments,
    and the current (unsmoothed) max‑convolution value.
    """
    n = a.shape[0]

    # self‑convolution and its maximum
    conv = conv_self_fft(a)
    max_conv = float(conv.max())

    # smoothing (log‑sum‑exp)
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    # Adam moment updates
    t = t_iter + 1
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # parameter step and projection
    step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)

    # enforce symmetry (a_i = a_{n-1-i})
    a_new = (a_new + a_new[::-1]) / 2.0

    # project back onto the ℓ₁‑simplex
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, m, v, max_conv

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects a time budget (~1000 s)
    and returns the best sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size (reuse previous construction if available)
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.array(base, dtype=np.float64)
    else:
        # choose a power‑of‑two that gives a fast FFT
        n_candidates = [4096, 8192, 16384]
        n = n_candidates[np.random.choice(len(n_candidates))]
        init_seq = None

    target_sum = math.sqrt(2.0 * n)           # normalisation: Σa = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – generate a few diverse initial candidates
    # --------------------------------------------------------------
    candidates = []

    # uniform start
    candidates.append(np.full(n, target_sum / n))

    # a few parametric U‑shapes
    shape_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
    param_grids = {
        "linear":       np.linspace(0.0, 6.0, 13),   # 0,0.5,…,6
        "quadratic":   np.linspace(0.0, 4.0, 11),
        "exponential": np.linspace(0.0, 5.0, 11),
        "power":       np.linspace(0.5, 4.0, 8),
        "inverse":     np.linspace(0.5, 4.0, 8),
    }
    for kind in shape_kinds:
        for p in param_grids[kind]:
            candidates.append(generate_shape(n, target_sum, kind, float(p)))

    # a few random simplex points
    for _ in range(5):
        rand_vec = np.random.rand(n)
        candidates.append(project_onto_simplex(rand_vec, target_sum))

    # include previous construction if present
    if init_seq is not None:
        candidates.append(init_seq.copy())

    best_seq = None
    best_score = np.inf

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for cand in candidates:
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best init → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during initialisation.")
            return best_seq.tolist()

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on the smoothed objective
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters (will be annealed)
    alpha = 0.3 * (target_sum / n)            # initial step size
    beta1 = 0.9
    beta2 = 0.999
    tau_factor = 0.06
    iteration = 0
    stagnation = 0

    print(f"[{time.time():.2f}] Entering Adam optimisation")
    while time.time() < deadline:
        a, m, v, cur_max = adam_step(
            a, m, v, iteration, target_sum,
            alpha, beta1, beta2, tau_factor
        )
        iteration += 1

        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            stagnation = 0
            print(f"Iter {iteration:07d} – improved to {best_score:.6f}")
        else:
            stagnation += 1

        # simple annealing schedules
        if iteration % 1500 == 0:
            alpha *= 0.97
            tau_factor = max(tau_factor * 0.95, 0.001)

        # occasional random pair transfer to shake the system
        if np.random.rand() < 0.01:
            a = random_pair_transfer(a, target_sum, scale=0.007)

        # if we have been stuck for a while, add a tiny jitter
        if stagnation >= 4000:
            jitter = np.random.randn(n) * 0.0003
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            stagnation = 0
            print(f"Iter {iteration:07d} – jitter applied")

    # --------------------------------------------------------------
    # Phase 2 – short polishing with random pair transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Adam phase finished; best score {best_score:.6f}")

    for _ in range(60):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.004)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish transfer improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence with score {best_score:.6f}")
    return best_seq.tolist()
