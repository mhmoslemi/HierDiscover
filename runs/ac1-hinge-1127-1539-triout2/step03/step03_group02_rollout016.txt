"""
Propose a non‑negative step‑function that minimises the maximum of its
self‑convolution (equivalently the value returned by evaluate_sequence)
under the ℓ₁‑normalisation Σ a_i = √(2 n).

Algorithm
---------
1. Problem size – reuse the length of the supplied `height_sequence_1`
   (if present); otherwise pick a power‑of‑two ≈16384 for fast FFTs.

2. Initialise – sweep a variety of symmetric “U‑shapes’’ (linear,
   quadratic, exponential, power, inverse) plus a batch of random
   symmetric Dirichlet‑type seeds.  Keep the best few candidates.

3. Adam‑optimised smoothed descent – minimise a log‑sum‑exp
   approximation of max(conv(a,a)).  The gradient is obtained by one
   FFT‑based convolution.  We use the Adam optimiser (per‑coordinate
   adaptive steps) together with Euclidean projection onto the simplex.
   The smoothing temperature τ is multiplied by a factor <1 each

   tiny.

4. Symmetry – after every update we symmetrise `a` (a ← (a+a[::-1])/2);
   the optimum is known to be symmetric, this reduces the search space.

5. Polishing – a final burst of tiny random pair‑transfers (mass moves
   between two coordinates) with projection.  Any improvement is kept.

The whole procedure respects a ~1000 s budget and returns the best
sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, Σ x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(
    n: int,
    target_sum: float,
    kind: str,
    param: float,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    param ≥ 0 controls the shape.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution of a."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence (i.e. the maximum of its self‑convolution).
    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.array(base, dtype=np.float64)
    else:
        # pick a power‑of‑two large enough for good optics
        n_candidates = [8192, 16384, 32768]
        n = int(np.random.choice(n_candidates))
        init_seq = None

    target_sum = math.sqrt(2.0 * n)          # Σ a_i = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – systematic + random sweep of initial shapes
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)    # uniform start
    best_score = compute_max_conv(best_seq)

    print(f"[{time.time():.2f}] Starting initial sweep (n={n})")
    shape_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),
        "quadratic": np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 11),
        "inverse": np.linspace(0.5, 5.0, 11),
    }

    # deterministic U‑shapes
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during deterministic sweep.")
                return best_seq.tolist()

    # random symmetric seeds (Dirichlet‑like)
    rnd_trials = 30
    half_len = (n + 1) // 2
    for _ in range(rnd_trials):
        half = np.random.rand(half_len)
        # mirror to obtain full symmetric vector
        if n % 2 == 0:
            cand = np.concatenate([half, half[::-1]])
        else:
            cand = np.concatenate([half, half[-2::-1]])
        cand = project_onto_simplex(cand, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  Random seed improved → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during random seed sweep.")
            return best_seq.tolist()

    # consider supplied baseline (if any)
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves initial score → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimiser on a smoothed max‑convolution
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # hyper‑parameters
    base_lr = 0.5 * (target_sum / n)   # base learning‑rate
    eta = base_lr                     # will be scaled by τ later
    tau_factor = 0.08                 # initial τ = τ_factor * max_conv
    tau_min = 1e-9
    tau_decay = 0.9995                # per‑iteration decay
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering Adam‑optimised smoothed descent")
    while time.time() < deadline:
        iteration += 1

        # current convolution statistics
        conv = conv_self_fft(a)
        max_conv = conv.max()

        # track best
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration:07d} – improved max_conv = {best_score:.6f}")
        else:
            no_improve += 1

        # decide smoothing temperature
        tau = max(tau_factor * max_conv, tau_min)

        # compute gradient (smooth or exact based on τ)
        if tau > 1e-6:
            # log‑sum‑exp smooth gradient
            w = np.exp((conv - max_conv) / tau)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            a_rev = a[::-1]
            L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
            W = np.fft.rfft(w, n=L)
            A_rev = np.fft.rfft(a_rev, n=L)
            full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
            grad = (2.0 / tau) * full[n - 1 : n - 1 + n]
        else:
            # true sub‑gradient (averaged over near‑max indices)
            eps_mask = max(1e-12, 1e-9 * max_conv)
            mask = (conv >= max_conv - eps_mask).astype(np.float64)
            cnt = mask.sum()
            if cnt == 0:
                mask[conv.argmax()] = 1.0
                cnt = 1.0
            a_rev = a[::-1]
            L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
            M = np.fft.rfft(mask, n=L)
            A_rev = np.fft.rfft(a_rev, n=L)
            full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
            grad = (2.0 / cnt) * full[n - 1 : n - 1 + n]

        # Adam update (per‑coordinate adaptive step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # step size scales with τ (gradient magnitude ∝ 1/τ)
        step_size = eta * tau
        a_new = a - step_size * m_hat / (np.sqrt(v_hat) + eps)

        # enforce feasibility
        a_new = np.maximum(a_new, 0.0)
        # symmetrise (optimal solution is symmetric)
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = project_onto_simplex(a_new, target_sum)

        a = a_new

        # temperature schedule
        tau_factor = max(tau_min / max_conv, tau_factor * tau_decay)

        # occasional jitter to escape plateaus
        if no_improve >= 3000:
            jitter = np.random.randn(n) * 3e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:07d} – jitter applied")

        # tiny random transfers for extra exploration
        if np.random.rand() < 0.006:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # periodic logging
        if iteration % 5000 == 0:
            print(f"[{time.time():.2f}] Iter {iteration:07d} – cur max_conv = {max_conv:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – final polishing with random pair transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Adam phase finished; best max_conv {best_score:.6f}")

    for _ in range(80):
        if time.time() >= deadline:
            break
        candidate = random_pair_transfer(best_seq.copy(), target_sum, scale=0.003)
        cand_score = compute_max_conv(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            print(f"Polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence with max_conv {best_score:.6f}")
    return best_seq.tolist()
