def adam_step(a, target_sum, lr, tau_factor, m, v, t, beta1=0.9, beta2=0.999, eps=1e-8):

    conv = conv_self_fft(a)

    max_conv = conv.max()

    tau = max(tau_factor * max_conv, 1e-12)

    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum != 0:
        w = w / w_sum

    # compute gradient
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n-1 : n-1 + n]

    # Adam update
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, m, v, max_conv

"""
Propose a candidate step‑function by:

1.  Using a richer parametric sweep (U‑shapes, block‑edge, cosine) to get a
    good symmetric start.
2.  Refining the start with an Adam‑style smoothed gradient descent.
    The smooth approximation to the max‑convolution uses a log‑sum‑exp
    with a temperature τ that is exponentially decreased.
    After each update the vector is projected onto the ℓ₁‑simplex
    (fixed sum = √(2 n)) and forced to be symmetric.
3.  Finishing with a short sub‑gradient phase and a polishing stage
    consisting of random symmetric pair‑transfers.

All heavy work is performed with FFT‑based convolutions (O(n log n)).
The routine respects a 1000‑second budget and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using an FFT of power‑of‑two size.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, ∑ x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def enforce_symmetry(x: np.ndarray) -> np.ndarray:
    """Force a vector to be symmetric w.r.t. its centre."""
    return (x + x[::-1]) * 0.5


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index to another
    and project back onto the simplex (symmetrised afterwards).
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    a = project_onto_simplex(a, target_sum)
    a = enforce_symmetry(a)
    return a


def generate_shape(n: int, target_sum: float, kind: str,
                   param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence.
    kind ∈ {linear, quadratic, exponential, power, inverse,
            block, cosine}.
    param controls the shape (exponent, scale, fraction, …).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                      # ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "block":
        # param is a fraction of n (0 < param ≤ 0.5)
        k = max(1, min(int(round(param * n)), n // 2))
        raw = np.zeros(n, dtype=np.float64)
        raw[:k] = 1.0
        raw[-k:] = 1.0
    elif kind == "cosine":
        # raw = 1 + param * cos(π·i/(n‑1)), stays non‑negative
        raw = 1.0 + param * np.cos(np.pi * idx / (n - 1))
        raw = np.clip(raw, 0.0, None)
    else:
        raise ValueError(f"unknown shape kind: {kind}")

    raw = np.clip(raw, 0.0, None)
    raw = enforce_symmetry(raw)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Adam‑style smoothed gradient step
# ----------------------------------------------------------------------


def adam_smooth_step(a: np.ndarray, target_sum: float,
                     lr: float, tau_factor: float,
                     m: np.ndarray, v: np.ndarray,
                     t: int, beta1: float = 0.9,
                     beta2: float = 0.999, eps: float = 1e-12):
    """
    One Adam iteration on the τ‑smoothed max‑convolution.
    Returns (new_a, new_m, new_v, max_conv).
    """
    n = a.shape[0]

    # Self‑convolution and its current maximum
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # τ – temperature
    tau = max(tau_factor * max_conv, 1e-12)

    # Log‑sum‑exp weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Gradient of the smoothed max (derived via convolution)
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    # Adam moment updates
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    a_new = enforce_symmetry(a_new)

    return a_new, m, v, max_conv


# ----------------------------------------------------------------------
# Sub‑gradient step (true max)
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray, target_sum: float,
                     step: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient step on the actual max‑convolution.
    Returns (new_a, max_conv_before_step).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()

    eps = max(1e-12, 1e-9 * max_conv)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:                     # safety
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    a_rev = a[::-1]
    L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
    grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    a_new = enforce_symmetry(a_new)
    return a_new, max_conv


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence().  The routine respects the time budget and
    returns the best sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3      # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        a0 = np.array(base, dtype=np.float64)
    else:
        # power of two for fast FFTs – a modest size that still allows
        # many iterations within the time budget.
        n = 16384
        a0 = None

    target_sum = math.sqrt(2.0 * n)           # we keep the classic normalisation

    # ------------------------------------------------------------------
    # Phase 0 – extensive parametric sweep
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)     # uniform start (will be improved)
    best_max = compute_max_conv(best_seq)

    shape_kinds = ("linear", "quadratic", "exponential",
                   "power", "inverse", "block", "cosine")
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),
        "quadratic": np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 10),
        "inverse": np.linspace(0.5, 5.0, 10),
        "block": np.linspace(0.01, 0.20, 10),          # fraction of n
        "cosine": np.linspace(0.1, 1.0, 10),           # amplitude multiplier
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            cur_max = compute_max_conv(cand)
            if cur_max < best_max:
                best_max = cur_max
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → max={best_max:.6e}")
            if time.time() >= deadline:
                print("  Time exhausted during sweep.")
                ratio = 2 * n * best_max / (target_sum ** 2)
                print(f"  Returning early with ratio {ratio:.6f}")
                return best_seq.tolist()

    # incorporate a provided baseline if it improves the current best
    if a0 is not None:
        a0 = project_onto_simplex(a0, target_sum)
        a0 = enforce_symmetry(a0)
        cur_max = compute_max_conv(a0)
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a0.copy()
            print(f"  Baseline improves best → max={best_max:.6e}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam‑style smoothed gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr = 0.5 * (target_sum / n)               # initial learning rate
    tau_factor = 0.09                         # initial temperature factor
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    iteration = 0
    no_improve = 0
    best_ratio = 2 * n * best_max / (target_sum ** 2)

    print(f"[{time.time():.2f}] Starting Adam‑gradient phase")
    while time.time() < deadline:
        iteration += 1
        a, m, v, cur_max = adam_smooth_step(
            a, target_sum, lr, tau_factor, m, v, iteration,
            beta1=beta1, beta2=beta2, eps=eps
        )
        cur_ratio = 2 * n * cur_max / (target_sum ** 2)

        if cur_ratio + 1e-12 < best_ratio:
            best_ratio = cur_ratio
            best_max = cur_max
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration:07d} – improved ratio {best_ratio:.6f}")
        else:
            no_improve += 1

        # adaptive schedules
        if iteration % 1500 == 0:
            lr *= 0.97
            tau_factor = max(0.001, tau_factor * 0.96)

        # occasional random mass transfer to escape plateaus
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.006)

        # jitter after long stagnation
        if no_improve >= 5000:
            jitter = np.random.randn(n) * 4e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            a = enforce_symmetry(a)
            no_improve = 0
            print(f"Iter {iteration:07d} – jitter injected")

    # ------------------------------------------------------------------
    # Phase 2 – true sub‑gradient refinement (short)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Adam phase finished; best ratio {best_ratio:.6f}")
    sub_step = 0.02 * (target_sum / n)
    sub_iter = 0
    while time.time() < deadline:
        a, cur_max = subgradient_step(a, target_sum, sub_step)
        sub_iter += 1
        cur_ratio = 2 * n * cur_max / (target_sum ** 2)
        if cur_ratio + 1e-12 < best_ratio:
            best_ratio = cur_ratio
            best_max = cur_max
            best_seq = a.copy()
            print(f"Sub‑grad {sub_iter:06d} – improved ratio {best_ratio:.6f}")

        if sub_iter % 400 == 0:
            sub_step *= 0.97
        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    # ------------------------------------------------------------------
    # Phase 3 – final polishing (random symmetric transfers)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Sub‑gradient phase finished; best ratio {best_ratio:.6f}")

    for _ in range(80):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.003)
        cand_max = compute_max_conv(cand)
        cand_ratio = 2 * n * cand_max / (target_sum ** 2)
        if cand_ratio < best_ratio:
            best_ratio = cand_ratio
            best_max = cand_max
            best_seq = cand.copy()
            print(f"Polish – improved ratio {best_ratio:.6f}")

    # ------------------------------------------------------------------
    # Return best sequence as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence; final ratio {best_ratio:.6f}")
    return best_seq.tolist()
