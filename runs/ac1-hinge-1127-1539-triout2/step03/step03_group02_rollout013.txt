"""
Propose a step‑function (non‑negative coefficient vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.

Algorithm
---------
1. **Problem size** – reuse the length of an existing construction
   (`height_sequence_1`) if present, otherwise pick a power of two
   (4096, 8192 or 16384).  The sum of the coefficients is fixed to
   `√(2 n)`; with this normalisation the evaluation simplifies to the
   maximal entry of the self‑convolution.

2. **Random sparse initialisations** – repeatedly draw a *k*‑sparse
   vector (k ≈ 1.2·√(2 n)) with equal weights, project it onto the
   ℓ₁‑simplex and keep the best one seen so far.

3. **Local optimisation** – for each sparse start run a short
   sub‑gradient descent (the exact gradient of the max‑convolution) and
   a few stochastic “pair‑transfer’’ moves.  The step size decays
   slowly; after every few hundred sub‑gradient steps a tiny random
   jitter is added to avoid stagnation.

4. **Simulated‑annealing style exploration** – keep a current candidate,
   propose a random mass transfer (move a random fraction of a random
   positive entry to a random destination) and accept it either if it
   improves the score or with probability exp(−Δ/τ) where τ slowly
   cools down.  Acceptances that improve the global best are recorded.

5. **Polishing** – after the time budget is almost exhausted perform a
   final burst of tiny random pair transfers and keep any improvement.

All heavy work uses FFT‑based convolution (O(n log n)).  The routine
returns the best sequence observed before the 1000 s deadline.
"""

import time
import math

def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Algorithm of Duchi et al. (2008), O(n).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case – fall back to uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (a≥0)."""
    return float(conv_self_fft(a).max())


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small amount of mass from one random index to another.
    The amount is proportional to the source entry; after the move the
    vector is projected back onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    # pick a source that currently carries some mass
    pos = np.flatnonzero(a)
    if pos.size == 0:
        return a
    i = np.random.choice(pos)
    j = np.random.choice(n)
    if i == j:
        return a
    delta = scale * a[i]
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def subgradient_step(a: np.ndarray, target_sum: float,
                     step: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient step on the true max‑convolution.
    Returns the new vector and the (pre‑step) maximal convolution value.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # mask of maximising indices (allow a tiny tolerance)
    eps = max(1e-12, 1e-9 * max_conv)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:            # safety fallback
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    # gradient = 2 / cnt * (mask * a_rev)  (convolution in Fourier domain)
    a_rev = a[::-1]
    L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
    grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, max_conv


def random_sparse_vector(n: int, k: int, target_sum: float) -> np.ndarray:
    """
    Create a k‑sparse non‑negative vector of length n, each non‑zero entry
    equal to target_sum/k.  The support is chosen uniformly without replacement.
    """
    idx = np.random.choice(n, k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = target_sum / k
    return a

        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[{time.time():.2f}] Polishing improved → {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score {best_score:.6f})")
    return best_seq.tolist()
