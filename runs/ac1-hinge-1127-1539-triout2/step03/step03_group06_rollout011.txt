def _run_softmax_opt(a0, n, deadline, eta0=0.5, beta=0.9, tau0=None, tau_decay=0.9995, eta_decay=0.9999, max_iters=200000):

    a = a0.copy()
    S = math.sqrt(2.0*n)
    best_max = np.inf
    best_seq = None
    m = np.zeros_like(a)
    it = 0
    while time... and it<max_iters:
        it += 1
        conv = _full_convolve(a, a)
        cur_max = np.max(conv)
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
        if tau0 is None:
            # set tau to cur_max - min, but ensure >0
            tau = max(cur_max - np.min(conv), 1e-3)
        else:
            tau = tau0
        # compute softmax weights
        exp_term = np.exp((conv - cur_max)/tau)
        w = exp_term / np.sum(exp_term)

        # gradient
        w_rev = w[::-1]
        grad_full = _full_convolve(a, w_rev)  # length 3n-2
        g = 2.0 * grad_full[n-1:2*n-1]

        # momentum
        m = beta * m + (1 - beta) * g

        # step scaling
        step = -eta0 * m / (np.mean(np.abs(m)) + 1e-12)

        # multiplicative update
        a = a * np.exp(step)

        # renorm to sum S
        total = a.sum()
        if total <= 0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # decay parameters
        eta0 *= eta_decay
        if tau0 is not None:
            tau0 *= tau_decay

        # jitter occasional
        if np.random.rand() < 0.02:
            # add noise
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0, None)
            a *= S / a.sum()

        if time.time() >= deadline - 0.1:
            break

    return best_seq, best_max

"""
Search for a non‑negative step function minimizing ``evaluate_sequence``.

The algorithm combines two optimisation schemes:

* **Soft‑max gradient descent** – a smooth surrogate of the
  max‑convolution (τ‑temperature log‑sum‑exp) is minimised using a
  multiplicative (exponentiated) update with momentum.  The temperature
  τ is gradually reduced, so the method starts with a smooth landscape
  and ends with a hard max‑sub‑gradient.

* **Top‑K exponentiated gradient** – the original fast method that
  updates only with the most violating convolution shifts.  It is kept
  as a cheap fallback.

Both schemes keep the ℓ₁‑norm fixed to ``√(2 n)`` (the external score is
scale‑invariant).  Periodically a tiny linear‑programming refinement
(solving the “max‑convolution’’ LP) is applied when SciPy is available.

The search runs many independent restarts with different lengths,
initialisations and hyper‑parameters inside the 1000 s wall‑clock
budget, always keeping the best sequence seen so far.
"""

# Basic convolution utilities (FFT based, full convolution)
# ----------------------------------------------------------------------
def _full_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    """
    n = a.shape[0]
    m = b.shape[0]
    conv_len = n + m - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, fft_len)
    Fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(Fa * Fb, fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray):
    """Return (full convolution, maximal entry, index of the maximum)."""
    conv = _full_convolve(a, a)
    max_idx = int(conv.argmax())
    return conv, float(conv[max_idx]), max_idx


# ----------------------------------------------------------------------
# Top‑K exponentiated‑gradient optimiser (the original method)
# ----------------------------------------------------------------------
def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the sum of selected convolution entries.
    For each shift k we add 2·a_{k‑i} to the i‑th component whenever the
    index lies inside the vector.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g


def _run_optimize_one_start(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            top_k_start: int = 3,
                            eta0: float = 0.8,
                            beta: float = 0.9,
                            decay: float = 0.99995,
                            jitter_prob: float = 0.02,
                            max_iters: int = 200_000):
    """
    Exponentiated‑gradient descent (with momentum) on the max‑convolution.
    Returns the best vector seen during the run and its max‑convolution.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)                # target ℓ₁‑norm (kept fixed)
    best_local_max = float('inf')
    best_local_seq = a.copy()

    v = np.zeros_like(a)                 # momentum buffer
    eta = eta0
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv, cur_max, _ = _conv_max(a)

        # adapt the number of active shifts – increase if we have stalled
        top_k = top_k_start
        if it - last_improve > 2_000 and top_k < 30:
            top_k = min(30, top_k * 2)

        if conv.size <= top_k:
            shifts = np.arange(conv.size)
        else:
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient of the sum of the selected shifts
        g = _gradient_from_shifts(a, shifts)

        # momentum (exponential moving average)
        v = beta * v + (1.0 - beta) * g

        # multiplicative (exponentiated) update – guarantees positivity
        a = a * np.exp(-eta * v)

        # renormalise to keep the ℓ₁‑norm fixed
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional random jitter to escape tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # step‑size decay
        eta *= decay

        # record improvement of the surrogate (which equals the external score)
        if cur_max < best_local_max - 1e-12:
            best_local_max = cur_max
            best_local_seq = a.copy()
            last_improve = it

        if time.time() >= deadline - 0.1:
            break

    return best_local_seq, best_local_max


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient descent optimiser
# ----------------------------------------------------------------------
def _run_softmax_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          eta0: float = 0.5,
                          beta: float = 0.9,
                          tau_factor: float = 0.2,
                          eta_decay: float = 0.9995,
                          tau_decay: float = 0.999,
                          jitter_prob: float = 0.02,
                          max_iters: int = 20_000):
    """
    Minimises a smoothed version of the max‑convolution:
        L(a) = τ·log Σ_k exp( (a∗a)[k] / τ )
    using exponentiated‑gradient updates with momentum.
    Returns the best vector seen and its max‑convolution.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)
    best_max = float('inf')
    best_seq = a.copy()
    m = np.zeros_like(a)          # momentum
    eta = eta0
    tau = None
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # full self‑convolution
        conv = _full_convolve(a, a)
        cur_max = float(conv.max())

        # keep best so far
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()

        # initialise / decay temperature
        if tau is None:
            diff = cur_max - float(conv.min())
            tau = max(diff * tau_factor, 1e-4)
        else:
            tau = max(tau * tau_decay, 1e-6)

        # soft‑max weights (stable log‑sum‑exp)
        exp_term = np.exp((conv - cur_max) / tau)
        w = exp_term / np.sum(exp_term)          # shape (2n‑1,)

        # gradient of L – can be obtained by a convolution with the reversed weight vector
        w_rev = w[::-1]
        grad_full = _full_convolve(a, w_rev)    # length 3n‑2
        g = 2.0 * grad_full[n - 1:2 * n - 1]     # length n

        # momentum
        m = beta * m + (1.0 - beta) * g

        # scale step to avoid overflow; use mean absolute magnitude as normaliser
        scale = eta / (np.mean(np.abs(m)) + 1e-12)
        step = -scale * m
        step = np.clip(step, -50.0, 50.0)        # guard against extreme exponents

        # exponentiated update and renormalisation
        a = a * np.exp(step)
        total = a.sum()
        if total <= 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # jitter (helps escape plateaus)
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # decay learning rate
        eta *= eta_decay

        if time.time() >= deadline - 0.1:
            break

    return best_seq, best_max

# Linear‑programming refinement (optional, SciPy based)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    """
    Construct (2n‑1) × n matrix H where (H @ g)[k] = (f∗g)[k].
    Used only inside the LP refinement (n ≤ 3000 for tractability).
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5):
    """
    Given a vector ``f`` solve the LP
        maximise   Σ g_i
        s.t.       g_i ≥ 0,
                   (f∗g)[k] ≤ M  ∀k,
    where M = max_k (f∗f)[k].
    The resulting ``g`` is mixed back with ``f`` and renormalised.
    Several iterations improve the maximal convolution entry.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:                 # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f

        print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    else:
        # ------------------------------------------------------------------
        # Fallback – uniform sequence (should never happen).
        # ------------------------------------------------------------------
        n = 2000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    return best_seq.tolist()
