"""
Propose a non‑negative step‑function (vector of heights) that minimizes  

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.

The algorithm works in four stages:

1. **Exploratory sweep** – several parametric families (linear, quadratic,
   exponential, Hamming) together with a few random Dirichlet vectors are
   generated, projected onto the ℓ₁‑simplex (∑a = √(2 n)) and the best
   candidate is kept.

2. **Adam‑based smoothed gradient descent** – we minimise a
   log‑sum‑exp smoothed version of the max‑convolution.  Gradients are
   obtained via FFT, the update follows the Adam optimiser and the result
   is projected back onto the simplex after each step.

3. **Diminishing‑step sub‑gradient descent** – after the smoothed phase
   we switch to the exact sub‑gradient of the max‑convolution (the gradient
   of the currently active shift).  The step size decays as 1/√t.

4. **Simulated‑annealing polish** – random pair‑mass transfers are
   accepted if they improve the objective, otherwise with a temperature‑
   dependent probability.  This helps to escape tiny plateaux that the
   deterministic phases may miss.

All stages respect a global time budget (default 1000 s) and the best
vector found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# Helper functions – top level, no closures or lambdas

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = target}.
    O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another
    and re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Symmetric “U” shaped sequence.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge‑enhancement.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_hamming_shape(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hamming) shape, naturally symmetric."""
    idx = np.arange(n, dtype=np.float64)
    raw = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    return project_onto_simplex(raw, target_sum)


def generate_random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """Draw a random point on the simplex via a Dirichlet‑like construction."""
    # Gamma(1,1) = Exp(1) → generate iid exponential variables
    raw = np.random.exponential(scale=1.0, size=n)
    return project_onto_simplex(raw, target_sum)


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of log‑sum‑exp smoothed max‑convolution.
    Returns (gradient, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # smoothing temperature (never too small)
    tau = max(tau_factor * max_conv, 1e-12)

    # stable weights for log‑sum‑exp
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # grad = (2/τ) * (w * a_rev) restricted to central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def _subgradient_max(a: np.ndarray):
    """
    Exact sub‑gradient of max(conv(a,a)).
    Returns (gradient, max_conv, idx_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())

    # i runs over all valid pairs i + j = idx_max
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    # diagonal terms (i=j) have derivative 2*a_i, the formula above already gives that.
    return grad, max_conv, idx_max

    # Adam hyper‑parameters

    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # learning‑rate scale – roughly 0.1 · average entry size
    eta0 = 0.15 * (target_sum / n)
    tau_factor = 0.05         # starting smoothing temperature factor
    no_improve = 0

    while time.time() < deadline:
        t_iter += 1
        grad, cur_max = _smoothed_gradient(a, tau_factor)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = eta0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)

        a = a_new

        # track improvement
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"Iter {t_iter:06d} – improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # occasional adaptive tweaks
        if t_iter % 800 == 0:
            eta0 *= 0.93
            tau_factor = max(0.001, tau_factor * 0.94)

        # random jitter (helps escape very flat plateaus)
        if no_improve >= 2500:
            jitter = np.random.randn(n) * 5e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {t_iter:06d} – jitter applied")

        # early exit if we beat the target bound
        if best_score < 1.503:
            print(f"Target bound reached ({best_score:.6f}) – stopping early")
            break

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent with diminishing step size
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – sub‑gradient refinement")
    a = best_seq.copy()
    # initial step roughly the size of a typical entry
    step = 0.5 * (target_sum / n)
    sub_iter = 0
    while time.time() < deadline and sub_iter < 5000:
        sub_iter += 1
        grad, cur_max, idx_max = _subgradient_max(a)
        a = a - step * grad
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)

        # step size schedule (classical 1/√t)
        step = max(step * 0.9995, (target_sum / n) * 0.01)

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            print(f" Sub‑grad iter {sub_iter:04d} – improved to {best_score:.6f}")
            # allow a brief step‑size increase after a success
            step = max(step, 0.5 * (target_sum / n))

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing polishing (pair transfers)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 4 – simulated‑annealing polishing")
    temperature = 0.02 * best_score          # initial temperature
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 20000:
        anneal_iter += 1
        # propose a tiny pair transfer
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.006)
        conv_cand = conv_self_fft(cand)
        cand_score = conv_cand.max()

        delta = cand_score - best_score
        if delta < 0:
            best_score = cand_score
            best_seq = cand
            # cool down a little faster after improvements
            temperature *= 0.985
            print(f"  SA iter {anneal_iter:05d} – improved to {best_score:.6f}")
        else:
            # accept with Metropolis probability
            if temperature > 0 and np.random.rand() < math.exp(-delta / temperature):
                best_seq = cand
                best_score = cand_score
            temperature *= 0.9999  # slow cooling

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Return the best sequence (Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Search finished – final best score {best_score:.6f}")
    return best_seq.tolist()
