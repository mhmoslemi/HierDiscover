def _optimize_local(a0, deadline, rng):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    # compute conv and max
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # hyperparameters
    eta = 0.6
    eta_decay = 0.9998
    tau = cur_max # initial tau
    tau_decay = 0.9995
    tau_min = 1e-4

    move_prob = 0.12
    epsilon_move = 0.05
    temp = cur_max * 0.1
    temp_decay = 0.9998
    jitter_prob = 0.015
    jitter_scale = 0.02

    while time.time() < deadline:
        # recompute conv and max (if not freshly computed)
        # We may have stored from previous step.

        # random mass swap
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    b_candidate = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_candidate)
                    accept = False
                    if cand_max <= cur_max:
                        accept = True
                    else:
                        prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                        if rng.random() < prob:
                            accept = True
                    if accept:
                        a = a_candidate
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                    temp *= temp_decay
        else:
            if tau > tau_min:
                # softmax gradient step
                a, b, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
                tau *= tau_decay
            else:
                # hard max gradient step with line search
                # compute k_max and gradient
                # gradient calculation
                b = _full_convolution_fft(a)
                cur_max = np.max(b)
                k_star = np.argmax(b)
                g = _grad_max_at_k(a, k_star)

                # line search
                alpha = eta  # treat eta as step scale
                improved = False
                for _ in range(8):
                    a_candidate = _project_to_simplex(a - alpha * g, S)
                    b_candidate = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_candidate)
                    if cand_max < cur_max:
                        a = a_candidate
                        cur_max = cand_max
                        improved = True
                        break
                    else:
                        alpha *= 0.5
                if not improved:
                    # fallback: jitter small
                    if rng.random() < jitter_prob:
                        noise = rng.normal(scale=a.mean()*jitter_scale, size=n)
                        a = np.clip(a + noise, 0.0, None)
                        a = _project_to_simplex(a, S)
                        cur_max = np.max(_full_convolution_fft(a))

                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
        # occasional jitter
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean()*jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        eta *= eta_decay

    return best_a, best_max

"""
Propose a non‑negative step‑function (list of heights) that minimizes the
自‑convolution peak.  The algorithm mixes three simple ideas:

1.  Initialise a random feasible vector (non‑negative, ℓ₁‑norm √(2·n)).
2.  Run a fast hybrid optimiser:
    • a soft‑max gradient descent phase (smooth approximation of the ∞‑norm),
    • a hard‑max projected gradient phase with an explicit back‑tracking line‑search,
    • occasional simulated‑annealing mass swaps and Gaussian jitter.
3.  Perform many independent restarts (different lengths n) within the
    1000‑second budget and keep the best sequence found.  The final best
    sequence is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (non‑negative, fixed ℓ₁‑norm)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]               # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution using a real FFT (linear‑time O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Initialise a feasible vector (ℓ₁‑norm = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = np.asarray(base_seq[:n], dtype=float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                                 # Dirichlet(1) seed
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient step (smooth approximation of the ∞‑norm)
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float) -> tuple[np.ndarray,
                                                                  np.ndarray,
                                                                  float]:
    """One soft‑max gradient step followed by projection onto the simplex."""
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # current convolution and its maximum
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau <= 1e-12:                # treat as hard max (sub‑gradient)
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]                     # diagonal counted once
    else:                                 # smooth soft‑max gradient
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)               # soft‑max weights

        # cross‑correlation a ⋆ w
        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
        g = 2.0 * conv[n - 1:n - 1 + n]

    # additive step + projection
    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    # recompute convolution after projection
    b_new = _full_convolution_fft(a_proj)
    max_new = np.max(b_new)

    return a_proj, b_new, max_new


# ----------------------------------------------------------------------
# Gradient of the hard max (sub‑gradient of max(b) at the current argmax)
# ----------------------------------------------------------------------
def _grad_max_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Compute the sub‑gradient of max(b) w.r.t. a at the index k where b attains its
    maximum.  The formula is g_i = 2·a_{k-i} (zero when the partner index is out
    of bounds).  Implemented via array slicing for speed.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    offset = n - 1 - k                  # shift needed on the reversed array
    g = np.zeros_like(a)
    start = max(0, -offset)
    end = min(n, n - offset)
    if start < end:
        g[start:end] = 2.0 * a_rev[offset + start: offset + end]
    return g


# ----------------------------------------------------------------------
# Local optimiser (one restart) – hybrid soft‑max / hard‑max descent
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mixed optimisation:
       – soft‑max projected gradient descent (smooth phase),
       – hard‑max projected gradient descent with back‑tracking (refinement),
       – occasional simulated‑annealing mass swaps,
       – rare Gaussian jitter.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best record
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters
    eta = 0.6
    eta_decay = 0.9997
    move_prob = 0.12
    epsilon_move = 0.04
    temp = cur_max * 0.1
    temp_decay = 0.9998
    jitter_prob = 0.01
    jitter_scale = 0.02

    tau = cur_max           # start temperature for soft‑max
    tau_decay = 0.9995
    tau_min = 1e-4

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) random mass swap (simulated‑annealing)
        # --------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta

                    b_candidate = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_candidate)

                    accept = False
                    if cand_max <= cur_max:
                        accept = True
                    else:
                        prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                        if rng.random() < prob:
                            accept = True

                    if accept:
                        a = a_candidate
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                    temp *= temp_decay
        # --------------------------------------------------------------
        # 2) gradient descent step
        # --------------------------------------------------------------
        else:
            if tau > tau_min:
                # soft‑max phase
                a, b, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
                tau *= tau_decay
            else:
                # hard‑max phase with back‑tracking line search
                b = _full_convolution_fft(a)
                cur_max = np.max(b)
                k_star = np.argmax(b)
                g = _grad_max_at_k(a, k_star)

                # start with a step comparable to the current learning rate
                alpha = eta
                improved = False
                for _ in range(8):
                    a_candidate = _project_to_simplex(a - alpha * g, S)
                    b_candidate = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_candidate)
                    if cand_max < cur_max:
                        a = a_candidate
                        cur_max = cand_max
                        improved = True
                        break
                    alpha *= 0.5
                if not improved:
                    # fallback jitter (treated later)
                    pass

                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()

        # --------------------------------------------------------------
        # 3) occasional jitter
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # decay schedules
        eta *= eta_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a low self‑convolution peak.
    The routine performs many restarts of a hybrid optimiser (soft‑max
    + hard‑max descent) within the allotted time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # small safety margin

    # try to reuse a previously discovered construction, if any
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # main loop – many restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # choose a length – explore a fairly wide interval
        if base_seq is not None:
            max_len = max(800, min(8000, int(base_seq.size * 1.5)))
        else:
            max_len = 8000
        n = rng.integers(800, max_len + 1)

        # initialise a feasible vector
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a slice of the remaining time to this restart
        slice_deadline = min(time.time() + 40.0, deadline)

        # run the optimiser
        a_opt, internal_max = _optimize_local(a0, slice_deadline, rng)

        # evaluate through the black‑box function
        seq = a_opt.tolist()
        external_score = evaluate_sequence(seq)
        if external_score < best_overall_score:
            best_overall_score = external_score
            best_overall_seq = seq
            print(f"[global] attempt={attempt} n={n} "
                  f"new best external score={external_score:.6f} "
                  f"(internal max={internal_max:.6f})")

    # ------------------------------------------------------------------
    # final refinement of the best sequence (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_overall_seq is not None:
        remaining_deadline = deadline
        a_start = np.asarray(best_overall_seq, dtype=float)
        a_refined, _ = _optimize_local(a_start, remaining_deadline, rng)
        refined_seq = a_refined.tolist()
        refined_score = evaluate_sequence(refined_seq)
        if refined_score < best_overall_score:
            best_overall_score = refined_score
            best_overall_seq = refined_seq
            print(f"[refine] improved external score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity check and return
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = [1.0]  # fallback (should never happen)

    print(f"[final] best external score = {best_overall_score:.6f} "
          f"length = {len(best_overall_seq)}")
    globals()['height_sequence_1'] = best_overall_seq   # store for possible reuse
    return best_overall_seq
