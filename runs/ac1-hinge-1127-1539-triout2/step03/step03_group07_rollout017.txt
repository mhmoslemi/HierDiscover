def project_simplex(v):
    # Ensure non-negative sum=1
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

        scaled = alpha * conv
        # stable softmax
        c = np.max(scaled)

        a_sum = a.sum()
        a = a / a_sum
        # schedule
        alpha = min(alpha * alpha_factor, alpha_max)
        lr *= lr_decay
    # final best conv

"""
Improved search for a low‑score step function.

The score is
    2·n·max(a*a) / (sum(a)²)

which is scale‑invariant.  We therefore keep the L1‑norm of the
candidate equal to √(2·n).  The algorithm consists of two phases:

1.  **Smooth‑max exponentiated gradient descent**.
    We minimise a smooth approximation of the “maximum” of the
    self‑convolution:
        f(a) = (1/α)·log Σₖ exp(α·(a∗a)ₖ) .
    For a fixed α the gradient is
        ∇f = 2·corr(w, a)    with   wₖ ∝ exp(α·(a∗a)ₖ) .
    The update is multiplicative (aᵢ←aᵢ·exp(−η·∇fᵢ)) which keeps the
    vector non‑negative; after each step we renormalise to Σa = √(2·n).

    α is increased gradually so that the smooth objective approaches the
    true maximum.  The iteration is stopped when the time budget is
    exhausted or a maximum number of steps is reached.

2.  **Local stochastic hill‑climbing (simulated annealing)**
    started from the best vector found in phase 1.  A small fraction of
    mass is moved between two random indices; the move is accepted if it
    improves the true maximum or with a probability that decays with a
    temperature schedule.

The two phases are run repeatedly with random restarts, various
lengths (2000, 4000, 8000) and Dirichlet concentration parameters.
The best sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a∗a (full length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights wₖ ∝ exp(α·convₖ) (stable)."""
    scaled = alpha * conv
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient ∇f = 2·corr(w,a) where corr(w,a)[i] = Σⱼ w_{i+j} aⱼ.
    Implemented via an FFT‑based convolution of w with a reversed.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(w.shape[0] + n - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    corr_full = np.fft.irfft(W * A_rev, n=L)
    start = n - 1
    g = 2.0 * corr_full[start : start + n]
    return g


def _exp_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8000,
    alpha0: float = 0.3,
    alpha_max: float = 9.0,
    alpha_factor: float = 1.02,
    lr0: float = 0.5,
    lr_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max exponentiated gradient descent.
    Returns the best vector (still summed to √(2·n)) and its maximal
    self‑convolution value (the external score, because Σa = √(2·n)).
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    conv = _conv_self(a)
    best_max = float(conv.max())

    alpha = alpha0
    lr = lr0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_max = float(conv.max())
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            print(
                f"[exp descent improve] t={time.time():.1f}s n={n} it={it} max={cur_max:.6f}"
            )

        # soft‑max weights
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth max
        g = _gradient_from_weights(a, w)

        # exponentiated (multiplicative) update
        exp_arg = -lr * g
        exp_arg = np.clip(exp_arg, -20, 20)  # avoid overflow/underflow
        a = a * np.exp(exp_arg)

        # keep positivity and renormalise to Σa = √(2·n)
        a = np.maximum(a, 1e-16)
        a_sum = a.sum()
        if a_sum == 0.0:
            a.fill(S / n)
        else:
            a *= (S / a_sum)

        # schedule updates
        alpha = min(alpha * alpha_factor, alpha_max)
        lr *= lr_decay

    return best_a, best_max


def _local_hill_climb(
    a_start: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style hill‑climbing on the true
    max‑convolution (no smoothing).  Works directly with vectors that
    already satisfy Σa = √(2·n).
    """
    a = a_start.copy()
    cur_max = float(np.max(_conv_self(a)))
    best_a = a.copy()
    best_max = cur_max

    temperature = max(0.05 * cur_max, 1e-4)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        frac = rng.uniform(0.04, 0.18)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = float(np.max(_conv_self(a_new)))

        if new_max < cur_max - 1e-12:
            a, cur_max = a_new, new_max
            if new_max < best_max:
                best_max, best_a = new_max, a_new.copy()
                print(
                    f"[hill climb improve] t={time.time():.1f}s n={n} it={it} max={new_max:.6f}"
                )
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a, cur_max = a_new, new_max

        temperature *= 0.9995

        # occasional jitter to escape plateaus
        if rng.random() < 0.001:
            a += rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()
            cur_max = float(np.max(_conv_self(a)))
            if cur_max < best_max:
                best_max, best_a = cur_max, a.copy()
                print(
                    f"[jitter] t={time.time():.1f}s n={n} it={it} max={cur_max:.6f}"
                )

    return best_a, best_max

            alpha_factor=1.02,
            lr0=rng.uniform(0.3, 0.7),
            lr_decay=0.9995,
        )
        score = max_conv  # because Σa = √(2·n)

        # ------------------- Phase 2 : hill climbing (optional) -----------
        if rng.random() < 0.35:  # occasionally polish the candidate
            a_opt, max_conv = _local_hill_climb(
                a_opt, n, deadline, rng, max_iters=15000
            )
            score = max_conv

        # ------------------- Record global improvement --------------------
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_opt.copy()
            print(
                f"[global improve] run={run_id} n={n} score={best_score:.6f}"
            )
        run_id += 1

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a valid return.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification (the platform will evaluate again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
