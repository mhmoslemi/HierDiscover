Propose a non‑negative step‑function (vector of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a multi‑start Adam‑based smoothed gradient descent, followed by a
sub‑gradient phase and a short simulated‑annealing polishing loop.
The algorithm explores several parametric families for the initial guess,
adds random Dirichlet / sparse starts, and continuously projects onto the
ℓ₁‑simplex (∑a = √(2 n)).  It respects a 1000 s budget and always returns the
best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / projection utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution via FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    O(n) algorithm (Duchi et al. 2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small amount of mass from a random index i to a random index j,
    then project back onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Families of parametric “U‑shapes” (edges higher than centre)
# ----------------------------------------------------------------------


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric U‑shaped sequence.
    kind ∈ {"linear","quadratic","exponential"}.
    param ≥ 0 controls edge enhancement.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance to centre ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_edge_decay(
    n: int,
    target_sum: float,
    power: float = 1.0,
) -> np.ndarray:
    """
    Edge‑heavy shape a_i ∝ (distance to nearest edge + 1)^(-power).
    """
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to the nearest edge
    raw = (d + 1) ** (-power)
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of τ‑smoothed max‑convolution (log‑sum‑exp).
    Returns (gradient, current_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-8)       # avoid division‑by‑zero
    # stable log‑sum‑exp weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = (2/τ) * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


# ----------------------------------------------------------------------
# Sub‑gradient of the (non‑smooth) max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a, a)).
    Returns (gradient, max_conv, idx_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())
    # indices of a that contribute to the maximal entry
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimizes
    `evaluate_sequence`.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    # --------------------------------------------------------------
    # Determine problem size (reuse previous construction if possible)
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_from_base = np.array(base, dtype=np.float64)
    else:
        # a length that is large enough to allow improvement but still cheap
        n = np.random.randint(7200, 9400)
        init_from_base = None

    target_sum = np.sqrt(2.0 * n)            # normalisation: objective = max(conv)

    # --------------------------------------------------------------
    # Helper to compute the objective (max of the self‑convolution)
    # --------------------------------------------------------------
    def max_conv(a: np.ndarray) -> float:
        return conv_self_fft(a).max()

    # --------------------------------------------------------------
    # Generate a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # uniform baseline
    candidates.append(np.full(n, target_sum / n))

    # previous best if available
    if init_from_base is not None:
        candidates.append(init_from_base.copy())

    # parametric U‑shapes – extended parameter ranges
    for kind in ("linear", "quadratic", "exponential"):
        # broader sweeps (more points) to explore extremes
        for p in np.linspace(0.0, 8.0, 33):
            if kind == "exponential" and p > 4.0:
                continue
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            candidates.append(cand)

    # edge‑decay family (mass concentrated near the ends)
    for p in np.linspace(0.0, 5.0, 21):
        cand = generate_edge_decay(n, target_sum, power=p)
        candidates.append(cand)

    # random Dirichlet draws with several concentration parameters
    for alpha in (0.2, 0.5, 1.0, 2.0):
        rand = np.random.dirichlet(np.full(n, alpha))
        candidates.append(rand * target_sum)

    # sparse random starts (few non‑zero entries)
    for k in (5, 10, 20):
        idx = np.random.choice(n, k, replace=False)
        vals = np.random.rand(k)
        vec = np.zeros(n)
        vec[idx] = vals
        candidates.append(project_onto_simplex(vec, target_sum))

    # -----------------------------------------------------------------
    # Pick the best among the crude pool
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        cur = max_conv(cand)
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] initial best score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 1 – Adam‑based smoothed gradient descent
    # -----------------------------------------------------------------
    a = best_seq.copy()
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    # initial learning‑rate scale (≈ step size relative to the average entry)
    eta0 = 0.6 * (target_sum / n)
    tau_factor = 0.2                       # τ = τ_factor * max_conv
    no_improve = 0

    while time.time() < deadline:
        grad, cur_max = _smoothed_gradient(a, tau_factor)
        t_iter += 1
        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        a = a_new

        # record improvement
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Adam iter {t_iter:06d} → {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # schedules
        if t_iter % 800 == 0:
            eta0 *= 0.9
            tau_factor = max(tau_factor * 0.92, 5e-4)
        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.015)

        # heavy jitter after long stagnation
        if no_improve >= 3000:
            jitter = np.random.randn(n) * 4e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"[{time.time():.2f}] jitter applied at iter {t_iter}")

        # early exit if we already beat the target bound
        if best_score < 1.503:
            print(f"[{time.time():.2f}] target bound reached ({best_score:.6f}) – stopping Adam")
            break

    # -----------------------------------------------------------------
    # Phase 2 – Sub‑gradient refinement (focus on the currently maximal entry)
    # -----------------------------------------------------------------
    eta_sub = 0.5 * (target_sum / n)
    sub_iter = 0
    while time.time() < deadline and sub_iter < 2500:
        grad, cur_max, idx_max = _subgradient_max(a)
        a = a - eta_sub * grad
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)
        sub_iter += 1

        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.2f}] sub‑grad iter {sub_iter:04d} → {best_score:.6f}")
            # slowly increase step size after a successful improvement
            eta_sub = min(eta_sub * 1.05, 0.8 * (target_sum / n))
        elif sub_iter % 500 == 0:
            eta_sub *= 0.85

        if best_score < 1.503:
            break

    # -----------------------------------------------------------------
    # Phase 3 – Simulated‑annealing style polishing with random pair transfers
    # -----------------------------------------------------------------
    temp = 0.02               # initial temperature for Metropolis acceptance
    decay = 0.9999            # geometric decay per move
    moves = 0
    while time.time() < deadline and moves < 20000:
        moves += 1
        # adapt the transfer scale gradually
        scale = 0.008 * (0.5 ** (moves // 2000))
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=scale)
        cand_score = max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] polishing improvement {best_score:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-8)):
                best_seq = cand.copy()
        temp *= decay
        # occasional reset to the globally best known configuration
        if moves % 3000 == 0:
            best_seq = best_seq.copy()

        if best_score < 1.503:
            break

    # -----------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # -----------------------------------------------------------------
    print(f"[{time.time():.2f}] search finished – final best score {best_score:.6f}")
    return best_seq.tolist()
