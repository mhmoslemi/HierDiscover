def subgradient(a, target_sum):

    conv = conv_self_fft(a)

    max_conv = conv.max()

    eps = max_conv * 1e-9

    mask = (conv >= max_conv - eps).astype(np.float64)

    cnt = mask.sum()

    if cnt == 0: mask[conv.argmax()] = 1; cnt=1

    # gradient: (2/cnt) * conv_full[n-1:n-1+n] where conv_full = mask convolved with a_rev

    a_rev = a[::-1]; L = 1 << ((len(mask)+len(a_rev)-1).bit_length())

    M = np.fft.rfft(mask, n=L)

    A_rev = np.fft.rfft(a_rev, n=L)

    conv_full = np.fft.irfft(M * A_rev, n=L)[:len(mask)+len(a_rev)-1]

    grad = (2.0 / cnt) * conv_full[n-1:n-1+n]

    return grad, max_conv

"""
Propose a non‑negative step‑function that minimizes

        2 * n * max( conv(a, a) ) / ( sum(a)**2 )

The ratio is scale‑invariant; we therefore fix the ℓ₁‑norm to
    S = √(2 n) .
Then the objective equals max(conv(a,a)).  The algorithm works in three
phases:

1. **Parametric sweep** – generate a variety of symmetric “U‑shapes’’
   (linear, quadratic, exponential, power, inverse) and a few
   Beta‑distributions.  Each candidate is projected onto the simplex
   {x≥0, Σx=S} and the best one (smallest max‑convolution) is kept.

2. **Adam sub‑gradient optimisation** – minimise the *true* max‑convolution
   via the sub‑gradient (average of the gradients of all maximal
   lags).  Adam’s adaptive moments accelerate convergence.  After every
   update the vector is projected back onto the simplex.

3. **Simulated‑annealing polishing** – random pair‑mass transfers are
   accepted if they improve the objective, otherwise with a
   temperature‑controlled probability.  Temperature slowly cools;
   occasional jitter helps escaping tiny plateaus.

The whole search respects a 1000 s time budget and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target }.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries become zero – fallback to uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution of a."""
    return float(conv_self_fft(a).max())


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another
    and project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    # projection guarantees exact ℓ₁‑norm = target_sum
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Shape generators for the parametric sweep
# ----------------------------------------------------------------------


def generate_u_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    param controls the curvature (≥0 for all but power/inverse where it is the exponent).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    # scale to the right ℓ₁‑norm and project for safety
    raw = raw / raw.sum() * target_sum
    return project_onto_simplex(raw, target_sum)


def generate_beta_shape(n: int, target_sum: float, alpha: float, beta: float) -> np.ndarray:
    """
    Shape based on the Beta distribution density (t^(α) (1‑t)^(β)).
    """
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    raw = (t ** alpha) * ((1.0 - t) ** beta)
    raw += 1e-6                     # avoid exact zeros
    raw = raw / raw.sum() * target_sum
    return project_onto_simplex(raw, target_sum)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (used by Adam)
# ----------------------------------------------------------------------


def subgradient(a: np.ndarray, target_sum: float) -> tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of the objective max(conv(a,a))
    and return (grad, max_conv).
    The gradient is the average of the gradients of all (approximately)
    maximal lags.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_c = conv.max()
    # a tiny tolerance makes the mask robust to floating‑point noise
    eps = max(1e-12, 1e-9 * max_c)
    mask = (conv >= max_c - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        # fallback – use the single maximal index
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    # gradient = (2 / cnt) * (mask * a_rev) convolution, central n entries
    a_rev = a[::-1]
    L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
    grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]
    return grad, max_c


# ----------------------------------------------------------------------
# Adam optimisation of the sub‑gradient
# ----------------------------------------------------------------------


def adam_optimize(a: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of max(conv(a,a)).  Stops when the deadline
    is reached and returns the best vector seen together with its score.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initial learning‑rate – a few percent of the average entry size
    alpha = 0.25 * (target_sum / n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_max_conv(a)

    t = 0
    while time.time() < deadline:
        t += 1
        grad, cur_max = subgradient(a, target_sum)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter {t:07d} – new best {best_score:.6f}")

        # simple learning‑rate decay
        if t % 2000 == 0:
            alpha *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------


def annealing_refine(a: np.ndarray,
                     target_sum: float,
                     cur_score: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Random pair‑mass transfers accepted with a Metropolis criterion.
    Temperature slowly cools; occasional jitter restarts help escaping
    tiny plateaus.
    """
    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_score = cur_score

    T = 0.02 * best_score      # initial temperature
    decay = 0.9999
    it = 0

    while time.time() < deadline:
        it += 1
        # scale of the random transfer is linked to the current temperature
        scale = max(1e-4, min(0.03, T / best_score))
        cand = random_pair_transfer(cur_a.copy(), target_sum, scale=scale)
        cand_score = compute_max_conv(cand)

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            print(f"[{time.time():.2f}] Anneal iter {it:07d} – new best {best_score:.6f}")

        T *= decay

        # occasional jitter around the best point
        if it % 5000 == 0:
            jitter = np.random.randn(len(a)) * (scale * 0.5)
            cur_a = project_onto_simplex(np.maximum(best_a + jitter, 0.0), target_sum)
            cur_score = compute_max_conv(cur_a)

    return best_a, best_score

    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.array(base, dtype=np.float64)
    else:
        # use a power‑of‑two that works well with FFTs
        n = 16384
        init_seq = None

    target_sum = math.sqrt(2.0 * n)               # ℓ₁‑normalisation

    # --------------------------------------------------------------
    # Phase 0 – parametric sweep over simple shapes
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)         # uniform start
    best_score = compute_max_conv(best_seq)

    shape_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),          # step 0.2
        "quadratic": np.linspace(0.0, 6.0, 31),      # step 0.2
        "exponential": np.linspace(0.0, 6.0, 31),    # step 0.2
        "power": np.linspace(0.5, 5.0, 11),          # exponent
        "inverse": np.linspace(0.5, 5.0, 11),
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_u_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during parametric sweep.")
                return best_seq.tolist()

    # Beta‑distribution shapes (a modest grid to keep time low)
    alphas = np.linspace(0.5, 5.0, 7)
    betas = np.linspace(0.5, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            cand = generate_beta_shape(n, target_sum, a_, b_)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best beta (α={a_:.2f}, β={b_:.2f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during beta sweep.")
                return best_seq.tolist()

    # consider the supplied baseline if it exists
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves initial score → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam sub‑gradient optimisation
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Entering Adam optimisation (best score {best_score:.6f})")
    # allocate roughly 70 % of the remaining budget to Adam
    adam_deadline = time.time() + 0.68 * (deadline - time.time())
    best_seq, best_score = adam_optimize(best_seq, target_sum, adam_deadline)

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing polishing
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting annealing phase (score {best_score:.6f})")
    best_seq, best_score = annealing_refine(best_seq, target_sum, best_score, deadline)

    print(f"[{time.time():.2f}] Returning best sequence with score {best_score:.6f}")
    return best_seq.tolist()
