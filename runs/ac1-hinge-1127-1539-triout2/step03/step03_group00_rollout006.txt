        # Keep global best

generates many random candidates on the ℓ¹‑simplex, refines each

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation: the maximum entry of a*a.
    `a` is assumed to already satisfy Σa = √(2·len(a)).
    """
    return _max_convolution_fft(a)

    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }  (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialiser (a few simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex of length `n`.
    Several families (power‑law, exponential, mixed, Dirichlet) are
    sampled uniformly at random.
    """
    target_sum = np.sqrt(2.0 * n)
    choice = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if choice == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif choice == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                   # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    else:                               # almost uniform, tiny jitter
        a = np.full(n, target_sum / n)

    # a small Gaussian jitter and a final projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Interpolation (to upscale a known good sequence)
# ----------------------------------------------------------------------


def _interpolate_sequence(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a non‑negative sequence to a new length.
    The result is projected onto the appropriate simplex afterwards.
    """
    old_n = old_seq.size
    if old_n == new_n:
        return old_seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, old_seq)
    target_sum = np.sqrt(2.0 * new_n)
    new_seq = _project_to_simplex(new_seq, target_sum)
    return new_seq


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact ℓ∞ convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent for `max(a*a)` with projection onto the
    ℓ¹‑simplex.  The step‑size decays like 1/√iteration.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)                # a is already normalised
    it = 0
    # Pre‑compute FFT size (used repeatedly)
    size_fft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and it < 5000:
        # current convolution and the maximal index/indices
        fa = np.fft.rfft(a, n=size_fft)
        b = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_val = b.max()
        ks = np.flatnonzero(b == max_val)        # all indices attaining the max

        # gradient of the max‑convolution (sub‑gradient)
        g = np.zeros_like(a)
        for k in ks:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            i_idx = np.arange(start, end + 1)
            j_idx = k - i_idx
            # each pair (i,j) contributes a[j] (once) and, when i≠j, another a[j]
            mask = i_idx != j_idx
            g[i_idx] += a[j_idx] * (1.0 + mask.astype(float))

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break

        # step‑size schedule (α / (‖g‖·√(it+1)))
        alpha = target_sum * 0.2
        lr = alpha / (norm_g * (np.sqrt(it + 1) + 1.0))

        a_next = a - lr * g
        a_next = np.maximum(a_next, 0.0)
        a_next = _project_to_simplex(a_next, target_sum)

        # evaluate improvement
        fa_next = np.fft.rfft(a_next, n=size_fft)
        b_next = np.fft.irfft(fa_next * fa_next, n=size_fft)[: 2 * n - 1]
        new_max = b_next.max()

        if new_max < max_val - 1e-12:
            a = a_next
        else:
            # if no progress, shrink step a little and continue
            if lr * norm_g < 1e-12:
                break
        it += 1

    return a


# ----------------------------------------------------------------------
# Very lightweight simulated‑annealing (local mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly moves a random fraction of mass between two
    indices and accepts uphill moves with a cooling schedule.
    """
    a = a0.copy()
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-6      # initial temperature
    cooling = 0.9996
    rng = np.random.default_rng()

    n = a.size
    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction of the mass at i to j
        frac = rng.random() * 0.15          # ≤15 % of a[i]
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to shake out of deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    competition objective.  The routine respects a wall‑clock budget,
    explores a few different lengths, samples many random vectors,
    refines each via sub‑gradient descent and a short SA pass, and
    finally returns the best sequence found as a Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is available, start from it
    # ------------------------------------------------------------------
    init = globals().get("height_sequence_1")
    if isinstance(init, (list, tuple, np.ndarray)):
        a0 = np.asarray(init, dtype=float)
        n0 = a0.size
        target0 = np.sqrt(2.0 * n0)
        a0 = _project_to_simplex(a0, target0)
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from global height_sequence_1 "
              f"(n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to experiment with (larger n ⇒ more freedom, still fast)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Phase 1 – massive random sampling on the simplex
        # --------------------------------------------------------------
        phase_end = min(deadline, time.time() + 15.0)   # ~15 s per length
        while time.time() < phase_end:
            a_rand = _random_initial_sequence(n, rng)
            sc = _score(a_rand)
            if sc < best_score:
                best_score = sc
                best_seq = a_rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) "
                      f"n={n} score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – take the current best of this length (or interpolated)
        # and apply a deterministic sub‑gradient refinement
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_start = _interpolate_sequence(best_seq, n)
        else:
            a_start = _random_initial_sequence(n, rng)

        # give each refinement a modest time slice
        refine_dead = min(deadline, time.time() + 12.0)
        a_refined = _subgradient_descent_max(a_start, refine_dead)
        sc_ref = _score(a_refined)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_refined.copy()
            print(f"[{time.time():.1f}] New best after sub‑grad "
                  f"n={n} score={sc_ref:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – a brief simulated‑annealing polish (uses whatever time left)
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_sa = _simulated_annealing(best_seq, deadline)
            sc_sa = _score(a_sa)
            if sc_sa < best_score:
                best_score = sc_sa
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST after SA "
                      f"n={n} score={sc_sa:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
