"""
Propose a step‑function sequence that minimises the max entry of its
self‑convolution (the evaluation score).  

Algorithm
---------
1. **Binary‑mass heuristic** –  for many random problem sizes `n`
   (800 … 5000) pick a random number `m` of non‑zero positions,
   assign each of them the same height so that the ℓ¹‑norm equals
   √(2 n).  The convolution of a sparse binary pattern is usually very
   flat, giving a low maximal entry.  The best pattern found so far is
   kept.

2. **Local refinement** –  starting from the best binary pattern,
   perform a projected smooth‑max gradient descent (with Adam‑style
   adaptivity) that directly reduces the maximal convolution entry.
   Random “mass‑swap’’ moves help escape plateaus.

3. The whole procedure respects the 1000 s time budget, prints
   progress, and finally returns the best list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (real input)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT (real‑valued)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(α·conv_i) / Σ_j exp(α·conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term g_j = 2·Σ_i w_{i+j}·a_i .
    Both a (len n) and w (len 2n‑1) are real vectors.
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # pick the right slice
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# One projected (Adam‑style) gradient step on the smooth max surrogate
# ----------------------------------------------------------------------
def _adam_step(a: np.ndarray,
               m1: np.ndarray,
               v1: np.ndarray,
               t: int,
               lr: float,
               alpha: float,
               beta1: float,
               beta2: float,
               eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform an Adam update for the smooth‑max objective,
    then project back onto the simplex Σa = √(2·n).
    Returns the updated vector and new first/second moment estimates.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)

    # Adam moments
    m1 = beta1 * m1 + (1.0 - beta1) * grad
    v1 = beta2 * v1 + (1.0 - beta2) * (grad * grad)

    m_hat = m1 / (1.0 - beta1 ** t)
    v_hat = v1 / (1.0 - beta2 ** t)

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

    # project onto the simplex Σa = √(2·n)
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)
    return a_new, m1, v1


# ----------------------------------------------------------------------
# Local optimiser (smooth‑max + occasional random mass‑swap)
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate `a0` using Adam + random swaps.
    Returns the best vector seen and its true maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m1 = np.zeros_like(a)
    v1 = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                      # initial learning rate
    lr_decay = 0.9995
    alpha = 1.0                   # temperature for soft‑max
    alpha_mul = 1.001

    # keep the best
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        a, m1, v1 = _adam_step(a, m1, v1, it, lr, alpha,
                               beta1, beta2, eps)

        cur_max = np.max(_full_convolution(a))

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # ---- random mass‑swap (simulated‑annealing flavour) ----
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()   # move up to 4 % of a[i]
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis acceptance
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.05, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # schedule updates
        lr *= lr_decay
        alpha *= alpha_mul

        # early stop if no improvement for a while
        if it - last_improve > 3000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Helpers for the binary‑mass heuristic
# ----------------------------------------------------------------------
def _binary_pattern(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a non‑negative vector of length n with exactly m non‑zero
    entries, all equal, and ℓ¹‑norm √(2·n).
    """
    S = np.sqrt(2.0 * n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = S / m
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low max‑self‑convolution value.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    overall_deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – massive random binary‑mass exploration
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + 0.75 * (budget_s - 5.0)

    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < phase1_deadline:
        attempts += 1
        n = rng.integers(800, 5000)          # problem size
        # fraction of non‑zeros – explore a wide range log‑uniformly
        frac = rng.uniform(0.08, 0.8)
        m = max(2, int(frac * n))
        if m > n:
            m = n
        a = _binary_pattern(n, m, rng)

        # max convolution (true objective for this scale)
        cur_max = np.max(_full_convolution(a))

        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[binary] attempt={attempts} n={n} m={m} "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – local continuous refinement of the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        # use the best binary solution as a seed and run a few Adam‑descent
        # runs with jitter.  All runs share the remaining time budget.
        while time.time() < overall_deadline:
            # small jitter around the current best (helps escape stagnation)
            a0 = best_seq.copy()
            jitter = rng.normal(scale=0.001 * a0.mean(), size=a0.size)
            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * a0.size))

            a_opt, score = _optimize_local(a0, overall_deadline, rng)

            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"[refine] new best score={best_score:.6f} "
                      f"len={len(best_seq)}")

    # ------------------------------------------------------------------
    # Final sanity check (black‑box evaluator) and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform tiny sequence
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score={external_score:.6f} "
          f"internal max={best_score:.6f} length={len(final_seq)}")
    return final_seq
