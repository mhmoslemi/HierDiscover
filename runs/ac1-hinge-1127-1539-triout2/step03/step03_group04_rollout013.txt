        # compute sensitivity

        hi = min(k_max, n - 1)
        sens[lo:hi+1] = a[k_max - np.arange(lo, hi+1)]
        # donor weight
        weight = a * sens

        else:

            continue

"""
fast_sa_optimizer.py

A two‑stage optimizer for the “max‑convolution / ℓ¹‑norm’’ benchmark.

Stage 1 – start from the best known step function (the global variable
`height_sequence_1` if present, otherwise a simple power‑law).  The vector
is normalised to the simplex Σa = √(2·n).

Stage 2 – Simulated‑annealing refinement.  In each iteration we
* compute the full convolution a*a by FFT (O(n log n));
* locate the index k* where the convolution attains its maximum;
* weight the coordinates by their contribution a_i·a_{k*‑i};
* either move a random fraction of mass from a high‑weight donor to a
  low‑weight receiver, or replace a short window by its average
  (local smoothing).  Moves are accepted if they improve the objective,
  otherwise with Metropolis probability exp(‑Δ/τ) where the temperature
  τ is exponentially cooled.
The algorithm respects non‑negativity and the ℓ¹‑norm constraint, and
continues until the time budget expires.  The best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact evaluation of the competition objective for a normalised sum."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """Projected gradient descent on the soft‑max approximation of the max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 256.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD   beta={beta:.2f}  score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _guided_hill_climb(a: np.ndarray, deadline: float):
    """Deterministic hill‑climbing that shifts mass from high‑sensitivity to low‑sensitivity places."""
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07

    while time.time() < deadline:
        conv = np.convolve(a, a)          # O(n) for moderate n
        k_max = int(np.argmax(conv))

        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens[lo:hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        weight = sens * a
        if weight.max() == 0:
            break
        donor = int(np.argmax(weight))

        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH   improved = {best_val:.6f}")
        else:
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 20) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves:
      * donor→receiver mass transfer (most common);
      * local smoothing (replace a short window by its average).
    Acceptance follows a Metropolis rule with exponentially cooling temperature.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = 2.0 * n * _max_convolution_fft(a) / (cur_sum ** 2)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial convolution (we will keep it for the first iteration)
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # progress‑dependent parameters
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        # exponential temperature schedule
        temp = start_temp * (end_temp / start_temp) ** tau
        # fractional move size shrinking with temperature
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # ---- data from current state ----
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity for the active convolution entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # ------ propose a move ------
        if rng.random() < 0.70:          # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_idx)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            # apply move
            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                            # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # ---- evaluate new state (single FFT) ----
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new / (cur_sum ** 2)

        # ---- Metropolis acceptance ----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:  # smooth
                a[saved_start:saved_end] = saved_seg
            # conv stays unchanged, cur_score unchanged

        # occasional jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            # re‑normalise to preserve the ℓ¹‑norm
            a = a * (cur_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = 2.0 * n * conv.max() / (cur_sum ** 2)

        it += 1

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Starts from the best known sequence (or a power‑law), then refines it with
    simulated annealing (and a brief gradient‑descent pass if time permits).
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # initialise from the globally supplied best construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n0 = seq.size
        target = np.sqrt(2.0 * n0)
        if seq.sum() > 0:
            seq = seq * (target / seq.sum())
        else:
            seq = _power_law_sequence(n0, 0.50, target)
        seq = _project_to_simplex(seq, target)
        best_seq = seq.copy()
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] start from global seq, score={best_score:.6f}")
    else:
        # fallback – a simple power‑law of moderate length
        n0 = 1500
        target = np.sqrt(2.0 * n0)
        best_seq = _power_law_sequence(n0, 0.55, target)
        best_seq = _project_to_simplex(best_seq, target)
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] start from power‑law, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optional brief annealed gradient descent (if we have a healthy margin)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + min(20.0, remaining / 2.0)
        gd_seq, gd_score = _annealed_gradient_descent(best_seq, sub_deadline)
        if gd_score < best_score:
            best_seq, best_score = gd_seq, gd_score
            print(f"[{time.time():.1f}] after GD, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # simulated annealing refinement (the core of the improvement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_seq, sa_score = _simulated_annealing(best_seq, deadline)
        if sa_score < best_score:
            best_seq, best_score = sa_seq, sa_score
            print(f"[{time.time():.1f}] after SA, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # final housekeeping – trim trailing zeros
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
