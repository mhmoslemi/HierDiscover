Define projection:

def simplex_proj(v, s):
    # v is 1D array
    # ensure nonnegative sum s
    # Implementation from Duchi et al. 2008.

    n = v.shape[0]
    # sort descending:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

def proj_simplex(v, s):
    # v: 1d np array
    # similar to Duchi et al.
    n = v.shape[0]
    if s < 0:
        raise ValueError("s must be non-negative")
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

   initialisations with random concentration, and varied hyper‑parameters
   (learning‑rate, τ‑growth) are tried until the time budget expires.
   The best vector seen so far (its max‑convolution equals the external
   score because the ℓ₁‑norm is kept at √(2·n)) is kept globally.

3. **Local stochastic hill‑climbing** (simulated‑annealing style) to
   polish the best vector.  A small fraction of mass is moved from a
   donor index to a recipient; moves that improve the maximum convolution
   are always accepted, worsening moves are accepted with a temperature‑
   dependent probability.  The temperature is geometrically cooled.

All arithmetic is performed with NumPy and FFTs (real‑valued) for
`O(n log n)` speed.  The routine respects the 1000 s budget, prints
progress information, and finally returns the best list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i >= 0, sum_i x_i = s}
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex size 's' must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    # Find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) for non‑negative lags.
    Returns the full correlation array of length len(w)+len(a)-1.
    """
    Fw = np.fft.rfft(w, n=fft_len)
    Fa = np.fft.rfft(a, n=fft_len)
    corr_full = np.fft.irfft(Fw * np.conj(Fa), n=fft_len)
    total_len = w.shape[0] + a.shape[0] - 1
    return corr_full[:total_len]


# ----------------------------------------------------------------------
# Stage 1 – log‑sum‑exp gradient descent
# ----------------------------------------------------------------------


def _run_logsumexp_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    init_lr: float = 0.7,
    lr_decay: float = 0.9999,
    init_tau: float = 0.1,
    tau_factor: float = 1.001,
    tau_max: float = 8.0,
    jitter_prob: float = 0.015,
    max_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate of max(a*a).
    Returns the best vector encountered and its max‑convolution value.
    """
    S = np.sqrt(2.0 * n)                 # required ℓ₁‑norm
    a = a0.copy()
    best_a = a.copy()
    best_val = np.max(_full_convolution(a))  # external score = max(conv)

    # Pre‑computed FFT length for correlation (size 3n‑2)
    corr_len = 3 * n - 2
    fft_len_corr = 1 << ((corr_len - 1).bit_length())

    lr = init_lr
    tau = init_tau
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- compute self‑convolution and soft‑max weights ----
        conv = _full_convolution(a)
        mx = np.max(conv)

        # soft‑max (stable)
        w = np.exp(tau * (conv - mx))
        w_sum = np.sum(w)
        if w_sum == 0.0:          # numerical safety
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        # correlation w ⋆ a gives Σ_k w_k * a_{k-i}
        corr = _correlation(w, a, fft_len_corr)
        offset = n - 1                     # lag 0 corresponds to this index
        grad = 2.0 * corr[offset : offset + n]   # 2·Σ w_k a_{k-i}

        # ---- projected gradient step ----
        a_new = a - lr * grad
        a = _project_onto_simplex(a_new, S)

        # ---- occasional jitter to escape plateaus ----
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.015
            a = _project_onto_simplex(a + noise, S)

        # ---- bookkeeping ----
        cur_val = np.max(_full_convolution(a))
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[descent] time={time.time():.1f}s n={n} iter={it} "
                  f"max={cur_val:.6f} lr={lr:.5f} tau={tau:.4f}")

        # decay / increase schedules
        lr *= lr_decay
        tau = min(tau_max, tau * tau_factor)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – local stochastic hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def _local_hill_climb(
    a_start: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style hill‑climbing around ``a_start``.
    Returns the best vector encountered and its max‑convolution.
    """
    a = a_start.copy()
    cur_val = np.max(_full_convolution(a))
    best_a = a.copy()
    best_val = cur_val

    temperature = max(0.05 * cur_val, 1e-4)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        frac = rng.uniform(0.07, 0.22)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = np.max(_full_convolution(a_new))

        if new_val < cur_val - 1e-12:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a_new.copy()
                print(f"[hill] time={time.time():.1f}s n={n} iter={it} max={new_val:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_val - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                a, cur_val = a_new, new_val

        temperature *= 0.9995

        # occasional noise to diversify
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.012, size=n)
            a = _project_onto_simplex(np.clip(a + noise, 0.0, None), np.sqrt(2.0 * n))
            cur_val = np.max(_full_convolution(a))
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[jitter] time={time.time():.1f}s n={n} iter={it} max={cur_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low evaluation
    score within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # if a previous construction exists, record it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {sc:.6f} "
                      f"length = {cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # search parameters
    # ------------------------------------------------------------------
    lengths = [1000, 2000, 4000, 8000]          # lengths we will try
    run_id = 0

    while time.time() < deadline:
        # pick a length, slightly favour larger sizes
        probs = np.array([0.15, 0.25, 0.35, 0.25])
        n = int(rng.choice(lengths, p=probs / probs.sum()))

        # initialise a starting vector
        if best_seq is not None and best_seq.size == n:
            a0 = best_seq.copy()
        else:
            conc = rng.uniform(0.05, 1.2)                    # Dirichlet concentration
            a0 = rng.dirichlet(np.full(n, conc))
            a0 = a0 * np.sqrt(2.0 * n)                       # enforce ℓ₁‑norm

        # ---------- phase 1 : log‑sum‑exp descent ----------
        a_opt, max_conv = _run_logsumexp_descent(
            a0,
            n,
            deadline,
            init_lr=rng.uniform(0.4, 0.9),
            lr_decay=0.99993,
            init_tau=rng.uniform(0.07, 0.15),
            tau_factor=rng.uniform(1.0008, 1.0025),
            tau_max=12.0,
            jitter_prob=0.018,
            max_iters=30000,
        )
        score = max_conv  # because ℓ₁‑norm = √(2·n)  →  external score equals max_conv

        # ---------- phase 2 : local hill‑climbing ----------
        a_opt2, max_conv2 = _local_hill_climb(
            a_opt,
            n,
            deadline,
            rng,
            max_iters=20000,
        )
        if max_conv2 < score:
            a_opt, score = a_opt2, max_conv2

        # ---------- record global improvement ----------
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final verification (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
