def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2*n-1]
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    # gradient: g_i = 2 * sum_j a_j * p_{i+j} (or similar). The earlier code used conv_rev = a_rev ⋆ p, then take central part.
    # They used a_rev = a[::-1] and convolve with p.
    a_rev = a[::-1]
    conv_len = n + (2*n - 1) - 1  # 3n-2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n-1 : n-1 + n]
    return g, b

def adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    step = 0
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 512.0
    # step_base adapts with beta; we could compute lr = target_sum*0.2/(beta+1)
    # We'll use constant learning rate factor to m_hat/v_hat: lr0 = target_sum * 0.2
    lr0 = target_sum * 0.2

    while time.time() < deadline:
        # compute gradient of smoothed max
        g, _ = _gradient_softmax(a, beta)

        step += 1
        # update moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        # adjust step size based on beta (higher beta -> smaller steps)
        # scaling factor
        scale = lr0 / (beta + 1.0)

        a = a - scale * m_hat / (np.sqrt(v_hat) + epsilon)

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # evaluate occasionally
        if step % 100 == 0:
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam step {step} beta={beta:.2f} best={best_val:.6f}")

        # anneal beta
        if step % 250 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

    return best_a, best_val

"""
Propose a step‑function that (approximately) minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm combines three ideas:

1. **Diverse random starts** – power‑law shapes, Dirichlet‑draws and the
   previously discovered construction (if ``height_sequence_1`` exists).

2. **Adam‑based projected optimisation** of a smooth approximation of the
   max‑convolution (soft‑max with an annealed temperature β).  After each
   gradient step the vector is projected onto the simplex
   {x≥0, Σx = √(2 n)} so the evaluation function reduces to the maximal
   convolution entry.

3. **Guided stochastic hill‑climbing** – after the smooth optimisation the
   current peak index k* is identified and mass is shifted from indices that
   contribute heavily to that peak to indices with little contribution.
   Moves are accepted either if they improve the score or with a small
   Metropolis probability (simulated‑annealing style) to escape shallow
   local minima.

All phases are time‑boxed; the best sequence found before the global budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the linear convolution a*a using an FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n‑1) using an FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Fast version of the competition objective.
    (When Σa = √(2 n) the score equals max(conv).)
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:          # should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max of the convolution a*a.
    Returns (g, b) where b = a*a (full convolution) and
    g = ∇_a log‑sum‑exp_{β}(b).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # gradient: 2·(a_rev ⋆ p) restricted to the original length
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]       # n‑dimensional gradient
    return g, b


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max approximation.
    β is annealed during optimisation.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step = 0

    # schedule for the smoothness parameter β
    beta = 1.0
    max_beta = 512.0

    # learning‑rate base (scaled with target sum)
    lr0 = target * 0.20

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the soft‑max objective
        g, _ = _gradient_softmax(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # step size shrinks as β grows
        scale = lr0 / (beta + 1.0)

        a = a - scale * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target)

        # occasional jitter to escape flat regions
        if step % 500 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # keep the best seen so far
        if step % 100 == 0:
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam step {step:5d} β={beta:.2f} best={best_val:.6f}")

        # anneal β
        if step % 250 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided stochastic hill‑climbing (simulated‑annealing flavour)
# ----------------------------------------------------------------------


def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Move mass from indices that heavily contribute to the current peak
    to indices that contribute little.  Accepts uphill moves with a small
    Metropolis probability to avoid getting stuck.
    """
    n = a.size
    target = np.sqrt(2.0 * n)

    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    temperature = 0.12          # initial acceptance temperature

    while time.time() < deadline:
        # full convolution and location of the maximal entry
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # contribution of each coordinate to conv[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k_max - idx
        weight = np.zeros(n, dtype=np.float64)
        weight[idx] = a[idx] * a[partner]

        # donor = index with largest contribution, receiver = a low‑contribution index
        donor = int(weight.argmax())
        low_thr = np.percentile(weight, 10)
        candidates = np.where((weight <= low_thr) & (np.arange(n) != donor))[0]
        if candidates.size == 0:
            # fallback: pick a random different index
            receiver = int(rng.integers(0, n))
            if receiver == donor:
                continue
        else:
            receiver = int(rng.choice(candidates))

        if a[donor] < 1e-12:
            continue

        # move a modest fraction of the donor's mass
        frac = rng.random() * 0.05                     # up to 5 %
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temperature, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] Hill‑climb improved = {best_val:.6f}")
        else:
            # revert the move
            a[receiver] -= delta
            a[donor] += delta

        # slowly cool down
        temperature = max(temperature * 0.9999, 1e-6)

    # final projection (tiny drift may appear)
    best_a = _project_to_simplex(best_a, target)
    return best_a, best_val

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) and normalise to the prescribed L1 norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, alpha: float, target_sum: float, rng) -> np.ndarray:
    """Draw from a Dirichlet(α,…,α) and scale to the target L1 norm."""
    g = rng.gamma(alpha, 1.0, size=n)
    a = g * target_sum / g.sum()
    return a
