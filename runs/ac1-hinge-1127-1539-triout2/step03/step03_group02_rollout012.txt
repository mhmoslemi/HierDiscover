"""
Propose a non‑negative step‑function (a list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  The objective, after the
normalisation Σa = √(2 n), is simply   max(conv(a, a)).
The algorithm consists of three stages:

1️⃣  A quick multi‑initialisation and a sub‑gradient descent with a
    Polyak step size (target lower bound 1.0).  This quickly pushes
    the maximum convolution down from the uniform value 2.

2️⃣  An alternating‑projection (AP) phase that tries to flatten the
    DFT magnitude of the zero‑padded sequence.  Constant spectral
    magnitude yields (approximately) a flat autocorrelation, i.e.
    a low max‑convolution.

3️⃣  A short polishing stage that performs random pair‑mass transfers
    (a simple stochastic hill‑climber) to catch any remaining tiny
    improvements.

All operations respect the simplex constraint
    {x ≥ 0, Σx = √(2 n)}  via Euclidean projection.
The code respects a 1000 s time budget; when time runs out the best
sequence found so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow_of_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of a (length n) using an FFT of power‑of‑two size.
    Returns an array of length 2*n‑1.
    """
    n = a.shape[0]
    L = next_pow_of_two(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return float(conv_self_fft(a).max())


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small amount of mass from a random index i to another index j,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    param controls the shape (exponent for power/inverse).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid  # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


# ----------------------------------------------------------------------
# Stage 1 – sub‑gradient descent with a Polyak step size
# ----------------------------------------------------------------------


def subgradient_descent(a: np.ndarray,
                        target_sum: float,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Perform sub‑gradient iterations on the true max‑convolution.
    Returns the best sequence and its score found during the loop.
    """
    n = a.shape[0]
    iter_cnt = 0
    while time.time() < deadline:
        # --- convolution and maximal value ---------------------------------
        conv_full = conv_self_fft(a)
        max_conv = conv_full.max()

        # keep a copy if we improved
        if max_conv < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it {iter_cnt:06d} → {best_score:.6f}")

        # --- construct sub‑gradient (average over all maximal indices) ---
        eps = max(1e-12, 1e-9 * max_conv)
        mask = (conv_full >= max_conv - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv_full.argmax()] = 1.0
            cnt = 1.0

        a_rev = a[::-1]
        L2 = next_pow_of_two(len(mask) + n - 1)
        M = np.fft.rfft(mask, n=L2)
        A_rev = np.fft.rfft(a_rev, n=L2)
        conv_full_grad = np.fft.irfft(M * A_rev, n=L2)[: len(mask) + n - 1]
        grad = (2.0 / cnt) * conv_full_grad[n - 1 : n - 1 + n]

        # --- Polyak step (target lower bound = 1.0) -----------------------
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq == 0:
            break

        step = (max_conv - 1.0) / grad_norm_sq
        # safety caps
        max_step = 0.3 * target_sum / n
        if step > max_step:
            step = max_step
        if step < 1e-12:
            step = 1e-12

        # --- update --------------------------------------------------------
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        iter_cnt += 1

        # occasional random shake to escape plateaus
        if iter_cnt % 800 == 0 and np.random.rand() < 0.4:
            a = random_pair_transfer(a, target_sum, scale=0.02)

        # early exit if we are already very close to the theoretical lower bound
        if best_score <= 1.504 and time.time() > deadline - 30:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – alternating projection towards a flat spectrum
# ----------------------------------------------------------------------


def alternating_projection(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly force the DFT magnitude of the zero‑padded sequence to be constant
    and project back onto the simplex.  This tends to flatten the autocorrelation.
    """
    n = a.shape[0]
    L = next_pow_of_two(2 * n - 1)          # length used for convolution elsewhere
    it = 0
    while time.time() < deadline:
        # pad to length L and take FFT
        a_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # constant magnitude that preserves total energy
        energy = np.sum(a ** 2)               # ∑a_i²
        const_mag = np.sqrt(L * energy / (len(A)))  # keep ∑|A_k|² unchanged

        A_new = const_mag * np.exp(1j * np.angle(A))

        # back to time domain
        a_full = np.fft.irfft(A_new, n=L)
        a_new = a_full[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)

        # evaluate
        cur_max = compute_max_conv(a_new)
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] AP iter {it:04d} → {best_score:.6f}")

        a = a_new
        it += 1

        # occasional jitter to avoid getting trapped in a limit cycle
        if it % 15 == 0 and np.random.rand() < 0.5:
            a = random_pair_transfer(a, target_sum, scale=0.005)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – final polishing with random pair transfers
# ----------------------------------------------------------------------


def polishing(a: np.ndarray,
              target_sum: float,
              deadline: float,
              best_seq: np.ndarray,
              best_score: float) -> tuple[np.ndarray, float]:
    """
    A tiny stochastic hill‑climber: random pair transfers are accepted only
    if they improve the maximal convolution.
    """
    while time.time() < deadline:
        cand = random_pair_transfer(a.copy(), target_sum, scale=0.003)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] Polish → {best_score:.6f}")
    return best_seq, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------

    candidates = []

                                      deadline,
                                      best_seq,
                                      best_score)

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence with max‑conv {best_score:.6f}")
    return best_seq.tolist()
