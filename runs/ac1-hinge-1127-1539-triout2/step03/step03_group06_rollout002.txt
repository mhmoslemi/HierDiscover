"""
Smooth‑max optimiser for the step‑function problem.

The external score returned by ``evaluate_sequence`` equals the maximum
entry of the self‑convolution when the vector is normalised to ℓ₁‑norm
√(2 n).  This implementation minimises a *smooth* approximation of that
maximum:  

    L(a) = log Σ_k exp(α·(a*a)_k) .

For α→∞, L → max(a*a).  The gradient of L can be obtained analytically and
computed with three fast FFT convolutions per iteration.  An exponentiated‑
gradient update with momentum keeps the vector non‑negative; after each
update the ℓ₁‑norm is re‑scaled to √(2 n).

The optimiser is run many times with random restarts and a few different
lengths.  When a promising candidate appears we optionally polish it with a
small linear‑programming refinement (if SciPy is available).  The routine
stops before the 1000 s wall‑clock budget expires and returns the best
sequence found.
"""

import time
import math
import numpy as np

# Optional LP refinement – only used when SciPy is present and the problem
# size is modest.
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _fft_convolve(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full (linear) convolution of two real vectors via FFT.
    Returns a vector of length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv_full = np.fft.irfft(X * Y, n=fft_len)
    return conv_full[: len(x) + len(y) - 1]


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute the full self‑convolution of ``a`` and return (conv,
    max(conv)).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv, float(conv.max())


def _smooth_max_gradient(a: np.ndarray, alpha: float,
                         fft_len: int) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of L(a)=log Σ exp(α·(a*a)) together with the
    current smooth‑max value (i.e. log‑sum‑exp itself).  Returned gradient
    has the same shape as ``a``.
    """
    n = a.shape[0]

    # ----- self‑convolution (a*a) and soft‑max weights -----
    Fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
    max_val = conv_full.max()
    # shift to avoid overflow, then exponentiate
    shift = conv_full - max_val                 # ≤ 0
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                               # soft‑max probabilities

    # ----- gradient = 2·α·(conv_a_w + conv_a_w_rev) -----
    # convolution a * w
    conv_a_w = _fft_convolve(a, w, fft_len)          # length 3n‑2
    term2 = conv_a_w[:n]

    # convolution a * w_rev
    w_rev = w[::-1]
    conv_a_w_rev = _fft_convolve(a, w_rev, fft_len)
    # indices 2n‑2‑j give Σ_i a_i·w_{i+j}
    term1 = conv_a_w_rev[2 * n - 2 - np.arange(n)]

    grad = 2.0 * alpha * (term1 + term2)

    # smooth‑max (log‑sum‑exp) value – it equals max_val + log Σ exp(α·(conv‑max))
    smooth_val = max_val + math.log(w_sum) / alpha   # not used for optimisation

    return grad, max_val


def _smooth_max_optimize(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2500,
                         eta0: float = 0.25,
                         beta: float = 0.9,
                         decay: float = 0.9999,
                         jitter_prob: float = 0.015,
                         alpha0: float = 2.0,
                         alpha_factor: float = 1.002,
                         alpha_max: float = 16.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth‑max surrogate.
    Returns the best vector seen and the corresponding (unsmoothed) max
    convolution value.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)          # target ℓ₁‑norm
    best_seq = a.copy()
    best_val = float('inf')

    # momentum buffer for the gradient
    m = np.zeros_like(a)

    # pre‑compute FFT length used for all convolutions of this run
    fft_len = 1 << ((3 * n - 2) - 1).bit_length()

    eta = eta0
    alpha = alpha0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # gradient of the smooth‑max and the *true* max‑convolution entry
        grad, cur_max = _smooth_max_gradient(a, alpha, fft_len)

        # record improvement (cur_max is the external score for this norm)
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_seq = a.copy()

        # momentum update
        m = beta * m + (1.0 - beta) * grad

        # exponentiated (mirror) step – keep positivity
        delta = -eta * m
        delta = np.clip(delta, -30.0, 30.0)      # avoid overflow
        a = a * np.exp(delta)

        # renormalise to ℓ₁‑norm √(2 n)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to escape flat plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # schedule updates
        eta *= decay
        alpha = min(alpha_max, alpha * alpha_factor)

        # early exit if we are about to exceed the deadline
        if time.time() >= deadline - 0.1:
            break

    return best_seq, best_val


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix H (size (2n‑1)×n) such that H@g = f*g.
    Used only by the optional LP refinement for moderate n (≤ 3000).
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i : i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float,
               steps: int = 4) -> np.ndarray:
    """
    Optional linear‑programming polishing step (identical to the one used
    in the reference implementation).  It solves:

        maximise   Σ g_i
        subject to 0 ≤ g_i  and  (f*g)_k ≤ M ∀k

    where M = max (f*f).  The resulting vector g is mixed back into f.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:            # keep LP tractable
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:                 # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # mix and renormalise
        f = (1.0 - 0.2) * f + 0.2 * g
        f = f / f.sum() * S

    return f


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point for the challenge.  Searches for a non‑negative
    step‑function that minimises ``evaluate_sequence``.  The routine obeys
    the 1000 s time budget, prints progress, and returns the best sequence
    (as a plain list of floats) it discovers.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1)  Try to use any existing construction supplied in the environment.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                n0 = cand.size
                S0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * S0
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] height_sequence_1 score = {best_score:.6f}, len = {n0}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 2)  Random‑restart smooth‑max optimisation.
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000, 12000, 16000]
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ----------------------------------------------------------------
        # Initialise a starting vector.
        # ----------------------------------------------------------------
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.35:
            a0 = best_seq[:n].copy()
        else:
            a0 = rng.random(n)

        # Normalise to ℓ₁‑norm √(2 n)
        S = math.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # ----------------------------------------------------------------
        # Run the smooth‑max optimiser.
        # ----------------------------------------------------------------
        seq, cur_max = _smooth_max_optimize(
            a0, n, deadline,
            max_iters=3000,
            eta0=0.3,
            beta=0.85,
            decay=0.9998,
            jitter_prob=0.02,
            alpha0=2.0,
            alpha_factor=1.003,
            alpha_max=14.0)

        if cur_max < best_score:
            best_score = cur_max
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        # ----------------------------------------------------------------
        # Optional LP polishing when a noteworthy improvement occurs.
        # ----------------------------------------------------------------
        if cur_max < best_score * 0.98:
            refined = _lp_refine(best_seq, deadline)
            _, refined_max, _ = _conv_max(refined)
            if refined_max < best_score:
                best_score = refined_max
                best_seq = refined.copy()
                print(f"[LP refine] run={run_id} n={n} refined score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 3)  Final polish of the best candidate (if any).
    # ------------------------------------------------------------------
    if best_seq is not None:
        # a few extra smooth‑max iterations with tighter parameters
        best_seq, best_score = _smooth_max_optimize(
            best_seq,
            best_seq.size,
            deadline,
            max_iters=6000,
            eta0=0.2,
            beta=0.9,
            decay=0.99995,
            jitter_prob=0.01,
            alpha0=4.0,
            alpha_factor=1.002,
            alpha_max=18.0)

        # final LP pass (if affordable)
        best_seq = _lp_refine(best_seq, deadline)

        # final exact evaluation
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    else:
        # Fallback – uniform distribution (should never happen).
        n = 1000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    return best_seq.tolist()
