```
def random_simplex_vector(n, target_sum):
    x = np.random.rand(n) # uniform (0,1)

    x = np.random.exponential(1.0, size=n)
    x_sum = x.sum()
    return x * (target_sum / x_sum)
```

The evaluation metric is  

\[
F(a)=\frac{2\,n\;\max_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `*` denotes the linear self‑convolution.  
The function is *scale‑invariant*: multiplying the whole vector by a constant does not change the
score, therefore we may (and do) keep the ℓ₁‑norm fixed to  

\[
\sum_i a_i = \sqrt{2n}.
\]

The optimisation problem is non‑convex and the gradients of the true max‑operator are
discontinuous.  The strategy is:

1. **Parametric warm‑start** – sweep a few one‑parameter families (exponential,
   linear and quadratic decays).  The best of those vectors seeds the subsequent
   optimisation.

2. **Smoothed Adam optimisation** – replace the hard max by a log‑sum‑exp
   smooth approximation with temperature `τ`.  The gradient of the smoothed
   objective can be computed with two FFT‑convolutions, hence each iteration costs
   `O(n log n)`.  Adam (adaptive moments) together with a projection onto the
   ℓ₁‑simplex gives fast, stable progress.  The temperature is decreased
   gradually so that the algorithm finally focuses on the true maximum.

3. **Sub‑gradient polishing** – once τ is very small we switch to the exact
   sub‑gradient of the max‑convolution (only the indices that realise the
   maximum receive a non‑zero gradient).  A short run of pure sub‑gradient steps
   cleans up the remaining non‑smoothness.

4. **Random jitter & pair transfers** – occasional random mass transfers
   (`random_pair_transfer`) and jitter help to escape plateaus.

All steps respect the 1000 s budget; the algorithm always returns the best
vector seen so far.

```python
"""
Propose a non‑negative step‑function (vector of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a three‑stage scheme:

1. coarse sweep over monotone decaying families (exponential, linear,
   quadratic);
2. Adam optimiser on a log‑sum‑exp smoothed version of the max‑convolution,
   with projection onto the ℓ₁‑simplex;
3. sub‑gradient polishing followed by a brief random‑perturbation pass.

The whole search stops after ~budget seconds and returns the best sequence
encountered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of a vector using FFT.
    Returned length is 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0 , Σ x_i = target }.
    O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then re‑project onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def random_simplex_vector(n: int, target_sum: float) -> np.ndarray:
    """
    Draw a random point on the ℓ₁‑simplex (Dirichlet‑like) and scale it to
    `target_sum`.
    """
    x = np.random.exponential(1.0, size=n)
    return x * (target_sum / x.sum())


def generate_decay_shape(
    n: int,
    target_sum: float,
    kind: str = "exponential",
    param: float = 1.0,
) -> np.ndarray:
    """
    Produce a monotone decreasing sequence of length n.
    * kind ∈ {"exponential", "linear", "quadratic"}.
    * param controls the strength of the decay (≥0).
    The result is projected onto the ℓ₁‑simplex.
    """
    idx = np.arange(n, dtype=np.float64)

    if kind == "exponential":
        # a_i ∝ exp(-param * i)
        raw = np.exp(-param * idx)
    elif kind == "linear":
        # a_i = 1 - param * i / (n-1)
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "quadratic":
        raw = 1.0 - param * (idx / (n - 1)) ** 2
    else:
        raise ValueError("unknown decay kind")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (grad, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # temperature, never smaller than machine epsilon
    tau = max(tau_factor * max_conv, 1e-12)

    # stable soft‑max weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # grad = (2/τ) * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def _subgradient_max(a: np.ndarray):
    """
    Exact sub‑gradient of max(conv(a,a)).
    Returns (grad, max_conv, idx_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())

    # indices that contribute to the maximal entry
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)

    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects a ~budget‐second time limit and always
    returns the best vector found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a previously discovered construction if it exists
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_vec = np.array(base, dtype=np.float64)
    else:
        n = np.random.randint(7500, 8700)   # comfortable size for FFT‑based work
        base_vec = None

    target_sum = math.sqrt(2.0 * n)           # ℓ₁‑norm we keep fixed

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric sweep (monotone decays)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)     # uniform placeholder
    best_score = float("inf")

    print(f"[{time.time():.2f}] Stage 1 – parametric sweep (n={n})")

    # exponential decay
    for p in np.linspace(0.0005, 0.003, 30):
        cand = generate_decay_shape(n, target_sum, kind="exponential", param=p)
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            print(f"  new best exp (λ={p:.5f}) → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # linear decay
    for p in np.linspace(0.0, 0.8, 20):
        cand = generate_decay_shape(n, target_sum, kind="linear", param=p)
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            print(f"  new best lin (p={p:.3f}) → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # quadratic decay
    for p in np.linspace(0.0, 1.5, 20):
        cand = generate_decay_shape(n, target_sum, kind="quadratic", param=p)
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            print(f"  new best quad (p={p:.3f}) → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # incorporate an existing construction if it beats the sweep
    if base_vec is not None:
        cur = conv_self_fft(base_vec).max()
        if cur < best_score:
            best_score = cur
            best_seq = base_vec.copy()
            print(f"  baseline improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam on smoothed max‑convolution
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Stage 2 – Adam optimisation")
    global_best_seq = best_seq.copy()
    global_best_score = best_score

    run_id = 0
    while time.time() < deadline:
        # choose start point
        if run_id == 0 and base_vec is not None:
            a = base_vec.copy()
            a = random_pair_transfer(a, target_sum, scale=0.02)
        else:
            a = random_simplex_vector(n, target_sum)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        alpha = 0.5 * (target_sum / n)          # initial step size

        tau_factor = 0.07                       # initial smoothing factor
        it = 0
        stagnant = 0

        while time.time() < deadline:
            grad, cur_max = _smoothed_gradient(a, tau_factor)
            it += 1
            # Adam updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = alpha * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)

            # keep global best
            if cur_max < global_best_score - 1e-12:
                global_best_score = cur_max
                global_best_seq = a.copy()
                print(f"Run {run_id:02d}, iter {it:06d} – new global best {global_best_score:.6f}")

            # simple stagnation detection
            if cur_max + 1e-12 < best_score:
                best_score = cur_max
                stagnant = 0
            else:
                stagnant += 1

            # schedule adaptations
            if it % 500 == 0:
                tau_factor = max(0.001, tau_factor * 0.94)
                alpha *= 0.97

            # occasional jitter to escape flat regions
            if np.random.rand() < 0.008:
                a = random_pair_transfer(a, target_sum, scale=0.015)

            if stagnant > 3000:
                a = random_pair_transfer(a, target_sum, scale=0.02)
                stagnant = 0

            if global_best_score < 1.503:
                break

            if it >= 8000:            # safe upper bound per run
                break

        run_id += 1
        # early exit if we already beat the target
        if global_best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient polishing (τ → 0)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Stage 3 – sub‑gradient refinement")
    a = global_best_seq.copy()
    eta_sub = 0.5 * (target_sum / n)
    sub_it = 0
    while time.time() < deadline and sub_it < 4000:
        grad, cur_max, _ = _subgradient_max(a)
        a = a - eta_sub * grad
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)
        sub_it += 1

        if cur_max < global_best_score - 1e-12:
            global_best_score = cur_max
            global_best_seq = a.copy()
            print(f" sub‑grad iter {sub_it:05d} – improved to {global_best_score:.6f}")

        if sub_it % 500 == 0:
            eta_sub *= 0.85

        if global_best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 4 – final random polishing
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Stage 4 – final polishing")
    for _ in range(80):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(global_best_seq.copy(), target_sum, scale=0.006)
        cand_score = conv_self_fft(cand).max()
        if cand_score < global_best_score:
            global_best_score = cand_score
            global_best_seq = cand.copy()
            print(f"  polishing improved to {global_best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished – best score {global_best_score:.6f}")
    return global_best_seq.tolist()
```
