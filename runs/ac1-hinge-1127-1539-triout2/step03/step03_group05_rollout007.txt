def _exp_update(a, grad, eta, target_sum):
    # a: np.ndarray
    # grad: np.ndarray same shape
    # compute a * exp(-eta*grad)
    a_new = a * np.exp(-eta * grad)
    # avoid zeros? if a is zero, stays zero; but we need sum >0; ensure no all zero.
    total = a_new.sum()
    if total == 0:
        # fallback: uniform distribution
        a_new = np.full_like(a, target_sum / a.size)
    else:
        a_new *= (target_sum / total)
    return a_new

"""
Propose a non‑negative step‑function (vector of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear self‑convolution.

Algorithm outline
-----------------
1. Gather a few sensible initial candidates (uniform, several symmetric
   “U‑shapes”, a centre‑heavy shape and a few random vectors).  Keep the
   best one.
2. Exponentiated‑gradient descent on a smooth log‑sum‑exp approximation
   of the max‑convolution.  The update is multiplicative, keeping the
   vector on the ℓ₁‑simplex.
3. Smoothed gradient descent with Nesterov‑style momentum.
4. Classic sub‑gradient descent on the actual max‑convolution entry
   (step‑size decays).
5. A max‑min balancing step that simultaneously pushes the current
   maximal convolution entry down and lifts the minimal non‑zero entry.
6. Final random polishing via tiny pair‑wise mass transfers.

All stages respect the time budget (≈1000 s) and the best vector found
throughout is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=target}."""
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(n: int, target_sum: float, kind: str = "linear", param: float = 0.0) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.

    kind ∈ {linear, quadratic, exponential, centre}
    * linear      – height grows linearly with distance from centre
    * quadratic   – height grows quadratically with distance
    * exponential – height grows exponentially with distance
    * centre      – opposite of the above: mass concentrated in the centre
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "centre":
        raw = 1.0 + param * (1.0 - d)                # high in the centre
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (grad, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)

    # stable log‑sum‑exp weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w  *  a_rev)  restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    # clip to avoid overflow in the exponentiated update
    grad = np.clip(grad, -1e3, 1e3)
    return grad, max_conv


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a, a)).
    Returns (grad, max_conv, idx_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max


def subgradient_at_index(a: np.ndarray, idx: int) -> np.ndarray:
    """Gradient of the convolution entry with index `idx`."""
    n = a.shape[0]
    low = max(0, idx - (n - 1))
    high = min(idx, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx - idx_range]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimizes the evaluation metric.
    The routine respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – use the existing construction if present
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_vec = np.array(base, dtype=np.float64)
    else:
        n = np.random.randint(7500, 8700)
        base_vec = None

    target_sum = math.sqrt(2.0 * n)           # after normalisation the objective = max(conv)

    # ------------------------------------------------------------------
    # Stage 0 – generate a rich pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(project_onto_simplex(np.ones(n), target_sum))

    # parametric U‑shapes (including centre‑heavy)
    for kind in ("linear", "quadratic", "exponential", "centre"):
        for p in np.linspace(0.0, 4.0, 41):
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            candidates.append(cand)

    # few random starts
    for _ in range(6):
        cand = np.random.rand(n)
        candidates.append(project_onto_simplex(cand, target_sum))

    # previous best, if any
    if base_vec is not None:
        candidates.append(base_vec.copy())

    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Initial best max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient descent on smoothed objective
    # ------------------------------------------------------------------
    a = best_seq.copy()
    eta = 0.25                 # multiplicative step size
    tau_factor = 0.12
    no_improve = 0
    it = 0
    while time.time() < deadline and it < 8000:
        grad, cur_max = _smoothed_gradient(a, tau_factor)
        a = a * np.exp(-eta * grad)
        a = a * (target_sum / a.sum())
        it += 1
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            no_improve = 0
            print(f"Exp‑Grad it {it:05d} → {best_score:.6f}")
        else:
            no_improve += 1

        # adaptive schedules
        if it % 1000 == 0:
            eta *= 0.85
            tau_factor = max(0.001, tau_factor * 0.85)

        # occasional jitter to shake off plateaus
        if no_improve >= 1200:
            jitter = np.random.randn(n) * 4e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Exp‑Grad jitter at it {it}")

    # ------------------------------------------------------------------
    # Stage 2 – smoothed gradient descent with momentum (Nesterov style)
    # ------------------------------------------------------------------
    eta = 0.18 * (target_sum / n)
    tau_factor = 0.06
    beta = 0.9
    velocity = np.zeros_like(a)
    it2 = 0
    no_improve2 = 0
    while time.time() < deadline and it2 < 6000:
        grad, cur_max = _smoothed_gradient(a, tau_factor)
        # Nesterov update
        velocity = beta * velocity + (1.0 - beta) * grad
        a_new = a - eta * velocity
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        a = a_new
        it2 += 1
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            no_improve2 = 0
            print(f"Momentum it {it2:05d} → {best_score:.6f}")
        else:
            no_improve2 += 1

        if it2 % 800 == 0:
            eta *= 0.93
            tau_factor = max(0.001, tau_factor * 0.94)
            beta = max(0.5, beta * 0.97)

        if no_improve2 >= 1800:
            jitter = np.random.randn(n) * 6e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve2 = 0
            print(f"Momentum jitter at it {it2}")

    # ------------------------------------------------------------------
    # Stage 3 – pure sub‑gradient descent on the true max‑conv entry
    # ------------------------------------------------------------------
    eta_sub = 0.6 * (target_sum / n)
    sub_it = 0
    no_improve3 = 0
    while time.time() < deadline and sub_it < 20000:
        grad, cur_max, idx_max = _subgradient_max(a)
        step = eta_sub / (1.0 + 0.0005 * sub_it)
        a = a - step * grad
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)
        sub_it += 1
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            no_improve3 = 0
            print(f"Sub‑grad it {sub_it:05d} → {best_score:.6f}")
        else:
            no_improve3 += 1
        if sub_it % 2000 == 0:
            eta_sub *= 0.88

    # ------------------------------------------------------------------
    # Stage 4 – max‑min balancing (flattens the convolution)
    # ------------------------------------------------------------------
    bal_it = 0
    eta_bal = 0.12 * (target_sum / n)
    while time.time() < deadline and bal_it < 5000:
        conv = conv_self_fft(a)
        idx_max = int(conv.argmax())
        # choose the smallest positive entry as the “minimum’’
        pos = np.where(conv > 0)[0]
        if pos.size == 0:
            break
        idx_min = int(conv[pos].argmin())
        grad_max = subgradient_at_index(a, idx_max)
        grad_min = subgradient_at_index(a, idx_min)
        direction = -grad_max + grad_min
        a = a - eta_bal * direction
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)
        bal_it += 1
        cur_max = conv[idx_max]
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            print(f"Balancing it {bal_it:04d} → {best_score:.6f}")
        if bal_it % 1000 == 0:
            eta_bal *= 0.85

    # ------------------------------------------------------------------
    # Stage 5 – final random polishing
    # ------------------------------------------------------------------
    polish_it = 0
    while time.time() < deadline and polish_it < 5000:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.005)
        cand_score = conv_self_fft(cand).max()
        polish_it += 1
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish it {polish_it:04d} → {best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished – final best max‑conv = {best_score:.6f}")
    return best_seq.tolist()
