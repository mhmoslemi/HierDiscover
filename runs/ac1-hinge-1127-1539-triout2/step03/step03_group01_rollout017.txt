    if i_high == i_low:
        i_low = (i_low + 1) % n
    delta = epsilon * a[i_high]
    a_new = a.copy()
    a_new[i_high] -= delta
    a_new[i_low] += delta
    # enforce non-negativity later via projection
    return a_new

def _optimize_local(a0, deadline, rng):
    a = a0.copy()
    S = np.sum(a)  # keep sum fixed (we treat as 1)
    n = len(a)
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # hyperparameters
    eta = 0.5
    eta_decay = 0.9998
    tau = cur_max
    tau_decay = 0.9999
    temp = cur_max * 0.1
    temp_decay = 0.9995
    epsilon_move = 0.05
    jitter_prob = 0.02
    jitter_scale = 0.01
    move_prob = 0.6
    iter_counter = 0
    while time.time() < deadline:
        # compute convolution and max each loop
        b = _full_convolution_fft(a)
        cur_max = np.max(b)
        k_max = np.argmax(b)

        # 1. targeted mass move with probability move_prob
        if rng.random() < move_prob:

            # project onto simplex (sum S)

            accept = False

            else:

            if accept:

                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temp *= temp_decay
        else:

            # update max

            cur_max = np.max(b)

            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        # decay parameters

        tau *= tau_decay
        iter_counter += 1
    return best_a, best_max

"""
Search for a non‑negative step‑function minimizing the provided
evaluate_sequence().  The approach combines three ideas:

1.  A *spectral initialisation*:  a is obtained as the (inverse) FFT of
    the square‑root of a uniform target convolution, i.e. a ≈ √(uniform)
    in the frequency domain.  After a small shift to make it non‑negative
    the vector is projected onto the ℓ₁‑simplex (∑a = 1).

2.  A fast *projected gradient / targeted‑mass‑move* local optimiser.
    At each iteration the maximal entry of the self‑convolution is found.
    The sub‑gradient of this entry is used for an additive step followed
    by a Euclidean projection onto the simplex.
    With a high probability a *biased mass swap* is performed:
    a small amount of mass is moved from an index that contributes a lot
    to the current maximal convolution entry to an index that contributes
    little.  This aggressively flattens the convolution.

3.  Many *short restarts* with random lengths (800 … 8000) and random
    initialisations (spectral, random Dirichlet or a mixture with a
    previously found best sequence).  The best vector seen so far is
    retained and, after the time budget is exhausted, a final short
    refinement using the original soft‑max optimiser from the baseline
    code is applied.

The algorithm runs until the allotted time (≈1000 s) is exhausted and
returns the best sequence found.  All numbers are non‑negative, so the
security checks in ``evaluate_sequence`` pass unchanged.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    # If already feasible, just scale
    if np.all(v >= 0) and np.isclose(v.sum(), S):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, S / n, dtype=float)
    rho = np.where(cond)[0][-1]               # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear (non‑circular) convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Spectral initialisation: square‑root of a uniform convolution
# ----------------------------------------------------------------------
def _spectral_initial_sequence(n: int) -> np.ndarray:
    """
    Produce a length‑n vector whose *circular* self‑convolution is close
    to uniform.  This is done by taking the square‑root of the FFT of a
    uniform distribution of length 2·n‑1, inverting, shifting to
    non‑negative and projecting onto the ℓ₁‑simplex (sum = 1).
    """
    L = 2 * n - 1
    target = np.full(L, 1.0 / L, dtype=float)          # uniform target
    N = 1 << ((L - 1).bit_length())                  # power‑of‑2 length
    U = np.fft.rfft(target, n=N)                     # FFT of target
    A = np.sqrt(U)                                   # principal sqrt
    a_complex = np.fft.irfft(A, n=N)[:n]             # inverse FFT, truncate
    a = np.real(a_complex)
    # shift to make non‑negative (tiny negatives may appear due to round‑off)
    min_val = a.min()
    if min_val < 0:
        a = a - min_val
    # handle the degenerate case (should not happen)
    if a.sum() == 0:
        a = np.random.rand(n)
    a = _project_to_simplex(a, 1.0)                  # normalise to sum = 1
    return a


# ----------------------------------------------------------------------
# Targeted mass move: shift a small fraction from a high‑contributing
# index (for the current maximal convolution entry) to a low‑contributing
# index.
# ----------------------------------------------------------------------
def _targeted_move(a: np.ndarray, k_max: int, eps: float, rng: np.random.Generator) -> np.ndarray:
    n = a.shape[0]
    partner = k_max - np.arange(n)
    valid = (partner >= 0) & (partner < n)
    contrib = np.zeros(n)
    contrib[valid] = a[valid] * a[partner[valid]]
    # pick a source with large contribution
    i_src = np.argmax(contrib)
    # pick a destination with small contribution (avoid equality)
    i_dst = np.argmin(contrib)
    if i_src == i_dst:
        i_dst = (i_dst + 1) % n
    delta = eps * a[i_src]
    a_new = a.copy()
    a_new[i_src] -= delta
    a_new[i_dst] += delta
    return a_new


# ----------------------------------------------------------------------
# Fast local optimiser (projected gradient + biased mass swaps)
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a fast local search:
      * hard‑max sub‑gradient step on the current peak of a*a,
      * biased mass swaps that move mass away from indices contributing
        heavily to that peak,
      * occasional Gaussian jitter.
    Returns the best vector found and its maximal self‑convolution entry.
    """
    a = a0.copy()
    S = a.sum()                     # we keep the ℓ₁‑mass constant (normally 1)
    n = a.shape[0]

    # initialise bookkeeping
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters (tuned for speed vs. stability)
    eta = 0.5                # additive step size for gradient step
    eta_decay = 0.9995
    move_prob = 0.65         # how often we attempt a biased mass move
    eps_move = 0.08          # fraction of source mass moved per swap
    temp = cur_max * 0.1     # temperature for simulated‑annealing acceptance
    temp_decay = 0.9996
    jitter_prob = 0.015
    jitter_scale = 0.015

    while time.time() < deadline:
        # recompute convolution & peak
        b = _full_convolution_fft(a)
        cur_max = np.max(b)
        k_max = int(np.argmax(b))

        # --------------------------------------------------------------
        # 1) biased mass move (simulated annealing)
        # --------------------------------------------------------------
        if rng.random() < move_prob:
            a_candidate = _targeted_move(a, k_max, eps_move, rng)
            a_candidate = _project_to_simplex(a_candidate, S)

            b_cand = _full_convolution_fft(a_candidate)
            cand_max = np.max(b_cand)

            accept = False
            if cand_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                a = a_candidate
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temp *= temp_decay
            continue   # skip the gradient step this round

        # --------------------------------------------------------------
        # 2) hard‑max projected gradient step
        # --------------------------------------------------------------
        g = np.zeros_like(a)
        partner = k_max - np.arange(n)
        valid = (partner >= 0) & (partner < n)
        g[valid] = 2.0 * a[partner[valid]]
        if k_max % 2 == 0:        # diagonal term appears only once
            d = k_max // 2
            if 0 <= d < n:
                g[d] += a[d]

        a = _project_to_simplex(a - eta * g, S)

        # check improvement
        b = _full_convolution_fft(a)
        cur_max = np.max(b)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # --------------------------------------------------------------
        # 3) occasional jitter to escape tiny plateaus
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _full_convolution_fft(a)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # decay schedules
        eta *= eta_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# Final refinement using the more sophisticated baseline optimiser.
# ----------------------------------------------------------------------
def _final_refine(a0: np.ndarray,

    so far, using the remaining time budget.
    """
    # The baseline code (copied from the reference solution) is reused
    # here with a tiny adaptation to work as a helper.
    def _softmax_update(a: np.ndarray, tau: float, eta_: float):

        b = _full_convolution_fft(a)
        max_b = np.max(b)

        if tau <= 1e-12:  # hard max
            k = np.argmax(b)
            idx = np.arange(n)
            partner = k - idx
            valid = (partner >= 0) & (partner < n)
            g = np.zeros_like(a)
            g[valid] = 2.0 * a[partner[valid]]
            if k % 2 == 0:
                d = k // 2
                if 0 <= d < n:
                    g[d] = a[d]
        else:
            diff = b - max_b
            exps = np.exp(diff / tau)
            w = exps / np.sum(exps)

            # cross‑correlation a ⋆ w
            a_rev = a[::-1]
            m = 1 << ((len(w) + n - 1).bit_length())
            Fw = np.fft.rfft(w, n=m)
            Fa_rev = np.fft.rfft(a_rev, n=m)
            conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
            g = 2.0 * conv[n - 1:n - 1 + n]

        a_new = a - eta_ * g
        a_proj = _project_to_simplex(a_new, np.sqrt(2.0 * n))
        b_new = _full_convolution_fft(a_proj)
        return a_proj, b_new, np.max(b_new)

    def _optimize_local_base(a0_inner: np.ndarray,
                             deadline_inner: float,
                             rng_inner: np.random.Generator):
        a = a0_inner.copy()
        n = a.shape[0]
        b = _full_convolution_fft(a)
        cur_max = np.max(b)
        best_a = a.copy()
        best_max = cur_max

        eta = 0.6
        eta_decay = 0.9997
        move_prob = 0.12
        eps_move = 0.04
        temp = cur_max * 0.1
        temp_decay = 0.9998
        jitter_prob = 0.01
        jitter_scale = 0.02
        tau = cur_max
        tau_decay = 0.9995

        while time.time() < deadline_inner:
            if rng_inner.random() < move_prob:
                i = rng_inner.integers(n)
                if a[i] <= 0.0:
                    continue
                j = rng_inner.integers(n)
                if i == j:
                    continue
                delta = eps_move * a[i]
                a_c = a.copy()
                a_c[i] -= delta
                a_c[j] += delta
                b_c = _full_convolution_fft(a_c)
                cand_max = np.max(b_c)
                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng_inner.random() < prob:
                        accept = True
                if accept:
                    a = a_c
                    cur_max = cand_max
                    if cur_max < best_max:
                        best_max = cur_max
                        best_a = a.copy()
                temp *= temp_decay
            else:
                a, b, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()

            if rng_inner.random() < jitter_prob:
                noise = rng_inner.normal(scale=a.mean() * jitter_scale, size=n)
                a = np.clip(a + noise, 0.0, None)
                a = _project_to_simplex(a, np.sqrt(2.0 * n))
                b = _full_convolution_fft(a)
                cur_max = np.max(b)
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()

            eta *= eta_decay
            tau *= tau_decay

        return best_a, best_max

    return _optimize_local_base(a0, deadline, rng)

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # reuse an earlier construction if available
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Main loop: many independent restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # choose a length – explore a fairly wide interval
        if base_seq is not None:
            max_len = max(800, min(8000, int(base_seq.size * 1.5)))
        else:
            max_len = 8000
        n = rng.integers(800, max_len + 1)

        # ------------------------------------------------------------------
        # initialise a feasible vector (ℓ₁‑norm = 1)
        # ------------------------------------------------------------------
        if rng.random() < 0.5:
            a0 = _spectral_initial_sequence(n)
        else:
            a0 = rng.random(n)
            a0 = _project_to_simplex(a0, 1.0)

        # occasionally blend with a previously good construction
        if best_seq is not None and rng.random() < 0.2:
            m = min(len(best_seq), n)
            a0[:m] = 0.5 * a0[:m] + 0.5 * best_seq[:m]
            a0 = _project_to_simplex(a0, 1.0)

        # allocate a slice of time to this restart
        slice_deadline = min(time.time() + 15.0, deadline)

        # run the fast local optimiser
        a_opt, max_b = _optimize_local(a0, slice_deadline, rng)

        # compute internal score (identical to evaluate_sequence up to
        # clipping, which does not fire for our small numbers)
        score = 2.0 * n * max_b / (np.sum(a_opt) ** 2)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement (if time left) using the more sophisticated baseline
    # optimiser.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        refine_deadline = deadline
        best_seq, _ = _final_refine(best_seq, refine_deadline, rng)

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the black‑box function
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist() if best_seq is not None else [1.0]
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) "
          f"length = {len(final_seq)}")

    # store the result for possible reuse by a later run
    globals()['height_sequence_1'] = final_seq
    return final_seq
