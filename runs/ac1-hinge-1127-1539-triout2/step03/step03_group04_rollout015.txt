def simulated_annealing(a, conv, target_sum, time_deadline):
    n = len(a)
    best_a, best_conv = a.copy(), conv.copy()
    best_score = _score(a)  # should equal np.max(conv)
    cur_a, cur_conv = a.copy(), conv.copy()
    cur_score = best_score

    start = time.time()
    # parameters
    T0 = 0.1 * best_score  # scaling
    T = T0
    iteration = 0
    while time.time() < time_deadline:
        # pick donor, receiver
        # donor selection: weighted by a_i * sens? maybe just random but with probability proportional to a_i
        donor = int(rng.integers(n))
        if cur_a[donor] < 1e-12:
            continue
        # receiver: choose random index not donor
        receiver = donor
        while receiver == donor:
            receiver = int(rng.integers(n))
        # choose fraction
        max_frac = 0.2
        frac = rng.random() * max_frac
        d = cur_a[donor] * frac
        if d <= 0:
            continue

        # compute updated conv
        new_conv = cur_conv.copy()
        # shift updates
        # using original a values (cur_a) for shift contributions
        # We'll use vectorized slices
        # Note: we must ensure we use cur_a values before change
        # So use a_old = cur_a
        a_old = cur_a
        new_conv[donor : donor + n] -= 2 * d * a_old
        new_conv[receiver : receiver + n] += 2 * d * a_old

        # quadratic adjustments
        idx_2d = 2 * donor
        if idx_2d < new_conv.size:
            new_conv[idx_2d] += d * d
        idx_2jr = 2 * receiver
        if idx_2jr < new_conv.size:
            new_conv[idx_2jr] += d * d
        idx_cross = donor + receiver
        if idx_cross < new_conv.size:
            new_conv[idx_cross] -= 2 * d * d

        # compute new max
        new_max = new_conv.max()
        new_score = new_max  # because sum constant ensures equality
        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            # accept
            # update a
            cur_a[donor] -= d
            cur_a[receiver] += d
            cur_conv = new_conv
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                best_conv = cur_conv.copy()
                print(f"[{time.time():.1f}] SA best {best_score:.6f}")
        # update temperature
        iteration += 1
        if iteration % 1000 == 0:
            T *= 0.99  # cooling
            # optional reheat if stuck? Not needed.

    return best_a, best_score

def apply_move(conv, a, donor, receiver, delta):
    # a: original vector before move
    # conv: original convolution array before move
    # returns new conv (numpy array) with move applied.
    n = a.size
    new_conv = conv.copy()
    # shift contributions
    new_conv[donor:donor+n] -= 2 * delta * a
    new_conv[receiver:receiver+n] += 2 * delta * a
    # quadratics
    idx2d = 2*donor
    if idx2d < new_conv.size:
        new_conv[idx2d] += delta*delta
    idx2r = 2*receiver
    if idx2r < new_conv.size:
        new_conv[idx2r] += delta*delta
    idx_cross = donor + receiver
    if idx_cross < new_conv.size:
        new_conv[idx_cross] -= 2 * delta * delta
    return new_conv
```

def simulated_annealing(a, conv, deadline):
    n = len(a)
    sum_target = a.sum()  # assumed constant
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_score = cur_conv.max()
    best_a, best_conv, best_score = cur_a.copy(), cur_conv.copy(), cur_score

    rng = np.random.default_rng()
    # schedule parameters
    T0 = cur_score * 0.5  # initial temperature maybe half of current score
    T = T0
    iter_count = 0
    # fraction range adapted to temperature; larger T larger moves
    while time.time() < deadline:
        # pick donor with probability proportional to a (but non-zero)
        donor = rng.integers(n)
        if cur_a[donor] <= 1e-12:
            continue
        receiver = rng.integers(n-1)
        if receiver >= donor:
            receiver += 1
        # choose delta fraction
        max_frac = 0.2  # could vary with T
        frac = rng.random() * max_frac
        delta = cur_a[donor] * frac
        if delta <= 0:
            continue
        # compute new conv fast
        new_conv = cur_conv.copy()
        # shift updates refer to original a (cur_a)
        new_conv[donor:donor+n] -= 2 * delta * cur_a
        new_conv[receiver:receiver+n] += 2 * delta * cur_a
        idx2d = 2*donor
        if idx2d < new_conv.size:
            new_conv[idx2d] += delta*delta
        idx2r = 2*receiver
        if idx2r < new_conv.size:
            new_conv[idx2r] += delta*delta
        idx_cross = donor + receiver
        if idx_cross < new_conv.size:
            new_conv[idx_cross] -= 2*delta*delta

        new_score = new_conv.max()
        # acceptance rule
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            # accept
            cur_a[donor] -= delta
            cur_a[receiver] += delta
            cur_conv = new_conv
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                best_conv = cur_conv.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")
        # cooling
        iter_count += 1
        if iter_count % 2000 == 0:
            T *= 0.9
            # re‑compute conv from scratch every now and then to avoid drift
            # maybe only if iter_count % 10000
            if iter_count % 10000 == 0:

                cur_score = cur_conv.max()
    return best_a, best_score

"""
A fast, time‑boxed optimiser for the “max‑convolution / ℓ¹‑norm’’ benchmark.

The algorithm consists of three phases:

1. **Coarse grid search** –  Power‑law sequences a_i = (i+1)^(-p) are
   generated for several lengths (up to 20 000) and a broad range of exponents.
   The best few candidates (lowest max‑convolution) are kept.

2. **Simulated annealing with an O(n) incremental convolution update** –  
   Starting from a candidate, mass is transferred from a donor index to a receiver
   index.  The effect on the full convolution is computed analytically, which
   avoids repeated FFTs.  An annealing schedule accepts occasional worsening
   moves, allowing the search to escape local minima.  The current best sequence
   per candidate is retained.

3. **Deterministic hill‑climbing** –  After SA a simple greedy improvement
   (moving a small fraction of mass from “high‑sensitivity’’ to “low‑
   sensitivity’’ coordinates) is performed until the time budget expires.

The whole routine respects the 1000 s budget, returns the best sequence found,
and works entirely with NumPy (no filesystem or network access).  All numbers
are non‑negative, and the sum is kept equal to √(2·n), which means the objective
value is simply `max(a * a)`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def conv_max(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    """Maximum entry of a*a.  If the convolution is already known, reuse it."""
    if conv is None:
        conv = full_convolution(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Simple generators
# ----------------------------------------------------------------------


def power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def dirichlet_sequence(rng: np.random.Generator,
                       n: int,
                       alpha: float,
                       target_sum: float) -> np.ndarray:
    """Gamma‑Dirichlet draw (α > 0) normalised to the required sum."""
    a = rng.gamma(shape=alpha, scale=1.0, size=n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a *= target_sum / s
    return a


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s , x_i ≥ 0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simulated annealing – O(n) convolution update
# ----------------------------------------------------------------------


def _apply_move(conv: np.ndarray,
                a: np.ndarray,
                donor: int,
                receiver: int,
                delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` mass from `donor`
    to `receiver`.  The update is exact (quadratic terms included) and
    runs in O(n) via vectorised slicing.
    """
    n = a.size
    new_conv = conv.copy()

    # linear part  – 2·δ·a shifted by donor / receiver
    new_conv[donor: donor + n] -= 2.0 * delta * a
    new_conv[receiver: receiver + n] += 2.0 * delta * a

    # quadratic corrections
    idx = 2 * donor
    if idx < new_conv.size:
        new_conv[idx] += delta * delta
    idx = 2 * receiver
    if idx < new_conv.size:
        new_conv[idx] += delta * delta
    idx = donor + receiver
    if idx < new_conv.size:
        new_conv[idx] -= 2.0 * delta * delta

    return new_conv


def simulated_annealing(a0: np.ndarray,
                        conv0: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected simulated annealing that keeps the ℓ¹‑norm fixed.
    Returns the best sequence seen and its max‑convolution value.
    """
    n = a0.size
    target_sum = math.sqrt(2.0 * n)

    cur_a = a0.copy()
    cur_conv = conv0.copy()
    cur_best = cur_conv.max()
    best_a = cur_a.copy()
    best_val = cur_best

    rng = np.random.default_rng()
    # initial temperature ≈ 30 % of the current objective
    T = max(1e-12, 0.30 * cur_best)
    it = 0

    while time.time() < deadline:
        # --- pick a donor (probability ∝ a) and a distinct receiver ------------
        if cur_a.sum() == 0.0:
            break
        donor = rng.choice(n, p=cur_a / cur_a.sum())
        if cur_a[donor] <= 1e-12:
            continue
        # receiver – uniform but different from donor
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        # how much to move (fraction of donor's mass)
        max_frac = 0.25
        frac = rng.random() * max_frac
        delta = cur_a[donor] * frac
        if delta <= 0.0:
            continue

        # ---- compute the updated convolution ---------------------------------
        new_conv = _apply_move(cur_conv, cur_a, donor, receiver, delta)
        new_score = new_conv.max()

        # ---- Metropolis acceptance --------------------------------------------
        accept = False
        if new_score < cur_best:
            accept = True
        else:
            # probability exp(-(new‑old)/T)
            if T > 0:
                prob = math.exp(-(new_score - cur_best) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            # commit the move
            cur_a[donor] -= delta
            cur_a[receiver] += delta
            cur_conv = new_conv
            cur_best = new_score
            if new_score < best_val:
                best_val = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA  new global best = {best_val:.6f}")

        # ---- cooling schedule -------------------------------------------------
        it += 1
        if it % 2000 == 0:
            T *= 0.85  # exponential cooling
            # occasional full recomputation to avoid drift
            if it % 10000 == 0:
                cur_conv = full_convolution(cur_a)
                cur_best = cur_conv.max()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple deterministic hill‑climbing (greedy improvement)
# ----------------------------------------------------------------------


def guided_hill_climb(a0: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy “move mass from high‑sensitivity to low‑sensitivity’’ steps.
    Works on moderate lengths (few thousand) and stops when the time budget
    expires.
    """
    n = a0.size
    a = a0.copy()
    conv = full_convolution(a)
    best_val = conv.max()
    best_a = a.copy()
    rng = np.random.default_rng()

    max_frac = 0.07  # maximal fraction of donor’s mass to move each step

    while time.time() < deadline:
        k = int(np.argmax(conv))

        # sensitivity = a shifted by the peak index (zero outside the support)
        sens = np.zeros(n, dtype=np.float64)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        sens[lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

        weight = sens * a
        if weight.max() == 0.0:
            break
        donor = int(np.argmax(weight))

        # choose a receiver among the 10 % lowest weighted positions
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac

        # apply move (exact update)
        new_conv = _apply_move(conv, a, donor, receiver, delta)
        new_val = new_conv.max()
        if new_val < best_val:
            a[donor] -= delta
            a[receiver] += delta
            conv = new_conv
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GH  improved = {best_val:.6f}")
        # otherwise reject – do nothing

    return best_a, best_val
