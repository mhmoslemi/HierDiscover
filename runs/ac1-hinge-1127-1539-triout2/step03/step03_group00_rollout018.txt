def solve_lp(f, M):
    n = len(f)
    rows = 2*n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for k in range(rows):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # columns from j_start to j_end
        idxs = np.arange(j_start, j_end+1)
        A[k, j_start:j_end+1] = f[k - idxs]
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return res.x

def _lp_refine(a, target_sum, deadline, max_iters=10):
    n = a.size
    while time.time() < deadline and max_iters > 0:
        M = _max_convolution_fft(a)
        # build A
        # Use toeplitz for speed.
        col = np.concatenate([a, np.zeros(n-1)], dtype=np.float64)
        row = np.concatenate([a[:1], np.zeros(n-1)], dtype=np.float64)
        A = toeplitz(col, row)   # shape (2n-1, n)
        # Solve LP
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        b_ub = np.full(2*n-1, M, dtype=np.float64)

        # Solve linear program
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        except Exception as e:
            print("linprog exception", e)
            break

        if not res.success:
            # Might be singular or limited; break or continue with damping.
            print("LP not successful:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g * (target_sum / sum_g)

        # Update step size
        t = 0.5
        # compute new candidate
        improved = False
        while t > 1e-5:
            a_new = (1 - t) * a + t * g
            new_M = _max_convolution_fft(a_new)
            # accept if max decreased
            if new_M < M - 1e-8:
                a = a_new
                improved = True
                break
            t *= 0.5
        if not improved:
            # no improvement with convex combo; break
            break
        max_iters -= 1
    return a

"""
search for a non‑negative step‑function that minimizes
evaluate_sequence(...).

The algorithm consists of three stages:

1. Random initialisation on the ℓ¹‑simplex (several families).
2. LP‑refinement: for the current sequence f, solve a linear program
   that maximises ∑ g_j subject to the linear constraints
   (f * g)_k ≤ M  for all k, where M = max_k (f * f)_k.
   The optimal g concentrates mass where f * g is small.
   We then move f towards g with a convex combination.
   The LP is solved with SciPy's ``highs`` driver (dense but modest size).
3. A short simulated‑annealing (SA) phase to shake the result and
   possibly escape shallow local minima.

The whole routine respects the supplied time budget (default 1000 s). 
If a global variable ``height_sequence_1`` exists it is used as an
extra starting point.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Create a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    choice = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if choice == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)
    elif choice == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)
    elif choice == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif choice == 3:                   # Dirichlet sample
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum
    else:                               # uniform plus tiny jitter
        a = np.full(n, target_sum / n)

    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# LP refinement (max‑sum direction)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the dense convolution matrix A such that A @ g = f * g
    (linear convolution).  Returned as a CSR sparse matrix.
    """
    n = f.shape[0]
    col = np.concatenate([f, np.zeros(n - 1)], dtype=np.float64)
    row = np.concatenate([f[:1], np.zeros(n - 1)], dtype=np.float64)
    A_dense = toeplitz(col, row)           # shape (2n‑1, n)
    return csr_matrix(A_dense)


def _lp_refine(a: np.ndarray, target_sum: float, deadline: float,
               max_iters: int = 12) -> np.ndarray:
    """
    Iteratively improve ``a`` by solving the linear programme

        maximise   1ᵀg
        subject to conv(a,g) ≤ M
                   g ≥ 0

    where M = max(conv(a,a)).  The optimal g is taken, normalised to the
    required ℓ¹‑norm and mixed with a using a convex combination.
    The loop stops after ``max_iters`` steps, after any successful step
    fails to improve, or when the time budget expires.
    """
    n = a.size
    it = 0
    while time.time() < deadline and it < max_iters:
        M = _max_convolution_fft(a)

        # --- build the linear constraints conv(a,g) ≤ M ---------------------------------
        A = _build_conv_matrix(a)                       # (2n‑1) × n, sparse
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # --- solve the LP (maximise sum(g)) ---------------------------------------------
        c = -np.ones(n, dtype=np.float64)               # minimise -∑g ⇔ maximise ∑g
        bounds = [(0, None)] * n

        # give the solver a few seconds at most; it will abort early if the
        # overall deadline is close.
        time_left = max(0.0, deadline - time.time())
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'time_limit': time_left, 'presolve': True})
        except Exception as exc:                         # pragma: no cover
            print("linprog exception:", exc)
            break

        if not res.success:
            # In practice the LP should be feasible; abort if something went wrong.
            print("LP failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # --- rescale g so that its ℓ¹‑norm equals the target sum -------------------------
        g = g * (target_sum / sum_g)

        # --- convex update ---------------------------------------------------------------
        # start with a moderate step size; back‑track if it does not improve.
        step = 0.5
        improved = False
        while step > 1e-5:
            a_new = (1.0 - step) * a + step * g
            new_M = _max_convolution_fft(a_new)
            if new_M < M - 1e-9:                     # genuine improvement
                a = a_new
                improved = True
                break
            step *= 0.5
        if not improved:                               # no progress – stop
            break

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated annealing (fine tuning)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Simple SA that moves a random fraction of mass between two indices."""
    a = a0.copy()
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    # initialise temperature as a small fraction of the current score
    T = cur_score * 0.02 + 1e-6
    cooling = 0.9996

    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction (up to 15%) of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) minimising the
    competition objective.  The routine runs for at most ``budget_s``
    seconds, trying several lengths, random restarts, a linear‑program
    refinement phase and a short simulated‑annealing polishing.
    The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # give ourselves a small safety margin before the hard deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use a global construction if one is already present
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a = np.asarray(global_seq, dtype=float)
        n0 = a.size
        a = _project_to_simplex(a, np.sqrt(2.0 * n0))
        sc = evaluate_sequence(a.tolist())
        best_seq = a.copy()
        best_score = sc
        print(f"[{time.time():.1f}] start from global height_sequence_1 (n={n0}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a few moderate values work well with the LP
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Phase A – generate a few random seeds on the simplex
        # ------------------------------------------------------------------
        seeds = []
        for _ in range(4):
            a0 = _random_initial_sequence(n, rng)
            seeds.append(a0)

        # ------------------------------------------------------------------
        # Phase B – LP refinement + short SA for each seed
        # ------------------------------------------------------------------
        for a0 in seeds:
            if time.time() >= deadline:
                break

            # LP iterative refinement
            a_ref = _lp_refine(a0, target_sum, deadline, max_iters=10)

            # a quick SA polishing (uses whatever time left)
            a_ref = _simulated_annealing(a_ref, deadline)

            sc = evaluate_sequence(a_ref.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] new global best (n={n}) score={sc:.6f}")

        # ------------------------------------------------------------------
        # Keep the best sequence of this length for a final, longer SA run
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            a_final = _simulated_annealing(best_seq.copy(), deadline)
            sc = evaluate_sequence(a_final.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] refined best after SA (n={n}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be hit)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_seq = a
        best_score = evaluate_sequence(a.tolist())
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # Trim trailing near‑zero entries for neatness
    best_seq = np.asarray(best_seq)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f} length={best_seq.size}")
    return best_seq.tolist()
