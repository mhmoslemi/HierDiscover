def _adam_subgradient_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # step size schedule: lr = target_sum * 0.2 / sqrt(n) maybe
    lr0 = target_sum * 0.2 / np.sqrt(n) # base
    while time.time() < deadline:
        t += 1
        g, max_b = _subgradient_max_conv(a, eps=1e-8) # returns gradient and max conv
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)
        # occasionally jitter to escape plateau
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (t+1), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
    return a

def _subgradient_max_conv(a, eps=1e-12):
    n = a.shape[0]
    # compute conv via FFT
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = conv.max()
    # mask for near max values tolerance relative
    # Use absolute tolerance maybe 1e-12 or relative.
    mask = (conv >= max_b - eps)  # boolean mask
    # cast to float
    p = mask.astype(float)
    # convolution of a_rev and p
    a_rev = a[::-1]
    conv_len = n + (2*n-1) -1  # 3n-2
    size2 = 1 << ((conv_len).bit_length())
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n-1:n-1+n]
    return g, max_b

"""
Propose a non‑negative step‑function (list of heights) that minimises
evaluate_sequence(...).  The algorithm combines:

* diverse random initialisations (power‑law, exponential, Dirichlet,
  sinusoidal, mixed families) projected onto the ℓ¹‑simplex,
* a projected Adam optimiser that uses the exact sub‑gradient of the
  max‑convolution (the piecewise‑linear objective),
* a short simulated‑annealing refinement to escape possible local minima,
* a dynamic allocation of the 1000 s time budget over several vector
  lengths (up to 15000 entries).

The objective after ℓ¹‑normalisation equals the maximum entry of the
self‑convolution, so we minimise that maximum directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: projection onto the ℓ¹‑simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast max‑convolution (used for scoring)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation – simply max(a*a)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact, not smoothed)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, eps: float = 1e-12):
    """Return (sub‑gradient, max value) of max(a*a)."""
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv.max()
    tol = max(eps, max_b * 1e-12)
    mask = (conv >= max_b - tol).astype(float)      # 0/1 mask of active indices

    # gradient = 2 * (a_rev * mask) sliced to length n
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3n‑2
    size2 = 1 << ((conv_len).bit_length())
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    mask_fft = np.fft.rfft(mask, n=size2)
    conv_rev = np.fft.irfft(a_rev_fft * mask_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the exact sub‑gradient
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Projected Adam on max‑convolution (sub‑gradient)."""
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # base learning‑rate (empirically good for all explored n)
    lr0 = target * 0.30
    step_limit = target * 0.30

    while time.time() < deadline:
        t += 1
        g, _ = _subgradient_max_conv(a)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -step_limit, step_limit)

        a -= step
        a = _project_to_simplex(a, target)

        # occasional jitter to break plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Mass‑transfer SA with geometric cooling, returns best solution seen."""
    a = a0.copy()
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.05 + 1e-6        # initial temperature
    cooling = 0.9995
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move up to 20 % of the mass at i towards j
        frac = rng.random() * 0.20
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score

        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initial sequences (several families)
# ----------------------------------------------------------------------
def _random_initial_seq(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative vector on the ℓ¹‑simplex."""
    target = np.sqrt(2.0 * n)
    choice = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if choice == 0:          # pure power‑law
        p = rng.uniform(0.1, 1.5)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:        # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif choice == 2:        # mixed power‑law + exponential
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:        # Dirichlet sample (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif choice == 4:        # sinusoidal bell (positive)
        a = np.sin(np.pi * (idx + 0.5) / n)
        a = np.maximum(a, 0.0)

    else:                    # uniform with tiny jitter
        a = np.full(n, 1.0)

    # add tiny Gaussian noise before projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)

    # occasionally enforce symmetry (helps flatten convolution)
    if rng.random() < 0.25:
        a = (a + a[::-1]) * 0.5

    a = _project_to_simplex(a, target)
    return a

    # ------------------------------------------------------------------

        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] FINAL SA IMPROVEMENT score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up tiny trailing values for a neat result
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform sequence (should never happen)

    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
