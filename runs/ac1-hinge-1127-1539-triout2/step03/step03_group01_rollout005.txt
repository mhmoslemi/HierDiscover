def _optimize_flatten(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2*n)
    L = 1 << ((3*n - 2).bit_length())
    # pre
    # precompute best
    b = _full_convolution_fft_fixed_len(a, L)  # need a function returning conv length 2n-1
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # parameters
    eta = 0.001  # step size
    eta_decay = 0.9995
    jitter_prob = 0.02
    jitter_scale_frac = 0.01
    swap_prob = 0.1
    epsilon_move = 0.05  # fraction moved

    temperature = cur_max * 0.2

    target_const = S**2 / (2*n - 1)  # constant

    while time.time() < deadline:
        # choose operation
        r = rng.random()
        if r < swap_prob:
            # random mass move
            i = rng.integers(n)
            if a[i] <= 0:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = epsilon_move * a[i]
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # projection not needed because sum preserved; ensure nonneg
            # compute b candidate
            b_candidate = _full_convolution_fft_fixed_len(a_candidate, L)
            cand_max = np.max(b_candidate)
            accept = False
            if cand_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(cand_max - cur_max) / max(temperature, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                a = a_candidate
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temperature *= 0.999
        elif r < swap_prob + jitter_prob:
            # jitter
            noise = rng.normal(scale=a.mean() * jitter_scale_frac, size=n)
            a_j = a + noise
            a_j = np.clip(a_j, 0.0, None)
            a_j = _project_to_simplex(a_j, S)
            b_j = _full_convolution_fft_fixed_len(a_j, L)
            cur_max_j = np.max(b_j)
            a = a_j
            cur_max = cur_max_j
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        else:
            # gradient step
            # compute b
            b = _full_convolution_fft_fixed_len(a, L)
            cur_max = np.max(b)
            # error e = b - target_const
            e = b - target_const
            # compute gradient
            a_rev = a[::-1]
            # convolution e * a_rev
            Fe = np.fft.rfft(e, n=L)
            Fa_rev = np.fft.rfft(a_rev, n=L)
            conv = np.fft.irfft(Fe * Fa_rev, n=L)[:len(e) + n - 1]
            grad = 2.0 * conv[n-1:n-1+n]
            # step
            a_new = a - eta * grad
            a = _project_to_simplex(a_new, S)
            # compute new b and cur_max
            b = _full_convolution_fft_fixed_len(a, L)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        # decay eta
        eta *= eta_decay

    return best_a, best_max

def _optimize_flatten_local(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    target_const = S*S / (2*n - 1)
    L = 1 << ((3*n - 2).bit_length()) # ensure enough length for conv e * a_rev

    # compute initial best
    b = _conv_fft(a, a, L)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # parameters
    eta = 1e-5
    eta_decay = 0.9995
    swap_prob = 0.15
    jitter_prob = 0.02
    jitter_scale = 0.02
    move_frac = 0.05
    temp = cur_max * 0.1
    temp_decay = 0.9998

    while time.time() < deadline:
        if rng.random() < swap_prob:
            # random move
            i = rng.integers(n)
            if a[i] <= 0:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = move_frac * a[i]
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # Keep non-negative
            a_candidate = np.clip(a_candidate, 0.0, None)
            # sum preserved, not need projection
            b_candidate = _conv_fft(a_candidate, a_candidate, L)
            cand_max = b_candidate.max()
            if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(temp, 1e-12)):
                a = a_candidate
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temp *= temp_decay
        elif rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _conv_fft(a, a, L)

"""
Flat‑Convolution optimiser.

The goal is to minimise 2·n·max(a*a)/(sum(a)^2).  Since
∑(a*a)=sum(a)^2 is constant, minimising the maximal entry of the
self‑convolution is equivalent to flattening the convolution as close
to a constant as possible.  The optimiser:

* normalises the ℓ₁‑norm to √(2·n) (projection onto the simplex),
* repeatedly computes the full convolution via an FFT,
* defines the fixed target value t = S²/(2·n‑1) (the exact mean of the
  convolution for any feasible vector),
* takes a gradient step for the L₂‑loss ∥a*a−t∥₂²,
* occasionally performs simulated‑annealing mass swaps and Gaussian
  jitter to escape local minima,
* projects back onto the simplex after every update.

Many short restarts with different lengths are performed within the
time budget; the best sequence found is returned and stored in the
global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution via real FFT (returns full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, L: int) -> np.ndarray:
    """Convolve u and v using FFT of length L, truncate to len(u)+len(v)-1."""
    Fu = np.fft.rfft(u, n=L)
    Fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Fu * Fv, n=L)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
# Initialise a feasible vector (ℓ₁‑norm = sqrt(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None, n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with ℓ₁‑norm √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=0.001 * a.mean(), size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                                 # Dirichlet(1)
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Local optimiser – flatten convolution using L2‑loss gradient steps
# ----------------------------------------------------------------------
def _optimize_flatten_local(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the maximal entry of a*a by flattening the convolution.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    target = S * S / (2 * n - 1)                     # exact mean of a*a
    L = 1 << ((3 * n - 2).bit_length())              # length for e*a_rev convolution

    # initial record
    b = _conv_fft(a, a, L)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters
    eta = 1e-5               # gradient step size
    eta_decay = 0.9995
    swap_prob = 0.12
    jitter_prob = 0.02
    jitter_scale = 0.02
    move_frac = 0.05
    temperature = cur_max * 0.1
    temp_decay = 0.9998

    while time.time() < deadline:
        r = rng.random()
        # --------------------------------------------------------------
        # 1) random mass move (simulated annealing)
        # --------------------------------------------------------------
        if r < swap_prob:
            i = rng.integers(n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = move_frac * a[i]
            a_cand = a.copy()
            a_cand[i] -= delta
            a_cand[j] += delta
            a_cand = np.clip(a_cand, 0.0, None)          # keep non‑negative

            b_cand = _conv_fft(a_cand, a_cand, L)
            cand_max = b_cand.max()
            accept = False
            if cand_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(cand_max - cur_max) / max(temperature, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                a = a_cand
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temperature *= temp_decay

        # --------------------------------------------------------------
        # 2) Gaussian jitter
        # --------------------------------------------------------------
        elif r < swap_prob + jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _conv_fft(a, a, L)
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # --------------------------------------------------------------
        # 3) gradient step towards a flat convolution
        # --------------------------------------------------------------
        else:
            b = _conv_fft(a, a, L)                      # a*a
            cur_max = b.max()
            e = b - target                               # deviation from flat target

            # gradient = 2 * (e * a_rev)   (slice of length n)
            a_rev = a[::-1]
            Fe = np.fft.rfft(e, n=L)
            Fa_rev = np.fft.rfft(a_rev, n=L)
            conv = np.fft.irfft(Fe * Fa_rev, n=L)[: len(e) + n - 1]
            grad = 2.0 * conv[n - 1:n - 1 + n]

            a_new = a - eta * grad
            a = _project_to_simplex(a_new, S)

            # recompute after projection
            b = _conv_fft(a, a, L)
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # decay step size
        eta *= eta_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Run many independent restarts (different lengths, different seeds)
    within the time budget, applying the flat‑convolution optimiser.
    Returns the sequence that achieves the smallest external score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # try to reuse a previously discovered construction, if any
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Main loop: many short restarts with varying lengths
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # choose a length – favour larger n (more degrees of freedom)
        if base_seq is not None and rng.random() < 0.6:
            max_len = max(2000, min(30000, int(base_seq.size * 1.5)))
        else:
            max_len = 30000
        n = rng.integers(2000, max_len + 1)   # n ∈ [2000, max_len]

        # initialise a feasible vector
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a slice of the remaining time to this restart
        slice_deadline = min(time.time() + 30.0, deadline)

        # run the optimiser
        a_opt, cur_max = _optimize_flatten_local(a0, slice_deadline, rng)

        # external score (same as evaluate_sequence but cheaper to compute)
        S = np.sqrt(2.0 * n)
        # sum is S by construction, but we recompute for safety
        sum_a = a_opt.sum()
        score = float(2.0 * n * cur_max / (sum_a * sum_a + 1e-12))

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} "
                  f"new best max={cur_max:.6f} score={score:.6f}")

    # ------------------------------------------------------------------
    # final sanity check – evaluate through the black‑box function
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq
        return final_seq
    else:
        # fallback – should never happen under normal circumstances
        return [1.0]
