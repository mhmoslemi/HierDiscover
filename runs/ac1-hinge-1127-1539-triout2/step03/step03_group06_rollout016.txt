def simplex_projection(v, S):
    # v: array, S: sum target (positive)
    # returns w = max(v - tau, 0) with sum w = S
    # find tau = (np.sort(v)[::-1].cumsum() - S) / (range)...
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, S):
    # Projects v onto {x >=0, sum x = S}
    # v can be any real vector.
    # Using sorting method O(n log n).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Simplex‑projected sub‑gradient descent on the maximum entry of the
self‑convolution.

The evaluation score equals the maximal entry of a*a when the vector is
normalised to ℓ₁‑norm √(2 n).  The optimiser works directly on that
objective: it repeatedly computes the convolution, builds a smooth
approximation of the max (soft‑max with a temperature that is slowly
increased) and takes a momentum step followed by a projection onto the
simplex {x ≥ 0, Σx = √(2 n)}.  All convolutions and correlations are
performed with FFTs, giving an O(n log n) cost per iteration.

Several random restarts with different lengths are launched until the
wall‑clock budget (≈ 1000 s) is exhausted.  The best sequence seen is
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper linear‑algebra utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a*a using a forward FFT.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    L = 1 << ((conv_len - 1).bit_length())          # next power of two
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[:conv_len]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute, for each i∈[0,n),  Σ_j w_{i+j}·a_j .
    This is the (non‑reversed) cross‑correlation of w and a.
    """
    n = a.shape[0]
    m = w.shape[0]                                   # = 2·n‑1
    L = 1 << ((m + n - 1 - 1).bit_length())          # length for FFT
    Fw = np.fft.rfft(w, n=L)
    Fa_rev = np.fft.rfft(a[::-1], n=L)               # FFT of reversed a
    full = np.fft.irfft(Fw * Fa_rev, n=L)[:m + n - 1]
    start = n - 1
    return full[start:start + n]                     # length n


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex {x≥0, Σx = S}
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=S}."""
    if S <= 0:
        raise ValueError("Target sum S must be positive")
    # Fast median‑of‑means style: sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine – sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _optimize_max_convolution(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.6,
    beta: float = 0.9,
    decay: float = 0.99995,
    jitter_prob: float = 0.015,
    alpha_start: float = 0.2,
    alpha_max: float = 20.0,
    alpha_inc: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Perform a single optimisation run starting from ``a0``.
    Returns the best sequence (still normalised to sum=√(2 n))
    and the corresponding maximal convolution entry.
    """
    a = a0.astype(np.float64).copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)               # required ℓ₁‑norm
    a *= S / a.sum()                     # safety normalisation

    best_a = a.copy()
    best_val = np.inf

    v = np.zeros_like(a)                 # momentum buffer
    eta = eta0
    alpha = alpha_start

    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # --- convolution and (soft) max weight ---------------------------------
        conv = _full_convolution_fft(a)                 # length 2n‑1
        M = conv.max()
        # soft‑max weights on the convolution entries, temperature = alpha
        # (the largest entry receives weight 1, others less)
        w = np.exp(alpha * (conv - M))
        w_sum = w.sum()
        if w_sum == 0.0:                     # should not happen, fallback
            w = np.where(conv == M, 1.0, 0.0)
            w_sum = w.sum()
        w /= w_sum

        # --- gradient of the (soft) max ---------------------------------------
        # corr[i] = Σ_j w_{i+j} a_j  = Σ_k w_k a_{k-i}
        corr = _correlation_fft(w, a)                 # length n

        # diagonal correction for the case k = 2i  (only 1 contribution)
        diag = np.zeros_like(a)
        idx2 = 2 * np.arange(n)
        mask = idx2 < w.shape[0]
        diag[mask] = w[idx2[mask]]

        grad = 2.0 * corr - diag * a                  # sub‑gradient direction

        # --- momentum + projected simplex step ---------------------------------
        v = beta * v + (1.0 - beta) * grad
        a_new = a - eta * v

        # clip to non‑negative then renormalise to keep Σa = S
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s > 0.0:
            a = a_new * (S / s)
        else:
            # rare degenerate case – fall back to uniform vector
            a = np.full(n, S / n)

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a = a * (S / a.sum())

        # --- schedule updates --------------------------------------------------
        eta *= decay
        alpha = min(alpha * alpha_inc, alpha_max)

        # record improvement
        if M < best_val - 1e-12:
            best_val = M
            best_a = a.copy()
            last_improve = it

        # early exit when we are about to run out of time
        if time.time() >= deadline - 0.2:
            break

    return best_a, best_val

    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine runs many independent restarts
    with different lengths, each using the optimiser above.  The best
    sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0)  Try the supplied baseline if present (height_sequence_1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                n0 = cand.size
                S0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * S0
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] height_sequence_1 score = {best_score:.6f}, n={n0}")
        except Exception:   # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1)  Lengths to explore – larger lengths usually give a better bound.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000, 24000, 30000]

    run_id = 0
    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ----------------------------------------------------------------
        # Initialise a starting vector.
        # ----------------------------------------------------------------
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.4:
            a0 = best_seq[:n].copy()
        else:
            a0 = rng.random(n)

        # Normalise to the required ℓ₁‑norm √(2 n)
        a0 = a0 / a0.sum() * math.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # Run a single optimisation.
        # ----------------------------------------------------------------
        seq, max_val = _optimize_max_convolution(a0, deadline,
                                                  max_iters=120_000,
                                                  eta0=0.8,
                                                  beta=0.9,
                                                  decay=0.99993,
                                                  jitter_prob=0.012,
                                                  alpha_start=0.2,
                                                  alpha_max=15.0,
                                                  alpha_inc=1.0012)

        # The surrogate value (max_val) equals the true score after our
        # normalisation, so we can compare directly.
        if max_val < best_score:
            best_score = max_val
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} max={best_score:.6f}")

        # ----------------------------------------------------------------
        # Occasionally perform a longer fine‑tuning on the current best.
        # ----------------------------------------------------------------
        if run_id % 7 == 0 and best_seq is not None:
            # short extra run with a tighter step‑size
            best_seq, best_score = _optimize_max_convolution(
                best_seq, deadline,
                max_iters=80_000,
                eta0=0.4,
                beta=0.95,
                decay=0.99997,
                jitter_prob=0.008,
                alpha_start=0.5,
                alpha_max=20.0,
                alpha_inc=1.002,
            )
            print(f"[refine] after run={run_id} new max={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 2)  Final polishing pass (if we have any candidate).
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_seq, best_score = _optimize_max_convolution(
            best_seq, deadline,
            max_iters=200_000,
            eta0=0.3,
            beta=0.97,
            decay=0.99998,
            jitter_prob=0.006,
            alpha_start=1.0,
            alpha_max=30.0,
            alpha_inc=1.003,
        )
        # compute the exact external score for logging
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] score = {final_score:.6f}, length = {len(best_seq)}")
    else:
        # ----------------------------------------------------------------
        # Fallback – uniform sequence (should never be needed).
        # ----------------------------------------------------------------
        n = 4000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {final_score:.6f}")

    return best_seq.tolist()
