def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient term g_j = 2 * Σ_i w_{i+j} * a_i.
    Both a (len n) and w (len 2n-1) are real vectors.
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)

    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # gradient slice: indices 2n-2 - j
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad

"""
Propose a candidate step‑function that reduces the evaluation score.
Improvements over the previous version:
- ADAM optimizer with adaptive learning rates.
- Two‑phase schedule: a smooth‑max phase gradually increasing the
  soft‑max temperature, then an exact‑max sub‑gradient phase.
- Targeted “mass‑swap” moves that preferentially move mass away from
  the shift that currently attains the maximal convolution.
- Wider exploration of sequence lengths (1500–8000) and many more
  short restarts.
The function respects the 1000 s budget and returns the best sequence
found as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of vector v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term g_j = 2 * Σ_i w_{i+j} * a_i.
    a : length n , w : length 2n‑1
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Weight generators
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max across the convolution vector (temperature α)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / np.sum(exp_shifted)
    return w


def _mask_weights(conv: np.ndarray) -> np.ndarray:
    """Binary mask that marks every index attaining the max."""
    max_val = np.max(conv)
    mask = (conv == max_val).astype(float)
    return mask


# ----------------------------------------------------------------------
# Optimiser (ADAM + two‑phase schedule)
# ----------------------------------------------------------------------


def _optimise_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   smooth_phase: bool = True) -> tuple[np.ndarray, float]:
    """
    ADAM optimiser.  In the smooth phase we use a soft‑max weight,
    in the final phase we switch to an exact max‑mask (sub‑gradient).
    Returns the best sequence seen and its true max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # ADAM state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # schedule parameters
    lr = 0.8                # initial step size
    lr_decay = 0.9995
    alpha = 1.0
    alpha_inc = 1.005

    # bookkeeping
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)

        # decide which weight formulation to use
        if smooth_phase and alpha < 50.0:
            w = _softmax_weights(conv, alpha)
            alpha *= alpha_inc
        else:
            w = _mask_weights(conv)          # exact sub‑gradient

        # gradient of the (smoothed) max‑norm
        grad = _conv_with_weights(a, w)

        # ADAM update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # projection onto the simplex {x≥0, Σx = S}
        a_new = _project_simplex(a_new, S)

        # occasionally perform a cheap “mass‑swap” move
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * rng.random() * 0.02
                cand = a_new.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # accept if helpful (or with a tiny temperature)
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a_new = cand
                    cur_max = cand_max

        # keep the best so far
        if cur_max < best_score:
            best_score = cur_max
            best_a = a_new.copy()
            last_improve = it

        # prepare for next iteration
        a = a_new
        lr *= lr_decay

        # early stop if no progress for a while
        if it - last_improve > 3000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If a base sequence is supplied, its first n entries are used with a
    tiny jitter; otherwise a Dirichlet(sample) is drawn.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimizes the evaluation score.
    The routine runs many short ADAM‑optimisation attempts with
    different lengths, keeps the best sequence found and finally
    returns it as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # a safety margin

    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # length selection – wider than before
        n = rng.integers(1500, 8000)

        # initialise
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a time slice (≈ 7 s per attempt)
        slice_deadline = min(time.time() + 7.0, deadline)

        # first perform a smooth‑max phase, then a hard max phase
        a_opt, score = _optimise_adam(a0, slice_deadline, rng, smooth_phase=True)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempts} n={n} "
                  f"new best score={best_score:.6f}")

    if best_seq is None:
        # fallback – a trivial uniform vector
        n = 2000
        a = np.full(n, 1.0)
        a = _project_simplex(a, np.sqrt(2.0 * n))
        best_seq = a

    # final sanity check with the public evaluator
    final = best_seq.tolist()
    external = evaluate_sequence(final)
    print(f"[final] external score = {external:.6f} "
          f"(internal max = {best_score:.6f}) length = {len(final)}")
    return final

    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)

        # choose the weighting scheme
        if smooth_phase and alpha < 50.0:
            w = _softmax_weights(conv, alpha)
            alpha *= alpha_inc
        else:
            w = _mask_weights(conv)                # exact sub‑gradient

        # gradient (or sub‑gradient)
        grad = _conv_with_weights(a, w)

        # ADAM update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new, S)

        # occasional random mass‑swap (helps escaping plateaux)
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * rng.random() * 0.02
                cand = a_new.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # tiny Metropolis acceptance when not improving
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.02, 1e-12)):
                    a_new = cand
                    cur_max = cand_max

        # keep the best vector seen so far
        if cur_max < best_score:
            best_score = cur_max
            best_a = a_new.copy()
            last_improve = it

        a = a_new
        lr *= lr_decay

        # stop if we have been idle for a long time
        if it - last_improve > 3000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    With probability 0.2 a geometric (decaying) shape is used;
    otherwise a Dirichlet (uniform‑over‑simplex) sample is drawn.
    If a base sequence is available its leading part is taken with a tiny
    jitter.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    elif rng.random() < 0.2:
        # geometric decay, r close to 1 for a smooth shape
        r = 1.0 - 1.0 / (n + 1.0)
        a = r ** np.arange(n, dtype=float)
    else:
        a = rng.random(n)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for the step‑function with the smallest evaluation score.
    The strategy is a collection of short ADAM‑optimisation runs with
    different lengths.  The best sequence seen is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    # optional warm‑start from a previously discovered construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # pick a length – a bit wider than before to give the optimiser more room
        n = rng.integers(1500, 8000)

        a0 = _initial_sequence(base_seq, n, rng)

        # each attempt gets a modest time slice (≈ 7 s)
        slice_deadline = min(time.time() + 7.0, deadline)

        a_opt, score = _optimise_adam(a0, slice_deadline, rng, smooth_phase=True)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempts} n={n} "
                  f"new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Light post‑processing: symmetrisation (can only improve or keep)
    # ------------------------------------------------------------------
    if best_seq is not None:
        sym = (best_seq + best_seq[::-1]) / 2.0
        sym = _project_simplex(sym, np.sqrt(2.0 * sym.shape[0]))
        sym_score = np.max(_full_convolution(sym))
        if sym_score < best_score:
            best_seq = sym
            best_score = sym_score
            print(f"[post] symmetrisation improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the public evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector of length 2000
        n = 2000
        best_seq = np.full(n, 1.0)
        best_seq = _project_simplex(best_seq, np.sqrt(2.0 * n))

    final = best_seq.tolist()
    external = evaluate_sequence(final)
    print(f"[final] external score = {external:.6f} "
          f"(internal max = {best_score:.6f}) length = {len(final)}")
    return final
