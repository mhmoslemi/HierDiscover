"""
Step‑function optimizer.

The evaluation score is 2·n·max(a*a) / (sum(a)^2).  
If we keep the L1‑norm Σa = √(2·n) the score equals the maximal
entry of the self‑convolution, i.e. `max(conv(a,a))`.  The algorithm
searches for a non‑negative vector with that norm that makes the
convolution as flat as possible.

Approach
--------
1. **Soft‑max gradient descent** – we smooth the “max’’ by a soft‑max
   with temperature *T*.  The gradient of the weighted sum
   `∑ w_k·(a*a)[k]` can be written as a cross‑correlation of the
   weight vector `w` with `a`.  Using FFT the gradient is obtained in
   O(n log n) time.  An exponentiated (log‑space) update keeps the
   variables non‑negative; after each step we renormalise the vector
   to the required L1‑norm.

2. **Random restarts** – many independent runs with different
   lengths (500 … 12000) and Dirichlet‑distributed initial vectors.
   The best result is kept globally.

3. **Local stochastic hill‑climbing** – after the gradient phase a
   short simulated‑annealing style hill‑climber performs small mass
   transfers between coordinates, accepting occasional worsening moves.
   This often fine‑tunes the solution.

4. **Symmetrisation** – a candidate can be forced to be symmetric
   (`a_i = a_{n‑1‑i}`) which often reduces the convolution peak.  Both
   the original and the symmetrised vector are evaluated.

The routine respects the 1000 s budget, prints progress,
and finally returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper: next power of two >= x."""
    return 1 << (x - 1).bit_length()


def conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n‑1) using FFT.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(np.max(conv_full(a)))


# ----------------------------------------------------------------------
# Gradient of the soft‑max weighted convolution
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, T: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k ∝ exp((conv_k - max(conv))/T).
    Returns a vector that sums to 1.
    """
    max_c = np.max(conv)
    # shift for numerical stability
    shifted = (conv - max_c) / T
    # exponentiate safely
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s == 0.0:
        # fallback to uniform weights (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] with respect to a.
    The i‑th component is 2·∑_k w_k·a_{k-i} (indices out of range ignored).
    Implemented via cross‑correlation using FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]

    # length of the correlation result = len(w) + n - 1
    corr_len = w.shape[0] + n - 1
    fft_len = _next_pow2(corr_len)

    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)

    # cross‑correlation w ⋆ a = irfft(W * conj(A_rev))
    corr = np.fft.irfft(W * np.conj(A_rev), n=fft_len)

    # the needed slice starts at index n‑1 and has length n
    start = n - 1
    g = corr[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Soft‑max (exponentiated) gradient descent
# ----------------------------------------------------------------------


def _run_softmax_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    T_start: float = 0.5,
    T_end: float = 0.01,
    decay_T: float = 0.9997,
    eta0: float = 0.30,
    decay_eta: float = 0.9998,
    max_iters: int = 40000,
    jitter_prob: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Soft‑max gradient descent with multiplicative (log‑space) updates.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)

    best_val = conv_max(a)
    best_a = a.copy()

    T = T_start
    eta = eta0
    no_improve = 0

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution and its max
        conv = conv_full(a)
        cur_max = float(np.max(conv))

        # record improvement
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            no_improve = 0
            print(f"[softmax improve] t={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}")
        else:
            no_improve += 1

        # soft‑max weights from current convolution
        w = _softmax_weights(conv, T)

        # gradient of weighted sum
        g = gradient_softmax(a, w)

        # log‑space multiplicative update (guarantees positivity)
        log_a = np.log(a + 1e-30) - eta * g
        a = np.exp(log_a)

        # renormalise to the exact L1‑norm required by the scoring function
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # schedule updates
        T = max(T_end, T * decay_T)
        eta *= decay_eta

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # early stop if stuck for a long time
        if no_improve > 2000:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def _local_hill_climb(
    a_start: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style hill‑climber.
    Moves a random fraction of mass from one index to another.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a_start.copy()
    cur_max = conv_max(a)
    best_a = a.copy()
    best_max = cur_max

    # initialise temperature as a small fraction of the current max
    temperature = max(0.05 * cur_max, 1e-4)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # move a random fraction (5‑20 %) of mass from i to j
        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = conv_max(a_new)

        # accept improvement or probabilistic worsening move
        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(f"[hill climb improve] t={time.time():.1f}s n={n} iter={it} max={new_max:.6f}")
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        # geometric cooling
        temperature *= 0.9995

        # occasional jitter to diversify search
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()
            cur_max = conv_max(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[jitter] t={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}")

    return best_a, best_max

    # Main optimisation loop (random restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length with a mild bias toward larger vectors
        probs = np.array([0.12, 0.10, 0.10, 0.15, 0.15, 0.12, 0.10, 0.08, 0.08])
        probs /= probs.sum()
        n = int(rng.choice(lengths, p=probs))

        # ------------------------------------------------------------------
        # Initialise a starting vector
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            # Perturb the current best of this length
            a0 = best_seq.copy()
            a0 *= rng.uniform(0.85, 1.15, size=n)
            a0 = np.clip(a0, 0.0, None)
        else:
            # Dirichlet initialisation with random concentration
            conc = rng.uniform(0.05, 1.5)
            a0 = rng.dirichlet(np.full(n, conc))
            a0 = a0.astype(float)

        # enforce the required L1‑norm exactly
        a0 *= np.sqrt(2.0 * n) / a0.sum()

        # ------------------------------------------------------------------
        # Phase 1 – soft‑max gradient descent
        # ------------------------------------------------------------------
        a_opt, max_conv = _run_softmax_descent(
            a0,
            n,
            deadline,
            T_start=rng.uniform(0.3, 0.8),
            T_end=0.008,
            decay_T=rng.uniform(0.9995, 0.9999),
            eta0=rng.uniform(0.20, 0.45),
            decay_eta=rng.uniform(0.9995, 0.9999),
            max_iters=30000,
            jitter_prob=0.02,
        )
        score = max_conv

        # ------------------------------------------------------------------
        # Symmetrise and keep the better of the two
        # ------------------------------------------------------------------
        a_sym = (a_opt + a_opt[::-1]) * 0.5
        a_sym *= np.sqrt(2.0 * n) / a_sym.sum()
        sym_score = conv_max(a_sym)
        if sym_score < score:
            a_opt, score = a_sym, sym_score

        # ------------------------------------------------------------------
        # Phase 2 – local hill‑climbing (short refinement)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_refined, refined_max = _local_hill_climb(
                a_opt, n, deadline, rng, max_iters=8000
            )
            if refined_max < score:
                a_opt, score = a_refined, refined_max

        # ------------------------------------------------------------------
        # Record global improvement
        # ------------------------------------------------------------------
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback – very unlikely to be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (exact L1‑norm) and verification
    # ------------------------------------------------------------------
    n_final = best_seq.size
    best_seq = best_seq.astype(float)
    best_seq *= np.sqrt(2.0 * n_final) / best_seq.sum()   # exact norm
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {n_final}")

    return best_seq.tolist()
