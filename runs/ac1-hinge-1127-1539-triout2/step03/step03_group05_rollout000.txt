"""
search_algorithm.py

A multi‑phase stochastic‑gradient + simulated‑annealing optimiser for the
step‑function problem.

The objective (with the ℓ₁‐normalisation  Σa = √(2 n)) is simply
    max_k (a * a)[k] .
The algorithm proceeds as follows:

1. **Rich parametric sweep** – generate many symmetric families
   (linear, quadratic, exponential, Gaussian, Beta‑shaped) and keep the
   best one.  The sweep also evaluates a previously discovered construction
   (``height_sequence_1``) if it exists.

2. **Adam optimisation on a log‑sum‑exp smoothed max** – minimise a smooth
   surrogate of the maximum convolution while staying on the ℓ₁‑simplex.
   Adam’s adaptive learning rates and a slowly decreasing temperature give
   fast, stable descent.

3. **Simulated‑annealing local search** – starting from the best vector
   found so far, repeatedly transfer a small amount of mass between a few
   random indices.  Improvements are always accepted; worsening moves are
   accepted with a temperature schedule to escape plateaus.

4. **Final polishing** – a short pass of random pairwise transfers.

The routine respects the 1000 s time budget, prints progress, and finally
returns the best list of heights found.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a vector with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small amount of mass from a random donor index to a random
    receiver index, then re‑project onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]               # magnitude of the transfer
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls edge enhancement.
    The result is scaled to have ℓ₁‑norm `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_gaussian_shape(n: int, target_sum: float, sigma: float) -> np.ndarray:
    """
    Symmetric discrete Gaussian centred at the middle.
    sigma is the relative width (0 < sigma ≤ 0.5).
    """
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    raw = np.exp(-0.5 * ((x - mid) / (sigma * mid)) ** 2)
    return project_onto_simplex(raw, target_sum)


def generate_beta_shape(n: int, target_sum: float, a: float) -> np.ndarray:
    """
    Symmetric Beta‑like shape: (x(1‑x))^{a‑1}
    where x = i/(n‑1) ∈ [0,1].  a > 0.
    """
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    raw = (x * (1 - x)) ** (a - 1) + 1e-12   # avoid exact zero
    return project_onto_simplex(raw, target_sum)


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max of the self‑convolution.
    Returns (gradient, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)               # length 2n‑1
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)

    # stable soft‑max weights over the convolution entries
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w ⋆ reverse(a)) restricted to central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv

    # Helper to evaluate the current objective (max of self‑convolution).
    # ------------------------------------------------------------------
    def max_conv(v: np.ndarray) -> float:
        return conv_self_fft(v).max()

    # ------------------------------------------------------------------
    # Phase 0 – collect a diverse set of candidate shapes.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    candidates = []

    # Uniform (just a baseline)
    uniform = np.full(n, target_sum / n)
    candidates.append(("uniform", uniform))

    # Linear / quadratic / exponential U‑shapes
    for p in np.linspace(0.0, 5.0, 51):
        candidates.append(("linear", generate_U_shape(n, target_sum, "linear", p)))
    for p in np.linspace(0.0, 4.0, 41):
        candidates.append(("quadratic", generate_U_shape(n, target_sum, "quadratic", p)))
    for p in np.linspace(0.0, 3.0, 31):
        candidates.append(("exponential", generate_U_shape(n, target_sum, "exponential", p)))

    # Gaussian families
    for sigma in np.linspace(0.08, 0.5, 9):
        candidates.append(("gaussian", generate_gaussian_shape(n, target_sum, sigma)))

    # Symmetric Beta‑like families
    for a_param in np.linspace(0.4, 4.5, 10):
        candidates.append(("beta", generate_beta_shape(n, target_sum, a_param)))

    # Include the previous construction if present
    if base_vec is not None:
        candidates.append(("previous", base_vec))

    # Evaluate all candidates (stop early if we run out of time)
    for name, vec in candidates:
        if time.time() >= deadline:
            break
        score = max_conv(vec)
        if score < best_score:
            best_score = score
            best_seq = vec.copy()
            print(f"[{time.time():.2f}] New best after sweep: {name} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on a log‑sum‑exp smoothed objective.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    # Adam hyper‑parameters
    lr = 0.25 * (target_sum / n)   # initial learning rate (scaled)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    tau_factor = 0.07

    while time.time() < deadline:
        grad, cur_max = _smoothed_gradient(a, tau_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        a = a_new
        iteration += 1

        # Record improvement
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Adam iter {iteration:06d} → {best_score:.6f}")

        # Adaptive schedules
        if iteration % 800 == 0:
            lr *= 0.85
            tau_factor = max(0.001, tau_factor * 0.94)

        # Occasional random perturbation to avoid stagnation
        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.012)

        # Early exit if we already beat the target bound
        if best_score < 1.503:
            print(f"[{time.time():.2f}] Target reached during Adam ({best_score:.6f}); stopping early.")
            break

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing style local search.
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    temperature_start = 0.08 * cur_score
    stagnation = 0

    while time.time() < deadline:
        # Propose a move that transfers a small amount of mass between a few indices
        cand = cur.copy()
        # Apply 1–3 random pair transfers in one move
        for _ in range(np.random.randint(1, 4)):
            cand = random_pair_transfer(cand, target_sum, scale=0.018)

        cand_score = max_conv(cand)

        if cand_score < cur_score:
            # Accept improvement
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[{time.time():.2f}] SA improved to {best_score:.6f}")
                stagnation = 0
        else:
            # Probabilistic acceptance (simulated annealing)
            T = temperature_start * (0.9995 ** stagnation)
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                cur, cur_score = cand, cand_score
            stagnation += 1

        # Periodic restart with a tiny jitter to explore new basins
        if stagnation > 6000:
            cur = best_seq.copy()
            jitter = np.random.randn(n) * 0.001
            cur = project_onto_simplex(np.maximum(cur + jitter, 0.0), target_sum)
            cur_score = max_conv(cur)
            stagnation = 0

        if best_score < 1.503:
            print(f"[{time.time():.2f}] Target reached during SA ({best_score:.6f}); breaking.")
            break

    # ------------------------------------------------------------------
    # Phase 3 – Final polishing (short random perturbations).
    # ------------------------------------------------------------------
    for _ in range(120):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.005)
        cand_score = max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Polishing improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Search terminated. Final max(conv) = {best_score:.6f}")
    return best_seq.tolist()
