def compute_smooth_max_grad(a, beta):
    n = a.shape[0]
    conv = conv_full_fft(a)  # length 2n-1
    cur_max = np.max(conv)
    # compute softmax weights
    exp_terms = np.exp(beta * (conv - cur_max))
    sum_exp = np.sum(exp_terms)
    p = exp_terms / sum_exp
    # compute gradient via correlation
    L = 1 << ((len(p) + n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    P = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(A * np.conj(P), n=L)[:n]
    grad = 2.0 * corr
    smooth_max = cur_max + np.log(sum_exp) / beta
    return smooth_max, grad

The algorithm works in three phases:

1. **Variance descent** – minimise the L2 distance of the linear self‑convolution
   to the flat target 1/(2n‑1).  The gradient is obtained by a single FFT‑based
   correlation, and after each step the vector is projected onto the simplex
   (non‑negative, sum = 1).

2. **Smoothed‑max descent** – use a log‑sum‑exp approximation of the max
   (parameter β) and perform projected gradient descent with a schedule that
   slowly raises β, turning the smooth objective into the true max.

3. **Local stochastic hill‑climbing** – fine‑tune the best candidate found so
   far by moving small amounts of mass between coordinates and accepting
   occasional worsening moves (simulated annealing).

All phases respect the non‑negativity and clipping performed by
`evaluate_sequence`.  The routine runs many random restarts of phases 1‑2 of
different lengths (500 … 8000) until the (≈1000 s) time budget expires, and
finally returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of appropriate length."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a via FFT (fast, no extra copies)."""
    return float(np.max(_conv_full_fft(a)))


# ----------------------------------------------------------------------
# Phase 1 – variance descent (flatten the convolution)
# ----------------------------------------------------------------------


def _run_variance_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 1500,
                          lr0: float = 0.4,
                          decay: float = 0.998) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the squared error
        L = Σ_k ( (a*a)[k] – 1/(2n‑1) )² .
    Returns the best vector seen (in terms of true max‑convolution) and that
    max value.
    """
    a = a0.copy()
    target = 1.0 / (2 * n - 1)          # desired flat value when Σa = 1
    lr = lr0
    best_a = a.copy()
    best_max = _conv_max_fft(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a)                # length 2n‑1
        diff = conv - target

        # gradient = 4 * correlation(a, diff)
        L = 1 << ((diff.size + n - 1).bit_length())   # avoid circular wrap‑around
        A = np.fft.rfft(a, n=L)
        D = np.fft.rfft(diff, n=L)
        corr = np.fft.irfft(A * np.conj(D), n=L)[:n]
        grad = 4.0 * corr

        a -= lr * grad
        a = np.clip(a, 0.0, None)               # non‑negativity
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s                               # project onto the simplex (Σa = 1)

        lr *= decay

        cur_max = np.max(conv)
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            print(f"[var descent improve] t={time.time():.1f}s n={n} it={it} max={cur_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max descent (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (smooth_max, gradient) of the smoothed max
          φ_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
    Gradient = 2 * correlation(a, p) where p is the softmax over the
    convolution entries.
    """
    conv = _conv_full_fft(a)
    cur_max = np.max(conv)

    # stable softmax
    exp_terms = np.exp(beta * (conv - cur_max))
    sum_exp = exp_terms.sum()
    p = exp_terms / sum_exp                     # Σ_k p_k = 1

    # gradient = 2 * correlation(a, p)
    L = 1 << ((p.size + a.size - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    P = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(A * np.conj(P), n=L)[:a.size]
    grad = 2.0 * corr

    smooth = cur_max + np.log(sum_exp) / beta   # exact value of φ_β
    return smooth, grad, conv


def _run_smooth_max_descent(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 2000,
                            lr0: float = 0.08,
                            beta0: float = 5.0,
                            beta_factor: float = 1.015,
                            lr_decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max.
    Returns the best vector seen (according to the true max‑convolution) and that max.
    """
    a = a0.copy()
    lr = lr0
    beta = beta0
    best_a = a.copy()
    best_max = _conv_max_fft(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        smooth, grad, conv = _smooth_max_and_grad(a, beta)

        # simple gradient step
        a -= lr * grad
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s

        # update best according to the *real* max (not the smoothed one)
        cur_max = np.max(conv)
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            print(f"[smooth descent improve] t={time.time():.1f}s n={n} it={it} max={cur_max:.6f}")

        # schedules
        lr *= lr_decay
        beta *= beta_factor

    return best_a, best_max


# ----------------------------------------------------------------------
# Phase 3 – local stochastic hill climbing / simulated annealing
# ----------------------------------------------------------------------


def _local_hill_climb(a_start: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """Simple annealed hill climbing around a_start."""
    a = a_start.copy()
    cur_max = _conv_max_fft(a)
    best_a = a.copy()
    best_max = cur_max

    # temperature scaled to the current max
    temperature = max(0.05 * cur_max, 1e-4)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        frac = rng.uniform(0.05, 0.18)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _conv_max_fft(a_new)

        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(f"[hill climb improve] t={time.time():.1f}s n={n} it={it} max={new_max:.6f}")
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        temperature *= 0.9995

        # occasional jitter to avoid becoming stuck
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            cur_max = _conv_max_fft(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[jitter] t={time.time():.1f}s n={n} it={it} max={cur_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low score.
    Uses variance descent → smoothed‑max descent → hill‑climbing.
    Returns a Python list of heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 4.0   # safety margin

    best_seq = None
    best_score = float("inf")
    run_id = 0

    # ------------------------------------------------------------------
    # Problem sizes we will try (larger n gives a lower theoretical bound)
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000]

    while time.time() < deadline:
        # pick a length, bias a little towards the larger ones
        n = int(rng.choice(lengths,
                           p=np.array([0.15, 0.15, 0.2, 0.25, 0.25])))

        # ------- random start -------------------------------------------------
        conc = rng.uniform(0.05, 1.2)            # Dirichlet concentration
        a0 = rng.dirichlet(np.full(n, conc))
        a0 = a0 / a0.sum()                       # enforce Σa = 1

        # ------- phase 1: variance descent ------------------------------------
        # iteration budget scaled roughly inversely to n
        max_iters_v = max(300, int(1200000 / n))
        a_mid, _ = _run_variance_descent(
            a0, n, deadline,
            max_iters=max_iters_v,
            lr0=rng.uniform(0.2, 0.6),
            decay=rng.uniform(0.997, 0.9995)
        )

        # ------- phase 2: smooth‑max descent (if time permits) ---------------
        if time.time() < deadline:
            max_iters_s = max(300, int(800000 / n))
            a_opt, _ = _run_smooth_max_descent(
                a_mid, n, deadline,
                max_iters=max_iters_s,
                lr0=rng.uniform(0.02, 0.12),
                beta0=rng.uniform(4.0, 9.0),
                beta_factor=1.008 + rng.random() * 0.015,
                lr_decay=0.9995
            )
        else:
            a_opt = a_mid

        # ----- evaluate with the provided function (scale does not matter) -----
        external_score = evaluate_sequence(a_opt.tolist())
        if external_score < best_score - 1e-12:
            best_score = external_score
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Final local refinement (hill climbing) if any time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_refined, _ = _local_hill_climb(
            best_seq, best_seq.shape[0], deadline, rng, max_iters=20000
        )
        final_score = evaluate_sequence(a_refined.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_refined.copy()
            print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # final verification (the platform will also evaluate)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
