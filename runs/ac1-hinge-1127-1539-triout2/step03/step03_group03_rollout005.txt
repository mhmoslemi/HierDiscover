def _resample_sequence(seq, n):
    m = len(seq)
    if m == 0:
        # just uniform
        return None
    # original normalized positions from 0 to 1
    x_orig = np.linspace(0,1,m)
    x_new = np.linspace(0,1,n)
    seq_new = np.interp(x_new, x_orig, seq)  # linear interpolation
    # ensure nonneg
    seq_new = np.clip(seq_new, 0.0, None)
    return seq_new

    v = np.zeros_like(a)

    # schedule for alpha and lr
    alpha = 0.5
    alpha_target = 400.0  # final large temperature
    alpha_increment = (alpha_target/alpha)**(1/5000) # multiplicative factor? but easier: alpha *= 1.001 each step

    step = 0

        # gradient step
        conv = _full_convolution(a)
        # smooth weights
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**step)
        v_hat = v/(1-beta2**step)

                a[j] += delta

"""
Hybrid Adam + subgradient optimisation for the max entry of the
self‑convolution under a fixed ℓ¹‑norm.  Random restarts explore a wide
range of lengths (up to 20 000).  Each run first minimises a smooth
approximation of the ℓ∞‑norm (soft‑max) using Adam and a gradually
increasing temperature, then switches to a pure sub‑gradient step that
directly reduces the maximal convolution entry.  After each update the
candidate is projected onto the simplex {x≥0, Σx=√(2·n)}.  The best
sequence found before the time budget expires is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector v onto the simplex
    {x >= 0, sum x = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the pivot
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using FFT.
    Returns array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum ∑_k w_k (a*a)_k w.r.t. a.
    Implements 2 * (a ⋆ w_rev) where ⋆ is linear convolution.
    """
    n = a.shape[0]
    # reverse w to obtain correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # gradient slice: indices 2n-2 - j
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max over a convolution vector
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i = exp(alpha * conv_i) / Σ_j exp(alpha * conv_j).
    The parameter alpha controls smoothness (larger → sharper).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Resample an existing sequence to a new length (linear interpolation)
# ----------------------------------------------------------------------


def _resample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resample `seq` (length m) linearly to length `new_n`.
    The shape of the distribution is roughly preserved.
    """
    m = seq.shape[0]
    if m == 0:
        return None
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    return new_seq


# ----------------------------------------------------------------------
# Core optimiser (smooth + sub‑gradient phases)
# ----------------------------------------------------------------------


def _optimize_sequence(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Optimise a single candidate `a0` until `deadline` (seconds since epoch)
    using a two‑phase scheme:

    1. Adam on the soft‑max smoothed max‑norm (temperature grows).
    2. Pure sub‑gradient descent on the true max entry.

    Returns the best sequence encountered and its true maximal
    self‑convolution value.
    """

    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # ---------- Phase 1 : Adam on smooth max ----------
    # hyper‑parameters
    lr = 0.08
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step = 0
    alpha = 0.5          # starting temperature
    alpha_target = 400.0

    while time.time() < deadline:
        step += 1

        # compute full self‑convolution
        conv = _full_convolution(a)

        # soft‑max weights
        w = _smooth_weights(conv, alpha)

        # gradient of the smoothed objective
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass‑swap (helps escaping plateaus)
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.04
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # evaluate true max for bookkeeping
        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # schedule updates
        alpha = min(alpha * 1.002, alpha_target)
        lr *= 0.9997

        # early stop if no improvement for a while
        if step % 2000 == 0:
            # if we have already spent a significant fraction of the
            # allocated time, break to the second phase
            if time.time() > deadline - 2.0:
                break

        # optionally switch after a fixed number of iterations
        if step >= 6000:
            break

    # ---------- Phase 2 : direct sub‑gradient on the max ----------
    sub_lr = 0.12
    sub_step = 0
    while time.time() < deadline:
        sub_step += 1

        conv = _full_convolution(a)
        idx = int(np.argmax(conv))

        # gradient of the maximal entry (sub‑gradient)
        # g_j = 2 * a_{idx-j} if the index is valid, else 0
        pad = np.pad(a, (n - 1, n - 1), mode='constant')
        grads = 2.0 * pad[idx: idx + n][::-1]   # reversed slice gives proper order

        a = a - sub_lr * grads
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # decay learning rate
        sub_lr *= 0.9995

        # random small swap occasionally
        if rng.random() < 0.03:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.03
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        if sub_step % 1500 == 0:
            if time.time() > deadline - 0.5:
                break

    return best_a, best_score

"""
Hybrid Adam + sub‑gradient optimisation for the max entry of the
self‑convolution under a fixed ℓ¹‑norm.  Random restarts explore a wide
range of lengths (up to 20 000).  Each run first minimises a smooth
approximation of the ℓ∞‑norm (soft‑max) using Adam and a gradually
increasing temperature, then switches to a pure sub‑gradient step that
directly reduces the maximal convolution entry.  After each update the
candidate is projected onto the simplex {x≥0, Σx=√(2·n)}.  The best
sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector v onto the simplex
    {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k (a*a)_k w.r.t. a.
    Implements 2·(a ⋆ w_rev) where ⋆ denotes linear convolution.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice giving the correct gradient entries
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max over a convolution vector
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j).
    Larger ``alpha`` yields a sharper approximation of the infinity norm.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Resample an existing sequence to a new length (linear interpolation)
# ----------------------------------------------------------------------


def _resample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly interpolate ``seq`` (length m) to a new length ``new_n``.
    The shape of the distribution is roughly preserved.
    """
    m = seq.shape[0]
    if m == 0:
        return None
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    return new_seq
