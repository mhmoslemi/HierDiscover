def adam_descent(a_init, target_sum, deadline, max_iters=inf, init_lr=0.2, tau_factor=0.5, tau_decay=0.999, lr_decay=0.9995):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_local_a = a.copy()
    best_local_score = compute_max_conv(a)

    iter = 0
    while time.time() < deadline and (max_iters is None or iter < max_iters):
        conv = conv_self_fft(a)
        max_conv = conv.max()
        # smoothing tau
        tau = max(tau_factor * max_conv, 1e-8)
        # compute softmax weights
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        w /= w_sum
        # grad = (2/tau) * conv(w, a_rev)
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[:len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[len(a)-1 : len(a)-1 + len(a)]

        # Adam update: 
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**(iter+1))
        v_hat = v / (1 - beta2**(iter+1))
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # projection onto simplex (nonneg and sum)
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # update best local
        if max_conv < best_local_score:
            best_local_score = max_conv
            best_local_a = a.copy()
            # possibly reset counters

        # schedule updates
        if iter % 1000 == 0 and iter > 0:
            init_lr *= lr_decay
            tau_factor = max(tau_factor * 0.95, 0.001)  # maybe
        # occasional jitter / random transfers
        if np.random.rand() < 0.005:
            a = random_pair_transfer(a, target_sum, scale=0.005)
        iter += 1

    return best_local_a, best_local_score
```

def hill_climb(a, target_sum, deadline):
    best = a.copy()
    best_score = compute_max_conv(best)
    while time.time() < deadline:
        i, j = np.random.choice(len(a), 2, replace=False)
        delta = best[i] * np.random.rand() * 0.1 # up to 10% of value at i
        # ensure new values nonnegative
        new_i = best[i] - delta
        if new_i < 0:
            continue
        new_j = best[j] + delta
        cand = best.copy()
        cand[i] = new_i
        cand[j] = new_j
        cand = project_onto_simplex(cand, target_sum)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # print improvement
    return best, best_score

def adam_descent(a0, target_sum, deadline, max_iters=100000, init_lr=0.2, init_tau_factor=0.5, min_tau=1e-8, tau_decay=0.9995, lr_decay=0.9995):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_a = a.copy()
    best_score = compute_max_conv(a)
    iter = 0
    tau_factor = init_tau_factor
    while time.time() < deadline and iter < max_iters:
        conv = conv_self_fft(a)
        max_conv = conv.max()
        # smoothing
        tau = max(tau_factor * max_conv, min_tau)
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[:len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[len(a)-1 : len(a)-1 + len(a)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (iter + 1))
        v_hat = v / (1.0 - beta2 ** (iter + 1))
        step = init_lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a = a - step
        # ensure nonnegative and project onto simplex
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # compute new max after projection
        cur_max = compute_max_conv(a)
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            # print improvement
            print(f"[{time.time():.2f}] Adam iter {iter} improved to {best_score:.6f}")

        # schedule updates
        if (iter + 1) % 1000 == 0:
            init_lr *= lr_decay
            tau_factor = max(tau_factor * tau_decay, 0.001)

        # occasional random jitter to escape local minima
        if np.random.rand() < 0.01:
            jitter = np.random.randn(len(a)) * 0.0005
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)

        iter += 1
    return best_a, best_score

"""
Propose a non‑negative step function (height sequence) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

by combining a lightweight parametric sweep, a few random simplex
initialisations and a projected Adam optimiser with a log‑sum‑exp
smooth approximation of the max‑convolution.  The optimiser works
directly on the coefficients, keeps the ℓ₁‑norm fixed (projection onto
the simplex) and gradually reduces the smoothing parameter τ.
After the main optimisation the candidate is optionally up‑sampled to a
larger power‑of‑two length and polished with tiny random mass transfers.
The whole routine respects a 1000 s time budget and prints progress
information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Basic linear‑algebra / FFT helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # extreme case – all entries become equal
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny mass from a random index to another and project."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def random_simplex_vector(n: int, target_sum: float) -> np.ndarray:
    """Sample a random point on the ℓ₁‑simplex (Dirichlet with α=1)."""
    x = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return target_sum * x / x.sum()


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    The parameter controls the aggressiveness of the shape.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # normalised distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4                      # avoid exact zeros
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the self‑convolution of a."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
#  Optimiser (projected Adam with a log‑sum‑exp smoothing)
# ----------------------------------------------------------------------


def adam_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
    init_lr: float = 0.4,
    init_tau_factor: float = 0.5,
    lr_decay: float = 0.97,
    tau_decay: float = 0.97,
    min_tau: float = 1e-6,
):
    """
    Projected Adam optimiser for the smoothed max‑convolution objective.
    Returns the best sequence found during the run and its max‑convolution.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_max_conv(a)

    it = 0
    tau_factor = init_tau_factor

    while time.time() < deadline and it < max_iters:
        # ----- current convolution & smoothing -----
        conv = conv_self_fft(a)
        max_conv = conv.max()
        tau = max(tau_factor * max_conv, min_tau)

        # log‑sum‑exp weights
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[len(a) - 1 : len(a) - 1 + len(a)]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # ----- record improvement -----
        cur_score = compute_max_conv(a)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter {it:07d} → {best_score:.6f}")

        # ----- schedules & occasional jitter -----
        if (it + 1) % 1000 == 0:
            init_lr *= lr_decay
            tau_factor = max(tau_factor * tau_decay, 0.001)

        if np.random.rand() < 0.01:        # small stochastic kick
            jitter = np.random.randn(len(a)) * 0.0004
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)

        it += 1

    return best_a, best_score


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of seq (length n) to length new_n.
    The result is projected onto the simplex of appropriate total mass.
    """
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    up = np.interp(x_new, x_old, seq)
    target_sum = math.sqrt(2.0 * new_n)
    return project_onto_simplex(up, target_sum)

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a height sequence that minimises the evaluation metric.
    The function respects the time budget and returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # a tiny safety margin

    # ----- problem size -------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_seq = np.array(base, dtype=np.float64)
    else:
        # start with a power‑of‑two that is fast for FFTs
        n = 8192
        base_seq = None

    target_sum = math.sqrt(2.0 * n)           # normalisation Σa = √(2n)

    # ----- Phase 0 – parametric sweep ---------------------------------
    best_seq = np.full(n, target_sum / n)
    best_score = compute_max_conv(best_seq)

    shape_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
    param_grids = {
        "linear": np.linspace(0.0, 12.0, 61),
        "quadratic": np.linspace(0.0, 10.0, 51),
        "exponential": np.linspace(0.0, 8.0, 41),
        "power": np.linspace(0.3, 8.0, 20),
        "inverse": np.linspace(0.3, 8.0, 20),
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  new best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # ----- Random simplex seeds (cheap evaluation) --------------------
    n_random = 64
    for _ in range(n_random):
        cand = random_simplex_vector(n, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  random seed improved → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during random seeding.")
            return best_seq.tolist()

    # consider the supplied baseline if it exists
    if base_seq is not None and base_seq.shape[0] == n:
        score = compute_max_conv(base_seq)
        if score < best_score:
            best_score = score
            best_seq = base_seq.copy()
            print(f"  baseline improved → {best_score:.6f}")

    # ----- Phase 1 – Adam descent on the best candidate ---------------
    remaining = deadline - time.time()
    # reserve roughly 30 % of the remaining budget for a possible up‑sample
    descent_deadline = time.time() + 0.7 * remaining

    print(f"[{time.time():.2f}] Starting Adam descent (budget ≈ {descent_deadline - time.time():.0f}s)")
    best_seq, best_score = adam_descent(
        best_seq,
        target_sum,
        descent_deadline,
        max_iters=500_000,
        init_lr=0.4,
        init_tau_factor=0.5,
        lr_decay=0.97,
        tau_decay=0.97,
        min_tau=1e-6,
    )
    print(f"[{time.time():.2f}] Adam descent finished – best score {best_score:.6f}")

    # ----- Phase 2 – optional up‑sampling to a larger power of two -----
    # Choose the next power of two (but not larger than 32768 to stay fast)
    up_n = 1 << ((2 * n - 1).bit_length())
    if up_n > n and up_n <= 32768 and time.time() + 60 < deadline:
        print(f"[{time.time():.2f}] Upsampling to {up_n} for final refinement")
        up_seq = upsample_sequence(best_seq, up_n)
        up_target = math.sqrt(2.0 * up_n)

        up_deadline = deadline - 5.0   # keep a few seconds for polishing
        up_seq, up_score = adam_descent(
            up_seq,
            up_target,
            up_deadline,
            max_iters=300_000,
            init_lr=0.2,
            init_tau_factor=0.5,
            lr_decay=0.98,
            tau_decay=0.98,
            min_tau=1e-6,
        )
        if up_score < best_score:
            best_seq = up_seq
            best_score = up_score
            print(f"[{time.time():.2f}] Upsampled refinement improved to {best_score:.6f}")

    # ----- Phase 3 – final polishing with small random transfers -------
    print(f"[{time.time():.2f}] Polishing phase (score {best_score:.6f})")
    polish_deadline = deadline - 0.1
    while time.time() < polish_deadline:
        cand = random_pair_transfer(best_seq.copy(),
                                    math.sqrt(2.0 * len(best_seq)),
                                    scale=0.002)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"  polish improved → {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    return best_seq.tolist()
