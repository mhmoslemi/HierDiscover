Implementation details for projection onto simplex: Use standard algorithm O(n log n). Or use np.maximum and iterative algorithm (like sorting). Use method from Duchi et al. 2008.

Pseudo code for projection:
```
def proj_simplex(v, s):
    # onto {x >= 0, sum x = s}
    # If sum(v) == s and all v >=0: return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1 # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```
But we need careful about shape.

Alternatively, we can use projection to simplex implemented by:
```
def simplex_proj(x, s):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0)
    return w
```

Improved convex‐smooth optimisation for the self‑convolution bound.

The external score equals the maximal entry of the autocorrelation
once the vector is normalised to ℓ₁‑norm √(2 n).  We therefore minimise
the smooth surrogate

    fβ(a) = 1/β · log∑ₖ exp(β·(a*a)_k)

which converges to the true max as β→∞.
Its gradient is a simple cross‑correlation that can be obtained with two
FFT calls.  The algorithm uses an exponentiated‑gradient update (which
keeps the vector non‑negative) together with a simplex‑renormalisation,
momentum, a slowly increasing β and occasional jitter.  Multiple
random restarts with different lengths are performed within the wall‑time
budget.  The best sequence found is returned.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable softmax with temperature β."""
    m = x.max()
    y = np.exp(beta * (x - m))
    return y / y.sum()


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if v.sum() == s and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Should not happen, but fallback to uniform
        return np.full_like(v, s / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _run_smooth_opt(
    a0: np.ndarray,
    deadline: float,
    beta_start: float = 0.5,
    beta_max: float = 30.0,
    beta_mul: float = 1.018,
    eta0: float = 0.7,
    eta_decay: float = 0.9997,
    momentum: float = 0.9,
    jitter_prob: float = 0.015,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth max‑convolution surrogate.
    Returns the best vector seen (as ndarray) and its true max‑convolution.
    """
    n = a0.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # FFT length – must accommodate both the convolution (2n‑1) and
    # the cross‑correlation (n + 2n‑1 – 1 = 3n‑2)
    L = _next_pow2(3 * n)

    a = a0.copy()
    v = np.zeros_like(a)  # momentum buffer
    beta = beta_start
    eta = eta0

    # best bookkeeping
    conv_full = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
    best_val = conv_full.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- forward FFT of current vector -----
        A = np.fft.rfft(a, n=L)

        # ----- self‑convolution (full) -----
        conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        # ----- smooth max (log‑sum‑exp) -----
        # weight vector w = softmax(β · conv)
        w = _softmax(conv_full, beta)

        # ----- gradient (cross‑correlation of a with w) -----
        W = np.fft.rfft(w, n=L)
        grad = 2.0 * np.fft.irfft(A * np.conj(W), n=L)[:n]
        grad = np.real(grad)

        # ----- momentum -----
        v = momentum * v + (1.0 - momentum) * grad

        # ----- exponentiated update (keeps positivity) -----
        # Clip gradient to avoid overflow in exp()
        clip_val = 80.0 / max(eta, 1e-12)  # heuristic bound
        g_clip = np.clip(v, -clip_val, clip_val)
        a = a * np.exp(-eta * g_clip)

        # ----- jitter to escape plateaus -----
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)

        # ----- renormalise to the ℓ₁‑norm constraint -----
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a.fill(target_sum / n)
        else:
            a *= (target_sum / s)

        # ----- book‑keeping -----
        cur_max = conv_full.max()
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()

        # ----- schedule updates -----
        beta = min(beta_max, beta * beta_mul)
        eta *= eta_decay

        # early stop check
        if time.time() >= deadline - 0.1:
            break

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` score within the given time budget.
    Returns the best sequence found as a plain list[float].
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 4.0  # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0)  Use the supplied construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                n0 = cand.size
                target0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * target0
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = float(sc)
                print(f"[baseline] height_sequence_1 score = {best_score:.6f}, length = {n0}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1)  Define a set of lengths – we mix short (fast) and long runs
    # ------------------------------------------------------------------
    lengths = [1200, 2000, 3000, 4000, 6000, 8000, 12000, 16000]
    run_id = 0

    while time.time() < deadline:
        # pick a length – favour shorter ones for more restarts
        n = int(rng.choice(lengths, p=np.array([0.25] + [0.75 / (len(lengths) - 1)] * (len(lengths) - 1))))
        target_sum = math.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # 2)  Initialise a starting vector
        # ----------------------------------------------------------------
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.35:
            a0 = best_seq[:n].copy()
            # Re‑normalise to the required ℓ₁‑norm
            a0 *= target_sum / a0.sum()
        else:
            # Dirichlet (exponential) start – guarantee strict positivity
            a0 = rng.exponential(scale=1.0, size=n)
            a0 *= target_sum / a0.sum()

        # ----------------------------------------------------------------
        # 3)  Run the smooth exponential‑gradient optimiser
        # ----------------------------------------------------------------
        seq, val = _run_smooth_opt(a0, deadline)

        # ----------------------------------------------------------------
        # 4)  Record improvement (val is the true max‑convolution)
        # ----------------------------------------------------------------
        if val < best_score - 1e-9:
            best_score = float(val)
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        # ----------------------------------------------------------------
        # 5)  Occasionally polish the current best with a longer run
        # ----------------------------------------------------------------
        if run_id % 7 == 0 and best_seq is not None:
            # tighten β and run a few extra iterations on the current best
            polished, val2 = _run_smooth_opt(best_seq, deadline, max_iters=80_000,
                                            beta_start=2.0, beta_max=80.0,
                                            eta0=0.4, eta_decay=0.9999,
                                            momentum=0.95)
            if val2 < best_score:
                best_score = float(val2)
                best_seq = polished.copy()
                print(f"[polish] run={run_id} n={best_seq.size} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 6)  Final fine‑tuning on the best candidate (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _run_smooth_opt(
            best_seq,
            deadline,
            max_iters=120_000,
            beta_start=5.0,
            beta_max=120.0,
            eta0=0.3,
            eta_decay=0.99995,
            momentum=0.97,
        )
        # recompute exact external score for reporting
        final_score = evaluate_sequence(final_seq.tolist())
        if final_score < best_score:
            best_score = float(final_score)
            best_seq = final_seq.copy()
        print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    else:
        # ------------------------------------------------------------------
        # Fallback – uniform sequence (should never be needed)
        # ------------------------------------------------------------------
        n = 2000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    # Return as a plain Python list
    return best_seq.tolist()
