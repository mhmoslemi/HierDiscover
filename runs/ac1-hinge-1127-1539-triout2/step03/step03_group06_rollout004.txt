Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

The algorithm is a projected Adam optimiser on a smooth surrogate of
the max‑convolution (log‑sum‑exp).  The surrogate gradually approaches the
true maximum by increasing a temperature parameter β.  Each Adam step is
projected onto the ℓ₁‑simplex of radius √(2 n) (the score is scale‑invariant)
so the iterates stay feasible.  A few inexpensive LP refinements (when
SciPy is available) are applied to promising candidates.

The search runs many short restarts with different lengths, adapts β,
adds occasional jitter and respects a 1000 s wall‑clock budget.
"""

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) self‑convolution of a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(Fa * Fa, fft_len)[:conv_len]
    return conv


def _softmax_surrogate_and_grad(a: np.ndarray, beta: float):
    """
    Smooth surrogate of max(conv) together with its gradient.

    The surrogate is   L(a) = max(conv) + (1/β)·log( Σ exp(β·(conv−max(conv))) )
    whose gradient w.r.t. a is
        g = 2·corr(a , softmax(conv)),
    where corr denotes linear cross‑correlation.
    """
    n = a.shape[0]

    # 1) full convolution a*a
    conv = _full_conv_fft(a)                     # length 2n‑1
    max_c = conv.max()

    # 2) softmax weights (numerically stable)
    shifted = conv - max_c                       # ≤ 0
    exp_beta = np.exp(beta * shifted)            # no overflow
    sum_exp = exp_beta.sum()
    p = exp_beta / sum_exp                       # softmax vector, length 2n‑1

    # 3) gradient = 2 * cross‑correlation(a , p) for lags 0 … n‑1
    #   Linear correlation length = n + (2n‑1) – 1 = 3n‑2.
    #   Zero‑lag corresponds to index (len(p)‑1) = 2n‑2.
    L = 1 << ((3 * n - 2 - 1).bit_length())
    Fa = np.fft.rfft(a, L)
    Fp = np.fft.rfft(p, L)
    corr_full = np.fft.irfft(Fa * np.conj(Fp), L)
    offset = 2 * n - 2
    grad = 2.0 * corr_full[offset:offset + n]

    # surrogate loss (equals max when β→∞)
    loss = max_c + (1.0 / beta) * math.log(sum_exp)

    return loss, grad, max_c


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   max_iters: int = 4000,
                   beta_start: float = 0.5,
                   beta_factor: float = 1.02,
                   eta_start: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max surrogate.
    Returns the best vector seen and its *actual* max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)          # required ℓ₁‑norm (scale‑invariant)
    a = _project_onto_simplex(a, S)

    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    eta = eta_start
    beta = beta_start

    best_local_max = float('inf')
    best_local_seq = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        loss, grad, cur_max = _softmax_surrogate_and_grad(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_onto_simplex(a, S)

        # record best *true* max‑convolution encountered
        if cur_max < best_local_max - 1e-12:
            best_local_max = cur_max
            best_local_seq = a.copy()

        # increase temperature slowly (focuses on the true max)
        beta *= beta_factor

        # mild learning‑rate decay
        eta *= 0.9999

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.008:
            jitter = np.random.randn(n) * a.mean() * 0.004
            a = np.clip(a + jitter, 0.0, None)
            a = _project_onto_simplex(a, S)

        # early exit if time is almost gone
        if time.time() >= deadline - 0.2:
            break

    return best_local_seq, best_local_max


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Dense matrix H (size (2n‑1) × n) such that (H @ g) = f * g.
    Used only for the optional LP refinement (n ≤ 3000).
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5) -> np.ndarray:
    """
    Small linear‑programming refinement (same idea as the baseline).
    Only invoked when SciPy is present and n ≤ 3000.
    """
    if not _SCIPY_AVAILABLE or f.size > 3000:
        return f

    n = f.size
    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        M = np.convolve(f, f).max()
        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg   → minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:  # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        g0_sum = g0.sum()
        if g0_sum == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / g0_sum)

        # convex combination and re‑normalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Main entry point required by the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) attains a score
    below the current best bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 0) Baseline from a possibly‑provided global variable.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                n0 = cand.size
                S0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * S0
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] height_sequence_1 score = {best_score:.6f}, length = {n0}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1) Search over a set of lengths.  Larger n tend to give better
    #    bounds, but each iteration becomes slower.  We cap n at 20000.
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 12000, 16000, 20000]
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ----------------------------------------------------------------
        # Initial vector for this run.
        # ----------------------------------------------------------------
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.4:
            a0 = best_seq[:n].copy()
            a0 += 0.02 * rng.random(n)          # slight jitter
        else:
            # blend a sine‑shape with uniform randomness
            idx = np.arange(n)
            sine = np.sin(math.pi * (idx + 0.5) / n)   # positive on (0,π)
            a0 = 0.7 * sine + 0.3 * rng.random(n)

        # Normalise to required ℓ₁‑norm √(2 n)
        S = math.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # ----------------------------------------------------------------
        # Optimise this start with Adam + soft‑max surrogate.
        # ----------------------------------------------------------------
        seq, est_max = _adam_optimize(a0, deadline,
                                      max_iters=int(8000 / (n / 4000)),
                                      beta_start=0.5,
                                      beta_factor=1.018,
                                      eta_start=0.28)

        # Record improvement (using the *actual* max‑convolution)
        if est_max < best_score:
            best_score = est_max
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} max_conv≈{best_score:.6f}")

        # ----------------------------------------------------------------
        # Optional LP refinement (moderate n only).
        # ----------------------------------------------------------------
        if n <= 2500 and est_max < best_score * 0.97:
            refined = _lp_refine(best_seq, deadline)
            refined_max = np.convolve(refined, refined).max()
            if refined_max < best_score:
                best_score = refined_max
                best_seq = refined.copy()
                print(f"[LP refine] run={run_id} n={n} refined max={best_score:.6f}")

        run_id += 1
        # quick exit if we are very close to the target
        if best_score < 1.505 and time.time() < deadline - 30:
            # give a tiny extra polishing budget
            extra_deadline = time.time() + 30
            best_seq, best_score = _adam_optimize(best_seq, extra_deadline,
                                                  max_iters=4000,
                                                  beta_start=0.8,
                                                  beta_factor=1.02,
                                                  eta_start=0.15)
            print(f"[polish] improved to max_conv≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final fine‑tuning of the best vector found.
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_deadline = time.time() + min(60.0, deadline - time.time())
        best_seq, _ = _adam_optimize(best_seq, final_deadline,
                                     max_iters=8000,
                                     beta_start=0.9,
                                     beta_factor=1.015,
                                     eta_start=0.12)
        best_seq = _lp_refine(best_seq, final_deadline)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    else:
        # Fallback – uniform distribution (should never be needed)
        n = 2000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    return best_seq.tolist()
