def _optimize_subgrad(a_init, deadline, rng):
    a = a_init.copy()
    n = len(a)
    S = np.sqrt(2*n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(best_a))
    iter_cnt = 0
    eta0 = 0.1
    while time.time() < deadline:
        iter_cnt += 1
        # compute conv and max index
        conv = _full_convolution(a)
        k_max = np.argmax(conv)
        # compute subgrad
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n-1))
        i_high = min(n-1, k_max)
        idx_i = np.arange(i_low, i_high+1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        # step size adaptation: eta = eta0 / sqrt(iter_cnt)
        eta = eta0 / np.sqrt(iter_cnt)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # random perturb occasionally
        if rng.random() < 0.02:

"""
Propose a step‑function (non‑negative vector) that minimizes the
self‑convolution max‑norm used by ``evaluate_sequence``.

Algorithm
---------
*   Work on the simplex Σa = √(2·n) (the objective is scale‑invariant).
*   Initialise either from the previously discovered best sequence
    (``height_sequence_1``) or from a random Dirichlet draw.
*   Run a *smooth* optimisation phase based on a soft‑max
    approximation of the max‑norm.  The gradient is obtained by a
    weighted convolution (FFT).  A simple SGD update with a
    decreasing learning rate and a slowly increasing temperature α
    is used; after each step we project back onto the simplex.
*   Switch to a *sub‑gradient* phase that directly reduces the
    currently maximal convolution entry.  The sub‑gradient of the
    max entry is simply the reversed copy of the vector aligned with
    the worst shift.
*   Occasionally perform a random “mass‑swap’’ move (transfer a
    small fraction of mass from one index to another) to escape
    plateaus.
*   The routine is time‑aware: it spends most of the budget
    refining the best known candidate, but restarts with a fresh
    random vector if no improvement is seen for a while.
*   The best sequence found is finally evaluated with
    ``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)  (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generator
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If ``base_seq`` is supplied and long enough, its leading part (with tiny jitter)
    is used; otherwise a Dirichlet/Uniform sample is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)               # Dirichlet(1) – uniform on simplex.

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Smooth optimisation phase (soft‑max + SGD)
# ----------------------------------------------------------------------


def _optimize_smooth(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Gradient descent on the smooth max‑norm (soft‑max approximation)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters
    lr = 0.3               # start learning rate
    lr_decay = 0.9997
    alpha = 0.5            # temperature – starts smooth
    alpha_mul = 1.005
    swap_prob = 0.06
    swap_frac = 0.02

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- 1. smooth gradient step ----
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        a = a - lr * grad
        a = _project_simplex(a, S)

        # ---- 2. optional random mass‑swap ----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # accept if improvement or with a tiny probability
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand
                    conv = _full_convolution(a)   # recompute for bookkeeping

        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # schedule updates
        lr *= lr_decay
        alpha = min(alpha * alpha_mul, 30.0)

        # early stop if nothing improves for a while
        if it % 4000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation phase (focus on the worst shift)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst entry)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.2                # initial step size constant
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # construct sub‑gradient: g_i = a_{k_max-i} if index in range
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation (helps escape tiny plateaus)
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # very cheap early‑stop test
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (smooth → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth phase for a few seconds, then finish with the
    sub‑gradient phase.  The best intermediate solution is returned.
    """
    # split the allotted time roughly 60 % smooth, 40 % sub‑grad
    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_smooth(a0, smooth_deadline, rng)

    # if we still have time, launch sub‑gradient refinement
    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimizing ``evaluate_sequence``.
    Uses a long refinement of the best known candidate (taken from the
    global ``height_sequence_1`` when available) combined with occasional
    random restarts.
    Returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Try to reuse the previously discovered construction
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    last_improve_time = start_time
    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # ------------------------------------------------------------------
        # Choose problem size
        # ------------------------------------------------------------------
        if best_overall_seq is None or (time.time() - last_improve_time > 30):
            # no recent improvement → fresh random start
            n = rng.integers(800, 5000)
            a0 = _initial_sequence(base_seq, n, rng)
        else:
            # continue refining the current best
            a0 = best_overall_seq.copy()
            n = a0.shape[0]

        # ------------------------------------------------------------------
        # Allocate a slice of the remaining budget for this attempt
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + 15.0, deadline)

        # ------------------------------------------------------------------
        # Optimise (smooth phase → sub‑gradient phase)
        # ------------------------------------------------------------------
        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        # ------------------------------------------------------------------
        # Record global best
        # ------------------------------------------------------------------
        if score < best_overall_score:

            best_overall_seq = a_opt.copy()
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={n} "

    # ------------------------------------------------------------------
    if best_overall_seq is None:

    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
