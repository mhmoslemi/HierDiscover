"""
Hybrid optimisation for the step‑function problem.

The objective to minimise is
    2·n·max(a*a) / (sum(a)**2)

Because the ratio is homogeneous, we fix ∑a = √(2·n) and simply
 minimise the peak of the self‑convolution.  The algorithm consists of
 three stages:

1. **Random initialisation** –  Dirichlet, Pareto or uniform draws on the
   simplex (∑a = √(2·n)).  A few very sparse draws are also tried because
   sparse vectors tend to have a smaller number of overlapping products.

2. **Exponentiated (multiplicative) gradient descent** –  a sub‑gradient of
   the max‑convolution is obtained from the K largest convolution entries
   (K≈20).  The update  a ← a·exp(‑η·g)  keeps non‑negativity; after each
   step the vector is re‑normalised to the required L1‑norm.  Momentum,
   learning‑rate decay and occasional jitter are used to escape plateaus.

3. **Local stochastic hill‑climbing / simulated annealing** –  from the
   best vector found so far a simple move transfers a random fraction of
   mass from one index to another.  Moves are accepted if they improve the
   peak; otherwise they are accepted with probability exp(‑Δ/τ) where τ is
   a temperature that fades geometrically.

The whole search runs for the allotted time (≈1000 s).  The best
sequence discovered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using an FFT (length 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return float(np.max(full_convolution(a)))


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max‑convolution with respect to ``a``.
    For each selected shift k the contribution to the gradient is
    2·a_{k‑j} for all admissible j.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g


# ----------------------------------------------------------------------
# Exponentiated (multiplicative) gradient descent
# ----------------------------------------------------------------------


def exponential_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    top_k: int = 20,
    eta0: float = 0.6,
    beta: float = 0.9,
    decay: float = 0.99993,
    jitter_prob: float = 0.015,
    max_iters: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative gradient descent with momentum.
    Returns the best vector encountered and its maximal convolution value.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)               # target L1‑norm
    cur_best = conv_max(a)
    best_a = a.copy()
    v = np.zeros_like(a)               # momentum buffer
    eta = eta0
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # compute full convolution and its maximum
        conv = full_convolution(a)
        cur_max = float(conv.max())

        # choose the K shifts that dominate the max
        if conv.size <= top_k:
            shifts = np.arange(conv.size)
        else:
            # unsorted top‑K indices
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient from those shifts
        g = _gradient_from_shifts(a, shifts)

        # momentum step
        v = beta * v + (1.0 - beta) * g

        # multiplicative update (always stays ≥0)
        a = a * np.exp(-eta * v)

        # projection onto the simplex ∑a = S
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to break symmetry
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # learning‑rate decay
        eta *= decay

        # record improvement
        if cur_max < cur_best - 1e-12:
            cur_best = cur_max
            best_a = a.copy()
            no_improve = 0
            print(
                f"[exp descent improve] t={time.time():.1f}s n={n} it={it} max={cur_max:.6f}"
            )
        else:
            no_improve += 1
            # if no improvement for a while, accelerate decay
            if no_improve > 5000:
                eta *= 0.99
                no_improve = 0

    return best_a, cur_best


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing / simulated annealing
# ----------------------------------------------------------------------


def hill_climb(
    a_start: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style hill‑climbing.
    Randomly moves a fraction of mass from a donor index to a recipient.
    """
    a = a_start.copy()
    cur_max = conv_max(a)
    best_a = a.copy()
    best_max = cur_max

    # initialise temperature as a modest fraction of the current peak
    temperature = max(0.05 * cur_max, 1e-4)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # pick donor (must have some mass) and recipient
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # move a random fraction of the donor's mass
        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = conv_max(a_new)

        # accept if improvement, else with Metropolis probability
        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(
                    f"[hill climb improve] t={time.time():.1f}s n={n} it={it} max={new_max:.6f}"
                )
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        # geometric cooling
        temperature *= 0.9995

        # occasional jitter
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()
            cur_max = conv_max(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(
                    f"[jitter] t={time.time():.1f}s n={n} it={it} max={cur_max:.6f}"
                )

    return best_a, best_max


# ----------------------------------------------------------------------
# Sampling helpers for diverse starts
# ----------------------------------------------------------------------


def sample_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    """Draw a Dirichlet vector (length‑n) with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc))
    return a * np.sqrt(2.0 * n)


def sample_pareto(n: int, rng: np.random.Generator, alpha: float = 1.5) -> np.ndarray:
    """Draw a heavy‑tailed vector from a Pareto distribution and project onto the simplex."""
    raw = rng.pareto(alpha, size=n) + 1.0
    raw = raw.astype(float)
    raw /= raw.sum()
    return raw * np.sqrt(2.0 * n)


def sample_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform distribution over the simplex."""
    a = np.full(n, 1.0 / n)
    return a * np.sqrt(2.0 * n)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    """

    deadline = time.time() + budget_s - 3.0   # a safety margin

    best_seq = None          # best numpy vector so far
    best_score = float("inf")  # its *external* score (as returned by evaluate_sequence)

    # ------------------------------------------------------------------

                    f"[baseline] height_sequence_1 score={sc:.6f} length={cand.size}"
                )
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter space – a mixture of short, medium and long vectors.
    # ------------------------------------------------------------------
    lengths = [800, 1600, 3200, 6400, 12800]  # up to ~13 k elements

    run_id = 0
    while time.time() < deadline:
        n = int(rng.choice(lengths, p=length_probs))

        # --------------------------------------------------------------
        # Phase 1 – multiplicative gradient descent (the heavy work)
        # --------------------------------------------------------------
        if rng.random() < 0.70:  # majority of the time
            # pick an initialisation method
            init_type = rng.choice(["dirichlet", "pareto", "uniform"], p=[0.5, 0.3, 0.2])

            elif init_type == "pareto":
                alpha = rng.uniform(1.1, 2.5)

            else:

            # random permutation to break any accidental symmetry

            a_opt, max_conv = exponential_descent(
                a0,
                n,
                deadline,
                top_k=rng.integers(15, 30),
                eta0=rng.uniform(0.4, 0.8),
                beta=0.9,
                decay=0.9999,
                jitter_prob=0.02,
                max_iters=20000,
            )
            score = max_conv                     # because of the normalisation Σa = √(2·n)

        # --------------------------------------------------------------
        # Phase 2 – local hill‑climbing (quick refinement)
        # --------------------------------------------------------------
        else:
            if best_seq is not None and best_seq.size == n:

            else:

            a_opt, max_conv = hill_climb(
                start_vec, n, deadline, rng, max_iters=15000
            )
            score = max_conv

        # --------------------------------------------------------------
        # Record a global improvement
        # --------------------------------------------------------------
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_opt.copy()
            print(
                f"[global improve] run={run_id} n={n} score={best_score:.6f}"
            )

        run_id += 1

    # ------------------------------------------------------------------
    # Final polishing – a last hill‑climb on the best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        final_n = best_seq.size
        best_seq, _ = hill_climb(
            best_seq, final_n, deadline, rng, max_iters=30000
        )

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification – the platform will also call evaluate_sequence
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
