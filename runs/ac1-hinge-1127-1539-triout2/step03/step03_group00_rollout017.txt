Let's conceive algorithm:

1. Choose initial n (maybe 4000). Generate initial vector via some distribution that is known to reduce correlation (e.g., Dirichlet with small concentration, i.e., many small entries). Or generate binary vector with p fraction ones as discussed.

2. Compute convolution, find top k peaks (peak indices, sorted). Compute contribution map: for each position i, compute its contributions to each peak (maybe only the max peak). The contribution of i to a particular peak index p is a_i * a_{p-i}*2 (except maybe self). Compute weight_i = sum_{p∈peaks} a_i * a_{p-i}. Use this as a measure of "involvement" in high peaks.

3. At each iteration, select i with probability proportional to weight_i (so indices that heavily contribute to high peaks). Choose j with low contribution (maybe random or lowest weight). Then transfer small mass δ from i to j. Accept if max_b reduces, else accept with probability etc.

def _adam_optimize(a0, deadline, beta_start=0.5, beta_max=1e6):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # hyperparameters
    lr0 = 0.2 * target_sum
    beta = beta_start
    beta_inc = 1.5
    beta_inc_iter = 300
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        t += 1
        m = 0.9 * m + (1 - 0.9) * g
        v = 0.999 * v + (1 - 0.999) * (g ** 2)

        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        step = lr0 / (np.sqrt(v_hat) + eps)

        a -= step * m_hat
        # projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum*0.001/(beta+1), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % beta_inc_iter == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        # early stop if a hasn't changed? not needed.

    return a

def _simulated_annealing_targeted(a0, deadline):
    a = a0.copy()
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score
    T = cur_score * 0.02 + 1e-6
    cooling = 0.9995
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # compute b, find peak
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        max_idx = int(np.argmax(b))
        max_val = float(b[max_idx])

        # contribution map
        i_min = max(0, max_idx - (n-1))
        i_max = min(n-1, max_idx)
        idx_i = np.arange(i_min, i_max + 1)
        idx_j = max_idx - idx_i
        prods = a[idx_i] * a[idx_j]
        contrib = np.zeros(n)
        # double contributions for i!=j
        mask = idx_i != idx_j
        contrib[idx_i[mask]] = 2.0 * prods[mask]
        # diagonal
        contrib[idx_i[~mask]] = prods[~mask]

        # select source i
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            i = int(rng.choice(n, p=probs))
        else:
            i = rng.integers(0, n)
        # select target j random < median
        low_idx = np.where(a <= np.percentile(a, 20))[0]
        if low_idx.size:
            j = int(rng.choice(low_idx))
        else:
            j = rng.integers(0, n)
        if i == j or a[i] <= 1e-14:
            continue
        # move fraction
        frac = rng.random() * 0.2  # up to 20%
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta
        # compute new score
        new_score = _score(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):

            a[j] -= delta
            a[i] += delta

"""
Propose a step‑function (non‑negative sequence) that minimises the
competition objective.  The search combines several ideas:

* Random draws on the ℓ¹‑simplex (different families of distributions).
* An Adam‑style projected gradient descent on a smooth log‑sum‑exp
  approximation of the convolution maximum.  The smoothing parameter β
  is annealed to focus on the true maximum.
* A targeted simulated‑annealing that preferentially moves mass from
  coefficents heavily contributing to the current peak of the
  self‑convolution.
* Progressive up‑scaling of the best solution to larger dimensions,
  re‑optimising after each scaling step.

The routine respects the 1000 s budget, keeps the best sequence seen,
and returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast helpers
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation (just the max of a*a)."""
    return _max_convolution_fft(a)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    choice = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if choice == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif choice == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                   # Dirichlet sample
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    else:                               # uniform plus tiny jitter
        a = np.full(n, target_sum / n)

    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


def _interpolate_sequence(a: np.ndarray, n_target: int) -> np.ndarray:
    """
    Linear interpolation of an existing sequence to a new length.
    The result is re‑projected onto the appropriate ℓ¹‑simplex.
    """
    n0 = a.shape[0]
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_target)
    b = np.interp(x1, x0, a)
    b = _project_to_simplex(b, np.sqrt(2.0 * n_target))
    return b


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (grad, conv) where
        conv = a*a (full convolution, length 2·n‑1)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) )
              = 2·(a ⋆ p)  with p_k ∝ exp(β·conv_k)
    """
    n = a.shape[0]

    # ---- convolution a*a ---------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities ---------------------------------------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))          # stable exp
    p = exps / np.sum(exps)

    # ---- gradient = 2·(a ⋆ p) ----------------------------------------
    # compute correlation of a with p (a reversed)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    corr = np.fft.irfft(a_rev_fft * p_fft, n=size2)[:conv_len]
    grad = 2.0 * corr[n - 1: n - 1 + n]

    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp smooth approximation of max(a*a).
    The iterate is re‑projected onto the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # hyper‑parameters
    beta = 0.5
    beta_max = 1e6
    lr_base = target_sum * 0.15          # base learning‑rate magnitude
    eps = 1e-8
    beta_inc = 1.6
    beta_inc_iter = 250

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr_base / (np.sqrt(v_hat) + eps)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # anneal β
        if t % beta_inc_iter == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        # occasional jitter to avoid plateaus
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing that preferentially moves mass away from the
    coefficients that contribute most to the current convolution peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score

    # initialise temperature as a small fraction of the current score
    T = cur_score * 0.02 + 1e-8
    cooling = 0.9996

    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- full convolution (needed to locate the peak) ------------
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_idx = int(np.argmax(b))

        # ----- contribution of each index to the peak ------------------
        i_min = max(0, max_idx - (n - 1))
        i_max = min(n - 1, max_idx)
        idx_i = np.arange(i_min, i_max + 1)
        idx_j = max_idx - idx_i
        prod = a[idx_i] * a[idx_j]

        contrib = np.zeros(n)
        mask = idx_i != idx_j
        contrib[idx_i[mask]] = 2.0 * prod[mask]          # off‑diagonal pairs
        contrib[idx_i[~mask]] = prod[~mask]             # diagonal (i=j)

        # ----- pick a source index weighted by contribution -----------
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i = int(rng.choice(n, p=prob))
        else:
            i = rng.integers(0, n)

        # ----- pick a destination index (prefer small coefficients) ---
        low_mask = a <= np.percentile(a, 20)
        low_idxs = np.nonzero(low_mask)[0]
        if low_idxs.size:
            j = int(rng.choice(low_idxs))
        else:
            j = rng.integers(0, n)

        if i == j or a[i] <= 1e-14:
            continue

        # ----- move a random fraction (up to 15 %) --------------------
        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score

        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        # temperature decay
        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    competition objective.  The routine honours the time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # leave a small safety margin before the hard deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction exists, try to refine it first
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        n0 = seq0.size
        target0 = np.sqrt(2.0 * n0)
        seq0 = _project_to_simplex(seq0, target0)
        # quick refinement
        if time.time() < deadline:
            seq0 = _adam_optimize(seq0, deadline)
        if time.time() < deadline:
            seq0 = _targeted_simulated_annealing(seq0, deadline)
        sc0 = _score(seq0)
        if sc0 < best_score:
            best_score = sc0
            best_seq = seq0.copy()
            print(f"[{time.time():.1f}] Refined global start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – we gradually increase the dimension
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000]

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # allocate a fair share of the remaining budget to this length
        remaining = deadline - time.time()
        remaining_lengths = len(lengths) - idx
        per_len_budget = min(remaining / remaining_lengths, 180.0)  # at most 3 min
        length_deadline = time.time() + per_len_budget

        # ------------------------------------------------------------------
        # Phase A – generate a few random seeds
        # ------------------------------------------------------------------
        candidates = []
        random_phase_end = min(length_deadline, time.time() + min(6.0, per_len_budget / 3))
        while time.time() < random_phase_end:
            a0 = _random_initial_sequence(n, rng)
            candidates.append(a0)
            sc = _score(a0)
            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase B – upscale the best solution from the previous size (if any)
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_up = _interpolate_sequence(best_seq, n)
            candidates.append(a_up)

        # keep only the most promising few seeds
        scored = [( _score(c), c ) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for (_, c) in scored[:5]]

        # ------------------------------------------------------------------
        # Phase C – refined optimisation (Adam + targeted SA)
        # ------------------------------------------------------------------
        for cand in top_candidates:
            if time.time() >= length_deadline:
                break
            # split the remaining time for this candidate
            sub_deadline = min(length_deadline, time.time() + (length_deadline - time.time()) * 0.8)

            a_opt = _adam_optimize(cand, sub_deadline)
            if time.time() >= length_deadline:
                # if we ran out of time during Adam, skip SA
                sc = _score(a_opt)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"[{time.time():.1f}] New best after Adam n={n} score={best_score:.6f}")
                break

            a_opt = _targeted_simulated_annealing(a_opt, length_deadline)

            sc = _score(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # keep the best sequence of this length for possible up‑scaling
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            # also store a version resized to the current length (helps the next step)
            best_seq = _interpolate_sequence(best_seq, n)

    # ------------------------------------------------------------------
    # Fallback: uniform distribution (should hardly ever be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    # Trim trailing near‑zeros for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} length = {best_seq.size}")
    return best_seq.tolist()
