"""
Improved step‑function search.

The goal is to minimise
    score = 2·n·max(conv(a,a)) / (sum(a)^2)

which, after normalising sum(a)=√(2n), is simply the maximal entry of the
self‑convolution.  The optimiser runs many independent restarts at several
lengths, using a mirror‑descent (exponentiated‑gradient) update with momentum,
top‑K surrogate gradients and occasional jitter.  Initial vectors are drawn
from several simple families (random, symmetric, exponential, sinusoidal,
triangular) to avoid being trapped in a single basin.  The best candidate
found before the time budget expires is returned.

The code respects the 1000 s wall‑clock limit, prints occasional progress,
and contains only top‑level functions (no closures or lambdas) as required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray):
    """Full convolution of a with itself (FFT) and its maximum entry."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray):
    """
    Sub‑gradient of the sum of the selected convolution entries.
    For each selected shift k we add 2·a_{k‑i} to component i whenever
    the index lies inside the vector.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        # each valid pair contributes 2·a[partner]
        g[valid] += 2.0 * a[partner[valid]]
    return g


# ----------------------------------------------------------------------
# Initialisation families (all produce non‑negative vectors)
# ----------------------------------------------------------------------


def _init_random(n, rng):
    """Plain i.i.d. uniform on [0,1)."""
    return rng.random(n)


def _init_symmetric_random(n, rng):
    """Random vector forced to be symmetric."""
    half = rng.random((n + 1) // 2)
    if n % 2 == 0:
        a = np.concatenate([half, half[::-1]])
    else:
        a = np.concatenate([half, half[-2::-1]])
    return a


def _init_exponential(n, rng, scale=0.005):
    """Decaying exponential from the centre."""
    centre = (n - 1) / 2.0
    i = np.arange(n)
    a = np.exp(-scale * np.abs(i - centre))
    return a


def _init_sinusoid(n, rng):
    """Positive half‑cosine shape."""
    i = np.arange(n)
    a = 0.5 * (1.0 + np.cos(math.pi * (i - (n - 1) / 2) / (n // 2 + 1)))
    a[a < 0] = 0.0
    return a


def _init_triangular(n, rng):
    """Simple triangle (peak in the centre)."""
    i = np.arange(n)
    a = 1.0 - np.abs(i - (n - 1) / 2) / ((n + 1) / 2)
    a[a < 0] = 0.0
    return a


_INITIALISERS = [
    _init_random,
    _init_symmetric_random,
    _init_exponential,
    _init_sinusoid,
    _init_triangular,
]


# ----------------------------------------------------------------------
# Optimisation core
# ----------------------------------------------------------------------


def _run_optimize_one_start(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            rng: np.random.Generator,
                            top_k_start: int = 5,
                            eta0: float = 0.6,
                            beta: float = 0.95,
                            decay: float = 0.99997,
                            jitter_prob: float = 0.015,
                            jitter_scale: float = 0.01,
                            max_iters: int = 250_000):
    """
    Mirror‑descent (exponentiated‑gradient) on the max‑convolution surrogate.
    Returns the best vector seen and its surrogate max‑convolution value.
    """
    a = a0.astype(np.float64, copy=True)

    S = math.sqrt(2.0 * n)          # the fixed ℓ₁‑norm
    total = a.sum()
    if total == 0.0:
        a.fill(S / n)
    else:
        a *= (S / total)

    best_max = float("inf")
    best_seq = a.copy()

    v = np.zeros_like(a)            # momentum buffer
    eta = eta0
    it = 0
    last_improve = 0

    # pre‑allocate a buffer for the exponent to avoid repeated allocations
    while time.time() < deadline and it < max_iters:
        it += 1

        # convolution and its maximum
        conv, cur_max, _ = _conv_max(a)

        # decide which shifts to use for the sub‑gradient
        if conv.size <= top_k_start:
            shifts = np.arange(conv.size)
        else:
            # top‑k (unsorted) indices – more efficient than full sort
            top_k = top_k_start
            # if we have not improved for a while increase top‑k modestly
            if it - last_improve > 2_000 and top_k < 30:
                top_k = min(30, top_k * 2)
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient for the selected shifts
        g = _gradient_from_shifts(a, shifts)

        # momentum update (exponential moving average)
        v = beta * v + (1.0 - beta) * g

        # exponentiated update – clamp exponent to avoid overflow/underflow
        exp_factor = np.exp(-eta * np.clip(v, -50.0, 50.0))
        a *= exp_factor

        # renormalise to the required ℓ₁‑norm
        tot = a.sum()
        if tot == 0.0 or not np.isfinite(tot):
            a.fill(S / n)
        else:
            a *= (S / tot)

        # random jitter to jump out of tiny plateaus
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a.fill(S / n)
            else:
                a *= (S / a_sum)

        # decay the step‑size
        eta *= decay

        # record improvement
        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_seq = a.copy()
            last_improve = it

        # early exit when deadline is very near
        if time.time() >= deadline - 0.05:
            break

    return best_seq, best_max


def _local_refine(seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator):
    """
    A short, aggressive refinement of the given sequence.
    Uses a few hundred exponentiated‑gradient steps with a larger
    top‑K and a higher jitter probability.
    """
    a0 = seq.copy()
    refined, _ = _run_optimize_one_start(
        a0,
        n,
        deadline,
        rng,
        top_k_start=10,
        eta0=0.8,
        beta=0.9,
        decay=0.99990,
        jitter_prob=0.03,
        jitter_scale=0.02,
        max_iters=100_000,
    )
    return refined


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1)  Use the supplied global construction if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                n0 = cand.size
                S0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * S0
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] height_sequence_1 score = {best_score:.6f}, length = {n0}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 2)  Main multi‑restart optimisation loop.
    # ------------------------------------------------------------------
    # A set of lengths that balances quality and speed.
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000, 24000]
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ----------------------------------------------------------------
        # 2a) initialise a starting vector (various families)
        # ----------------------------------------------------------------
        init_fn = rng.choice(_INITIALISERS)
        a0 = init_fn(n, rng)

        # Normalise to the required ℓ₁‑norm √(2 n)
        S = math.sqrt(2.0 * n)
        total = a0.sum()
        if total == 0.0:
            a0.fill(S / n)
        else:
            a0 = a0 / total * S

        # ----------------------------------------------------------------
        # 2b) run one optimisation restart
        # ----------------------------------------------------------------
        seq, surrogate_max = _run_optimize_one_start(
            a0,
            n,
            deadline,
            rng,
            top_k_start=5,
            eta0=0.6,
            beta=0.95,
            decay=0.99997,
            jitter_prob=0.015,
            jitter_scale=0.01,
            max_iters=150_000,
        )

        # surrogate_max equals the external score for the normalised vector
        if surrogate_max < best_score:
            best_score = surrogate_max
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

            # optional short local refinement when a clear improvement appears
            if time.time() < deadline - 1.0:
                refined = _local_refine(best_seq, n, deadline, rng)
                conv, refined_max, _ = _conv_max(refined)
                if refined_max < best_score:
                    best_score = refined_max
                    best_seq = refined.copy()
                    print(f"[local refine] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 3)  Final polish on the best candidate (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_n = best_seq.size
        # a few more gradient steps with tighter parameters
        best_seq, _ = _run_optimize_one_start(
            best_seq,
            final_n,
            deadline,
            rng,
            top_k_start=8,
            eta0=0.5,
            beta=0.9,
            decay=0.99995,
            jitter_prob=0.01,
            jitter_scale=0.015,
            max_iters=200_000,
        )
        # final exact evaluation
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best score = {best_score:.6f}, length = {final_n}")
    else:
        # ----------------------------------------------------------------
        # Fallback – uniform sequence (guaranteed validity)
        # ----------------------------------------------------------------
        n = 2000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq.tolist()
