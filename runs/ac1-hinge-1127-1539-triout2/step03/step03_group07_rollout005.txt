def projected_adam(a0, n, deadline, gamma_start=5.0, gamma_end=50.0, max_iters=30000, lr=0.1):

    a = a0 copy.

    m = np.zeros_like(a); v = np.zeros_like(a)

    beta1=0.9, beta2=0.999

    t = 0

    while time.time() < deadline and t < max_iters:

        t += 1

        # compute conv
        conv = compute_conv(a)  # length 2n-1
        # compute gamma schedule
        gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iters)
        # compute w_i = exp(gamma*conv_i) / sum
        maxc = np.max(gamma*conv) (to avoid overflow)
        expc = np.exp(gamma*conv - maxc)
        w = expc / np.sum(expc)
        # gradient = 2 * correlate(w, a)
        g = 2 * np.correlate(w, a, mode='valid')
        # Adam update
        m = beta1*m + (1-beta1) * g
        v = beta2*v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # projection
        a = np.maximum(a, 0)
        s = a.sum()
        if s == 0:
            a = np.full(n, np.sqrt(2.0 * n) / n)
        else:
            a *= (np.sqrt(2.0 * n) / s)

def run_adam_descent(a0, n, deadline, gamma_start=5.0, gamma_end=40.0, lr=0.5, max_iters=20000):

    a = a0.copy()
    best_a = a.copy()
    best_val = conv_max(a)  # or compute directly conv
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    while time.time() < deadline and t < max_iters:
        t+=1
        # conv
        conv = compute_conv_fft(a)
        cur_max = conv.max()
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            print(f"[adam improve] time={time.time():.1f}s n={n} t={t} max={cur_max:.6f}")

        # gamma schedule
        gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iters)
        # compute w via softmax
        scaled = gamma * conv
        # shift for numerical stability
        max_scaled = scaled.max()
        expw = np.exp(scaled - max_scaled)
        w = expw / expw.sum()
        # gradient
        g = gradient_corr_fft(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        mhat = m / (1 - beta1**t)
        vhat = v / (1 - beta2**t)
        a = a - lr * mhat / (np.sqrt(vhat) + eps)

        # projection onto non-negatives and L1 norm sqrt(2n)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, np.sqrt(2.0*n) / n)
        else:
            a *= (np.sqrt(2.0*n) / total)

        # occasional jitter
        if rng.random() < 0.005:
            a += rng.normal(scale=a.mean()*0.01, size=n)
            a = np.clip(a, 0.0, None)
            a *= (np.sqrt(2.0*n) / a.sum())
    # after loop, return best_a, best_val
    return best_a, best_val

"""
Propose a candidate step‑function (list of non‑negative heights) that minimises

    2·n·max(a*a) / (sum(a)^2)

The algorithm consists of three stages:

1. **Smooth‑max (log‑sum‑exp) surrogate optimisation**.
   The hard max of the self‑convolution is replaced by a differentiable
   soft‑max with a temperature γ that is increased during the run.
   Gradients are obtained analytically: for a given weight vector w (the
   soft‑max of the convolution) the gradient equals
        2 * correlate(w, a) .
   An Adam optimiser with projection onto the feasible set
   (a_i ≥ 0 and Σa_i = √(2·n)) performs the updates.  Symmetrisation of the
   vector is applied every few iterations because the optimal solution is
   known to be symmetric.

2. **Local stochastic hill‑climbing**.
   After the smooth‑max phase the best vector found is refined by moving a
   random fraction of mass between two coordinates.  Worse moves are accepted
   with a temperature schedule (simulated‑annealing style).

3. **Multi‑restart global search**.
   The procedure is run many times with different random initialisations,
   lengths (500 … 8000) and hyper‑parameters until the time budget (≈1000 s)
   is exhausted.  The best sequence overall is returned.

All operations respect the non‑negativity and clipping performed by
`evaluate_sequence`; the final score printed at the end is the external
score (max of the self‑convolution) because the L1‑norm is kept equal to
√(2·n).  The code uses only NumPy and the supplied `evaluate_sequence`
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / correlation utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def gradient_corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max surrogate:
        g = 2 * correlate(w, a)   (valid mode, length = len(a))

    Implemented via FFT to stay O(n log n).
    """
    n = a.shape[0]
    # correlation = convolution of w with reversed a
    a_rev = a[::-1]
    conv_len = len(w) + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    fft_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(Fw * Fa_rev, n=fft_len)
    start = n - 1                       # valid part starts here
    g = corr_full[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  max_iters: int = 25000,
                  gamma_start: float = 5.0,
                  gamma_end: float = 40.0,
                  lr: float = 0.5,
                  symmetrize_every: int = 10) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate with projection onto
    {a_i ≥ 0, Σa_i = √(2·n)}.  Returns the best vector seen and its
    (hard) max‑convolution value.
    """
    a = a0.copy()
    sqrt2n = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_val = np.max(conv_self_fft(a))

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # --- current convolution and soft‑max weights ---------------------------------
        conv = conv_self_fft(a)                     # length 2n‑1
        cur_max = conv.max()
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            print(f"[adam improve] time={time.time():.1f}s n={n} iter={t} max={cur_max:.6f}")

        # gamma ramps from gamma_start to gamma_end
        gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iters)

        # soft‑max over the convolution entries (numerically stable)
        scaled = gamma * conv
        shift = scaled.max()
        exp_w = np.exp(scaled - shift)
        w = exp_w / exp_w.sum()                     # shape (2n‑1,)

        # --- gradient (smooth) ---------------------------------------------------------
        g = gradient_corr_fft(a, w)                 # shape (n,)

        # --- Adam update ---------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- projection (non‑negativity + L1 normalisation) ----------------------------
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a.fill(sqrt2n / n)
        else:
            a *= (sqrt2n / total)

        # --- occasional symmetrisation (helps because optimum is symmetric) ----------
        if t % symmetrize_every == 0:
            a = (a + a[::-1]) * 0.5
            a *= (sqrt2n / a.sum())

        # --- occasional jitter to avoid stagnation ------------------------------------
        if rng.random() < 0.004:
            a += rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a, 0.0, None)
            a *= (sqrt2n / a.sum())

    return best_a, best_val


def _local_hill_climb(a_start: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style refinement.
    Moves a random fraction of mass from one index to another.
    Returns the best vector encountered and its hard max‑convolution.
    """
    a = a_start.copy()
    cur_max = np.max(conv_self_fft(a))
    best_a = a.copy()
    best_max = cur_max

    # temperature starts as a modest fraction of the current max
    temperature = max(0.05 * cur_max, 1e-4)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = np.max(conv_self_fft(a_new))

        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(f"[hill climb improve] time={time.time():.1f}s n={n} iter={it} max={new_max:.6f}")
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        temperature *= 0.9995

        # tiny random jitter every now and then
        if rng.random() < 0.001:
            a += rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a, 0.0, None)
            a *= (np.sqrt(2.0 * n) / a.sum())
            cur_max = np.max(conv_self_fft(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[hill climb jitter] time={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}")

    return best_a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low score.
    Returns the list of heights.  The routine respects the 1000 s budget,
    runs many independent restarts of Adam‑based smooth‑max optimisation
    followed by a short stochastic hill‑climbing refinement, and finally
    reports the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the final score printout
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional: evaluate a few simple Gaussian initialisations as cheap baselines
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000]
    sigma_factors = [0.20, 0.30, 0.40, 0.50, 0.60]

    for n in lengths:
        mid = (n - 1) / 2.0
        for sf in sigma_factors:
            sigma = sf * n
            x = np.arange(n, dtype=float)
            a = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a *= (np.sqrt(2.0 * n) / a_sum)
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[gaussian init] n={n} sigma_factor={sf:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    # ------------------------------------------------------------------
    run_id = 0
    while time.time() < deadline:
        # pick a size, bias a little towards the larger candidates
        n = int(rng.choice(lengths,
                           p=np.array([0.18, 0.18, 0.24, 0.24, 0.16])))
        sqrt2n = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on smooth‑max surrogate
        # ------------------------------------------------------------------
        # Random Dirichlet start; sometimes initialise symmetrically
        conc = rng.uniform(0.05, 1.5)
        if rng.random() < 0.4:                     # symmetric initialisation
            half = n // 2
            a_half = rng.dirichlet(np.full(half, conc))
            if n % 2 == 0:
                a0 = np.concatenate([a_half, a_half[::-1]])
            else:
                centre = rng.dirichlet([conc])[0]
                a0 = np.concatenate([a_half, [centre], a_half[::-1]])
        else:
            a0 = rng.dirichlet(np.full(n, conc))

        a0 *= (sqrt2n / a0.sum())

        a_opt, max_conv = _adam_descent(a0, n, deadline, rng,
                                        max_iters=20000,
                                        gamma_start=5.0,
                                        gamma_end=45.0,
                                        lr=0.6,
                                        symmetrize_every=12)

        # ------------------------------------------------------------------
        # Phase 2 – local stochastic hill‑climbing refinement
        # ------------------------------------------------------------------
        a_opt, max_conv = _local_hill_climb(a_opt, n, deadline, rng,
                                            max_iters=15000)

        # ------------------------------------------------------------------
        # Record global best
        # ------------------------------------------------------------------
        if max_conv < best_score - 1e-12:
            best_score = max_conv
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback – in the (very unlikely) case no candidate was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
