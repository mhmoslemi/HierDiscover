def _hard_optimize(a, deadline, rng):
    a_cur = a.copy()
    n = a_cur.shape[0]
    S = math.sqrt(2.0 * n)
    best_a = a_cur.copy()
    best_max = np.max(_full_convolution(a_cur))
    lr = 0.5
    lr_decay = 0.999
    iter_cnt = 0
    last_improve = 0
    while time.time() < deadline:
        iter_cnt += 1
        conv = _full_convolution(a_cur)
        k_max = int(np.argmax(conv))
        # gradient
        grad = np.zeros_like(a_cur)
        # vectorized: indices where k_max - i in range
        # i from max(0, k_max - (n-1)) to min(n-1, k_max)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # for i in [i_start, i_end]
        # grad[i] = 2 * a_cur[k_max - i]
        grad[i_start:i_end+1] = 2.0 * a_cur[k_max - np.arange(i_start, i_end+1)]
        a_new = a_cur - lr * grad
        a_new = _project_simplex(a_new, S)
        cur_max = np.max(_full_convolution(a_new))
        if cur_max < best_max:
            best_max = cur_max
            best_a = a_new.copy()
            last_improve = iter_cnt
        a_cur = a_new
        lr *= lr_decay
        if iter_cnt - last_improve > 1000:
            break
    return best_a, best_max

"""
Propose a step‑function (list of non‑negative heights) that minimises the
evaluation score 2·n·max(a∗a)/(sum(a)²).

Algorithm
---------
*   Many short independent optimisation runs are performed within the time
    budget.  For each run a random problem size *n* (≈1 k–8 k) and a random
    initialisation (uniform, Dirichlet, or a √‑decay shape a_i∝1/√i) are
    selected.
*   **Phase 1 – smoothed max gradient descent**  
    The true max‑norm is approximated by a soft‑max with a temperature
    parameter *α*.  In each iteration
        1.   The full self‑convolution is computed by FFT.
        2.   Soft‑max weights w_i = exp(α·c_i)/∑exp(α·c) are built.
        3.   The gradient ∇ = 2·(a * w) (correlation) is obtained via FFT.
        4.   A projected gradient step with a decaying learning rate is taken.
    The temperature *α* is slowly increased so that the landscape sharpens
    from smooth to the true max‑norm.
*   **Phase 2 – hard‑max (sub‑gradient) refinement**  
    Starting from the best candidate of Phase 1, the index *k* of the
    maximal convolution entry is identified.  The exact sub‑gradient of the
    max‑norm is g_i = 2·a_{k−i} (zero outside the array).  A projected
    gradient step with a decreasing step size is performed until no
    improvement is seen.
*   The best sequence over all runs (including Phase 2 refinements) is
    returned.  The routine respects the 1000 s budget, prints progress and
    finally evaluates the external score via the provided `evaluate_sequence`.

The code uses only NumPy and the already‑imported `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient term g_j = 2 * Σ_i w_{i+j} * a_i.
    Both a (len n) and w (len 2n-1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse for correlation
    conv_len = n + (2 * n - 1) - 1       # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # indices that correspond to the required gradient
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max smoothing
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights from a convolution vector.
    w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Gradient‑descent steps
# ----------------------------------------------------------------------


def _gradient_step_smooth(a: np.ndarray, lr: float, alpha: float):
    """
    One projected gradient step for the smoothed max‑norm.
    Returns the updated vector and the true maximal convolution value.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    a_new = _project_simplex(a_new, math.sqrt(2.0 * n))
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


def _gradient_step_hard(a: np.ndarray, lr: float):
    """
    One projected sub‑gradient step for the true max‑norm.
    Returns the updated vector and the new maximal convolution value.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k = int(np.argmax(conv))                     # index of the current maximum
    # build the sub‑gradient g_i = 2 * a_{k‑i} (zero outside the array)
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    # i runs from i_start … i_end  (inclusive)
    idx = np.arange(i_start, i_end + 1)
    grad[idx] = 2.0 * a[k - idx]
    a_new = a - lr * grad
    a_new = _project_simplex(a_new, math.sqrt(2.0 * n))
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(n: int,
                     rng: np.random.Generator,
                     mode: str = "rand",
                     base_seq: np.ndarray | None = None) -> np.ndarray:
    """
    Produce a non‑negative vector of length *n* whose L¹‑norm equals √(2·n).
    Three simple initialisation schemes are supported:
        * "rand"   – Dirichlet(1) (uniform over the simplex)
        * "sqrt"   – a_i ∝ 1/√(i+1)  (captures the analytically motivated shape)
        * "uniform"– constant values (for completeness)
    If *base_seq* is supplied and long enough, its leading part is used with
    a tiny jitter; this allows reuse of a previously discovered good sequence.
    """
    S = math.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    elif mode == "sqrt":
        a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    elif mode == "uniform":
        a = np.full(n, 1.0, dtype=float)
    else:                      # random Dirichlet
        a = rng.random(n)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0, dtype=float)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Local optimiser – two phases
# ----------------------------------------------------------------------


def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a two‑phase local optimisation starting from *a0*.
    Phase 1 uses a smooth‑max surrogate; Phase 2 refines the result with a
    hard‑max sub‑gradient step.  Returns the best sequence encountered and its
    true maximal self‑convolution value.
    """
    # -------------------------------------------------
    # Phase 1 – smoothed objective (soft‑max)
    # -------------------------------------------------
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    lr = 0.35
    lr_decay = 0.9996
    alpha = 1.0
    alpha_mul = 1.002

    last_improve = 0
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1
        a, cur_max = _gradient_step_smooth(a, lr, alpha)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = iter_cnt

        # occasional random mass‑swap (helps escape plateaus)
        if rng.random() < 0.06:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.02 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis‑style acceptance
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.04, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = iter_cnt

        lr *= lr_decay
        alpha *= alpha_mul

        # stop if no improvement for a while
        if iter_cnt - last_improve > 2500:
            break

    # -------------------------------------------------
    # Phase 2 – hard max (sub‑gradient) refinement
    # -------------------------------------------------
    a = best_a.copy()
    lr_h = 0.7
    lr_decay_h = 0.9997
    last_improve_h = 0
    iter_cnt_h = 0

    while time.time() < deadline:
        iter_cnt_h += 1
        a, cur_max = _gradient_step_hard(a, lr_h)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve_h = iter_cnt_h

        lr_h *= lr_decay_h
        if iter_cnt_h - last_improve_h > 2000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Entry point expected by the evaluation harness.
    Runs many independent short optimisation attempts and returns the best
    sequence found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # optional warm‑start from a previously discovered sequence
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Main loop – many short independent attempts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # pick a problem size – explore a broad range
        n = rng.integers(1000, 8000)

        # choose an initialisation mode at random
        mode = rng.choice(['rand', 'sqrt', 'uniform'])

        # initialise a candidate
        a0 = _initial_sequence(n, rng, mode=mode, base_seq=base_seq)

        # allocate a time slice for this attempt (≈5 s)
        slice_deadline = min(time.time() + 5.0, deadline)

        # run the two‑phase optimiser
        a_opt, score = _optimize_local(a0, slice_deadline, rng)

        # keep the overall best
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} init={mode} "
                  f"new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the external evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – should never happen
        return [1.0]

    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal max = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
