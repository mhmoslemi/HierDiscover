"""
Propose a step‑function (non‑negative sequence) with a low maximal
self‑convolution entry.

Algorithm
---------

*   Work with a probability‑simplex representation:  the evaluation
    score equals the maximal entry of the self‑convolution, so the
    ℓ₁‑norm of the vector is irrelevant (the code normalises to √(2·n)).

*   For a given length *n* we start from a random feasible vector
    (Dirichlet‑type, projected onto the simplex).

*   **Linear‑programming improvement** – a classic “active‑set” step:
        given the current vector *f* with maximal self‑convolution *M*,
        solve the LP  

            maximise   Σ g_i
            subject to g_i ≥ 0,   (f * g)_k ≤ M   for all k

        The LP finds a direction *g* that can be added without increasing
        the current maximum.  After scaling *g* to the required ℓ₁‑norm
        we move a fraction *t* (default 0.5) towards *g*.
        The process is repeated until no improvement is observed or the
        time budget for this restart runs out.

*   **Fine‑tuning** – once the LP phase stagnates we run a short
    projected‑gradient descent (soft‑max surrogate of the ∞‑norm) with
    occasional jitter and simulated‑annealing moves.  This is the
    same optimiser that was used in the previous state‑of‑the‑art run.

*   The whole optimisation is performed many independent restarts
    (different lengths, random seeds).  The best sequence discovered
    before the overall 1000 s budget expires is returned.

The LP is solved with `scipy.optimize.linprog` (method “highs”).  If
SciPy is unavailable the code falls back to the pure gradient method.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (LP solver)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog  # type: ignore
    _HAS_LP = True
except Exception:      # pragma: no cover
    _HAS_LP = False
    linprog = None

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                    # fallback – uniform distribution
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Full convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

# ----------------------------------------------------------------------
# Initialise a feasible vector of length n with ℓ₁‑norm = sqrt(2·n)
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with sum √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                                 # Dirichlet(1)
    a = _project_to_simplex(a, S)
    return a

# ----------------------------------------------------------------------
# Soft‑max (or hard‑max) gradient update for the smooth surrogate
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float):
    """
    One update step:
        – compute b = a*a,
        – soft‑max weights w ∝ exp(b/tau),
        – gradient g_i = 2 Σ_j w_{i+j} a_j,
        – a ← a – η·g and project onto the ℓ₁‑simplex.
    Returns (a_new, b_new, max(b_new)).
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau <= 1e-12:                     # hard max (sub‑gradient)
        k = np.argmax(b)
        # gradient consists of the slice of a that pairs to index k
        g = np.zeros_like(a)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:                   # diagonal term counted once
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]
    else:                                 # smooth soft‑max
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # length 2n‑1
        # compute cross‑correlation a ⋆ w
        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fa_rev = np.fft.rfft(a_rev, n=m)
        Fw = np.fft.rfft(w, n=m)
        conv = np.fft.irfft(Fa_rev * Fw, n=m)[:len(w) + n - 1]
        g = 2.0 * conv[n - 1:n - 1 + n]

    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    b_new = _full_convolution_fft(a_proj)
    max_new = np.max(b_new)

    return a_proj, b_new, max_new


# ----------------------------------------------------------------------
# Local optimiser based on projected soft‑max gradient + occasional SA
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected soft‑max gradient descent with occasional random moves."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters (tuned empirically)
    eta = 0.6
    eta_decay = 0.9997
    move_prob = 0.12
    epsilon_move = 0.04
    temp = cur_max * 0.1
    temp_decay = 0.9998
    jitter_prob = 0.01
    jitter_scale = 0.02
    tau = cur_max
    tau_decay = 0.9995

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) Simulated‑annealing mass swap
        # ------------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    b_cand = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_cand)
                    accept = False
                    if cand_max <= cur_max:
                        accept = True
                    else:
                        prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                        if rng.random() < prob:
                            accept = True
                    if accept:
                        a = a_candidate
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                    temp *= temp_decay
            continue

        # ------------------------------------------------------------------
        # 2) Projected soft‑max gradient step
        # ------------------------------------------------------------------
        a, b, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # ------------------------------------------------------------------
        # 3) Occasional jitter
        # ------------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _full_convolution_fft(a)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # decay schedules
        eta *= eta_decay
        tau *= tau_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# LP‑based improvement step
# ----------------------------------------------------------------------
def _optimize_lp_local(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP:
        maximise Σ g_i  subject to  g ≥ 0  and  (a * g) ≤ M,
    where M = max(a*a).  Mix the solution back into a and repeat.
    Returns the best vector seen and its maximal self‑convolution entry.
    """
    if not _HAS_LP:
        # SciPy not available – fall back to gradient method
        return _optimize_local(a0, deadline, rng)

    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initial record
    cur_max = np.max(_full_convolution_fft(a))
    best_a = a.copy()
    best_max = cur_max

    # parameters
    mix_frac = 0.5          # primary mixing weight
    mix_small = 0.2         # fallback mixing weight
    min_improve = 1e-6

    while time.time() < deadline:
        # build constraint matrix A where A @ g = a * g
        # A has shape (2n‑1) × n, each column is a shifted copy of a
        A = np.empty((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        M = cur_max
        c = -np.ones(n)                         # maximise Σ g_i  → minimise –Σ g_i
        b_ub = np.full(2 * n - 1, M)

        # solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            break

        g0 = res.x
        if np.sum(g0) <= 0:
            break

        # scale g to the required ℓ₁‑norm
        g = g0 * (S / np.sum(g0))

        # first try a relatively aggressive mixing
        a_new = (1 - mix_frac) * a + mix_frac * g
        max_new = np.max(_full_convolution_fft(a_new))

        if max_new + min_improve < cur_max:
            a = a_new
            cur_max = max_new
        else:
            # try a more conservative step
            a_new = (1 - mix_small) * a + mix_small * g
            max_new = np.max(_full_convolution_fft(a_new))
            if max_new + min_improve < cur_max:
                a = a_new
                cur_max = max_new
            else:
                # no improvement – try a small random jitter and continue,
                # otherwise we are stuck.
                if rng.random() < 0.1:
                    noise = rng.normal(scale=a.mean() * 0.01, size=n)
                    a = np.clip(a + noise, 0.0, None)
                    a = _project_to_simplex(a, S)
                    cur_max = np.max(_full_convolution_fft(a))
                else:
                    break

        # record best
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Run many short restarts; each restart first applies a few LP
    refinement steps and then a short projected‑gradient polish.
    The best sequence discovered before the time limit is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # try to reuse a previously discovered construction, if any
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # main optimisation loop
    while time.time() < deadline:
        attempt += 1

        # choose a length – explore a medium range (800 … 1800)
        if base_seq is not None:
            max_len = max(800, min(1800, int(base_seq.size * 1.2)))
        else:
            max_len = 1800
        n = rng.integers(800, max_len + 1)

        # initialise a feasible vector
        a0 = _initial_sequence(base_seq, n, rng)

        # time slice for the LP phase (≈ 15 s)
        slice_deadline = min(time.time() + 15.0, deadline)
        a_lp, max_lp = _optimize_lp_local(a0, slice_deadline, rng)

        # polish with the gradient optimiser if time permits
        if time.time() < deadline - 5.0:
            slice_deadline2 = min(time.time() + 15.0, deadline)
            a_opt, max_opt = _optimize_local(a_lp, slice_deadline2, rng)
        else:
            a_opt, max_opt = a_lp, max_lp

        seq = a_opt.tolist()
        external_score = evaluate_sequence(seq)

        if external_score < best_overall_score:
            best_overall_score = external_score
            best_overall_seq = seq
            print(f"[global] attempt={attempt} n={n} new best score={best_overall_score:.6f}")

    # fallback – should never happen
    if best_overall_seq is None:
        best_overall_seq = [1.0]

    # store for potential reuse by a later run
    globals()['height_sequence_1'] = best_overall_seq
    return best_overall_seq
