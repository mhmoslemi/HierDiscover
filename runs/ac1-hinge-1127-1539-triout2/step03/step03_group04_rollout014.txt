        a = np.maximum(a, 0.0)

            a = np.maximum(a + jitter, 0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

"""
Fast multi‑stage search for a step‑function that minimises
evaluate_sequence(seq).

The algorithm:
  1️⃣  Coarse enumeration of several parametric families
      (power‑law, exponential, sin‑power, half‑circle, triangle,
      half‑sine).  Each candidate is projected onto the simplex
      Σa = √(2·n) and scored with a fast FFT‑based max‑convolution.
  2️⃣  The best few candidates are refined with a *sub‑gradient descent*
      on the exact max‑convolution (instead of the soft‑max used
      previously).  After each step the vector is projected back onto
      the simplex.
  3️⃣  A short guided hill‑climb (same as the reference implementation)
      finishes the search.

All stages respect the time budget (default 1000 s).  The best sequence
found is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog  # kept for possible future extensions


# ----------------------------------------------------------------------
# Core utilities (FFT‑based convolution, scoring, simplex projection)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Exact competition objective (max‑convolution entry) for a sum‑normalised vector."""
    # a is assumed to be already normalised to Σa = √(2·n)
    return _max_convolution_fft(a)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of parametric sequences (used in the coarse grid stage)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  →  normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return a * (target_sum / a.sum())


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam * i)  →  normalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    return a * (target_sum / a.sum())


def _sin_power_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    """a_i = sin(π(i+1)/(n+1))^q  →  normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** q
    return a * (target_sum / a.sum())


def _half_circle_sequence(n: int, factor: float, target_sum: float) -> np.ndarray:
    """
    Semi‑circular shape.
    radius = factor * (n‑1)/2 .  a_i = sqrt(1 - ((i‑c)/radius)^2) clipped at 0.
    """
    c = (n - 1) / 2.0
    radius = factor * c
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(np.maximum(0.0, 1.0 - ((i - c) / radius) ** 2))
    if a.sum() == 0:
        a = np.ones(n, dtype=np.float64) * (target_sum / n)
    else:
        a = a * (target_sum / a.sum())
    return a


def _triangle_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i = (i+1)*(n-i)  (mountain shape) → normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    return a * (target_sum / a.sum())


def _half_sine_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i = sin(π(i+0.5)/n)  → normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    return a * (target_sum / a.sum())


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent on   f(a) = max_k (a*a)[k] .
    The sub‑gradient at a is the sum over all active shift indices
    of 2·a shifted appropriately.  After each step we project back
    onto the simplex Σa = √(2·n).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    # initial step size – a small fraction of the allowed L1‑norm
    step0 = target_sum * 0.03

    while time.time() < deadline:
        # full convolution (O(n) via np.convolve – cheap for our moderate n)
        conv = np.convolve(a, a)
        max_val = conv.max()

        # indices attaining the maximum (within a tiny tolerance)
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: for each active shift k, add 2·a_{k-i} to component i
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # vectorised contribution
            idx = np.arange(lo, hi + 1)
            g[lo: hi + 1] += 2.0 * a[k - idx]

        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            break

        # diminishing step size, normalised by gradient norm
        lr = step0 / (norm_g * (1.0 + 0.001 * it))

        # projected gradient step
        a = a - lr * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] SUBG  improvement → {best_val:.6f}")

        # occasional random jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = np.random.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (same as the reference implementation)
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray, deadline: float):
    """Mass‑transfer hill climbing that respects the simplex constraint."""
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07  # max fraction of a donor’s mass that may be moved

    while time.time() < deadline:
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # sensitivity of each coordinate for the peak at k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens = np.zeros(n)
        sens[lo: hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        weight = sens * a
        if weight.max() == 0:
            break
        donor = int(np.argmax(weight))

        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH    improved → {best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs a fast parametric grid, refines the most promising
    candidates with sub‑gradient descent and finishes with a guided
    hill‑climbing phase.  The best sequence found before the time budget
    expires is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the caller
    deadline = time.time() + budget_s - 1.0

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Stage 1 – coarse multi‑family grid search
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000, 10000, 15000]
    # parameter grids for the various families
    power_ps = np.arange(0.3, 2.01, 0.3)          # power‑law exponent
    exp_lams = np.arange(0.2, 2.01, 0.2)          # exponential decay
    sin_qs = np.arange(0.5, 5.01, 0.5)           # sin‑power exponent
    half_factors = [0.6, 0.8, 1.0]               # half‑circle radius factor

    # store the top‑k raw candidates (by score) for the refinement stage
    top_k = 12
    raw_candidates = []  # list of (score, array)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ------ power‑law ------------------------------------------------
        for p in power_ps:
            if time.time() >= deadline:
                break
            a = _power_law_sequence(n, p, target_sum)
            val = _score(a)
            raw_candidates.append((val, a.copy()))

        # ------ exponential ----------------------------------------------
        for lam in exp_lams:
            if time.time() >= deadline:
                break
            a = _exp_sequence(n, lam, target_sum)
            val = _score(a)
            raw_candidates.append((val, a.copy()))

        # ------ sin‑power ------------------------------------------------
        for q in sin_qs:
            if time.time() >= deadline:
                break
            a = _sin_power_sequence(n, q, target_sum)
            val = _score(a)
            raw_candidates.append((val, a.copy()))

        # ------ half‑circle -----------------------------------------------
        for f in half_factors:
            if time.time() >= deadline:
                break
            a = _half_circle_sequence(n, f, target_sum)
            val = _score(a)
            raw_candidates.append((val, a.copy()))

        # ------ triangle --------------------------------------------------
        a = _triangle_sequence(n, target_sum)
        raw_candidates.append((_score(a), a.copy()))

        # ------ half‑sine -------------------------------------------------
        a = _half_sine_sequence(n, target_sum)
        raw_candidates.append((_score(a), a.copy()))

    # keep only the globally best few candidates
    raw_candidates.sort(key=lambda x: x[0])
    raw_candidates = raw_candidates[:top_k]

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient descent (exact max) on the shortlisted candidates
    # ------------------------------------------------------------------
    for base_val, a0 in raw_candidates:
        if time.time() >= deadline:
            break
        n = a0.size
        # allocate a slice of the remaining budget (≈30 s per candidate)
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(30.0, remaining / 2.0)

        a_opt, val = _subgradient_descent(a0, sub_deadline)

        # ------------------------------------------------------------------
        # Stage 3 – guided hill‑climbing refinement (final polishing)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_opt, val = _guided_hill_climb(a_opt, deadline)

        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"*** GLOBAL BEST ***  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – use the previously‑found global variable if it exists
    # ------------------------------------------------------------------
    try:
        hs1 = globals().get("height_sequence_1")
        if hs1 is not None:
            val = evaluate_sequence(list(hs1))
            if val < best_score:
                best_seq = np.array(hs1, dtype=np.float64)
                best_score = val
                print(f"*** REPLACED BY height_sequence_1 → {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fallback – simple normalised constant sequence (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a = _project_to_simplex(a, np.sqrt(2.0 * n))
        best_seq = a

    # Trim trailing (near‑zero) entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # convert to a plain list for the competition harness
    return best_seq.tolist()
