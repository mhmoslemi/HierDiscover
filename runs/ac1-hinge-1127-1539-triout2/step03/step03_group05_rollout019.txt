"""
Search for a non‑negative step‐function minimizing
    2·n·max(conv(a,a)) / (sum(a)**2)

Approach:
1. initialise a from the best known construction (global variable
   `height_sequence_1`) or by a quick parametric sweep of several
   symmetric “U‑shapes”.
2. run a simulated‑annealing style local search that repeatedly
   transfers a small amount of mass from one index to another.
   The convolution is updated incrementally (O(n) per move) so
   millions of proposals can be examined within the 1000 s budget.
3. occasional Gaussian jitter followed by a projection onto the
   ℓ₁‑simplex is used to escape plateaus.
4. the best vector encountered is returned as a plain list.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT, length 2*len(a)-1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]

def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x_i ≥ 0, sum x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param controls the edge enhancement (≥0).  The result is scaled
    to have ℓ₁‑norm `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)

def generate_sine_shape(n: int, target_sum: float) -> np.ndarray:
    """
    Produce a non‑negative sinusoidal shape a_i ∝ sin(π·i/(n‑1)).
    The vector is projected onto the simplex to enforce the ℓ₁‑norm.
    """
    x = np.arange(n, dtype=np.float64)
    raw = np.sin(np.pi * x / (n - 1))
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)

# ----------------------------------------------------------------------
# Simulated annealing helper – incremental convolution update
# ----------------------------------------------------------------------
def _update_convolution(B: np.ndarray, a: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` mass from entry i
    to entry j (i != j).  The update is performed exactly using the
    linear‑algebraic identity derived in the analysis section.
    """
    n = a.shape[0]
    B_new = B.copy()
    # linear terms (shifted copies of the original a)
    B_new[i:i + n] -= 2.0 * delta * a
    B_new[j:j + n] += 2.0 * delta * a
    # quadratic terms
    B_new[2 * i] += delta * delta          # (−δ)² = δ²
    B_new[2 * j] += delta * delta          # (+δ)² = δ²
    B_new[i + j] -= 2.0 * delta * delta    # 2·(−δ)·(+δ) = −2δ²
    return B_new

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimizes `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size and starting point
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        a = np.array(base, dtype=np.float64)
        n = a.shape[0]
        print(f"Using provided base sequence of length {n}")
    else:
        # choose a comfortable size; large enough to give flexibility,
        # small enough to keep the per‑iteration cost modest.
        n = np.random.randint(7000, 9000)
        target_sum = np.sqrt(2.0 * n)

        # coarse sweep over a few parametric families
        best_local = None
        best_max = np.inf
        print(f"[{time.time():.2f}] Parametric sweep, n={n}")

        # linear U‑shapes
        for p in np.linspace(0.0, 4.0, 41):
            cand = generate_U_shape(n, target_sum, kind="linear", param=p)
            cur_max = conv_self_fft(cand).max()
            if cur_max < best_max:
                best_max = cur_max
                best_local = cand.copy()
        # quadratic U‑shapes
        for p in np.linspace(0.0, 3.0, 31):
            cand = generate_U_shape(n, target_sum, kind="quadratic", param=p)
            cur_max = conv_self_fft(cand).max()
            if cur_max < best_max:
                best_max = cur_max
                best_local = cand.copy()
        # exponential U‑shapes
        for p in np.linspace(0.0, 2.0, 21):
            cand = generate_U_shape(n, target_sum, kind="exponential", param=p)

            if cur_max < best_max:

                best_local = cand.copy()

        cur_max = conv_self_fft(cand).max()
        if cur_max < best_max:
            best_max = cur_max
            best_local = cand.copy()

        a = best_local
        print(f"Initial sweep best max conv = {best_max:.6f}")

    # ------------------------------------------------------------------
    # Normalise sum (the security checks clip values to [0,1000] but the
    # objective is scale‑invariant, we keep the original ℓ₁‑norm).
    # ------------------------------------------------------------------
    S = a.sum()
    if S <= 0:
        # fallback to uniform if something went wrong
        S_target = np.sqrt(2.0 * n)
        a = np.full(n, S_target / n, dtype=np.float64)
        S = a.sum()
    # keep the exact sum – it never changes during the mass‑transfer moves
    target_sum = S

    # ------------------------------------------------------------------
    # Pre‑compute the convolution of the starting vector.
    # ------------------------------------------------------------------
    B = conv_self_fft(a)
    cur_max = B.max()
    best_a = a.copy()
    best_max = cur_max
    best_score = 2.0 * n * cur_max / (target_sum * target_sum)   # = cur_max when S = sqrt(2n)

    print(f"[{time.time():.2f}] Starting local SA, initial max conv {cur_max:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    iteration = 0
    no_improve = 0
    T = 0.2 * cur_max          # initial temperature (≈20 % of current value)
    T_min = 1e-9
    cooling = 0.9995           # multiplicative cooling factor per 1000 moves
    scale_factor = 0.05        # max proportion of the source entry that can be moved

    while time.time() < deadline:
        # pick source (i) with non‑zero mass and destination (j) distinct
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1   # ensure j != i

        # choose a random fraction of the source mass
        beta = np.random.rand() * scale_factor
        delta = beta * a[i]
        if delta <= 0.0:
            continue

        # incremental update of convolution
        B_new = _update_convolution(B, a, i, j, delta)
        new_max = B_new.max()
        new_score = 2.0 * n * new_max / (target_sum * target_sum)

        cur_score = 2.0 * n * cur_max / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # apply the move
            a[i] -= delta
            a[j] += delta
            B = B_new
            cur_max = new_max
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_max = new_max
                best_a = a.copy()
                no_improve = 0
                print(f"[{time.time():.2f}] iter {iteration:07d} new best {best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        # cooling schedule – apply every 1000 iterations
        if iteration % 1000 == 0:
            T = max(T * cooling, T_min)

        # occasional full‑vector jitter to break stagnation
        if no_improve >= 2000:
            jitter = np.random.randn(n) * 1e-3 * target_sum / n
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            B = conv_self_fft(a)
            cur_max = B.max()
            cur_score = 2.0 * n * cur_max / (target_sum * target_sum)
            no_improve = 0
            print(f"[{time.time():.2f}] jitter applied at iter {iteration}")

        # occasional recompute of B from scratch to keep numerical errors low
        if iteration % 5000 == 0:
            B = conv_self_fft(a)
            cur_max = B.max()
            cur_score = 2.0 * n * cur_max / (target_sum * target_sum)

        # early exit if we already beat the target bound
        if best_score < 1.503:
            print(f"[{time.time():.2f}] target bound reached ({best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # Final brief polishing – a few gradient‑descent steps on the
    # smoothed max (log‑sum‑exp) to clean up any remaining roughness.
    # ------------------------------------------------------------------
    def _smoothed_grad(a_vec: np.ndarray, tau_factor: float = 0.02):
        n_loc = a_vec.shape[0]
        conv = conv_self_fft(a_vec)
        m = conv.max()
        tau = max(tau_factor * m, 1e-12)
        w = np.exp((conv - m) / tau)
        w /= w.sum()
        a_rev = a_vec[::-1]
        L = _next_pow2(len(w) + len(a_rev) - 1)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[n_loc - 1 : n_loc - 1 + n_loc]
        return grad, m

    a = best_a.copy()
    B = conv_self_fft(a)
    cur_max = B.max()
    eta = 0.1 * target_sum / n
    beta_mom = 0.8
    vel = np.zeros_like(a)
    for step in range(5000):
        if time.time() >= deadline:
            break
        grad, m_val = _smoothed_grad(a, tau_factor=0.02)
        vel = beta_mom * vel + (1 - beta_mom) * grad
        a_new = a - eta * vel
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        a = a_new
        B = conv_self_fft(a)
        cur_max = B.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_score = 2.0 * n * cur_max / (target_sum * target_sum)
            print(f"[{time.time():.2f}] polishing step {step} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Search finished – best score {best_score:.6f}")
    return best_a.tolist()

"""
Search for a non‑negative step‑function that minimizes

    2·n·max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.
The method consists of three phases:

1.  Initialise a vector either from the global variable
    ``height_sequence_1`` (if present) or by a quick parametric sweep of a
    few symmetric “U‑shapes’’ and a sinusoid, keeping the best one.

2.  Simulated‑annealing style local search.  In each iteration a small
    amount of mass is transferred from one index to another; the
    convolution is updated *incrementally* (O(n) per move) by an exact
    algebraic formula, allowing millions of proposals within the 1000 s
    budget.  A temperature schedule permits occasional uphill moves.
    When progress stalls a tiny Gaussian jitter is applied and the
    convolution is recomputed from scratch.

3.  A brief smoothed‑gradient polishing stage (log‑sum‑exp smoothing)

The best vector encountered is returned as a plain Python list.

def _update_convolution(
    B: np.ndarray, a: np.ndarray, i: int, j: int, delta: float
) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` mass from entry i
    to entry j (i != j).  The update follows the exact identity

        B' = B + 2·δ_i·shift(a,i) + 2·δ_j·shift(a,j)
             + δ_i²·e_{2i} + δ_j²·e_{2j} + 2·δ_i·δ_j·e_{i+j},

    B_new[i + j] -= 2.0 * delta * delta    # 2·(–δ)(+δ) = –2δ²
    return B_new

    T = 0.2 * cur_max          # initial temperature
    T_min = 1e-9
    cooling = 0.9995           # cooling per 1000 accepted moves
    scale_factor = 0.05        # max fraction of source mass to move

    while time.time() < deadline:
        # pick a source with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # pick a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random proportion of the source
        beta = np.random.rand() * scale_factor
        delta = beta * a[i]
        if delta <= 0.0:
            continue

        # proposer – incremental convolution update
        B_new = _update_convolution(B, a, i, j, delta)
        new_max = B_new.max()
        new_score = 2.0 * n * new_max / (target_sum * target_sum)
        cur_score = 2.0 * n * cur_max / (target_sum * target_sum)

        # Metropolis accept/reject
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            B = B_new
            cur_max = new_max
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_max = new_max
                best_a = a.copy()
                no_improve = 0
                print(f"[{time.time():.2f}] iter {iteration:07d}  new best {best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        # cooling
        if iteration % 1000 == 0:
            T = max(T * cooling, T_min)

        # occasional jitter to escape stagnation
        if no_improve >= 2000:
            jitter = np.random.randn(n) * 1e-3 * target_sum / n
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            B = conv_self_fft(a)
            cur_max = B.max()
            cur_score = 2.0 * n * cur_max / (target_sum * target_sum)
            no_improve = 0
            print(f"[{time.time():.2f}] jitter applied at iter {iteration}")

        # refresh convolution periodically (numeric drift)
        if iteration % 5000 == 0:
            B = conv_self_fft(a)
            cur_max = B.max()
            cur_score = 2.0 * n * cur_max / (target_sum * target_sum)

        # early exit if target reached
        if best_score < 1.503:
            print(f"[{time.time():.2f}] target bound reached ({best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # Final smoothed‑gradient polishing (few hundred iterations)
    # ------------------------------------------------------------------
    def _smoothed_grad(v: np.ndarray, tau_factor: float = 0.02):
        """Gradient of log‑sum‑exp smoothed max‑convolution."""
        n_loc = v.shape[0]
        conv = conv_self_fft(v)
        m = conv.max()
        tau = max(tau_factor * m, 1e-12)
        w = np.exp((conv - m) / tau)
        w /= w.sum()
        rev = v[::-1]
        L = _next_pow2(len(w) + len(rev) - 1)
        W = np.fft.rfft(w, n=L)
        R = np.fft.rfft(rev, n=L)
        full = np.fft.irfft(W * R, n=L)[: len(w) + len(rev) - 1]
        grad = (2.0 / tau) * full[n_loc - 1 : n_loc - 1 + n_loc]
        return grad, m

    a = best_a.copy()
    B = conv_self_fft(a)
    cur_max = B.max()
    eta = 0.1 * target_sum / n
    beta_mom = 0.8
    vel = np.zeros_like(a)

    for step in range(5000):
        if time.time() >= deadline:
            break
        grad, m_val = _smoothed_grad(a, tau_factor=0.02)
        vel = beta_mom * vel + (1 - beta_mom) * grad
        a_new = a - eta * vel
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        a = a_new
        B = conv_self_fft(a)
        cur_max = B.max()
        cur_score = 2.0 * n * cur_max / (target_sum * target_sum)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time():.2f}] polishing iter {step:04d} improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished – best score {best_score:.6f}")
    return best_a.tolist()
