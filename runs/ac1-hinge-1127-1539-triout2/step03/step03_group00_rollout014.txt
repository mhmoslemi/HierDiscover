def _adam_optimize(a0, target_sum, deadline, beta_schedule):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t = 0
    beta = 0.5
    max_beta = 2048.0
    while time.time() < deadline:
        t += 1
        # compute gradient g
        g, b = _gradient_softmax(a, beta)
        # update moments
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.01 * target_sum / (np.sqrt(v_hat) + eps)  # adapt
        a = a - step * m_hat
        # projection
        a = _project_to_simplex(a, target_sum)
        # annealing beta schedule
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta*1.5, max_beta)
        # maybe jitter occasionally
        if t % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum*1e-3, size=a.shape)
            a += jitter

"""
A fast mixed‑strategy optimiser for the step‑function problem.

The goal is to minimise   2·n·max(conv(a,a)) / (sum a)^2 .
Because the objective is homogeneous we can restrict ourselves to the
ℓ¹‑simplex   Σ a_i = √(2·n)   and then the quantity to minimise is simply
max(conv(a,a)).  The algorithm proceeds as follows:

1.  Several families of random vectors are drawn on the simplex
    (exponential, power‑law, mixed, Dirichlet, triangular).
2.  The best few seeds for a given length are refined by
    * Adam optimisation of a log‑sum‑exp smooth approximation of the
      maximal convolution (β is annealed up to a large value);
    * a pure sub‑gradient descent that works directly on the true
      max‑convolution (gradient = 2·(a ⋆ e_k) where k are the maximal
      indices);
    * a lightweight simulated‑annealing phase that swaps small amounts
      of mass between coordinates.
3.  The whole search is performed for a list of lengths; larger n
    gives more flexibility and typically lower maximal convolution.
4.  The best sequence found before the time budget expires is returned
    as a plain Python list.

All steps respect the non‑negativity and simplex constraints by using
the Euclidean projection onto the simplex (Duchi et al. 2008).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a computed via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation: max(conv(a,a))."""
    return _max_convolution_fft(a)

# Random initialisers (several families)

def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σx = √(2·n)."""
    target = np.sqrt(2.0 * n)
    choice = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if choice == 0:            # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:          # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif choice == 2:          # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:          # Dirichlet (uniformly random simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif choice == 4:          # triangular – increase then decrease
        peak = rng.integers(n // 4, 3 * n // 4)
        left = np.linspace(0.0, 1.0, peak + 1)
        right = np.linspace(1.0, 0.0, n - peak)
        a = np.concatenate([left, right[1:]])  # keep length n

    elif choice == 5:          # two‑exponential mixture (bimodal)
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        mix = rng.random()
        a = mix * np.exp(-lam1 * idx) + (1 - mix) * np.exp(-lam2 * (n - 1 - idx))

    else:                       # uniform plus tiny jitter
        a = np.full(n, target / n)

    # small Gaussian jitter + projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) approximation of max(conv)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where
        grad = ∇_a ( (1/β)·log Σ_k exp(β·(a*a)_k) )
        conv = (a*a) (full convolution)
    Formula: grad = 2·(a ⋆ p) with p_k ∝ exp(β·conv_k).
    """
    n = a.shape[0]

    # full convolution
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_c = np.max(conv)
    exp_shifted = np.exp(beta * (conv - max_c))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv

# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """Projected Adam on the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    beta = 0.5          # starting temperature for the soft‑max
    max_beta = 2048.0

    # step‑size scale – roughly the typical size of a coordinate
    base_lr = target_sum * 0.12

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        # gradient of the smooth max
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # adaptive step
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to leave plateaus
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # anneal β (make the soft‑max sharper)
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

    return a

def _subgradient_from_max_indices(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of   max_k (a*a)_k   at the current a.
    For each k where the max is attained we add the contribution
    2·a_{k-i} to ∂/∂a_i, and finally average over all maximal k.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    if max_idx.size > 1:
        g /= float(max_idx.size)
    return g


def _subgradient_descent_opt(a0: np.ndarray, target_sum: float,
                             deadline: float) -> np.ndarray:
    """Simple projected sub‑gradient method on max(conv)."""
    a = a0.copy()
    step0 = target_sum * 0.5
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        conv = _full_convolution_fft(a)
        max_val = conv.max()
        max_idx = np.where(conv == max_val)[0]

        # sub‑gradient of the max
        g = _subgradient_from_max_indices(a, max_idx)
        norm = np.linalg.norm(g)
        if norm == 0.0:
            break

        # slowly decreasing step size
        step = step0 / (1.0 + iteration * 0.005)
        a -= step * g / norm
        a = _project_to_simplex(a, target_sum)

        # stop early if the objective is already extremely low
        if max_val < 1.0:
            break
        if iteration > 3000:
            break

    return a

def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Lightweight SA that moves random fractions of mass between two positions."""
    a = a0.copy()
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-6      # initial temperature
    cooling = 0.9996

    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a fraction of the mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best

                break

            # Reserve a slice of the budget for this seed
            sub_dead = min(deadline, time.time() + 22.0)   # ~22 s per refinement

            # 1) Adam on smooth max (log‑sum‑exp)
            a_opt = _adam_optimize(seed_vec, target, sub_dead)

            # 2) Switch to true max‑convolution sub‑gradient descent
            a_opt = _subgradient_descent_opt(a_opt, target, sub_dead)

            # 3) Final simulated‑annealing polish (uses whatever time is left)
            a_opt = _simulated_annealing(a_opt, deadline)

            sc = _score(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} score={sc:.6f}")

        # discard candidates before moving to the next length
        top_candidates.clear()

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_seq = a
        best_score = _score(a)

    # Trim trailing near‑zero entries for a cleaner output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
