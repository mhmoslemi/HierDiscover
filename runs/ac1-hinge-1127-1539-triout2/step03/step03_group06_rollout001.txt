def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    best_seq, best_score = None, float('inf')

    # optionally load height_sequence_1 if exists; store normalized.
    # We'll accept but rely on new search.

    # define length options
    lengths = [2000, 4000, 8000, 12000, 20000, 30000, 40000] # limit perhaps.

    # While time left, sample n
    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # initialization
        init_type = rng.choice(["random","uniform","bestpad"])
        if init_type == "random":
            a = rng.random(n) + 1e-8
        elif init_type == "uniform":
            a = np.full(n, 1.0)
        else: # bestpad, using previous best if exists and not too short
            if best_seq is not None and best_seq.size <= n:
                # embed
                a = np.concatenate([best_seq, rng.random(n - best_seq.size)])
            else:
                a = rng.random(n)

        # renorm to S
        S = math.sqrt(2.0 * n)
        a = a / a.sum() * S

        # run gradient descent optimization
        seq, val = _run_sd_exponential(a, n, deadline)

        # external score = val (since normalized)
        if val < best_score:
            best_score = val
            best_seq = seq
            print(f"[improve] n={n} val={best_score:.6f}")

        # optionally LP refine if size <= 3000
        if _SCIPY_AVAILABLE and n <= 3000:
            seq_ref = _lp_refine(best_seq, deadline)
            conv, max_conv, _ = _conv_max(seq_ref)
            if max_conv < best_score:
                best_score = max_conv
                best_seq = seq_ref

        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] score={best_score:.6f}, len={len(best_seq)}")

"""
search_step_function.py

A heuristic search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The external score equals the maximum entry of the
self‑convolution once the vector is normalised to ℓ₁‑norm √(2 n).
The algorithm therefore directly minimises the peak of the autocorrelation.

Strategy
--------
1.  Pick a length n (several values are tried, up to a few‑tens of thousands).
2.  Initialise a non‑negative vector a (random, uniform or padded from the
    current best solution) and scale it to the required ℓ₁‑norm S = √(2 n).
3.  Optimise a using exponentiated‑gradient descent on a *smooth* surrogate
    of the max‑convolution:
        L(a) = log‑sum‑exp( α·(a*a) )
    with a temperature parameter α that is slowly increased.
    The gradient of L can be computed in O(n log n) by FFTs.
    After each update the vector is renormalised to keep the ℓ₁‑norm fixed,
    guaranteeing feasibility.
4.  The best vector seen over all restarts is kept.  When time permits a
    small linear‑programming refinement (the same refinement used in the
    reference implementation) is applied for modest lengths.
5.  Finally a long polishing run is performed on the current best vector.

The whole procedure respects the 1000 s wall‑clock budget and prints progress
information.  It returns a plain ``list[float]`` that can be fed to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# Optional LP refinement (same as the reference code)

try:

    _SCIPY_AVAILABLE = True
except Exception:   # pragma: no cover
    _SCIPY_AVAILABLE = False


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Construct dense matrix H with (H @ g)[k] = (f * g)[k] (used by LP)."""
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5) -> np.ndarray:
    """Linear‑programming refinement (same as the reference implementation)."""
    if not _SCIPY_AVAILABLE or f.shape[0] > 3000:
        return f
    n = f.shape[0]
    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:                  # pragma: no cover
            break
        if not res.success:
            break
        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S
    return f

# Core optimisation utilities (exponentiated‑gradient on a soft‑max surrogate)

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of ``a`` using FFT (real input)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of  L(a) = log Σ exp(alpha·conv)   w.r.t. a,
    where conv = a * a (full linear convolution).
    """
    n = a.shape[0]
    # soft‑max weights (stable)
    m = conv.max()
    # avoid overflow – clip the exponent
    exps = np.exp(np.clip(alpha * (conv - m), a_min=None, a_max=700.0))
    w = exps / exps.sum()                     # shape (2n‑1,)

    # gradient = 2 * (w  *  a_rev)  (cross‑correlation)
    a_rev = a[::-1]                           # view, no copy
    # convolution of w (length 2n‑1) with a_rev (length n)
    conv_len = conv.shape[0]
    conv2_len = conv_len + n - 1               # = 3n‑2
    fft_len = 1 << (conv2_len - 1).bit_length()
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    tmp = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:conv2_len]
    start = n - 1
    g = 2.0 * tmp[start:start + n]             # shape (n,)
    return g


def _run_exponential_softmax(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 200_000,
                             eta0: float = 0.6,
                             beta: float = 0.9,
                             decay: float = 0.99996,
                             jitter_prob: float = 0.015,
                             alpha0: float = 8.0,
                             alpha_factor: float = 1.0015,
                             alpha_max: float = 400.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the soft‑max surrogate.
    Returns the best vector seen and its (true) maximal convolution value.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)
    best_val = float('inf')
    best_seq = a.copy()

    v = np.zeros_like(a)          # momentum buffer
    eta = eta0
    alpha = alpha0
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a)                # full self‑convolution
        cur_max = conv.max()

        # gradient of the smooth surrogate
        g = _gradient_softmax(a, conv, alpha)

        # momentum (exponential moving average)
        v = beta * v + (1.0 - beta) * g

        # exponentiated‑gradient update (protect against overflow)
        update_factor = np.exp(-eta * np.clip(v, -500.0, 500.0))
        a = a * update_factor

        # keep positivity and renormalise to Σa = S
        a = np.clip(a, 1e-12, None)
        total = a.sum()
        a *= (S / total)

        # occasional random jitter to escape tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.012
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # schedule updates
        eta *= decay
        alpha = min(alpha * alpha_factor, alpha_max)

        # record best (true) max‑convolution value
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_seq = a.copy()
            last_improve = it

        # early termination if the wall‑clock is about to be exceeded
        if time.time() >= deadline - 0.1:
            break

    return best_seq, best_val

        n = int(rng.choice(lengths))

        # ----- initial vector ------------------------------------------------
        choice = rng.integers(0, 3)
        if choice == 0:                     # random positive entries
            a0 = rng.random(n) + 1e-6
        elif choice == 1:                   # uniform start
            a0 = np.full(n, 1.0)
        else:                               # pad the current best (if any)
            if best_seq is not None and best_seq.size <= n:
                pad = rng.random(n - best_seq.size) + 1e-6
                a0 = np.concatenate([best_seq, pad])
            else:
                a0 = rng.random(n) + 1e-6

        # normalise to the required ℓ₁‑norm √(2 n)
        S = math.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # ----- optimisation run ---------------------------------------------
        seq, surf_max = _run_exponential_softmax(a0, n, deadline)

        # Since the vector is kept normalised, ``surf_max`` equals the true
        # external score.
        if surf_max < best_score:
            best_score = surf_max
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        # ----- occasional LP refinement (only for moderate sizes) ----------
        if _SCIPY_AVAILABLE and n <= 3000:
            refined = _lp_refine(best_seq, deadline)
            conv = _conv_fft(refined)
            refined_max = conv.max()
            if refined_max < best_score:
                best_score = refined_max
                best_seq = refined.copy()
                print(f"[LP refine] run={run_id} n={n} refined score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 2)  Final polishing on the best candidate (if any).
    # ------------------------------------------------------------------
    if best_seq is not None:
        # a few extra gradient steps with a tighter schedule
        best_seq, final_max = _run_exponential_softmax(
            best_seq,
            best_seq.size,
            deadline,
            max_iters=150_000,
            eta0=0.4,
            beta=0.92,
            decay=0.99997,
            jitter_prob=0.01,
            alpha0=12.0,
            alpha_factor=1.001,
            alpha_max=500.0
        )
        # final optional LP refinement (kept cheap)
        best_seq = _lp_refine(best_seq, deadline)

        # final exact evaluation (the external score)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    else:
        # ------------------------------------------------------------------
        # Fallback – uniform sequence (should never happen in practice).
        # ------------------------------------------------------------------
        n = 2000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    return best_seq.tolist()
