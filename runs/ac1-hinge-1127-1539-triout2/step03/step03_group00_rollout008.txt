def _random_mix_sequence(n, rng):
    # random number of modes
    m = rng.integers(1, 5)
    a = np.zeros(n)
    for _ in range(m):

    a = np.maximum(a, 0)

"""
Propose a non‑negative step‑function (sequence) that minimises
evaluate_sequence(...).  The algorithm:

* tries several vector lengths (up to ~15000);
* draws many random initialisations from diverse families (power‑law,
  exponential, Dirichlet, Gaussian mixtures);
* refines the most promising seeds with an Adam‑style projected gradient
  descent on a smooth log‑sum‑exp surrogate of the max‑convolution;
* finishes with a simulated‑annealing mass‑transfer phase;
* finally up‑samples the current best solution to a larger length and
  refines it again.

All operations respect the ℓ¹‑simplex constraint Σa_i = √(2·n) and
non‑negativity.  The search stops before the provided budget (default
1000 s) and returns the best sequence found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    (Duchi et al. 2008)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))               # stable
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation: max entry of a*a.
    a is assumed to satisfy Σa = √(2·n).
    """
    return _max_convolution_fft(a)

# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target = np.sqrt(2.0 * n)
    choice = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if choice == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif choice == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                   # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    else:                               # uniform baseline
        a = np.full(n, target / n)

    # add a little Gaussian noise and re‑project
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _random_mixture_sequence(n: int, rng) -> np.ndarray:
    """
    Random mixture of a few Gaussian bumps.  This creates diverse
    multi‑modal shapes that are often good starting points.
    """
    target = np.sqrt(2.0 * n)
    a = np.zeros(n)
    idx = np.arange(n, dtype=float)
    n_modes = rng.integers(1, 5)            # 1 to 4 modes
    for _ in range(n_modes):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.02, 0.20) * n
        weight = rng.uniform(0.2, 1.0)
        a += weight * np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    # tiny noise to avoid exact zeros
    a += rng.normal(scale=target * 0.0005, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smooth max‑conv objective
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float, rng) -> np.ndarray:
    """
    Adam optimiser with projection onto the ℓ¹‑simplex.
    Uses an annealed soft‑max temperature to approach the true max.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # temperature schedule
    beta = 0.5
    max_beta = 2048.0
    beta_inc_iter = 200

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    base_lr = target_sum * 0.02   # a modest base learning rate

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal temperature
        if t % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 1.6, max_beta)

    return a

# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that transfers a random fraction of mass between two indices.
    Accepts uphill moves with probability exp(-Δ/T).  Temperature decays
    multiplicatively.  Returns the best sequence seen.
    """
    a = a0.copy()
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    # initialise temperature as a modest fraction of current score
    T = cur_score * 0.02 + 1e-6
    cooling = 0.9998

    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # pick distinct indices, ensure source has some mass
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 15 %) from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    competition objective.  The routine runs for at most ``budget_s``
    seconds, trying several vector lengths, random restarts, a fast
    Adam‑gradient phase and a simulated‑annealing refinement.
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5      # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Initialise from a global construction if one is present
    # ------------------------------------------------------------------
    init_from_global = globals().get("height_sequence_1")
    if isinstance(init_from_global, (list, tuple, np.ndarray)):
        a = np.asarray(init_from_global, dtype=float)
        n0 = a.size
        a = _project_to_simplex(a, np.sqrt(2.0 * n0))
        sc = _score(a)
        best_seq = a.copy()
        best_score = sc
        print(f"[{time.time():.1f}] Starting from global height_sequence_1 "
              f"(n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1)  Choose lengths to explore – larger n gives more flexibility
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 15000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Phase A – random sampling on the simplex (quick)
        # ------------------------------------------------------------------
        phase_end = min(deadline, time.time() + max(5.0, 0.02 * budget_s))
        top_candidates = []          # keep a few promising seeds for this n

        while time.time() < phase_end:
            # randomly pick a generator
            if rng.random() < 0.3:
                a0 = _random_mixture_sequence(n, rng)
            else:
                a0 = _random_initial_sequence(n, rng)

            sc = _score(a0)
            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} "
                      f"score={best_score:.6f}")

            # maintain a small pool of the best seeds for refinement
            if len(top_candidates) < 5 or sc < top_candidates[-1][0]:
                top_candidates.append((sc, a0.copy()))
                top_candidates.sort(key=lambda x: x[0])
                top_candidates = top_candidates[:5]

        # ------------------------------------------------------------------
        # Phase B – refine the promising seeds (Adam + SA)
        # ------------------------------------------------------------------
        for _, seed_seq in top_candidates:
            if time.time() >= deadline:
                break

            # allocate a slice of the remaining budget for this candidate
            sub_dead = min(deadline, time.time() + 30.0)

            # 1) Adam‑style projected gradient descent
            a_gd = _adam_opt(seed_seq, sub_dead, rng)

            if time.time() >= deadline:
                break

            # 2) Simulated‑annealing refinement (uses whatever time is left)
            a_sa = _simulated_annealing(a_gd, deadline)

            sc = _score(a_sa)
            if sc < best_score:
                best_score = sc
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} "
                      f"score={best_score:.6f}")

        # clear temporary pool before the next length
        top_candidates.clear()

    # ------------------------------------------------------------------
    # 2)  Optional up‑sampling of the best solution to an even larger size
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        for factor in [1.5, 2.0]:
            if time.time() >= deadline:
                break
            n_old = best_seq.size
            n_new = int(min(20000, n_old * factor))
            if n_new == n_old:
                continue

            # linear interpolation to the larger grid
            x_old = np.linspace(0, 1, n_old)
            x_new = np.linspace(0, 1, n_new)
            a_up = np.interp(x_new, x_old, best_seq)

            # project onto the appropriate simplex
            a_up = _project_to_simplex(a_up, np.sqrt(2.0 * n_new))

            # refine the up‑sampled candidate
            sub_dead = min(deadline, time.time() + 60.0)
            a_ref = _adam_opt(a_up, sub_dead, rng)
            a_ref = _simulated_annealing(a_ref, deadline)

            sc = _score(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] UP‑SAMPLED BEST (n={n_new}) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Final tidy‑up and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform sequence (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    # trim trailing tiny entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
