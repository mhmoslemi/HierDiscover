Implementation:

def proj_simplex(v, S):
    if np.all(v == 0):
        return np.full_like(v, S / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, len(v)+1)
    cond = u - cssv/ind > 0
    if not np.any(cond):
        # fallback: all zero?
        theta = cssv[-1]/len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho]/(rho+1)
    w = np.maximum(v - theta, 0)
    return w

Pseudo:

def _pgd_optimize(a0, n, deadline, ...):
    a = a0.copy()
    S = sqrt(2*n)
    best_max = np.inf
    best_a = a.copy()
    # momentum vector
    m = np.zeros_like(a)
    beta = 0.9
    eta = eta0
    decay = 0.9999
    iter = 0
    while time.time() < deadline and iter < max_iters:
        # compute conv
        conv = np.fft.ifft(np.fft.fft(a, n_fft) * np.fft.fft(a, n_fft)).real[:2*n-1]
        # For speed, we can compute via np.fft.rfft for real.
        conv_np = np.fft.irfft(np.fft.rfft(a, n_fft) * np.fft.rfft(a, n_fft), n=n_fft)[:2*n-1]
        M = conv_np.max()
        if M < best_max:
            best_max = M
            best_a = a.copy()
        # gradient of max index
        k = conv_np.argmax()
        # gradient vector:
        g = np.zeros_like(a)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] = 2.0 * a[partner[valid]]
        # momentum
        m = beta * m + (1 - beta) * g
        # update
        a = a - eta * m
        # project onto simplex
        a = proj_simplex_nonneg(a, S)
        # jitter occasionally
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * jitter_scale
            a = proj_simplex_nonneg(a + noise, S)
        # step decay
        eta *= decay
        iter += 1
    return best_a, best_max

def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    n_list = [2000, 4000, 8000, 12000] # maybe also 16000

    # Stage1 param random search
    param_trials = 300
    for trial in range(param_trials):
        if time.time() >= deadline:
            break
        n = int(rng.choice(n_list))
        shape_type = rng.choice(['exp','pow','lin','quad','piecewise'])
        # generate a base array: raw values before normalization
        i = np.arange(n, dtype=float)
        if shape_type == 'exp':
            r = rng.uniform(1.01, 2.0)
            a_raw = r**i
        elif shape_type == 'pow':
            p = rng.uniform(0.0, 3.0)
            offset = rng.uniform(0.0, 5.0)
            a_raw = (i + offset)**p
        elif shape_type == 'lin':
            slope = rng.uniform(0.0, 0.01)  # small slope? Actually need to ensure large variation maybe up to 0.1.
            a_raw = 1.0 + slope * i
        elif shape_type == 'quad':
            a_raw = (i + 1)**2
            # random scaling factor
            scale = rng.uniform(0.0, 1.0)
            a_raw = a_raw**scale
        elif shape_type == 'piecewise':
            # split into 4 segments
            seg = rng.integers(2,5) # 2-4 segments
            # random heights
            heights = rng.uniform(0.1, 1.0, size=seg)
            # assign lengths proportionally random
            cuts = np.sort(rng.integers(1, n, size=seg-1))
            segments = np.split(i, cuts)
            a_raw = np.concatenate([np.full_like(seg_i, fill_value=heights[j])
                                     for j, seg_i in enumerate(segments)])
        else:
            continue
        # add optional small random noise
        if rng.random() < 0.3:
            noise = rng.normal(scale=0.01*a_raw.mean(), size=n)
            a_raw = np.maximum(a_raw + noise, 0.0)
        # convert to simplex sum S:
        S = math.sqrt(2.0 * n)
        a = a_raw
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, S/n)
        else:
            a = a * (S / a_sum)

        # evaluate surrogate (max conv) quickly using _conv_max
        _, cur_max, _ = _conv_max(a)
        # record best
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[stage1 new best] trial={trial} n={n} shape={shape_type} score={best_score:.6f}")

    # Stage2 refined gradient descent on best candidates
    # We'll keep a list of top candidates (maybe 5) based on their max conv
    # But we only stored one best. We could also keep a small pool of best (list of tuples)
    # For simplicity, we can just use best_seq as start for further descent.
    if best_seq is not None:
        # run gradient descent with many iterations
        seq, score = _run_optimize_one_start(best_seq, best_seq.size, deadline,
                                             top_k_start=5, eta0=0.5, beta=0.9,
                                             decay=0.9999, jitter_prob=0.01,
                                             max_iters=500_000)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[stage2 improve] score={best_score:.6f}")

    # Stage3 final LP refinement if feasible
    if best_seq is not None:
        refined = _lp_refine(best_seq, deadline)
        _, refined_max, _ = _conv_max(refined)
        if refined_max < best_score:
            best_score = refined_max
            best_seq = refined.copy()
            print(f"[LP refine final] refined_score={best_score:.6f}")

    # compute final evaluate
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final result] score={final_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

def _conv_max(a):
    n = a.shape[0]
    conv_len = 2*n-1
    n_fft = 1 << (conv_len-1).bit_length()
    Fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(Fa*Fa, n=n_fft)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx

def _run_optimize_pg(a0, n, deadline, eta0=0.2, beta=0.9, decay=0.9999, jitter_prob=0.02, jitter_scale=0.01, max_iters=200_000):
    a = a0.copy()
    S = math.sqrt(2.0*n)
    best_seq = a.copy()
    best_val = float('inf')
    v = np.zeros_like(a)
    eta = eta0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute conv max
        conv, cur_max, idx = _conv_max(a)
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
        # gradient of max entry
        g = np.zeros_like(a)
        idx_arr = np.arange(n)
        partner = idx - idx_arr
        valid = (partner >= 0) & (partner < n)
        g[valid] = 2.0 * a[partner[valid]]
        # momentum
        v = beta * v + (1 - beta) * g
        # additive update
        a_temp = a - eta * v
        # projection onto simplex S
        # use proj_simplex
        a = proj_simplex(a_temp, S)

        # jitter
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * jitter_scale
            a = proj_simplex(a + noise, S)

        # step decay
        eta *= decay

        # exit ahead if out of time
        if time.time() >= deadline - 0.1:
            break

    return best_seq, best_val

Classic algorithm:

def proj_simplex(v, z=1.0):
    # projection of v onto {x: sum x = z, x >= 0}
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function (list of heights) that minimises the
external ``evaluate_sequence`` score.

Algorithm
---------

1. **Random‑parameter seeding** –  Sample a few thousand candidate
   sequences from simple parametric families (exponential, power‑law,
   linear, quadratic, piecewise‑constant).  After normalising the ℓ₁‑norm
   to √(2 n) we evaluate the surrogate loss (the maximal entry of the
   self‑convolution) and keep the best vector seen so far.

2. **Projected‑gradient refinement** – Starting from the best seed,
   perform a large number of additive gradient‑descent steps on the
   ℓ∞‑loss.  The gradient of the maximal convolution entry is
   `g_i = 2·a_{k‑i}` where `k` is the index of the current maximum.
   After each step the iterate is projected onto the simplex
   `{x≥0, Σx = √(2 n)}` (guaranteeing feasibility).  Momentum,
   step‑size decay and occasional jitter are used to accelerate
   convergence and escape plateaus.

3. **Exponentiated‑gradient polishing** –  The same best vector is further
   polished with the multiplicative update from the previous state‑of‑the‑art
   code (`_run_optimize_one_start`).  This works well because the update
   automatically respects non‑negativity.

4. **Optional linear‑program refinement** – If SciPy is available and the
   current length is ≤ 3000 we solve a tiny LP that maximises the ℓ₁‑norm
   of a partner vector under the current convolution bound.  The partner
   is mixed back into the current vector (small step size) and renormalised.
   This can shave a few‑hundredths off the bound.

All steps respect a wall‑clock deadline (default 1000 s).  The function
returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# optional SciPy for the LP refinement
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray):
    """
    Compute the full self‑convolution of ``a`` using FFT and return
    (conv_array, maximal entry, index of the maximum).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    N = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Fa * Fa, n=N)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _proj_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        {x : x_i ≥ 0 and Σ x_i = z}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]            # sort descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; put all mass on a single coordinate
        w = np.zeros_like(v)
        w[np.argmax(v)] = z
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_max_entry(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry with index ``k``.
    (∂/∂a_i) (a*a)_k = 2·a_{k‑i} for valid indices, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)
    g[valid] = 2.0 * a[partner[valid]]
    return g

# Optimisers
# ----------------------------------------------------------------------
def _run_optimize_pg(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     eta0: float = 0.3,
                     beta: float = 0.9,
                     decay: float = 0.9999,
                     jitter_prob: float = 0.02,
                     jitter_scale: float = 0.01,
                     max_iters: int = 300_000):
    """
    Projected (additive) gradient descent on the ℓ∞‑convolution loss.
    Returns the best vector and its (surrogate) maximal convolution entry.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)          # required ℓ₁‑norm
    best_a = a.copy()
    best_val = float('inf')
    v = np.zeros_like(a)           # momentum buffer
    eta = eta0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv, cur_max, idx_max = _conv_max(a)

        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()

        # gradient of the active entry
        g = _gradient_max_entry(a, idx_max)

        # momentum
        v = beta * v + (1.0 - beta) * g

        # additive update
        a_tmp = a - eta * v

        # projection onto the ℓ₁‑simplex (non‑negative)
        a = _proj_simplex(a_tmp, S)

        # occasional jitter
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * jitter_scale
            a = _proj_simplex(a + noise, S)

        # decay step size
        eta *= decay

        # early termination guard
        if time.time() >= deadline - 0.1:
            break

    return best_a, best_val


def _run_optimize_one_start(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            top_k_start: int = 3,
                            eta0: float = 0.8,
                            beta: float = 0.9,
                            decay: float = 0.99995,
                            jitter_prob: float = 0.02,
                            max_iters: int = 200_000):
    """
    Exponentiated‑gradient descent (multiplicative update) on the same
    surrogate loss.  This is the optimiser from the previous state‑of‑the‑art
    code, kept for polishing.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)                # ℓ₁‑norm we keep fixed
    best_local_max = float('inf')
    best_local_seq = a.copy()

    v = np.zeros_like(a)                 # momentum buffer
    eta = eta0
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv, cur_max, _ = _conv_max(a)

        # adapt the number of active shifts – increase if we have stalled
        top_k = top_k_start
        if it - last_improve > 2_000 and top_k < 30:
            top_k = min(30, top_k * 2)

        if conv.size <= top_k:
            shifts = np.arange(conv.size)
        else:
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient of the sum of the selected shifts
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in shifts:
            partner = k - idx
            valid = (partner >= 0) & (partner < n)
            g[valid] += 2.0 * a[partner[valid]]

        # momentum (exponential moving average)
        v = beta * v + (1.0 - beta) * g

        # multiplicative (exponentiated) update – guarantees positivity
        a = a * np.exp(-eta * v)

        # renormalise to keep the ℓ₁‑norm fixed
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional random jitter to escape tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # step‑size decay
        eta *= decay

        # record improvement
        if cur_max < best_local_max - 1e-12:
            best_local_max = cur_max
            best_local_seq = a.copy()
            last_improve = it

        # early exit guard
        if time.time() >= deadline - 0.1:
            break

    return best_local_seq, best_local_max

# Optional LP refinement (uses SciPy if available)

def _build_conv_matrix(f: np.ndarray):
    """
    Build dense matrix H (size (2n‑1) × n) such that (H @ g)[k] = (f * g)[k].
    Used only for the LP refinement (moderate n).
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5):
    """
    Linear‑programming refinement (Frank‑Wolfe style).  Given ``f`` we solve

        maximise   Σ g_i
        s.t.       g_i ≥ 0,
                   (f * g)[k] ≤ M  ∀k,

    where M = max_k (f * f)[k].  The optimum ``g`` is mixed back into ``f``
    with step size ``t``.  The procedure is repeated a few times.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:          # avoid gigantic dense LPs
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:                 # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a step‑function with a low score.
    The routine respects the given wall‑clock budget and returns a list of
    non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 0)  Use a supplied baseline if present.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                n0 = cand.size
                S0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * S0
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] height_sequence_1 score = {best_score:.6f}, n={n0}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1)  Random‑parameter seeding (stage 1)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000]
    param_trials = 350
    for trial in range(param_trials):
        if time.time() >= deadline:
            break
        n = int(rng.choice(lengths))
        S = math.sqrt(2.0 * n)

        # ----- generate a raw (un‑normalised) shape -----
        i = np.arange(n, dtype=float)
        shape_type = rng.choice(['exp', 'pow', 'lin', 'quad', 'piecewise'])
        if shape_type == 'exp':
            r = rng.uniform(1.01, 1.8)
            a_raw = r ** i
        elif shape_type == 'pow':
            p = rng.uniform(0.0, 3.0)
            offset = rng.uniform(0.0, 5.0)
            a_raw = (i + offset) ** p
        elif shape_type == 'lin':
            slope = rng.uniform(0.0, 0.02)
            a_raw = 1.0 + slope * i
        elif shape_type == 'quad':
            a_raw = (i + 1) ** 2
            scale = rng.uniform(0.1, 0.5)
            a_raw = a_raw ** scale
        elif shape_type == 'piecewise':
            seg = rng.integers(2, 5)               # 2‑4 segments
            cuts = np.sort(rng.integers(1, n, size=seg - 1))
            segments = np.split(i, cuts)
            heights = rng.uniform(0.1, 1.0, size=seg)
            a_raw = np.concatenate([np.full_like(seg_i, fill_value=heights[j])
                                    for j, seg_i in enumerate(segments)])
        else:
            continue

        # optional small noise
        if rng.random() < 0.3:
            noise = rng.normal(scale=0.01 * a_raw.mean(), size=n)
            a_raw = np.maximum(a_raw + noise, 0.0)

        if a_raw.sum() == 0.0:
            a = np.full(n, S / n)
        else:
            a = a_raw * (S / a_raw.sum())

        # evaluate surrogate loss (max‑convolution)
        _, cur_max, _ = _conv_max(a)

        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[stage1] trial={trial:03d} n={n} shape={shape_type} "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Projected‑gradient refinement (stage 2)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        seq_pg, score_pg = _run_optimize_pg(
            best_seq, n, deadline,
            eta0=0.3, beta=0.9, decay=0.9999,
            jitter_prob=0.02, jitter_scale=0.02,
            max_iters=300_000)

        if score_pg < best_score:
            best_score = score_pg
            best_seq = seq_pg.copy()
            print(f"[pgd] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Exponentiated‑gradient polishing (stage 3)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        seq_eg, score_eg = _run_optimize_one_start(
            best_seq, n, deadline,
            top_k_start=5, eta0=0.5, beta=0.9,
            decay=0.9999, jitter_prob=0.01,
            max_iters=500_000)

        if score_eg < best_score:
            best_score = score_eg
            best_seq = seq_eg.copy()
            print(f"[eg] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Optional LP refinement (stage 4)
    # ------------------------------------------------------------------
    if best_seq is not None:
        refined = _lp_refine(best_seq, deadline)
        _, refined_max, _ = _conv_max(refined)
        if refined_max < best_score:
            best_score = refined_max
            best_seq = refined.copy()
            print(f"[lp refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5)  Final evaluation using the official function
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector
        n = 1000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")
    else:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best score = {final_score:.6f}, length = {len(best_seq)}")
        best_score = final_score

    # Convert to plain Python list as required by the harness
    return best_seq.tolist()
