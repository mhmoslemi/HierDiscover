assistantfinal**Solution Overview**

The evaluation function  

\[
\text{score}(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2}
\]

is minimized when the sum of the vector is fixed to  
\(S=\sqrt{2n}\).  In that case the score is simply the maximum entry of the
self‑convolution `max(a*a)`.  

The program below searches for a vector `a` with a very small
convolution maximum.  
It combines three complementary ideas:

1. **Exponentiated (multiplicative) gradient descent** – a fast, cheap
   optimiser that works with a full sub‑gradient of the *current*
   convolution maximum.
2. **Linear‑programming refinement** – for a fixed vector `f` we solve  

   \[
   \max_{b\ge0}\ \sum b_j \qquad
   \text{s.t.}\quad (f*b)_k\le M\;(k=0..2n-2),
   \]

   where \(M=\max(f*f)\).  The optimum `b` is the most “mass‑heavy’’
   vector that still respects the current convolution bound.
   After normalising `b` we mix it with `f`.  Repeating this step
   (alternating optimisation) strictly decreases the bound.
3. **Local stochastic hill‑climbing** – a short simulated‑annealing
   phase removes the last few imperfections.

All steps respect the non‑negativity and clipping constraints required
by `evaluate_sequence`.  The search runs until the 1000 s budget is
exhausted and returns the best sequence found.

```python
"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises the score  2·n·max(a*a)/(sum(a)^2).

The algorithm consists of three phases:

1. fast exponentiated gradient descent (multiplicative updates);
2. LP‑refinement – alternating optimisation of a partner vector via
   a linear program that maximises the L1‑norm under the current
   convolution bound;
3. a short simulated‑annealing hill‑climb.

All phases respect the positivity, clipping and normalisation required
by `evaluate_sequence`.  The best vector found within the time budget is
returned.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (full mode)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return float(np.max(conv))


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (hard) max‑convolution w.r.t. a,
    summed over the supplied shift indices.
    For a shift k the contribution to g_j is 2·a_{k‑j}
    whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g


# ----------------------------------------------------------------------
# Phase 1 – exponentiated (multiplicative) gradient descent
# ----------------------------------------------------------------------


def _run_exponential_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             top_k: int = 30,
                             eta0: float = 0.5,
                             beta: float = 0.9,
                             decay: float = 0.99995,
                             jitter_prob: float = 0.02,
                             max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent with momentum.
    Returns the best vector found and its *max‑convolution* value.
    """
    a = a0.copy()
    target_sum = np.sqrt(2.0 * n)

    cur_best = conv_max(a)
    best_a = a.copy()
    v = np.zeros_like(a)          # momentum buffer
    eta = eta0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution and its maximum
        conv_len = 2 * n - 1
        fft_len = 1 << (conv_len - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
        cur_max = float(np.max(conv_full))

        # choose K shifts with the largest convolution values
        if conv_full.size <= top_k:
            shifts = np.arange(conv_full.size)
        else:
            shifts = np.argpartition(conv_full, -top_k)[-top_k:]

        # sub‑gradient from those shifts
        g = _gradient_from_shifts(a, shifts)

        # momentum step
        v = beta * v + (1.0 - beta) * g

        # exponentiated update (guarantees non‑negativity)
        a = a * np.exp(-eta * v)

        # renormalise to keep Σa = √(2·n)
        total = a.sum()
        if total == 0.0:
            a.fill(target_sum / n)
        else:
            a *= (target_sum / total)

        # occasional random jitter to escape flat regions
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= target_sum / a.sum()

        # learning‑rate decay
        eta *= decay

        # record improvement
        if cur_max < cur_best - 1e-12:
            cur_best = cur_max
            best_a = a.copy()
            print(f"[exp descent improve] time={time.time():.1f}s "
                  f"n={n} iter={it} max={cur_max:.6f}")

    return best_a, cur_best


# ----------------------------------------------------------------------
# Phase 2 – LP refinement (alternating optimisation)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ b)[k] = (f * b)[k].
    Entry A[k, j] = f[k‑j] if 0 ≤ k‑j < n else 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_refine_step(f: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """
    One LP refinement step.
    Maximise sum(b) under the constraints (f * b) ≤ M, b ≥ 0, b ≤ 1000.
    The resulting b is normalised to the required L1‑norm and mixed
    with f (mixing factor t=0.2).  Returns the new vector and its
    convolution maximum.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)

    # Variables
    b = cp.Variable(n)

    # Constraints
    constraints = [
        A @ b <= M,
        b >= 0,
        b <= 1000.0
    ]

    # Objective: maximise L1‑norm of b
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        # fallback to SCS if ECOS is unavailable
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000, verbose=False)

    if b.value is None:
        # LP failed – return original vector unchanged
        return f, M

    b_val = np.asarray(b.value).flatten()
    sum_b = b_val.sum()
    if sum_b <= 0.0:
        return f, M

    # Normalise to Σ = √(2·n)
    scale = np.sqrt(2.0 * n) / sum_b
    g = b_val * scale

    # Mix a few percent of g into f
    t = 0.2
    f_new = (1.0 - t) * f + t * g

    new_M = conv_max(f_new)
    return f_new, new_M


# ----------------------------------------------------------------------
# Phase 3 – simple simulated‑annealing hill climbing
# ----------------------------------------------------------------------


def _local_hill_climb(a_start: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill climbing / simulated annealing around ``a_start``.
    Returns the best vector encountered and its max‑convolution.
    """
    a = a_start.copy()
    cur_max = conv_max(a)
    best_a = a.copy()
    best_max = cur_max

    # temperature set as a modest fraction of the current max
    temperature = max(0.05 * cur_max, 1e-4)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # Pick donor and recipient indices
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # Transfer a random fraction of the donor's mass
        frac = rng.uniform(0.03, 0.15)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = conv_max(a_new)

        # Accept improvement or Metropolis move
        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(f"[hill climb improve] time={time.time():.1f}s "
                      f"n={n} iter={it} max={new_max:.6f}")
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        # Cool down
        temperature *= 0.9995

        # Small random jitter now and then
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()
            cur_max = conv_max(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[jitter] time={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}")

    return best_a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low score.
    The routine respects the 1000 s budget, runs many independent
    restarts of exponentiated gradient descent, refines them with a
    linear‑programming alternation, finishes with a short hill‑climb,
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")
    run_id = 0

    # ------------------------------------------------------------------
    # Use an already‑available construction if present (baseline)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {sc:.6f} "
                      f"length = {cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter space
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000]   # reasonable sizes for the LP step
    while time.time() < deadline:
        # pick a length, slightly favouring larger choices
        n = int(rng.choice(lengths,
                           p=np.array([0.20, 0.20, 0.30, 0.30])))

        # -----------------------------------------------------------------
        # Phase 1 – fast multiplicative gradient descent
        # -----------------------------------------------------------------
        conc = rng.uniform(0.05, 1.2)               # Dirichlet concentration
        a0 = rng.dirichlet(np.full(n, conc))
        a0 *= np.sqrt(2.0 * n)                      # enforce Σa = √(2·n)

        a, cur_max = _run_exponential_descent(
            a0, n, deadline,
            top_k=int(rng.integers(30, 80)),
            eta0=rng.uniform(0.4, 0.7),
            beta=0.9,
            decay=0.9999,
            jitter_prob=0.02,
            max_iters=20000
        )
        # cur_max already equals the external score for this vector

        # -----------------------------------------------------------------
        # Phase 2 – LP refinement (alternating optimisation)
        # -----------------------------------------------------------------
        # Perform a small number of refinement steps; each step solves a
        # linear program of size O(n^2) but is still cheap for n≤4000.
        for _ in range(7):
            if time.time() >= deadline:
                break
            M = conv_max(a)
            a_new, M_new = _lp_refine_step(a, M)
            # stop the inner loop if no improvement
            if M_new < M - 1e-8:
                a = a_new
            else:
                break

        # -----------------------------------------------------------------
        # Phase 3 – local hill climbing for the final polish
        # -----------------------------------------------------------------
        a, final_max = _local_hill_climb(a, n, deadline, rng,
                                         max_iters=12000)

        # -----------------------------------------------------------------
        # Record the best result
        # -----------------------------------------------------------------
        score = evaluate_sequence(a.tolist())
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification (the platform will evaluate again)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
```
