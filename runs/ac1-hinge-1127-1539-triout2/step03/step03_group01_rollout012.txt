def _optimize_adam(a0, deadline, rng):
    a = a0.copy()
    n=len(a)
    S = sqrt(2*n)
    # initialize Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.1 # step size base (to experiment)
    # schedule: alpha = alpha0 * (1 - iter/ max_iter) maybe.
    # track iteration number t
    t=0
    best_a = a.copy()
    max_b = np.max(_full_convolution_fft(a))
    best_max = max_b
    # hyperparameters for temp schedule
    tau = max_b * 0.5 # start at maybe half of max? or max?
    tau_decay = 0.9995
    # mass move probability
    move_prob = 0.08
    epsilon_move = 0.03
    jitter_prob = 0.02
    jitter_scale = 0.02
    # compute power accumulators
    beta1_pow = beta1
    beta2_pow = beta2

    while time.time() < deadline:
        t += 1
        # compute convolution and max
        b = _full_convolution_fft(a)
        cur_max = np.max(b)

        # update best
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        # decide action
        if rng.random() < move_prob:
            # random mass move as in baseline
            i = rng.integers(n)
            if a[i] > 0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    b_cand = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_cand)
                    # accept if better or with temperature
                    if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(tau, 1e-12)):
                        a = a_candidate
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
            # after move maybe shrink tau? Not necessary
        else:
            # gradient step via Adam
            # compute gradient via softmax weight
            max_b = cur_max
            # tau schedule: slowly reduce
            tau = max(tau * tau_decay, 1e-8)  # maybe keep > 0
            # compute weights
            diff = b - max_b
            # stable exponentials
            exps = np.exp(diff / max(tau, 1e-12))
            w = exps / np.sum(exps)

            # compute cross-correlation gradient
            a_rev = a[::-1]
            # full conv length
            conv_len = len(w) + n - 1
            fft_len = 1 << ((conv_len - 1).bit_length())
            Fw = np.fft.rfft(w, n=fft_len)

"""
Optimiser for the step‑function height sequence.

The algorithm combines:
  • a smooth soft‑max (log‑sum‑exp) approximation of the ∞‑norm of the
    self‑convolution,
  • Adam‑style projected optimisation on the ℓ₁‑simplex,
  • occasional simulated‑annealing “mass‑swap’’ moves,
  • occasional Gaussian jitter.
Multiple independent restarts with random lengths are performed until the
time budget expires.  The best sequence found (lowest maximal self‑convolution
entry) is returned.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                    # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (complexity O(n log n))."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a feasible non‑negative vector of length n with ℓ₁‑norm = √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                       # Dirichlet(1) – uniform on simplex
    a = _project_to_simplex(a, S)
    return a


def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     tau: float) -> np.ndarray:
    """
    Gradient of τ·logΣexp(b/τ) w.r.t. a.
    Returns 2·(a ⋆ w) where w_k ∝ exp(b_k/τ).
    """
    n = a.shape[0]
    max_b = np.max(b)
    # soft‑max weights (stable)
    diff = b - max_b
    exps = np.exp(diff / max(tau, 1e-12))
    w = exps / np.sum(exps)

    # cross‑correlation a ⋆ w = Σ_j a_j w_{i+j}
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:conv_len]
    g = 2.0 * conv[len(w) - 1 : len(w) - 1 + n]
    return g


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam‑based projected optimisation of the smoothed ℓ_∞ loss.
    Returns the best vector encountered and its maximal self‑convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = beta1
    beta2_pow = beta2

    # algorithmic hyper‑parameters
    base_lr = 0.2                # learning‑rate before bias‑correction
    move_prob = 0.08
    epsilon_move = 0.03          # fraction of mass that may be moved
    jitter_prob = 0.02
    jitter_scale = 0.02

    tau = None                   # will be initialised after first convolution
    tau_decay = 0.9995
    tau_min_factor = 0.01        # keep τ ≥ 0.01·initial τ

    # keep the best so far
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max
    init_tau = max(cur_max * 0.5, 1e-3)
    tau = init_tau

    while time.time() < deadline:
        # ---------- occasional random mass swap (simulated annealing) ----------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    b_candidate = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_candidate)
                    # accept if better, otherwise Metropolis criterion
                    if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(tau, 1e-12)):
                        a = a_candidate
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
            # temperature decay for simulated annealing moves
            tau = max(tau * tau_decay, init_tau * tau_min_factor)
            continue   # skip gradient step this iteration

        # ---------- gradient step (Adam) ----------
        # compute convolution and current max
        b = _full_convolution_fft(a)
        cur_max = np.max(b)

        # update best
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # temperature schedule for the soft‑max
        tau = max(tau * tau_decay, init_tau * tau_min_factor)

        # gradient of smoothed max
        g = _gradient_softmax(a, b, tau)

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # step size – we let it slowly decay with the iteration count
        # a simple linear decay, never below zero
        # (the iteration counter is implicit in the powers of β)
        alpha = base_lr * (1.0 - (1.0 - beta1_pow) / (1.0 - beta1))

        a_new = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new, S)

        # bias‑correction factor update
        beta1_pow *= beta1
        beta2_pow *= beta2

        # ---------- occasional jitter ----------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    # final recompute of best max (in case the loop exited just after an improvement)
    b = _full_convolution_fft(best_a)
    best_max = np.max(b)
    return best_a, best_max

Adam‑based optimiser for the step‑function height sequence.

The algorithm minimises the maximal entry of the self‑convolution
under the constraints a_i ≥ 0 and Σ a_i = √(2·n).  It combines:

  • a smooth log‑sum‑exp (soft‑max) approximation of the ∞‑norm,
  • Adam‑style projected optimisation on the ℓ₁‑simplex,
  • occasional simulated‑annealing mass‑swap moves,
  • occasional Gaussian jitter.

# Projection onto the simplex {x ≥ 0, sum(x) = S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]               # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power‑of‑two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

# Initialise a feasible vector (ℓ₁‑norm = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:

        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                     # Dirichlet(1) – uniform on simplex
    a = _project_to_simplex(a, S)

# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     tau: float) -> np.ndarray:

    n = a.shape[0]
    max_b = np.max(b)
    diff = b - max_b
    exps = np.exp(diff / max(tau, 1e-12))
    w = exps / np.sum(exps)

    conv_len = len(w) + n - 1

    g = 2.0 * conv[len(w) - 1: len(w) - 1 + n]
    return g
