def _adam_optimize(a0, deadline, rng):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    alpha = 1.0
    alpha_mul = 1.001
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        # compute weights softmax
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        lr = 0.5 / (np.sqrt(t) + 1)  # or fixed schedule
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a, np.sqrt(2.0*a.shape[0]))
        # evaluate
        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # schedule
        alpha *= alpha_mul
        # early stop if no improvement for many iter.
        if t > 2000 and best_score < cur_max * 0.9999:
            # not improving, break maybe
            pass
    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes the maximal entry
of its self‑convolution.  The score of a sequence a of length n (with the
mandatory L1‑norm √(2·n)) equals  max(a∗a).  The optimizer works in three
stages:

1.  A fast Adam optimiser on a *smooth* max (soft‑max of the convolution).
    This gives a strong initial reduction while staying far from the
    non‑differentiable corners.

2.  A projected sub‑gradient descent that directly minimises the true
    maximum (the weight vector is a Kronecker delta on the current worst
    index).  Momentum and a simple learning‑rate schedule are used.

3.  A short random “mass‑swap’’ refinement that tries local perturbations.

All updates are projected onto the simplex {x ≥ 0, Σx = √(2·n)}.
The routine runs many independent restarts (different lengths, initial
distributions and seeds) within the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σ x = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=float)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient term g_j = 2 * Σ_i w_{i+j} * a_i.
    Both a (len n) and w (len 2n-1) are real vectors.
    """
    n = a.shape[0]
    # reverse w for the correlation needed in the gradient
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)   # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (smooth max) helpers
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights: w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initialisation helpers – several families of usable start‑vectors
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None, n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L1‑norm √(2·n).
    If a base_seq is supplied, a (interpolated) copy of it is used;
    otherwise one of a few parametric families or a Dirichlet draw is chosen.
    """
    S = np.sqrt(2.0 * n)

    # 1) reuse a previous solution (interpolated)
    if base_seq is not None and base_seq.size >= 2:
        # linear interpolation to required length
        idx = np.linspace(0, base_seq.size - 1, n)
        a = np.interp(idx, np.arange(base_seq.size), base_seq)
        a = np.maximum(a, 0.0)
        a = a / a.sum() * S
        return a

    # 2) choose a random family
    fam = rng.integers(0, 5)
    if fam == 0:
        # uniform Dirichlet(1) – random simplex point
        a = rng.random(n)
    elif fam == 1:
        # geometric decay a_i ∝ r^i
        r = rng.uniform(0.90, 0.9995)
        a = np.power(r, np.arange(n, dtype=float))
    elif fam == 2:
        # power‑law a_i ∝ (i+1)^(-p)
        p = rng.uniform(0.3, 2.0)
        a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
    elif fam == 3:
        # linear descending (shifted to stay positive)
        start = rng.uniform(0.0, 1.0)
        stop = rng.uniform(0.0, 0.5)
        a = np.linspace(start, stop, n)
    else:
        # a single spike (one element gets the whole mass)
        a = np.zeros(n, dtype=float)
        a[rng.integers(0, n)] = 1.0

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Optimiser – Phase 1: Adam on a smooth max
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of max(a∗a).
    Returns the best sequence encountered and its true max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # scheduling
    alpha = 0.7           # initial soft‑max temperature
    alpha_mul = 1.001     # increase per iteration
    lr0 = 0.55            # base learning rate
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)

        # soft‑max weights
        w = _smooth_weights(conv, alpha)

        # gradient of the smoothed objective
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)   # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetrisation – the optimum is usually symmetric
        a = (a + a[::-1]) * 0.5

        # project back onto the simplex
        a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # schedule updates
        alpha *= alpha_mul

        # early exit if stagnating (very cheap check)
        if it >= 3000:
            # after a few thousand steps the improvement stalls for most runs
            if cur >= best_score * 0.9999:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimiser – Phase 2: Sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray, deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a) = max_k (a∗a)[k].
    Uses a simple momentum term and a decaying stepsize.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # momentum
    mom = np.zeros_like(a)
    beta = 0.9
    lr = 0.45
    iter_no_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k = int(np.argmax(conv))                # index of current worst entry
        w = np.zeros_like(conv)
        w[k] = 1.0                               # exact max‑weight

        grad = _conv_with_weights(a, w)

        # momentum update
        mom = beta * mom + (1.0 - beta) * grad
        a = a - lr * mom

        # enforce symmetry (helps reduce the peak at the centre)
        a = (a + a[::-1]) * 0.5

        a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        lr *= 0.9995  # gentle decay

        # stop after many non‑improving steps
        if iter_no_improve > 2500:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Small random “mass‑swap’’ local search
# ----------------------------------------------------------------------


def _random_swap_refine(a0: np.ndarray, deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic local moves: move a fraction of the mass from one index
    to another and keep the change if it improves the max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] == 0.0:
            continue
        # move a random fraction of the source mass
        delta = a[i] * rng.uniform(0.01, 0.07)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cur = np.max(_full_convolution(cand))
        if cur < best_score:
            best_score = cur
            best_a = cand.copy()
            a = cand                 # continue from the improved point
    return best_a, best_score

# Full optimisation of a single restart
# ----------------------------------------------------------------------


def _optimise_one_restart(a0: np.ndarray, full_deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the three optimisation phases (Adam → sub‑gradient → random swap)
    within the allotted time.
    """
    now = time.time()
    # split the remaining budget roughly 5 s / 3, 5 s / 3, remaining
    phase1_dead = min(now + 5.0, full_deadline)
    a_mid, sc_mid = _adam_optimize(a0, phase1_dead, rng)

    now = time.time()
    phase2_dead = min(now + 5.0, full_deadline)
    a_mid2, sc_mid2 = _subgradient_optimize(a_mid, phase2_dead, rng)

    now = time.time()
    a_final, sc_final = _random_swap_refine(a_mid2, full_deadline, rng)

    # pick the best among the three checkpoints
    candidates = [(a0, np.max(_full_convolution(a0))),
                  (a_mid, sc_mid),
                  (a_mid2, sc_mid2),
                  (a_final, sc_final)]
    best_seq, best_score = min(candidates, key=lambda t: t[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises the evaluation score.
    The routine respects the 1000 s budget, runs many short independent
    restarts (different lengths, initialisations and optimiser runs) and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered solution
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # pick a problem size – we sample uniformly in a fairly wide band
        n = rng.integers(800, 5400)      # up to ~5 k entries (still cheap)

        # initialise – sometimes reuse the previous solution (interpolated)
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a time slice for this attempt
        slice_deadline = min(time.time() + 12.0, deadline)

        # run the three‑phase optimiser
        a_opt, score = _optimise_one_restart(a0, slice_deadline, rng)

        # update the global best
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} "
                  f"new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # extremely unlikely – fallback to a trivial feasible vector
        return [1.0]

    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal max = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
