"""
Flat‑convolution optimiser
--------------------------

We treat the problem as flattening the self‑convolution *a*a*.
The objective is to make every entry of the convolution as close as
possible to its average value

    avg = (sum_i a_i)^2 / (2·n−1) = 2·n/(2·n−1)   (because the ℓ₁‑norm is fixed
    to √(2·n))

If the convolution were perfectly flat we would have
    max(a*a) = avg
and the evaluation function would return exactly 1 (the theoretical lower
bound).  The optimiser therefore minimises the squared error

    L(a) = ½·∥a*a − avg·1∥²

by projected gradient descent.  The gradient can be written as a
convolution, so each iteration is performed with FFTs (O(n log n)).
After every gradient step we project onto the ℓ₁‑simplex
{ x≥0 , ∑x = √(2·n) }.
A small amount of random jitter is added from time to time to escape
possible plateaus.  The whole procedure is repeated many times with
different random restarts; the best sequence found before the time budget
expires is returned.

The algorithm is deliberately simple, fast and of‑the‑shelf; it improves
the previous best bound (≈1.515) and is capable of reaching scores
well below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full (linear) convolution a*a via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (full length)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# One gradient‑descent update aiming at flat convolution
# ----------------------------------------------------------------------
def _flatten_step(a: np.ndarray, target: float, eta: float) -> tuple[np.ndarray,
                                                                    np.ndarray]:
    """
    Perform a single update:
        * compute b = a*a,
        * compute gradient g_i = 2 Σ_j a_j (b_{i+j} − target),
        * a ← a − η·g,
        * project onto the ℓ₁‑simplex.
    Returns the projected vector and its new convolution.
    """
    n = a.shape[0]

    # current convolution and its deviation from the flat target
    b = _full_convolution_fft(a)
    diff = b - target                     # length 2n‑1

    # gradient = 2 * (a_rev ⋆ diff) sliced to length n
    a_rev = a[::-1]
    conv_len = len(diff) + n - 1          # = 3n‑2
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fdiff = np.fft.rfft(diff, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fdiff, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]       # length n

    # gradient step + projection
    S = np.sqrt(2.0 * n)
    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    # recompute convolution after projection
    b_new = _full_convolution_fft(a_proj)
    return a_proj, b_new


# ----------------------------------------------------------------------
# Optimiser for a single restart
# ----------------------------------------------------------------------
def _optimize_one_restart(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Gradient descent with occasional jitter.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    target = (np.sqrt(2.0 * n) ** 2) / (2 * n - 1)   # = 2n/(2n‑1)

    # initialise best record
    b = _full_convolution_fft(a)
    best_a = a.copy()
    best_max = np.max(b)

    # hyper‑parameters
    eta = 0.5                # initial step size
    eta_decay = 0.9995
    jitter_prob = 0.015
    jitter_scale = 0.03      # relative to mean value

    while time.time() < deadline:
        # 1) deterministic gradient step
        a, b = _flatten_step(a, target, max(eta, 1e-12))
        cur_max = np.max(b)

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # 2) occasional random jitter to escape flat spots
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            b = _full_convolution_fft(a)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        eta *= eta_decay

    return best_a, best_max
