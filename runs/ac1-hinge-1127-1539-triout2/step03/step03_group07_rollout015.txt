"""
Improved search for a low‑score step function.

The algorithm works in two phases:

1. **Projected sub‑gradient descent** on the exact max‑convolution.
   At each iteration the full self‑convolution is computed (FFT),
   the indices attaining the maximum are identified and the exact
   sub‑gradient of the max‑function is built.
   After a gradient step the vector is projected onto the simplex
   (fixed L1‑norm = √(2·n)).  Many random restarts are performed and the
   best vector (lowest max‑convolution) is kept.

2. **Linear‑programming refinement**.
   For the current best vector `a` we solve the LP

        maximise   Σ g_j
        s.t.       (a * g)_k ≤ M    for all k,
                   g_j ≥ 0,

   where `M = max(a*a)`.  The LP finds a direction `g` that can add a
   large amount of mass without increasing the cross‑convolution with `a`.
   The solution is normalised to the same L1‑norm and either replaces `a`
   (if its max‑convolution improves) or is mixed with `a`.  The process
   is repeated while time permits.

Both phases respect the non‑negativity and L1‑norm constraints that the
evaluation function uses.  The final vector is returned as a Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_full_and_max(a: np.ndarray):
    """Return the full linear self‑convolution of a and its maximum value."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv, float(np.max(conv))


def conv_max(a: np.ndarray) -> float:
    """Convenient wrapper returning only the maximum entry of a*a."""
    _, m = conv_full_and_max(a)
    return m


def project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_max_conv(a: np.ndarray, conv_full: np.ndarray, cur_max: float):
    """
    Exact sub‑gradient of the max‑convolution at a.
    For each shift k attaining the maximum we add the reversed slice of a.
    """
    n = a.shape[0]
    shifts = np.where(np.abs(conv_full - cur_max) <= 1e-12)[0]
    g = np.zeros_like(a)
    for k in shifts:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        partner = k - i
        g[i] += a[partner]
    return g


def subgradient_descent(a0: np.ndarray, deadline: float,
                        max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max‑convolution.
    Returns the best vector found and its max‑convolution value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # target L1‑norm
    a = a0.copy()
    best_a = a.copy()
    best_val = conv_max(a)

    eta = 0.5                     # initial step size
    decay = 0.9995                # geometric decay

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv_full, cur_max = conv_full_and_max(a)
        # build exact sub‑gradient
        g = subgradient_max_conv(a, conv_full, cur_max)

        # gradient step and projection onto simplex
        a = a - eta * g
        a = project_simplex(a, S)

        # evaluate after projection
        new_max = conv_max(a)
        if new_max < best_val - 1e-12:
            best_val = new_max
            best_a = a.copy()
            print(f"[sgd improve] time={time.time():.1f}s n={n} it={it} max={new_max:.6f}")

        # step‑size schedule
        eta *= decay

    return best_a, best_val


def build_conv_matrix(a: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A·g)_k = (a * g)[k].
    The matrix consists of n diagonals; diagonal i (offset i) is filled
    with the constant a_i.
    """
    n = a.shape[0]
    # each diagonal has length n (the minimal dimension of the strip)
    diags = [np.full(n, a_i) for a_i in a]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def lp_refine(a: np.ndarray, deadline: float,
              max_steps: int = 12) -> np.ndarray:
    """
    Linear‑programming refinement.
    Starting from a, repeatedly solve the LP that maximises additional
    L1‑mass while keeping the cross‑convolution bounded by the current
    maximum.  The LP solution is normalised to the target L1‑norm and,
    if it improves the max‑convolution, replaces a.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        M = conv_max(a)
        A = build_conv_matrix(a)
        b = np.full(2 * n - 1, M)

        # maximise Σ g_j  <=> minimise -Σ g_j
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b,
                      bounds=(0, None), method='highs')
        if not res.success:
            print("[lp] linprog failed")
            break

        g = res.x
        # normalise g to the same L1‑norm as a
        g = project_simplex(g, S)
        M_g = conv_max(g)

        if M_g < M - 1e-9:
            a = g
            print(f"[lp replace] step={step} max={M_g:.6f}")
            continue

        # try a simple average of a and g
        a_mix = (a + g) / 2.0
        a_mix = project_simplex(a_mix, S)
        M_mix = conv_max(a_mix)

        if M_mix < M - 1e-9:
            a = a_mix
            print(f"[lp mix] step={step} max={M_mix:.6f}")
            continue

        # no improvement – stop
        break

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the time budget, runs many gradient‑descent restarts
    and finishes with a linear‑programming refinement.
    Returns a Python list of heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional: load a previously discovered construction
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] height_sequence_1 score={sc:.6f} n={cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase A – projected sub‑gradient descent with many restarts
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000]   # candidate lengths
    while time.time() < deadline:
        # pick a length (favoring larger vectors a little)
        n = int(rng.choice(lengths,
                           p=np.array([0.2, 0.2, 0.25, 0.25, 0.1])))
        conc = rng.uniform(0.05, 1.0)          # Dirichlet concentration
        a0 = rng.dirichlet(np.full(n, conc))
        a0 = a0 * np.sqrt(2.0 * n)            # enforce Σa = √(2n)

        a_opt, max_conv = subgradient_descent(a0, deadline)

        score = max_conv                       # because Σa = √(2n)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global improve] n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – linear‑programming refinement (only if we have a candidate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[LP refinement start]")
        refined = lp_refine(best_seq, deadline)
        refined_score = conv_max(refined)      # sum is fixed → score = max
        if refined_score < best_score - 1e-12:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"[LP improved] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event we have no sequence yet
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
