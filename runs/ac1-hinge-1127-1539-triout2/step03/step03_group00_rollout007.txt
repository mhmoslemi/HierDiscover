def _random_initial_sequence_family(n, rng):
    # pick family
    choice = rng.integers(0,6)
    idx = np.arange(n, dtype=float)
    if choice == 0:
        # power law
        p = rng.uniform(0.05, 2.0)
        a = (idx + 1) ** (-p)
    elif choice == 1:
        # exponential
        lam = rng.uniform(0.0005, 0.02)  # small lam for slower decay
        a = np.exp(-lam * idx)
    elif choice == 2:
        # mixed
        p = rng.uniform(0.05, 2.0)
        lam = rng.uniform(0.0005, 0.02)
        a = (idx+1)**(-p) * np.exp(-lam * idx)
    elif choice == 3:
        # beta shape (U-shaped)
        alpha = rng.uniform(0.0,3.0)
        beta = rng.uniform(0.0,3.0)
        a = (idx+1)**alpha * (n - idx)**beta
    elif choice == 4:
        # Dirichlet
        a = rng.dirichlet(alpha=np.ones(n))
    else:
        # uniform jitter
        a = np.full(n, 1.0)
    # ensure nonnegativity
    a = np.maximum(a,0.0)
    # Renormalize to simplex
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)
    return a

"""
A fast, multi‑phase heuristic for Min‑max self‑convolution.

The objective after ℓ¹‑normalisation is simply  max(a * a) .
We search for a non‑negative vector a with
    Σ a_i = √(2·n)
that minimises this maximum.

The search consists of:

1. **Randomised families** – power‑law, exponential, mixed,
   beta‑shaped (U‑shape) and Dirichlet draws.  Each family is
   sampled many times on several lengths and the few best seeds are kept.

2. **Projected sub‑gradient descent** – we minimise the non‑smooth
   max‑convolution directly by repeatedly moving mass away from the
   indices contributing to the current maximal entry.  The step is
   scaled by the gradient norm and projected onto the ℓ¹‑simplex.

3. **Simulated‑annealing refinement** – a light SA that transfers a
   random fraction of mass between two positions while accepting
   uphill moves with a slowly cooling temperature.

All phases respect the time budget (default 1000 s).  The best sequence
found is returned as a plain Python list; it can be passed to the
provided ``evaluate_sequence`` function for scoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities for the linear convolution a*a
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _max_convolution_fft_and_index(a: np.ndarray):
    """Return (max value, index of max, full convolution) of a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx, conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score: max entry of a*a (the quantity to be minimised)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective = max(a*a) after ℓ¹‑normalisation."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Random initialisers – a handful of parametric families
# ----------------------------------------------------------------------


def _random_initial_sequence_family(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a non‑negative sequence on the ℓ¹‑simplex from random families."""
    target_sum = np.sqrt(2.0 * n)
    choice = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if choice == 0:                           # pure power‑law
        p = rng.uniform(0.05, 2.0)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                         # exponential decay
        lam = rng.uniform(0.0005, 0.02)
        a = np.exp(-lam * idx)

    elif choice == 2:                         # mixed power‑law + exponential
        p = rng.uniform(0.05, 2.0)
        lam = rng.uniform(0.0005, 0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                         # beta‑shaped (U‑shape)
        alpha = rng.uniform(0.0, 3.0)
        beta = rng.uniform(0.0, 3.0)
        a = (idx + 1.0) ** alpha * (n - idx) ** beta

    elif choice == 4:                         # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    else:                                     # uniform + tiny jitter
        a = np.full(n, 1.0)

    # guarantee non‑negativity, add a little Gaussian noise and project
    a = np.maximum(a, 0.0)
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> np.ndarray:
    """
    Sub‑gradient descent that follows the gradient of the active (max) entry.
    Momentum and a decreasing step size are used, with periodic jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    rho = 0.9                       # momentum decay
    momentum = np.zeros_like(a)
    step_base = target_sum * 0.2    # base magnitude for learning rate
    it = 0

    while time.time() < deadline:
        maxb, kmax, _ = _max_convolution_fft_and_index(a)

        # sub‑gradient of the active entry (2 * a reversed at shift kmax)
        g = np.zeros_like(a)
        start_i = max(0, kmax - (n - 1))
        end_i = min(kmax, n - 1)
        idxs = np.arange(start_i, end_i + 1, dtype=int)
        g[idxs] = 2.0 * a[kmax - idxs]

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm_m * np.sqrt(it + 1))
        a -= lr * momentum

        # project back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional small Gaussian jitter to escape plateaus
        if it % 400 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (np.sqrt(it + 1) + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> np.ndarray:
    """Mass‑transfer SA that accepts uphill moves with a geometric cooling."""
    a = a0.copy()
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    # initialise temperature as a small fraction of the current score
    T = cur_score * 0.02 + 1e-8
    cooling = 0.9996

    n = a.size
    while time.time() < deadline:
        # pick two distinct indices (i has mass, j receives mass)
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 15 %) of a[i] to a[j]
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    competition objective.  The routine respects the given time budget,
    explores several vector lengths, keeps a small pool of promising seeds,
    refines them with projected sub‑gradient descent and simulated annealing,
    and finally returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Initialise from any previously discovered global sequence
    # ------------------------------------------------------------------
    init_global = globals().get("height_sequence_1")
    if isinstance(init_global, (list, tuple, np.ndarray)):
        a = np.asarray(init_global, dtype=float)
        n0 = a.size
        a = _project_to_simplex(a, np.sqrt(2.0 * n0))
        sc = _score(a)
        best_seq = a.copy()
        best_score = sc
        print(f"[{time.time():.1f}] Starting from global height_sequence_1 (n={n0}) "
              f"score={sc:.6f}")

    # ------------------------------------------------------------------
    # Vector lengths to explore – larger n gives more freedom
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1500, 2000, 2500, 3000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        top_candidates = []                      # (score, vector) tuples

        # ------------------------------------------------------------------
        # Phase A – random sampling from several families
        # ------------------------------------------------------------------
        phase_end = min(deadline, time.time() + 8.0)   # short burst per length
        while time.time() < phase_end:
            a0 = _random_initial_sequence_family(n, rng)
            sc = _score(a0)

            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={sc:.6f}")

            # keep a few best random seeds for deeper work
            if len(top_candidates) < 5:
                top_candidates.append((sc, a0.copy()))
                top_candidates.sort(key=lambda x: x[0])
            elif sc < top_candidates[-1][0]:
                top_candidates[-1] = (sc, a0.copy())
                top_candidates.sort(key=lambda x: x[0])

        # ------------------------------------------------------------------
        # Phase B – refinement of the few most promising seeds
        # ------------------------------------------------------------------
        for _, a_seed in top_candidates:
            if time.time() >= deadline:
                break

            # allocate a bounded slot for sub‑gradient descent
            sub_dead = min(deadline, time.time() + 20.0)
            a_gd = _subgradient_descent(a_seed, sub_dead, rng)

            # remaining time (if any) goes to SA
            a_refined = _simulated_annealing(a_gd, deadline, rng)

            sc = _score(a_refined)
            if sc < best_score:
                best_score = sc
                best_seq = a_refined.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} "
                      f"score={sc:.6f}")

        # clear the pool before moving to the next length
        top_candidates.clear()

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_seq = a
        best_score = _score(a)

    # Trim insignificant trailing zeros for a tidy answer
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
