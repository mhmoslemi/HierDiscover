```
    eta = eta0

        conv_full = conv_fft(a)

        # find all indices within tolerance

    return best_a, best_val
```

```
def _simplex_projection(v, S):
    # (project onto nonnegative simplex of sum S)
    if v.sum() == 0:
        return np.full_like(v, S / len(v))
    # shift to non-negative?
    # Use algorithm: sort v desc, find rho, tau.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_onto_simplex(v, s):
    # ensure non-negative output that sums to s
    # reference: Wang & Carreira-Perpinán 2013
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and abs(v.sum() - s) < 1e-12:
        return v
    n = v.shape[0]

    else:

        theta = cssv[rho] / (rho + 1)

"""
Propose a non‑negative step‑function that minimises the maximum entry of its
self‑convolution (the value returned by ``evaluate_sequence``).

Algorithm
---------
1. **Exact projected gradient descent**  
   The objective `f(a)=max_k (a*a)[k]` is convex.  Its sub‑gradient at the
   current point is the sum of the gradients of all convolutions that attain the
   maximum.  The gradient of a single convolution with shift `k` is simply
   `2*a_{k‑i}` for each coordinate `i`.  We therefore
   * compute the full convolution by FFT,
   * collect the active shift indices,
   * build the sub‑gradient,
   * take a (negative) gradient step,
   * and project back onto the simplex `{a_i≥0, Σa_i=√(2·n)}`.
   The step size decays slowly; if a step does not improve the objective the
   step size is halved.  Occasional random jitter helps to escape plateaus.

2. **Local stochastic hill‑climbing**  
   From the best vector found by the gradient phase we run a short simulated‑
   annealing style hill‑climber that moves a random fraction of mass from one
   index to another.  Moves that improve the maximum convolution are always
   accepted; worsening moves are accepted with probability `exp(-Δ/τ)` where the
   temperature `τ` decays geometrically.

3. **Repeated restarts of varying length**  
   The whole two‑phase routine is executed many times with random Dirichlet
   initialisations and lengths taken from `[500, 1 000, 2 000, 4 000]` until the
   time budget (≈1000 s) is exhausted.  The best sequence over all runs is
   returned.  If a previously discovered sequence `height_sequence_1` exists it
   is used as an initial baseline.

All operations respect the non‑negativity and clipping performed inside
``evaluate_sequence``; the returned list is therefore a valid candidate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_linear(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of the next power of two."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return float(np.max(_conv_linear(a)))


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max_k (a*a)[k]` with respect to `a`,
    summed over the supplied shift indices.
    For a shift `k` the contribution to the gradient is 2·a_{k‑j}
    wherever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g


# ----------------------------------------------------------------------
# Projection onto the simplex Σa = S, a_i ≥ 0
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = S}."""
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and abs(v.sum() - S) <= 1e-12:
        return v
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected (additive) gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _run_proj_gradient(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       eta0: float = 0.08,
                       decay: float = 0.9999,
                       max_iters: int = 4000,
                       jitter_prob: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Exact projected gradient descent on `f(a)=max_k (a*a)[k]`.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)

    cur_max = _conv_max(a)
    best_a = a.copy()
    best_val = cur_max
    eta = eta0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution and active shift set
        conv = _conv_linear(a)
        cur_max = float(conv.max())
        active = np.where(conv >= cur_max - 1e-12)[0]

        # sub‑gradient from all active shifts
        g = _gradient_from_shifts(a, active)

        # propose a gradient step and project back to the simplex
        a_new = a - eta * g
        a_new = _project_onto_simplex(a_new, S)
        new_max = _conv_max(a_new)

        if new_max <= cur_max:          # accept improvement
            a = a_new
            cur_max = new_max
            if new_max < best_val - 1e-12:
                best_val = new_max
                best_a = a.copy()
                print(f"[proj‑grad improve] t={time.time():.1f}s n={n} it={it} max={new_max:.6f}")
        else:                           # reject – shrink the step size
            eta *= 0.5

        # schedule
        eta *= decay

        # occasional jitter to explore flat regions
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a = _project_onto_simplex(a, S)
            cur_max = _conv_max(a)

    return best_a, best_val

def _local_hill_climb(a_start: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 12000,
                      cooling: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Simple annealing procedure that moves a random fraction of mass
    from one index to another.  Returns the best vector encountered.
    """
    a = a_start.copy()
    cur_max = _conv_max(a)
    best_a = a.copy()
    best_max = cur_max
    temperature = max(0.05 * cur_max, 1e-4)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        frac = rng.uniform(0.05, 0.20)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _conv_max(a_new)

        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max

            cur_max = _conv_max(a)

        # --- Phase 1: exact projected gradient descent -------------------------------
        a_opt, val = _run_proj_gradient(
            a0, n, deadline,
            eta0=rng.uniform(0.04, 0.12),
            decay=rng.uniform(0.9995, 0.99995),
            max_iters=4000,
            jitter_prob=0.015,
        )

        # --- Phase 2: local hill‑climbing -------------------------------------------
        a_opt2, val2 = _local_hill_climb(a_opt, n, deadline, rng, max_iters=12000)

        # --- Record improvement ------------------------------------------------------
        if val2 < best_score - 1e-12:
            best_score = val2
            best_seq = a_opt2.copy()
            print(f"[global improve] n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
