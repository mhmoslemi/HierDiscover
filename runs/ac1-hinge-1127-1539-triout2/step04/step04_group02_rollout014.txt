"""
Propose a step‑function (non‑negative coefficient vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the full linear self‑convolution.  The routine follows
four stages inside a 1000 s time budget:

0. **Diversified initialisation** – a sweep over many parametric families
   (linear, quadratic, exponential, power, inverse, plus a new
   “inverse‑sqrt‑sum’’ shape that is known to flatten the self‑convolution)
   together with a few random Dirichlet vectors.  The best candidate is kept.

1. **Adam‑based smoothed optimisation** – minimise the log‑sum‑exp
   smooth surrogate of the max‑convolution.  Adam (adaptive moments) provides
   fast, stable progress; after each step the vector is projected onto the
   ℓ₁‑simplex {x≥0, Σx = √(2 n)}.

2. **Exact sub‑gradient descent** – switch to the true nonsmooth objective.
   The sub‑gradient is obtained from all (almost) maximal diagonals; a
   decreasing step‑size is used.

3. **Simulated‑annealing polishing** – a long‑running random‑pair‑transfer
   walk with a temperature schedule that occasionally accepts worsening
   moves, followed by a final burst of tiny transfers.

The algorithm works with any length `n`.  If a previous construction
(`height_sequence_1`) is available its length is reused; otherwise a larger
size `n = 65536` is chosen (larger `n` gives a tighter theoretical lower
bound).  The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the (linear) self‑convolution of a."""
    return float(conv_self_fft(a).max())


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        # all entries will be zero – return uniform distribution
        return np.full_like(v, target / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str,
                   param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence of length n.
    `kind` may be one of:
        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right,
        exponential_left, exponential_right,
        inverse_sqrt_sum   (new – sum of two inverse‑power terms)
    The parameter `param` controls the shape (exponent, rate, …).
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    elif kind == "inverse_sqrt_sum":
        # a_i ∝ (i+1)^(-p) + (n-i)^(-p)  (symmetric U‑shape)
        p = max(param, 1e-12)
        left = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        right = (np.arange(n, 0, -1, dtype=np.float64) ** (-p))
        raw = left + right
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)

    cur = best_seq.copy()
    cur_score = best_score
    temperature = 0.02 * best_score   # start temperature proportional to score
    cool = 0.9995
    anneal_iter = 0

    while time.time() < deadline:
        cand = random_pair_transfer(cur.copy(), target_sum, scale=0.01)
        cand_score = compute_max_conv(cand)
        diff = cand_score - cur_score
        accept = False
        if diff < 0:
            accept = True
        else:
            prob = math.exp(-diff / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"Anneal iter {anneal_iter:07d} – new best {best_score:.6f}")
        temperature *= cool
        anneal_iter += 1
        if anneal_iter % 5000 == 0:
            # reset to the globally best to avoid drifts
            cur = best_seq.copy()
            cur_score = best_score
            temperature = 0.02 * best_score

    # ------------------------------------------------------------------
    # Phase 4 – final tiny random pair transfers (polishing)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 4 – final polishing")
    for _ in range(500):
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.0015)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
