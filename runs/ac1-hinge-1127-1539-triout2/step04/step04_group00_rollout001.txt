def smooth_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2*n)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2
    it = 0
    best_a = a.copy()
    best_score = np.max(conv_full(a))
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = conv_full(a)
        # compute smooth weights with alpha schedule
        alpha = min(200.0, 0.1 * it)  # gradually increase
        w = smooth_weights(conv, alpha)

        # Adam update
        lr = lr0 * np.sqrt(1 - beta2**it) / (1 - beta1**it)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)

            best_a = a.copy()

            break
    return best_a, best_score

def anneal_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = sqrt(2n)
    best_a = a.copy()
    best_score = np.max(conv_full(a))
    # define schedule: start T = initial_temp (maybe 0.1) decays to 0.
    start_T = 0.5
    total = deadline - time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # time fraction
        elapsed = time.time() - (deadline - total)
        T = start_T * (1 - elapsed/total)  # linear decay
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] == 0: continue
        delta = rng.random() * 0.05 * a[i]  # up to 5% of mass at i
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand_score = np.max(conv_full(cand))
        cur_score = np.max(conv_full(a))
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score)/max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

def full_convolution(a):
    n = a.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

        a2, s2 = _subgradient_optimize(a1, now + slice_time + min(8.0, remaining - slice_time - 5.0), rng)
        # anneal short
        a3, s3 = _anneal_optimize(a2, deadline, rng)   # remaining time

            best_seq = a3.copy()
            last_improve = time.time()
            print(f"[global] n={n} new best score={best_score:.6f} time={last_improve-start:.1f}s")
    # final
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external_score={external_score:.6f} internal_best={best_score:.6f} length={len(final_seq)}")
    return final_seq

"""
Improved step‑function optimizer.

The objective of ``evaluate_sequence`` is (after normalising ∑a_i=√(2·n))
simply the maximum entry of the self‑convolution a∗a.  We minimise this
maximum by a three‑phase optimisation:

1. **Adam on a soft‑max approximation** – the max is replaced by a
   log‑sum‑exp with a temperature α that is slowly increased up to 200.
   Adam’s per‑coordinate step‑size together with projection onto the
   simplex gives fast, stable descent.

2. **Exact sub‑gradient descent** – after the smooth phase we focus on the
   currently worst shift k*.  The sub‑gradient at this shift is known in
   closed form (g_i = a_{k*−i}); a decreasing step‑size 𝜂∝1/√t is used.

3. **Simulated annealing (mass‑swap moves)** – a short stochastic phase
   explores the neighbourhood of the current best point and helps escape
   shallow plateaux.

The optimiser works for any length n (chosen randomly between 1500 and
4000) and repeatedly restarts when no improvement is seen for a while.
All updates keep the vector on the simplex {x≥0, Σx=√(2·n)}; the final
sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max‑norm.
    g_j = 2 Σ_i w_{i+j} a_i   (j = 0…n−1)
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generator (uniformly random on the simplex)
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If ``base_seq`` is supplied and long enough, its leading part (with
    tiny jitter) is used; otherwise a Dirichlet(1) sample is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)               # uniform on simplex

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Phase 1 – Adam on the smooth max‑norm
# ----------------------------------------------------------------------


def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimisation on a soft‑max approximation of the max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.25

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)

        # temperature slowly grows to concentrate on the true max
        alpha = min(200.0, 0.05 * it)   # linear ramp, cap at 200
        w = _smooth_weights(conv, alpha)

        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # crude early stop
        if it % 4000 == 0 and best_score == cur_score:
            break

    return best_a, best_score

# Phase 2 – exact sub‑gradient descent on the max‑norm
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent directly on the worst convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # Sub‑gradient for the worst shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = 0.15 / np.sqrt(it)          # decreasing step size
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:

            best_a = a.copy()

        if rng.random() < 0.05:

            if i != j and a[i] > 0:

                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)

                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 6000 == 0 and best_score == cur_score:

    return best_a, best_score

def _anneal_optimize(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis‑Hastings moves to explore the neighbourhood."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    start = time.time()
    total = max(deadline - start, 1e-6)
    it = 0

    while time.time() < deadline:
        it += 1
        # linear temperature schedule (starts at 0.5, ends near 0)
        elapsed = time.time() - start
        T = max(1e-4, 0.5 * (1.0 - elapsed / total))

        # simple mass‑transfer move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j and a[i] > 0:
            delta = rng.random() * 0.05 * a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand, S)

            cur_score = np.max(_full_convolution(a))
            cand_score = np.max(_full_convolution(cand))

            if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / T):
                a = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()

        # once in a while try to reduce the contribution of the
        # presently worst shift directly
        if rng.random() < 0.02:
            conv = _full_convolution(a)
            k = int(np.argmax(conv))
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            idx_i = np.arange(i_low, i_high + 1)
            idx_j = k - idx_i
            contrib = a[idx_i] * a[idx_j]
            worst = np.argmax(contrib)
            i_w = idx_i[worst]
            j_w = idx_j[worst]
            if a[i_w] > 0:
                delta = rng.random() * 0.1 * a[i_w]
                cand2 = a.copy()
                cand2[i_w] -= delta
                cand2[j_w] += delta
                cand2 = _project_simplex(cand2, S)

                if cand2_score < best_score:
                    a = cand2
                    best_score = cand2_score
                    best_a = a.copy()

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The optimiser consists of a smooth‑max Adam phase, a sub‑gradient phase
    and a short annealing refinement.  Restarts are performed when progress
    stalls; the best sequence found within the time budget is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load previously discovered construction if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    last_improve_time = start_time
    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # ------------------------------------------------------------------
        # Choose problem size (length of the step‑function)
        # ------------------------------------------------------------------
        if best_overall_seq is None or (time.time() - last_improve_time > 30):
            n = rng.integers(1500, 4000)          # fresh random start
            a0 = _initial_sequence(base_seq, n, rng)
        else:
            # continue refining the current best
            a0 = best_overall_seq.copy()
            n = a0.shape[0]

        # ------------------------------------------------------------------
        # Allocate a slice of the remaining budget for this attempt
        # ------------------------------------------------------------------
        now = time.time()
        time_left = deadline - now
        if time_left < 12.0:
            break  # not enough time for a useful pass
        # 60 % smooth, 30 % sub‑gradient, 10 % annealing (approx.)
        smooth_deadline = now + 0.6 * time_left
        subgrad_deadline = smooth_deadline + 0.3 * time_left
        anneal_deadline = deadline

        # ------------------------------------------------------------------
        # Run the three optimisation phases
        # ------------------------------------------------------------------
        a_s, score_s = _smooth_optimize(a0, smooth_deadline, rng)
        a_sg, score_sg = _subgradient_optimize(a_s, subgrad_deadline, rng)
        a_final, score_final = _anneal_optimize(a_sg, anneal_deadline, rng)

        # ------------------------------------------------------------------
        # Record global best
        # ------------------------------------------------------------------
        if score_final < best_overall_score:
            best_overall_score = score_final
            best_overall_seq = a_final.copy()
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={n} "
                  f"new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]                 # should never happen
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
