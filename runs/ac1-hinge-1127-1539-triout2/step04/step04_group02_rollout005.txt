Propose a step‑function (non‑negative height vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

which, after normalising the ℓ₁‑norm to √(2 n), is simply the maximal
value of the (linear) self‑convolution.  The search consists of

1. **Broad initial sweep** – several parametric families (linear,
   quadratic, exponential, power, inverse, Gaussian, sine) together
   with a few Dirichlet‑draws.  All candidates are projected onto the
   simplex {x≥0, Σx = √(2 n)} and the best one is kept.

2. **Smoothed Adam optimisation** – minimise a log‑sum‑exp smooth
   surrogate of the max using the Adam optimiser.  The smoothing
   parameter τ starts as a fraction of the current maximum and is
   reduced geometrically; the learning‑rate is also decayed.  After each
   step the vector is projected back onto the simplex.

3. **True sub‑gradient phase** – once τ becomes tiny we switch to the
   genuine non‑smooth objective.  The sub‑gradient is the average of the
   gradients at all (approximately) maximal lags.

4. **Polishing** – many tiny random pair‑mass transfers are tried;
   every improvement is kept.

The routine respects the 1000 s budget, logs progress and finally
returns the best sequence found (as a plain Python list).
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]               # sorted descending
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(
    n: int, target_sum: float, kind: str, param: float
) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence of length n.
    Kinds supported: linear, quadratic, exponential, power, inverse,
    linear_left, linear_right, exponential_left, exponential_right,
    gaussian, sine.
    The resulting vector is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-6
    elif kind == "inverse":
        raw = 1.0 / (1e-6 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    elif kind == "gaussian":
        sigma = param * n          # param is a fraction of n (0 < param < 0.5)
        raw = np.exp(-0.5 * ((idx - mid) / sigma) ** 2)
    elif kind == "sine":
        raw = np.sin(np.pi * idx / (n - 1)) + 0.001   # ensure positivity
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a (non‑negative)."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    evaluate_sequence objective.  Returns the best vector discovered
    before the 1000 s budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25          # small safety margin

    # --------------------------------------------------------------
    # Problem size (use an existing construction if available)
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.asarray(base, dtype=np.float64)
        init_seq = project_onto_simplex(init_seq, math.sqrt(2.0 * n))
    else:
        n = 16384                      # default length (power‑of‑two for FFT)
        init_seq = None

    target_sum = math.sqrt(2.0 * n)    # we keep Σa = √(2 n) → objective = max(conv)

    # --------------------------------------------------------------
    # Phase 0 – diverse initial candidates
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)   # uniform start
    best_score = compute_max_conv(best_seq)

    print(f"[{time.time():.2f}] Phase 0 – initial sweep (n={n})")

    # ------------------------------------------------------------------
    # 0.1  parametric families
    # ------------------------------------------------------------------
    shape_kinds = (
        "linear", "quadratic", "exponential",
        "power", "inverse",
        "linear_left", "linear_right",
        "exponential_left", "exponential_right",
        "gaussian", "sine",
    )
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),          # step 0.2
        "quadratic": np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 12),
        "inverse": np.linspace(0.5, 5.0, 12),
        "linear_left": np.linspace(0.0, 0.9, 19),
        "linear_right": np.linspace(0.0, 0.9, 19),
        "exponential_left": np.linspace(0.0, 6.0, 31),
        "exponential_right": np.linspace(0.0, 6.0, 31),
        "gaussian": np.linspace(0.04, 0.25, 12),     # sigma as fraction of n
        "sine": np.array([0.0]),                     # one dummy value (no param)
    }

    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind:12s} (p={p:6.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during parametric sweep.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 0.2  random Dirichlet‑like vectors
    # ------------------------------------------------------------------
    for _ in range(20):
        cand = np.random.rand(n)
        cand = project_onto_simplex(cand, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best random init → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during random init.")
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # 0.3  supplied baseline (if any)
    # ------------------------------------------------------------------
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves start → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Smoothed Adam optimisation
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.55
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = target_sum / (8.0 * n)    # modest initial learning rate
    tau_factor = 0.18
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Phase 1 – smoothed Adam")
    while time.time() < phase1_deadline:
        # current convolution and smoothing parameter
        conv = conv_self_fft(a)
        max_conv = conv.max()
        tau = max(tau_factor * max_conv, 1e-12)

        # soft‑max weights
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        w = w / w_sum

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = 2.0 * conv_full[n - 1 : n - 1 + n]

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # evaluate improvement
        new_max = compute_max_conv(a)
        if new_max + 1e-12 < best_score:
            best_score = new_max
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration:07d} – improved to {best_score:.6f}")
        else:
            no_improve += 1

        # schedules
        if iteration % 1000 == 0:
            lr *= 0.94
            tau_factor = max(5e-4, tau_factor * 0.94)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.008)

        if no_improve >= 4000:
            # apply a tiny random displacement to shake things up
            jitter = np.random.randn(n) * 5e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:07d} – jitter applied")

        if time.time() >= deadline:
            print("Budget exhausted during Phase 1.")
            return best_seq.tolist()

    # --------------------------------------------------------------
    # Phase 2 – true sub‑gradient refinement
    # --------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.90
    step = 0.045 * (target_sum / n)   # start step size
    sub_iter = 0
    no_improve_sub = 0

    print(f"[{time.time():.2f}] Phase 2 – sub‑gradient refinement")
    while time.time() < phase2_deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()
        eps = max(1e-12, 1e-9 * max_conv)

        mask = (conv >= max_conv - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0

        a_rev = a[::-1]
        L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        new_max = compute_max_conv(a)
        sub_iter += 1
        if new_max + 1e-12 < best_score:
            best_score = new_max
            best_seq = a.copy()
            no_improve_sub = 0
            print(f"Sub‑grad iter {sub_iter:06d} – improved to {best_score:.6f}")
        else:
            no_improve_sub += 1

        # decay the step size
        if sub_iter % 500 == 0:
            step *= 0.93

        # occasional random pair transfer to diversify
        if np.random.rand() < 0.018:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # if stuck for a long while, inject jitter
        if no_improve_sub >= 5000:
            jitter = np.random.randn(n) * 4e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve_sub = 0
            print(f"Sub‑grad iter {sub_iter:06d} – jitter applied")

        if time.time() >= deadline:
            print("Budget exhausted during Phase 2.")
            return best_seq.tolist()

    # --------------------------------------------------------------
    # Phase 3 – final polishing (tiny random pair transfers)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – polishing")
    while time.time() < deadline:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.0022)
        cand_score = compute_max_conv(cand)
        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
