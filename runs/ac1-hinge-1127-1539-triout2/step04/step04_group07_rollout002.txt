"""
Optimisation by exhaustive search over geometric (exponential) step functions.

For a fixed length `n` the objective after the normalisation Σa = √(2n)
reduces to the maximal entry of the self‑convolution `a*a`.  An exponential
profile `a_i ∝ q^i` gives a closed‑form expression for this maximum and the
optimal decay is attained when the decay parameter `k` (such that q = exp(-k/n))
satisfies `k ≈ ln n`.  The programme enumerates a modest set of lengths and
decay‑factors around this theoretical optimum, evaluates the exact maximum
via FFT‑based convolution and keeps the best sequence found before the 1000 s
budget expires.  A brief random‑swap refinement is applied at the end.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Exponential warm‑start (geometric progression)
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """
    Generate a decreasing exponential sequence of length n.
    The decay is controlled by `k_factor` (the total exponent k in exp(-k/n)).
    The result is projected onto the simplex Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    lam = k_factor / n                     # λ = k / n
    q = np.exp(-lam)                       # q = e^{-λ}
    # exact normalisation (avoid loss of precision for tiny λ)
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)   # = 1 - q**n
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)   # safety projection to enforce exact sum
    return a

# ----------------------------------------------------------------------

    # 1) systematic scan over lengths and decay parameters
    # ------------------------------------------------------------------
    # deterministic list of lengths (up to 20 000 – still fast with FFT)
    lengths = list(range(800, 20001, 800))

    # sprinkle a few random lengths to increase diversity (while respecting time)
    while time.time() < deadline and len(lengths) < 40:
        n_rand = int(rng.integers(800, 20001))
        if n_rand not in lengths:
            lengths.append(n_rand)

    lengths.sort()

    for n in lengths:
        if time.time() >= deadline:
            break

        # theoretical optimum decay: k ≈ ln n
        k_opt = math.log(n)

        # explore a small neighbourhood of the optimum
        for factor in (0.7, 0.9, 1.0, 1.1, 1.3):
            k = k_opt * factor
            a = _exponential_sequence(n, k)

            # exact maximum of the convolution (FFT based)
            conv = _full_convolution(a)
            max_b = np.max(conv)
            sum_a = np.sum(a)
            # the objective after normalisation is 2·n·max_b / sum_a²
            score = 2.0 * n * max_b / (sum_a * sum_a)

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[search] n={n:5d} k={k:6.3f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) short random‑swap refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        S = np.sqrt(2.0 * n)
        # a few thousand cheap proposals are cheap compared to the remaining budget
        for _ in range(5000):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j or best_seq[i] == 0.0:
                continue
            # move a small random fraction of mass from i to j
            delta = best_seq[i] * rng.random() * 0.02
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand, S)
            conv = _full_convolution(cand)
            max_b = np.max(conv)
            sum_a = np.sum(cand)
            score = 2.0 * n * max_b / (sum_a * sum_a)
            if score < best_score:
                best_score = score
                best_seq = cand
                print(f"[refine] n={n:5d} improved score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) final sanity check via the external evaluator and return result
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (very unlikely to happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
