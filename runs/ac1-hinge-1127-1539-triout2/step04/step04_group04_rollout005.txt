"""...

import time
import numpy as np
from scipy.optimize import linprog

# define projection functions, convolution, etc.

def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _gradient_softmax(a: np.ndarray, tau: float):
    # copy from original code
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)
    if tau < 1e-12:
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, b, max_b

def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6
    t = 0
    # initial temperature based on current max
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8
    best_a = a.copy()
    best_max = cur_max
    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, S)
        max_new = np.max(_full_convolution_fft(a_new))
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
        a = a_new
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
    return best_a, best_max

def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular", "power", "cosine"]
    )
    idx = np.arange(n, dtype=float)
    if typ == "uniform":
        a = rng.random(n)
    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)
    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))
    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])
    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)
    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)
    return a

def _lp_iterative_opt(f0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Iterative linear‑program refinement (Kantorovich‑type scheme)."""
    f = f0.copy()
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    # initial max
    best_max = np.max(_full_convolution_fft(f))
    best_f = f.copy()
    t = 0.5
    it = 0
    while time.time() < deadline and it < 50:
        it += 1
        M = best_max
        # build constraint matrix A_ub (dense)
        A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A_ub[i:i + n, i] = f[i]
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            # If LP fails, break out – keep current best
            break
        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            break
        scale = S / sum_g0
        g = g0 * scale
        # convex combination
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        max_new = np.max(_full_convolution_fft(f_new))
        if max_new < best_max - 1e-8:
            best_max = max_new
            best_f = f_new.copy()
            f = f_new
            t = max(0.1, t * 0.92)   # slowly reduce step as we improve
        else:
            # no improvement – shrink step size a little
            t = max(0.1, t * 0.85)
            # occasionally jitter to escape plateaus
            if rng.random() < 0.01:
                eps = rng.normal(scale=f.mean() * 0.02, size=n)
                f = np.clip(f + eps, 0.0, None)
                f = _project_to_simplex(f, S)
    # final projection (numerical clean‑up)
    best_f = _project_to_simplex(best_f, S)
    best_max = np.max(_full_convolution_fft(best_f))
    return best_f, best_max

"""
Step‐function optimiser.

The algorithm works in three stages within the time budget:

1️⃣ Random parametric sampling (≈45 % of the budget).  
   Generates vectors from a variety of families (uniform, exponential,
   symmetric‑exponential, Gaussian, triangular, power‑law, raised‑cosine),
   projects them onto the ℓ₁‑simplex ∑a = √(2 n) and keeps the best
   (smallest self‑convolution peak).

2️⃣ Linear‑program (LP) refinement of the current best vector.  
   For a vector `f` with maximal convolution entry `M = max(f*f)` we solve

        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k,
                   g_i ≥ 0.

   The solution `g₀` is scaled to the required ℓ₁–norm and mixed with `f`
   (convex combination).  Repeating this step decreases `M` and drives the
   convolution towards a flatter shape.

3️⃣ A short Adam optimiser that minimises a smooth soft‑max approximation
   of the ∞‑norm, keeping the vector on the simplex throughout.

The best sequence found before the deadline is returned and stored in the
global variable `height_sequence_1` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the ℓ₁‑simplex of sum S."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard max → sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:                 # diagonal contribution
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights w_k ∝ exp(b_k / τ)
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)          # shape (2n‑1,)

    # cross‑correlation a ⋆ w = Σ_j a_j w_{i+j}
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # length n
    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection (soft‑max temperature annealing)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.6
    t = 0

    # initialise temperature
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        max_new = np.max(_full_convolution_fft(a_new))
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        # occasional jitter to escape plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Generate a random parametric distribution of length n
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular", "power", "cosine"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    else:
        a = rng.random(n)

    # small jitter (helps diversity)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# LP‑based iterative refinement (Kantorovich‑type scheme)
# ----------------------------------------------------------------------
def _lp_iterative_opt(f0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Starting from f0, repeatedly:
      • solve LP maximise Σ g_i  s.t. (f * g) ≤ M,
      • scale g to the required ℓ₁‑norm,
      • replace f ← (1‑t) f + t g.
    The loop stops when time runs out or when no progress is made.
    """
    f = f0.copy()
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    best_max = np.max(_full_convolution_fft(f))
    best_f = f.copy()
    t = 0.5
    it = 0

    while time.time() < deadline and it < 50:
        it += 1
        M = best_max

        # Build dense constraint matrix A_ub: each column i holds f[i] on rows i … i+n‑1
        A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A_ub[i:i + n, i] = f[i]

        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)               # maximize Σ g_i

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            break

        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            break

        # Scale g0 to have the correct ℓ₁‑norm
        scale = S / sum_g0
        g = g0 * scale

        # Convex update
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)

        max_new = np.max(_full_convolution_fft(f_new))

        if max_new < best_max - 1e-8:
            best_max = max_new
            best_f = f_new.copy()
            f = f_new
            t = max(0.1, t * 0.92)      # shrink step size slowly as we improve
        else:
            t = max(0.1, t * 0.85)      # otherwise reduce step size aggressively
            if rng.random() < 0.01:     # occasional jitter to escape plateaus
                eps = rng.normal(scale=f.mean() * 0.02, size=n)
                f = np.clip(f + eps, 0.0, None)
                f = _project_to_simplex(f, S)

    # final clean‑up projection
    best_f = _project_to_simplex(best_f, S)
    best_max = np.max(_full_convolution_fft(best_f))
    return best_f, best_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (height vector) with a low evaluation score.
    The procedure consists of:
      1. random parametric sampling,
      2. LP‑based refinement of the best candidate,
      3. a short Adam polishing step.
    The best sequence discovered before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0                     # keep a small safety margin
    deadline = time.time() + budget_s - safety

    best_seq = None
    best_max = np.inf

    # ------------------------------------------------------------------
    # Use any previously‑found construction as a starting point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            max0 = np.max(_full_convolution_fft(seq0))
            best_seq = seq0.copy()
            best_max = max0
            print(f"[init] imported previous solution: n={len(seq0)}, max={best_max:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – random parametric search (≈45 % of the remaining budget)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.45
    attempts = 0
    while time.time() < phase1_deadline:
        attempts += 1
        n = rng.integers(500, 1500 + 1)           # moderate lengths keep LP cheap
        raw = _random_distribution(n, rng)
        S = np.sqrt(2.0 * n)
        cand = _project_to_simplex(raw, S)
        cur_max = np.max(_full_convolution_fft(cand))
        if cur_max < best_max:
            best_max = cur_max
            best_seq = cand.copy()
            print(f"[search] attempt={attempts} n={n} new best max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP refinement of the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None:
        # leave at least ~15 s for the final Adam polish
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(0.0, remaining - 15.0)
        refined_seq, refined_max = _lp_iterative_opt(best_seq, phase2_deadline, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_seq.copy()
            print(f"[lp] improved max to {best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short Adam polishing (uses whatever time is left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        refined_seq, refined_max = _adam_optimize(best_seq, deadline, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_seq.copy()
            print(f"[adam] final polishing improved max to {best_max:.6f}")

    # ------------------------------------------------------------------
    # Final return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_max:.6f}) length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
