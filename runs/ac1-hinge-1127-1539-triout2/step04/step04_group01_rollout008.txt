"""
Step‑function optimizer (enhanced).

The routine searches for a non‑negative sequence a with
    Σ a_i = √(2 n)
that minimises the peak of its self‑convolution.
The algorithm works in three stages:

1. **Multi‑scale exploration** – a set of lengths (512 … 8192) is tried.
   For each length many random initial shapes (Dirichlet, exponential,
   power‑law, cosine, Gaussian) are generated and refined by a
   p‑norm smooth‑maximum descent (the gradient of (∑ c_k^p)^{1/p},
   c = a*a).  The schedule of powers (2 → 4 → 8 → 16 → 32 → 64 → 128)
   lets the optimisation first spread the mass and then focus on the
   largest convolution entries.

2. **Targeted sub‑gradient polishing** – after stage 1 the best sequence
   for the current length is further improved by a simple sub‑gradient
   step that directly reduces the maximal convolution entry.

3. **Final simulated‑annealing (SA) polishing** – a fast SA that moves
   small fractions of mass between indices, accepting uphill moves with
   a geometrically cooling temperature.  SA runs with whatever time is
   left.

All updates keep the vector on the simplex (ℓ¹‑projection) and respect
the non‑negativity constraint.  The whole search respects the 1000 s
wall‑clock budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT version)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation of a (length n) with w (length m) producing a vector of length n:
        (a ⋆ w)[i] = Σ_j a_j * w_{i+j}
    Implemented via FFT (a is reversed before convolution).
    """
    n = a.size
    m = w.size
    conv_len = n + m - 1
    size = _next_pow2(conv_len)
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    w_fft = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(ar_fft * w_fft, n=size)[:conv_len]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# p‑norm (smooth‑max) gradient descent
# ----------------------------------------------------------------------


def _p_norm_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform a multi‑stage descent on the p‑norm of the convolution vector.
    The schedule of powers is 2, 4, 8, 16, 32, 64, 128.
    Projection onto the ℓ¹‑simplex is applied after every update.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # base step magnitude – a fraction of the ℓ¹ norm
    step_base = target * 0.20
    rng = np.random.default_rng()

    # power schedule (p, minimal seconds for this stage)
    schedule = [(2, 0.5), (4, 0.5), (8, 1.0), (16, 1.0),
                (32, 2.0), (64, 2.0), (128, 4.0)]

    for p, min_sec in schedule:
        # respect the global deadline
        stage_end = min(deadline, time.time() + min_sec)
        while time.time() < stage_end:
            # --- compute convolution a*a ---------------------------------
            size = _next_pow2(2 * n - 1)
            a_fft = np.fft.rfft(a, n=size)
            conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

            # --- weight w = conv^{p-1} for the gradient ------------------
            # avoid 0**(p-1) = 0 which is fine; np.power works for all p
            w = np.power(conv, p - 1, dtype=float)

            # --- gradient g = 2 * p * (a ⋆ w) ---------------------------
            g = 2.0 * p * _correlation_fft(a, w)

            # --- adaptive step size ------------------------------------
            norm_g = np.linalg.norm(g)
            if norm_g > 0.0:
                lr = step_base / (norm_g * (p + 1.0))
                a -= lr * g

                # --- projection back onto the simplex -------------------
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, target)

        # occasional jitter to escape shallow plateaus
        if time.time() < deadline:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (focused on the worst shift)
# ----------------------------------------------------------------------


def _max_subgradient_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient method that always reduces the currently largest convolution entry.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    it = 0
    base_lr = target * 0.10

    while time.time() < deadline:
        # convolution
        size = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        k_max = int(np.argmax(conv))          # shift with maximal entry
        max_val = conv[k_max]

        # build sub‑gradient g_i = 2 * a_{k_max-i} (if index in range)
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        # step size decays slowly
        lr = base_lr / (1.0 + 0.0005 * it)
        a -= lr * g

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        it += 1
        if it % 500 == 0:
            print(f"[{time.time():.1f}] Sub‑grad iter {it} max={max_val:.6f}")

        # occasional random jitter
        if it % 2500 == 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves) – final polishing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a random fraction of mass between
    two indices.  Accepts uphill moves with probability exp(-Δ/T) where T
    decays geometrically.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    # initial temperature – a small fraction of the current score
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

# Random initialisation families (on the simplex)

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the simplex
    Σ a = √(2 n).  Several families are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:          # uniform
        a = np.full(n, target / n)

    elif family == 1:        # Dirichlet (flat on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 2:        # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 3:        # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 4:        # half‑cosine (raised cosine window)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2

    elif family == 5:        # Gaussian
        sigma = n / 6.0
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)

    else:                    # mixed exponential × power‑law
        p = rng.uniform(0.2, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Log the final value as seen by the official evaluator
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:                      # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
