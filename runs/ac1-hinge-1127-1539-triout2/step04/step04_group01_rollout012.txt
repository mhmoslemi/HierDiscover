def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = np.inf

    # list of lengths to try (starting from small, increase)
    lengths = [2000, 4000, 6000, 8000, 10000]
    # allocate time proportionally
    # we will loop lengths while time left, break if insufficient.
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        # initial candidate: sorted Dirichlet to be decreasing
        # generate several initial restarts
        init_dead_end = min(deadline, time.time() + min(20.0, (deadline - time.time())/4))
        while time.time() < init_dead_end:
            # use Dirichlet sample and sort descending to produce decreasing shape
            a0 = rng.dirichlet(alpha=np.ones(n))
            a0 = np.sort(a0)[::-1]  # descending
            a0 = a0 * target_sum / a0.sum()
            # Gradient descent
            gd_dead = min(deadline, time.time() + 4.0)
            a_opt = _gradient_descent_opt(a0, gd_dead, beta_max=1e6)
            # Evaluate
            val = _score(a_opt)
            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] New best length {n} val {val:.6f}")
        # After initial restarts, run a deeper refinement if best_seq length matches n
        if best_seq is not None and best_seq.size == n:
            # Additional long gradient descent to refine
            refine_dead = min(deadline, time.time() + min(120.0, deadline - time.time()))
            a_ref = _gradient_descent_opt(best_seq, refine_dead, beta_max=1e6)
            val = _score(a_ref)
            if val < best_val:
                best_val = val
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined best with gd length {n} val {val:.6f}")
        # apply targeted simulated annealing
        if best_seq is not None and best_seq.size == n:
            # Use half remaining time for SA
            remaining = deadline - time.time()
            if remaining > 5.0:
                a_sa = _targeted_sa(best_seq, deadline)
                val = _score(a_sa)
                if val < best_val:
                    best_val = val
                    best_seq = a_sa.copy()
                    print(f"[{time.time():.1f}] Targeted SA improved length {n} val {val:.6f}")

    # Finally evaluate with evaluate_sequence if available
    best_seq = best_seq.tolist() if best_seq is not None else [0.0]
    # Trim trailing zeros
    seq_arr = np.asarray(best_seq)
    nz_idx = np.where(seq_arr > 1e-12)[0]
    if nz_idx.size > 0:
        seq_arr = seq_arr[: nz_idx[-1] + 1]
    best_seq = seq_arr.tolist()
    try:
        val = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] final evaluate_sequence = {val:.6f}, length {len(best_seq)}")
    except Exception as e:
        print("Error in evaluate_sequence:", e)
    return best_seq

def _targeted_sa(a0, deadline):
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _score(a)
    # compute number of iterations based on time
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution, find max index
        b = np.convolve(a, a)
        max_b = b.max()
        k = np.argmax(b)
        # compute contributions
        # compute indices i for which j = k - i is valid
        i_vals = np.arange(n)
        j_vals = k - i_vals
        valid = (j_vals >= 0) & (j_vals < n)
        # contributions for valid i
        contrib = np.zeros_like(a)
        contrib[valid] = a[valid] * a[j_vals[valid]]
        total_contrib = contrib.sum()
        if total_contrib == 0.0:
            # no overlapping? break
            break
        # sample i with probability proportional to contrib
        prob = contrib / total_contrib
        i = rng.choice(n, p=prob)
        # pick j random distinct index that is not i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        # transfer fraction
        frac = rng.random() * 0.1  # up to 10%
        delta = a[i] * frac
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        # keep a non-negative; due to delta from i ensures non-negative
        # evaluate new score:
        new_score = _score(a)
        if new_score < best_score:
            best_score = new_score
            best = a.copy()
            print(f"[{time.time():.1f}] Targeted SA improved to {new_score:.6f}")
        # Accept or reject based on improvement (only accept if non-worse)
        if new_score > max_b: # not improvement
            # revert move with some probability?
            # simplistic: revert
            a[j] -= delta
            a[i] += delta
        # else keep (improved)
    return best

  approximation of the convolution peak.  The temperature (β) is annealed up
  to 1e6 and a momentum/Adam‑style step‑size control is used.
* Finally applies a **targeted simulated‑annealing** phase that repeatedly
  moves a small fraction of mass away from the indices that contribute most
  to the current peak of the convolution.  Only moves that lower the peak
  (or are accepted with a tiny uphill probability) are kept.
* The whole search respects a 1000 s wall‑clock budget and returns the best
  sequence found as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Linear‑algebra utilities
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation: max(a*a)."""
    return _max_convolution_fft(a)


# ------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) and projected gradient descent
# ------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p),   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    # subtract max for numerical stability
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    # correlation a ⋆ p = (a_rev * p) evaluated at indices n‑1 … n‑1+n‑1
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray,
                          deadline: float,
                          beta_max: float = 1e6) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max.
    Uses momentum and an Adam‑style adaptive step size.
    The inverse temperature β is annealed until ``beta_max``.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_factor = 1.4
    step_base = target_sum * 0.2          # base magnitude of the step

    # Adam‑style buffers
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    eps = 1e-12
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of smooth max
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # adaptive learning rate – larger β ⇒ smaller steps
        lr = step_base / (np.sqrt(v_hat) + eps) / (beta + 1.0)

        a -= lr * m_hat

        # keep on the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t_iter % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if t_iter % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ------------------------------------------------------------
# Targeted simulated‑annealing (mass‑transfer moves aimed at the peak)
# ------------------------------------------------------------


def _targeted_sa(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly move a small fraction of mass away from the indices that
    contribute most to the current convolution peak.
    Only moves that lower the peak (or are accepted with a tiny uphill
    probability) are kept.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    # temperature for occasional uphill steps – decays slowly
    T = best_score * 0.02 + 1e-12
    cooling = 0.9997

    while time.time() < deadline:
        # full convolution and peak location
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        cur_max = float(conv[k_max])

        # contributions of each index to this peak
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[valid] * a[j[valid]]

        # if there is no contribution (should not happen) break
        if contrib.sum() == 0.0:
            break

        # pick a source index proportional to its contribution
        prob = contrib / contrib.sum()
        src = rng.choice(n, p=prob)

        # pick a destination index uniformly (different from src)
        dst = rng.integers(0, n)
        while dst == src:
            dst = rng.integers(0, n)

        # transfer a small random fraction of the source mass
        frac = rng.random() * 0.07        # up to 7 %
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        a[src] -= delta
        a[dst] += delta

        new_max = _score(a)   # cheap call – uses FFT
        dE = new_max - cur_max

        # accept if we improve, otherwise maybe accept uphill
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))
        if not accept:
            # revert the move
            a[dst] -= delta
            a[src] += delta
        else:
            if new_max < best_score:
                best_score = new_max
                best = a.copy()
                print(f"[{time.time():.1f}] Targeted SA improved → {best_score:.6f}")

        # temperature schedule
        T *= cooling

        # occasional global jitter to avoid being stuck
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_max = _score(a)
            if cur_max < best_score:
                best_score = cur_max
                best = a.copy()
                print(f"[{time.time():.1f}] Global jitter improved → {best_score:.6f}")

    return best


# ------------------------------------------------------------
# Random initialisers on the ℓ¹‑simplex
# ------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence with Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    # a few handcrafted families (sorted Dirichlet gives a decreasing shape)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # sorted Dirichlet → decreasing shape
        a = rng.dirichlet(alpha=np.ones(n))
        a = np.sort(a)[::-1]

    # add a tiny gaussian noise and project onto the simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, tries several lengths, uses
    projected‑gradient descent with an aggressive annealing schedule and
    finishes with a targeted simulated‑annealing polish.
    Returns the best sequence found (plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # -----------------------------------------------------------------
    # Seed from the previous construction if it exists
    # -----------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_val = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # -----------------------------------------------------------------
    # Lengths to explore – larger n gives more freedom but is slower
    # -----------------------------------------------------------------
    lengths = [2000, 3500, 5000, 7000, 9000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # -------------------------------------------------------------
        # Phase A – cheap random restarts with a short GD run
        # -------------------------------------------------------------
        phase_end = min(deadline,
                        time.time() + min(25.0,
                                          (deadline - time.time()) / 4.0))
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)

            # quick GD (≈2 s) to get into a decent basin
            sub_dead = min(deadline, time.time() + 2.5)
            a_gd = _gradient_descent_opt(a0, sub_dead, beta_max=5e5)

            val = _score(a_gd)
            if val < best_val:
                best_val = val
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={val:.6f}")

        # -------------------------------------------------------------
        # Phase B – deep refinement of the current best of this length
        # -------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            sub_dead = min(deadline, time.time() + min(150.0, deadline - time.time()))
            a_ref = _gradient_descent_opt(best_seq, sub_dead, beta_max=1e6)
            val = _score(a_ref)
            if val < best_val:
                best_val = val
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined best (length {n}) score={val:.6f}")

    # -----------------------------------------------------------------
    # Final targeted simulated‑annealing polish
    # -----------------------------------------------------------------
    if best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 5.0:
            a_final = _targeted_sa(best_seq, deadline)
            val = _score(a_final)
            if val < best_val:
                best_val = val
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final targeted SA improvement: score={val:.6f}")

    # -----------------------------------------------------------------
    # Clean up tiny trailing zeros for neatness
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # -----------------------------------------------------------------
    # Report final objective via the official evaluator
    # -----------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
