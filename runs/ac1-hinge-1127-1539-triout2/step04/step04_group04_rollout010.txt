```
from scipy.sparse import csr_matrix

def lp_maximize_sum(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # for each i in [i_start, i_end]:
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

"""
Improved search for low‑evaluation step‑function heights.

The algorithm consists of:

1️⃣ Random parametric sampling (short time budget).
2️⃣ Multi‑restart Adam optimisation using a soft‑max approximation
   of the ∞‑norm of the self‑convolution.
3️⃣ Linear‑programming refinement: given a current vector *f*,
   maximise the ℓ₁‑norm of a new vector *g* while keeping the
   mixed convolution (f⋆g) below the current bound.  Mixing
   (1−t) f + t g with a small step *t* provably never raises the
   maximal self‑convolution entry, and it often reduces it.
4️⃣ A final short Adam run to polish the result.

The code observes the time budget (≈ 1000 s) and stores the best
candidate in the global variable `height_sequence_1` for reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT ( O(n log n) )."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Gradient of τ·log Σ exp(b/τ) where b = a*a,
    plus raw convolution b and its maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard‑max → sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal term (i=j) counted once
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)           # shape (2n‑1,)

    # compute gradient = 2 * a ⋆ w
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]

    return g, b, max_b


def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Projected Adam optimisation of the soft‑max approximation of
    max(a*a).  Returns the best vector found and its maximal convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6
    t = 0

    # initial temperature (scale of convolution) – a heuristic
    _, _, cur_max = _gradient_softmax(a, tau=0.0)   # hard max just for scale
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        # gradient step
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluation
        new_max = np.max(_full_convolution_fft(a_new))

        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        # occasional jitter to escape plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate an unnormalised non‑negative vector from a random family."""
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian",
         "triangular", "power", "cosine", "beta"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "beta":
        alpha = rng.uniform(0.3, 5.0)
        beta = rng.uniform(0.3, 5.0)
        a = rng.beta(alpha, beta, size=n)

    else:
        a = rng.random(n)

    # small random jitter (helps diversity)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


def _lp_maximize_sum(f: np.ndarray, M: float):
    """
    Solve the LP:
        maximize   Σ g_i
        subject to conv(f, g)_k ≤ M  (k = 0..2n-2)
                   g_i ≥ 0
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # build constraint matrix row by row
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)                     # minimise –Σ g_i  ⇒ maximise Σ g_i
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=(0, None), method='highs')
    if res.success:
        return res.x
    else:
        return None


def _lp_refine(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Iteratively improve a vector by linear‑programming mixing.
    At each iteration we solve the LP that maximises the ℓ₁‑norm of a new
    vector g under the current maximal convolution bound, then mix
    (1‑t) f + t g with a small step size t.  The process never raises the
    bound and usually drives it down.
    Returns the refined vector and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_max = np.max(_full_convolution_fft(a))
    best_a = a.copy()
    t = 0.3                     # initial mixing weight

    while time.time() < deadline:
        g = _lp_maximize_sum(best_a, best_max)
        if g is None:
            print("[lp] LP failed – aborting refinement")
            break

        sum_g = np.sum(g)
        if sum_g <= 1e-12:
            break

        # Scale g to have the required ℓ₁‑norm
        g_scaled = g * (S / sum_g)

        # Mix and re‑project onto the simplex
        a_new = (1.0 - t) * best_a + t * g_scaled
        a_new = _project_to_simplex(a_new, S)

        new_max = np.max(_full_convolution_fft(a_new))

        if new_max < best_max - 1e-7:
            best_a = a_new
            best_max = new_max
            print(f"[lp_refine] improved max to {best_max:.6f}")
            # shrink step size a little for stability
            t = max(t * 0.9, 0.05)
        else:
            # no progress – halve the mixing weight
            t = max(t * 0.5, 0.01)
            if t < 0.02:
                break

    return best_a, best_max

"""
Improved search for low‑evaluation step‑function heights.

The algorithm consists of:

1️⃣ Random parametric sampling (short time budget).
2️⃣ Multi‑restart Adam optimisation using a soft‑max approximation
   of the ∞‑norm of the self‑convolution.
3️⃣ Linear‑programming refinement: given a current vector *f*,
   maximise the ℓ₁‑norm of a new vector *g* while keeping the
   mixed convolution (f⋆g) below the current bound.  Mixing
   (1−t) f + t g with a small step *t* provably never raises the
   maximal self‑convolution entry, and it often reduces it.
4️⃣ A final short Adam run to polish the result.

The code observes the time budget (≈ 1000 s) and stores the best
candidate in the global variable `height_sequence_1` for reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT ( O(n log n) )."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Gradient of τ·log Σ exp(b/τ) where b = a*a,
    plus raw convolution b and its maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard‑max → sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)           # shape (2n‑1,)

    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]

    return g, b, max_b


def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Projected Adam optimisation of the soft‑max approximation of
    max(a*a).  Returns the best vector found and its maximal convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6
    t = 0

    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        new_max = np.max(_full_convolution_fft(a_new))

        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()

        a = a_new

        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate an unnormalised non‑negative vector from a random family."""
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian",
         "triangular", "power", "cosine", "beta"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "beta":
        alpha = rng.uniform(0.3, 5.0)
        beta = rng.uniform(0.3, 5.0)
        a = rng.beta(alpha, beta, size=n)

    else:
        a = rng.random(n)

    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


def _lp_maximize_sum(f: np.ndarray, M: float):
    """
    Solve the LP:
        maximise   Σ g_i
        subject to conv(f, g)_k ≤ M   (k = 0…2n‑2)
                   g_i ≥ 0
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)                     # minimise –Σ g_i  ⇒ maximise Σ g_i
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=(0, None), method='highs')
    if res.success:
        return res.x
    else:
        return None


def _lp_refine(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Iteratively improve a vector by linear‑programming mixing.
    At each iteration we solve the LP that maximises the ℓ₁‑norm of a new
    vector g under the current maximal convolution bound, then mix
    (1‑t) f + t g with a small step size t.  The process never raises the
    bound and usually drives it down.
    Returns the refined vector and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_max = np.max(_full_convolution_fft(a))
    best_a = a.copy()
    t = 0.3                     # initial mixing weight

    while time.time() < deadline:
        g = _lp_maximize_sum(best_a, best_max)
        if g is None:
            print("[lp] LP failed – aborting refinement")
            break

        sum_g = np.sum(g)
        if sum_g <= 1e-12:
            break

        g_scaled = g * (S / sum_g)

        a_new = (1.0 - t) * best_a + t * g_scaled
        a_new = _project_to_simplex(a_new, S)

        new_max = np.max(_full_convolution_fft(a_new))

        if new_max < best_max - 1e-7:
            best_a = a_new
            best_max = new_max
            print(f"[lp_refine] improved max to {best_max:.6f}")
            t = max(t * 0.9, 0.05)
        else:
            t = max(t * 0.5, 0.01)
            if t < 0.02:
                break

    return best_a, best_max
