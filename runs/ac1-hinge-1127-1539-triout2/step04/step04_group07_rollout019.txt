def solve_lp_max_sum_b(f, M):
    n = len(f)
    S = np.sqrt(2*n)
    # Build constraint matrix A_ub = (conv matrix) (dimensions (2n-1) x n)
    # constraint: conv_i = (f * b)[i] <= M  => A_ub[i,:] @ b <= M
    # plus non-neg constraints handled by bounds.

    # Use scipy.sparse.lil_matrix for building rows.
    from scipy.sparse import lil_matrix
    rows = 2*n - 1
    A = lil_matrix((rows, n), dtype=float)
    for i in range(rows):
        start_j = max(0, i-(n-1))
        end_j = min(i, n-1)
        # indices j in [start_j, end_j]
        # for each j, coefficient = f[i-j]
        idx_range = np.arange(start_j, end_j+1)
        coeff = f[i - idx_range]
        A[i, idx_range] = coeff
    A = A.tocsr()

    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(rows, M)  # upper bound
    # bounds: b>=0
    bounds = [(0, None)] * n

    # Use linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b_opt = res.x
    return b_opt

def solve_lp_max_sum_b(f, M):
    n = len(f)
    rows = 2*n - 1
    # Build matrix
    from scipy.sparse import lil_matrix
    A = lil_matrix((rows, n), dtype=float)
    for i in range(rows):
        # compute start_j, end_j
        start_j = max(0, i-(n-1))
        end_j = min(i, n-1)
        # fill
        # using slice assignment
        A[i, start_j:end_j+1] = f[i - np.arange(start_j, end_j+1)]
        # vectorization: maybe use np.arange and assign
    A = A.tocsr()
    c = -np.ones(n, dtype=float)
    b_ub = np.full(rows, M, dtype=float)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence().

The method combines three ideas:

1.  **Exponential warm–starts** – for a given length n we scan a dense
    grid of decay rates λ (a_i = c·e^{-λ i}) that are projected onto the
    ℓ1‑simplex Σa = √(2·n).  The best exponential candidate is kept.

2.  **Projected smoothed‑max gradient descent** – the hard ∞‑norm
    max_k (a*a)[k] is replaced by a soft‑max with temperature α.
    A gradient step (computed via FFT) is taken and the vector is
    projected back onto the simplex.  Random mass‑swaps are added to
    escape plateaus.

3.  **Linear‑programming “boost”** – after a few gradient steps we solve
    a linear program
        maximise   Σ b_j
        subject to   (a * b)[k] ≤ M   (for all k) ,   b_j ≥ 0
    where M is the current maximal convolution entry.
    The optimal b has a larger ℓ1‑norm than a; after scaling it to the
    required sum we mix it with a (mixing factor t) which provably
    reduces the maximal convolution value.  The LP is solved with
    scipy.optimize.linprog (HiGHS) using a sparse Toeplitz constraint
    matrix.

All candidates are evaluated with the cheap “max‑convolution” objective
(the true objective after normalisation is exactly that maximum).  The
best sequence found before the budget expires is returned as a plain
list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max helpers (used for gradient descent)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(α·conv_i)/Σ_j exp(α·conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute gradient = 2·(a ⋆ w)   (correlation of a with w).
    """
    n = a.shape[0]
    # correlation a ⋆ w  (full)
    w_rev = w[::-1]
    conv_len = n + w.size - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # extract the n entries needed for the gradient
    # The formula derived in the original code:
    idx = w.size - 1 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the smoothed max‑norm.
    Returns the updated sequence and the *exact* maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Exponential warm‑start generator (parameterised by decay λ)
# ----------------------------------------------------------------------


def _exponential_by_lambda(n: int, lam: float) -> np.ndarray:
    """
    Generate a decreasing exponential a_i = c·exp(-lam·i) of length n,
    then project onto the simplex Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    q = np.exp(-lam)
    # avoid loss of precision for tiny lam
    if lam < 1e-12:
        a = np.full(n, S / n)
    else:
        denom = 1.0 - q ** n
        if denom == 0.0:
            denom = 1.0  # fallback (should not happen)
        c = S * (1.0 - q) / denom
        a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (maximise ℓ1‑norm under convolution bound)
# ----------------------------------------------------------------------


def _solve_lp_max_sum_b(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j
    subject to (f * b)[k] ≤ M   for all k = 0..2n‑2
               b_j ≥ 0
    Returns the optimal b (or None on failure).
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A (dense enough for the LP, but stored sparse)
    A = lil_matrix((rows, n), dtype=float)

    # Each row i corresponds to (f * b)[i] = Σ_j f_j·b_{i-j}
    for i in range(rows):
        start_j = max(0, i - (n - 1))
        end_j = min(i, n - 1)
        cols = np.arange(start_j, end_j + 1, dtype=int)
        vals = f[i - cols]                # because i - cols yields the needed f-index
        A[i, cols] = vals

    A = csr_matrix(A)

    c = -np.ones(n, dtype=float)          # maximise Σ b_j  → minimise -Σ b_j
    b_ub = np.full(rows, M, dtype=float)
    bounds = [(0.0, None)] * n

    # HiGHS interior‑point simplex works well with sparse matrices
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Local optimiser – gradient + occasional random swaps + LP boost
# ----------------------------------------------------------------------


def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence using projected gradient descent on a
    smooth‑max approximation, random mass‑swap moves and occasional LP
    “boost” steps.  The routine stops when the deadline is reached.
    Returns the best sequence seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (tuned empirically)
    lr = 0.8
    lr_decay = 0.9997
    alpha = 0.5
    alpha_mul = 1.008
    swap_prob = 0.12
    swap_frac = 0.07
    lp_prob = 0.07           # chance to trigger an LP boost per iteration
    mix_factor = 0.4         # t in a←(1‑t)a + t·g
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ---- gradient descent step on the smoothed max ----
        a, cur_max = _gradient_step(a, lr, alpha)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # ---- random mass‑swap (exploration) ----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis‑type acceptance
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # ---- occasional LP boost ----
        if rng.random() < lp_prob:
            M = np.max(_full_convolution(a))
            b = _solve_lp_max_sum_b(a, M)
            if b is not None:
                sum_b = b.sum()
                # if we can increase the ℓ1‑norm while respecting the bound,
                # mixing will strictly lower the maximal entry.
                if sum_b > S * 1.001:
                    g = b * (S / sum_b)            # scale to the required sum
                    a = (1.0 - mix_factor) * a + mix_factor * g
                    a = _project_simplex(a, S)
                    cur_max = np.max(_full_convolution(a))
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # ---- schedule updates ----
        lr *= lr_decay
        alpha *= alpha_mul

        # safety: stop if learning rate becomes negligible
        if lr < 1e-5:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises evaluate_sequence().
    The routine:
      * scans exponential families for several lengths,
      * runs a gradient+LP local optimiser from the most promising candidates,
      * keeps the overall best sequence, and finally returns it as a list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0) use the previous solution if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size:
                n = prev.size
                S = np.sqrt(2.0 * n)
                prev = _project_simplex(prev, S)
                cur = np.max(_full_convolution(prev))
                best_seq, best_score = prev, cur
                print(f"[init] using previous height_sequence_1, score={cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) scan exponential families (quick, cheap evaluation)
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2500, 3000, 3500, 4000]
    for n in lengths:
        if time.time() >= deadline:
            break
        S = np.sqrt(2.0 * n)
        # decay λ is roughly 1/(n) … 3/(n); we scan a fine grid
        lam_grid = np.linspace(0.5 / n, 3.0 / n, 80)   # 80 candidates per length
        for lam in lam_grid:
            a = _exponential_by_lambda(n, lam)
            cur = np.max(_full_convolution(a))
            if cur < best_score:
                best_score, best_seq = cur, a.copy()
                print(f"[exp] n={n:4d} λ={lam:.6f}  score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2) local optimisation of the best few candidates
    # ------------------------------------------------------------------
    # keep a small pool of promising candidates (length, λ)
    pool = []
    for n in lengths:
        for _ in range(2):   # two random initial λ values per length
            lam = rng.uniform(0.5 / n, 3.0 / n)
            a = _exponential_by_lambda(n, lam)
            pool.append(a)

    # also add the current best sequence (if any)
    if best_seq is not None:
        pool.append(best_seq)

    # optimise each candidate until time runs out
    while time.time() < deadline and pool:
        a0 = pool.pop(rng.integers(len(pool)))
        n = a0.shape[0]

        slice_deadline = min(deadline, time.time() + 30.0)   # 30 s per run
        a_opt, score_opt = _local_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score, best_seq = score_opt, a_opt.copy()
            print(f"[opt] new best n={n:4d} score={score_opt:.6f}")

        # occasionally re‑inject a tweaked version to keep diversity
        if rng.random() < 0.3 and time.time() < deadline:
            jitter = rng.normal(scale=0.005 * a_opt.mean(), size=n)
            a_jit = np.clip(a_opt + jitter, 0.0, None)
            a_jit = _project_simplex(a_jit, np.sqrt(2.0 * n))
            pool.append(a_jit)

    # ------------------------------------------------------------------
    # 3) fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
