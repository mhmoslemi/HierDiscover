"""
Advanced step‑function optimizer.

The optimizer starts from the best known sequence (or a random one),
applies an Adam‑based projected gradient descent on a smooth approximation
of the L∞‑norm of the self‑convolution, then iteratively solves a
linear‑program that provably never increases the peak value.
A short simulated‑annealing phase finishes the run.
The whole routine respects a wall‑clock budget (default 1000 s) and
returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full, not circular)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a returning a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns
        g  – gradient w.r.t. a,
        b  – the full convolution array (used only for debugging).
    The analytical form is  g = 2·(a ⋆ p)  where
        p_k ∝ exp(β·b_k)    (soft‑max probabilities).
    """
    n = a.shape[0]

    # b = a*a (full convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # correlation a ⋆ p  (a reversed convolved with p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑based projected gradient descent on the smoothed max (log‑sum‑exp).
    The loop stops when time.time() >= deadline.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4
    # base learning‑rate (fraction of the ℓ¹‑norm)
    step_base = target_sum * 0.30

    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = step_base / (np.sqrt(v_hat) + eps)

        a -= lr * m_hat

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Linear‑program step (the “LP‑boost” from the paper)
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A ∈ ℝ^{(2n‑1) × n} such that (A @ g)_k = (a * g)_k.
    It is a banded Toeplitz matrix: column i contains `a` shifted down by i rows.
    """
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # each column i: rows i … i+n‑1 get a[0 … n‑1]
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_improve(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    One or several LP‑based improvement steps.
    Returns a (potentially) improved sequence that never exceeds the previous
    peak value.  The loop terminates when the global deadline is reached.
    """
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    current_score = _max_convolution_fft(a)

    while time.time() < deadline:
        # Build the convolution matrix for the current a
        A = _build_conv_matrix(a)

        # Solve   max 1ᵀ g    s.t. A g ≤ current_score , g ≥ 0
        c = -np.ones(n)                     # –sum(g)  (linprog minimizes)
        b_ub = np.full(2 * n - 1, current_score, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Allocate a small time slice for the LP call; hi‑GHS respects a time limit.
        time_left = deadline - time.time()
        if time_left <= 0.0:
            break
        options = {"presolve": True, "time_limit": max(0.1, time_left)}
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options=options)

        if not res.success:
            # If the LP fails (e.g. time out) we abort the LP phase.
            break

        g = np.maximum(res.x, 0.0)           # numerical safety
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # Scale g back to the required ℓ¹‑norm.
        g = g * (target_sum / sum_g)

        # Mix the new vector with the old one.
        # Start with a relatively aggressive mixing factor and back‑off
        # if the peak would increase.
        t = 0.5
        improved = False
        while t > 1e-4:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            new_score = _max_convolution_fft(a_candidate)
            if new_score <= current_score + 1e-12:
                a = a_candidate
                current_score = new_score
                improved = True
                print(f"[{time.time():.1f}] LP mix t={t:.4f} → score={new_score:.6f}")
                break
            t *= 0.5

        if not improved:
            # No admissible mixing factor found – stop the LP phase.
            break

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (same as the previous implementation)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or rng.random() < np.exp(-dE / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # uniform plus tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under a 1000 s wall‑clock budget.

    The strategy is:
    1. Start from the previously stored sequence (if any) or a random one.
    2. Adam‑based projected gradient descent on a smooth max.
    3. A few LP‑based “boost” steps that are provably non‑increasing.
    4. A short simulated‑annealing polishing phase.
    The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use the previously discovered construction if present
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a = np.asarray(earlier, dtype=float)
        n = a.size
        a = _project_to_simplex(a, np.sqrt(2.0 * n))
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n})")
    else:
        # If no seed, pick a modest length and generate a random start
        n = 2000
        a = _random_initial_sequence(n, rng)

    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_seq = a.copy()
    best_score = _max_convolution_fft(a)
    print(f"[{time.time():.1f}] Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam projected gradient descent
    # ------------------------------------------------------------------
    phase_end = min(deadline, time.time() + min(120.0, (deadline - time.time()) * 0.5))
    a = _adam_gradient_descent_opt(a, phase_end)
    sc = _max_convolution_fft(a)
    if sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"[{time.time():.1f}] After Adam GD: {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase B – LP “boost” iterations (guaranteed non‑increase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        lp_dead = deadline - 10.0   # leave a few seconds for the final SA
        a = _lp_improve(a, lp_dead)
        sc = _max_convolution_fft(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] After LP phase: {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase C – final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_final = _simulated_annealing(best_seq, deadline)
        sc = _max_convolution_fft(a_final)
        if sc < best_score:
            best_score = sc
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] After final SA: {sc:.6f}")

    # ------------------------------------------------------------------
    # Trim insignificant trailing zeros for tidiness
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Report the final objective under the official evaluator
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()

"""
Advanced step‑function optimizer.

The optimizer starts from the best known sequence (or a random one),
applies an Adam‑based projected gradient descent on a smooth approximation
of the L∞‑norm of the self‑convolution, then iteratively solves a
linear‑program that provably never increases the peak value.
A short simulated‑annealing phase finishes the run.
The whole routine respects a wall‑clock budget (default 1000 s) and
returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full, not circular)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a returning a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Smooth‑max (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns
        g  – gradient w.r.t. a,
        b  – the full convolution array.
    The analytical form is  g = 2·(a ⋆ p)  where
        p_k ∝ exp(β·b_k)    (soft‑max probabilities).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑based projected gradient descent on the smoothed max (log‑sum‑exp).
    Terminates when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4
    step_base = target_sum * 0.30

    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = step_base / (np.sqrt(v_hat) + eps)

        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Linear‑program “boost” step
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A ∈ ℝ^{(2n‑1) × n} such that (A @ g)_k = (a * g)_k.
    It is a banded Toeplitz matrix: column i contains `a` shifted down by i rows.
    """
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_improve(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeated LP‑based improvement steps.
    Each step solves:
        max   1ᵀ g
        s.t.  a * g ≤ current_peak ,  g ≥ 0
    The feasible solution is scaled back to the required ℓ¹‑norm and mixed
    with the current vector.  The mixing factor is reduced until the peak
    does not increase, guaranteeing a non‑increasing objective.
    """
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    current_score = _max_convolution_fft(a)

    while time.time() < deadline:
        A = _build_conv_matrix(a)
        c = -np.ones(n)                                 # linprog minimizes
        b_ub = np.full(2 * n - 1, current_score, dtype=np.float64)
        bounds = [(0.0, None)] * n

        time_left = deadline - time.time()
        if time_left <= 0.0:
            break
        options = {"presolve": True, "time_limit": max(0.1, time_left)}
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options=options)

        if not res.success:
            break

        g = np.maximum(res.x, 0.0)
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        g = g * (target_sum / sum_g)

        # try to mix; if the peak would rise we halve the mixing factor
        t = 0.5
        improved = False
        while t > 1e-4:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            new_score = _max_convolution_fft(a_candidate)
            if new_score <= current_score + 1e-12:
                a = a_candidate
                current_score = new_score
                improved = True
                print(f"[{time.time():.1f}] LP mix t={t:.4f} → score={new_score:.6f}")
                break
            t *= 0.5

        if not improved:
            break

    return a

# Simulated‑annealing polishing (unchanged)

def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or rng.random() < np.exp(-dE / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # uniform plus tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a
