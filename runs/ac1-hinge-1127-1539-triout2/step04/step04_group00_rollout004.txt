def _optimize_subgrad(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0*n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        # gradient = 2 * reversed a aligned with shift k_max
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        # 2*a[idx_j] factor
        g[idx_i] = 2.0 * a[idx_j]

        # step size: diminishing 1/sqrt(t) scaled by current max
        eta = 0.2 / np.sqrt(t)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # occasional random perturbation (swap)
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur or rng.random() < 0.2:
                    a = cand
                    cur = cand_max
                    if cur < best_score:
                        best_score = cur
                        best_a = a.copy()
        if t % 5000 == 0 and best_score == cur:
            break
    return best_a, best_score

"""
Propose a step‑function that (approximately) minimizes the self‑convolution
max‑norm used by ``evaluate_sequence``.
==========================================================================

The objective is scale‑invariant, therefore we always keep the ℓ₁‑norm
of the candidate equal to √(2·n).  The algorithm works as follows:

1. **Multiple restarts** – while time permits we repeatedly initialise a
   candidate of a randomly chosen length ``n`` (800…3000).  The support
   size is chosen roughly 1.6·√n and the non‑zero entries are placed at
   random positions with equal mass (a sparse start).

2. **Smooth Adam phase** – a soft‑max approximation of the max‑norm is
   minimised using the Adam optimiser.  The gradient of the soft‑max
   weighted sum of the convolution entries can be computed by a single
   FFT‑based convolution (see ``_conv_with_weights``).  The temperature
   of the soft‑max is increased slowly so that the optimisation ends
   with a good approximation of the true maximum.

3. **Sub‑gradient refinement** – the worst shift is identified and a
   (scaled) sub‑gradient of the true max‑norm is taken.  A diminishing
   step‑size together with occasional random “mass‑swap’’ moves helps to
   escape plateaus.

4. **Global bookkeeping** – the best vector (and its achieved max)
   encountered over all attempts is kept.  If no improvement is made
   for a while a fresh random start is performed; otherwise the current
   best vector is refined further.

The whole search respects the 1000 s budget (leaving a 5 s safety
margin).  The final candidate is returned as a plain Python list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max helper (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j) (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser for the smooth phase
# ----------------------------------------------------------------------


def _optimize_smooth_adam(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max approximation of max(conv).
    Returns the best vector seen and the associated max(conv) value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # hyper‑parameters
    lr0 = 0.07               # initial learning rate
    lr = lr0
    lr_decay = 0.9998
    alpha = 0.4              # temperature start
    alpha_mul = 1.008
    max_alpha = 30.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    t = 0
    while time.time() < deadline:
        t += 1

        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a, S)

        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # schedule updates
        lr *= lr_decay
        alpha = min(alpha * alpha_mul, max_alpha)

        # occasional random mass‑swap (helps escape plateaus)
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.02 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.15:
                    a = cand
                    # recompute a few helpers for the next iteration
                    # (we keep the Adam moments – they remain sensible)

        # early stop if stagnating
        if t % 3000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation (focus on the worst shift)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑norm (only the worst entry).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the worst entry (factor 2 included)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = 2.0 * a[idx_j]

        eta = 0.25 / np.sqrt(it)            # diminishing step size
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random “mass‑swap’’ perturbation
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # occasional early stop
        if it % 6000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (smooth → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase followed by a sub‑gradient refinement.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 60 % of the remaining budget to the smooth phase
    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_smooth_adam(a0, smooth_deadline, rng)

    # 40 % (or whatever is left) to sub‑gradient
    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    performs many random restarts, and uses a smooth‑Adam phase
    followed by a sub‑gradient refinement.  The best candidate found
    is returned as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Prepare a possible warm‑start from a previously discovered sequence
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    last_improve_time = start_time
    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # ------------------------------------------------------------------
        # Choose problem size and initialise a candidate
        # ------------------------------------------------------------------
        if best_overall_seq is None or (time.time() - last_improve_time > 30):
            # fresh random start – sparse initialisation
            n = int(rng.integers(800, 3000))
            # support size roughly 1.6·√n (a factor a bit larger than the
            # theoretical √n bound of Sidon sets, allowing a few collisions)
            m = max(2, int(rng.uniform(1.5, 2.0) * np.sqrt(n)))
            idx = rng.choice(n, size=m, replace=False)
            a0 = np.zeros(n, dtype=float)
            a0[idx] = 1.0
            a0 = a0 / a0.sum() * np.sqrt(2.0 * n)   # simplex normalisation
        else:
            # refine the current best – keep its length
            a0 = best_overall_seq.copy()
            n = a0.shape[0]

        # ------------------------------------------------------------------
        # Allocate a slice of the remaining budget for this attempt
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + 12.0, deadline)

        # ------------------------------------------------------------------
        # Optimise (smooth Adam → sub‑gradient)
        # ------------------------------------------------------------------
        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        # ------------------------------------------------------------------
        # Record global best
        # ------------------------------------------------------------------
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={n} m={m if 'm' in locals() else 'NA'} "
                  f"new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [np.sqrt(2.0)]          # fallback: single‑element sequence
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
