def _optimize_smooth_adam(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    lr0 = 0.4
    alpha = 0.5
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_simplex(a, S)

        # occasional random swap
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.02 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # schedule alpha to increase
        alpha = min(alpha * 1.005, 200.0)
        # early break if not improving
        if t % 5000 == 0 and best_score == cur_score:
            break
    return best_a, best_score

def _optimize_subgrad_proper(a0, deadline, rng):

        if rng.random() < 0.04:

def _subgrad_from_conv(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    # first term: a_{k - i}
    i_low = max(0, k - (n - 1))
    i_high = min(k, n - 1)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        g[idx] += a[k - idx]
    # second term: a_{i - k}
    i_low2 = max(k, 0)
    i_high2 = n - 1
    if i_low2 <= i_high2:
        idx2 = np.arange(i_low2, i_high2 + 1)
        g[idx2] += a[idx2 - k]
    return g

"""
Improved step‑function search.

The goal is to minimise  max_k  Σ_i a_i a_{i+k}
under a_i ≥ 0 and Σ_i a_i = √(2·n).
The evaluation value equals that maximum, so we directly minimise it.

Algorithm
---------
1.  Initialise a candidate on the simplex (∑ = √(2·n)).
    Several seeds are tried: a jittered copy of a previous best,
    a Hann window, or a random Dirichlet draw.
2.  **Smooth phase** – minimise a soft‑max approximation of the
    max‑convolution using Adam (adaptive momentum).  The gradient of
    Σ w_k·conv_k is obtained by a weighted convolution (FFT).  The
    temperature α grows so the smooth objective approaches the true
    max.
3.  **Sub‑gradient phase** – work with the exact worst shift.
    The true sub‑gradient of conv_{k_max} w.r.t. a is
        g_i = a_{k_max-i} (if in range) + a_{i-k_max} (if in range).
    A decreasing step‑size η = η₀/√t is used, followed by a projection
    onto the simplex.
4.  Random “mass‑swap” perturbations are inserted in both phases to
    escape plateaux.
5.  The whole process is repeatedly restarted (different n, different
    seeds) until the global time budget (≈ 1000 s) is exhausted.
   The best vector found is returned as a plain list – it is finally
   evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ w_k·(a*a)_k.
    Returns 2·Σ_i w_{i+j}·a_i for each j.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # picks the needed slice
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (smooth approximation of the max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generator
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ₁‑norm √(2·n).
    Tries a copy of ``base_seq`` (if long enough) or a Hann window
    or a random Dirichlet draw.
    """
    S = np.sqrt(2.0 * n)

    # 1. copy from previous best (tiny jitter)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=1e-3 * a.mean(), size=n)   # tiny noise
        a = np.clip(a, 0.0, None)

    else:
        # 2. deterministic start – Hann window (zero at the ends)
        kind = rng.choice(["hann", "uniform", "random"])
        if kind == "hann":
            i = np.arange(n)
            a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        elif kind == "uniform":
            a = rng.random(n)
        else:  # random Dirichlet (uniform on simplex)
            a = rng.random(n)

        a = np.clip(a, 0.0, None)

    # 3. scale to the required ℓ₁‑norm
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Sub‑gradient for a given shift
# ----------------------------------------------------------------------


def _subgrad_from_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact sub‑gradient of the convolution entry at shift k:
        g_i = a_{k‑i} (if defined) + a_{i‑k} (if defined)
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    # term a_{k-i}
    i_low = max(0, k - (n - 1))
    i_high = min(k, n - 1)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        g[idx] += a[k - idx]

    # term a_{i-k}
    i_low2 = max(k, 0)
    i_high2 = n - 1
    if i_low2 <= i_high2:
        idx2 = np.arange(i_low2, i_high2 + 1)
        g[idx2] += a[idx2 - k]

    return g


# ----------------------------------------------------------------------
# Smooth phase – Adam optimiser
# ----------------------------------------------------------------------


def _optimize_smooth_adam(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.4

    # soft‑max temperature
    alpha = 0.5

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    t = 0
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_simplex(a, S)

        # occasional tiny mass‑swap to escape plateaux
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.02 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase temperature (make the surrogate sharper)
        alpha = min(alpha * 1.005, 200.0)

        # cheap early‑stop test
        if t % 5000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase – exact max entry
# ----------------------------------------------------------------------


def _optimize_subgrad_proper(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.2
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = _subgrad_from_conv(a, k_max)
        eta = eta0 / np.sqrt(t)
        a -= eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random perturbation (mass‑swap) with small probability
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early‑stop test
        if t % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (smooth → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined_v2(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase for ~55 % of the remaining budget,
    then finish with the exact sub‑gradient phase.
    The best intermediate solution is returned.
    """
    now = time.time()
    remain = deadline - now
    if remain <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # smooth part
    smooth_deadline = now + 0.55 * remain
    a_s, score_s = _optimize_smooth_adam(a0, smooth_deadline, rng)

    # sub‑gradient refinement (if time left)
    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad_proper(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s
