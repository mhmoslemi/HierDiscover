"""
Propose a non‑negative step‑function (a list of heights) that minimises
the competition objective

    evaluate_sequence(seq)  =  2·n·max(conv(seq,seq)) / (sum(seq)**2)

Because the normalisation used in the search (∑a_i = √(2·n)) makes the
denominator constant, the problem is equivalent to *minimising the
maximum entry of the self‑convolution*.

The algorithm works in three stages:

1. **Exploration of many lengths** – we start with fairly large vectors
   (up to 20 000 entries).  For each length a handful of random
   “edge‑heavy’’ families (power‑law, exponential, mixed) are generated
   and projected onto the ℓ¹‑simplex {x ≥ 0, Σx = √(2n)}.  The best few
   candidates are kept.

2. **Projected gradient descent on a log‑sum‑exp smooth surrogate** – a
   fast FFT‑based gradient of the smoothed maximum is taken, a momentum
   term is added and the iterate is projected back onto the simplex.  The
   smoothing parameter β is annealed aggressively so the algorithm quickly
   approaches the true max‑norm.

3. **Simple sub‑gradient flattening** – starting from the best vector of
   step 2 we repeatedly compute the exact convolution, identify the shift
  (s) achieving the current maximum and take a small step opposite to the
   sub‑gradient (the sum of the partner entries).  After each step the
   vector is projected onto the simplex again.  This step is cheap and
   drives the maximum down close to the theoretical lower bound 1.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence found.  If a global variable ``height_sequence_1``
is present it is used as a warm‑start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Duchi et al. (2008) Euclidean projection onto the ℓ¹‑simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ★ p)  with   p_k ∝ exp(β·b_k)
    The convolution a ★ p is performed with FFTs.
    """
    n = a.shape[0]

    # ----- b = a*a -------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities -----------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stability
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ★ p) ----------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    A_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(A_fft * P_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # centre n entries
    return g, b


# ----------------------------------------------------------------------
# Projected (momentum) gradient descent on the smoothed max
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp surrogate.
    Stops when the time budget expires.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 10.0               # start a bit away from the pure max
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9                 # momentum decay
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum) + 1e-12
        # step size scales inversely with gradient norm and with β
        lr = 0.6 * target_sum / (norm_m * (beta + 1.0))
        a -= lr * momentum

        # projection back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0007 / (beta + 1.0),
                                 size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # geometric increase of the smoothing parameter
        if it % 50 == 0 and beta < max_beta:
            beta = min(beta * 1.12, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Simple sub‑gradient flattening of the exact max‑norm
# ----------------------------------------------------------------------


def _flatten_max_subgrad(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly moves a tiny amount of mass from coordinates that
    contribute most to the current maximum convolution value to
    those that contribute least, always staying on the simplex.
    """
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    it = 0
    while time.time() < deadline and it < 5000:
        conv = _full_convolution_fft(a)
        max_b = conv.max()
        # identify all shifts attaining the maximum (within tiny tolerance)
        eps = max_b * 1e-12
        ks = np.where(conv >= max_b - eps)[0]

        # build sub‑gradient (sum of partner entries for each maximal shift)
        g = np.zeros_like(a)
        for k in ks:
            d = k - (n - 1)                     # actual shift distance
            if d == 0:
                g += 2.0 * a
            elif d > 0:
                g[:n - d] += a[d:]
                g[d:] += a[:n - d]
            else:  # d < 0
                dneg = -d
                g[dneg:] += a[:n - dneg]
                g[:n - dneg] += a[dneg:]

        # a small step opposite to the sub‑gradient
        norm = np.linalg.norm(g) + 1e-12
        step = 0.35 * target_sum / (norm * (it + 1))
        a -= step * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        it += 1

    return a

# Random initialisers (edge‑heavy families)
# ----------------------------------------------------------------------


def _random_edge_heavy(n: int, rng) -> np.ndarray:
    """Generate a random edge‑heavy sequence and project onto the simplex."""
    target_sum = np.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)

    kind = rng.integers(0, 5)
    if kind == 0:               # pure power‑law (decreasing)
        p = rng.uniform(0.05, 1.5)
        a = (idx + 1.0) ** (-p)
    elif kind == 1:             # exponential decay from the left
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)
    elif kind == 2:             # mixed power‑law + exponential
        p = rng.uniform(0.05, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif kind == 3:             # symmetric edge‑heavy (high at both ends)
        p = rng.uniform(0.15, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right
    else:                       # uniform with tiny jitter
        a = np.full(n, target_sum / n)

    # add a modest Gaussian perturbation and project
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the competition
    objective within the allotted time budget.
    Returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any.
    # ------------------------------------------------------------------
    init = globals().get("height_sequence_1")
    if isinstance(init, (list, tuple, np.ndarray)):
        a0 = np.asarray(init, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        val = _max_convolution_fft(a0)
        best_seq, best_val = a0.copy(), val
        print(f"[{time.time():.1f}] Warm‑start from global n={n0} val={val:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – we start with fairly large vectors.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000]   # will be truncated by time

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        print(f"[{time.time():.1f}] Exploring length n={n}")

        # --------------------------------------------------------------
        # Phase 0 – diverse random initialisation (quick evaluations)
        # --------------------------------------------------------------
        phase0_end = min(deadline, time.time() + 6.0)   # ~6 s per length
        while time.time() < phase0_end:
            a = _random_edge_heavy(n, rng)
            val = _max_convolution_fft(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} val={val:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – projected gradient descent on smoothed max
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            sub_dead = min(deadline, time.time() + 22.0)    # allocate ~22 s
            a_gd = _gradient_descent_opt(best_seq, sub_dead)
            val = _max_convolution_fft(a_gd)
            if val < best_val:
                best_val, best_seq = val, a_gd.copy()
                print(f"[{time.time():.1f}] GD improvement n={n} val={val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – simple sub‑gradient flattening of the exact max
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n and time.time() < deadline:
            sub_dead = min(deadline, time.time() + 20.0)    # allocate ~20 s
            a_flat = _flatten_max_subgrad(best_seq, sub_dead)
            val = _max_convolution_fft(a_flat)
            if val < best_val:
                best_val, best_seq = val, a_flat.copy()
                print(f"[{time.time():.1f}] Sub‑grad improvement n={n} val={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_val = _max_convolution_fft(best_seq)

    # Trim trailing near‑zero entries for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final sanity check using the official evaluator (cheap)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished – best value = {best_val:.6f}, "
          f"length = {best_seq.size}")
    return best_seq.tolist()
