def _lp_step(a, conv):
    n = a.shape[0]
    S = np.sqrt(2.0*n)

    # Build sparse matrices
    from scipy.sparse import diags, eye, hstack, vstack, csr_matrix

    # diagonal values: 2*a[d]; generate list of arrays of length n for each diagonal
    diag_vals = [np.full(n, 2*a[d]) for d in range(n)]
    offsets = np.arange(n)
    A_conv = diags(diag_vals, offsets, shape=(2*n-1, n), format='csr')
    # Append t column = -1
    t_col = csr_matrix((-np.ones((2*n-1, 1)), (np.arange(2*n-1), np.zeros(2*n-1))), shape=(2*n-1, 1))
    # Actually easier: t_col = -np.ones((2*n-1,1))
    t_col = csr_matrix(-np.ones((2*n-1, 1)))
    A_ub_conv = hstack([A_conv, t_col])

    # delta bound constraints: -delta_i <= a_i => -delta_i - 0*t <= a_i
    A_ub_bound = hstack([-eye(n), csr_matrix((n,1))])
    b_ub_bound = a.copy()

    A_ub = vstack([A_ub_conv, A_ub_bound])
    b_ub = np.concatenate([-conv, a])

    # equality sum(delta) = 0
    A_eq = csr_matrix(np.concatenate([np.ones((1,n)), np.zeros((1,1))], axis=1))
    b_eq = np.array([0.0])

    # bounds
    bounds = [(-a[i], None) for i in range(n)] + [(0, None)]

    # objective: minimize t
    c = np.zeros(n+1)
    c[-1] = 1.0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5})
    if not res.success:
        return np.zeros_like(a), np.max(conv) # fallback no change
    delta = res.x[:n]
    t_opt = res.x[-1]
    return delta, t_opt
```

"""Propose a step‐function (non‑negative vector) that minimizes the
self‑convolution max‑norm used by ``evaluate_sequence``.

Algorithm
---------
* Choose a random length n (≈ 1200‑2500) and initialise a non‑negative
  vector a on the simplex Σa = √(2·n) (either from the previous best
  construction or a fresh Dirichlet draw).
* **LP refinement** – repeatedly solve the linear program  

      maximise   Σ g_j
      subject to  (a * g)_k ≤ M   for all k          (M = max(a*a))
                  g_j ≥ 0

  The optimal g has a larger ℓ₁‑mass than a while respecting the same
  convolution bound.  Scale g back to the required ℓ₁‑norm and move a
  towards it (convex combination).  If the move reduces the max‑convolution
  value we keep it; otherwise we try smaller steps or a tiny random mass
  transfer.  This step is the core of the *linear‑programming boost* that
  improves on pure gradient descent.
* **Sub‑gradient polishing** – after the LP phase we run a short
  sub‑gradient descent on the true max‑norm (the worst shift gives a
  sub‑gradient).  This cleans up remaining small peaks.
* The whole routine is time‑aware: it performs many independent restarts
  until the global time budget is exhausted and keeps the best sequence
  seen.  The final sequence is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, eye, hstack, vstack, csr_matrix

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution (FFT, length 2·n‑1)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Linear program: maximise ℓ₁‑mass of g under a * g ≤ M
# ----------------------------------------------------------------------


def _solve_lp_max_mass(a: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """
    Solve:
        maximise   Σ g_j
        subject to  (a * g)_k ≤ M   (k = 0 … 2n‑2)
                    g_j ≥ 0

    Returns the optimal g and its ℓ₁‑norm.
    """
    n = a.shape[0]

    # Build the convolution matrix A where A[k, j] = a_{k‑j} (or 0 if out‑of‑range)
    # It is a Toeplitz matrix with constant diagonals equal to the entries of a.
    diag_data = [np.full(n, a[d]) for d in range(n)]          # constant on each diagonal
    offsets = np.arange(n)                                    # diagonal offsets 0 … n‑1
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')

    b = np.full(2 * n - 1, M, dtype=float)                   # RHS: M for every shift
    c = -np.ones(n)                                           # minimise -Σg ⇒ maximise Σg
    bounds = [(0.0, None)] * n                                # g ≥ 0

    # HiGHS (default) handles sparse A efficiently.
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method='highs',
        options={'presolve': True},
    )
    if not res.success:
        return np.zeros_like(a), 0.0
    g = res.x
    return g, g.sum()


# ----------------------------------------------------------------------
# LP‑based refinement of a single vector
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray,
               S: float,
               deadline: float,
               rng: np.random.Generator) -> np.ndarray:
    """
    Perform a few LP‑boost steps on ``a``.
    The ℓ₁‑norm of ``a`` is assumed to be ``S`` (sqrt(2·n)).
    Stops when time runs out or when no improvement can be made.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = np.max(_full_convolution(a))

    while time.time() < deadline:
        conv = _full_convolution(a)
        M = np.max(conv)

        # keep global best inside this refinement
        if M < best_val:
            best_val = M
            best = a.copy()

        # Solve the LP that maximises the ℓ₁‑mass under the current bound M
        g, sum_g = _solve_lp_max_mass(a, M)

        # If the LP cannot increase the mass, stop the refinement
        if sum_g <= S + 1e-12:
            break

        # Scale g back to the required ℓ₁‑norm
        scale = S / sum_g
        g_scaled = g * scale

        # Try to move a towards g_scaled; several step sizes are tried.
        improved = False
        for t in (0.5, 0.3, 0.1, 0.05):
            cand = (1.0 - t) * a + t * g_scaled
            # numerical safety – enforce non‑negativity and exact sum
            cand = np.maximum(cand, 0.0)
            cand = cand * (S / cand.sum())
            cand_max = np.max(_full_convolution(cand))
            if cand_max < M - 1e-12:
                a = cand
                improved = True
                break

        if not improved:
            # Small random “mass‑swap” to escape a possible plateau
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.02 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
            # If even the random swap does not help, give up on this vector.
            break

    return best


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑norm (focus on the worst shift)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent to polish a vector."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: g_i = a_{k_max‑i} (if the index exists)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional tiny random perturbation
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # cheap early stop
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimizing ``evaluate_sequence``.
    The method combines an LP‑boost phase (theoretical improvement step)
    with a short sub‑gradient polishing phase and many random restarts.
    Returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the external call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Try to use the previously discovered construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    while time.time() < deadline:
        attempt += 1

        # --------------------------------------------------------------
        # Pick a problem size (n) and an initial vector
        # --------------------------------------------------------------
        n = int(rng.integers(1200, 2500))            # moderate size
        S = np.sqrt(2.0 * n)

        if base_seq is not None and base_seq.size >= n and rng.random() < 0.5:
            a = base_seq[:n].astype(float).copy()
            a = np.clip(a, 0.0, None)
        else:
            a = rng.random(n)                       # Dirichlet(1) – uniform on simplex

        a = a / a.sum() * S                           # project onto the simplex

        # --------------------------------------------------------------
        # LP‑boost refinement (takes a few seconds)
        # --------------------------------------------------------------
        lp_deadline = min(time.time() + 12.0, deadline)
        a = _lp_refine(a, S, lp_deadline, rng)

        # --------------------------------------------------------------
        # Short sub‑gradient polishing (a few seconds left)
        # --------------------------------------------------------------
        if time.time() < deadline:
            sub_deadline = min(time.time() + 8.0, deadline)
            a, _ = _optimize_subgrad(a, sub_deadline, rng)

        # --------------------------------------------------------------
        # Record the best candidate seen so far
        # --------------------------------------------------------------
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
            print(f"[global] attempt={attempt} n={n} "
                  f"new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
