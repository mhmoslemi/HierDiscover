    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n and rng.random() < 0.5:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean()*1e-3, size=n)

    else:

        if style == 'dirichlet':

"""
Proposes a step‑function (non‑negative vector) that minimizes the
self‑convolution max‑norm used by ``evaluate_sequence``.
The algorithm splits the available time into many short attempts with
different lengths and initialisations.  Each attempt runs a smooth
Adam optimisation on a soft‑max approximation of the max‑norm followed
by a sub‑gradient phase that directly reduces the current worst
convolution entry.  Updates are performed with multiplicative (mirror)
steps and occasional random “mass‑swap’’ moves to escape plateaus.
The best sequence found before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – required after additive updates.
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1                     # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)                     # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights – smooth approximation of the max entry.
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j) (stable)."""
    scaled = alpha * conv
    m = np.max(scaled)
    exp_shifted = np.exp(scaled - m)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generator – several families of shapes.
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    If ``base_seq`` is supplied and sufficiently long, a jittered copy
    is used; otherwise a random shape (uniform, exponential, sinusoidal,
    linear) is sampled.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n and rng.random() < 0.5:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        shape = rng.choice(['dirichlet', 'exp', 'sin', 'lin'])
        if shape == 'dirichlet':
            a = rng.random(n)                     # uniform on simplex
        elif shape == 'exp':
            lam = rng.uniform(0.0002, 0.005)        # decay rate
            a = np.exp(-lam * np.arange(n))
        elif shape == 'sin':
            x = np.arange(n) + 0.5
            a = np.sin(np.pi * x / n) ** 2
        else:  # 'lin'
            a = np.linspace(1.0, 0.0, n)           # decreasing linear ramp

        # add a tiny floor to avoid exact zeros (helps multiplicative updates)
        a += rng.uniform(0.0, 1e-6, size=n)

    # normalise to the required ℓ¹‑norm
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Smooth optimisation phase – Adam on a soft‑max surrogate.
# ----------------------------------------------------------------------


def _optimize_smooth_adam(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam‑based descent on soft‑max approximation of the max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature (smoothness) schedule
    alpha = 0.5
    max_alpha = 80.0
    alpha_mul = 1.015

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    t = 0
    while time.time() < deadline:
        t += 1
        conv = _full_convection := _full_convolution(a)  # compute once
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)                 # shape (n,)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a, S)

        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # temperature increase
        alpha = min(alpha * alpha_mul, max_alpha)

        # occasional random mass‑swap for escaping tiny plateaus
        if rng.random() < 0.03:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.015 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # accept if better or with small probability
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # cheap early stop if stagnating
        if t % 6000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase – direct reduction of the worst convolution entry.
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (worst shift only)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: g_i = a_{k_max-i}   (if index in range)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random perturbation (helps leave tiny plateaus)
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.02 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # early stop if no progress for a while
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser – smooth phase → sub‑gradient refinement.
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase (≈60 % of the allocated time) followed by
    a sub‑gradient phase (≈40 %).  Returns the best intermediate solution.
    """
    now = time.time()
    if now >= deadline:
        return a0, np.max(_full_convolution(a0))

    remaining = deadline - now
    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_smooth_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine performs many short optimisation attempts with
    different lengths and initialisations, keeping the best solution
    found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # try to reuse a previously discovered construction if available
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main search loop – short attempts interleaved with occasional
    # refinements of the current global best.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # decide whether to start a fresh random attempt or to refine
        # the current best (if recent improvement happened)
        if best_seq is None or (time.time() - last_improve > 25.0):
            n = rng.integers(2000, 12000)          # favour larger problems
            a0 = _initial_sequence(base_seq, n, rng)
        else:
            # mostly keep the same length, occasionally perturb it
            if rng.random() < 0.2:
                # a small random change of length (adds/removes a few zeros)
                delta_n = rng.integers(-200, 200)
                n = max(800, min(15000, best_seq.shape[0] + delta_n))
                a0 = _initial_sequence(best_seq, n, rng)
            else:
                a0 = best_seq.copy()
                n = a0.shape[0]

        # allocate a slice of the remaining budget for this attempt
        slice_deadline = min(time.time() + 8.0, deadline)   # ~8 s per trial

        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final elongation / refinement of the best candidate (use rest of time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # give the remaining budget to a longer refinement
        final_seq, final_score = _optimize_combined(best_seq, deadline, rng)
        best_seq = final_seq
        best_score = final_score

    # ------------------------------------------------------------------
    # Return the result as a plain Python list.
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]               # fallback, should never happen
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
