def _local_search(a, S, max_iters, rng):
    best_a = a.copy()
    best_max = np.max(_full_convolution_fft(best_a))
    n = len(a)
    for it in range(max_iters):
        # pick i random, j random different
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # choose epsilon relative to a[i]
        eps = rng.uniform(0, min(best_a[i], S) * 0.1)  # move some mass
        # propose new a
        a_new = best_a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # Ensure non-negative
        if a_new[i] < 0:
            continue
        # no need to project since sum unchanged and nonnegative
        cur_max = np.max(_full_convolution_fft(a_new))
        if cur_max < best_max - 1e-9:  # accept improvement
            best_max = cur_max
            best_a = a_new
    return best_a, best_max
```

# Projection onto the ℓ₁‑simplex {x ≥ 0, Σx = S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum S."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (linear convolution) using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient (soft‑max) of max‑convolution – used by the Adam phase
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Gradient of τ·log Σ exp(b/τ) where b = a*a.
    Returns (gradient, full convolution, current max entry).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                # exact max → sub‑gradient
        k = np.argmax(b)
        # compute gradient of b_k = Σ a_i a_{k-i}
        g = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            g[i] += a[j]
            if i != j:
                g[j] += a[i]
        return g, b, max_b

    # soft‑max weights
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)            # shape (2n‑1,)

    # gradient = 2·(a ⋆ w)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # central part, length n
    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max of the max‑convolution)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """Adam optimisation of the max‑convolution (soft‑max version)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6
    t = 0

    # initialise temperature
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam pre‑projection step
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluate
        cur_max_new = np.max(_full_convolution_fft(a_new))

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        # occasional jitter
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a raw (unnormalised) vector from a random family."""
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular",
         "power", "cosine", "beta", "twostep"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.3)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.3)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.4) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        dec = inc[::-1] if n % 2 == 0 else inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 3.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "beta":
        alpha = rng.uniform(0.1, 5.0)
        beta = rng.uniform(0.1, 5.0)
        x = (idx + 0.5) / n               # avoid extremes
        a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))

    elif typ == "twostep":
        # two‑level step: first m entries high, rest low
        m = rng.integers(1, n)
        high = rng.uniform(0.5, 2.0)
        low = rng.uniform(0.0, 0.1)
        a = np.empty_like(idx)
        a[:m] = high
        a[m:] = low

    else:
        a = rng.random(n)

    # small jitter for diversity
    if rng.random() < 0.2:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# Local stochastic refinement (hill climbing with occasional uphill moves)
# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray, deadline: float, rng: np.random.Generator):
    """Greedy hill‑climbing / simulated‑annealing improvement."""
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_max = np.max(_full_convolution_fft(best_a))

    # initial temperature (allow some uphill moves)
    T0 = best_max * 0.05
    T = T0
    decay = 0.9995

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        # move a small amount from i to j
        eps = rng.uniform(0.0, best_a[i] * 0.2 + 1e-12)
        a_new = best_a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:
            continue

        # projection not needed – sum unchanged and non‑negative preserved
        cur_max = np.max(_full_convolution_fft(a_new))

        delta = cur_max - best_max
        if delta < -1e-12 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            best_a = a_new
            best_max = cur_max
            T = max(T * decay, 1e-8)   # cool down

    # final projection to guarantee exact sum (helps the Adam phase)
    best_a = _project_to_simplex(best_a, S)
    best_max = np.max(_full_convolution_fft(best_a))
    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (height vector) with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0                                 # keep a small safety margin
    deadline = time.time() + budget_s - safety

    best_seq = None
    best_max = np.inf

    # ------------------------------------------------------------------
    # Initialise from a previously found construction (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = seq0.shape[0]
            S0 = np.sqrt(2.0 * n0)
            seq0 = _project_to_simplex(seq0, S0)
            cur_max = np.max(_full_convolution_fft(seq0))
            best_seq = seq0.copy()
            best_max = cur_max
            print(f"[init] imported previous solution: n={n0}, max={best_max:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – intensive random parametric search
    # ------------------------------------------------------------------
    t_start = time.time()
    phase1_deadline = t_start + 0.55 * (deadline - t_start)

    attempts = 0
    while time.time() < phase1_deadline:
        attempts += 1
        n = rng.integers(800, 5000 + 1)          # moderate to large sizes
        raw = _random_distribution(n, rng)
        S = np.sqrt(2.0 * n)
        a = _project_to_simplex(raw, S)

        cur_max = np.max(_full_convolution_fft(a))

        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
            print(f"[search] attempt={attempts} n={n} new best max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – local stochastic refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        phase2_deadline = time.time() + 0.25 * (deadline - time.time())
        refined_seq, refined_max = _local_refine(best_seq, phase2_deadline, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_seq.copy()
            print(f"[local] improved max to {best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam polishing (runs for the remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None:
        refined_seq, refined_max = _adam_optimize(best_seq, deadline, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_seq.copy()
            print(f"[adam] improved max to {best_max:.6f}")

    # ------------------------------------------------------------------
    # Final return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]               # fallback – should never happen
    else:
        final_seq = best_seq.tolist()

    # external evaluation (the harness expects this printout)
    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_max:.6f}) length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
