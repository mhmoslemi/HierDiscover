def _poly_sequence(n: int, degree: int, max_val: float = 1000.0) -> list[float]:
    """
    Return a list of length ``n`` with entries proportional to i**degree
    (i = 0,…,n‑1) scaled so that the maximum entry equals ``max_val``.
    All entries are non‑negative and the list is suitable for
    ``evaluate_sequence``.
