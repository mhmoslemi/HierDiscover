def subgradient_step(a, target_sum, step_size):
    n = len(a)
    conv = _conv_fft(a)   # linear conv length 2n-1
    max_val = conv.max()
    # active indices
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # reversed slice of a[lo:hi+1], but need a[k - np.arange(lo, hi+1)]
        idx = np.arange(lo, hi+1)
        g[lo:hi+1] += a[k - idx]
    g *= 2.0
    # Normalize gradient step size
    norm = np.linalg.norm(g)
    if norm > 0:
        eta = step_size / norm
        a = a - eta * g
    # projection to simplex
    a = _project_to_simplex(a, target_sum)
    # remove negative rounding errors.
    a = np.maximum(a, 0.0)
    return a, max_val
```

"""
Propose a step‑function with a low max‑convolution value.

The evaluation score for a sequence a (with ℓ¹‑norm √(2·n))
is exactly the maximal entry of the linear convolution a*a.
The goal is therefore to make the convolution as flat as possible,
i.e. to bring its maximum down towards its average value
≈ 2·n/(2·n‑1) ≈ 1.

Method
------
1.  Try several lengths n (shorter vectors give much cheaper
    iterations – we can afford many more updates).
2.  Initialise a with a random power‑law and project onto the simplex
    Σa_i = √(2·n) (the projection keeps the ℓ¹‑norm fixed, which makes
    the evaluation scale‑invariant).
3.  Repeatedly perform a *sub‑gradient descent* on the max‑convolution:
        • compute the full convolution by FFT,
        • locate the active indices where the maximum is attained,
        • build the sub‑gradient (for each active k the gradient is
          2·a_{k-i} on the overlapping window),
        • step opposite to the gradient with a decaying learning rate,
        • project back onto the simplex.
    A small Gaussian jitter is added every few thousand iterations to
    escape plateaus.
4.  After the gradient phase, a short simulated‑annealing refinement
    (mass‑transfer / local smoothing moves) is applied if time permits.
5.  The best sequence found among all explored lengths is returned.

The whole search respects the 1000 s budget, uses only NumPy (no file
I/O) and never produces negative or overly large numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = s, x ≥ 0)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s} (Duchi et al., 2008)."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple power‑law initialisation
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create (i+1)^(-p) of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient step for the max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, target_sum: float,
                      step_size: float) -> tuple[np.ndarray, float]:
    """
    Perform one sub‑gradient update for ``max(a*a)`` and re‑project.
    Returns the updated vector and the new maximum convolution value.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    # build sub‑gradient (sum of gradients of all active maxima)
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[lo:hi + 1] += a[k - idx]
    g *= 2.0

    norm = np.linalg.norm(g)
    if norm > 0:
        a = a - step_size * g / (norm + 1e-12)

    # keep feasibility (non‑negativity + ℓ¹‑norm)
    a = _project_to_simplex(a, target_sum)
    a = np.maximum(a, 0.0)
    # recompute the maximum after projection
    new_max = _max_convolution_fft(a)
    return a, new_max

        if rng.random() < 0.70:                     # mass transfer
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)
            low_thr = np.percentile(weight, 10)
            low_idxs = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idxs) if low_idxs.size else rng.integers(n)
            if donor == receiver:
                continue
            delta = a[donor] * rng.random() * max_frac
            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                                        # local smoothing
            start_w = rng.integers(0, n)
            length = rng.integers(2, min(12, n - start_w) + 1)
            end_w = start_w + length
            seg = a[start_w:end_w].copy()
            avg = seg.mean()
            a[start_w:end_w] = avg
            move_type = "smooth"
            saved_seg = seg
            saved_start, saved_end = start_w, end_w

        # ---- evaluate new state -------------------------------------------
        new_score = _max_convolution_fft(a)

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved_seg

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Optimisation for a single length n
# ----------------------------------------------------------------------
def _optimise_length(n: int, deadline: float, rng: np.random.Generator) -> tuple[list[float], float]:
    """
    Run sub‑gradient descent (plus optional SA) for a specific vector length.
    Returns the best sequence (as a plain list) and its evaluation score.
    """
    target_sum = np.sqrt(2.0 * n)

    # random power‑law exponent for diversity
    p = rng.uniform(0.55, 0.85)
    a = _power_law_sequence(n, p, target_sum)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    # step‑size schedule
    step = target_sum * 0.30
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        a, cur_score = _subgradient_step(a, target_sum, step)
        step *= decay

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} improved = {best_score:.6f}")

        # occasional jitter to avoid stagnation
        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    # final quick SA if a slice of time is still left
    remaining = deadline - time.time()
    if remaining > 5.0:
        sa_seq, sa_score = _simulated_annealing(best_a, deadline)
        if sa_score < best_score:
            best_a, best_score = sa_seq, sa_score

    # trim trailing zeros for a tidy output
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    return arr.tolist(), float(best_score)


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low max‑convolution value.
    Several vector lengths are tried; the best result found before the budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # candidate lengths – smaller n gives many cheap iterations,
    # larger n gives finer granularity; explore a few of them.
    lengths = [400, 600, 800, 1100, 1500]

    best_seq = None
    best_score = float("inf")

    # allocate time roughly equally, but give earlier (shorter) runs a bit more
    for i, n in enumerate(lengths):
        time_left = deadline - time.time()
        if time_left < 10.0:
            break
        # give each length up to half of the remaining time (but never zero)
        sub_deadline = time.time() + max(5.0, time_left * 0.45)
        seq, score = _optimise_length(n, sub_deadline, rng)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[{time.time():.1f}] overall best updated: score={best_score:.6f} (n={n})")

    # fallback – never happen, but keep a safe default
    if best_seq is None:
        n = 500
        target = np.sqrt(2.0 * n)
        best_seq = _power_law_sequence(n, 0.65, target).tolist()
        best_score = float('inf')

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq
