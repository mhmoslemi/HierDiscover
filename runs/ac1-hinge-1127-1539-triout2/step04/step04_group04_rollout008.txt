"""
Propose a step‑function (non‑negative height vector) that minimizes
evaluate_sequence(sequence) = 2·n·max(a*a)/(∑a)².
The algorithm uses three stages within the allotted time budget:

1️⃣  Broad random search over many lengths and parametric families
    (exponential, symmetric‑exponential, Gaussian, power‑law,
    Beta, and simple linear ramps).  Each candidate is projected onto
    the ℓ₁‑simplex {x ≥ 0, ∑x = √(2·n)} so that the external score
    equals the maximal self‑convolution entry.
    The best candidate seen so far is retained.

2️⃣  Several independent Adam‑optimisation runs (soft‑max approximation
    of the ∞‑norm) starting from the current best vector and from
    slightly perturbed copies.  Hyper‑parameters (learning‑rate,
    temperature decay) are varied between runs to escape shallow local
    minima.

3️⃣  A final “local‑perturbation” phase that repeatedly adds tiny
    Gaussian noise, re‑projects onto the simplex and keeps the move
    only if the maximal convolution entry improves (simple stochastic
    hill‑climbing).

The whole procedure respects the 1000 s time limit, stores the final
vector in the global `height_sequence_1` for reuse, and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the ℓ₁‑simplex of sum S (non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution (full) via real FFT – O(n log n)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())       # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of soft‑max approximation of max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Compute gradient of τ·log∑exp(b/τ) where b = a*a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                     # hard max → sub‑gradient
        k = np.argmax(b)               # index of the maximal entry
        # gradient = 2 * a convolved with a Kronecker delta at shift k
        g = np.zeros_like(a)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] = 2.0 * a[partner[valid]]
        return g, b, max_b

    # soft‑max weights (stable)
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)            # shape (2n‑1,)

    # convolution of a_rev with w  (cross‑correlation a ⋆ w)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # extract length‑n slice
    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection (soft‑max temperature annealing)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   rng: np.random.Generator,
                   lr_init: float = 0.6,
                   tau_factor: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max approximation of the max‑convolution entry.
    Returns the best vector found (projected on the simplex) and its
    maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lr = lr_init

    # initial temperature roughly half of the current max entry
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = max(cur_max * tau_factor, 1e-8)

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1

        # gradient w.r.t. the soft‑max surrogate
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluate true objective (hard max) for possible improvement
        cur_max_new = np.max(_full_convolution_fft(a_new))

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)    # cool temperature slowly
        lr *= 0.9999                    # decay learning rate

        # occasional jitter to jump out of tiny plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    # final fine‑tuning with hard‑max sub‑gradient (small step size)
    # (run only if we still have a few seconds left)
    final_deadline = deadline + 0.0
    if time.time() < final_deadline:
        # one additional pass with decreasing step size
        step_size = lr_init * 0.1
        while time.time() < final_deadline:
            g, _, _ = _gradient_softmax(best_a, tau=0.0)   # true gradient of max
            best_a = best_a - step_size * g
            best_a = _project_to_simplex(best_a, S)
            cur_max_new = np.max(_full_convolution_fft(best_a))
            if cur_max_new < best_max:
                best_max = cur_max_new
            step_size *= 0.995

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families for the initial search
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create an unnormalised non‑negative vector using a random family."""
    typ = rng.choice([
        "uniform", "exp", "symexp", "gaussian", "triangular",
        "power", "cosine", "beta", "linear"
    ])
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.02)          # slower decay favours longer tails
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.02)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.08, 0.25) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        # simple decreasing ramp
        a = np.linspace(1.0, 0.0, n)

    elif typ == "power":
        p = rng.uniform(0.0, 2.5)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "beta":
        # draw from a Beta distribution on [0,1] then map to index space
        alpha = rng.exponential(scale=2.0) + 0.1
        beta = rng.exponential(scale=2.0) + 0.1
        a = rng.beta(alpha, beta, size=n)
        # skew the distribution a little (avoid being too flat)
        a = np.sqrt(a)

    elif typ == "linear":
        # random slope (may be increasing or decreasing)
        slope = rng.uniform(-1.0, 1.0)
        intercept = rng.uniform(0.0, 1.0)
        a = intercept + slope * (idx / (n - 1))
        a = np.clip(a, 0.0, None)

    else:
        a = rng.random(n)

    # occasional tiny random jitter before projection (helps diversity)
    if rng.random() < 0.25:
        a += rng.normal(scale=a.mean() * 0.02, size=n)
        a = np.clip(a, 0.0, None)

    return a

        S = np.sqrt(2.0 * n)
        while time.time() < deadline:
            # tiny random perturbation
            a_new = best_seq.copy()
            scale = a_new.mean() * 0.008
            a_new += rng.normal(scale=scale, size=n)
            a_new = np.clip(a_new, 0.0, None)
            a_new = _project_to_simplex(a_new, S)

            new_obj = np.max(_full_convolution_fft(a_new))
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = a_new.copy()
                print(f"[local] improved max={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Finalise – convert to plain Python list and evaluate via the supplied
    # black‑box to obtain the external score (required by the harness).
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        best_obj = np.inf
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_obj:.6f}) length = {len(final_seq)}"
    )

    # make the solution available for the next run (if any)
    globals()["height_sequence_1"] = final_seq
    return final_seq
