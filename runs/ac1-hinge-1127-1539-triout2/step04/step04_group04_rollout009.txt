assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the heights of a step function) that minimizes  

\[
\text{score}(a)=\frac{2\,n\;\max(a\ast a)}{\bigl(\sum_i a_i\bigr)^2},
\]

where `n = len(a)` and `a∗a` is the full linear convolution.  
Because the score is scale‑invariant, we can fix  

\[
\sum_i a_i = \sqrt{2n}\qquad\Longrightarrow\qquad
\text{score}(a)=\max(a\ast a).
\]

Thus we have to **minimise the largest entry of the self‑convolution** while keeping the vector on the simplex.

The optimiser works in three phases:

1. **Diverse random‑family search** – many parametric families (uniform,
   left/right exponentials, “U‑shaped” edge exponentials, Beta‑shapes,
   Dirichlet draws, power‑law, triangular, cosine) are sampled with random
   lengths `n`.  After projecting onto the ℓ₁‑simplex the convolution
   maximum is evaluated.  The best candidate is retained.

2. **Projected Adam optimisation** – the best candidate from phase 1 is
   refined with Adam on the *soft‑max* approximation of the `max`
   (temperature `τ`).  After each Adam step the vector is projected back
   onto the simplex.

3. **Exponentiated‑gradient (mirror‑descent) optimisation** – the same
   candidate is further polished using an entropy‑mirror step
   `p ← p·exp(‑η·∇)` on the probability vector `p = a / Σa`.  This respects
   the simplex automatically and often escapes shallow local minima
   that Adam can get stuck in.

All phases respect the time budget (default 1000 s).  The final
sequence is stored in the global variable `height_sequence_1` for reuse.

```python
"""
Advanced three‑phase optimiser for the step‑function problem.

Phase 1 – random parametric families (including new “U‑shaped” families).
Phase 2 – Adam optimiser on a soft‑max approximation of the max‑convolution.
Phase 3 – Mirror‑descent (exponentiated‑gradient) optimiser on the simplex.

The code respects the time budget, keeps the best solution seen,
stores it in the global variable `height_sequence_1` and returns it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the ℓ₁‑simplex {x ≥ 0, Σx = S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the simplex Σx = S, x ≥ 0."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via a real FFT ( O(n log n) )
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient (or sub‑gradient) of the soft‑max approximation of max(b)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Compute the gradient of τ·log(∑exp(b/τ)), where b = a*a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                     # hard max → sub‑gradient
        k = np.argmax(b)
        # gradient of b_k w.r.t. a is 2 * shifted version of a
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal term (i=j) already handled above; no extra work needed
        return g, b, max_b

    # soft‑max weights w ∝ exp(b/τ)
    diff = b - max_b                # for numerical stability
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)          # shape (2n‑1,)

    # convolution of a (reversed) with the weight vector w
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # gradient of the smooth objective
    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with projection onto the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Adam optimiser on the soft‑max approximation.
    Returns the best vector and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6                       # initial learning rate
    t = 0

    # initialise temperature τ from a hard‑max scale
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluate hard max for possible improvement
        cur_max_new = np.max(_full_convolution_fft(a_new))
        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)   # cool the soft‑max
        lr *= 0.9999                    # decay learning rate

        # occasional tiny jitter to escape plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser on the simplex
# ----------------------------------------------------------------------
def _eg_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Mirror‑descent optimiser using the natural (entropy) geometry.
    Works directly on the probability vector p = a / Σa.
    Returns the best vector and its maximal convolution entry.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    p = a0 / S                     # now Σp = 1, p ≥ 0

    best_p = p.copy()
    best_max = np.max(_full_convolution_fft(p * S))

    # hyper‑parameters
    eta = 0.5                      # learning‑rate on the mirror space
    # initialise temperature τ similarly to Adam
    _, _, cur_max = _gradient_softmax(p * S, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    while time.time() < deadline:
        # gradient of the smooth objective w.r.t. a, then translate to p
        g_a, _, cur_max = _gradient_softmax(p * S, tau)
        grad_p = g_a * S            # ∂J/∂p = S·∂J/∂a

        # exponentiated‑gradient step: p ← p ⊙ exp(‑η·grad)
        p = p * np.exp(-eta * grad_p)
        p_sum = np.sum(p)
        if p_sum == 0.0:
            p = np.full_like(p, 1.0 / n)
        else:
            p = p / p_sum

        a = p * S
        cur_max_new = np.max(_full_convolution_fft(a))
        if cur_max_new < best_max:
            best_max = cur_max_new
            best_p = p.copy()

        # schedules
        eta *= 0.9999
        tau = max(tau * 0.9995, 1e-7)

    best_a = best_p * S
    return best_a, best_max


# ----------------------------------------------------------------------
# Random‑family generator (new families added)
# ----------------------------------------------------------------------
def _random_family(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an unnormalised non‑negative vector using a randomly chosen
    parametric family.  The families include several U‑shaped / edge‑peak
    constructions that are not present in the original baseline.
    """
    family = rng.choice([
        "uniform", "exp_left", "exp_right", "edge_exp",
        "beta", "dirichlet", "power", "triangular", "cosine"
    ])
    idx = np.arange(n, dtype=float)

    if family == "uniform":
        a = rng.random(n)

    elif family == "exp_left":
        lam = rng.uniform(0.001, 0.5)
        a = np.exp(-lam * idx)                     # peak at left end

    elif family == "exp_right":
        lam = rng.uniform(0.001, 0.5)
        a = np.exp(-lam * (n - 1 - idx))           # peak at right end

    elif family == "edge_exp":
        # high at both ends, decays towards the centre
        lam = rng.uniform(0.001, 0.5)
        d = np.minimum(idx, n - 1 - idx)
        a = np.exp(-lam * d)

    elif family == "beta":
        # U‑shaped Beta distribution (α,β < 1 give peaks at the ends)
        a_shape = rng.uniform(0.1, 2.0)
        b_shape = rng.uniform(0.1, 2.0)
        # normalised coordinate in (0,1)
        x = idx / max(1, n - 1)
        eps = 1e-12
        x = np.clip(x, eps, 1.0 - eps)
        a = (x ** (a_shape - 1)) * ((1 - x) ** (b_shape - 1))

    elif family == "dirichlet":
        # random Dirichlet draws (α_i ∈ (0,2) produces diverse shapes)
        alphas = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alphas, 1.0)

    elif family == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif family == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif family == "cosine":
        # raised cosine that is zero in the centre and maximal at the edges
        a = (1.0 - np.cos(np.pi * idx / (n - 1))) / 2.0

    else:
        a = rng.random(n)

    # optional tiny jitter for extra diversity
    if rng.random() < 0.25:
        a += rng.normal(scale=a.mean() * 0.02, size=n)
        a = np.clip(a, 0.0, None)

    return a

        S = np.sqrt(2.0 * n)
        cand = _project_to_simplex(raw, S)
        cur_max = np.max(_full_convolution_fft(cand))
        if cur_max < best_max:
            best_max = cur_max
            best_seq = cand.copy()
            print(f"[search] attempt={attempts} n={n} new best max={best_max:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam refinement (≈ 45 % of the remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None:
        adam_deadline = time.time() + (deadline - time.time()) * 0.45
        refined_a, refined_max = _adam_optimize(best_seq, adam_deadline, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_a.copy()
            print(f"[adam] improved max to {best_max:.6f}")

    # ------------------------------------------------------------------
    # 4) Exponentiated‑gradient refinement (remainder of the budget)
    # ------------------------------------------------------------------
    if best_seq is not None:
        eg_deadline = deadline
        refined_a, refined_max = _eg_optimize(best_seq, eg_deadline, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_a.copy()
            print(f"[eg] improved max to {best_max:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence found
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                     # should never happen
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_max:.6f}) length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
```
