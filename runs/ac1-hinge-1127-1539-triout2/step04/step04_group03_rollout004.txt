def _mirror_descent_opt(a0, deadline):

    a = a0.copy()

    n = len(a)

    target_sum = np.sqrt(2.0*n)

    beta = 1.0

    beta_max = 1e5

    step = 0

    # parameters
    eta0 = 0.5   (base learning rate?)

    while time.time() < deadline:
        # compute gradient using softmax with beta
        g, b = _gradient_softmax(a, beta)
        # exponentiated gradient update:
        eta = eta0 / np.sqrt(step+1) # decaying
        # avoid zeros; add epsilon
        a = a * np.exp(-eta * g)
        # renormalize sum to target_sum
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # schedule beta increase
        if step % 200 == 0 and beta < beta_max:
            beta = min(beta * 1.07, beta_max)

        step += 1

    return a

def _targeted_simulated_annealing(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)
    b = _full_convolution_fft(a)  # we might recalc each iteration anyway.
    # set temperature
    T = best_score * 0.02 + 1e-8
    cooling = 0.9997
    rng = np.random.default_rng()
    max_transfer = 0.3

    while time.time() < deadline:
        # compute current b and k_max
        b = _full_convolution_fft(a)
        max_idx = int(np.argmax(b))
        # compute weights for selecting i
        # only indices i where 0 <= max_idx - i < n
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        # i range [low, high]
        # compute w = a[i] * a[max_idx-i]
        # We'll compute vectorized:
        idx_range = np.arange(low, high + 1)
        w = a[idx_range] * a[max_idx - idx_range]
        if w.sum() <= 0:
            i = rng.integers(0, n)
        else:
            probs = w / w.sum()
            i = rng.choice(idx_range, p=probs)
        # pick j random
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        # move
        delta = a[i] * rng.random() * max_transfer
        if delta < 1e-15:
            continue
        a[i] -= delta
        a[j] += delta

        # compute new score
        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or (T > 0 and rng.random() < np.exp(-dE / T)):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasionally jitter global
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum()*0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            # normalize
            a_sum = np.sum(a)
            if a_sum > 0:
                a = a * (np.sqrt(2.0*n) / a_sum)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best

"""
Hybrid random search → mirror‑descent (exponentiated gradient) → 
targeted simulated annealing for minimizing the maximal entry of
the self‑convolution (the evaluated objective after normalising the
ℓ¹‑norm of the step‑function).

The algorithm works in three phases under a user‑supplied time budget:

1. **Diverse random seeding** – several families (Dirichlet,
   power‑law, exponential, symmetric edge‑heavy and Beta‑shaped)
   are projected onto the ℓ¹‑simplex Σa = √(2 n).  A short scan over a
   handful of lengths (2 k … 6 k) quickly discovers a promising
   candidate.

2. **Mirror‑descent optimisation** – a smoothed surrogate of
   max(b) (log‑sum‑exp of the convolution) is minimised using an
   exponentiated‑gradient (mirror descent) step that stays on the
   simplex automatically.  The smoothing parameter β is annealed to
   approximate the true ℓ∞ norm.

3. **Targeted simulated‑annealing** – the current best sequence is
   refined by repeatedly moving a random fraction of mass from an
   index that heavily contributes to the current maximal convolution
   entry to a random destination.  Uphill moves are accepted with a
   temperature schedule; occasional global jitter helps escape deep
   minima.

All helper functions are top‑level, no I/O is performed and the
search respects the supplied time budget (default 1000 s).  The
returned value is a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ------- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ------- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exp
    p = exps / np.sum(exps)

    # ------- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]                            # reversed a
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (exponentiated gradient) on the smoothed max
# ----------------------------------------------------------------------


def _mirror_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate.
    Keeps the iterate on the ℓ¹‑simplex automatically.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    beta_max = 1e5
    eta0 = 0.5                         # base learning rate
    step = 0
    eps = 1e-12

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(step + 1.0)
        a = a * np.exp(-eta * g)       # element‑wise
        a = np.maximum(a, eps)         # avoid exact zeros for the log

        # renormalise to the ℓ¹‑simplex
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # anneal β slowly
        if step % 200 == 0 and beta < beta_max:
            beta = min(beta * 1.07, beta_max)

        step += 1

    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    SA that repeatedly moves a fraction of mass from an index that
    heavily contributes to the current maximal convolution entry to a
    random destination.  Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()
    max_transfer = 0.25           # up to 25 % of the selected mass

    while time.time() < deadline:
        # current convolution and its maximal index
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that actually affect b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)                     # possible i
        contrib = a[idx_range] * a[k_max - idx_range]        # a_i * a_{k‑i}
        weight_sum = np.sum(contrib)

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick a destination j ≠ i uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of the mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = np.sum(a)
            if a_sum > 0:
                a = a * (np.sqrt(2.0 * n) / a_sum)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families are mixed to increase diversity,
    including a Beta‑shaped (U‑shaped or bell‑shaped) distribution.
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                 # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                           # Beta‑shaped
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    # small Gaussian perturbation and projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence`` under the supplied time budget.  The routine
    performs a quick multi‑length scan, refines the best candidate with
    mirror‑descent, and finishes with targeted simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Initialise from a global construction if it exists
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc = _max_convolution_fft(a0)
        best_seq, best_score = a0.copy(), sc
        print(f"[{time.time():.1f}] Start from global height_sequence_1 "
              f"(n={n0})  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – quick scan over several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000]
    scan_time_per_len = 8.0    # seconds dedicated to each length

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        end_phase = min(deadline, time.time() + scan_time_per_len)

        while time.time() < end_phase:
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New global best (scan) n={n}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was generated
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent refinement (half of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        md_deadline = time.time() + remaining * 0.5
        a_md = _mirror_descent_opt(best_seq, md_deadline)
        md_score = _max_convolution_fft(a_md)
        if md_score < best_score:
            best_seq, best_score = a_md.copy(), md_score
            print(f"[{time.time():.1f}] New best after mirror descent: score={md_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – targeted simulated‑annealing (use whatever time left)
        # ------------------------------------------------------------------
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        final_score = _max_convolution_fft(a_sa)
        if final_score < best_score:
            best_seq, best_score = a_sa.copy(), final_score
            print(f"[{time.time():.1f}] New best after SA: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up tiny trailing entries and return as plain list
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check with the official evaluator
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max‑conv) = "
          f"{best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
