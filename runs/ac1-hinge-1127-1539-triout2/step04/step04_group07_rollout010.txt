"""
A fast heuristic for the non‑negative step‑function problem.

Overview
--------
* Choose a length `n` (bias towards larger n – the lower bound of the
  objective is 2·n·max qₖ ≈ 1, so larger n gives more room for improvement).
* Initialise the probability vector with a *U‑shaped* Beta(α,α) density
  (α∈(0,1]) – this puts more mass near the ends and tends to flatten the
  additive convolution.
* Optimise the vector on the simplex Σa=√(2·n) using
  – a smooth‑max surrogate of the max‑convolution,
  – Adam‑style adaptive steps,
  – cheap random mass‑swaps,
  – a targeted move that moves mass from the largest contributing pair
    to the smallest one for the current worst convolution index.
* When progress stalls the algorithm restarts with a fresh β‑shape.
* The whole search runs until the 1000 s budget expires; the best
  sequence seen is finally evaluated with the supplied `evaluate_sequence`
  and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers (projection onto the ℓ1‑simplex)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective.
    Returns 2·(w * a) where `*` denotes the (full) correlation of w
    with a (i.e. convolution with the reversed weights).
    """
    n = a.shape[0]
    # correlation a ⋆ w  ==   conv(a, w_rev)
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # slice that corresponds to Σ_i w_{i+j}·a_i
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smooth approximation of max(conv)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Candidate generation (U‑shaped Beta densities)
# ----------------------------------------------------------------------


def _beta_initial(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a symmetric Beta(α,α) shaped probability vector of length n,
    normalised to Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    # barycentric points in (0,1)
    x = (np.arange(n) + 0.5) / n
    # pdf ∝ x^{α-1} (1-x)^{α-1}; add a tiny epsilon to avoid zeros.
    eps = 1e-12
    pdf = np.power(x * (1.0 - x) + eps, alpha - 1.0)
    pdf = pdf / pdf.sum() * S
    a = _project_simplex(pdf, S)
    return a


# ----------------------------------------------------------------------
# Main optimisation routine (Adam + swaps)
# ----------------------------------------------------------------------

    """
    Search for a sequence that minimises `evaluate_sequence`.
    Returns the best list of non‑negative heights found before the
    budget expires.
    """

    best_seq = None

    # ------------------------------------------------------------------
    length_candidates = [1800, 2500, 3000, 4000, 6000, 8000, 10000]

    for n in length_candidates:

        S = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:

    n = a.size
