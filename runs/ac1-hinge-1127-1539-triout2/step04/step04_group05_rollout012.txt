def _lp_iterative_optimize(f0, deadline, rng):
   f = f0.copy()
   n = len(f)
   S = np.sqrt(2*n)
   best_f = f.copy()
   best_M = _full_convolution_fft(f).max()
   t = 0.5  # mixing factor maybe.
   # loop while time < deadline
   while time.time() < deadline:
       # compute constraints
       M = _full_convolution_fft(f).max()
       # Build matrix A_ub: each column i: rows i..i+n-1 = f[i].
       # Use dense array for speed.
       A = np.zeros((2*n-1, n))
       for i in range(n):
           A[i:i+n, i] = f[i]
       b = np.full(2*n-1, M)
       # Solve LP
       # objective maximize sum(g) => minimize -sum(g)
       c = -np.ones(n)
       # bounds: (0,None)
       res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
       if not res.success:
           # fallback: break
           break
       g = res.x
       sum_g = g.sum()
       if sum_g <= 0:
           break
       scale = S / sum_g
       g_scaled = g * scale
       # mixing
       f_new = (1 - t) * f + t * g_scaled
       # project to simplex to maintain S exactly (though mixing ensures sum preserved? Let's compute:
       # sum(f_new) = (1-t)*S + t*S = S; so sum preserved, but numeric errors maybe.
       # Ensure nonnegativity
       f_new = np.maximum(f_new, 0.0)
       # proceed
       f = f_new
       # update best
       cur_M = _full_convolution_fft(f).max()
       if cur_M < best_M:
           best_M = cur_M
           best_f = f.copy()
           # maybe reduce t as we converge
           t = max(0.1, t * 0.95)  # gradually reduce mixing step
   return best_f, best_M

"""
Fast LP‑guided search for a step‑function with small max self‑convolution.

Algorithm
---------
1.  Initialise a feasible vector a (non‑negative, ℓ₁‑norm = √(2·n)).
2.  Repeatedly improve a by a Frank‑Wolfe‑type step:
        – compute M = max(a∗a)
        – solve the linear programme
              maximise   ∑ g_i
              subject to (a∗g)[k] ≤ M   for all k
                         g_i ≥ 0
        – scale the LP solution to the required ℓ₁‑norm and mix it into a
          (a ← (1‑t)·a + t·g_scaled).  The mixing factor t is reduced
          gradually.
   The LP drives the vector toward a shape where the convolution peaks are
   evenly spread, which typically reduces the evaluation value.
3.  The LP phase is run for a few seconds on several candidate lengths
    (800 … 2000).  The best result is then polished with the stochastic
    optimiser from the previous baseline (gradient, sub‑gradient,
    simulated‑annealing moves).
4.  The final best sequence is returned; it is stored in the global variable
    ``height_sequence_1`` for possible warm‑starts in later runs.

The code respects the 1000 s budget, uses only NumPy, SciPy and the
``evaluate_sequence`` black‑box, and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility: power of two (used by the FFT helpers of the old optimiser)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass S."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Linear scaling / interpolation of a sequence to a new length
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Build the (dense) convolution constraint matrix for the LP.
# For a given f, the matrix A satisfies   A @ g = f ∗ g .
# ----------------------------------------------------------------------
def _build_convolution_matrix_dense(f: np.ndarray) -> np.ndarray:
    n = len(f)
    rows = 2 * n - 1
    # broadcasting trick:
    idx = np.arange(rows)[:, None] - np.arange(n)[None, :]
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((rows, n), dtype=float)
    A[mask] = f[idx[mask]]
    return A


# ----------------------------------------------------------------------
# Build a sparse version of the same matrix (handy for larger n)
# ----------------------------------------------------------------------
def _build_convolution_matrix_sparse(f: np.ndarray) -> csr_matrix:
    n = len(f)
    # each column i appears in rows i … i+n‑1
    rows = np.arange(n)[:, None] + np.arange(n)  # shape (n,n)
    rows = rows.ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


# ----------------------------------------------------------------------
# LP‑guided Frank‑Wolfe step
# ----------------------------------------------------------------------
def _lp_iterative_optimize(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Starting from a0 (already on the simplex), repeatedly:
        – compute M = max(a∗a)
        – solve max ∑g_i  s.t. (a∗g) ≤ M, g ≥ 0
        – scale g to have the same ℓ₁‑norm as a0 and mix it back.
    Returns the best vector seen and its maximal self‑convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_M = _full_convolution_fft(a).max()

    # mixing factor – start relatively aggressive and decay
    t = 0.4
    it = 0
    while time.time() < deadline:
        # current worst convolution value
        M = _full_convolution_fft(a).max()

        # Build LP matrix; use dense for simplicity (n ≤ ~2000)
        A = _build_convolution_matrix_dense(a)   # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M)

        # maximise sum(g)  → minimise -sum(g)
        c = -np.ones(n)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            # If the LP fails we abort the LP phase for this length.
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Scale g to the required ℓ₁‑norm and mix back.
        scale = S / sum_g
        g_scaled = g * scale
        a = (1 - t) * a + t * g_scaled
        a = _project_to_simplex(a, S)   # numerical safety

        cur_M = _full_convolution_fft(a).max()
        if cur_M < best_M - 1e-9:
            best_M = cur_M
            best_a = a.copy()
            it = 0          # reset stagnation counter
        else:
            it += 1
            if it >= 8:    # stop if no improvement for several rounds
                break

        # decay mixing factor – we want finer adjustments later
        t = max(0.05, t * 0.93)

    return best_a, best_M

# --- Stochastic local optimiser (the one from the baseline) ----------
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float):
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau <= 1e-12:
        g = np.zeros_like(a)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)       # (2n‑1,)

        a_rev = a[::-1]
        conv_len = len(w) + n - 1
        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(w, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    b_new = _full_convolution_fft(a_proj)
    max_new = b_new.max()
    return a_proj, b_new, max_new


def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)

    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)            # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]               # length n
    return g, max_b


def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Hybrid optimiser (gradient, sub‑gradient, simulated‑annealing moves)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # basic hyper‑parameters (can be tweaked)
    eta = 0.7
    eta_decay = 0.9997
    move_prob = 0.13
    eps_move = 0.04
    temperature = cur_max * 0.12
    temp_decay = 0.9996
    jitter_prob = 0.015
    jitter_scale = 0.015
    use_softmax = True
    tau = cur_max
    tau_decay = 0.9995

    while time.time() < deadline:
        r = rng.random()

        if use_softmax and r < 0.55:
            a, b, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
            tau *= tau_decay

        elif r < 0.80:
            g, cur_max = _subgradient_max(a)
            improved = False
            for factor in (0.5, 1.0, 1.5):
                cand = a - eta * factor * g
                cand = _project_to_simplex(cand, S)
                max_cand = _full_convolution_fft(cand).max()
                if max_cand < cur_max - 1e-9:
                    a = cand
                    cur_max = max_cand
                    improved = True
                    break
            if not improved:
                cand = a - eta * g
                cand = _project_to_simplex(cand, S)
                max_cand = _full_convolution_fft(cand).max()
                if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                    a = cand
                    cur_max = max_cand
                temperature *= temp_decay

        elif r < 0.95:
            i = rng.integers(n)
            if a[i] > 0:
                j = rng.integers(n)
                if i != j:
                    delta = eps_move * a[i]
                    cand = a.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    max_cand = _full_convolution_fft(cand).max()
                    if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                        a = cand
                        cur_max = max_cand
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                        temperature *= temp_decay

        else:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            cand = np.clip(a + noise, 0.0, None)
            cand = _project_to_simplex(cand, S)
            max_cand = _full_convolution_fft(cand).max()
            if max_cand < cur_max:
                a = cand
                cur_max = max_cand

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        eta *= eta_decay

        if rng.random() < 0.01:
            use_softmax = not use_softmax

    return best_a, best_max

# Main entry point required by the harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Run a time‑limited search for a non‑negative step sequence that
    minimises the evaluation function.  The algorithm combines a linear‑program
    based Frank‑Wolfe step with the stochastic local optimiser from the
    baseline.  The best sequence found before the budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load the previous construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Candidate lengths (moderate sizes – large enough to improve the bound,
    # small enough to keep the LP tractable)
    # ------------------------------------------------------------------
    candidate_lengths = [800, 1000, 1200, 1400, 1600, 1800, 2000]

    for n in candidate_lengths:
        if time.time() >= end_time:
            break

        attempts += 1

        # ---- initialise a feasible vector ---------------------------
        if base_seq is not None and base_seq.size > 0:
            a0 = _scale_sequence(base_seq, n).astype(float)
            a0 = np.clip(a0, 0.0, None)
        else:
            a0 = rng.random(n)
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n))

        # ---- allocate time slice for this length --------------------
        remaining_lengths = len(candidate_lengths) - (attempts - 1)
        slice_time = min(45.0, (end_time - time.time()) / max(remaining_lengths, 1))
        slice_deadline = time.time() + slice_time

        # ---- LP‑guided iteration ------------------------------------
        a_lp, score_lp = _lp_iterative_optimize(a0, slice_deadline, rng)

        # ---- short polishing with the stochastic optimiser ----------
        if time.time() < slice_deadline - 1.0:
            a_lp, score_lp = _local_optimize(a_lp, slice_deadline, rng)

        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[candidate] length={n} attempt={attempts} new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing of the overall best sequence (time permitting)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time:
        a_final, score_final = _local_optimize(best_seq, end_time, rng)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[final polish] improved bound={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guarantees a non‑empty return
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = _project_to_simplex(rng.random(n), np.sqrt(2.0 * n))

    final_seq = best_seq.tolist()

    # ------------------------------------------------------------------
    # Evaluate through the supplied black‑box (for logging only)
    # ------------------------------------------------------------------
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal max = {best_score:.6f}) length = {len(final_seq)}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
