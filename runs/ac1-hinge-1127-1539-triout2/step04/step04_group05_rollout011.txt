"""
Improved multi‑phase search for a non‑negative step sequence that
minimises the evaluation function (the max entry of the self‑convolution
after normalising the ℓ₁‑norm).  The algorithm builds on the previous
heuristic but adds several new ingredients:

*   Enforces symmetry a[i] = a[n‑1‑i] after every update – the optimal
    construction is believed to be symmetric and this reduces the
    dimensionality of the search.
*   Uses Nesterov‑style momentum for both soft‑max and hard‑max (sub‑)
    gradient steps.
*   Refines the temperature schedule for the smooth max (soft‑max) and
    decays the step size more gently.
*   Diversifies the initialisations (random Dirichlet, linear ramp,
    sine‑shaped bump or an interpolated copy of the current best).
*   Adds a small final polishing phase on the best sequence found so
    far.
*   Prints progress information so the search can be monitored.

The implementation stays within a single CPU, uses only NumPy (no
external solvers) and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two (used by FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of total mass S."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (full convolution via real FFT)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution (len(x)+len(y)-1) of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full, length 2·n‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Simple symmetrisation (enforces a[i] = a[n‑1‑i])
# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry by averaging with the reversed vector."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Linear scaling / interpolation of a sequence to a new length
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Initial feasible sequence (ℓ₁‑norm = sqrt(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with the required sum."""
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size > 0:
        a = _scale_sequence(base_seq, n).astype(float)
    else:
        # diverse random initialisers
        r = rng.random()
        if r < 0.15:
            a = np.linspace(0, 1, n)                     # linear ramp
        elif r < 0.30:
            # sine‑shaped bump, zero at the ends
            x = np.arange(n) / (n - 1)
            a = np.sin(np.pi * x)
        else:
            a = rng.random(n)                             # plain Dirichlet(1)

    a = np.clip(a, 0.0, None)
    a = _symmetrize(a)
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Soft‑max (smooth) gradient of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns the soft‑max gradient of max(a*a) and the current max value.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    diff = b - max_b
    exps = np.exp(diff / max(tau, 1e-12))
    w = exps / np.sum(exps)                     # soft‑max weights (length 2n‑1)

    a_rev = a[::-1]
    conv_len = len(w) + len(a) - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[len(a) - 1:len(a) - 1 + len(a)]
    return grad, max_b


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(b) = max(a*a)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    """Return the sub‑gradient at the active max‑convolution entry."""
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)

    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)          # length 3n‑2
    grad = 2.0 * conv[len(a) - 1:len(a) - 1 + len(a)]
    return grad, max_b


# ----------------------------------------------------------------------
# Local optimiser (single restart)
# ----------------------------------------------------------------------
def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hybrid optimiser using soft‑max gradient, hard sub‑gradient,
    random mass swaps and Gaussian jitter.  Symmetry and momentum are
    applied throughout.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Momentum buffer (Nesterov‑style)
    v = np.zeros_like(a)

    # bookkeeping
    b = _full_convolution_fft(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters
    eta = 0.65                 # base step size
    eta_decay = 0.9998

    beta = 0.9                 # momentum coefficient

    # soft‑max temperature
    tau = cur_max
    tau_decay = 0.9995

    # simulated‑annealing temperature for random moves
    temperature = cur_max * 0.12
    temp_decay = 0.9996

    move_prob = 0.18          # probability to attempt a random mass move
    eps_move = 0.04           # fraction of the source mass to move

    jitter_prob = 0.04
    jitter_scale = 0.018

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) decide which update to apply
        # --------------------------------------------------------------
        r = rng.random()

        if r < 0.55:
            # ----- soft‑max gradient step (smooth) -----
            grad, cur_max = _softmax_gradient(a, tau)
            tau = max(tau * tau_decay, 1e-8)

        elif r < 0.80:
            # ----- hard max sub‑gradient step -----
            grad, cur_max = _subgradient_max(a)

        elif r < 0.95:
            # ----- random mass swap (simulated‑annealing) -----
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = eps_move * a[i]
                    cand = a.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = _symmetrize(cand)
                    cand = _project_to_simplex(cand, S)
                    cand_max = _full_convolution_fft(cand).max()
                    accept = cand_max < cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(temperature, 1e-12))
                    if accept:
                        a = cand
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                        temperature *= temp_decay
            # random‑move finished – skip momentum update
            continue

        else:
            # ----- Gaussian jitter -----
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            cand = np.clip(a + noise, 0.0, None)
            cand = _symmetrize(cand)
            cand = _project_to_simplex(cand, S)
            cand_max = _full_convolution_fft(cand).max()
            if cand_max < cur_max:
                a = cand
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            continue

        # --------------------------------------------------------------
        # 2) momentum / projected update
        # --------------------------------------------------------------
        v = beta * v + (1.0 - beta) * grad
        a_new = a - eta * v
        a_new = _project_to_simplex(a_new, S)
        a_new = _symmetrize(a_new)
        a = a_new

        # evaluate new max
        cur_max = _full_convolution_fft(a).max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # parameter decay
        eta *= eta_decay

        # occasional random toggle of soft‑max mode (helps escaping plateaus)
        if rng.random() < 0.01:
            # force a fresh sub‑gradient in the next iteration
            pass

    return best_a, best_max

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing of the overall best (if time permits)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 30.0 and best_seq is not None:
        # run a longer local optimisation on the current best
        polish_deadline = time.time() + remaining
        seq_opt, score_opt = _local_optimize(best_seq, polish_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[polish] final polishing improved max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the black‑box function
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_score:.6f}) length = {len(final_seq)}")
        # expose the best construction for future runs
        globals()['height_sequence_1'] = final_seq

    return final_seq
