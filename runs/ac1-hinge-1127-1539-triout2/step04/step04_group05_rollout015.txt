def _extended_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    # ensure feasible sum:
    a = _project_to_simplex(a, S)

    # initial parameters
    b = _full_convolution_fft(a)
    max_b = b.max()
    best_a = a.copy()
    best_max = max_b

    # Adam hyperparameters
    alpha = 0.1   # learning rate (scale to sum)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # temperature for softmax
    tau = max_b
    tau_decay = 0.9995
    # add jitter schedule
    jitter_prob = 0.02
    jitter_scale = 0.02

    while time.time() < deadline:
        # compute gradient of smooth max
        b = _full_convolution_fft(a)
        max_b = b.max()
        # update tau
        if tau > 1e-8:
            tau = max(tau * tau_decay, max_b * 0.08)  # ensure not too small relative
        # Compute softmax weights
        # Use scaling to avoid overflow:
        # subtract max_b to get exp((b - max_b)/tau)
        exps = np.exp((b - max_b) / tau)
        w = exps / exps.sum()
        # gradient g:
        conv = _fft_convolve(w, a[::-1])
        g = 2 * conv[n-1:n-1+n]

        # Adam update
        t_iter += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection
        a = _project_to_simplex(a, S)

        # occasional jitter
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=jitter_scale * a.mean(), size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

        # evaluate current max
        cur_max = _full_convolution_fft(a).max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # Occasionally do subgradient step for final refinement
        if rng.random() < 0.02:
            g_sub, max_sub = _subgradient_max(a)
            # try step size scaling
            for factor in (0.5, 1.0, 1.5):
                a_cand = a - step * factor * g_sub
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max:
                    a = a_cand
                    cur_max = max_cand
                    if cur_max < best_max:
                        best_max = cur_max
                        best_a = a.copy()
                    break

    return best_a, best_max

"""
Hybrid optimiser for the self‑convolution L∞ problem.

* Starts from a deterministic “arcsine’’ shape (a_i ∝ 1/√(i·(n‑i))) – this
  distribution is known to flatten the pair‑sum distribution and gives a
  good initial bound (often < 1.6 for moderate n).

* Refines the candidate with an Adam‑style projected gradient descent on a
  smooth approximation of the max (τ·log‑sum‑exp).  The temperature τ is
  slowly reduced (continuation) so the optimiser first sees a smooth
  landscape and then focuses on the true sup‑norm.

* Occasionally adds random jitter / mass‑swap moves (simulated‑annealing
  style) to escape shallow plateaus.

* Runs many independent restarts of varying length; each restart is given a
  time slice (≈30 s) and the best overall sequence is returned.  The final
  sequence is checked with the black‑box ``evaluate_sequence`` function.

The code is written with only NumPy (no external solvers) and respects the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=S}."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast for large vectors)."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence with linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a feasible non‑negative vector of length n (ℓ₁‑norm = √(2n))."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size > 0:
        a = _scale_sequence(base_seq, n).astype(float)
        a += rng.normal(scale=max(a.mean(), 1e-12) * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)          # Dirichlet(1)
    a = _project_to_simplex(a, S)
    return a


def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    """
    Sub‑gradient of max(a*a) together with the maximal entry.
    ``eps`` treats entries within eps of the maximum as active.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)
    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)          # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


def _max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a*a)_k."""
    return _full_convolution_fft(a).max()


def _arcsine_initial(n: int) -> np.ndarray:
    """
    Deterministic “arcsine’’ start: a_i ∝ 1/√((i+½)(n‑½‑i)).
    The vector is scaled to have ℓ₁‑norm = √(2n) (the scaling is irrelevant
    for the objective but helps with numerical stability).
    """
    idx = np.arange(n, dtype=float)
    i = idx + 0.5
    j = n - 0.5 - idx
    a = 1.0 / np.sqrt(i * j)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the τ‑soft‑max approximation of the L∞‑norm.
    Returns the best vector seen and its exact max self‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0

    # temperature schedule
    b = _full_convolution_fft(a)
    tau = b.max()                     # start from current max
    tau_decay = 0.9995

    best_a = a.copy()
    best_val = _max_convolution(a)

    while time.time() < deadline:
        # ----- smooth‑max gradient -----
        b = _full_convolution_fft(a)
        max_b = b.max()
        # keep τ from collapsing too quickly
        tau = max(tau * tau_decay, max_b * 0.08)

        # soft‑max weights (log‑sum‑exp trick for stability)
        exp_shift = np.exp((b - max_b) / tau)          # max_b subtracted to avoid overflow
        w = exp_shift / exp_shift.sum()

        # gradient ∂ (τ·log‑sum‑exp) / ∂a = 2 * (w ⋆ a_rev)
        conv = _fft_convolve(w, a[::-1])
        g = 2.0 * conv[n - 1:n - 1 + n]

        # ----- Adam update -----
        t_iter += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)

        step = 0.2 * m_hat / (np.sqrt(v_hat) + eps)   # learning‑rate ≈0.2

        a = a - step
        a = _project_to_simplex(a, S)

        # ----- occasional stochastic moves (helps escape plateaus) -----
        if rng.random() < 0.02:          # jitter
            noise = rng.normal(scale=0.02 * a.mean(), size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

        # ----- bookkeeping -----
        cur_val = _max_convolution(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # ----- optional hard sub‑gradient refinement -----
        if rng.random() < 0.01:
            g_sub, _ = _subgradient_max(a)
            # simple backtracking line‑search for the sub‑gradient direction
            step_sz = 0.5 * np.mean(np.abs(g_sub)) + 1e-6
            a_cand = a - step_sz * g_sub
            a_cand = _project_to_simplex(a_cand, S)
            cand_val = _max_convolution(a_cand)
            if cand_val < cur_val:
                a = a_cand
                cur_val = cand_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()

    return best_a, best_val

def propose_candidate(seed: int = 42,
