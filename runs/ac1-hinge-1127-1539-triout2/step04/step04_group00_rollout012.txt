def optimize_adam_subgrad(a, deadline, rng):
    n = len(a); S = np.sqrt(2*n)
    a = a.copy()
    best_a = a.copy(); best_val = np.max(_full_convolution(a))

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    beta = 0.5
    iter = 0

    while time.time() < deadline:
        iter += 1
        # compute conv and weights
        conv = _full_convolution(a)
        max_conv = conv.max()
        if max_conv < best_val:
            best_val = max_conv; best_a = a.copy()
        # adapt beta gradually:
        if iter % 100 == 0:
            beta = min(beta * 1.01, 50)

        # gradient weights:
        w = _smooth_weights(conv, beta)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**iter)
        v_hat = v / (1 - beta2**iter)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex:
        a = _project_simplex(a, S)

        # random mass swap occasionally
        if rng.random() < 0.02:
            i = rng.integers(n); j = rng.integers(n)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta; a[j] += delta
                a = _project_simplex(a, S)

        # subgradient step after some iterations?
        if iter % 2000 == 0:
            # compute subgrad using worst shift as before but with step size decaying
            k_max = int(np.argmax(conv))
            g = np.zeros_like(a)
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            idx_i = np.arange(i_low, i_high + 1)
            idx_j = k_max - idx_i
            g[idx_i] = a[idx_j]
            step = 0.1 / np.sqrt(iter)
            a = a - step * g
            a = _project_simplex(a, S)

    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) that minimizes the
self‑convolution max‑norm used by ``evaluate_sequence``.

Algorithm
---------
*   Work on the simplex Σa = √(2·n).  The objective
    ``2·n·max(conv)/sum(a)²`` reduces to ``max(conv)`` under this scaling.
*   For each trial we pick a length ``n`` (800 – 10 000) and initialise
    the vector either from the best known construction or from a random
    Dirichlet draw.
*   Three optimisation phases are applied to the same trial:
    1. **Flattening phase** – minimise the L² deviation of all
       convolution entries from their average value
       ``avg = Σa²/(2·n‑1) = 2·n/(2·n‑1)``.  Gradient descent with a tiny
       learning rate is used (the gradient is a simple convolution).
    2. **Smooth max‑phase** – minimise a soft‑max approximation of the
       max‑norm using Adam (the gradient is a weighted convolution with
       soft‑max weights).  The temperature is increased slowly so that
       the approximation becomes tighter.
    3. **Sub‑gradient phase** – a classic sub‑gradient descent that
       directly reduces the currently maximal diagonal of the
       self‑convolution.
    Each phase projects the vector back onto the simplex.
*   The best vector (smallest internal objective) seen over all trials
    is kept.  Random “mass‑swap’’ moves are performed occasionally to
   escape plateaus.
*   After the time budget (≈ 1000 s) the best vector is evaluated with
   ``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for a weighted sum of the convolution entries.
    Returns g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)  (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generator
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None, n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    If ``base_seq`` is supplied and long enough, its leading part (with tiny jitter)
    is used; otherwise a Dirichlet (uniform‑on‑simplex) sample is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # tiny jitter to escape exact repetitions
        jitter = rng.normal(scale=max(a.mean(), 1e-6) * 1e-3, size=n)
        a += jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)              # uniform on the simplex (Dirichlet(1))

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Phase 1 – flattening (L2 loss on the whole convolution)
# ----------------------------------------------------------------------
def _optimize_flatten(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the variance of the convolution entries by minimising
    Σ_k (conv_k - avg)².  This helps to obtain a nearly flat convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    target = (S * S) / (2 * n - 1)          # average convolution value

    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    lr = 0.12                               # small learning rate
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        err = conv - target                 # signed error vector (len 2n‑1)

        # gradient of Σ(err_k)²  => 4 * Σ err_{i+j} a_i
        grad = 4.0 * _conv_with_weights(a, err)

        a = a - lr * grad
        a = _project_simplex(a, S)

        # optional random “mass‑swap’’ to increase diversity
        if rng.random() < 0.015:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.025 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()

        # decay learning rate very gently
        lr *= 0.9999

        # early stop if no progress for a while
        if it % 3000 == 0 and best_val == cur_max:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – smooth max optimisation (Adam on soft‑max)
# ----------------------------------------------------------------------
def _optimize_smooth_adam(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a soft‑max approximation of the max‑norm.
    ``alpha`` (temperature) is increased slowly so that the surrogate
    becomes tighter as optimisation proceeds.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.22
    alpha = 0.6               # start rather smooth
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()

        # soft‑max weights – smooth approximation of the max
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)          # 2·Σ w_{i+j} a_i

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # occasional random perturbation (mass‑swap)
        if rng.random() < 0.025:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule tweaks
        lr *= 0.9997                      # very mild decay
        alpha = min(alpha * 1.008, 140.0) # sharpen the soft‑max

        # early stop if stagnant
        if it % 2500 == 0 and best_val == cur_max:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – sub‑gradient optimisation on the true max‑norm
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent that reduces the currently worst diagonal
    of the convolution.  Step size η_t = η₀/√t.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    eta0 = 0.18
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient g_j = a_{k_max-j} (zero when out of range)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()

        # occasional tiny random move (helps to escape plateaus)
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.022 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        if it % 8000 == 0 and best_val == cur_max:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Combined optimiser: flatten → smooth (Adam) → sub‑gradient
# ----------------------------------------------------------------------
def _optimize_combined_v2(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the three phases sequentially, dividing the remaining time
    roughly 20 % – 50 % – 30 %.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # Phase 1 – flattening (most cheap)
    t1 = now + 0.20 * remaining
    a1, v1 = _optimize_flatten(a0, t1, rng)

    # Phase 2 – smooth max with Adam
    now = time.time()
    t2 = now + 0.50 * (deadline - now)
    a2, v2 = _optimize_smooth_adam(a1, t2, rng)

    # Phase 3 – sub‑gradient refinement
    if time.time() < deadline:
        a3, v3 = _optimize_subgrad(a2, deadline, rng)
        # return the best of the three
        if v3 < v2 and v3 < v1:
            return a3, v3
    if v2 < v1:
        return a2, v2
    return a1, v1


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    The routine repeatedly starts a new trial (random length 800–10 000,
    random or warm start) and refines it with the three‑phase optimiser
    described above.  The best sequence found is finally evaluated and
    returned as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # safety margin for external eval

    # try to reuse the previous construction if it exists
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_val = np.inf
    attempts = 0
    last_improve = start_time

    while time.time() < deadline:
        attempts += 1

        # ------------------------------------------------------------------
        # Choose problem size and initialise a starting vector
        # ------------------------------------------------------------------
        if best_seq is None or rng.random() < 0.25:
            n = int(rng.integers(800, 10000))
            a0 = _initial_sequence(base_seq, n, rng)
        else:
            # continue refining the current best, with a tiny jitter
            a0 = best_seq.copy()
            n = a0.shape[0]
            jitter = rng.normal(scale=max(a0.mean(), 1e-6) * 1e-3, size=n)
            a0 += jitter
            a0 = np.clip(a0, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))

        # ------------------------------------------------------------------
        # Allocate a slice of the remaining budget for this attempt
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + 28.0, deadline)

        # ------------------------------------------------------------------
        # Optimise (flatten → smooth Adam → sub‑gradient)
        # ------------------------------------------------------------------
        a_opt, val = _optimize_combined_v2(a0, slice_deadline, rng)

        # ------------------------------------------------------------------
        # Record global best
        # ------------------------------------------------------------------
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best= {best_val:.6f}")

        # occasional hard restart if we have been idle for a while
        if time.time() - last_improve > 70.0:
            best_seq = None   # force a fresh random start next loop

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_val:.6f}) length = {len(final_seq)}")
    return final_seq
