"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  The objective is

    2·n·max(a*a) / (sum(a)^2) .

For a fixed length *n* the scaling of *a* does not matter, so we
normalise the ℓ¹‑norm to √(2·n).  In this normalisation the objective
equals the maximum entry of the linear convolution a*a.

The algorithm consists of four stages:

1. **Initialisation** – start from the globally supplied best sequence
   (``height_sequence_1``) if it exists, otherwise from a power‑law
   distribution.  A cheap grid search over a few lengths and power‑law
   exponents supplies a better starting point.

2. **Flattening gradient descent** – minimise the L² deviation of the
   convolution from its average value (which is the theoretical lower
   bound).  This smooth objective yields a simple analytic gradient that
   can be evaluated with two FFTs per iteration.  After each step we
   project back onto the simplex {x≥0, Σx=√(2·n)}.

3. **Simulated‑annealing refinement** – a classic SA with two move
   types (mass transfer and local smoothing) is applied to the result
   of step 2.  The temperature and the maximum transferred fraction are
   exponentially cooled.

4. **Optional length perturbation** – the current best shape is
   resampled to nearby lengths (±200) to see whether a slightly larger
   or smaller support gives a lower maximum convolution.

All steps keep strict non‑negativity and respect the time budget
(≈ 1000 s).  The best sequence found is returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _flatten_gradient(a: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k ( (a*a)_k - avg )²   w.r.t. a,
    where avg = (sum a)^2 / (2·n‑1).  The gradient can be written as
    4 * cross‑correlation( a, (a*a) - avg ).
    """
    n = a.size
    # full convolution a*a
    b = _conv_fft(a)                      # length 2·n‑1
    avg = (a.sum() ** 2) / (2 * n - 1)    # theoretical average of b
    r = b - avg                           # residual (length 2·n‑1)

    a_rev = a[::-1]
    size = 1 << ((len(r) + n - 1).bit_length())   # enough for linear conv
    fr = np.fft.rfft(r, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(fr * fa_rev, n=size)[: len(r) + n - 1]
    # grad_i = 4 * Σ_k r_k * a_{k-i}  → taken from the reversed prefix of conv
    grad = 4.0 * conv[n - 1 :: -1]        # n entries, reversed order
    return grad


def _resample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to a new length and ℓ¹‑normalisation."""
    old_n = a.size
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    if new_a.sum() == 0.0:
        # fallback to uniform (should never happen)
        new_a = np.full(new_n, target / new_n, dtype=np.float64)
    else:
        new_a = new_a * (target / new_a.sum())
    new_a = _project_to_simplex(new_a, target)
    return new_a


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.15,
                         max_frac_end: float = 0.004,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """
    Refine ``a0`` with classic simulated‑annealing.
    Moves:
        – donor → receiver mass transfer,
        – local smoothing (replace a short window by its average).
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = 2.0 * n * _max_convolution_fft(a) / (cur_sum ** 2)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial convolution (kept for the first iteration)
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # linear schedule for temperature and maximal transferred fraction
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current maximal convolution entry and its index
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity of each coordinate w.r.t. the active entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # ---- propose a move ----
        if rng.random() < 0.70:            # (1) donor → receiver
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_idx)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            # apply move
            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"

        else:                               # (2) local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # recompute convolution (single FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new / (cur_sum ** 2)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the rejected move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved_seg
                # conv stays unchanged

        # occasional global jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())   # simple ℓ¹‑renorm
            conv = _conv_fft(a)
            cur_score = 2.0 * n * conv.max() / (cur_sum ** 2)

        it += 1

    return best_a, best_score

# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Helper to evaluate the (normalised) objective (max convolution)
    # ------------------------------------------------------------------
    def obj_val(a: np.ndarray) -> float:
        # a is assumed to have ℓ¹‑norm = sqrt(2·len(a))
        return _max_convolution_fft(a)

    # ------------------------------------------------------------------
    # Phase 0 – initialise from the global best if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n0 = seq.size
        target = math.sqrt(2.0 * n0)
        seq = seq * (target / seq.sum()) if seq.sum() > 0 else seq
        seq = _project_to_simplex(seq, target)
        best_a = seq.copy()
        best_score = obj_val(best_a)
        best_n = n0
        print(f"[{time.time():.1f}] start from global seq, n={n0} score={best_score:.6f}")
    else:
        # simple power‑law fallback
        n0 = 1500
        target = math.sqrt(2.0 * n0)
        best_a = (np.arange(1, n0 + 1, dtype=np.float64) ** (-0.55))
        best_a = best_a * (target / best_a.sum())
        best_a = _project_to_simplex(best_a, target)
        best_score = obj_val(best_a)
        best_n = n0
        print(f"[{time.time():.1f}] start from power‑law, n={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – coarse grid search over lengths and power‑law exponents
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2500]   # moderate sizes, cheap to evaluate
    exponents = np.linspace(0.30, 0.90, 13)   # 13 values between 0.3 and 0.9

    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        for p in exponents:
            a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            a = a * (target / a.sum())
            a = _project_to_simplex(a, target)
            score = obj_val(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] grid n={n} p={p:.3f} score={score:.6f}")

        # a few random starts for the same length
        for _ in range(2):
            a = rng.random(n)
            a = a * (target / a.sum())
            a = _project_to_simplex(a, target)
            score = obj_val(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] random init n={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – flattening gradient descent (L2‑error of convolution)
    # ------------------------------------------------------------------
    it = 0
    step0 = best_a.mean() * 0.5 if best_a.mean() > 0 else 0.01
    while time.time() < deadline:
        it += 1
        grad = _flatten_gradient(best_a)                # O(n log n)
        # simple diminishing step size
        step = step0 / (1.0 + it * 0.01)
        new_a = best_a - step * grad
        new_a = np.maximum(new_a, 0.0)
        # renormalise to the simplex (sum = √(2·n))
        target = math.sqrt(2.0 * best_n)
        new_a = _project_to_simplex(new_a, target)

        new_score = obj_val(new_a)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a
            print(f"[{time.time():.1f}] flatten iter {it} improved score={best_score:.6f}")
        else:
            # occasional stochastic acceptance to escape plateaus
            if rng.random() < 0.001:
                best_a = new_a
                print(f"[{time.time():.1f}] flatten iter {it} random accept")

        # jitter every few hundred steps
        if it % 250 == 0:
            jitter = rng.normal(scale=target * 0.001, size=best_n)
            best_a = np.maximum(best_a + jitter, 0.0)
            best_a = _project_to_simplex(best_a, target)

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing refinement (core improvement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_seq, sa_score = _simulated_annealing(best_a, deadline,
                                                start_temp=0.02,
                                                end_temp=1e-6,
                                                max_frac_start=0.15,
                                                max_frac_end=0.004,
                                                max_window=30)
        if sa_score < best_score:
            best_a = sa_seq
            best_score = sa_score
            print(f"[{time.time():.1f}] after SA new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – explore nearby lengths by simple resampling
    # ------------------------------------------------------------------
    for delta in (-200, +200):
        if time.time() >= deadline:
            break
        new_n = best_n + delta
        if new_n < 50:
            continue
        candidate = _resample_sequence(best_a, new_n)
        candidate_score = obj_val(candidate)
        if candidate_score < best_score:
            best_a = candidate
            best_score = candidate_score
            best_n = new_n
            print(f"[{time.time():.1f}] resample n={new_n} improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for a compact representation
    # ------------------------------------------------------------------
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_a)}")
    return best_a.tolist()
