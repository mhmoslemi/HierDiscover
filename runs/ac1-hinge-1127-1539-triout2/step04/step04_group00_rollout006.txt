"""
Propose a non‑negative step function (vector) that minimises the
self‑convolution ∥a∗a∥∞.  The optimisation proceeds in three stages:

1. **Initialisation** – generate a reasonable starting shape (uniform or
   “U‑shaped”) and, when possible, upscale a previously discovered good
   vector by simple linear interpolation.

2. **Smooth optimisation** – minimise a smooth approximation of the
   max‑norm using an Adam optimiser on the soft‑max (log‑sum‑exp)
   surrogate.  After each update the vector is projected back onto the
   simplex Σa = √(2·n).  A small random mass‑swap is occasionally tried
   to escape plateaux.

3. **Sub‑gradient refinement** – directly decrease the currently maximal
   convolution entry by taking a sub‑gradient step for the true
   max‑norm, again with simplex projection.

The optimiser is run on a schedule of increasing problem sizes
(e.g. 800 → 1600 → 3200 → 6400) so that a good shape discovered for a
smaller length can be interpolated and refined for larger lengths.
The whole search respects the 1000 s budget; the best sequence found
is finally evaluated with the provided ``evaluate_sequence`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for a weighted sum of the convolution entries:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j) (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If ``base_seq`` is supplied and long enough, its leading part (with tiny jitter)
    is used; otherwise a random Dirichlet draw or a simple “U‑shape” is returned.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        # random decide between uniform and a “U‑shape” (high at the ends)
        if rng.random() < 0.5:
            a = rng.random(n)          # Dirichlet(1) – uniform on simplex.
        else:
            # simple U‑shape: weight inversely proportional to distance from edges
            idx = np.arange(n)
            a = 1.0 / (np.minimum(idx + 1, n - idx) + 1e-6)
            a = np.clip(a, 0.0, None)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


def _upscale_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly interpolate a vector to a new length and renormalise to
    Σa = √(2·new_n).  The shape of the original vector is preserved.
    """
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.clip(new_a, 0.0, None)
    S_new = np.sqrt(2.0 * new_n)
    new_a = new_a / new_a.sum() * S_new
    return new_a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑norm (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of the max‑norm.
    Returns the best vector and its (true) max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05  # modest learning rate
    t = 0

    # scheduling for the soft‑max temperature
    alpha = 0.5
    alpha_mul = 1.005

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # occasional random “mass‑swap” parameters
    swap_prob = 0.02
    swap_frac = 0.02

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        cur_score = np.max(conv)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # --- smooth surrogate gradient ---
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)               # ∂/∂a of Σ w_i conv_i

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a, S)

        # --- occasional random perturbation (mass‑swap) ---
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.1:
                    a = cand
                    conv = _full_convolution(a)   # recompute for next iteration
                    cur_score = cand_score

        # increase temperature (make the surrogate sharper)
        alpha = min(alpha * alpha_mul, 80.0)

        # very cheap early stop if nothing improves for a while
        if t % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (direct minimisation of the true max)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst entry)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.2                # initial step‐size constant
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation (helps escape tiny plateaus)
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (smooth → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase for roughly 60 % of the allotted time,
    then improve the result with the sub‑gradient phase.
    """
    now = time.time()
    if deadline - now <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * (deadline - now)
    a_s, score_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f

    return a_s, score_s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    The routine builds a hierarchy of problem sizes, refines each size
    with a smooth Adam phase followed by a sub‑gradient phase, and
    finally returns the best vector found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Prepare a list of target lengths.  We start modestly and grow
    # roughly by a factor 2, never exceeding a size that would be too
    # costly for the remaining budget.
    # ------------------------------------------------------------------
    max_len = 12000
    target_lengths = []
    n = 800
    while n <= max_len:
        target_lengths.append(n)
        n = int(n * 1.7)                # growth factor ≈ √3

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Main optimisation loop over the length schedule
    # ------------------------------------------------------------------
    for idx, n in enumerate(target_lengths):
        if time.time() >= deadline:
            break

        # allocate a fair slice of the remaining budget
        remaining = deadline - time.time()
        # leave a little time for future levels (including a final refine)
        levels_left = len(target_lengths) - idx + 1   # +1 for final fine‑tune
        slice_time = min(remaining * 0.85 / levels_left, max(30.0, remaining * 0.5))
        slice_deadline = time.time() + slice_time

        # ------------------------------------------------------------------
        # Initialise the vector for this size
        # ------------------------------------------------------------------
        if best_seq is None:
            a0 = _initial_sequence(None, n, rng)
        else:
            a0 = _upscale_sequence(best_seq, n)

        # ------------------------------------------------------------------
        # Optimise (smooth Adam → sub‑gradient)
        # ------------------------------------------------------------------
        attempts += 1
        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        # ------------------------------------------------------------------
        # Record global best
        # ------------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempts} n={n} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the current global best (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_final, score_final = _optimize_combined(best_seq, deadline, rng)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[final‑refine] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Convert to plain list and evaluate via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                     # fallback, should never occur
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
