def propose_candidate(seed=1234, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.1

    best_seq = None
    best_val = np.inf

    lengths = [2000, 3500, 5000, 7000, 9000, 12000]   # optionally larger

    for n in lengths:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)
        # Phase 1: random restarts + GD
        phase_end = min(deadline, time.time() + min(40.0, (deadline - time.time())/2))
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)
            sub_dead = min(deadline, time.time() + 5.0)  # short GD for quick screening
            a_gd = _gradient_descent_opt(a0, sub_dead)
            score = _score(a_gd)
            if score < best_val:
                best_val = score
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New candidate n={n} score={score:.6f}")

        # Phase 2: deep GD on current best of this length
        if best_seq is not None and best_seq.size == n:
            a_ref = _gradient_descent_opt(best_seq, min(deadline, time.time() + 120.0))
            sc = _score(a_ref)
            if sc < best_val:
                best_val = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined best n={n} score={sc:.6f}")

        # Phase 3: permutation SA
        if best_seq is not None and best_seq.size == n and time.time() < deadline - 5.0:
            a_perm = _permutation_annealing(best_seq, deadline)
            sc = _score(a_perm)
            if sc < best_val:
                best_val = sc
                best_seq = a_perm.copy()
                print(f"[{time.time():.1f}] Perm SA improved n={n} score={sc:.6f}")

    # final SA
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = _simulated_annealing(best_seq, deadline)
        sc = _score(a_final)
        if sc < best_val:
            best_val = sc
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final SA improved final score={sc:.6f}")

    # Trim zeros and return but also ensure sum close to sqrt(2n). Not required.
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[:nz[-1]+1]
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as e:
        print("Error evaluating:", e)

    return best_seq.tolist()

"""step_function_optimizer_v2

A search routine that improves upon the previous implementation by
* allocating more refinement time to the most promising length,
* using a higher inverse‑temperature schedule,
* adding a permutation‑based simulated‑annealing phase that only swaps
  entries (preserving the ℓ¹‑norm) to escape ordering‑related plateaus,
* keeping the same projected‑gradient descent core but with tighter
  parameters.

The code respects a wall‑clock budget (default 1000 s) and returns a
plain list of non‑negative heights which can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a computed by FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation: max(conv(a,a))."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) and projected gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)
    where p_k ∝ exp(β·b_k) and ⋆ denotes correlation.
    """
    n = a.shape[0]

    # ---- b = a*a ----------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # subtract max for stability
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1              # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smoothed maximum.
    Uses momentum, an adaptive step size and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0          # higher final temperature
    beta_factor = 1.4

    # base step size – a fraction of the ℓ¹‑norm
    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90                  # momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # tiny jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that transfers a random fraction of mass between two indices.
    Accepts uphill moves with probability exp(-Δ/T). Returns the best sequence.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    SA that repeatedly swaps two entries (keeping the ℓ¹‑norm intact).
    Helpful for breaking ordering plateaus that pure mass‑transfer cannot.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.05 + 1e-9
    cooling = 0.9998
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # swap
        a[i], a[j] = a[j], a[i]

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or rng.random() < np.exp(-dE / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] Perm SA improved: {best_score:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional random reshuffle to escape deep traps
        if rng.random() < 0.0005:
            rng.shuffle(a)
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] Perm SA reshuffle improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisation on the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the simplex Σ a = √(2 n).
    Several families (power‑law, exponential, mixed, Dirichlet,
    piecewise exponential) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # uniform plus tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the given time budget,
    performs several random restarts, a long projected‑gradient phase,
    a permutation‑based annealing phase and a final mass‑transfer SA.
    The best sequence found (as a plain Python list) is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # If a previous construction exists, seed with it
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_val = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a mix of modest and larger sizes
    # ------------------------------------------------------------------
    lengths = [2000, 3500, 5000, 7000, 9000, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Phase A – cheap random restarts with a short GD refinement
        # --------------------------------------------------------------
        phase_end = min(deadline,
                        time.time() + min(30.0, (deadline - time.time()) / 4.0))
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)

            # 2‑second GD quick‑launch
            sub_dead = min(deadline, time.time() + 2.0)
            a_gd = _gradient_descent_opt(a0, sub_dead)

            sc = _score(a_gd)
            if sc < best_val:
                best_val = sc
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New candidate (rand) n={n} score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase B – intensive GD on the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            sub_dead = min(deadline, time.time() + min(120.0, deadline - time.time()))
            a_ref = _gradient_descent_opt(best_seq, sub_dead)
            sc = _score(a_ref)
            if sc < best_val:
                best_val = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined best n={n} score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase C – permutation‑based simulated annealing
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n and time.time() < deadline - 5.0:
            a_perm = _permutation_annealing(best_seq, deadline)
            sc = _score(a_perm)
            if sc < best_val:
                best_val = sc
                best_seq = a_perm.copy()
                print(f"[{time.time():.1f}] Permutation SA improved n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final mass‑transfer SA with the remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = _simulated_annealing(best_seq, deadline)
        sc = _score(a_final)
        if sc < best_val:
            best_val = sc
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final SA improved final score={sc:.6f}")

    # ------------------------------------------------------------------
    # Strip trailing zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform sequence (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Log the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
