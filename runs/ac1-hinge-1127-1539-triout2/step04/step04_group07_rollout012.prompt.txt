# ----------------------------------------------------------------------
# Geometry helpers

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the rho index
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power‑of‑two for speed
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_j = 2 Σ_i w_{i+j}·a_i.
    a has length n, w has length 2·n‑1.

    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad

# Smooth‑max helpers

def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:

        w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)

def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:

    Generate a decreasing exponential sequence of length n.
    k_factor controls the decay (larger → faster decay).
    The result is projected onto the simplex Σa = √(2·n).

    # exact normalisation (avoids loss of precision for tiny λ)

    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    # final projection guarantees the exact sum
    a = _project_simplex(a, S)
    return a
