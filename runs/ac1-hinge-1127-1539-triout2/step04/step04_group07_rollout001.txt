"""
Propose a step‑function that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Warm–start** – generate an exponential decay sequence and project it onto
   the ℓ₁‑simplex Σa = √(2·n).  Several lengths (1200 … 3000) are tried.

2. **Iterative linear‑programming refinement** – given a candidate `f`,
   compute M = max(conv(f,f)).  Solve the LP

        maximise   Σ g_i
        subject to conv(f,g) ≤ M      (entrywise)
                    g_i ≥ 0
                    g_i ≤ 10·√(2·n)

   The solution `g` is scaled to the correct ℓ₁‑norm and mixed with `f`
   (convex combination) using a few step‑sizes; the best reduction of `M`
   is kept.  This step reduces the maximal auto‑convolution entry
   dramatically and converges after only a handful of iterations.

3. **Local fine‑tuning** – a projected‑gradient descent on a smooth
   approximation of the max‑norm (soft‑max) together with cheap “mass‑swap”
   moves further decreases the objective.  This stage runs for a short
   time slice (≈12 s) after the LP phase.

4. The best sequence among all lengths is returned.  The external
   `evaluate_sequence` function is called once at the end for reporting.

The whole routine respects the 1000 s budget and prints intermediate
scores.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Exponential warm‑start
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """
    Generate a decreasing exponential sequence of length n.
    The decay rate is controlled by k_factor (larger → faster decay).
    The result is projected onto the simplex Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    # exact normalisation (avoids loss of precision for tiny λ)
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)   # = 1 – qⁿ
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Linear‑programming helpers
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that (A @ g) == conv(f, g) (full linear convolution).
    Shape of A: (2·n‑1) × n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp(A: np.ndarray, M: float, ub: float) -> np.ndarray:
    """
    Solve:
        maximise   Σ g_i
        subject to A @ g ≤ M, 0 ≤ g ≤ ub
    Returns the optimal vector g as a dense NumPy array.
    """
    n = A.shape[1]
    g = cp.Variable(n)
    constraints = [A @ g <= M,
                   g >= 0,
                   g <= ub]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver='ECOS', warm_start=True, max_iters=5000, verbose=False)
    except (cp.error.SolverError, ValueError):
        prob.solve(solver='SCS', max_iters=20000, verbose=False)
    if g.value is not None:
        return np.asarray(g.value).flatten()
    # fallback – return a zero vector (should not happen)
    return np.zeros(n, dtype=np.float64)


def _iterative_lp(f0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  max_iters: int = 12) -> tuple[np.ndarray, float]:
    """
    Refine a candidate `f` by alternating linear‑programming steps and
    convex combinations.  Returns the improved sequence and its max‑conv
    value.
    """
    f = f0.copy()
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    best_M = np.max(_full_convolution(f))

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = best_M
        A = _build_conv_matrix(f)
        ub = 10.0 * S
        g = _solve_lp(A, M, ub)
        sum_g = np.sum(g)
        if sum_g < 1e-12:
            break
        g_scaled = g * (S / sum_g)

        improved = False
        # try a few step sizes, keep the best reduction of the max‑conv
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            f_new = (1.0 - t) * f + t * g_scaled
            f_new = _project_simplex(f_new, S)
            M_new = np.max(_full_convolution(f_new))
            if M_new < best_M - 1e-9:
                f = f_new
                best_M = M_new
                improved = True
                break
        if not improved:
            # small random jitter to escape possible plateaus
            jitter = rng.normal(scale=0.001 * S / n, size=n)
            f_new = f + jitter
            f_new = np.clip(f_new, 0.0, None)
            f_new = _project_simplex(f_new, S)
            f = f_new
            best_M = np.max(_full_convolution(f))

    return f, best_M

# Local optimiser (gradient + targeted swaps) – re‑used from the reference
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights for the smoothed max‑norm:
        w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smoothed max‑norm.
    Returns 2·Σ_i w_{i+j}·a_i for each j (size n).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the smoothed max‑norm.
    Returns the updated sequence and the exact maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using projected gradient descent on a smooth max
    and occasional mass‑swap moves.
    Returns the best sequence seen and its max‑conv value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (tuned empirically)
    lr = 0.5
    lr_decay = 0.9999
    alpha = 0.5
    alpha_mul = 1.005
    swap_prob = 0.15
    swap_frac = 0.04
    target_prob = 0.2
    eps_target = 0.03

    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # gradient step
        a, cur_max = _gradient_step(a, lr, alpha)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # random mass‑swap (simulated‑annealing flavour)
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # cheap targeted move aimed at the current maximal index
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            src_idx = np.argmax(contributions)
            i_src = i_range[src_idx]

            tgt_idx = np.argmin(contributions)
            i_tgt = i_range[tgt_idx]

            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # schedule updates
        lr *= lr_decay
        alpha *= alpha_mul

        if it - last_improve > 5000:
            break

    return best_a, best_score

        S = np.sqrt(2.0 * n)

        # ---- initialise candidate ------------------------------------
        if 'height_sequence_1' in globals():
            try:
                prev = np.asarray(height_sequence_1, dtype=float)
                if prev.size == n:
                    a0 = prev.copy()
                elif prev.size > n:
                    a0 = prev[:n].copy()
                else:
                    a0 = np.concatenate([prev, np.zeros(n - prev.size)])
                a0 = _project_simplex(a0, S)
            except Exception:
                a0 = _exponential_sequence(n, rng.uniform(0.6, 2.5))
        else:
            a0 = _exponential_sequence(n, rng.uniform(0.6, 2.5))

        # ---- LP refinement phase ------------------------------------
        slice_deadline = min(deadline, time.time() + 20.0)
        f_lp, score_lp = _iterative_lp(a0, slice_deadline, rng, max_iters=12)

        # ---- local gradient‑swap fine‑tuning -----------------------
        if time.time() < deadline:
            slice_deadline = min(deadline, time.time() + 12.0)
            f_refined, score_refined = _local_optimize(f_lp, slice_deadline, rng)
        else:
            f_refined, score_refined = f_lp, score_lp

        if score_refined < best_score:
            best_score = score_refined
            best_seq = f_refined.copy()
            print(f"[new best] n={n:4d} max‑conv={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) fallback if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
