import numpy as np, from scipy.optimize import linprog

def solve_lp(f, M, target_sum):
    n = f.size
    # Build A_ub
    # Use vectorized building.
    # Create zeros matrix A.
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # indices j = i_min ... i_max
        # compute f[k - j] for each j
        idx_range = np.arange(i_min, i_max+1)
        A[k, idx_range] = f[k - idx_range]
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimize negative sum => maximize sum
    # bounds
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        print("LP failed:", res.message)
        return None
    return res.x

"""
Step‑function optimizer – alternating linear‑programming + simple
projected‑gradient refinement.

The objective after ℓ¹‑normalisation (∑a = √(2 n)) is simply the
maximum entry of the self‑convolution a*a.  The algorithm works in two
phases:

1. **LP‑step** – with the current sequence f and its bound M = max(f*f) we
   solve a linear program that maximises the ℓ¹‑norm of a new sequence g
   under the linear constraints   (f∗g)[k] ≤ (1+ε)·M .  Because the
   constraints are linear in g this can be solved exactly with a LP
   solver (SciPy HiGHS).  The optimal g usually has a larger ℓ¹‑norm;
   after rescaling it to the required sum we obtain a candidate that can
   strictly reduce the bound after a convex combination with f.

2. **Mix & refine** – we search a few mixing coefficients t∈{0.05,…,0.25}
   and pick the one that gives the smallest new max‑convolution.
   Small Gaussian jitter and a short projected‑gradient descent on a
   smooth‑max surrogate are applied as polishing steps.

The loop runs while time permits (≤ 1000 s).  It starts from the
previous best construction stored in the global variable ``height_sequence_1``
(if present) and always returns a plain ``list[float]`` suitable for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the linear programme.
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (full) using FFT."""
    n = a.size
    m = b.size
    size = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: n + m - 1]


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_full_convolution(a, a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming helpers
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = f∗g.
    Column j of A contains f shifted by j positions.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _solve_lp(f: np.ndarray, M_bound: float) -> np.ndarray | None:
    """
    Maximise Σ g_i  subject to   g_i ≥ 0  and  (f∗g)[k] ≤ M_bound.
    Returns the optimal g (or None on failure).
    """
    if linprog is None:      # pragma: no cover
        print("SciPy linprog not available – skipping LP step")
        return None

    n = f.size
    A = _conv_matrix(f)                     # shape (2n‑1, n)
    b = np.full(2 * n - 1, M_bound, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # minimise –Σg_i  ⇒ maximise Σg_i
    bounds = [(0, None)] * n

    # HiGHS is fast and robust for moderate‑size dense LPs.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        print(f"LP failure: {res.message}")
        return None
    return res.x

# ----------------------------------------------------------------------


def _smooth_max_gradient(a: np.ndarray, beta: float):
    """
    Gradient of g(a)= (1/β)·log Σ exp(β·b)  where b = a∗a .
    Returns (grad, b) where grad = 2·(a ⋆ p),  p_k ∝ exp(β·b_k).
    """
    n = a.size
    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # correlation a ⋆ p = (a_rev * p) evaluated at indices n‑1 … n‑1+n‑1
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    ar = np.fft.rfft(a_rev, n=size2)
    pr = np.fft.rfft(p, n=size2)
    corr = np.fft.irfft(ar * pr, n=size2)[:conv_len]
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad, b


def _proj_grad_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short projected‑gradient descent on the smooth‑max approximation.
    Stops when the time budget expires.
    """
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)

    beta = 1.0
    beta_max = 4096.0
    beta_factor = 1.4
    step_base = target * 0.20
    momentum = np.zeros_like(a)
    rho = 0.90

    it = 0
    while time.time() < deadline:
        g, _ = _smooth_max_gradient(a, beta)
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # Euclidean projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target)

        # anneal β every few iterations
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
        it += 1
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine starts from the previous best construction (if any),
    performs several alternating “LP‑mix” iterations and finishes with a
    short projected‑gradient refinement.  The returned list is a plain
    Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the previously found sequence if available.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a = np.asarray(earlier, dtype=float)
    else:
        # fallback: uniform Dirichlet on a new length
        n = 2000
        target = math.sqrt(2.0 * n)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    n = a.size
    target_sum = math.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    conv_a = _full_convolution(a, a)
    cur_obj = float(conv_a.max())
    best_seq = a.copy()
    best_obj = cur_obj
    print(f"[{time.time():.1f}] init  n={n}  max‑conv={cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Main alternating LP + mixing loop
    # ------------------------------------------------------------------
    max_iters = 30
    slack = 1e-4                       # tiny relaxation of the bound
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = best_obj * (1.0 + slack)   # relaxed bound for the LP

        g = _solve_lp(best_seq, M)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= target_sum * 0.999:          # not enough mass → stop
            print(f"[{time.time():.1f}] iter {it}: LP sum too small ({sum_g:.3f})")
            break

        # rescale g to the required ℓ¹‑norm
        scale = target_sum / sum_g
        g_scaled = g * scale

        # pre‑compute the three convolutions needed for mixing
        conv_g = _full_convolution(g_scaled, g_scaled)
        conv_fg = _full_convolution(best_seq, g_scaled)

        # try a handful of mixing parameters and keep the best improvement
        best_t = 0.0
        best_new_obj = best_obj
        best_conv = None
        for t in (0.05, 0.10, 0.15, 0.20, 0.25):
            coeff_self = (1.0 - t) ** 2
            coeff_cross = 2.0 * t * (1.0 - t)
            coeff_g = t ** 2
            conv_new = (coeff_self * conv_a
                        + coeff_cross * conv_fg
                        + coeff_g * conv_g)
            new_obj = float(conv_new.max())
            if new_obj < best_new_obj - 1e-9:
                best_new_obj = new_obj
                best_t = t
                best_conv = conv_new

        if best_t == 0.0:
            print(f"[{time.time():.1f}] iter {it}: no improving mix found")
            break

        # update the current sequence and its convolution
        best_seq = (1.0 - best_t) * best_seq + best_t * g_scaled
        conv_a = best_conv
        best_obj = best_new_obj
        print(f"[{time.time():.1f}] iter {it}: t={best_t:.3f}  max‑conv={best_obj:.6f}")

        # occasional tiny jitter to escape plateaus
        if it % 5 == 0 and time.time() < deadline:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            best_seq += jitter
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = _project_to_simplex(best_seq, target_sum)
            conv_a = _full_convolution(best_seq, best_seq)
            best_obj = float(conv_a.max())

    # ------------------------------------------------------------------
    # Final projected‑gradient refinement (uses the same smooth‑max surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_dead = deadline
        refined = _proj_grad_descent(best_seq, sub_dead)
        conv_ref = _full_convolution(refined, refined)
        obj_ref = float(conv_ref.max())
        if obj_ref < best_obj:
            best_seq = refined
            best_obj = obj_ref
            print(f"[{time.time():.1f}] final GD improved to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print("evaluate_sequence error:", exc)

    return best_seq.tolist()
