"""
A fast multi‑phase optimizer for the step‑function problem.

1️⃣  Systematic grid search over a few well‑known parametric families
    (exponential, symmetric‑exponential, power‑law and Gaussian) for a
    range of lengths.  For each candidate the exact self‑convolution
    maximum is computed via an FFT and the best vector is retained.

2️⃣  A broad random search that creates vectors from many parametric
    families (including a Dirichlet draw) with random lengths.  This
    phase explores far more diverse shapes.

3️⃣  The best candidate is refined with a projected Adam optimiser that
    minimises a soft‑max approximation of the ∞‑norm of the convolution.
    After each update the vector is projected onto the ℓ₁‑simplex
    (∑a = √(2·n), a ≥ 0).

4️⃣  A tiny local‑search phase (mass transfers between two coordinates)
    attempts to squeeze out the last improvement.

The whole routine respects the given time budget (default 1000 s) and
stores the final sequence in the global variable `height_sequence_1`
so that subsequent runs can start from it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the ℓ₁‑simplex of sum S."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Compute gradient of τ·log∑exp(b/τ) where b = a*a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard max ⇒ sub‑gradient
        k = np.argmax(b)
        # gradient of b_k = Σ_i a_i a_{k-i}
        # ∂b_k/∂a_i = a_{k-i} + a_{k-i} (if i≠k-i) else 2*a_i
        g = np.zeros_like(a)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += a[j]
                if i != j:
                    g[i] += a[j]   # count the symmetric term
        return g, b, max_b

    # soft‑max weights w_k ∝ exp(b_k / τ) (stable version)
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)          # shape (2n‑1,)

    # convolution of reversed a with w gives ∂/∂a of Σ_k w_k b_k
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # length n

    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Minimise the soft‑max approximation of max(a*a) using Adam.
    Returns the best vector found and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6                       # initial learning rate
    t = 0

    # initialise temperature for soft‑max
    _, _, cur_max = _gradient_softmax(a, tau=0.0)   # hard max just to get a scale
    tau = max(cur_max * 0.5, 1e-8)

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        # ----- gradient step -------------------------------------------------
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # ----- evaluation ----------------------------------------------------
        cur_max_new = np.max(_full_convolution_fft(a_new))

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # ----- schedules -----------------------------------------------------
        tau = max(tau * 0.995, 1e-8)   # a slightly faster cooling
        lr *= 0.9999                  # very gradual decay

        # occasional jitter to escape tiny plateaus
        if rng.random() < 0.01:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families (including a Dirichlet draw)
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create an un‑normalised non‑negative vector using a random family."""
    typ = rng.choice(
        [
            "uniform",
            "exp",
            "symexp",
            "gaussian",
            "triangular",
            "power",
            "cosine",
            "dirichlet",
        ]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "dirichlet":
        # sample Dirichlet via Gamma draws, then normalise
        alpha = rng.uniform(0.2, 5.0, size=n)
        a = rng.gamma(alpha)
        a = np.clip(a, 0.0, None)

    else:
        a = rng.random(n)

    # tiny random jitter (helps diversity)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a
