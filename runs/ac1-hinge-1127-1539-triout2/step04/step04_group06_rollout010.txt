def subgradient(a, conv, top_k=5):
    # conv full length 2n-1
    idx = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    n = a.size
    for k in idx:
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        i_range = np.arange(lo, hi+1)
        grad[i_range] += a[k - i_range]
    grad *= 2.0
    return grad

def propose_candidate(seed=0, budget_s=1000):

    start_time = time.time()

    best_seq = None
    best_score = np.inf

    # define list of n values
    n_list = [1000, 1500, 2000]  # maybe add 2500 depending on time

    for n in n_list:
        target_sum = np.sqrt(2.0 * n)
        # number of restarts per n
        restarts = 3
        for r in range(restarts):
            # random exponent
            p = rng.uniform(0.4, 0.8)
            a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
            # add small noise
            a += rng.normal(scale=0.01 * a.mean(), size=n)
            # clip nonneg
            a = np.maximum(a, 0.0)
            # normalize to target sum
            sum_a = a.sum()
            if sum_a == 0:

            else:
                a = a * (target_sum / sum_a)
            # start gradient descent stage
            # compute initial max
            cur_max = _max_convolution_fft(a)
            cur_score = cur_max
            # subgradient descent for limited time e.g., 5 seconds

"""
Multi‑restart gradient + simulated‑annealing optimizer for the max‑convolution
benchmark.

* Random power‑law initialisations (exponent ∈ [0.4, 0.8]) of various
  lengths.
* A short projected sub‑gradient descent that directly reduces the
  active convolution peak (gradient of the max‑convolution).
* An enhanced simulated‑annealing phase with three move types
  (mass transfer, swap, local smoothing).  Temperature starts at 0.5 and
  ends at 1e‑6; the maximal transferred fraction shrinks from 0.25 to 0.01.
* The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# convolution helpers
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    """
    Compute a (sub‑)gradient of max_k sum_i a_i a_{k-i}.
    We sum the gradients of the `top_k` largest convolution entries.
    """
    n = a.size
    if top_k >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]

    grad = np.zeros_like(a)
    for k in idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += a[k - i_range]          # ∂/∂a_i a_i * a_{k-i}
    grad *= 2.0                                    # both (i,k-i) and (k-i,i)
    return grad


def _descent_step(a: np.ndarray,
                  grad: np.ndarray,
                  cur_max: float,
                  target_sum: float) -> tuple[np.ndarray, float]:
    """
    One back‑tracking projected sub‑gradient step.
    Returns the possibly updated array and its new max‑convolution value.
    """
    norm_g = np.linalg.norm(grad)
    if norm_g == 0.0:
        return a, cur_max

    # initial step size – proportional to current peak, inverse to gradient norm
    step = cur_max / (norm_g + 1e-12) * 0.5
    for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
        step_eff = step * factor
        a_new = a - step_eff * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new = a_new * (target_sum / s)          # re‑normalise ℓ¹‑norm
        new_max = _max_convolution_fft(a_new)
        if new_max < cur_max - 1e-12:              # accept any strict improvement
            return a_new, new_max
    # no improvement – return unchanged
    return a, cur_max


# ----------------------------------------------------------------------
# simulated annealing with several move types
# ----------------------------------------------------------------------
def _simulated_annealing2(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine `a0` using simulated annealing.
    Moves:
        * donor → receiver mass transfer (weighted by current peak sensitivity)
        * swap two entries
        * replace a short window by its average (local smoothing)
    Temperature decays from 0.5 to 1e‑6; max transfer fraction from 0.25 to 0.01.
    """
    a = a0.copy()
    n = a.size
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_max

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial convolution (used for sensitivities)
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time                       # progress ∈ [0,1]
        temp = 0.5 * (1e-6 / 0.5) ** tau                 # exponential schedule
        max_frac = 0.25 * (0.01 / 0.25) ** tau

        # ------------------------------------------------------------------
        # data for current state
        # ------------------------------------------------------------------
        k_max = int(np.argmax(conv))

        # sensitivity for the active shift
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens                                 # contribution to current peak

        # ------------------------------------------------------------------
        # propose a move
        # ------------------------------------------------------------------
        move_type = None
        saved = None                                     # for revert

        if rng.random() < 0.6:                           # mass transfer (most common)
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_idx)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"
                    saved = (donor, receiver, delta)

        elif rng.random() < 0.8:                         # swap two positions
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                a[i], a[j] = a[j], a[i]
                move_type = "swap"
                saved = (i, j)

        else:                                            # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, 20)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved = (start, end, old_seg)

        # ------------------------------------------------------------------
        # evaluate new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_max = conv_new.max()

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            conv = conv_new
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                donor, receiver, delta = saved
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "swap":
                i, j = saved
                a[i], a[j] = a[j], a[i]
            elif move_type == "smooth":
                start, end, old_seg = saved
                a[start:end] = old_seg
            # conv remains unchanged, cur_max unchanged

        # occasional jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a * (cur_max ** 0.5 * np.sqrt(2 * n) / s)   # keep sum ≈ sqrt(2n)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved best = {best_score:.6f}")

        it += 1

    return best_a, best_score

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin for the harness

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1️⃣  If a previously discovered construction exists, use it as a seed.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n0 = seq.size
        target = np.sqrt(2.0 * n0)
        if seq.sum() > 0:
            seq = seq * (target / seq.sum())
        else:
            seq = np.full(n0, target / n0)
        cur = _max_convolution_fft(seq)
        best_seq = seq.copy()
        best_score = cur
        print(f"[{time.time():.1f}] start from global seq, length={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Multi‑restart optimisation on several lengths.
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1500, 1800]
    for n in lengths:
        if time.time() >= deadline - 20.0:
            break
        target_sum = np.sqrt(2.0 * n)

        # a few random restarts per length
        for restart in range(4):
            if time.time() >= deadline - 10.0:
                break

            # ----- initialise a random power‑law -------------------------
            p = rng.uniform(0.4, 0.8)
            a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            a += rng.normal(scale=0.01 * a.mean(), size=n)   # tiny noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a * (target_sum / s)
            else:
                a = np.full(n, target_sum / n)

            # blend with the global construction when lengths match (optional)
            if best_seq is not None and len(best_seq) == n:
                a = 0.5 * a + 0.5 * best_seq
                a = a * (target_sum / a.sum())

            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] init n={n} p={p:.3f} new best={best_score:.6f}")

            # ----- short sub‑gradient descent ---------------------------
            sub_deadline = time.time() + min(8.0, deadline - time.time() - 5.0)
            while time.time() < sub_deadline:
                conv = _conv_fft(a)
                grad = _subgradient(a, conv, top_k=5)
                a, cur_score = _descent_step(a, grad, cur_score, target_sum)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] grad descent improved to {best_score:.6f}")

            # ----- simulated annealing refinement ------------------------
            if time.time() < deadline - 3.0:
                sa_seq, sa_score = _simulated_annealing2(a, deadline)
                if sa_score < best_score:
                    best_score = sa_score
                    best_seq = sa_seq.copy()
                    print(f"[{time.time():.1f}] SA refined best={best_score:.6f}")

    # ------------------------------------------------------------------
    # final housekeeping – drop trailing zeros for a tidy output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of length 1500
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
