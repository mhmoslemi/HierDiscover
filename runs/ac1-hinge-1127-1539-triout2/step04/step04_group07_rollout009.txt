    A = np.zeros((r, n))
    for idx, k in enumerate(rows):
        i_start = max(0, k - (n - 1))

        i_range = np.arange(i_start, i_end + 1)

"""
propose_candidate – search for a non‑negative step function with minimal
maximal auto‑convolution (the value returned by evaluate_sequence).

Algorithm
---------
1.  Warm‑start with an exponential decay sequence of a chosen length.
    The sequence is projected onto the ℓ1‑simplex Σa = √(2·n).

2.  Iteratively improve the candidate by solving a small linear programme.
    For the current sequence f we compute M = max (f∗f).  Any sequence g
    satisfying the linear constraints   f·g ≤ M   (convolution written
    as a linear operator in g) can be mixed with f while keeping the
    maximal convolution bounded.  We therefore
        * collect the indices where (f∗f) is within a tiny tolerance of M
          (plus the central index, which guarantees a full‑rank constraint);
        * build the corresponding linear constraints;
        * maximise Σ g_j subject to g ≥ 0 and those constraints (LP);
        * rescale g so that Σ g = √(2·n) and replace
          f ← (1‑t)·f + t·g   with a decreasing step size t.
    The LP is tiny because only the “active’’ rows of the convolution
    matrix are used; it solves in a few milliseconds.

3.  Occasionally add a small random jitter and enforce symmetry
    (a ← (a + a[::-1])/2) – this drives the search away from poor local
    minima and respects the natural symmetry of the problem.

4.  The whole search runs until the time budget (≈ 1000 s) is exhausted,
    keeping the best sequence seen.  The final sequence is returned as a
    plain list of floats; its quality is finally reported by
    evaluate_sequence().
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Warm‑start: exponential decay sequences
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """Decreasing exponential sequence of length n, projected onto the ℓ1‑simplex."""
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    # normalisation constant so that Σ a_i = S
    c = S * (1 - q) / (1 - q ** n)
    a = c * q ** np.arange(n, dtype=float)
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Build the (few) active rows of the convolution matrix
# ----------------------------------------------------------------------


def _build_active_matrix(a: np.ndarray, rows: np.ndarray) -> np.ndarray:
    """
    For a given vector a (length n) construct the matrix A where
        (A @ g)[k] = (a * g)[k]   for every k in `rows`.
    Only the requested rows are returned (shape len(rows) × n).
    """
    n = a.shape[0]
    r = rows.size
    A = np.zeros((r, n), dtype=float)
    for idx, k in enumerate(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        A[idx, k - i_range] = a[i_range]
    return A


# ----------------------------------------------------------------------
# One LP‑based improvement step
# ----------------------------------------------------------------------


def _lp_step(a: np.ndarray, t: float) -> tuple[np.ndarray, float]:
    """
    Perform a single LP‑based update.
    Returns the updated sequence and its maximal convolution entry.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    conv = _full_convolution(a)
    M = conv.max()

    # tolerance for the active set (relative)
    eps = max(1e-12, 1e-6 * M)
    active = np.where(conv >= M - eps)[0]

    # guarantee a full‑rank row (the central convolution entry)
    centre = n - 1
    if centre not in active:
        active = np.append(active, centre)

    # build the active constraint matrix
    A = _build_active_matrix(a, active)

    # linear programme: maximise Σ g_j  ⇔ minimise –Σ g_j
    c = -np.ones(n)
    b_ub = M * np.ones(A.shape[0])
    bounds = [(0.0, None)] * n

    # solve the LP (highs is fast and reliable)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")

    # if the LP fails for any reason, fall back to the unchanged vector
    if not res.success:
        return a, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return a, M

    # rescale to the required ℓ1‑norm
    g = g * (S / sum_g)

    # convex combination with a small step size
    a_new = (1.0 - t) * a + t * g

    # enforce symmetry (helps the objective) and project back onto the simplex
    a_new = 0.5 * (a_new + a_new[::-1])
    a_new = _project_simplex(a_new, S)

    new_M = _full_convolution(a_new).max()
    # accept the move only if it improves the objective
    if new_M < M:
        return a_new, new_M
    else:
        return a, M


# ----------------------------------------------------------------------
# Local optimiser (many LP steps + occasional jitter)
# ----------------------------------------------------------------------


def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Iterate LP‑steps until the time budget expires."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _full_convolution(a).max()

    t = 0.5                     # initial mixing weight
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        a, cur_max = _lp_step(a, t)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # slowly decay the step size
        t = max(0.05, t * 0.985)

        # occasional random jitter to escape tiny plateaus
        if rng.random() < 0.07:
            noise = rng.normal(scale=0.001 * a.mean(), size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_simplex(a, S)

        # early stop if no improvement for a while
        if it - last_improve > 2000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a sequence that minimises evaluate_sequence().
    The routine explores several lengths, warm‑starts each with a random
    exponential decay and then refines the candidate using the LP‑based
    scheme described above.  The best sequence found before the time budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) try a few lengths with a cheap exponential warm‑start
    # ------------------------------------------------------------------
    candidate_lengths = [800, 1200, 1500, 2000, 2500, 3000, 3500]
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        for _ in range(3):   # a few random decay rates per length
            k = rng.uniform(0.5, 3.0)
            a0 = _exponential_sequence(n, k)
            cur_score = _full_convolution(a0).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a0.copy()
                print(f"[warm‑start] n={n:4d} k={k:4.2f} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2) incorporate the globally‑available previous construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                n = prev.size
                S = np.sqrt(2.0 * n)
                prev = _project_simplex(prev, S)
                cur = _full_convolution(prev).max()
                if cur < best_score:
                    best_score = cur
                    best_seq = prev.copy()
                    print(f"[prev] re‑using height_sequence_1, score={cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) main optimisation phase – run until the budget runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a length; favour previously successful lengths
        if best_seq is not None and rng.random() < 0.6:
            n = best_seq.size
        else:
            n = rng.integers(800, 4001)

        # Initialise either from the current best (jittered) or a fresh exponential
        if best_seq is not None and best_seq.size == n and rng.random() < 0.5:
            a0 = best_seq.copy()
            jitter = rng.normal(scale=0.001 * a0.mean(), size=n)
            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))
        else:
            k = rng.uniform(0.5, 3.0)
            a0 = _exponential_sequence(n, k)

        slice_deadline = min(deadline, time.time() + 30.0)   # 30 s per slice
        a_opt, score_opt = _local_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global] new best n={n:4d} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final sanity check via the external black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of length 1000
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
