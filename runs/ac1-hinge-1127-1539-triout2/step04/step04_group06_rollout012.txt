def refine_via_lp(a, t, max_iter, deadline):
    n = len(a)
    target_sum = np.sqrt(2*n)
    # Precompute row_idx, col_idx once
    row_idx = np.concatenate([...])   # size n*n
    col_idx = np.repeat(np.arange(n), n)
    shape = (2*n -1, n)

    best_a = a.copy()
    best_M = max_convolution_fft(a)

    for iter_i in range(max_iter):
        if time.time() >= deadline:
            break
        M = max_convolution_fft(a)   # compute current max conv
        # Build convex matrix via data = np.tile(a, n)
        data = np.tile(a, n)
        A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=shape)
        # Solve LP
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # fallback to break
            break
        g = res.x
        # Scale g to target sum; sum_g = g.sum()
        if g.sum() == 0:
            break
        g_scaled = g * (target_sum / g.sum())
        # Combine
        a = (1 - t) * a + t * g_scaled
        # Compute new max convolve
        new_M = max_convolution_fft(a)
        if new_M < best_M:
            best_M = new_M
            best_a = a.copy()
        # optionally increase t with iteration?
    return best_a, best_M

    conv = _conv_fft(a)
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        # exponential schedule
        temp = 0.02 * (1e-5/0.02) ** tau
        max_frac = 0.20 * (0.01/0.20) ** tau

        # compute weight
        max_b = conv.max()
        k_max = int(np.argmax(conv))
        # compute sens and weight like earlier
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        # propose move
        if rng.random() < 0.7:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight/total_w)
            else:
                donor = rng.integers(n)
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_idx)
            if donor == receiver:
                continue
            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue
            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:
            # smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, 21) # max window 20
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            a[start:end] = old_seg.mean()
            move_type = "smooth"
            saved_seg = old_seg
            saved_start = start
            saved_end = end

        # evaluate new state via FFT
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[:2*n-1]
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new / (cur_sum ** 2)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved_seg
        it += 1
    return best_a, best_score

"""
Hybrid LP–SA optimizer for the “max‑convolution / ℓ¹‑norm’’ benchmark.

The objective after normalising the ℓ¹‑norm to √(2·n) is simply the
maximum entry of the linear convolution a*a.  The algorithm works in two
phases:

1.  Linear‑programming refinement.
    For a fixed sequence f we maximise Σ g_i under the linear constraints
    (f*g)_k ≤ M where M = max(f*f).  The optimiser chooses g that pushes
    mass towards indices where f is small.  By blending f with a suitably
    normalised copy of g the peak M tends to decrease.  The LP is solved
    with SciPy’s HiGHS interior‑point method; the convolution matrix is
    built once per length and reused.

2.  Simulated‑annealing fine‑tuning.
    Starting from the LP‑refined sequence a short SA run (FFT‑based
    convolution, Metropolis acceptance, mass‑transfer / local‑smoothing
    moves) searches the neighbourhood for a better peak.

The whole search is wrapped in a time‑budget loop that repeatedly
initialises new candidates (random power‑law shapes of various lengths)
and applies the two phases.  The best sequence found before the deadline
is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Small utility caches
# ----------------------------------------------------------------------
_conv_idx_cache: dict[int, tuple[np.ndarray, np.ndarray]] = {}


def _conv_matrix_indices(n: int) -> tuple[np.ndarray, np.ndarray]:
    """Return row/column index arrays for the (2n‑1)×n convolution matrix."""
    if n in _conv_idx_cache:
        return _conv_idx_cache[n]

    # each column i contributes the vector f shifted by i positions
    row = np.empty(n * n, dtype=np.int64)
    col = np.empty(n * n, dtype=np.int64)
    pos = 0
    for i in range(n):
        row[pos:pos + n] = np.arange(i, i + n)
        col[pos:pos + n] = i
        pos += n
    _conv_idx_cache[n] = (row, col)
    return row, col


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Exact evaluation of the competition objective for a normalised sum."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create (i+1)^(-p) of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _lp_refine(
    a: np.ndarray,
    blend: float = 0.5,
    max_iter: int = 10,
    deadline: float = 1e20,
) -> tuple[np.ndarray, float]:
    """
    One‑step linear‑programming refinement.

    Maximise Σ g_i under f*g ≤ M (M = max(f*f)).
    The returned vector has the same ℓ¹‑norm as the input (√(2n)).
    """
    n = a.size
    target = np.sqrt(2.0 * n)
    row_idx, col_idx = _conv_matrix_indices(n)
    shape = (2 * n - 1, n)

    best = a.copy()
    best_M = _max_convolution_fft(a)

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        M = _max_convolution_fft(a)

        # Build the constraint matrix A (sparse)
        data = np.tile(a, n)                 # a repeated n times
        A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=shape)

        # Linear programme: maximise 1^T g   ⇔   minimise -1^T g
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M, dtype=np.float64),
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s_g = g.sum()
        if s_g <= 0.0:
            break

        # Normalise g to the required ℓ¹‑norm and blend
        g = g * (target / s_g)
        a = (1.0 - blend) * a + blend * g

        new_M = _max_convolution_fft(a)
        if new_M < best_M - 1e-12:
            best_M = new_M
            best = a.copy()

    return best, best_M


def _simulated_annealing_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast FFT‑based simulated annealing.

    Moves:
      * transfer a random fraction of mass from a “high‑weight’’ donor to a
        low‑weight receiver (weights derived from the active convolution
        entry);
      * replace a short random window by its average (local smoothing).
    Acceptance follows a Metropolis rule with an exponential temperature
    schedule.
    """
    n = a.size
    cur_sum = a.sum()
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    total = max(deadline - start, 1e-12)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial convolution (cached for speed)
    conv = np.fft.irfft(np.fft.rfft(a, n=size_fft) ** 2, n=size_fft)[: 2 * n - 1]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total

        # exponential schedules (tuned for the 1000 s budget)
        temp = 0.02 * (1e-5 / 0.02) ** tau
        max_frac = 0.20 * (0.01 / 0.20) ** tau

        # ---------------------------------------------------------
        #   data derived from the current peak of the convolution
        # ---------------------------------------------------------
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = np.zeros(n)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        # ---------------------------------------------------------
        #   propose a move
        # ---------------------------------------------------------
        if rng.random() < 0.70:          # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                             # local smoothing
            start_i = rng.integers(0, n)
            length = rng.integers(2, 21)          # window ≤ 20
            end_i = min(start_i + length, n)
            if end_i - start_i < 2:
                continue
            saved = a[start_i:end_i].copy()
            a[start_i:end_i] = saved.mean()
            move_type = "smooth"
            saved_start, saved_end = start_i, end_i

        # ---------------------------------------------------------
        #   evaluate the new state (single FFT)
        # ---------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = 2.0 * n * conv_new.max() / (cur_sum ** 2)

        # ---------------------------------------------------------
        #   Metropolis acceptance
        # ---------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved

        it += 1

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.

    The routine mixes a linear‑programming “mass‑reallocation’’ step with a
    short simulated‑annealing refinement.  Multiple random restarts are
    performed until the time budget expires, and the best sequence found
    is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    #   initialise from the globally supplied construction if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n0 = init.size
        target = np.sqrt(2.0 * n0)
        if init.sum() > 0:
            init = init * (target / init.sum())
        else:
            init = _power_law_sequence(n0, 0.55, target)
        best_seq = init.copy()
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] start from global seq, score={best_score:.6f}")
    else:
        # fallback – a simple power‑law of moderate length
        n0 = 1500
        target = np.sqrt(2.0 * n0)
        best_seq = _power_law_sequence(n0, 0.55, target)
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] start from power‑law, score={best_score:.6f}")

    # ------------------------------------------------------------------
    #   a first deterministic refinement (LP + SA) on the starter
    # ------------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 30.0)
    refined, _ = _lp_refine(best_seq, blend=0.5, max_iter=20, deadline=sub_deadline)
    # short SA to polish
    refined, sa_score = _simulated_annealing_refine(refined, deadline=deadline)
    cand_score = _score(refined)
    if cand_score < best_score:
        best_seq, best_score = refined, cand_score
        print(f"[{time.time():.1f}] after deterministic refinement, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    #   random multi‑start exploration while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:           # leave a few seconds for cleanup
        # random length (1200 – 3000) and power‑law exponent
        n = int(rng.integers(1200, 3001))
        p = float(rng.uniform(0.35, 1.10))
        target = np.sqrt(2.0 * n)
        a = _power_law_sequence(n, p, target)

        # allocate a modest time slice for this candidate
        now = time.time()
        remaining = deadline - now
        if remaining < 20.0:
            break
        # split remaining time roughly 30 % LP, 30 % SA, rest for next candidate
        lp_time = min(remaining * 0.30, 20.0)
        sa_time = min(remaining * 0.30, 20.0)

        lp_deadline = now + lp_time
        a_lp, _ = _lp_refine(a, blend=0.5, max_iter=15, deadline=lp_deadline)

        sa_deadline = lp_deadline + sa_time
        a_sa, _ = _simulated_annealing_refine(a_lp, deadline=sa_deadline)

        cand_score = _score(a_sa)
        if cand_score < best_score:
            best_seq, best_score = a_sa.copy(), cand_score
            print(f"[{time.time():.1f}] new global best {best_score:.6f} (n={n}, p={p:.3f})")

    # ------------------------------------------------------------------
    #   final housekeeping – trim trailing near‑zeros
    # ------------------------------------------------------------------
    best_arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(best_arr > 1e-12)[0]
    if nz.size:
        best_arr = best_arr[: nz[-1] + 1]
    else:
        best_arr = np.array([0.0])

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_arr)}")
    return best_arr.tolist()
