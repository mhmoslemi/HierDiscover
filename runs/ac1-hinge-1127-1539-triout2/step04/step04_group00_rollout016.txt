```
def _lp_optimize(a0, num_iters, step_size, rows, cols, diff, deadline):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_M = np.max(_full_convolution(a))
    for it in range(num_iters):
        if time.time() >= deadline:
            break
        M = np.max(_full_convolution(a))
        b_raw = _lp_step(a, rows, cols, diff, M)
        if b_raw is None:
            break
        sum_b = b_raw.sum()
        if sum_b == 0:
            continue
        g = b_raw * (S / sum_b)
        # optionally clip to [0, 1000]
        g = np.clip(g, 0, 1000)
        a = (1 - step_size) * a + step_size * g
        a = _project_simplex(a, S)
        cur_M = np.max(_full_convolution(a))
        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()
        # adapt step size maybe reduce over iterations
        step_size = max(0.01, step_size * 0.95)
    return best_a, best_M
```

It combines a **linear‑programming oracle** (the Frank–Wolfe step described in the
paper) with the previous smooth/sub‑gradient optimisation.  
The LP step maximises the ℓ₁‑norm of a new vector `b` under the constraint
`f * b ≤ M` (where `M = max(f∗f)`), then rescales `b` to the simplex and
replaces `f` with it.  Repeating this operation quickly drives the peak of the
auto‑convolution down.  The whole routine runs until the allotted 1000 s are
exhausted, performing several random restarts and keeping the best sequence
found.

```python
"""
step‑function optimiser for the “max‑convolution” objective.

*  Uses a linear‑programming (LP) Frank–Wolfe oracle: for a current vector
   f, maximise Σ b_j subject to f*b ≤ M (M = max(f*f)) and b ≥ 0.
   The optimiser rescales the LP solution to the simplex Σ b = √(2n)
   and replaces f with it.  Repeating this step monotonically reduces the
   peak of the auto‑convolution.

*  After a short LP phase a smooth‑gradient / sub‑gradient refinement (the
   previous code) is applied to squeeze the remaining margin.

*  Random restarts, different lengths (800‑4000) and occasional “mass‑swap”
   moves explore the search space.  All operations respect the security checks
   performed inside ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for n up to a few 10⁴)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """soft‑max: w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial vector generator (simplex normalised)
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ₁‑norm √(2·n).
    If a base sequence is given and long enough, it is used (with tiny jitter);
    otherwise a Dirichlet(1) sample is drawn.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)               # uniform Dirichlet(1)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Smooth optimisation phase (soft‑max + SGD)
# ----------------------------------------------------------------------


def _optimize_smooth(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (tuned for this problem)
    lr = 0.3
    lr_decay = 0.9997
    alpha = 0.5
    alpha_mul = 1.005
    swap_prob = 0.06
    swap_frac = 0.02

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        a = a - lr * grad
        a = _project_simplex(a, S)

        # occasional random “mass‑swap”
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand
                    conv = _full_convolution(a)

        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        lr *= lr_decay
        alpha = min(alpha * alpha_mul, 30.0)

        if it % 4000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (act on the worst shift)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # tiny random perturbation
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined smooth → sub‑gradient optimiser (as in the previous code)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_smooth(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Linear‑programming oracle (Frank–Wolfe step)
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray,
             rows: np.ndarray,
             cols: np.ndarray,
             diff: np.ndarray,
             M: float) -> np.ndarray | None:
    """
    Solve  max Σ b_j   s.t.   f * b ≤ M   and   b ≥ 0.
    Returns the optimal vector b (or None on failure).
    """
    data = f[diff]                     # coeffs of the convolution matrix
    n = f.shape[0]
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n, dtype=float)       # minimise -Σb  ↔ maximise Σb
    bounds = [(0, None)] * n

    try:
        # “highs” works with sparse matrices and is very fast for this size
        res = opt.linprog(c, A_ub=A, b_ub=ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
    except Exception:
        return None

    if not res.success:
        return None
    return res.x


def _lp_optimize(a0: np.ndarray,
                 rows: np.ndarray,
                 cols: np.ndarray,
                 diff: np.ndarray,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeated Frank–Wolfe (LP) steps.
    Stops after ~20 iterations or when time runs out.
    Returns the best vector found and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_M = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        M = np.max(_full_convolution(a))

        b_raw = _lp_step(a, rows, cols, diff, M)
        if b_raw is None:
            break
        sum_b = b_raw.sum()
        if sum_b == 0:
            break

        # scale to the simplex Σb = √(2n)
        g = b_raw * (S / sum_b)
        # keep values inside the evaluator’s clipping range
        g = np.clip(g, 0.0, 1000.0)

        a = g                               # replace current iterate
        cur_M = np.max(_full_convolution(a))

        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()

        # a very small safety stop
        if it >= 20:
            break

    return best_a, best_M

        if best_overall_seq is None or rng.random() < 0.25:
            n = int(rng.integers(800, 4001))          # 800 … 4000
            a0 = _initial_sequence(base_seq, n, rng)
        else:
            # continue refining the current best
            a0 = best_overall_seq.copy()
            n = a0.shape[0]

        # pre‑compute the index‑patterns needed for the LP matrix
        i = np.arange(n, dtype=np.int32)[:, None]     # (n,1)
        j = np.arange(n, dtype=np.int32)[None, :]     # (1,n)
        rows = (i + j).ravel()
        cols = j.ravel()
        diff = i.ravel()                              # i = row – col

        # allocate a slice of the remaining budget for this attempt
        slice_deadline = min(time.time() + 30.0, deadline)

        # ------------------------------------------------------------------
        # 1) LP (Frank–Wolfe) phase
        # ------------------------------------------------------------------
        a_lp, score_lp = _lp_optimize(a0, rows, cols, diff, slice_deadline)

        # ------------------------------------------------------------------
        # 2) Optional smooth / sub‑gradient refinement
        # ------------------------------------------------------------------
        if time.time() < slice_deadline:
            a_ref, score_ref = _optimize_combined(a_lp, slice_deadline, rng)
            if score_ref < score_lp:
                a_lp, score_lp = a_ref, score_ref

        # ------------------------------------------------------------------
        # Record the best sequence discovered so far
        # ------------------------------------------------------------------
        if score_lp < best_overall_score:
            best_overall_score = score_lp
            best_overall_seq = a_lp.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} "
                  f"new best score={best_overall_score:.6f}")

        # occasional “hard reset” if we have stalled for a while
        if time.time() - last_improve > 45.0:
            best_overall_seq = None
            print("[info] stalled – discarding current best and restarting")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
```
