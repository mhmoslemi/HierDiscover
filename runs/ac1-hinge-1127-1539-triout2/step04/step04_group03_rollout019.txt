def propose_candidate(...):
    start = time.time()
    deadline = start + budget_s - 0.5

    # Choose a set of candidate lengths
    lengths = [2000, 3000, 4000]   # may try multiple sequentially
    best_seq = None
    best_score = np.inf

    for n in lengths:
        if time.time() >= deadline: break
        # generate population
        popsize = 30
        population = []
        # create initial individuals using several random families
        for i in range(popsize):
            seq = _random_initial_sequence(n, np.random.default_rng())
            pop_score = _max_convolution_fft(seq)
            population.append((pop_score, seq))
            if pop_score < best_score:
                best_score = pop_score; best_seq = seq.copy()
        
        # sort
        population.sort(key=lambda x: x[0])
        # Evolutionary loop
        gen=0
        while time.time() < deadline:
            # select parents
            p1_score, p1 = population[0]    # elite
            # random parent from top half
            p2_score, p2 = population[np.random.randint(popsize//2)]
            alpha = np.random.rand()
            child = alpha * p1 + (1-alpha) * p2
            # optional mutation:
            if np.random.rand() < 0.3:
                # move random fraction from random index to another
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j and child[i] > 1e-12:
                    frac = np.random.rand() * 0.2
                    delta = child[i] * frac
                    child[i] -= delta
                    child[j] += delta
            # project to simplex
            child = _project_to_simplex(child, np.sqrt(2.0 * n))
            child_score = _max_convolution_fft(child)
            # replace worst if better
            if child_score < population[-1][0]:
                population[-1] = (child_score, child)
                population.sort(key=lambda x: x[0])
                if child_score < best_score:
                    best_score = child_score; best_seq = child.copy()
            # occasional local refinement using gradient descent
            if gen % 50 == 0:
                # refine best candidate
                refined = _gradient_descent_opt(population[0][1], time.time() + 0.1)
                refined_score = _max_convolution_fft(refined)
                if refined_score < best_score:
                    best_score = refined_score; best_seq = refined.copy()
                population[0] = (refined_score, refined)
                population.sort(key=lambda x: x[0])
            gen += 1

def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_obj = np.inf

    # optionally start from existing height_sequence_1 if exists
    init = globals().get('height_sequence_1')
    if init is not None:
        a_init = np.array(init, dtype=float)
        # project to correct sum for its length
        n_init = a_init.size
        a_init = _project_to_simplex(a_init, np.sqrt(2.0 * n_init))
        obj_init = _max_convolution_fft(a_init)
        if obj_init < best_obj:
            best_obj = obj_init; best_seq = a_init

    # Candidate lengths
    lengths = [1500, 2000, 2500, 3000, 3500, 4000]
    for n in lengths:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        # exponentials
        for lam in np.logspace(-5, -1, 30):
            if time.time() >= deadline:
                break
            raw = np.exp(-lam * np.arange(n))
            a = raw * (target / raw.sum())
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj = obj; best_seq = a.copy()
                print(f"[{time.time():.1f}] New best exp lam={lam:.5g} n={n} obj={obj:.6f}")

        # power laws
        for p in np.linspace(0.1, 2.0, 30):
            if time.time() >= deadline:
                break
            raw = (np.arange(1, n+1, dtype=float)) ** (-p)
            a = raw * (target / raw.sum())
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj = obj; best_seq = a.copy()
                print(f"[{time.time():.1f}] New best power p={p:.3f} n={n} obj={obj:.6f}")

        # linear decreasing from 1 to epsilon
        for start_val in np.linspace(0.2, 1.0, 10):
            if time.time() >= deadline:
                break
            raw = np.linspace(start_val, 1.0, n)  # decreasing? Actually start > end? we want decreasing from start to 0
            raw = raw[::-1]  # make decreasing from start to end
            a = raw * (target / raw.sum())
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj = obj; best_seq = a.copy()
                print(f"[{time.time():.1f}] New best linear start={start_val:.3f} n={n} obj={obj:.6f}")

        # mixture of two exponentials (asymmetric)
        for _ in range(20):
            if time.time() >= deadline:
                break
            lam1 = rng.uniform(0.0005, 0.02)
            lam2 = rng.uniform(0.0005, 0.02)
            w = rng.random()
            raw = w * np.exp(-lam1 * np.arange(n)) + (1 - w) * np.exp(-lam2 * np.arange(n)[::-1])
            a = raw * (target / raw.sum())
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj = obj; best_seq = a.copy()
                print(f"[{time.time():.1f}] New best mix lam1={lam1:.5g} lam2={lam2:.5g} w={w:.3f} n={n} obj={obj:.6f}")

        # random perturbation of best sequence for local search
        if best_seq is not None:
            for _ in range(5):
                if time.time() >= deadline:
                    break
                a = best_seq.copy()
                # move small random fraction from random index to another
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.random() * 0.2
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, target)
                obj = _max_convolution_fft(a)
                if obj < best_obj:
                    best_obj = obj; best_seq = a.copy()
                    print(f"[{time.time():.1f}] New best perturbed n={n} obj={obj:.6f}")

    # final gradient descent on the best sequence
    if best_seq is not None and time.time() < deadline:
        sub_deadline = deadline
        a_refined = _gradient_descent_opt(best_seq, sub_deadline)
        obj = _max_convolution_fft(a_refined)
        if obj < best_obj:
            best_obj = obj; best_seq = a_refined.copy()
            print(f"[{time.time():.1f}] After gradient descent obj={obj:.6f}")

    # final simulated annealing for last few seconds
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = _simulated_annealing(best_seq, deadline)
        obj = _max_convolution_fft(a_final)
        if obj < best_obj:
            best_obj = obj; best_seq = a_final.copy()
            print(f"[{time.time():.1f}] After SA final obj={obj:.6f}")

    # Trim trailing zeros
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
    # Evaluate finally via official evaluate_sequence for sanity
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished best obj {best_obj:.6f} length {len(best_seq)}")
    return best_seq.tolist()

"""
Fast optimisation of the step‑function evaluation problem.

The objective is                2·n·max(conv(a,a)) / (sum(a)^2)
which is invariant to a global scaling of ``a``.
If we enforce the ℓ¹‑simplex Σ a_i = √(2·n) the objective reduces to
    max_k (a*a)[k]                 (the maximal entry of the self‑convolution).

A very effective family for this problem is the (left‑hand) exponential
shape

        a_i ∝ exp(‑λ·i) ,   i = 0 … n‑1

with λ = c / n.  For large n the maximal convolution value depends only on
the parameter c and is approximately

        f(c) = 2·c·e^{‑c} / (1‑e^{‑c})² .

The minimum of f(c) is attained for a moderate constant c (≈ W(n) ≈ 6–8 for
our problem sizes) and yields values of the objective well below 1.0.
The code below scans a fine grid of c values for several vector lengths,
keeps the best sequence, and finally refines it with a short projected‑
gradient descent on a smooth log‑sum‑exp surrogate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, max entry)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the full linear convolution a * a using FFT."""
    n = a.size
    # size must be at least 2*n‑1 and a power of two for fast FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex  {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used for a short refinement)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of (1/β)·log Σ exp(β·(a*a)[k]) """
    n = a.shape[0]

    # convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Quick projected gradient descent on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 5e5
    momentum = np.zeros_like(a)
    rho = 0.9
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0:
            lr = 0.7 * target_sum / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1
    return a

# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative sequence that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # Optionally start from a globally supplied construction
    # ------------------------------------------------------------------
    init = globals().get("height_sequence_1")
    if isinstance(init, (list, tuple, np.ndarray)):
        a0 = np.asarray(init, dtype=float)
        n0 = a0.size

        obj0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_obj = obj0
        print(f"[{time.time():.1f}] Using global height_sequence_1 (n={n0}) → obj={obj0:.6f}")

    # ------------------------------------------------------------------
    # Search over several lengths – the optimum improves with larger n
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----- exponential family (a_i ∝ exp(-c/n * i)) -----
        # parameter c roughly corresponds to the solution of c·e^c = n
        # we sweep a generous interval and keep the best.
        c_vals = np.linspace(0.5, 15.0, 200)          # 200 different shapes

                best_seq = a.copy()
