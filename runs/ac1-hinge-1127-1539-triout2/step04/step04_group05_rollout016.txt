    best_a = a.copy()
    best_max = _full_convolution_fft(a).max()

    while time.time() < deadline:
        # compute convolution and max
        b = _full_convolution_fft(a)
        max_b = b.max()

        # if max_b < best_max: update best
        if max_b < best_max:
            best_max = max_b
            best_a = a.copy()

        # compute gradient using softmax weights
        tau = max(tau * 0.999, max_b * 1e-4)   # decreasing schedule
        # Softmax gradient
        diff = b - max_b
        # note: we need smoothing param: smaller tau reduces to near-hinge
        exps = np.exp(diff / max(tau, 1e-12))
        w = exps / np.sum(exps)

        # cross-correlation
        a_rev = a[::-1]
        conv_len = len(w) + n - 1
        f_len = _next_pow2(conv_len)
        conv = np.fft.irfft(np.fft.rfft(w, n=f_len) * np.fft.rfft(a_rev, n=f_len), n=f_len)[:conv_len]

        # momentum update
        momentum = beta * momentum - eta * grad
        a_candidate = a + momentum
        a_candidate = _project_to_simplex(a_candidate, 1.0)
        # reflect symmetry
        a_candidate = (a_candidate + a_candidate[::-1]) / 2
        a_candidate = _project_to_simplex(a_candidate, 1.0)

        # accept if improvement else maybe accept with small prob
        b_cand_max = _full_convolution_fft(a_candidate).max()

def _gradient_descent_optimize(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = 1.0

    best_a, best_score = a.copy(), _compute_score(a)

    # parameters
    eta = 0.2           # step size
    beta = 0.9          # momentum
    v = np.zeros_like(a)

    tau = 0.1   # initial smoothing, will adapt later
    # measure initial max to set tau
    max_b = _full_convolution_fft(a).max()
    tau = max_b * 0.5  # initial

    # schedule parameters
    tau_decay = 0.999
    eta_decay = 0.9995

    while time.time() < deadline:
        # compute convolution
        b = _full_convolution_fft(a)
        max_b = b.max()

        # record best
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # softmax gradient
        # use current tau > tiny
        if tau > 1e-12:
            diff = b - max_b
            # to avoid overflow, subtract max(b) (already done)
            # scale with tau
            exps = np.exp(diff / tau)
            w = exps / np.sum(exps)
            a_rev = a[::-1]
            conv_len = len(w) + n - 1
            f_len = _next_pow2(conv_len)
            conv = np.fft.irfft(np.fft.rfft(w, n=f_len) * np.fft.rfft(a_rev, n=f_len), n=f_len)[:conv_len]
            grad = 2.0 * conv[n - 1:n - 1 + n]
        else:
            # subgradient for max entry
            mask = (b >= max_b - 1e-12).astype(float)
            a_rev = a[::-1]
            conv = np.fft.irfft(np.fft.rfft(mask, n=_next_pow2(len(mask)+n-1)) * np.fft.rfft(a_rev, n=_next_pow2(len(mask)+n-1)), n=_next_pow2(len(mask)+n-1))[:len(mask)+n-1]
            grad = 2.0 * conv[n-1:n-1+n]

        # momentum
        v = beta * v - eta * grad
        a_new = a + v

        # enforce nonnegativity and simplex projection
        a_new = _project_to_simplex(a_new, S)

        # add optional random jitter sometimes
        if rng.random() < 0.01:
            noise = rng.normal(scale=0.01, size=n)
            a_new = np.clip(a_new + noise, 0, None)
            a_new = _project_to_simplex(a_new, S)

        a = a_new

        # decay
        tau *= tau_decay
        eta *= eta_decay

        # random mass swap occasionally
        if rng.random() < 0.05:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n)
                if i != j:
                    delta = a[i] * 0.1  # move 10% of mass
                    a_swap = a.copy()
                    a_swap[i] -= delta
                    a_swap[j] += delta
                    a_swap = _project_to_simplex(a_swap, S)
                    a = a_swap

    return best_a, best_score

def _fft_convolve(x, y):
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv

"""Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence().

The algorithm has three stages:

1. **Exploration via low‑dimensional parametric families** –  
   for several lengths `n` we scan simple U‑shaped families
   `a_i ∝ (|i−(n−1)/2|+1)^p` and
   `a_i ∝ (i+1)^p+(n−i)^p` (the latter is uniform for p=1).  
   The best candidate of this cheap grid search is kept.

2. **Intensive local optimisation** –  
   starting from the best candidate we run an accelerated gradient
   descent on the smooth soft‑max approximation of the ℓ∞‑norm of the
   self‑convolution, with momentum and a projection onto the simplex
   (∑a_i = 1, a_i ≥ 0).  The smoothing parameter τ is annealed to zero,
   so the method automatically switches to a sub‑gradient step on the
   true maximum.

3. **Additional short restarts** –  
   while time permits we launch cheap 10‑second refinements from
   random initialisations (or the previously known sequence) to guard
   against being trapped in a local minimum.

All convolutions are performed with a real FFT (O(n log n)).
The sequence is finally returned as a plain Python list; the global
variable ``height_sequence_1`` is updated so that later calls can start
from the current best construction.  The routine respects the 1000 s
budget and prints intermediate scores for tracing progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, S: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto {x ≥ 0, Σx = S}.
    Returns a new array (does not modify v).
    """
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """a * a (full convolution, length 2·len(a)‑1)."""
    return _fft_convolve(a, a)


def _compute_score(a: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a (Σa = 1)."""
    n = a.shape[0]
    max_b = _full_convolution_fft(a).max()
    # Σa is 1 → denominator = 1
    return 2.0 * n * max_b


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to length ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0], dtype=float)
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient‑descent optimiser (soft‑max + momentum)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the τ‑soft‑max approximation of max(a*a).
    Returns a vector of the same shape as a.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    # prevent overflow – shift by the max (already done)
    exps = np.exp((b - max_b) / max(tau, 1e-12))
    w = exps / np.sum(exps)                     # shape (2n‑1,)
    # cross‑correlation a ⋆ w
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)             # length 3n‑2
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def _subgradient(a: np.ndarray, eps: float = 0.0) -> np.ndarray:
    """
    Exact sub‑gradient of max(a*a).  All entries whose value is within
    ``eps`` of the maximum are treated as active.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)
    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def _local_optimize(start_a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Accelerated gradient descent with a decreasing soft‑max temperature.
    Returns the best vector found and its score.
    """
    a = start_a.copy()
    n = a.shape[0]
    S = 1.0                                          # we keep Σa = 1

    # bookkeeping
    best_a = a.copy()
    best_score = _compute_score(a)

    # hyper‑parameters
    eta = 0.8                                        # base step size
    eta_decay = 0.9997
    beta = 0.9                                        # momentum coeff
    momentum = np.zeros_like(a)

    tau = _full_convolution_fft(a).max() * 0.5       # initial smoothing
    tau_decay = 0.9995

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) gradient step (soft‑max while tau > 0, otherwise sub‑grad)
        # --------------------------------------------------------------
        if tau > 1e-12:
            grad = _softmax_gradient(a, tau)
        else:
            grad = _subgradient(a, eps=1e-12)

        # momentum update
        momentum = beta * momentum - eta * grad
        a_candidate = a + momentum

        # keep the vector non‑negative and on the simplex
        a_candidate = _project_to_simplex(a_candidate, S)

        # enforce symmetry (the optimum can be taken symmetric)
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        a_candidate = _project_to_simplex(a_candidate, S)

        # occasional random jitter to escape plateaus
        if rng.random() < 0.02:
            noise = rng.normal(scale=0.01, size=n)
            a_candidate = np.clip(a_candidate + noise, 0.0, None)
            a_candidate = _project_to_simplex(a_candidate, S)

        # evaluate
        cand_score = _compute_score(a_candidate)

        # accept if improvement or with a tiny annealing probability
        cur_score = _compute_score(a)
        if cand_score < cur_score or rng.random() < 0.001:
            a = a_candidate
            cur_score = cand_score

        # bookkeeping
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # schedule updates
        tau *= tau_decay
        eta *= eta_decay

        # random mass‑swap move (simulated‑annealing style)
        if rng.random() < 0.10:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n)
                if i != j:
                    delta = 0.08 * a[i]                     # move 8 % of the mass
                    a_swap = a.copy()
                    a_swap[i] -= delta
                    a_swap[j] += delta
                    a_swap = _project_to_simplex(a_swap, S)
                    a_swap = (a_swap + a_swap[::-1]) * 0.5
                    a_swap = _project_to_simplex(a_swap, S)
                    a = a_swap

    return best_a, best_score
