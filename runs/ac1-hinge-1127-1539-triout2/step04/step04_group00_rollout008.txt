"""
Improved optimiser for the self‑convolution max‑norm.

The algorithm works on the L¹‑simplex Σa = √(2·n) – the objective
becomes simply max(conv(a,a)).  A fast FFT‑based convolution is used.
We minimise the smooth log‑sum‑exp approximation of the max with an
Adam optimiser (momentum, adaptive steps) while projecting onto the
simplex after each update.  The temperature (soft‑max sharpness) is
grown gradually up to a large value, so that the smooth objective
converges to the true max‑norm.  Random “mass‑swap’’ moves occasionally
inject noise to escape plateaus.  The routine performs a few short
re‑starts with different dimensions and interpolates the current best
solution when the dimension changes.

The code respects the 1000 s budget, always returns a feasible sequence
and prints progress information.
"""
import time
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights – smooth approximation of max
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)  (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Sequence interpolation / initialisation utilities
# ----------------------------------------------------------------------
def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to length ``new_n`` by linear interpolation."""
    if seq.size == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, seq.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    return new_seq


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with L¹‑norm √(2·n)."""
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= 1:
        # Either interpolate an existing best solution or take a slice
        a = _interpolate_sequence(base_seq, n)
        # tiny jitter to avoid stagnation
        jitter = rng.normal(scale=max(a.mean(), 1e-6) * 1e-3, size=n)
        a = np.clip(a + jitter, 0.0, None)
    else:
        a = rng.random(n)               # Dirichlet(1) – uniform on simplex.

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑norm (log‑sum‑exp)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 400.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max‑norm.
    Returns the best vector seen and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.08                     # base learning rate
    lr_decay = 0.9999
    alpha = 0.5                   # start temperature
    alpha_mul = 1.01

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- 1. evaluate current conv and soft‑max weights ----
        conv = _full_convolution(a)
        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # ---- 2. compute gradient of smoothed max ----
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # ---- 3. Adam update ----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a, S)

        # ---- 4. occasional random mass‑swap (helps escaping plateaus) ----
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = rng.random() * 0.03 * a[i]
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # accept if it improves or with a small probability
                if cand_max < cur_max or rng.random() < 0.1:
                    a = cand
                    cur_max = cand_max

        # ---- 5. schedule updates ----
        lr *= lr_decay
        alpha = min(alpha * alpha_mul, max_alpha)

        # early stop if no improvement for a while
        if it % 4000 == 0 and best_score == cur_max:
            break

    return best_a, best_score
