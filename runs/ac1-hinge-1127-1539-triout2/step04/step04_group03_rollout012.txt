def _subgradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0*n)
    momentum = np.zeros_like(a)
    rho = 0.9
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute conv
        conv = _full_convolution_fft(a)
        max_b = conv.max()
        mask = conv >= max_b - 1e-12
        p = mask.astype(float)
        if p.sum() == 0:  # no active? not possible
            break
        p = p / p.sum()

        # gradient via convolution
        a_rev = a[::-1]
        # convolve a_rev with p (like before)
        conv_len = a_rev.size + p.size - 1  # 3n-2
        size2 = 1 << (conv_len.bit_length())
        Ar_fft = np.fft.rfft(a_rev, n=size2)
        Pr_fft = np.fft.rfft(p, n=size2)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n-1 : n-1+n]   # central n entries

        # step size
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = 0.5 * target_sum / (norm_g * (1.0 + it/(1000+1)))  # maybe decaying
            # incorporate momentum
            momentum = rho * momentum + (1 - rho) * g
            a -= lr * momentum

        # project
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (1.0 + it/(500)), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a

"""
Hybrid sub‑gradient / simulated‑annealing optimizer.

The objective is 2·n·max(a*a)/(∑a)².  Because the ratio is scale‑invariant,
we fix the ℓ¹‑simplex Σa_i = √(2 n).  The algorithm works as follows.

1. **Random starts** – a handful of diverse families (power‑law,
   exponential, Dirichlet, symmetric, uniform) are projected onto the
   simplex.

2. **Sub‑gradient descent on the exact max** – at each iteration the
   current maximal convolution entry is identified, a sub‑gradient
   (the average of the gradients of all active entries) is computed by
   a second FFT, and a projected (momentum) step is taken.  The step‑size
   decays like 1/√t and occasional Gaussian jitter is added to escape
   plateaus.

3. **Simulated annealing polishing** – from the best vector obtained so
   far a fast SA loop repeatedly transfers a random fraction of mass
   between two indices, accepting uphill moves with a slowly decreasing
   temperature.  Global jitter restructures the vector from time to
   time.

All work is done inside the supplied time budget (default 1000 s).  The
best sequence found (as a plain ``list[float]``) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialiser families (projected onto the simplex)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative sequence a with Σa = √(2 n).
    Several families are mixed to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # decreasing power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # left‑edge exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law & exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (full simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif kind == 4:                 # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                           # uniform plus tiny jitter
        a = np.full(n, target / n)

    # tiny Gaussian perturbation and final projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max (convex objective)
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max of the self‑convolution.
    Uses a simple momentum scheme, a 1/√t step‑size schedule and occasional
    Gaussian jitter to avoid stagnation.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # momentum parameters
    rho = 0.9
    momentum = np.zeros_like(a)

    # step‑size scaling (empirically good for the simplex size)
    step0 = target_sum * 0.6

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # ----- convolution and active set -----
        conv = _full_convolution_fft(a)
        max_b = conv.max()
        # treat everything within 1e‑12 of the top as active
        active = conv >= max_b - 1e-12
        p = active.astype(float)
        cnt = p.sum()
        if cnt == 0:
            cnt = 1.0
        p /= cnt

        # ----- sub‑gradient (average over active indices) -----
        a_rev = a[::-1]
        conv_len = n + (2 * n - 1) - 1            # = 3·n‑2
        size2 = 1 << (conv_len.bit_length())
        Ar_fft = np.fft.rfft(a_rev, n=size2)
        Pr_fft = np.fft.rfft(p, n=size2)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]     # central n entries

        # ----- momentum update & projection -----
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step0 / (norm * np.sqrt(it + 1))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # ----- occasional jitter to jump out of plateaus -----
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a random fraction of mass between
    two indices, accepting uphill moves with probability exp(-Δ/T).  Temperature
    decays multiplicatively; a very occasional global jitter reshuffles the
    vector.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # start temperature a small fraction of the objective
    T = best_score * 0.05 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        # choose a source with non‑zero mass
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        # choose a distinct destination
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 15 %) of the source mass
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter (≈ 0.2 % of iterations)
        if rng.random() < 0.002:
            jitter = rng.normal(scale=best.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, best.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget,
    explores several vector lengths, runs many random restarts,
    applies a long sub‑gradient descent phase and finishes with a
    simulated‑annealing polishing step.  The best sequence found is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # 0.5 s safety margin
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_obj = np.inf

    # --------------------------------------------------------------
    # 1) initialise from a globally supplied solution, if present
    # --------------------------------------------------------------
    gl_seq = globals().get("height_sequence_1")
    if isinstance(gl_seq, (list, tuple, np.ndarray)):
        a_gl = np.asarray(gl_seq, dtype=float)
        n0 = a_gl.size
        a_gl = _project_to_simplex(a_gl, np.sqrt(2.0 * n0))
        obj0 = _max_convolution_fft(a_gl)
        best_seq = a_gl.copy()
        best_obj = obj0
        print(f"[{time.time():.1f}] Starting from global height_sequence_1 "
              f"(n={n0})  obj={best_obj:.6f}")

    # --------------------------------------------------------------
    # 2) explore a range of lengths
    # --------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # number of random starts per length (adjustable)
        seeds_per_n = 4

        for _ in range(seeds_per_n):
            if time.time() >= deadline:
                break

            a0 = _random_initial_sequence(n, rng)

            # allocate a slice of the remaining budget for the GD phase
            sub_dead = min(deadline, time.time() + 12.0)   # ≈12 s per start
            a_gd = _subgradient_descent_opt(a0, sub_dead)

            # evaluate
            obj = _max_convolution_fft(a_gd)
            if obj < best_obj:
                best_obj = obj
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New global best (GD) n={n}  obj={best_obj:.6f}")

    # --------------------------------------------------------------
    # 3) final simulated‑annealing polishing on the current best
    # --------------------------------------------------------------
    if best_seq is not None:
        a_sa = _simulated_annealing(best_seq, deadline)
        obj = _max_convolution_fft(a_sa)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] New global best after SA  obj={best_obj:.6f}")

    # --------------------------------------------------------------
    # 4) tidy up and return
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # trim trailing near‑zero entries – makes the output nicer
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # optional sanity check with the official evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max conv) = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
