"""
Search for a non‑negative step function (height sequence) that minimises the
``evaluate_sequence`` objective.

The algorithm proceeds in four stages within the 1000 s time budget:

1. **Parametric sweep** – generate a variety of deterministic shapes
   (linear, quadratic, exponential, power, inverse and several biased
   versions) together with a few random Dirichlet‑like vectors.
   Every candidate is projected onto the ℓ₁‑simplex so that the sum of the
   coefficients equals √(2 n).  The best candidate is kept.

2. **Smoothed gradient descent** – minimise a log‑sum‑exp surrogate of the
   max‑self‑convolution.  The gradient is obtained analytically via FFT,
   an Adam optimiser is used, and after each step the vector is
   re‑normalised to the required sum (simple ℓ₁‑normalisation).  The
   smoothing parameter τ is decreased geometrically; occasional random
   pair‑mass transfers and jitter keep the iterates from stagnating.

3. **Sub‑gradient refinement** – switch to the true nonsmooth objective.
   The sub‑gradient is the average of gradients of all (approximately)
   maximal lags.  A diminishing step size together with occasional random
   pair‑transfers is employed.

4. **Polishing / simulated annealing** – a final burst of tiny random
   pair‑transfers is performed; moves that improve the objective are kept,
   otherwise they are accepted with a small temperature‑dependent
   probability.  The best sequence found is returned.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT of length
    equal to the next power of two.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (assumed non‑negative)."""
    return float(conv_self_fft(a).max())


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence of length n.
    `kind` may be one of:
        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right, exponential_left, exponential_right.
    The parameter `param` controls the shape (exponent, rate, etc.).
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        # avoid the zero at d=0
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        # all entries will be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def simple_normalise(v: np.ndarray, target: float) -> np.ndarray:
    """
    Faster ℓ₁‑normalisation used inside the iterative phases.
    Sets negatives to zero, then rescales so the sum equals ``target``.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, target / v.size)
    return v * (target / s)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑normalise to the required sum.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return simple_normalise(a, target_sum)

        "inverse": np.linspace(0.5, 5.0, 11),
        "linear_left": np.linspace(0.0, 0.9, 10),
        "linear_right": np.linspace(0.0, 0.9, 10),
        "exponential_left": np.linspace(0.0, 6.0, 21),
        "exponential_right": np.linspace(0.0, 6.0, 21),
    }

    print(f"[{time.time():.2f}] Phase 0 – parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # a few random Dirichlet‑like vectors
    for _ in range(25):
        cand = np.random.rand(n)
        cand = simple_normalise(cand, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best random init → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during random init.")
            return best_seq.tolist()

    # incorporate supplied baseline if it was up‑sampled
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves start → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed gradient descent (Adam) -----------------------
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.55
    a = best_seq.copy()
    m = np.zeros_like(a)       # first‑moment estimate
    v = np.zeros_like(a)       # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.8 * (target_sum / n)   # base learning rate
    tau_factor = 0.12
    iter_cnt = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Phase 1 – smoothed gradient descent (Adam)")
    while time.time() < phase1_deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()
        tau = max(tau_factor * max_conv, 1e-12)

        # log‑sum‑exp weights
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via FFT (see derivation in the docstring)
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

        iter_cnt += 1

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep non‑negative and renormalise to the required sum
        a = simple_normalise(a, target_sum)

        # occasional perturbations to escape plateaux
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.008)

        if iter_cnt % 2000 == 0:
            lr *= 0.94
            tau_factor = max(0.001, tau_factor * 0.94)

        # evaluate
        cur_score = compute_max_conv(a)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iter_cnt:07d} – improved to {best_score:.6f}")
        else:
            no_improve += 1

        if no_improve >= 8000:
            # inject a tiny jitter
            jitter = np.random.randn(n) * 0.0005
            a = simple_normalise(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iter_cnt:07d} – jitter applied")

        if time.time() >= phase1_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – true sub‑gradient refinement ----------------------------
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.90
    a = best_seq.copy()
    step = 0.045 * (target_sum / n)    # a bit larger than before
    sub_iter = 0
    print(f"[{time.time():.2f}] Phase 2 – sub‑gradient refinement")
    while time.time() < phase2_deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()
        eps_local = max(1e-12, 1e-9 * max_conv)
        mask = (conv >= max_conv - eps_local).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0

        a_rev = a[::-1]
        L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        a = a - step * grad
        a = simple_normalise(a, target_sum)

        sub_iter += 1
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Sub‑grad iter {sub_iter:06d} – improved to {best_score:.6f}")

        if sub_iter % 1500 == 0:
            step *= 0.94

        if np.random.rand() < 0.018:
            a = random_pair_transfer(a, target_sum, scale=0.006)

        if time.time() >= phase2_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – polishing / simulated annealing -------------------------
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – polishing (simulated annealing)")
    cur_seq = best_seq.copy()
    cur_score = best_score
    temperature = max(best_score * 0.008, 1e-6)
    cooling = 0.9996

    while time.time() < deadline:
        # propose a tiny random pair transfer (scale adapts with temperature)
        scale = 0.0025 * (temperature / (best_score * 0.008) + 0.1)
        cand = random_pair_transfer(cur_seq.copy(), target_sum, scale=scale)
        cand_score = compute_max_conv(cand)

        # deterministic acceptance for improvements
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

        # simulated‑annealing acceptance
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / temperature):
            cur_seq = cand
            cur_score = cand_score

        temperature = max(temperature * cooling, 1e-9)

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
