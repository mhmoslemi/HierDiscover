maximum entry of its self‑convolution under the ℓ₁‑norm constraint
∑a_i = √(2·n).

The algorithm works as follows:

1.  Gather a set of candidate initial shapes (Hann, Hamming, Blackman,
    Sine, semi‑circle, quadratic, uniform) and, if available,
    a scaled copy of the previously best known sequence.
    For each restart a length n is sampled and the best‑initialising
    shape (with the smallest current max‑convolution) is selected.

2.  A time‑limited local optimiser is then run.  The optimiser mixes
    several move types:
        – exact sub‑gradient step on the active convolution entries,
        – multiplicative (exponentiated‑gradient) update,
        – smoothing by a small kernel,
        – random mass‑transfer between two coordinates,
        – Gaussian jitter on the whole vector.
    Moves that increase the objective are accepted with a simulated‑annealing
    probability that decays over time.

3.  The best sequence seen over all restarts is stored in the global
    variable ``height_sequence_1`` and returned.

All updates stay on the simplex {x≥0,∑x=S} where S=√(2·n) by
projecting onto the simplex after each move.

The implementation uses only NumPy and a real FFT for fast full
convolutions (O(n log n)).  It respects the 1000 s time budget.

import time

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, ∑x=S}."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback uniform
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·n‑1)."""
    return _fft_convolve(a, a)


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a one‑dimensional sequence to length new_n by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]

def _shape_uniform(n: int) -> np.ndarray:
    return np.ones(n)


def _shape_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def _shape_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def _shape_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    a0, a1, a2 = 0.42, 0.5, 0.08
    return a0 - a1 * np.cos(2.0 * np.pi * i / (n - 1)) + a2 * np.cos(4.0 * np.pi * i / (n - 1))


def _shape_sine(n: int) -> np.ndarray:
    i = np.arange(n)
    return np.sin(np.pi * i / (n - 1))


def _shape_semicircle(n: int) -> np.ndarray:
    i = np.arange(n)
    c = (n - 1) / 2.0
    x = (i - c) / c
    y = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))
    return y


def _shape_quadratic(n: int) -> np.ndarray:
    i = np.arange(n)
    c = (n - 1) / 2.0
    # parabola that is 0 at ends and 1 in middle
    y = 1.0 - ((i - c) / c) ** 2
    y = np.maximum(0.0, y)
    return y

    _shape_blackman,
    _shape_sine,

# Sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, eps: float = 1e-12):
    """
    Returns (g, max_b) where g is a sub‑gradient of max(a*a) and
    max_b = max_i (a*a)_i.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    # mask of active entries (those within eps of the maximum)
    mask = (b >= max_b - eps).astype(float)
    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)          # length 3n‑2
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g, max_b

def _initial_candidate(base_seq: np.ndarray | None,
                       n: int,
                       rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible starting vector of length n.
    Several candidates (scaled base, deterministic shapes, random) are
    generated and the one with the smallest current max‑convolution is
    returned.
    """
    S = np.sqrt(2.0 * n)
    candidates = []

    # 1) scaled previous best, if we have it
    if base_seq is not None and base_seq.size > 0:
        a = _scale_sequence(base_seq, n).astype(float)
        a = np.clip(a, 0.0, None)
        a = _project_to_simplex(a, S)
        candidates.append(a)

    # 2) deterministic shapes
    for fn in _SHAPE_FUNCS:
        a = fn(n).astype(float)
        a = np.clip(a, 0.0, None)
        a = _project_to_simplex(a, S)
        candidates.append(a)

    # 3) pure random Dirichlet‑like draw
    a = rng.random(n)
    a = _project_to_simplex(a, S)
    candidates.append(a)

    # Pick the best among them (few extra convolutions are cheap)
    best = None
    best_val = np.inf
    for a in candidates:
        cur = _full_convolution_fft(a).max()
        if cur < best_val:
            best_val = cur
            best = a
    return best

    # step‑size for additive sub‑gradient moves
    step_scale = 0.02

    # step‑size for exponentiated‑gradient moves
    exp_eta = 0.02

    # smoothing kernel (length‑3)
    smooth_kernel = np.array([0.25, 0.5, 0.25])

    # probabilities for move types
    # 0.40 sub‑gradient, 0.25 multiplicative, 0.15 smoothing,
    # 0.13 random mass transfer, 0.07 jitter
    while time.time() < deadline:
        r = rng.random()

        if r < 0.40:
            # --- exact sub‑gradient step with simple back‑tracking ---
            g, max_b = _subgradient_max(a, eps=1e-12)
            # magnitude‑aware step size
            base_alpha = step_scale * S / (np.linalg.norm(g) + 1e-12)
            improved = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                step = base_alpha * factor
                cand = a - step * g
                cand = _project_to_simplex(cand, S)
                cand_max = _full_convolution_fft(cand).max()
                if cand_max < cur_max - 1e-11:
                    a = cand
                    cur_max = cand_max
                    improved = True
                    break
            if not improved:
                # accept worse move with SA probability
                cand = a - step_scale * g
                cand = _project_to_simplex(cand, S)
                cand_max = _full_convolution_fft(cand).max()
                if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(temperature, 1e-12)):
                    a = cand
                    cur_max = cand_max
                temperature *= temp_decay

        elif r < 0.65:
            # --- exponentiated‑gradient (multiplicative) update ---
            g, max_b = _subgradient_max(a, eps=1e-12)
            cand = a * np.exp(-exp_eta * g)
            cand = _project_to_simplex(cand, S)
            cand_max = _full_convolution_fft(cand).max()
            if cand_max < cur_max - 1e-11:
                a = cand
                cur_max = cand_max
            else:
                if rng.random() < np.exp(-(cand_max - cur_max) / max(temperature, 1e-12)):
                    a = cand
                    cur_max = cand_max
                temperature *= temp_decay

        elif r < 0.80:
            # --- smoothing move (local averaging) ---
            cand = np.convolve(a, smooth_kernel, mode='same')
            cand = _project_to_simplex(cand, S)
            cand_max = _full_convolution_fft(cand).max()
            if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(temperature, 1e-12)):
                a = cand
                cur_max = cand_max
                temperature *= temp_decay

        elif r < 0.93:
            # --- random mass transfer between two coordinates ---
            i = rng.integers(n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = rng.random() * 0.10 * a[i]   # up to 10 % of source mass
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_max = _full_convolution_fft(cand).max()
            if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(temperature, 1e-12)):
                a = cand
                cur_max = cand_max
                temperature *= temp_decay

        else:
            # --- Gaussian jitter on the whole vector ---
            noise = rng.normal(scale=a.mean() * 0.02, size=n)
            cand = np.clip(a + noise, 0.0, None)
            cand = _project_to_simplex(cand, S)
            cand_max = _full_convolution_fft(cand).max()
            if cand_max < cur_max:
                a = cand
                cur_max = cand_max

        # bookkeeping
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # keep temperature from collapsing completely
        temperature = max(temperature, 1e-8)

    return best_a, best_max

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Time‑limited search for a step sequence that minimises the
    evaluation function.  Returns the best sequence (list of floats) found.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # retrieve any previously stored construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – attempt to polish the known construction (if any)
    # ------------------------------------------------------------------
    if base_seq is not None and base_seq.size > 0:
        n0 = base_seq.shape[0]
        a0 = _project_to_simplex(base_seq, np.sqrt(2.0 * n0))
        phase_deadline = min(time.time() + 180.0, end_time)
        seq_opt, score_opt = _local_optimize(a0, phase_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[phase 1] refined length={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many independent restarts with varied lengths
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempts += 1

        # pick a length (mix moderate and larger sizes)
        if rng.random() < 0.6:
            n = rng.integers(1500, 8000)          # moderate
        else:
            n = rng.integers(8000, 35000)         # larger (still fast enough)

        # initialise a feasible vector
        a0 = _initial_candidate(base_seq, n, rng)

        # allocate a short time slice for this restart
        slice_deadline = min(time.time() + 12.0, end_time)

        seq_opt, score_opt = _local_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[global] attempt={attempts} n={n} new best max={best_score:.6f}")

        # occasionally refresh the base_seq with the current best
        if attempts % 20 == 0 and best_seq is not None:
            base_seq = best_seq.copy()

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the black‑box function
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        try:
            external_score = evaluate_sequence(final_seq)
            print(f"[final] external score = {external_score:.6f} "
                  f"(internal max = {best_score:.6f}) length = {len(final_seq)}")
        except Exception:
            print("[final] evaluation failed (bad sequence)")

    # store for possible future runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
