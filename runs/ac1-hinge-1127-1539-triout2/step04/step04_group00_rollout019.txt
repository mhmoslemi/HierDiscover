"""step‑function optimizer

This version improves the previous search by
* using a longer single optimisation run (instead of many short restarts);
* employing Adam‑style adaptive steps for both the smooth‑max phase and the
  sub‑gradient phase;
* increasing the temperature schedule so the smooth approximation approaches the true
  max more closely;
* adding occasional “mass‑swap’’ perturbations to escape plateaus;
* keeping the best vector seen and returning it at the end.

The objective that ``evaluate_sequence`` finally returns is
    max_k (a * a)[k]  (the infinity‑norm of the self‑convolution),
because the sum of the coefficients is normalised to √(2·n).  The algorithm
works entirely on the simplex {a_i≥0, Σa_i=√(2·n)}.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth soft‑max objective.
    g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weighting (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Compute stable soft‑max weights."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w

# Main optimisation routine
# ----------------------------------------------------------------------

        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # 2 * a[j] for i != j, a[i] for i == j (i.e. 2*i == k_max)
        g[mask] = 2.0 * a[j[mask]]
        diag = (2 * idx == k_max) & mask
        g[diag] = a[idx[diag]]            # replace 2*a_i by a_i on the diagonal

        # Adam‑style adaptive step
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / np.sqrt(it)          # base decay
        step = eta / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a, S)

        cur_score = np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score:
                    a = cand
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

    # ------------------------------------------------------------------
    # Return the best sequence found.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal best = {best_score:.6f}  "
          f"length = {len(final_seq)}")
    return final_seq
