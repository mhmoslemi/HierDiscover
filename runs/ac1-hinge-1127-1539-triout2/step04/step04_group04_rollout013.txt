def _subgradient_descent(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    b = _full_convolution_fft(a)
    best_max = np.max(b)
    # initial step size: set as a fraction of average element
    avg = S / n
    step = avg * 0.4  # initial step magnitude
    # schedule: decay factor
    decay = 0.9995
    while time.time() < deadline:
        # compute max and argmax
        b = _full_convolution_fft(a)
        cur_max = np.max(b)
        k = np.argmax(b)  # index of max
        # compute subgradient g = 2 * a_rev shifted
        # g_i = 2*a_{k-i} if 0 <= k-i < n else 0
        indices = np.arange(n)
        partner = k - indices
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[partner[mask]]

        # gradient step (descent)
        a_new = a - step * g
        # guarantee nonnegatives via simplex projection (maintains sum)
        a_new = _project_to_simplex(a_new, S)

        # evaluate
        b_new = _full_convolution_fft(a_new)
        max_new = np.max(b_new)
        if max_new < best_max - 1e-12:
            best_max = max_new
            best_a = a_new.copy()
        # accept if improvement else maybe keep a (or still accept)
        # We accept always for simplicity.
        a = a_new
        # decays
        step *= decay
        # maybe random jitter occasionally
        if rng.random() < 0.005:
            noise = rng.normal(scale=avg*0.05, size=n)
            a = np.clip(a + noise, 0, None)
            a = _project_to_simplex(a, S)
    return best_a, best_max

"""
Random‑parameter search (with a focus on scaled exponential families) +
sub‑gradient descent + Adam refinement + optional mixing with a uniform vector.

The core idea:
* For a vector a with ∑a = √(2·n) the evaluation score equals max(a*a).
* The average of the self‑convolution entries is ≈1, so the theoretical
  optimum is 1.  By using a one‑sided (or symmetric) exponential shape
  with decay λ≈c/n (c≈1.0–2.5) the convolution becomes almost flat and the
  maximum approaches the lower bound.
* The program first explores many random families (including the tuned
  exponential ones), keeps the best candidate and then refines it with a
  sub‑gradient method and an Adam optimiser that works on a soft‑max
  approximation of the ∞‑norm.
* Finally the best vector is blended with the uniform vector to exploit
  any remaining margin.

The whole search respects the 1000 s budget; a small safety margin is
kept so that the harness does not kill the process.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project onto the ℓ₁‑simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """Return (grad, conv, max) for τ·log∑exp(b/τ) where b = a*a."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard max → sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        return g, b, max_b

    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)                # soft‑max weights, shape (2n‑1,)

    # cross‑correlation a ⋆ w
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]        # length n
    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection (minimise soft‑max of convolution)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6               # initial learning rate
    t = 0

    # initialise temperature for soft‑max
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam pre‑projection update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluation
        cur_max_new = np.max(_full_convolution_fft(a_new))

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedule
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        # occasional jitter
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Sub‑gradient descent on the hard max (direct reduction of max entry)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_max = np.max(_full_convolution_fft(a))

    # initial step size proportional to the average entry
    step = (S / n) * 0.6
    decay = 0.9995

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        cur_max = np.max(b)
        k = np.argmax(b)                     # index where max occurs

        # sub‑gradient of max entry (see analysis)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[partner[mask]]

        a_new = a - step * g
        a_new = _project_to_simplex(a_new, S)

        cur_max_new = np.max(_full_convolution_fft(a_new))

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new
        step *= decay

        # tiny random perturbation to escape flat regions
        if rng.random() < 0.004:
            noise = rng.normal(scale=S / n * 0.04, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Blend the current sequence with the uniform distribution (convex combination)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, deadline: float, rng: np.random.Generator):
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    u = np.full(n, S / n)                     # uniform vector (already normalised)
    best_a = a.copy()
    best_max = np.max(_full_convolution_fft(a))

    # systematic scan of mixing coefficients
    for t in np.linspace(0.0, 1.0, 31):
        cand = (1.0 - t) * a + t * u
        cur_max = np.max(_full_convolution_fft(cand))
        if cur_max < best_max:
            best_max = cur_max
            best_a = cand.copy()

    # a few random mixes if time permits
    while time.time() < deadline:
        t = rng.random()
        cand = (1.0 - t) * a + t * u
        cur_max = np.max(_full_convolution_fft(cand))
        if cur_max < best_max:
            best_max = cur_max
            best_a = cand.copy()
        if rng.random() < 0.01:   # break chance to avoid endless loop
            break

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families (including the crucial scaled exponential)
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    typ = rng.choice(
        ["uniform", "exp_scaled", "symexp_scaled", "gaussian_scaled",
         "triangular", "power", "cosine"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp_scaled":
        # λ = c / n,  c ∈ [0.7, 2.5]
        c = rng.uniform(0.7, 2.5)
        lam = c / n
        a = np.exp(-lam * idx)

    elif typ == "symexp_scaled":
        c = rng.uniform(0.7, 2.5)
        lam = c / n
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian_scaled":
        # σ = s·n,  s ∈ [0.04, 0.12]
        s = rng.uniform(0.04, 0.12)
        sigma = s * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    else:
        a = rng.random(n)

    # small jitter for added diversity
    if rng.random() < 0.25:
        a += rng.normal(scale=a.mean() * 0.012, size=n)
        a = np.clip(a, 0.0, None)

    return a
