def solve_lp(f, M):
    n = len(f)
    # Build convolution matrix
    # each diagonal offset i has value f[i], length n
    data = [np.full(n, f[i]) for i in range(n)]

    res = opt.linprog(c=-np.ones(n), A_ub=C, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')

"""
minimise evaluate_sequence()  →  minimise max(a*a)  (sum of a is fixed = √(2·n))

Algorithm
---------
1. Warm‑start with an exponential (or Dirichlet) sequence and project onto the
   ℓ1‑simplex (Σa = √(2·n)).
2. For the current sequence f compute M = max(conv(f,f)).
   Solve the linear programme

        maximise   Σ g_j
        subject to  conv(f,g) ≤ M      (2·n‑1 linear inequalities)
                    g_j ≥ 0, g_j ≤ Σf   (upper bound prevents unboundedness)

   This gives a “partner” g that can be added without increasing the worst
   convolution entry.
3. Rescale g to the correct ℓ1‑norm and replace

        f ← (1‑t)·f + t·g          (t≈0.2)

   and repeat the LP a few times.  M drops monotonically.
4. When the LP phase finishes, finish the refinement with the projected‑
   gradient / random‑swap routine from the previous implementation.
5. Keep the best sequence seen before the time budget expires and return it.

The LP is built with a sparse Toeplitz matrix (one diagonal per element of f)
so that even n≈2500 (≈6 M non‑zeros) fits comfortably in memory.  All steps
run inside the 1000 s budget and print progress.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·Σ_i w_{i+j}·a_i   (gradient of Σ w·conv(a,a) w.r.t. a).
    a : length n, w : length 2·n‑1
    """
    n = a.shape[0]
    # correlation a ⋆ w
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # indices that give Σ_i w_{i+j}·a_i
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Warm‑start helpers
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """
    decreasing exponential a_i ∝ exp(-λ·i)  with λ = k_factor / n,
    projected onto the simplex Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    # exact normalisation (avoids loss of precision for tiny λ)
    c = S * (1 - q) / (1 - q ** n)
    a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (core of the new improvement)
# ----------------------------------------------------------------------


def _solve_lp(f: np.ndarray, M: float, S: float):
    """
    Solve
        maximise   Σ g_j
        subject to  conv(f,g) ≤ M   (2·n‑1 constraints)
                    0 ≤ g_j ≤ S
    Returns (g, success).
    """
    n = f.shape[0]
    # Build sparse Toeplitz matrix C where C[k, j] = f_{k-j} (0 if out‑of‑range)
    # One diagonal per element of f
    data = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)               # diagonal at offset i
    C = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')

    c = -np.ones(n)                      # maximise Σg  → minimise -Σg
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, S) for _ in range(n)]

    try:
        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x, True
        else:
            return None, False
    except Exception:
        return None, False


def _lp_optimize(a0: np.ndarray, t: float, max_iters: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑based update described in the comment above.
    Returns the best sequence found and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a)
        M = np.max(conv)

        # solve LP
        g, ok = _solve_lp(a, M, S)
        if not ok or g is None:
            break

        sum_g = np.sum(g)
        if sum_g < 1e-12:
            break

        # rescale to exact simplex sum and mix
        g = g * (S / sum_g)
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # stop when LP no longer improves the bound
        if cur_score >= M - 1e-9:
            break

    return best_a, best_score

# Local optimiser (gradient + random swaps) – same as before

def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the smoothed max‑norm.
    Returns the updated vector and the exact maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    # smooth soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    a_new = _project_simplex(a_new, np.sqrt(2.0 * n))
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Refines a candidate with gradient steps and cheap mass swaps."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (empirically chosen)
    lr = 0.5
    lr_decay = 0.9999
    alpha = 0.5
    alpha_mul = 1.005
    swap_prob = 0.15
    swap_frac = 0.04
    target_prob = 0.2
    eps_target = 0.03

    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # gradient step
        a, cur_max = _gradient_step(a, lr, alpha)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # random mass‑swap (simulated‑annealing flavour)
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # cheap targeted move aiming at the current maximal index
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            src_idx = np.argmax(contributions)
            i_src = i_range[src_idx]

            tgt_idx = np.argmin(contributions)
            i_tgt = i_range[tgt_idx]

            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # schedule updates
        lr *= lr_decay
        alpha *= alpha_mul

        if it - last_improve > 5000:
            break

    return best_a, best_score

    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of length 1000
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
