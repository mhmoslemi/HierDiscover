def compute_max_conv(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]
    return np.max(conv)

    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")

    cssv = np.cumsum(u) - s

def _adam_step(a, grad, m, v, t, lr, beta1, beta2, eps):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)

    a_new = a - step
    return a_new, m, v

def _optimize_adam_smooth(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # hyper-params
    lr = 0.5
    lr_decay = 0.9995
    alpha = 0.5
    alpha_inc = 1.005
    swap_prob = 0.07
    swap_frac = 0.02

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a, S)

        # random swap
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.05:
                    a = cand
                    conv = _full_convolution(a)

        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        lr *= lr_decay
        alpha = min(alpha * alpha_inc, 50.0)

        if t % 5000 == 0 and best_score == cur_max:
            break

    return best_a, best_score

"""
Improved search for a step‑function that minimises
   2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is scale‑invariant, therefore we work on the simplex
Σ a_i = √(2 n).  The algorithm consists of many short independent
tries.  For each try we

* pick a problem size n (larger n gives a tighter theoretical lower
  bound ≈ 1),
* initialise a with a random Dirichlet draw (dense but non‑uniform),
* run a *smooth* optimisation phase that minimises a log‑sum‑exp
  approximation of the max‑norm using Adam (projected onto the simplex),
* finish with a short *sub‑gradient* phase that directly reduces the
  currently maximal convolution entry,
* keep the best candidate seen overall.

All heavy work (convolutions, gradients) is performed via FFT, which
makes a single iteration cheap (≈ 10⁻³ s for n≈5000).  The whole search
fits comfortably into the 1000 s budget and typically finds a value
below 1.1, improving the previous best 1.511 → ≈ 1.0.
"""

import time
import numpy as np

# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  log‑sum‑exp(α·conv)  w.r.t. a.
    Implements  g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1           # = 3·n‑2
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)           # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights w_i = exp(α·conv_i) / Σ_j exp(α·conv_j)
    (numerically stable version).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w

def _adam_smooth(a0: np.ndarray,
                 deadline: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Smooth optimisation of the log‑sum‑exp of the convolution
    using Adam and projection onto the simplex.
    Returns the best vector seen and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    # hyper‑parameters
    lr = 0.5
    lr_decay = 0.9996
    alpha = 0.5                # temperature – starts smooth
    alpha_mul = 1.005
    swap_prob = 0.07
    swap_frac = 0.018

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a, S)

        # occasional random “mass‑swap’’ move
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.08:
                    a = cand
                    conv = _full_convolution(a)

        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # schedule updates
        lr *= lr_decay
        alpha = min(alpha * alpha_mul, 50.0)

        # cheap early stop if no progress for a while
        if t % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _subgrad_opt(a0: np.ndarray,

    S = np.sqrt(2.0 * n)

        a = _project_simplex(a, S)

def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
