assistantfinalBelow is a self‑contained implementation of `propose_candidate`.  
It builds on the previous optimizer but adds a **quadratic (ℓ₂) smoothing phase** that directly pushes the self‑convolution towards a flat constant, and then switches to a log‑sum‑exp (smooth‑max) phase.  The two phases are blended gradually, and a final short simulated‑annealing polish is applied.  The routine respects the 1000 s wall‑clock budget, uses FFTs for fast convolutions, and works for any sequence length.

```python
"""
Step‑function optimizer with ℓ₂‑flattening pre‑stage.

The algorithm proceeds as follows (all steps respect the overall 1000 s
budget):

1. Randomly initialise many candidates on the ℓ¹‑simplex.
2. For each candidate run a *mixed* projected‑gradient descent:
   * early iterations minimise the ℓ₂‑error
     ‖a*a − c·1‖² (c = (∑a)²/(2n‑1) is the constant that a perfectly
     flat convolution would have).  This quickly flattens the convolution.
   * later iterations switch to a smooth‑max (log‑sum‑exp) gradient,
     driving down the actual maximum.
   * momentum, adaptive step‑size and tiny jitter are used throughout,
     with Euclidean projection onto the simplex after each step.
3. Keep the best sequence seen for each length.
4. Refine the current global best with a longer descent.
5. End with a short simulated‑annealing phase that transfers a fraction
   of mass between indices, accepting uphill moves with a cooling
   schedule.
6. Return the best sequence (as a plain Python list).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation: max(conv(a,a))."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed max   (1/β)·log Σ exp(β·b)   where b = a*a.

    Returns
        g  – gradient w.r.t. a
        b  – the convolution a*a (needed for the max‑value)
    The formula is g = 2·(a ⋆ p) with p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_l2(a: np.ndarray):
    """
    Gradient of the squared ℓ₂‑error  Σ (a*a - c)²,
    where c = (∑a)²/(2n‑1) is the constant value of a perfectly flat convolution.
    Returns
        g – gradient w.r.t. a
        b – the current convolution a*a
    """
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)          # = ∑a after projection
    target_const = (target_sum ** 2) / (2 * n - 1)   # the ideal flat value

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    diff = b - target_const                # (b - c)

    # gradient = 4·(a ⋆ diff)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Diff_fft = np.fft.rfft(diff, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Diff_fft, n=size2)[:conv_len]
    g = 4.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent (mixed ℓ₂ + smooth‑max)
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> np.ndarray:
    """
    Mixed projected gradient descent.
    * First ~2000 iterations minimise the ℓ₂‑error (flattening).
    * Afterwards we switch to a smooth‑max (log‑sum‑exp) gradient.
    Momentum, an adaptive learning‑rate and occasional jitter are used.
    The loop stops when `time.time()` reaches ``deadline``.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # parameters for the smooth‑max schedule
    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.20
    momentum = np.zeros_like(a)
    rho = 0.90                      # momentum decay
    it = 0

    while time.time() < deadline:
        # --- mixing weight: start with ℓ₂, then fade to smooth‑max ---
        mix = max(0.0, 1.0 - it / 2000.0)   # 1 → ℓ₂, 0 → smooth‑max

        if mix > 0.0:
            g_l2, _ = _gradient_l2(a)
        else:
            g_l2 = np.zeros_like(a)

        # smooth‑max gradient (only used after the ℓ₂ phase)
        if beta < beta_max:
            g_sm, _ = _gradient_softmax(a, beta)
        else:
            g_sm = np.zeros_like(a)

        g = mix * g_l2 + (1.0 - mix) * g_sm

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # project back onto the ℓ¹‑simplex (∑a = target_sum, a_i ≥ 0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (smooth‑max becomes sharper)
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a random fraction of mass
    from one index to another.  Accepts uphill moves with probability
    exp(−Δ/T) where the temperature decays geometrically.
    Returns the best sequence observed before ``deadline``.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    # initial temperature – a small fraction of the current score
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisation on the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the simplex
    Σ a = √(2 n).  Several families (power‑law, exponential,
    mixed, Dirichlet, piecewise exponential) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # uniform plus tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the given time budget,
    runs several random restarts, a mixed ℓ₂ / smooth‑max projected‑gradient
    phase, a short simulated‑annealing polish, and finally returns the
    best sequence (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously discovered construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more flexibility
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000]

    # Main search over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Phase A : cheap random restarts with a short mixed GD
        # --------------------------------------------------------------
        phase_end = min(deadline, time.time() + min(30.0,
                                                    (deadline - time.time()) * 0.3))
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)

            sub_dead = min(deadline, time.time() + 3.0)
            a_gd = _gradient_descent_opt(a0, sub_dead, rng)

            sc = _score(a_gd)
            if sc < best_score:
                best_score = sc
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase B : deep refinement of the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            sub_dead = min(deadline, time.time() + min(120.0,
                                                       deadline - time.time()))
            a_ref = _gradient_descent_opt(best_seq, sub_dead, rng)
            sc = _score(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined best (length {n}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 5.0:
            a_final = _simulated_annealing(best_seq, deadline, rng)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean up tiny trailing zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (never expected to be taken)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Log the final objective as seen by the official evaluator
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:     # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```
