"""
Optimisation of a non‑negative step function for the black‑box
evaluate_sequence().

The evaluator returns                2·n·max(conv) / (sum(a))**2.
If we keep the ℓ1‑norm exactly √(2·n) the denominator cancels and the
objective reduces to the maximal entry of the linear convolution
a∗a.  The goal is therefore to find a vector a≥0 with Σa=√(2·n) that makes
the convolution as flat as possible.

Algorithm
---------

1. **Initial pool** – generate a few hundred candidate vectors of
   different lengths (2000–8000) and shapes (exponential, Gaussian,
   cosine‑bump, uniform, random Dirichlet).  Every candidate is projected
   onto the simplex Σa=√(2·n).

2. **Projected Adam on a smooth‑max** – the hard max is replaced by a
   soft‑max with temperature α.  For a given α the objective is
   `∑_k w_k (a∗a)_k` where `w_k = exp(α·c_k)/∑_j exp(α·c_j)`.  Its
   gradient can be written as a convolution and is computed by FFTs.
   An Adam‑style adaptive step‑size is used, followed by projection onto
   the simplex.

   The temperature α is annealed slowly from a small value (≈0.2) to a
   larger one (≈5.0) so that the optimisation first moves many
   coefficients simultaneously and later concentrates on the current
   maximal entry.

3. **Targeted mass‑swap moves** – after each batch of Adam steps we
   locate the index `k_max` where the (real) convolution is maximal.
   We move a small fraction of mass from the index contributing the
   largest product `a_i·a_{k_max‑i}` to the index with the smallest
   such product.  This cheap move strictly reduces the current maximal
   convolution entry (or is accepted with a Metropolis probability).

4. **Random swaps** – occasional simulated‑annealing style swaps of a
   random amount of mass between two indices help to escape shallow
   plateaus.

5. **Best‑ever bookkeeping** – the best vector seen (smallest real
   max‑conv) is stored.  The whole procedure runs until the 1000 s time
   budget expires; the final candidate is returned as a plain list of
   floats.

The code is deliberately vectorised (NumPy + FFT) and fits comfortably
into the 1‑CPU budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k (a∗a)_k  w.r.t. a.
    It equals 2 * (reverse(w) ⋆ a), where ⋆ denotes the linear
    convolution (correlation).  Implemented via FFT.
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # extract the slice that corresponds exactly to Σ_i w_{i+j}·a_i
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(alpha * conv_k)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------


def _exponential_seq(n: int, rate: float) -> np.ndarray:
    """Decreasing exponential a_i = C·exp(-rate·i) projected onto simplex."""
    S = np.sqrt(2.0 * n)
    i = np.arange(n)
    a = np.exp(-rate * i)
    a = a / a.sum() * S
    return _project_simplex(a, S)


def _gaussian_seq(n: int, sigma_rel: float = 0.2) -> np.ndarray:
    """Symmetric Gaussian centred at middle, projected onto simplex."""
    S = np.sqrt(2.0 * n)
    mid = (n - 1) / 2.0
    sigma = sigma_rel * n
    i = np.arange(n)
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a = a / a.sum() * S
    return _project_simplex(a, S)


def _cosine_bump_seq(n: int) -> np.ndarray:
    """Non‑negative cosine bump (half‑wave) centred, projected onto simplex."""
    S = np.sqrt(2.0 * n)
    i = np.arange(n)
    a = np.cos(np.pi * (i - (n - 1) / 2) / n)
    a = np.maximum(a, 0.0)
    a = a / a.sum() * S
    return _project_simplex(a, S)


def _uniform_seq(n: int) -> np.ndarray:
    """Uniform sequence with the required ℓ1‑norm."""
    S = np.sqrt(2.0 * n)
    a = np.full(n, S / n)
    return a


def _random_dirichlet_seq(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw (α=1) – automatically non‑negative."""
    S = np.sqrt(2.0 * n)
    a = rng.dirichlet(np.ones(n))
    a = a * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   *,
                   lr0: float = 0.8,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   alpha_start: float = 0.2,
                   alpha_end: float = 5.0,
                   alpha_mul: float = 1.005,
                   swap_prob: float = 0.12,
                   target_prob: float = 0.22,
                   eps_target: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of the soft‑max of the convolution.
    Returns the best sequence encountered and its *exact* maximal convolution
    entry (i.e. the value that will be fed to evaluate_sequence).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # bookkeeping
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # initialise temperature
    alpha = alpha_start

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- compute smooth‑max gradient ----
        conv = _full_convolution(a)
        w = _softmax_weights(conv, alpha)
        grad = _conv_with_weights(a, w)               # ∇_a Σ w_k conv_k

        # Adam update (projected)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a_new = a - lr * m_hat
        a = _project_simplex(a_new, S)

        # ---- keep exact best ----
        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # ---- occasional random mass‑swap (exploration) ----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = rng.random() * 0.05 * a[i]    # up to 5 % of source mass
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis acceptance
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # ---- cheap targeted move on the current maximum ----
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            # indices that contribute to conv[k_max]
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contrib = a[i_range] * a[j_range]

            src_idx = i_range[np.argmax(contrib)]
            tgt_idx = i_range[np.argmin(contrib)]

            if a[src_idx] > 1e-12 and src_idx != tgt_idx:
                delta = eps_target * a[src_idx] * rng.random()
                cand = a.copy()
                cand[src_idx] -= delta
                cand[tgt_idx] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # ---- temperature annealing ----
        alpha = min(alpha_end, alpha * alpha_mul)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence that minimises evaluate_sequence().
    The routine explores many lengths and shapes, refines each candidate
    with a projected Adam optimiser and occasional targeted moves,
    and finally returns the best sequence discovered before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0      # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) pool of diverse initial vectors (cheap warm start)
    # ------------------------------------------------------------------
    init_shapes = [
        _exponential_seq,
        _gaussian_seq,
        _cosine_bump_seq,
        _uniform_seq,
        _random_dirichlet_seq,
    ]

    # pick a handful of lengths (biased towards larger values)
    length_candidates = [2000, 3000, 4000, 5000, 6000, 8000]

    for n in length_candidates:
        if time.time() >= deadline:
            break
        for shape_fn in init_shapes:
            if time.time() >= deadline:
                break
            # some shapes need an extra random parameter
            if shape_fn is _exponential_seq:
                rate = rng.uniform(0.0005, 0.0015) * n   # scale with n
                a0 = shape_fn(n, rate)
            elif shape_fn is _gaussian_seq:
                sigma_rel = rng.uniform(0.12, 0.30)
                a0 = shape_fn(n, sigma_rel)
            elif shape_fn is _random_dirichlet_seq:
                a0 = shape_fn(n, rng)
            else:
                a0 = shape_fn(n)

            # quick evaluation (exact max conv)
            cur_score = np.max(_full_convolution(a0))
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a0.copy()
                print(f"[init] n={n:4d} shape={shape_fn.__name__:15s} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2) local refinement phases – run until budget runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose length: favour previously successful lengths (randomly jitter)
        if best_seq is not None and rng.random() < 0.55:
            n = best_seq.size
        else:
            n = int(rng.integers(2500, 9001))

        # initialise either from the current best (plus jitter) or a fresh random shape
        if best_seq is not None and best_seq.size == n and rng.random() < 0.5:
            a0 = best_seq.copy()
            jitter = rng.normal(scale=0.001 * a0.mean(), size=n)
            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))
        else:
            # random shape choice
            sh = rng.choice(init_shapes)
            if sh is _exponential_seq:
                rate = rng.uniform(0.0005, 0.0015) * n
                a0 = sh(n, rate)
            elif sh is _gaussian_seq:
                sigma_rel = rng.uniform(0.12, 0.30)
                a0 = sh(n, sigma_rel)
            elif sh is _random_dirichlet_seq:
                a0 = sh(n, rng)
            else:
                a0 = sh(n)

        # allocate a slice of time (≈ 25 s) for this candidate
        slice_deadline = min(deadline, time.time() + 25.0)

        a_opt, score_opt = _adam_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global] new best n={n:4d} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final sanity check via the external black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # Very safe fallback – uniform length‑1000 sequence
        n = 1000
        best_seq = _uniform_seq(n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)      # type: ignore  # provided externally
    print(f"[final] external score = {external:.6f}   length = {len(final_seq)}")
    # expose for possible next‑run warm‑start
    globals()["height_sequence_1"] = final_seq
    return final_seq
