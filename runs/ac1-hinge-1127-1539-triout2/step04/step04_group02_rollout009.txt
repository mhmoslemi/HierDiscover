"""
Propose a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

using a three‑stage optimisation:

1. **Broad initial sweep** – several parametric “U‑shapes”, “edge‑heavy”
   shapes and a few random Dirichlet vectors are projected onto the
   ℓ₁‑simplex {x≥0, Σx = √(2 n)}.  The best candidate is kept.

2. **Smoothed Adam descent** – minimise a log‑sum‑exp surrogate of the
   maximum.  The surrogate gradient is computed by a single FFT‑based
   convolution.  Adam with decaying learning‑rate and a slowly decreasing
   smoothing parameter τ guides the iterates, while occasional random
   pair‑transfers keep the search from stagnating.

3. **True‑max refinement + polishing** – replace the soft‑max weights by
   a uniform mask over (approximately) maximal lags, continue with Adam,
   and finish with a burst of tiny random pair‑transfers.

All phases respect the 1000 s time budget, and the best vector found is
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# Utility functions
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        # all entries would become zero – fallback to uniform distribution
        return np.full_like(v, target / len(v))

    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a candidate shape of length n and project onto the ℓ₁‑simplex.
    Supported ``kind`` values:

        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right,
        exponential_left, exponential_right,
        edge_power, edge_exponential, edge_linear

    ``param`` is a shape parameter (exponent, rate, …).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]
    eps = 1e-8

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + eps
    elif kind == "inverse":
        raw = 1.0 / (eps + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    elif kind == "edge_power":
        raw = (idx + 1) ** (-param) + (n - idx) ** (-param)
    elif kind == "edge_exponential":
        raw = np.exp(-param * idx) + np.exp(-param * (n - 1 - idx))
    elif kind == "edge_linear":
        raw = 1.0 / (idx + 1 + eps) + 1.0 / (n - idx + eps)
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())


def weighted_gradient(a: np.ndarray, w: np.ndarray, factor: float) -> np.ndarray:
    """
    Compute ∇_a  (factor * Σ_k w_k * a_{k-i})  efficiently via FFT.
    ``w`` is a vector of length (2n‑1) – either a softmax weight vector
    (sums to 1) or a mask over maximal lags (sums to cnt).  ``factor`` is
    either 2/τ (smooth surrogate) or 2/cnt (true sub‑gradient).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = factor * conv_full[n - 1 : n - 1 + n]
    return grad
