def _hill_climbing(a, b, deadline):
    n = a.shape[0]
    best_a = a.copy()
    best_b = b.copy()
    best_score = b.max()
    rng = np.random.default_rng()
    while time.time() < deadline:
        # Choose source index i weighted by a_i (higher probability for larger values)
        # can sample from a distribution proportionally to a_i.
        probs = a / a.sum()
        i = rng.choice(n, p=probs)  # heavy mass transfers
        # Choose destination j from uniform random steps or choose low mass.
        # We'll pick j uniformly among indices where a_j is small.
        j_candidates = np.where(a < a.mean()/2)[0]
        if j_candidates.size == 0:
            j = rng.integers(0, n)
        else:
            j = rng.choice(j_candidates)
        if i == j: continue
        # Determine max transferable mass
        max_delta = a[i] * 0.5  # move at most half of source
        if max_delta < 1e-12: continue
        delta = rng.random() * max_delta
        # compute new b via slice updates (no copy)
        # apply on copy of b for testing
        b_new = b.copy()
        # apply contributions
        b_new[i:i+n] -= 2.0 * delta * a   # using old a before change
        b_new[j:j+n] += 2.0 * delta * a
        # term C:
        k2i = 2*i
        k2j = 2*j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta*delta
        if k2j < b_new.size:
            b_new[k2j] += delta*delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        # compute new score
        new_score = b_new.max()
        if new_score < best_score:
            # accept move
            a[i] -= delta
            a[j] += delta
            b = b_new
            best_score = new_score
            best_a = a.copy()
            best_b = b.copy()
            # optionally print progress
            print(f"[{time.time():.1f}] hill_climb improved: {best_score:.6f}")

def _hill_climbing(a, b, deadline):
    n = a.shape[0]
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()
    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9997
    it = 0
    while time.time() < deadline:
        # sample source i proportional to a (higher)
        if cur_score == 0:
            break
        # sample i
        probs = a / a.sum()
        i = rng.choice(n, p=probs)
        if a[i] < 1e-12:
            continue
        # sample j uniform
        j = rng.integers(0, n)
        if i == j:
            continue
        # compute delta
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        # compute new b via incremental update
        b_new = b.copy()
        # slice updates
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[j:j+n] += 2.0 * delta * a
        k2i = 2*i
        k2j = 2*j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta*delta
        if k2j < b_new.size:
            b_new[k2j] += delta*delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        new_score = b_new.max()
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Simulated annealing acceptance
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True
        if accept:
            # commit
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")
        # update temperature
        T *= cooling
        it += 1
    return best_a, best_b
```

"""
Step‑function optimizer (improved).

The algorithm works in three stages:

1. **Exploration** – for several lengths (up to ~15 k) we create a random
   non‑negative vector on the ℓ¹‑simplex Σa = sqrt(2 n) and run a long
   projected‑gradient descent on a smooth log‑sum‑exp approximation of the
   max‑convolution.  The gradient uses FFTs, momentum and an annealed
   inverse temperature β.

2. **Mass‑transfer hill‑climbing** – starting from the best vector for a
   given length we keep the full linear convolution `b = a*a` and repeatedly
   move a random fraction of mass from a large entry to a small entry.
   The change in `b` can be updated in O(n) time using the analytical
   formula
        Δbₖ = 2·δ·(a_{k‑j} – a_{k‑i})   (+ δ² at 2i,2j and –2·δ² at i+j)
   which allows thousands of cheap moves.  An annealed‐temperature rule
   accepts occasional uphill steps.

3. **Polishing** – a final hill‑climb (or a short simulated‑annealing run)
   uses the remaining time budget.

All numbers are kept non‑negative and are projected onto the simplex after
each stochastic move, so the official ``evaluate_sequence`` sees a valid
step‑function.  The routine respects the given wall‑clock budget (default
1000 s) and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                          # reversed a
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Uses momentum, an annealed inverse temperature β and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

# Random initialisers

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the simplex Σa = sqrt(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing with mass transfers.
    a0 – initial sequence (ℓ¹‑normalised)
    b0 – its full convolution a0*a0
    The loop runs until ``deadline`` and returns the best pair (a, b)
    observed.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        # pick a source index proportional to its mass
        if a.sum() == 0:
            break
        probs = a / a.sum()
        i = rng.choice(n, p=probs)

        if a[i] < 1e-14:
            continue

        # pick a destination uniformly (could be biased towards small entries)
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount of mass to move (up to 50 % of source)
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---------- compute new convolution ----------
        b_new = b.copy()
        # contribution from the -δ at i
        b_new[i:i + n] -= 2.0 * delta * a
        # contribution from the +δ at j
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---------- acceptance ----------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b
