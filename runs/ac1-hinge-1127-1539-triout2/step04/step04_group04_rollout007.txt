"""
Propose a step‑function (non‑negative vector) that minimises
    2·n·max(a∗a)/(sum a)²
using a two‑phase strategy:

1️⃣  Random parametric sampling (exponential, cosine, “U‑shaped” exponential,
    power‑law, etc.) to obtain a decent starting vector.

2️⃣  Iterative linear‑programming refinement inspired by the classic
    f←(1−t)f+t g update:
      – Given the current vector f, compute M = max(f∗f).
      – Solve the LP   maximise Σ b_j
                        s.t.   b_j ≥ 0  and  (f∗b)_k ≤ M  ∀k .
        The solution b is the “most massive” vector that does not increase the
        convolution peak beyond M.
      – Rescale b to satisfy the ℓ₁‑simplex Σ b = √(2·n) and mix
        f←(1−t)f + t·b (t starts at 0.25 and slowly decays).
      – Project the mixture back onto the simplex.
      – Keep the best vector seen so far.

The LP is solved with CVXPY/ECOS; a full convolution is obtained via a
real FFT for speed.  Multiple restarts are performed if the process stalls.
The final vector is returned and also cached in the global variable
`height_sequence_1` for possible reuse by later runs.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the ℓ₁‑simplex of sum S (dense, O(n log n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Random parametric families (including a “U‑shaped” exponential)
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector from a randomly chosen parametric family."""
    typ = rng.choice(
        ["uniform", "exp", "symexp", "Uexp", "gaussian", "triangular",
         "power", "cosine", "cosineU"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":                               # peak in the centre
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "Uexp":                                 # high at the edges
        lam = rng.uniform(0.001, 0.15)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "cosineU":                               # U‑shaped raised cosine
        a = (1.0 - np.cos(np.pi * idx / (n - 1))) / 2.0
        a = np.clip(a, 0.0, None)

    else:
        a = rng.random(n)

    # occasional tiny jitter (helps diversity)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# Linear‑programming step: maximise Σb under (f∗b)≤M
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, S: float, solver: str = "ECOS") -> np.ndarray:
    """
    Given the current vector f (length n) and its convolution bound M,
    solve the LP
        maximise Σ b
        s.t.    b ≥ 0,
                (f∗b)_k ≤ M   for all k .
    The returned b is rescaled to satisfy Σ b = S.
    """
    n = f.shape[0]
    # Build convolution matrix C such that C @ b = f∗b (full length 2n‑1)
    C = np.zeros((2 * n - 1, n), dtype=float)
    for i in range(n):
        C[i:i + n, i] = f
    # cvxpy model
    b = cp.Variable(n, nonneg=True)
    constraints = [C @ b <= M + 1e-12]             # tiny slack for numerics
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        prob.solve(solver=solver, warm_start=False, max_iters=5000, verbose=False)

    if b.value is None:
        # fall back to a uniform simplex point
        b_val = np.full(n, S / n)
    else:
        b_val = np.asarray(b.value).reshape(-1)

    b_val = np.clip(b_val, 0.0, None)
    sum_b = b_val.sum()
    if sum_b > 0:
        b_val = b_val * (S / sum_b)
    return b_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height vector that minimises the evaluation
    function within the allocated time.  The routine combines random
    sampling with an LP‑based refinement loop.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0                     # keep a short safety cushion
    deadline = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # Initialise with the previously found construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_max = np.inf
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = seq0.shape[0]
            S0 = np.sqrt(2.0 * n0)
            seq0 = _project_to_simplex(seq0, S0)
            cur_max = np.max(_full_convolution_fft(seq0))
            best_seq = seq0.copy()
            best_max = cur_max
            print(f"[init] imported previous solution: n={n0}, max={best_max:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # If nothing usable, generate a fresh random start
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = rng.integers(800, 2000)
        a_raw = _random_distribution(n0, rng)
        S0 = np.sqrt(2.0 * n0)
        best_seq = _project_to_simplex(a_raw, S0)
        best_max = np.max(_full_convolution_fft(best_seq))
        print(f"[init] random start: n={n0}, max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – quick random search (≈20 % of the remaining budget)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.20
    attempts = 0
    while time.time() < phase1_deadline:
        attempts += 1
        n = rng.integers(800, 2000)
        a_raw = _random_distribution(n, rng)
        S = np.sqrt(2.0 * n)
        a = _project_to_simplex(a_raw, S)
        cur_max = np.max(_full_convolution_fft(a))
        if cur_max < best_max:

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based iterative refinement
    # ------------------------------------------------------------------
    t = 0.25                      # initial mixing weight

        new_seq = _project_to_simplex(new_seq, S)

        if new_max < best_max - 1e-7:          # worthwhile improvement
            best_max = new_max
            best_seq = new_seq.copy()

    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "

    )

    return final_seq
