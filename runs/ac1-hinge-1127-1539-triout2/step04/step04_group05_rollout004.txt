def _logsumexp_gradient(a, tau):
    # compute convolution b = a*a
    b = _full_convolution_fft(a)
    # compute log-sum-exp safely
    max_b = np.max(b)
    # use shift for stability: w_i = exp((b_i - max_b)/tau)
    exp_vals = np.exp((b - max_b) / tau)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp  # shape (2n-1)
    # compute gradient: g = 2 * conv(w, a_rev)  (only the central part of length n)
    a_rev = a[::-1]
    # conv w * a_rev length (len(w) + n -1)
    conv_len = len(w) + len(a) - 1
    f_len = _next_pow2(conv_len)
    W = np.fft.rfft(w, n=f_len)
    A_rev = np.fft.rfft(a_rev, n=f_len)
    conv_result = np.fft.irfft(W * A_rev, n=f_len)[:conv_len]
    # central slice corresponds to indices n-1 to n-1+n-1 (since convolution offset)
    g = 2.0 * conv_result[len(a)-1: len(a)-1 + len(a)]  # match shape n
    return max_b, g

def _adam_optimize(a0, S, tau_schedule, max_iter_per_tau, deadline, rng):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_max = _full_convolution_fft(a).max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0.1 # learning rate maybe dynamic.

    t = 1
    for tau in tau_schedule:
        for iteration in range(max_iter_per_tau):
            if time.time() >= deadline:
                break
            max_b, g = _logsumexp_gradient(a, tau)
            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)

"""
Step‑function optimizer for the L∞ self‑convolution objective.

The evaluation function reduces (for sequences having ℓ₁‑norm √(2·n))
to the maximal entry of the full self‑convolution.
We minimise this value by a continuation method:
  * a smooth surrogate (log‑sum‑exp) of the max is minimised with
    Adam (projected onto the simplex {x≥0, Σx = √(2·n)}).
  * the temperature τ is decreased geometrically, sharpening the surrogate.
  * many independent restarts with different lengths and initialisations
    (random, scaled best‑known, and a 1/√(i) shape) explore the landscape.
  * a short sub‑gradient refinement finishes each run.
The whole search runs for the allotted budget (≤ 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = S}."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - S
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    N = 2 * n - 1
    f_len = _next_pow2(N)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:N]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of seq to length new_n."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _logsumexp_and_grad(a: np.ndarray, tau: float):
    """
    Returns (max_b, g) where max_b = max(a*a) and g = ∇_{a} Lτ,
    Lτ(a) = log Σ_i exp( (a*a)_i / τ ).
    """
    n = a.shape[0]
    # self‑convolution
    b = _full_convolution_fft(a)
    max_b = b.max()

    # stable soft‑max weights
    shift = max_b           # for numerical stability
    exp_vals = np.exp((b - shift) / tau)
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # gradient = 2 * (w ⋆ a_rev)  (central slice length n)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    W = np.fft.rfft(w, n=f_len)
    A_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(W * A_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return max_b, g


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   S: float,
                   tau_schedule,
                   max_iter_per_tau: int,
                   deadline: float,
                   rng: np.random.Generator):
    """
    Projected Adam on the log‑sum‑exp surrogate.
    Returns (best_a, best_max_b) discovered before deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # bookkeeping
    best_a = a.copy()
    best_max = _full_convolution_fft(a).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 1
    base_lr = 0.2                     # base learning rate

    for tau in tau_schedule:
        for _ in range(max_iter_per_tau):
            if time.time() >= deadline:
                return best_a, best_max
            # compute surrogate value and gradient
            cur_max, grad = _logsumexp_and_grad(a, tau)

            # Adam update (pre‑projection)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr / (np.sqrt(v_hat) + eps)

            a = a - step * m_hat
            # project back onto the simplex
            a = _project_to_simplex(a, S)

            # record maximal convolution entry
            cur_max = _full_convolution_fft(a).max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
            t += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Short sub‑gradient refinement (hard max)
# ----------------------------------------------------------------------
def _subgradient_refine(a: np.ndarray,
                        S: float,
                        max_steps: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Exact sub‑gradient descent on the hard max, using a few line‑search
    scalings. Returns the refined vector and its max‑convolution.
    """
    best_a = a.copy()
    best_max = _full_convolution_fft(a).max()
    eta = 0.5  # base step size for sub‑gradient moves

    for _ in range(max_steps):
        if time.time() >= deadline:
            break
        # compute sub‑gradient of the active max entries
        n = a.shape[0]
        b = _full_convolution_fft(a)
        max_b = b.max()
        mask = (b == max_b).astype(float)          # (2n‑1,)
        a_rev = a[::-1]
        # convolution mask⋆a_rev
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        M = np.fft.rfft(mask, n=f_len)
        A_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(M * A_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

        improved = False
        for factor in (0.5, 1.0, 1.5):
            cand = a - eta * factor * g
            cand = _project_to_simplex(cand, S)
            cand_max = _full_convolution_fft(cand).max()
            if cand_max < best_max - 1e-12:
                a = cand
                best_max = cand_max
                best_a = cand.copy()
                improved = True
                break
        if not improved:
            # accept a small worsening with tiny probability – helps escape plateaus
            if rng.random() < 0.02:
                cand = a - eta * g
                cand = _project_to_simplex(cand, S)
                a = cand
                best_max = _full_convolution_fft(cand).max()
                best_a = cand.copy()
        # optional tiny decrease of step size
        eta *= 0.98

    return best_a, best_max

# Candidate proposal – entry point for the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """Run a time‑limited search for a non‑negative step sequence that
    minimises the evaluation function (max self‑convolution)."""

    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Load any previously discovered construction
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Helper to create a feasible start vector of length n
    # ------------------------------------------------------------------
    def _make_start(n):
        S = np.sqrt(2.0 * n)
        mode = rng.random()
        if base_seq is not None and mode < 0.3:
            a = _scale_sequence(base_seq, n)
        elif mode < 0.6:
            # deterministic 1/√ shape (decreasing)
            idx = np.arange(n, dtype=float) + 1.0
            a = 1.0 / np.sqrt(idx)
        else:
            a = rng.random(n)                       # Dirichlet(1)
        a = _project_to_simplex(a, S)
        return a

    # ------------------------------------------------------------------
    # Phase 1 – polish existing construction (if any)
    # ------------------------------------------------------------------
    if base_seq is not None and base_seq.size > 0:
        n0 = base_seq.shape[0]
        S0 = np.sqrt(2.0 * n0)
        a0 = _project_to_simplex(base_seq.astype(float), S0)
        phase_deadline = min(time.time() + 180.0, end_time)
        # simple Adam run (few taus) followed by a short sub‑gradient polish
        init_max = _full_convolution_fft(a0).max()
        tau_sched = [max(1e-6, init_max * f) for f in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01)]
        a_opt, max_opt = _adam_optimize(a0, S0,
                                         tau_schedule=tau_sched,
                                         max_iter_per_tau=200,
                                         deadline=phase_deadline,
                                         rng=rng)
        # final sub‑gradient polish
        a_opt, max_opt = _subgradient_refine(a_opt, S0,
                                             max_steps=30,
                                             deadline=phase_deadline,
                                             rng=rng)
        score_opt = evaluate_sequence(a_opt.tolist())
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[phase 1] refined length={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many independent restarts with varied lengths
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # pick a length (moderate or large)
        if rng.random() < 0.65:
            n = rng.integers(800, 6000)          # moderate
        else:
            n = rng.integers(6000, 25000)        # larger
        S = np.sqrt(2.0 * n)

        a0 = _make_start(n)

        # allocate a slice of the remaining budget
        slice_deadline = min(time.time() + 12.0, end_time)

        # initialise τ‑schedule based on the starting max‑convolution
        start_max = _full_convolution_fft(a0).max()
        tau_sched = [max(1e-6, start_max * f) for f in (1.0, 0.75, 0.5, 0.25,
                                                        0.1, 0.05, 0.02, 0.01)]

        # Adam optimisation
        a_opt, max_opt = _adam_optimize(a0, S,
                                         tau_schedule=tau_sched,
                                         max_iter_per_tau=200,
                                         deadline=slice_deadline,
                                         rng=rng)

        # short sub‑gradient fine‑tuning
        a_opt, max_opt = _subgradient_refine(a_opt, S,
                                             max_steps=30,
                                             deadline=slice_deadline,
                                             rng=rng)

        # evaluation (the external score equals max_opt for properly normalised vectors)
        ext_score = evaluate_sequence(a_opt.tolist())
        if ext_score < best_score:
            best_score = ext_score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best score={best_score:.6f} "
                  f"(max={max_opt:.6f})")
        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – attempt a final upscale if time permits
    # ------------------------------------------------------------------
    if time.time() < end_time and best_seq is not None:
        # try doubling the length (or go up to 100 k)
        n_big = min(100_000, best_seq.shape[0] * 2)
        S_big = np.sqrt(2.0 * n_big)
        a_big = _scale_sequence(best_seq, n_big)
        a_big = _project_to_simplex(a_big, S_big)

        upscale_deadline = min(time.time() + 30.0, end_time)
        start_max = _full_convolution_fft(a_big).max()
        tau_sched = [max(1e-6, start_max * f) for f in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01)]

        a_big, max_big = _adam_optimize(a_big, S_big,
                                         tau_schedule=tau_sched,
                                         max_iter_per_tau=150,
                                         deadline=upscale_deadline,
                                         rng=rng)
        a_big, max_big = _subgradient_refine(a_big, S_big,
                                             max_steps=20,
                                             deadline=upscale_deadline,
                                             rng=rng)
        big_score = evaluate_sequence(a_big.tolist())
        if big_score < best_score:
            best_score = big_score
            best_seq = a_big.copy()
            print(f"[upscale] length={n_big} improved best score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        best_score = evaluate_sequence(final_seq)
    else:
        final_seq = best_seq.tolist()
        # sanity – the external evaluation should equal the internal max
        external = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external:.6f} "
              f"(internal max ≈ {best_score:.6f}) length = {len(final_seq)}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
