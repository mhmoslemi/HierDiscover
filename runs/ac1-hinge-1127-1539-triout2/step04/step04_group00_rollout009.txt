    new_seq = np.interp(new_x, old_x, seq)
    # Ensure non-negative:
    new_seq = np.maximum(new_seq, 0.0)
    # Renormalize to sum 1
    s = new_seq.sum()
    if s == 0:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq = new_seq / s
    return new_seq

"""
Simulated‑annealing step‑function optimiser.

The goal is to minimise the value returned by ``evaluate_sequence``:
    2·n·max(conv(a,a)) / (sum(a)**2)

The function is homogeneous, therefore we keep the L¹‑norm fixed to
√(2·n) (the “simplex” constraint).  The optimiser works by random
mass‑transfer moves (a small fraction of a[i] is moved to a[j]) and
accepts them with a Metropolis probability that gradually cools.
A few seconds are spent on each candidate length; the best vector
found over all lengths is finally refined for the remaining time.

The algorithm is deliberately simple, fast and free of closures – all
helpers are top‑level functions.  It can escape local minima via the
temperature schedule and it benefits from re‑using the best known
construction (``height_sequence_1``) after resizing it to a new length.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: maximum entry of the linear convolution a * a
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a, a)) using an FFT for speed."""
    n = a.shape[0]
    m = 2 * n - 1
    # length for FFT: next power of two (real FFT)
    fft_len = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    # linear convolution = inverse FFT of product (not conjugate)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:m]
    return float(conv.max())


# ----------------------------------------------------------------------
# Resize a probability vector (sum = 1) to a different length
# ----------------------------------------------------------------------


def _resize_prob(p: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector ``p`` to size ``new_n``.
    The output is again a probability vector (non‑negative, sum = 1).
    """
    old_n = p.shape[0]
    if old_n == new_n:
        return p.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_p = np.interp(new_x, old_x, p)
    new_p = np.maximum(new_p, 0.0)
    s = new_p.sum()
    if s == 0.0:
        new_p = np.full(new_n, 1.0 / new_n)
    else:
        new_p /= s
    return new_p


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser for a fixed length
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple (1+1) SA that preserves the L¹‑norm.
    Returns the best vector seen and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule
    T = max(0.01, cur_val * 0.2)      # start a fraction of the current value
    T_min = 1e-8
    decay = 0.9995                    # per iteration decay
    max_frac = 0.3                    # max fraction of a[i] to move
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- random mass‑transfer move ----
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * max_frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = _max_convolution(cand)

        # ---- Metropolis acceptance ----
        if cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / (T + 1e-12)):
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
                print(f"[SA] it={it} new best={best_val:.6f}")

        # ---- temperature decay ----
        T = max(T_min, T * decay)

        # ---- occasional large perturbation to keep diversity ----
        if it % 5000 == 0:
            # replace a small random fraction of entries by fresh Dirichlet noise
            mask = rng.random(n) < 0.02
            if mask.any():
                noise = rng.random(mask.sum())
                noise = noise / noise.sum() * np.sqrt(2.0 * n) * 0.05
                # preserve sum by adjusting the rest
                rest = a[~mask]
                rest_sum = rest.sum()
                rest = rest / rest_sum * (np.sqrt(2.0 * n) - noise.sum())
                a = np.empty_like(a)
                a[mask] = noise
                a[~mask] = rest
                cur_val = _max_convolution(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[SA] it={it} post‑perturb best={best_val:.6f}")

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------

    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        # Re‑normalise precisely to the simplex (protects against tiny drift)
        best_seq = best_seq / best_seq.sum() * np.sqrt(2.0 * n)
        seq, score = _simulated_annealing(best_seq, deadline, rng)
        if score < best_score:
            best_seq = seq
            best_score = score
            print(f"[final‑refine] improved best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Prepare the final answer for the external harness
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]   # fallback – should never happen
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
