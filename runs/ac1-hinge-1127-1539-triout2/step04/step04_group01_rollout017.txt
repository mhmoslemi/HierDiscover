def _adam_descent(a0, deadline):
    a = a0.copy()
    target_sum = np.sqrt(2.0 * a.size)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    beta_max = 32768.0
    beta_factor = 1.5
    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # step size: adapt based on target_sum
        # use base = target_sum * 0.1
        base_lr = target_sum * 0.1
        lr = base_lr / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)
        if t % 500 == 0:
            # jitter
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum*0.001/(beta+1.0), size=a.size)

            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
    return a

assistantfinal**Solution Overview**

The goal is to minimise  

```
evaluate_sequence(a) = 2·n·max(conv(a,a)) / (∑a)²
```

Scaling `a` does not change the value, therefore we normalise every candidate to  

```
∑ a_i = √(2·n)          (ℓ¹‑simplex)
```

and minimise simply `max(conv(a,a))`.  
The problem is non‑convex and the objective is the ∞‑norm of a quadratic map, so a combination of stochastic search and first‑order optimisation works well in practice.

The optimiser below consists of three stages

1. **Diverse random seeding** – many families of shapes (power‑law, exponential, beta‑pdf, Dirichlet, mixtures) are sampled on the simplex.
2. **Projected Adam optimisation** – a fast Adam style optimiser on a *soft‑max* smooth approximation of the max, with an annealing parameter `β`.  
   After every step the iterate is projected back onto the ℓ¹‑simplex.
3. **Late‑stage polishing** – either a short simulated‑annealing run or a final Adam run that uses the *exact* max (β = ∞) to drive the solution into a tighter local minimum.

All helper functions are pure and top‑level, the entry point is `propose_candidate`.  
The routine respects the 1000 s wall‑clock budget, prints progress and finally returns the best sequence found as a plain Python list.

```python
"""
Step‑function optimiser (improved).

The optimiser works in three phases:

1.  Randomly sample many shapes on the ℓ¹‑simplex (power‑law,
    exponential, beta‑pdf, Dirichlet, mixtures).  Each sample is
    quickly refined with a short Adam optimiser.

2.  The current best candidate is refined for a longer period with
    Adam on a smoothed max (log‑sum‑exp).  The temperature β is
    annealed up to a large value (≈ 1e5) so that the gradient focuses
    on the true maximum.

3.  A final polishing stage either runs a short simulated‑annealing
    walk or a last Adam step that uses the *exact* maximum (β = ∞).

All iterates are projected onto the simplex {x ≥ 0, Σx = √(2 n)}.
The routine respects the 1000 s wall‑clock limit and returns the
best sequence as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation: max(conv(a,a))."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p),   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # b = a * a (linear convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    expb = np.exp(beta * (b - max_b))
    p = expb / expb.sum()

    # correlation a ⋆ p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    corr = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * corr[n - 1 : n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Exact‑max gradient (β = ∞)
# ----------------------------------------------------------------------


def _gradient_exact_max(a: np.ndarray):
    """
    Gradient of the true max(b) where b = a*a.
    Returns (g, b) with g_i = 2·a_{k*‑i} for k* = argmax b.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    k = int(b.argmax())

    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    i_vals = np.arange(i_min, i_max + 1, dtype=int)
    j_vals = k - i_vals
    g = np.zeros_like(a)
    g[i_vals] = 2.0 * a[j_vals]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser (projected)
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float,
              beta_start: float = 0.5,
              beta_max: float = 65536.0,
              beta_factor: float = 1.5,
              base_lr_factor: float = 0.15) -> np.ndarray:
    """
    Adam optimiser on the smoothed max.
    After each step the iterate is projected onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    beta = beta_start
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = (target_sum * base_lr_factor) / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection onto ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if t % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


def _adam_opt_exact_max(a0: np.ndarray, deadline: float,
                        base_lr_factor: float = 0.08) -> np.ndarray:
    """
    Adam optimiser that uses the *exact* max (β = ∞) in the gradient.
    This is used only in the final polishing stage.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_exact_max(a)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = (target_sum * base_lr_factor) / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to avoid getting stuck
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

# Simulated annealing (mass transfer)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Simple SA that transfers a random fraction of mass between two indices."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9          # initial temperature
    cooling = 0.9994
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.20        # up to 20 % transfer
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or rng.random() < np.exp(-dE / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter (very low prob)
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best

def _random_initial_sequence_v2(n: int, rng) -> np.ndarray:
    """
    Produce a diverse random non‑negative sequence on the simplex
    Σ a = √(2 n).  Several parametric families are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 9)          # nine families
    idx = np.arange(n, dtype=float)

    if family == 0:                      # pure power‑law
        p = rng.uniform(0.05, 3.0)       # allow steeper decays
        a = (idx + 1.0) ** (-p)

    elif family == 1:                    # exponential decay
        lam = rng.exponential(scale=0.03) + 1e-4
        a = np.exp(-lam * idx)

    elif family == 2:                    # mixed power‑law × exponential
        p = rng.uniform(0.05, 2.0)
        lam = rng.exponential(scale=0.02) + 1e-4
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                    # beta‑distribution shape
        alpha = rng.uniform(0.3, 5.0)
        beta = rng.uniform(0.3, 5.0)
        a = (idx + 1) ** (alpha - 1) * (n - idx) ** (beta - 1)

    elif family == 4:                    # Dirichlet with small concentration (sparse)
        conc = rng.uniform(0.02, 0.3)
        a = rng.dirichlet(alpha=np.full(n, conc))

    elif family == 5:                    # piecewise exponential (two rates)
        split = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02) + 1e-4
        lam2 = rng.exponential(scale=0.02) + 1e-4
        a = np.concatenate([np.exp(-lam1 * np.arange(split)),
                            np.exp(-lam2 * np.arange(split, n))])

    elif family == 6:                    # sinusoidal bump (always non‑negative)
        a = np.sin(np.pi * (idx + 0.5) / (n + 0.5))
        a = np.maximum(a, 0.0)

    elif family == 7:                    # mixture of two random families
        # pick two families among the first five and mix
        f1, f2 = rng.choice([0, 1, 2, 3, 4], size=2, replace=False)
        a1 = _random_initial_sequence_v2_one_family(n, f1, rng)
        a2 = _random_initial_sequence_v2_one_family(n, f2, rng)
        w = rng.random()
        a = w * a1 + (1.0 - w) * a2

    else:                                # uniform plus tiny jitter
        a = np.full(n, target / n)

    # add a small Gaussian noise, clip negatives and project onto the simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _random_initial_sequence_v2_one_family(n: int, family: int, rng) -> np.ndarray:
    """Helper to generate a shape for a single family (used for mixtures)."""
    idx = np.arange(n, dtype=float)
    if family == 0:                     # pure power‑law
        p = rng.uniform(0.05, 3.0)
        a = (idx + 1.0) ** (-p)
    elif family == 1:                   # exponential
        lam = rng.exponential(scale=0.03) + 1e-4
        a = np.exp(-lam * idx)
    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.05, 2.0)
        lam = rng.exponential(scale=0.02) + 1e-4
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif family == 3:                   # beta‑pdf shape
        alpha = rng.uniform(0.3, 5.0)
        beta = rng.uniform(0.3, 5.0)
        a = (idx + 1) ** (alpha - 1) * (n - idx) ** (beta - 1)
    else:                               # Dirichlet sparse (fallback)
        conc = rng.uniform(0.02, 0.3)
        a = rng.dirichlet(alpha=np.full(n, conc))
    return a

    """

    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # If a previous construction exists, start from it
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – more flexibility for larger n
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 3000, 4000, 6000, 8000, 11000]

    for n in lengths:
        if time.time() >= deadline:

        # --------------------------------------------------------------
        # Phase A – cheap random restarts with a short Adam refinement
        # --------------------------------------------------------------
        phase_a_end = min(deadline,
                          time.time() + min(45.0, (deadline - time.time()) * 0.25))
        while time.time() < phase_a_end:
            a0 = _random_initial_sequence_v2(n, rng)

            # short Adam run (≈ 1 s) to bring the draw into a decent basin
            sub_dead = min(deadline, time.time() + 1.0)
            a_gd = _adam_opt(a0, sub_dead, base_lr_factor=0.18)

            sc = _score(a_gd)
            if sc < best_score:
                best_score = sc
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase B – deep refinement of the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            # allocate the remaining budget for a thorough Adam optimisation
            sub_dead = min(deadline, time.time() + min(140.0, deadline - time.time()))
            a_ref = _adam_opt(best_seq, sub_dead,
                               beta_start=0.5,
                               beta_max=1e5,
                               beta_factor=1.8,
                               base_lr_factor=0.12)
            sc = _score(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined best (length {n}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – either SA or a last exact‑max Adam run
    # ------------------------------------------------------------------
    if best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 20.0:
            # a short SA run to escape possible plateaus
            a_sa = _simulated_annealing(best_seq, deadline)
            sc_sa = _score(a_sa)
            if sc_sa < best_score:
                best_score = sc_sa
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] SA improvement: score={sc_sa:.6f}")

        # final Adam using the exact max (β = ∞)
        if time.time() < deadline - 2.0:
            a_final = _adam_opt_exact_max(best_seq, deadline)
            sc_fin = _score(a_final)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Exact‑max Adam improvement: score={sc_fin:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up tiny trailing zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence (should never happen)
        n = 1200
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Log the final objective as seen by the official evaluator
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```
