def _adam_optim(a0, deadline):
    a = a0.copy()
    S = a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    t=0
    while time.time() < deadline:
        t+=1
        # compute convolution via FFT
        size_fft = 1 << ((2*a.size-1).bit_length())
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa*fa, n=size_fft)[:2*a.size-1]
        max_b = conv.max()
        kstar = int(conv.argmax())
        # compute subgradient for max
        # but we might combine with softmax if early iterations
        if t < 2000:
            # use softmax gradient
            beta = 1 + (t / 2000) * 255.0  # schedule from 1 to 256
            # compute softmax gradient over conv
            max_c = max_b
            exps = np.exp(beta * (conv - max_c))
            p = exps / exps.sum()
            # conv_rev = convolution of a_rev and p
            a_rev = a[::-1]
            size2 = 1 << ((len(p)+len(a_rev)-1).bit_length())
            Ar_fft = np.fft.rfft(a_rev, n=size2)
            Pr_fft = np.fft.rfft(p, n=size2)
            conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:len(p)+len(a_rev)-1]
            g = 2 * conv_rev[a.size-1 : a.size-1 + a.size]  # same as earlier _gradient_softmax
        else:
            # subgradient of max
            g = np.zeros_like(a)
            lo = max(0, kstar - (a.size - 1))
            hi = min(kstar, a.size - 1)
            # For i in [lo,hi], we have j = kstar - i
            g[lo:hi+1] = 2 * a[kstar - np.arange(lo, hi+1)]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.1 * S / (np.sqrt(v_hat) + epsilon)  # maybe scale based on sum
        # Actually global lr factor for all coordinates
        a = a - (0.01 / (np.sqrt(t) + 1)) * m_hat / (np.sqrt(v_hat) + epsilon)  # simplistic
        # Ensure nonnegativity and keep sum
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, S)
        # optionally keep best
        # compute objective (max_b) after projection
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa*fa, n=size_fft)[:2*a.size-1]
        cur_max = conv.max()
        if cur_max < best_val: store.
    return best_a, best_val

"""
Propose_candidate: a three‑stage optimizer for the max‑convolution benchmark.
Stage 1 – Construct an initial non‑negative step function (power‑law or
interpolated global best) and iteratively improve it by solving a linear
program that maximises the total mass while keeping the self‑convolution
under the current peak.  A small convex combination with the current
function reduces the peak.
Stage 2 – Fine‑tune the result with an Adam‑style projected gradient
descent on a soft‑max approximation of the peak.  The algorithm
maintains the ℓ¹‑norm constraint by Euclidean projection onto the simplex.
Stage 3 – Deterministic hill‑climbing that shifts a tiny fraction of mass
from indices that contribute most to the current peak to low‑contributing
indices; this cleans remaining local irregularities.
The routine respects the 1000 s time budget and returns the best sequence
found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the maximal convolution entry.
    Returns a gradient vector of the same length as a and the current convolution.
    """
    n = a.shape[0]
    size_fft = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]
    max_b = conv.max()
    # soft‑max weights
    exps = np.exp(beta * (conv - max_b))
    p = exps / exps.sum()

    # convolution of reversed a with p
    a_rev = a[::-1]
    conv_len = n + p.size - 1
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    # gradient w.r.t. a_i: 2 * (a_rev * p)[i + n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, conv

def _lp_refine(f: np.ndarray, target_sum: float, t: float, deadline: float):
    """
    One linear‑programming refinement step.
    Maximises Σ g_i under constraints (f * g)_k ≤ M for all k,
    where M = max(f*f).  The obtained g is scaled to Σ g_i = target_sum
    and mixed with f using coefficient t.
    Returns the new f and the achieved max‑convolution value.
    """
    n = f.size
    M = _max_convolution_fft(f)

    # Build sparse matrix A where A[k,i] = f_i if i ≤ k ≤ i+n-1 else 0
    rows = []
    cols = []
    data = []
    # loop over i to fill column i
    for i in range(n):
        fi = f[i]
        if fi == 0.0:
            continue
        start = i
        end = i + n      # exclusive
        rows.extend(range(start, end))
        cols.extend([i] * n)
        data.extend([fi] * n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # RHS vector for the inequalities
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear program: maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if not res.success:
        # LP failed – fall back to returning the unchanged vector
        return f, M
    g = res.x
    sum_g = g.sum()
    if sum_g < 1e-12:
        return f, M
    # scale to target_sum
    scale = target_sum / sum_g
    g_scaled = g * scale

    # convex combination
    new_f = (1.0 - t) * f + t * g_scaled
    # re‑project to guarantee exact sum (numeric errors)
    new_f = _project_to_simplex(new_f, target_sum)

    new_M = _max_convolution_fft(new_f)
    return new_f, new_M

def _adam_softmax_optimize(a0: np.ndarray, target_sum: float,
                           deadline: float, max_iters: int = 200000):
    """
    Projected Adam‑style optimisation of the soft‑max approximation of
    the maximal convolution entry.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step_base = target_sum * 0.15      # base learning‑rate factor
    beta = 1.0
    max_beta = 256.0
    it = 0

    while time.time() < deadline and it < max_iters:
        # compute gradient (soft‑max)
        g, conv = _softmax_gradient(a, beta)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        # adaptive step size
        lr = step_base / (np.sqrt(v_hat) + eps)

        a = a - lr * m_hat
        # enforce non‑negativity and ℓ¹‑norm
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # evaluate
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD   iter={it}  β={beta:.1f}  max={best_val:.6f}")

        # beta schedule
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # occasional jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def _guided_hill_climb(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Deterministic hill‑climbing that moves a small fraction of mass
    from a high‑weight donor (large a_i·a_{k*‑i}) to a low‑weight receiver.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    cur_val = best_val
    rng = np.random.default_rng()
    max_frac = 0.07

    while time.time() < deadline:
        conv = np.convolve(a, a)      # full conv, O(n²) but n moderate
        k_max = int(np.argmax(conv))

        # contribution weight for each index i: a_i * a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        indices = np.arange(lo, hi + 1)
        weight = a[indices] * a[k_max - indices]

        if weight.max() == 0:
            break

        donor = int(lo + np.argmax(weight))
        # choose a receiver among low‑weight positions (10‑th percentile)
        thresh = np.percentile(weight, 10)
        low_candidates = indices[weight <= thresh]
        if low_candidates.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = int(rng.choice(low_candidates))

        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _max_convolution_fft(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH   new best={best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val

def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random but sensible initial non‑negative sequence of length n.
    The function randomly picks one of several parametric families.
    """
    target = np.sqrt(2.0 * n)
    choice = rng.integers(0, 4)
    if choice == 0:
        # simple power‑law a_i ∝ (i+1)^(-p)
        p = rng.uniform(0.3, 0.8)
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    elif choice == 1:
        # raised cosine (Hann) window
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    elif choice == 2:
        # exponential decay
        lam = rng.uniform(0.5, 2.0)
        a = np.exp(-lam * np.arange(n, dtype=np.float64) / n)
    else:
        # uniform (as a fallback)
        a = np.ones(n, dtype=np.float64)
    a = a * (target / a.sum())
    a = _project_to_simplex(a, target)
    return a

"""
Propose_candidate: a three‑stage optimizer for the max‑convolution benchmark.

Stage 1 – Linear‑programming refinement.
  Given the current sequence f, we compute its peak M = max(f*f) and solve
  the LP   maximise Σ g_i   subject to   (f * g)_k ≤ M  for all k and g_i ≥ 0.
  The solution g (scaled to the correct ℓ¹‑norm) is mixed with f:
        f ← (1‑t) f + t g   (t starts at 0.25 and shrinks).
  Each LP step never raises the peak and typically lowers it.

Stage 2 – Adam‑style projected gradient descent on a soft‑max surrogate of
  the peak.  The gradient of  β‑soft‑max(max (f*f)) is obtained analytically
  via FFTs; Adam moments are updated and the iterate is projected onto the
  simplex {x_i ≥ 0, Σx_i = √(2 n)}.

Stage 3 – Deterministic hill‑climbing.
  The index k* where the current convolution attains the maximum is found.
  For each i the contribution a_i · a_{k*‑i} is a “weight”.  A small
  fraction of mass is moved from a high‑weight donor to a low‑weight
  receiver; moves that improve the peak are kept.

All stages respect the time budget (default 1000 s) and the non‑negativity /
ℓ¹‑norm constraints.  The best sequence encountered is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the β‑soft‑max approximation of max(a*a).
    Returns (grad, full_convolution).
    """
    n = a.shape[0]
    size_fft = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]
    max_b = conv.max()
    exps = np.exp(beta * (conv - max_b))
    p = exps / exps.sum()
    a_rev = a[::-1]
    conv_len = n + p.size - 1
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Stage 1 – Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, target_sum: float, t: float, deadline: float):
    """
    Perform a single LP‑based refinement.
    Returns (new_f, new_max) where new_f = (1‑t)·f + t·g_scaled.
    """
    n = f.size
    M = _max_convolution_fft(f)

    # Build sparse matrix A with A[k,i] = f_i if i ≤ k ≤ i+n‑1 else 0.
    rows = []
    cols = []
    data = []
    for i in range(n):
        fi = f[i]
        if fi == 0.0:
            continue
        start = i
        end = i + n  # exclusive
        rows.extend(range(start, end))
        cols.extend([i] * n)
        data.extend([fi] * n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M, dtype=np.float64)      # RHS: ≤ M
    c = -np.ones(n, dtype=np.float64)                # maximise Σg ⇔ minimise –Σg
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success:
        return f, M

    g = res.x
    sum_g = g.sum()
    if sum_g < 1e-12:
        return f, M

    g_scaled = g * (target_sum / sum_g)
    new_f = (1.0 - t) * f + t * g_scaled
    new_f = _project_to_simplex(new_f, target_sum)
    new_M = _max_convolution_fft(new_f)
    return new_f, new_M


# ----------------------------------------------------------------------
# Stage 2 – Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_softmax_optimize(a0: np.ndarray, target_sum: float,
                           deadline: float, max_iters: int = 200_000):
    """Adam optimisation on the soft‑max surrogate of the peak."""
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step_base = target_sum * 0.15
    beta = 1.0
    max_beta = 256.0
    it = 0

    while time.time() < deadline and it < max_iters:
        grad, _ = _softmax_gradient(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        lr = step_base / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD  it={it}  β={beta:.1f}  max={best_val:.6f}")

        # β schedule (soft‑max becomes sharper)
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # occasional jitter to leave flat regions
        if it % 2000 == 0 and it > 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------


def _guided_hill_climb(a0: np.ndarray, target_sum: float, deadline: float):
    """Shift tiny mass fractions from high‑ to low‑weight indices."""
    a = a0.copy()

    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    cur_val = best_val
    rng = np.random.default_rng()
