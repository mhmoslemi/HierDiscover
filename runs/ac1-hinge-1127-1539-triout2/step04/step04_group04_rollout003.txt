def _build_convolution_matrix(f):
    n = f.shape[0]
    # create lists
    row_idx = np.empty(n * n, dtype=np.int32)
    col_idx = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    idx = 0
    for i in range(n):
        r_start = i
        r_end = i + n
        row_idx[idx: idx+n] = np.arange(r_start, r_end)
        col_idx[idx: idx+n] = i
        data[idx: idx+n] = f
        idx += n
    A = coo_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    return A.tocsr()

"""
Search for a low‑score non‑negative step‑function.

The algorithm mixes three ideas:
1.  Random parametric families generate many candidate vectors.
2.  A custom linear‑programming refinement (inspired by the paper
    “alternating optimisation of f·g ≤ M”) repeatedly mixes the current
    vector with a direction that maximises its ℓ₁‑mass while keeping the
    self‑convolution bounded by the current peak M.
3.  A short Adam‑based smooth‑max minimisation fine‑tunes the best
    candidate found so far.

All vectors are projected onto the ℓ₁‑simplex
   {a ≥ 0, Σ a = √(2·n)},
so that the external evaluation score reduces to the maximal
self‑convolution entry M.  The routine respects the 1000 s budget,
stores the best sequence in the global `height_sequence_1`, and
returns it as a plain Python list.
"""

import time
import numpy as np

# Optional SciPy – used for the LP refinement.
try:
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix, csr_matrix
    SCIPY_AVAILABLE = True
except Exception:                       # pragma: no cover
    SCIPY_AVAILABLE = False


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (O(n log n))."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate an un‑normalised non‑negative vector from a random family."""
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular", "power", "cosine"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    else:                                 # pragma: no cover
        a = rng.random(n)

    # tiny jitter for extra diversity
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


def _build_convolution_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that (A @ g) == f * g (full convolution).
    Row k corresponds to index k in the convolution output.
    """
    n = f.shape[0]
    # Each column i contributes f to rows i … i+n‑1
    row_idx = np.empty(n * n, dtype=np.int32)
    col_idx = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    idx = 0
    for i in range(n):
        row_idx[idx: idx + n] = np.arange(i, i + n, dtype=np.int32)
        col_idx[idx: idx + n] = i
        data[idx: idx + n] = f
        idx += n

    A = coo_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A.tocsr()


def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Linear program:
        maximise   Σ g_i
        subject to f * g ≤ M   (element‑wise)
                    g_i ≥ 0
    Returns the optimal g (numpy array) or None on failure.
    """
    if not SCIPY_AVAILABLE:               # pragma: no cover
        return None
    n = f.shape[0]
    A = _build_convolution_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)     # minimise -Σg_i  → maximise Σg_i
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        # enforce tiny non‑negativity due to numeric noise
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a0: np.ndarray, max_iter: int, t0: float,
               t_decay: float, deadline: float,
               rng: np.random.Generator) -> np.ndarray:
    """
    Repeatedly:
        1. solve the LP that maximises Σg under f*g ≤ current max
        2. scale g to the ℓ₁‑simplex
        3. mix f ← (1‑t) f + t g
    The mixing parameter t is decayed geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    t_cur = t0
    best_a = a.copy()
    best_M = np.max(_full_convolution_fft(a))

    while time.time() < deadline and max_iter > 0:
        M = np.max(_full_convolution_fft(a))
        g0 = _solve_lp_max_sum(a, M)

        if g0 is None:
            # fallback: small random jitter
            noise = rng.normal(scale=a.mean() * 0.015 + 1e-8, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            max_iter -= 1
            continue

        sum_g0 = np.sum(g0)
        if sum_g0 <= 0.0:
            break

        scaling = S / sum_g0
        g = g0 * scaling

        # convex combination
        a = (1.0 - t_cur) * a + t_cur * g
        a = _project_to_simplex(a, S)

        cur_M = np.max(_full_convolution_fft(a))
        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()

        t_cur *= t_decay
        max_iter -= 1

    return best_a


def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Gradient of τ·log Σ exp(b/τ) where b = a*a (full convolution).
    Returns (gradient, raw convolution b, max(b)).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard max → sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:                     # diagonal term
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)                # shape (2n‑1,)

    # compute gradient = 2·a ⋆ w
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]       # centre part, length n
    return g, b, max_b


def _adam_optimize(a0: np.ndarray, deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser minimising the soft‑max approximation of max(a*a)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6
    t = 0

    # initial temperature (scale of the convolution)
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        cur_max_new = np.max(_full_convolution_fft(a_new))
        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        # occasional jitter to escape plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max
