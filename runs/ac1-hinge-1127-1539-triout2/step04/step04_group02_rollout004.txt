def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of `a` using an FFT of suitable length."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence of length n.
    Kinds: linear, quadratic, exponential, power, inverse,
           linear_left, linear_right, exponential_left, exponential_right.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_beta_shape(n: int, target_sum: float, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑family shape: a_i ∝ (i+1)^{α-1} (n-i)^{β-1}.
    α,β > 0; α=β=1 gives the uniform distribution.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    raw = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    raw = np.maximum(raw, 0.0)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())

        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 12),
        "inverse": np.linspace(0.5, 5.0, 12),
        "linear_left": np.linspace(0.0, 0.9, 19),
        "linear_right": np.linspace(0.0, 0.9, 19),
        "exponential_left": np.linspace(0.0, 6.0, 31),
        "exponential_right": np.linspace(0.0, 6.0, 31),
    }

    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during parametric sweep.")
                return best_seq.tolist()

    # Beta‑family sweep
    alphas = [0.3, 0.5, 0.7, 1.0, 2.0]
    betas  = [0.3, 0.5, 0.7, 1.0, 2.0]
    for a_alpha in alphas:
        for b_beta in betas:
            cand = generate_beta_shape(n, target_sum, a_alpha, b_beta)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best Beta(α={a_alpha:.2f},β={b_beta:.2f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during Beta sweep.")
                return best_seq.tolist()

    # random Dirichlet‑like vectors
    for _ in range(15):
        cand = np.random.rand(n)
        cand = project_onto_simplex(cand, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best random init → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during random init.")
            return best_seq.tolist()

    # include supplied baseline if it exists
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves start → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed Adam descent
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.55
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    lr = 0.6 * (target_sum / n)           # initial learning rate
    tau_factor = 0.25
    iter_cnt = 0
    no_improve_cnt = 0

    print(f"[{time.time():.2f}] Phase 1 – smoothed Adam descent")
    while time.time() < phase1_deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()
        tau = max(tau_factor * max_conv, 1e-12)

        # log‑sum‑exp weights
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # parameter update (gradient descent)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        iter_cnt += 1

        # keep best
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            no_improve_cnt = 0
            print(f"Iter {iter_cnt:07d} – improved to {best_score:.6f}")
        else:
            no_improve_cnt += 1

        # scheduling tweaks
        if iter_cnt % 4000 == 0:
            lr *= 0.85
            tau_factor = max(0.001, tau_factor * 0.88)

        # occasional random tiny transfer to shake‑up
        if np.random.rand() < 0.018:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # if we've stagnated for a while, add a small jitter
        if no_improve_cnt >= 6000:
            jitter = np.random.randn(n) * 0.00035
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve_cnt = 0
            print(f"Iter {iter_cnt:07d} – jitter applied")

    # ------------------------------------------------------------------
    # Phase 2 – true sub‑gradient refinement
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.88
    step_size = 0.04 * (target_sum / n)
    sub_iter = 0
    print(f"[{time.time():.2f}] Phase 2 – sub‑gradient refinement")
    while time.time() < phase2_deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()

        eps_mask = max(1e-12, 1e-9 * max_conv)
        mask = (conv >= max_conv - eps_mask).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0

        a_rev = a[::-1]
        L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        a = a - step_size * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        sub_iter += 1
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Sub‑grad {sub_iter:06d} – improved to {best_score:.6f}")

        if sub_iter % 500 == 0:
            step_size *= 0.93

        if np.random.rand() < 0.014:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    # ------------------------------------------------------------------
    # Phase 3 – final polishing (tiny random pair transfers)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – polishing")
    while time.time() < deadline:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.0025)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
