def compute_conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

"""
Propose candidate step function for the max-convolution benchmark using
a multi‑move simulated‑annealing + final soft‑max gradient descent.
The optimizer employs donor→receiver mass transfers, local smoothing,
random power‑transform reshaping and occasional jitter, with an
exponential temperature schedule.  The best sequence found before the
time budget expires is returned.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    # Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, v.size+1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n - 1]
    return conv

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    if sum_a < 0.01:
        return np.inf
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _softmax_gradient(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n - 1]
    max_b = np.max(b)
    # soft‑max over convolution entries
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # compute gradient of sum_i p_i (a * a)[i] w.r.t a
    # derivative = 2 * (a * reverse(p)) (restrict to length n)
    a_rev = a[::-1]
    conv_len = n + (2*n - 1) - 1  # = 3n-2? actually we only need part length n
    size2 = 1 << (conv_len.bit_length())
    A_rev_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, b

def _gradient_descent_softmax(a0: np.ndarray, deadline: float):
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = _score_from_conv(_conv_fft(a), n, a.sum())
    beta = 1.0
    beta_max = 512.0
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = (target * 0.15) / (norm * (beta + 1.0))
            a -= step * grad
        a = _project_to_simplex(a, target)

        if it % 200 == 0:
            # small jitter to escape shallow minima
            jitter = rng.normal(scale=target*0.001/(beta+1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        cur_score = _score_from_conv(_conv_fft(a), n, a.sum())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD  beta={beta:.2f} score={best_score:.6f}")

        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * 1.6, beta_max)

        it += 1
    return best_a, best_score

def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.05,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.008,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    cur_sum = a.sum()
    # initial convolution
    conv = _conv_fft(a)
    cur_score = _score_from_conv(conv, n, cur_sum)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac_done = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** frac_done
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** frac_done

        # compute sensitivity for the current maximal convolution index
        k_max = int(np.argmax(conv))
        # contribution array (weight) = a[i] * a[k_max-i] for overlapping i
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = a[k_max - idx]   # a[j] where j = k_max - i
        weight = a[idx] * sens

        # pick move type
        if rng.random() < 0.65:
            # ------------------- donor / receiver transfer -------------------
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = idx[weight <= low_thr]
            if low_idx.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_idx)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue
            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:
            move_selector = rng.random()
            if move_selector < 0.5:
                # ------------------- local smoothing -------------------
                start = rng.integers(0, n)
                length = rng.integers(2, max_window + 1)
                end = min(start + length, n)
                if end - start < 2:
                    continue
                seg = a[start:end]
                avg = seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = seg.copy()
                saved_start, saved_end = start, end
            else:
                # ------------------- power‑transform reshaping -------------------
                gamma = rng.uniform(0.85, 1.15)  # exponent close to 1
                a = np.power(a, gamma)
                # re‑project to the simplex (preserves sum)
                a = _project_to_simplex(a, target_sum)
                move_type = "power"

        # recompute convolution via FFT
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[:2*n - 1]
        new_score = _score_from_conv(conv_new, n, cur_sum)

        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA  new best {best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "power":
                # undo by performing an inverse power‑transform is not trivial;
                # instead we simply revert to previous best copy of a
                a = best_a.copy()
                conv = _conv_fft(a)
                cur_score = best_score
                # continue; we skip the rest of this loop
                continue

        # occasional jitter to shake the system
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum*0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score_from_conv(conv, n, target_sum)

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine follows three phases:
        1.  Initialise a collection of power‑law candidates of various lengths
            and exponents; keep the best.
        2.  Refine the best candidate with a custom simulated‑annealing
            optimiser that mixes donor/receiver transfers, local smoothing,
            power‑transform reshaping and occasional jitter.
        3.  Perform a short soft‑max gradient‑descent fine‑tuning.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1.  Build a set of deterministic starting points (power‑law families)
    # ------------------------------------------------------------------
    lengths = [1000, 1200, 1500, 1800, 2000, 2500]
    exponents = [0.35, 0.45, 0.55, 0.65, 0.75]
    best_seq = None
    best_score = np.inf

    for n in lengths:
        target = np.sqrt(2.0 * n)
        for p in exponents:
            a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
            a = a * (target / a.sum())
            a = _project_to_simplex(a, target)
            # quick evaluation using fast convolution
            conv = _conv_fft(a)
            sc = _score_from_conv(conv, n, a.sum())
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] init n={n} p={p:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  If a globally supplied construction exists, consider it as well.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        seq_global = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n0 = seq_global.size
        target = np.sqrt(2.0 * n0)
        if seq_global.sum() > 0:
            seq_global = seq_global * (target / seq_global.sum())
        else:
            seq_global = _project_to_simplex(_power_law_sequence(n0, 0.55, target), target)
        seq_global = _project_to_simplex(seq_global, target)
        conv = _conv_fft(seq_global)
        sc = _score_from_conv(conv, n0, seq_global.sum())
        if sc < best_score:
            best_seq = seq_global.copy()
            best_score = sc
            print(f"[{time.time():.1f}] global seq score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Main simulated‑annealing optimisation on the current best.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a simple power law
        n0 = 1500
        target = np.sqrt(2.0 * n0)
        best_seq = (np.arange(1, n0+1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target / best_seq.sum())
        best_seq = _project_to_simplex(best_seq, target)

    # Run SA until time runs out (leaving a few seconds for final GD)
    sa_deadline = deadline - 5.0
    if sa_deadline > time.time():
        seq_sa, sc_sa = _simulated_annealing(best_seq, sa_deadline)
        if sc_sa < best_score:
            best_seq, best_score = seq_sa, sc_sa
            print(f"[{time.time():.1f}] after SA best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final short soft‑max gradient descent (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = deadline
        seq_gd, sc_gd = _gradient_descent_softmax(best_seq, gd_deadline)
        if sc_gd < best_score:
            best_seq, best_score = seq_gd, sc_gd
            print(f"[{time.time():.1f}] after GD final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Trim trailing zeros (tiny entries) for a compact representation.
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

"""
Propose candidate step function for the max-convolution benchmark.

The search proceeds in three stages:
  * Exhaustively scan a small family of power‑law sequences of various
    lengths and exponents and keep the best one (plus the optional
    global ``height_sequence_1`` if present).
  * Refine the best candidate with a simulated‑annealing optimiser that
    mixes several elementary moves:
        – donor→receiver mass transfer,
        – local smoothing of a short window,
        – random power‑transform reshaping,
        – occasional Gaussian jitter.
    An exponential temperature schedule together with a shrinking
    maximum transfer fraction provides a balance between exploration
    and exploitation.
  * Finish with a short soft‑max gradient‑descent pass that directly
    minimises a smooth approximation of the max‑convolution term.
The algorithm respects the non‑negativity and ℓ¹‑norm constraints
through projection onto the simplex after each move.  The best sequence
found before the time budget expires is returned as a plain list of
floats.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, v.size + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact value of the benchmark objective given a pre‑computed convolution."""
    if sum_a < 0.01:
        return np.inf
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _softmax_gradient(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv) w.r.t. a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    max_b = np.max(b)
    exp_terms = np.exp(beta * (b - max_b))
    p = exp_terms / np.sum(exp_terms)

    a_rev = a[::-1]
    conv_len = 3 * n - 2
    size2 = 1 << (conv_len.bit_length())
    A_rev_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size2)[: conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _gradient_descent_softmax(a0: np.ndarray, deadline: float):
    """Projected gradient descent on the soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = _score_from_conv(_conv_fft(a), n, a.sum())
    beta = 1.0
    beta_max = 512.0
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = (target * 0.15) / (norm * (beta + 1.0))
            a -= step * grad
        a = _project_to_simplex(a, target)

        if it % 200 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        cur_score = _score_from_conv(_conv_fft(a), n, a.sum())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD  beta={beta:.2f} score={best_score:.6f}")

        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * 1.6, beta_max)

        it += 1
    return best_a, best_score

def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.05,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.008,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """Custom SA with several move types."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    conv = _conv_fft(a)
    cur_score = _score_from_conv(conv, n, target_sum)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # --- sensitivity for the presently maximal entry ----------------
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = a[k_max - idx]               # partner values
        weight = a[idx] * sens               # contribution to the maximal entry

        # pick a move
        if rng.random() < 0.65:
            # ----------- donor → receiver mass transfer ----------------
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_candidates = idx[weight <= low_thr]
            if low_candidates.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_candidates)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:
            # choose between smoothing and power‑transform
            if rng.random() < 0.5:
                # ------------------- local smoothing -------------------
                start = rng.integers(0, n)
                length = rng.integers(2, max_window + 1)
                end = min(start + length, n)
                if end - start < 2:
                    continue
                segment = a[start:end].copy()
                avg = segment.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = segment
                saved_start, saved_end = start, end
            else:
                # ------------------- power‑transform reshaping ----------
                a_old = a.copy()
                gamma = rng.uniform(0.85, 1.15)
                a = np.power(a, gamma)
                a = _project_to_simplex(a, target_sum)
                move_type = "power"
                a_old_saved = a_old   # will be used to revert if needed

        # evaluate the new state
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score_from_conv(conv_new, n, target_sum)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA  new best {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "power":
                a = a_old_saved.copy()
                # conv and cur_score stay unchanged (still old)
                # no need to recompute anything

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score_from_conv(conv, n, target_sum)

        it += 1

    return best_a, best_score

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:

    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = a * (target_sum / a.sum())
    return a

        target = np.sqrt(2.0 * n0)
        best_seq = _power_law_sequence(n0, 0.55, target)
        best_seq = _project_to_simplex(best_seq, target)

    # reserve a few seconds for the final gradient pass
    sa_deadline = deadline - 5.0
    if sa_deadline > time.time():
        seq_sa, sc_sa = _simulated_annealing(best_seq, sa_deadline)
        if sc_sa < best_score:
            best_seq, best_score = seq_sa, sc_sa
            print(f"[{time.time():.1f}] after SA best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Short gradient‑descent polishing (if time permits).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_gd, sc_gd = _gradient_descent_softmax(best_seq, deadline)
        if sc_gd < best_score:
            best_seq, best_score = seq_gd, sc_gd
            print(f"[{time.time():.1f}] after GD final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Trim insignificant trailing zeros for a compact representation.
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
