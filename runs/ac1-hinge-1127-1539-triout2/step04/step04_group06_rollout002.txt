"""
propose_candidate – hybrid optimisation for the max‑convolution benchmark.

The objective after normalisation Σa = √(2·n) is simply
    max_k (a * a)[k] .
We seek a non‑negative step function a that minimises this peak.

Algorithm (time‑budget aware):
1. Initialise from the global best `height_sequence_1` if present,
   otherwise from a power‑law of length ≈2000.
2. **Annealed soft‑max gradient descent** – a projected gradient method on a
   smooth approximation of the max‑convolution.  β (soft‑max sharpness) is
   raised progressively up to 512, while the iterates are projected onto the
   ℓ¹‑simplex (Σa = √(2·n), a_i ≥ 0).  This quickly drives the large‑weight
   entries down.
3. **Simulated annealing** – mass‑transfer, local smoothing and occasional
   value‑swap moves, evaluated by a single FFT per iteration.  Temperature
   and move‑size decay exponentially.
4. **Random‑restart phase** – while time permits we generate short‑lived
   restarts (random length, random power‑law exponent) and run a brief
   gradient‑descent + SA pass; any improvement updates the global best.
5. Trim trailing near‑zero entries and return the best sequence found.

The code respects all constraints (non‑negativity, ℓ¹‑norm, value caps) and
uses only NumPy.  It is deliberately written with no closures, no file I/O,
and a clear top‑level API `propose_candidate`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Exact evaluation of the competition objective for a normalised sum.
    The function equals max(conv) when Σa = √(2·n).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).

    Returns (gradient, softmax_weights) where gradient has the same shape as a.
    """
    n = a.shape[0]
    # FFT of a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max over convolution entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (rev(a) * p) (restricted to first n entries)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Optimisation primitives

def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on a soft‑max approximation of max(conv).
    β (soft‑max sharpness) is increased exponentially up to max_beta.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 512.0           # higher than the original 256
    step_base = target_sum * 0.25
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(grad)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * grad

        # projection onto the ℓ¹‑simplex Σa = target_sum, a_i ≥ 0
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD   β={beta:.2f}  score={best_val:.6f}")

        # increase β gradually
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 20) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Three move types:
      * mass transfer (donor → receiver)
      * value swap (exchange two entries)
      * local smoothing (replace a short window by its average)
    Acceptance follows a Metropolis rule with an exponential temperature schedule.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = 2.0 * n * _max_convolution_fft(a) / (cur_sum ** 2)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial convolution
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity for the active convolution entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        # decide move type
        r = rng.random()
        move_type = None

        if r < 0.60:  # mass transfer (most common)
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"

        elif r < 0.75:  # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        else:  # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # evaluate the new state (single FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new / (cur_sum ** 2)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]   # swap back
            else:  # smooth
                a[saved_start:saved_end] = saved_seg
            # conv and cur_score stay unchanged

        # occasional global jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = 2.0 * n * conv.max() / (cur_sum ** 2)

        it += 1

    return best_a, best_score

    if remaining > 30.0:
        gd_time = max(20.0, remaining * 0.30)
        sub_deadline = time.time() + gd_time
        gd_seq, gd_score = _annealed_gradient_descent(best_seq, sub_deadline)
        if gd_score < best_score:
            best_seq, best_score = gd_seq, gd_score
            print(f"[{time.time():.1f}] after GD, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Simulated‑annealing refinement (uses the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_seq, sa_score = _simulated_annealing(best_seq, deadline)
        if sa_score < best_score:
            best_seq, best_score = sa_seq, sa_score
            print(f"[{time.time():.1f}] after SA, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Random‑restart phase – while time permits we try fresh starts
    # ------------------------------------------------------------------
    while True:
        remaining = deadline - time.time()
        if remaining < 20.0:
            break

        # Random length and exponent for a fresh power‑law start
        n_new = int(rng.integers(1200, 2600))
        target_new = np.sqrt(2.0 * n_new)
        p_new = rng.uniform(0.30, 1.30)
        seq = _power_law_sequence(n_new, p_new, target_new)
        seq = _project_to_simplex(seq, target_new)

        # Allocate a modest budget for the restart (at most 60 s)
        sub_budget = min(remaining * 0.45, 60.0)
        sub_deadline = time.time() + sub_budget

        # Short gradient descent → SA
        gd_seq, _ = _annealed_gradient_descent(seq, sub_deadline * 0.40 + time.time())
        sa_seq, sa_score = _simulated_annealing(gd_seq, sub_deadline)

        if sa_score < best_score:
            best_seq, best_score = sa_seq, sa_score
            print(f"[{time.time():.1f}] new global best from restart "
                  f"n={n_new}, p={p_new:.3f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ post‑processing – drop trailing near‑zero heights
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
