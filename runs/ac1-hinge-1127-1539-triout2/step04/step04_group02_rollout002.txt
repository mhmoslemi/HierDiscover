"""
Propose a step‑function (non‑negative coefficient vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

using a three‑stage strategy:

1. **Broad parametric sweep** – generate many candidate shapes
   (U‑shapes, beta‑distributions, linear/exponential/skewed families,
   random Dirichlet vectors).  All candidates are projected onto the
   ℓ₁‑simplex {x_i≥0, Σx_i = √(2 n)} and the best (lowest
   max‑self‑convolution) is kept.

2. **Smoothed Adam optimisation** – minimise a log‑sum‑exp
   surrogate of the maximum with an Adam‑style update.  The smoothing
   parameter τ is decreased geometrically; after each update the
   vector is projected back onto the simplex.  This stage dominates the
   time budget.

3. **Sub‑gradient refinement & polishing** – switch to the true
   non‑smooth objective, continue a few Adam steps on the sub‑gradient,
   and finish with tiny random pair‑transfers.

The routine respects the 1000 s budget, constantly checks the wall‑clock,
and returns the best sequence discovered.
"""

def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT of length
    equal to the next power of two.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float,
                   kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence of
    length n.  The result is projected onto the ℓ₁‑simplex with the
    required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_beta_shape(n: int, target_sum: float,
                        alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) beta‑distribution shape.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # centre of each bin
    # avoid infinities when alpha or beta < 1
    eps = 1e-12
    raw = (np.maximum(x, eps) ** (alpha - 1)) * \
          (np.maximum(1.0 - x, eps) ** (beta - 1))
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())

        # Adam update
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        lr = base_lr * (np.sqrt(1.0 - beta2 ** t_iter) /
                        (1.0 - beta1 ** t_iter))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # decay the smoothness parameter
        tau = max(tau * tau_decay, 1e-12)

        # occasional exploratory jump
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.006)

    # --------------------------------------------------------------
    # Phase 2 – true sub‑gradient refinement (Adam style)
    # --------------------------------------------------------------
    phase2_end = start_time + budget_s * 0.94   # ~94 % of total time
    sub_m = np.zeros_like(a)
    sub_v = np.zeros_like(a)
    sub_lr = 0.04 * (target_sum / n)   # initial step
    sub_t = 0

    print(f"[{time.time():.2f}] Phase 2 – sub‑gradient Adam refinement")
    while time.time() < phase2_end:
        conv = conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (target_sum ** 2)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_max = cur_max
            best_seq = a.copy()
            print(f"  sub‑grad iter {sub_t:07d} → new best {best_score:.6f}")

        # mask of (approximately) maximal entries
        eps_mask = max(1e-12, 1e-9 * cur_max)
        mask = (conv >= cur_max - eps_mask).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0

        # sub‑gradient of the true max‑convolution
        a_rev = a[::-1]
        L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        # Adam‑style update on the sub‑gradient
        sub_t += 1
        sub_m = beta1 * sub_m + (1.0 - beta1) * grad
        sub_v = beta2 * sub_v + (1.0 - beta2) * (grad ** 2)
        m_hat = sub_m / (1.0 - beta1 ** sub_t)
        v_hat = sub_v / (1.0 - beta2 ** sub_t)

        step = sub_lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step

        # feasibility
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # adapt step size a little
        if sub_t % 400 == 0:
            sub_lr *= 0.94

        # occasional random mass transfer
        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    # --------------------------------------------------------------
    # Phase 3 – final polishing with tiny random pair transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – polishing")
    while time.time() < deadline:
        cand = random_pair_transfer(best_seq.copy(),
                                    target_sum, scale=0.0025)
        cand_max = compute_max_conv(cand)
        cand_score = 2.0 * n * cand_max / (target_sum ** 2)
        if cand_score < best_score:
            best_score = cand_score
            best_max = cand_max
            best_seq = cand.copy()
            print(f"  polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
