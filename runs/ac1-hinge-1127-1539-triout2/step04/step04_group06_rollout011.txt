def gradient_max(seq):
    n = len(seq)
    conv = _conv_fft(seq) # length 2n-1
    k = int(np.argmax(conv))
    # gradient
    grad = np.zeros_like(seq)
    lo = max(0, k - (n-1))
    hi = min(k, n-1)
    # for i from lo to hi, corresponds index i in seq, with partner j = k - i
    # gradient contribution 2*seq[j]
    idx = np.arange(lo, hi+1)
    partner = k - idx
    grad[idx] = 2 * seq[partner]
    return grad, conv[k]
```

def _project_to_simplex(v, s):
    if s < 0:
        raise ...
    # Add epsilon for numeric stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return float(conv.max())

"""
Propose_candidate – multi‑stage optimizer for the max‑convolution / ℓ¹‑norm benchmark.

Stage 1  : fast grid search over length, power‑law exponent and a simple
           two‑segment power‑law.  All candidates are projected onto the
           simplex Σa = √(2·n) so that the objective reduces to max(a*a).

Stage 2  : projected sub‑gradient descent on the exact max‑convolution
           (gradient of the active entry).  A diminishing step‑size is used
           and the iterates are kept on the simplex.

Stage 3  : simulated‑annealing refinement (mass‑transfer & local smoothing
           moves) with an exponential cooling schedule.

Stage 4  : short deterministic hill‑climb that shifts mass from high‑sensitivity
           to low‑sensitivity coordinates.

The best sequence found before the time budget expires is returned as a
plain list of non‑negative floats.
"""

import time
import math
import itertools
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a computed via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (FFT implementation)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact competition score (the evaluator returns the same value)."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    max_b = _max_convolution_fft(a)
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """
    Produce a (i+1)^(-p) sequence of length n and scale it to have sum = target_sum.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq *= target_sum / seq.sum()
    return seq


def _piecewise_power_law(
    n: int, p1: float, p2: float, split_frac: float, target_sum: float
) -> np.ndarray:
    """
    Two‑segment power law:
        a_i = (i+1)^(-p1) for i < split,
        a_i = (i+1)^(-p2) for i ≥ split,
    where split = floor(split_frac * n).  The vector is finally projected onto
    the simplex Σa = target_sum.
    """
    split = int(np.floor(split_frac * n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = np.empty(n, dtype=np.float64)
    seq[:split] = idx[:split] ** (-p1)
    seq[split:] = idx[split:] ** (-p2)
    seq *= target_sum / seq.sum()
    return seq


# ----------------------------------------------------------------------
# Stage 2 – sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the objective max(a*a)
    (equivalently the competition score).  The iterates stay on the simplex
    Σa = Σa0.  A simple diminishing step‑size η_t = α / √t is used.
    """
    a = a0.copy()
    n = a.shape[0]
    target = a.sum()
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    alpha = target * 0.5  # scaling of the step size

    while time.time() < deadline:
        # compute the active convolution entry and its index
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (2 * reversed a at the appropriate shift)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k - idx
        grad[idx] = 2.0 * a[partner]

        # diminishing step size
        eta = alpha / (np.sqrt(it + 1) * (np.linalg.norm(grad) + 1e-12))

        # gradient step + projection
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target)

        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] SGD iter {it:5d}  score={best_val:.6f}")

        # accept the new point if it does not increase the objective
        if val <= _score(a):
            a = a_new

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – simulated annealing
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.02,
    end_temp: float = 1e-6,
    max_frac_start: float = 0.25,
    max_frac_end: float = 0.01,
    max_window: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Classic SA with two move types:
       * donor → receiver mass transfer,
       * local smoothing (replace a short window by its mean).

    The temperature and the maximal transferred fraction shrink exponentially
    with the elapsed proportion τ ∈ [0,1].
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # keep the convolution for the current state (updated only when a move is accepted)
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time

        # exponential schedules
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current most‑active index
        k_max = int(np.argmax(conv))

        # sensitivity vector for the active index (same as in the gradient)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # ------------------------------------------------------------------
        # propose a move
        # ------------------------------------------------------------------
        if rng.random() < 0.70:  # mass transfer (the dominant move)
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"

        else:  # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # ------------------------------------------------------------------
        # evaluate the new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)   # uses the updated a (sum unchanged)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:  # smooth
                a[saved_start:saved_end] = saved_seg
            # conv and cur_score stay unchanged

        # occasional jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 4 – deterministic hill climbing (mass redistribution)
# ----------------------------------------------------------------------


def _guided_hill_climb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Shift a random fraction of mass from a high‑weight donor to a low‑weight receiver."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07  # maximal fraction of donor mass moved in one step

    while time.time() < deadline:
        conv = np.convolve(a, a)  # O(n) for moderate n
        k_max = int(np.argmax(conv))

        # sensitivity for the active index
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens[lo : hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        weight = sens * a
        if weight.max() == 0:
            break

        donor = int(np.argmax(weight))

        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))

        if donor == receiver or a[donor] <= 1e-14:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH improved = {best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search consists of a fast grid search, a short sub‑gradient descent,
    a simulated‑annealing phase and a final hill‑climb.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # initialise – either from the globally supplied best or from scratch
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq.size
        target = math.sqrt(2.0 * n0)
        seq = seq * (target / seq.sum()) if seq.sum() > 0 else _power_law_sequence(n0, 0.55, target)
        seq = _project_to_simplex(seq, target)
        best_seq = seq.copy()
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] start from global seq, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – rapid grid search over (n, exponent) and a simple two‑segment law
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 1500, 2000, 3000, 4000]
    exponents = np.linspace(0.30, 0.80, 26)          # step 0.02
    split_fractions = [0.3, 0.5, 0.7]

    for n in lengths:
        target = math.sqrt(2.0 * n)
        # pure power law
        for p in exponents:
            if time.time() >= deadline:
                break
            seq = _power_law_sequence(n, p, target)
            seq = _project_to_simplex(seq, target)
            sc = _score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] grid (n={n}, p={p:.2f})  score={sc:.6f}")

        # two‑segment power law (only on the best length found so far)
        if best_seq is not None and best_seq.size == n:
            p1_vals = np.linspace(0.30, 0.65, 8)
            p2_vals = np.linspace(0.70, 1.10, 9)
            for p1, p2, sf in itertools.product(p1_vals, p2_vals, split_fractions):
                if time.time() >= deadline:
                    break
                seq = _piecewise_power_law(n, p1, p2, sf, target)
                seq = _project_to_simplex(seq, target)
                sc = _score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time():.1f}] piecewise (n={n}, p1={p1:.2f}, p2={p2:.2f}, split={sf:.2f})  score={sc:.6f}"
                    )
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # If grid search yielded nothing (should not happen), fall back to a simple power law
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target = math.sqrt(2.0 * n)
        best_seq = _power_law_sequence(n, 0.55, target)
        best_seq = _project_to_simplex(best_seq, target)
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] fallback power‑law, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short projected sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + min(20.0, remaining / 2.0)
        seq_sgd, sc_sgd = _subgradient_descent(best_seq, sub_deadline)
        if sc_sgd < best_score:
            best_seq, best_score = seq_sgd, sc_sgd
            print(f"[{time.time():.1f}] after SGD, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing (core refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, sc_sa = _simulated_annealing(best_seq, deadline)
        if sc_sa < best_score:
            best_seq, best_score = seq_sa, sc_sa
            print(f"[{time.time():.1f}] after SA, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final deterministic hill climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_hc, sc_hc = _guided_hill_climb(best_seq, deadline)
        if sc_hc < best_score:
            best_seq, best_score = seq_hc, sc_hc
            print(f"[{time.time():.1f}] after hill‑climb, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros and return
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
```

This implementation explores a broader family of step‑functions (including longer sequences and two‑segment power‑laws), refines them with a dedicated sub‑gradient method, and then applies a carefully tuned simulated‑annealing and hill‑climbing phase. The whole search respects the 1000 s budget and guarantees a non‑negative sequence on return.
