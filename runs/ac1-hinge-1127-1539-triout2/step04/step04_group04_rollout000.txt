def hill_climb(a_start, time_deadline):
    a = a_start.copy()
    cur_max = np.max(_full_convolution_fft(a))
    best_a = a.copy()
    best_max = cur_max
    T = cur_max - 1.0  # start temperature approx above lower bound
    steps = 0
    n = len(a)
    while time.time() < time_deadline:
        steps += 1
        # Randomly move a small amount
        # Choose source index with a_i > 1e-12
        i = rng.choice(np.where(a > 1e-12)[0])
        # Choose target index diff maybe near i or random
        j = rng.integers(0, n)  # could be same? ensure i != j
        if j == i:
            continue
        # delta fraction of source mass
        max_move = a[i] * 0.1  # up to 10%
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0, max_move)
        a_test = a.copy()
        a_test[i] -= delta
        a_test[j] += delta

        # Ensure still nonnegative
        if a_test[i] < -1e-12:
            continue
        # No need projection; sum unchanged

        # Compute new max via convolution
        b_test = _full_convolution_fft(a_test)
        new_max = np.max(b_test)
        # Accept based on simulated annealing
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-9)):
            a = a_test
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        # Anneal
        T *= 0.9999  # gradually reduce
        # Occasionally random global restart? Not needed.
    return best_a, best_max

"""
Search algorithm for minimizing the maximum self‑convolution of a
non‑negative step function.

The algorithm proceeds in three stages:
1. Broad random search over a large family of parametric shapes (exponential,
   power‑law, arcsine, beta‑type, etc.) with lengths between 500 and 15000.
   Each candidate is projected onto the ℓ₁‑simplex {∑a = √(2n)} and its
   maximal convolution entry is recorded.
2. Local improvement of the K best candidates by a simulated‑annealing style
   hill‑climbing that repeatedly moves a small amount of mass from one index
   to another while keeping the ℓ₁‑norm fixed.  The temperature is gradually
   reduced, and the best intermediate vector is retained.
3. Final refinement of the current best vector with the projected‑gradient
   Adam optimiser that minimises a smooth soft‑max approximation of the
   max‑convolution; after each Adam step the vector is projected back onto
   the simplex.

The whole procedure respects the 1000 s time budget (a small safety margin
is kept).  The final sequence is stored in the global variable
`height_sequence_1` and returned as a plain Python list.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full (non‑circular) convolution a * a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _gradient_softmax(a: np.ndarray, tau: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Gradient of the τ·log∑exp(b/τ) where b = a*a.
    Returns (grad, b, max_b).  When τ is very small we fall back to a sub‑gradient.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:
        # sub‑gradient of the hard max
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal contribution (i=j) when k is even
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)  # shape (2n‑1,)

    # convolution a ⋆ w  (cross‑correlation)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, b, max_b


def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam optimiser on the soft‑max approximation of max(a*a)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6
    t = 0

    # initialise temperature
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        b_new = _full_convolution_fft(a_new)
        cur_max_new = np.max(b_new)

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        # occasional jitter
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


def _random_distribution_extended(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative vector of length n from a diverse set of parametric families."""
    typ = rng.choice(
        [
            "uniform", "exp", "symexp", "gaussian", "triangular",
            "power", "cosine", "arcsine", "beta", "mixexp"
        ]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "arcsine":
        # a_i ∝ ((i+1)*(n-i))^{-p}
        p = rng.uniform(0.2, 1.8)
        a = ((idx + 1) * (n - idx)) ** (-p)

    elif typ == "beta":
        # a_i ∝ (i+1)^{α} (n-i)^{β}
        alpha = rng.uniform(-1.5, 2.0)
        beta = rng.uniform(-1.5, 2.0)
        a = (idx + 1) ** alpha * (n - idx) ** beta
        a = np.maximum(a, 0.0)  # ensure non‑negative

    elif typ == "mixexp":
        # mixture of two exponentials placed at the ends
        lam = rng.uniform(0.001, 0.2)
        w = rng.random()
        left = np.exp(-lam * idx)
        right = np.exp(-lam * (n - 1 - idx))
        a = w * left + (1 - w) * right

    else:
        a = rng.random(n)

    # Small random jitter before projection
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.02, size=n)
        a = np.clip(a, 0.0, None)

    return a


def _hill_climb(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that conserves the ℓ₁‑norm.
    Small amounts of mass are moved from a source index to a destination index.
    The best vector seen during the run is returned together with its maximal
    convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # temperature starts mildly above the current excess over the trivial lower bound (≈1)
    T = max(cur_max - 1.0, 0.05)

    # Pre‑compute the total mass (should be S = √(2n))
    total = a.sum()

    # Main loop
    while time.time() < deadline:
        # Choose a source index with probability proportional to its mass.
        if total == 0:
            break
        src = rng.choice(n, p=a / total)
        # Choose a destination index uniformly at random (different from src)
        dst = rng.integers(0, n)
        if dst == src:
            continue

        # Move a random fraction (up to 20 %) of the source mass.
        max_move = a[src] * 0.2
        if max_move <= 0:
            continue
        delta = rng.uniform(0.0, max_move)

        a_test = a.copy()
        a_test[src] -= delta
        a_test[dst] += delta

        # Evaluate the new candidate.
        new_max = np.max(_full_convolution_fft(a_test))

        # Accept if better or with a Boltzmann probability.
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):
            a = a_test
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # Cool down slowly.
        T *= 0.99995

        # Occasionally re‑project to the simplex to wipe out accumulated rounding error.
        if rng.random() < 0.001:
            a = _project_to_simplex(a, total)

    return best_a, best_max


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point required by the harness.
    Performs a multi‑stage search (random families → hill‑climbing → Adam)
    within the allotted time budget and returns the best step‑function
    (list of non‑negative heights) found.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0
    end_time = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # initialise from a previously stored solution if available
    # ------------------------------------------------------------------
    best_seq = None
    best_max = np.inf
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = seq0.shape[0]
            S0 = np.sqrt(2.0 * n0)
            seq0 = _project_to_simplex(seq0, S0)
            cur_max = np.max(_full_convolution_fft(seq0))
            best_seq = seq0.copy()
            best_max = cur_max
            print(f"[init] imported previous solution: n={n0}, max={best_max:.6f}")
        except Exception as e:
            print("[init] failed to reuse previous solution:", e)

    # ------------------------------------------------------------------
    # Phase 1 – broad random sampling over many shapes and lengths
    # ------------------------------------------------------------------
    # allocate roughly 35 % of the remaining budget to this phase
    phase1_deadline = time.time() + (end_time - time.time()) * 0.35
    attempts = 0
    top_k = 5
    best_candidates = []   # list of (max, vector)

    while time.time() < phase1_deadline:
        attempts += 1
        n = rng.integers(500, 12000 + 1)      # moderate to large lengths
        raw = _random_distribution_extended(n, rng)
        S = np.sqrt(2.0 * n)
        a = _project_to_simplex(raw, S)
        cur_max = np.max(_full_convolution_fft(a))

        # Keep the global best
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
            print(f"[search] new global best at attempt {attempts}, n={n}, max={best_max:.6f}")

        # Maintain a short list of the best few candidates for hill‑climbing
        if len(best_candidates) < top_k:
            best_candidates.append((cur_max, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        elif cur_max < best_candidates[-1][0]:
            best_candidates[-1] = (cur_max, a.copy())
            best_candidates.sort(key=lambda x: x[0])

    print(f"[search] completed {attempts} random attempts; best max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – local simulated‑annealing hill‑climbing on the top candidates
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 0 and best_candidates:
        # allocate equally among the candidates, leaving a tiny buffer for the final phase
        per_candidate_budget = remaining * 0.7 / len(best_candidates)
        for idx, (cand_max, cand_vec) in enumerate(best_candidates):
            sub_deadline = time.time() + per_candidate_budget
            print(f"[hill] starting hill‑climb #{idx+1} (init max={cand_max:.6f})")
            cl_seq, cl_max = _hill_climb(cand_vec, sub_deadline, rng)
            if cl_max < best_max:
                best_max = cl_max
                best_seq = cl_seq.copy()
                print(f"[hill] improvement after climb #{idx+1}: new best max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine‑grained Adam optimisation on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time:
        print("[adam] starting final Adam refinement")
        refined_seq, refined_max = _adam_optimize(best_seq, end_time, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_seq.copy()
            print(f"[adam] improvement: refined max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list; also store it globally for reuse.
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        best_max = np.nan
    else:
        final_seq = best_seq.tolist()

    # we deliberately evaluate via the external black‑box to obtain the official score
    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_max:.6f}) length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Search algorithm for minimizing the maximal self‑convolution of a
non‑negative step function.

The algorithm has three stages:

1️⃣  Random sampling over many parametric families (exponential,
    power‑law, arcsine, beta‑type, mixed exponentials, …) with lengths
    ranging from a few hundred to over ten thousand.  Each candidate is
    projected onto the ℓ₁‑simplex {∑a = √(2·n)} and its maximal convolution
    entry is recorded.

2️⃣  Simulated‑annealing hill‑climbing on the K best candidates.  The
    procedure repeatedly moves a small amount of mass from one index to
    another while keeping the ℓ₁‑norm fixed.  A temperature schedule
    allows occasional uphill moves; the best intermediate vector is kept.

3️⃣  Final refinement of the current best vector with a projected‑gradient
    Adam optimiser that minimises a smooth soft‑max approximation of the
    max‑convolution.  After each Adam step the vector is projected back onto
    the simplex.

All stages respect the 1000 s time budget (a small safety margin is kept).
The resulting sequence is stored in the global variable `height_sequence_1`
and returned as a plain Python list.
"""

import time
import operator
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the ℓ₁‑simplex of sum S."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Compute gradient of τ·log∑exp(b/τ) where b = a*a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard max ⇒ sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal contribution (i=j) occurs only once -> gradient = 2*a_i
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights w ∝ exp(b/τ) (stable version)
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)          # shape (2n‑1,)

    # cross‑correlation a ⋆ w = Σ_j a_j w_{i+j}
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # length n

    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max approximation of max(a*a) using Adam.
    Returns the best vector found and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6                      # initial learning rate
    t = 0

    # initialise temperature for soft‑max
    _, _, cur_max = _gradient_softmax(a, tau=0.0)   # hard max just to get a scale
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        # ----- gradient step -------------------------------------------------
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # ----- evaluation ----------------------------------------------------
        b_new = _full_convolution_fft(a_new)
        cur_max_new = np.max(b_new)

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # ----- schedules -----------------------------------------------------
        tau = max(tau * 0.9995, 1e-7)           # cooling of the soft‑max temperature
        lr *= 0.9999                           # gradual learning‑rate decay

        # occasional tiny Gaussian jitter to escape plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Generate a random parametric distribution of length n (larger family)
# ----------------------------------------------------------------------
def _random_distribution_extended(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create an unnormalised non‑negative vector using a richer set of families."""
    typ = rng.choice(
        [
            "uniform", "exp", "symexp", "gaussian", "triangular",
            "power", "cosine", "arcsine", "beta", "mixexp"
        ]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "arcsine":
        # a_i ∝ ((i+1)*(n-i))^{-p}
        p = rng.uniform(0.2, 1.8)
        a = ((idx + 1) * (n - idx)) ** (-p)

    elif typ == "beta":
        # a_i ∝ (i+1)^{α} (n-i)^{β}
        alpha = rng.uniform(-1.5, 2.0)
        beta = rng.uniform(-1.5, 2.0)
        a = (idx + 1) ** alpha * (n - idx) ** beta
        a = np.maximum(a, 0.0)

    elif typ == "mixexp":
        # mixture of two exponentials located at the two ends
        lam = rng.uniform(0.001, 0.2)
        w = rng.random()
        left = np.exp(-lam * idx)
        right = np.exp(-lam * (n - 1 - idx))
        a = w * left + (1 - w) * right

    else:
        a = rng.random(n)

    # small random jitter before projection (helps diversity)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.02, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# Hill‑climbing (simulated annealing) that conserves the ℓ₁‑norm
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that transfers a tiny amount of mass
    between indices while keeping the total sum constant.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_max = np.max(_full_convolution_fft(a))
    best_a = a.copy()
    best_max = cur_max

    # temperature starts a little above the trivial lower bound (≈1)
    T = max(cur_max - 1.0, 0.05)

    total = a.sum()                     # should equal √(2·n)

    while time.time() < deadline:
        # source index ∝ current mass (larger entries move more often)
        src = rng.choice(n, p=a / total)
        dst = rng.integers(0, n)
        if dst == src:
            continue

        max_move = a[src] * 0.2          # move up to 20 % of source mass
        if max_move <= 0:
            continue
        delta = rng.uniform(0.0, max_move)

        a_test = a.copy()
        a_test[src] -= delta
        a_test[dst] += delta

        new_max = np.max(_full_convolution_fft(a_test))

        # accept if improvement or with Boltzmann probability
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):
            a = a_test
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # geometric cooling
        T *= 0.99995

        # occasional re‑projection to eliminate round‑off drift
        if rng.random() < 0.001:
            a = _project_to_simplex(a, total)

    return best_a, best_max

    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
