def subgradient_descent(a0, deadline, step_factor=0.2):

    a = a0.copy()
    n = len(a)
    target = sqrt(2*n)
    while time.time() < deadline:
        b = np.convolve(a, a)
        max_idx = np.argmax(b)
        max_val = b[max_idx]
        # compute subgradient
        # Note: b_len = 2n-1
        # for each i, partner index j = max_idx - i
        # if 0 <= j < n, then contribution
        # else zero.
        g = np.zeros_like(a)
        # vectorized:
        # we can compute partner indices as j = max_idx - np.arange(n)
        # valid mask
        idx = np.arange(n)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
        # for each i where mask true, term a[partner[i]]
        g[mask] += a[partner[mask]]
        # handle double count when i == partner (i.e., 2i == max_idx)
        double_mask = mask & (partner == idx)
        g[double_mask] += a[idx[double_mask]]  # add extra a_i (so total 2*a_i)
        # So g = a_{partner} + (partner==i) a_i
        # Actually we want 2*a_i when i==partner
        # Already added a_i, need another a_i
        # So g[double_mask] += a[idx[double_mask]] again.
        # Or maybe simpler: for all i, g_i = a_{k - i} if partner valid; plus a_i if partner==i.
        # Let's compute g = np.where(mask, a[partner], 0) + np.where(double_mask, a[idx], 0)
        g = np.where(mask, a[partner], 0.0) + np.where(double_mask, a[idx], 0.0)
        # This yields 2*a_i when i==partner.

        # Step
        # Determine step size based on max_val to gradually reduce.
        lr = step_factor * (max_val) / (np.linalg.norm(g) + 1e-12)  # scale
        a -= lr * g

        # enforce non-negativity and simplex
        a = np.maximum(a, 0)
        a = _project_to_simplex(a, target)

    return a

Now write final answer.assistantfinal**Solution Overview**

The goal is to find a non‑negative sequence `a` that minimises

```
E(a) = 2·n·max(a∗a) / ( Σ a_i )²
```

`Σ a_i` can be fixed – we project every candidate onto the ℓ¹‑simplex  

```
S_n = { a_i ≥ 0 , Σ a_i = √(2·n) } .
```

Because `Σ a_i` is constant, `E(a)` is exactly the maximal entry of the
autoconvolution `b = a∗a`.  
The algorithm therefore tries to **flatten the convolution**.

The search consists of three stages

1. **Exploratory families** – exponential, power‑law, symmetric edge‑heavy
   and triangular shapes are generated for several parameters and are
   projected onto `S_n`.  
2. **Sub‑gradient descent on the true max** – for a current vector we locate
   the index `k` where the convolution is maximal, build the sub‑gradient
   (which is simply a shifted copy of the vector), take a step opposite to
   that gradient and project back onto the simplex.  This directly reduces
   the current peak.
3. **Refinement by simulated‑annealing** – random mass transfers between
   two coordinates are accepted with the Metropolis rule; the temperature
   decays slowly.  The SA uses the fast FFT based routine for the maximal
   convolution value.

All heavy work (convolution, projection) is implemented with
vectorised NumPy / FFT code, so even 3000‑dimensional vectors finish
comfortably inside the 1000 s budget.

The best sequence seen during the whole run is returned as a plain
`list[float]`.  The final value printed by `evaluate_sequence` is the
objective that will be scored.

```python
"""
Fast search for a step‑function with minimal max‑autoconvolution.
The algorithm explores several parametric families, performs a
sub‑gradient descent on the exact max (projected onto the ℓ¹‑simplex)
and finishes with a simulated‑annealing refinement.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (computed by FFT)."""
    conv = _full_convolution_fft(a)
    return float(conv.max())


def _objective(a: np.ndarray) -> float:
    """
    Objective for a vector that already lies on the simplex Σa=√(2n).
    In that case `evaluate_sequence` returns exactly the maximal convolution.
    """
    return _max_convolution_fft(a)


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Diverse random start, projected onto the simplex Σa=√(2n)."""
    target = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                         # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                       # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                       # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                       # Dirichlet draw (already on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif kind == 4:                       # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                                 # uniform + jitter
        a = np.full(n, target / n)

    # small Gaussian perturbation, then project
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _parametric_exp(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i) (left‑heavy)."""
    a = np.exp(-lam * np.arange(n))
    return _project_to_simplex(a, np.sqrt(2.0 * n))


def _parametric_pow(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=float)) ** (-p)
    return _project_to_simplex(a, np.sqrt(2.0 * n))


def _parametric_sym_exp(n: int, lam: float) -> np.ndarray:
    """Symmetric edge‑heavy: a_i ∝ exp(-lam·min(i, n‑1‑i))."""
    dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-lam * dist)
    return _project_to_simplex(a, np.sqrt(2.0 * n))


def _parametric_triangle(n: int) -> np.ndarray:
    """Triangular shape, peak in the centre."""
    i = np.arange(n, dtype=float)
    a = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    a[a < 0] = 0.0
    return _project_to_simplex(a, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (projected onto the simplex)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Direct sub‑gradient steps on  max(a*a).  At each iteration we locate the
    index k where the convolution attains its maximum, build the sub‑gradient
    g_i = a_{k-i} (with a double contribution when i = k‑i) and move opposite
    to g, finally projecting back onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    idx = np.arange(n)

    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        k = int(np.argmax(conv))
        max_val = float(conv[k])

        partner = k - idx
        mask = (partner >= 0) & (partner < n)

        # g_i = a_{k-i}, extra a_i when i == k-i
        g = np.zeros_like(a)
        g[mask] = a[partner[mask]]
        double_mask = mask & (partner == idx)
        g[double_mask] += a[idx[double_mask]]       # second copy for i = partner

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break

        # step size proportional to current peak, shrinking with time
        # (a simple schedule works well)
        step = 0.5 * max_val / (norm_g + 1e-12)
        a -= step * g

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple Metropolis‑type SA that repeatedly transfers a random fraction
    of the mass of one coordinate to another.  The temperature starts
    proportionally to the current objective and decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature – 2 % of the current max value
    T = best_score * 0.02 + 1e-12
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move up to 15 % of the mass at i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        if cur_score <= best_score or (T > 0.0 and rng.random() < np.exp(-dE / T)):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  A mixture of parametric families, a dedicated
    sub‑gradient descent and a final simulated‑annealing phase are used.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None
    best_val = np.inf

    def _register(seq: np.ndarray):
        """Register a candidate, updating the global best if improved."""
        nonlocal best_seq, best_val
        val = _objective(seq)        # because Σa = √(2n) → objective = max conv
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] NEW BEST  val={best_val:.6f}  n={len(seq)}")

    # ------------------------------------------------------------------
    # If a previously discovered construction exists, start from it
    # ------------------------------------------------------------------
    init_global = globals().get("height_sequence_1")
    if isinstance(init_global, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_global, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        _register(a0)

    # ------------------------------------------------------------------
    # Lengths to try – a moderate range works well under the time budget
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2100, 2400]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # exponential decay
        for lam in np.linspace(0.0005, 0.0200, 20):
            a = _parametric_exp(n, lam)
            _register(a)

        # power‑law
        for p in np.linspace(0.1, 2.0, 20):
            a = _parametric_pow(n, p)
            _register(a)

        # symmetric edge‑heavy
        for lam in np.linspace(0.0005, 0.0200, 15):
            a = _parametric_sym_exp(n, lam)
            _register(a)

        # triangular
        a = _parametric_triangle(n)
        _register(a)

        # ------- 2. random starts ---------------------------------------
        for _ in range(12):
            a = _random_initial_sequence(n, rng)
            _register(a)

        # Keep the current best as a seed for a deeper optimisation pass.
        if best_seq is None:
            continue

        # ------- 3. sub‑gradient descent + SA refinement -----------------
        # allocate remaining time proportionally to the size of the current problem
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # half of the remaining time for sub‑gradient, the rest for SA
        sub_dead = time.time() + max(4.0, remaining * 0.45)
        a_sub = _subgradient_descent(best_seq, sub_dead)
        _register(a_sub)

        if time.time() < deadline:
            a_refined = _simulated_annealing(a_sub, deadline)
            _register(a_refined)

        # ---------------------------------------------------------------
        # clear any temporary data – move on to next length
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_val = _objective(best_seq)

    # Trim trailing near‑zero entries for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final sanity check using the official evaluator (cheap)
    try:
        val_check = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val_check:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_val:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
```
