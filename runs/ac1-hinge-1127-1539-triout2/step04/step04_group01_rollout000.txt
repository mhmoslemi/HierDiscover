"""
Step‑function optimiser (improved)

The optimiser searches for a non‑negative vector *a* of length *n* that
minimises the quantity

    evaluate_sequence(a) = max_k (a * a)[k]

subject to the ℓ¹‑normalisation  Σ a_i = √(2 n).

Key ideas
---------

* **Projection on the ℓ¹‑simplex** – all iterates stay feasible.
* **Smooth‑max gradient** – use the log‑sum‑exp approximation of the
  maximum with an annealed inverse temperature β.
* **Adam‑style adaptive steps** – per‑coordinate learning‑rates
  accelerate convergence compared with naive momentum.
* **Continuation in the length** – if a good construction for a smaller
  length exists (available as ``height_sequence_1``) it is linearly
  interpolated to the target length and refined.
* **Final mass‑transfer SA** – a cheap simulated‑annealing phase
  explores the neighbourhood of the best candidate.

The whole routine respects a wall‑clock budget (default 1000 s) and
returns a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full mode) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) – the core of the Adam step
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = ∇_a  ( 1/β·log Σ exp(β·b) ) = 2·(a ⋆ p)
    where p_k ∝ exp(β·b_k) and ⋆ denotes correlation.
    """
    n = a.shape[0]

    # ----------- b = a*a ---------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----------- soft‑max probabilities p -----------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))      # stability
    p = exps / np.sum(exps)

    # ----------- gradient g = 2·(a ⋆ p) -------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation of the smoothed max‑convolution.
    The loop stops when ``time.time()`` reaches ``deadline``.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # learning‑rate – a small fraction of the ℓ¹‑norm
    base_lr = target_sum * 0.08

    # temperature (inverse) schedule for the smooth‑max
    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5
    beta_step = 250      # increase β every this many iterations

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        # gradient of the smooth maximum
        g, _ = _gradient_softmax(a, beta)

        # Adam update (pre‑projection)
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # keep feasibility
        a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % beta_step == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # occasional tiny jitter to shake off plateaus
        if it % 500 == 0:
            jitter = np.random.default_rng().normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing (mass‑transfer moves) for polishing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Mass‑transfer SA that repeatedly moves a random fraction of mass
    from one index to another.  Accepts uphill moves with probability
    exp(-Δ/T) where the temperature decays geometrically.
    Returns the best sequence observed before ``deadline``.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    # initial temperature – a few per‑cent of the current score
    T = cur_score * 0.025 + 1e-9
    cooling = 0.9993

    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction of the mass from i to j
        frac = rng.random() * 0.20
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers on the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the simplex
    Σ a = √(2 n).  Several families are mixed so that the search explores
    both dense and sparse candidates.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.1, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.012)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.1, 1.0)
        lam = rng.exponential(scale=0.012)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                   # sparse spikes – K random positions
        K = max(2, int(n * rng.uniform(0.01, 0.05)))
        a = np.zeros(n)
        pos = rng.choice(n, size=K, replace=False)
        vals = rng.random(K) + 0.1
        a[pos] = vals

    elif family == 5:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.018)
        lam2 = rng.exponential(scale=0.018)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # uniform plus tiny jitter
        a = np.full(n, target / n)

    # add a little Gaussian noise, enforce non‑negativity and project
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Helper – resize an existing sequence (linear interpolation)
# ----------------------------------------------------------------------


def _resize_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate an old sequence to length ``new_n``."""
    if old.size == new_n:
        return old.copy()
    old_x = np.linspace(0, 1, old.size)
    new_x = np.linspace(0, 1, new_n)
    new = np.interp(new_x, old_x, old)
    # after interpolation the ℓ¹‑norm is changed – re‑project
    new = _project_to_simplex(new, np.sqrt(2.0 * new_n))
    return new
