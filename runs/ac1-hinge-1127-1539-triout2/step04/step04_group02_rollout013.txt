"""
Optimisation of a non‑negative step function for the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

Because the sum is fixed (the evaluation routine is homogeneous),
the objective is simply the maximal entry of the self‑convolution.
The code below searches an n‑dimensional vector `a ≥ 0`,
∑a = √(2 n), that minimises this maximum.

Algorithm
---------
0️⃣  Determine the problem size (use the supplied `height_sequence_1`
    if present, otherwise n = 32768).

1️⃣  **Broad initial sweep** – a collection of parametric families
    (“U‑shapes”, asymmetric linear/exponential skews, cosine‑U,
    logarithmic‑U, power‑U) together with many random Dirichlet draws.
    Every candidate is projected onto the ℓ₁‑simplex and its maximal
    convolution is recorded.

2️⃣  **Many short Adam‑style sub‑gradient runs** – the sub‑gradient of
    `max(conv)` w.r.t. `a` is obtained by convolving the mask of
    maximal lags with the reversed vector.  Starting from a random
    feasible point we run a few hundred Adam updates (first‑order
    method for non‑smooth convex‑like problems) and keep the best
    local optimum.  The loop is repeated with fresh random starts
    until the allotted time is exhausted.  This gives a wide
    exploration of the landscape while still allowing fast convergence
    thanks to adaptive step‑sizes.

3️⃣  **Polishing (simulated‑annealing style)** – from the best vector
    found so far we repeatedly perform tiny random pair‑mass transfers
    (the `random_pair_transfer` helper) with a temperature that
    decays linearly to zero.  Each proposal is accepted only if it
    strictly lowers the maximal convolution, guaranteeing monotonic
    improvement.

All operations respect the non‑negativity and ℓ₁‑normalisation
constraints via the Euclidean projection onto the simplex.
The programme stops well before the 1000 s budget and returns the
best sequence discovered as a plain Python list.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of `a` with itself using an FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another
    and re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence.
    Supported `kind`:
        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right,
        exponential_left, exponential_right,
        cosine, log.
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    elif kind == "cosine":
        # cosine‑U: low in the middle, high at the ends
        raw = 1.0 + param * (1.0 - np.cos(np.pi * d))
    elif kind == "log":
        raw = 1.0 + param * np.log1p(d)  # log(1+d) grows slowly
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the non‑negative self‑convolution."""
    return float(conv_self_fft(a).max())


def subgradient(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of `max(conv(a,a))` given the mask of the
    (approximately) maximal lags.  The gradient is
        2 / |mask| * (mask   *   a_rev)   (central n entries)
    where * denotes linear convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((mask.size + a_rev.size - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: mask.size + a_rev.size - 1]
    cnt = mask.sum()
    if cnt == 0:
        cnt = 1.0
    grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    `evaluate_sequence` objective.  The routine respects the 1000 s
    time budget and returns the best vector discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25            # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        a0 = np.asarray(base, dtype=np.float64)
        a0 = project_onto_simplex(a0, math.sqrt(2.0 * n))
    else:
        n = 32768            # a comfortable power‑of‑two length
        a0 = None

    target_sum = math.sqrt(2.0 * n)   # Σ a_i = √(2 n) (normalisation)

    # --------------------------------------------------------------
    # Phase 0 – diverse parametric sweep + random sketches
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)   # uniform start
    best_score = compute_max_conv(best_seq)

    shape_kinds = (
        "linear", "quadratic", "exponential",
        "power", "inverse",
        "linear_left", "linear_right",
        "exponential_left", "exponential_right",
        "cosine", "log",
    )
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),
        "quadratic": np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 12),
        "inverse": np.linspace(0.5, 5.0, 12),
        "linear_left": np.linspace(0.0, 0.9, 19),
        "linear_right": np.linspace(0.0, 0.9, 19),
        "exponential_left": np.linspace(0.0, 6.0, 31),
        "exponential_right": np.linspace(0.0, 6.0, 31),
        "cosine": np.linspace(0.0, 4.0, 21),
        "log": np.linspace(0.0, 4.0, 21),
    }

    print(f"[{time.time():.2f}] Phase 0 – parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # a few Dirichlet‑like random vectors
    for _ in range(30):
        cand = np.random.rand(n)
        cand = project_onto_simplex(cand, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best random init → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during random init.")
            return best_seq.tolist()

    # consider supplied baseline if present
    if a0 is not None:
        score = compute_max_conv(a0)
        if score < best_score:
            best_score = score
            best_seq = a0.copy()
            print(f"  Baseline improves start → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – many short Adam sub‑gradient runs (exploration)
    # --------------------------------------------------------------
    subgrad_deadline = start_time + budget_s * 0.80
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    base_lr = 0.6 * (target_sum / n)          # moderate step size
    iteration = 0
    no_improve_counter = 0

    print(f"[{time.time():.2f}] Phase 1 – Adam sub‑gradient exploration")
    while time.time() < subgrad_deadline:
        # fresh random start for each short run
        a = np.random.rand(n)
        a = project_onto_simplex(a, target_sum)

        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # a few hundred Adam steps; stop early if time runs out
        for inner in range(400):
            if time.time() >= subgrad_deadline:
                break

            # current convolution & mask of maximal entries
            conv = conv_self_fft(a)
            max_val = conv.max()
            eps = max(1e-12, max_val * 1e-9)
            mask = (conv >= max_val - eps).astype(np.float64)
            if mask.sum() == 0:
                mask[conv.argmax()] = 1.0

            grad = subgradient(a, mask)

            iteration += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = a - step
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a, target_sum)

            # evaluate
            cur_score = compute_max_conv(a)
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve_counter = 0
                print(f"Iter {iteration:07d} – improved to {best_score:.6f}")
            else:
                no_improve_counter += 1

            # occasional jitter to avoid stagnation
            if np.random.rand() < 0.003:
                a = random_pair_transfer(a, target_sum, scale=0.008)

        # after a short run, decay the learning rate a little
        base_lr *= 0.95

        # early exit if we have very little time left
        if time.time() + 1.0 >= subgrad_deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – final polishing (simulated‑annealing style)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 2 – polishing")
    # temperature linearly decays to zero
    while time.time() < deadline:
        remaining = deadline - time.time()
        # scale proportional to remaining fraction of total budget
        temp = 0.02 * (remaining / (budget_s - (deadline - start_time)))
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=temp)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
