def _gradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    s = np.sqrt(2.0*n)

    beta = 0.5
    max_beta = 1e8
    momentum = np.zeros_like(a)
    rho = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum
        momentum = rho * momentum + (1 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            # step size schedule decays with iteration
            lr = 0.8 * s / (norm * (beta + 1.0) * np.sqrt(it + 1))
            a -= lr * momentum

        # projection
        a = _project_to_simplex(a, s)

        # jitter
        if it % 300 == 0:
            jitter_scale = s * 0.0005 / (beta+1.0)
            jitter = rng.normal(scale=jitter_scale, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, s)

        if it % 250 == 0 and beta < max_beta:
            # increase beta more aggressively as we converge
            beta = min(beta * 1.2, max_beta)

        it += 1
    return a

def _simulated_annealing(a0, deadline, rng):

   a = a0.copy()
   n = a.shape[0]
   best = a.copy()
   best_score = _max_convolution_fft(best)

   T = best_score * 0.05 + 1e-8
   cooling = 0.9995

   while time.time() < deadline:
       # choose i weighted by a
       probs = a / a.sum()
       i = rng.choice(n, p=probs)

       # pick j random distinct
       j = rng.integers(0, n)
       while j == i:
           j = rng.integers(0, n)

       # move random fraction up to 0.2 of a[i]
       frac = rng.random() * 0.2
       delta = a[i] * frac
       if delta == 0.0:
           continue

       a[i] -= delta
       a[j] += delta

       cur_score = _max_convolution_fft(a)
       dE = cur_score - best_score
       accept = False
       if cur_score <= best_score:
           accept = True
       else:
           if rng.random() < np.exp(-dE / T):
               accept = True

       if accept:
           if cur_score < best_score:
               best_score = cur_score
               best = a.copy()
               # print improvement
               print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
       else:
           # revert
           a[j] -= delta
           a[i] += delta

       T *= cooling

       # occasional global jitter
       if rng.random() < 0.001:
           jitter = rng.normal(scale=a.sum() * 0.0005, size=n)
           a += jitter
           a = np.maximum(a, 0.0)
           a = _project_to_simplex(a, a.sum())

           cur_score = _max_convolution_fft(a)
           if cur_score < best_score:
               best_score = cur_score
               best = a.copy()
               print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

   return best

def _random_initial_sequence(n, rng):
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 8)  # 8 families

    idx = np.arange(n, dtype=float)

    if kind == 0:
        # uniform
        a = np.full(n, target_sum / n)
    elif kind == 1:
        # power law decreasing
        p = rng.uniform(0.1, 1.2)
        a = (idx + 1) ** (-p)
    elif kind == 2:
        # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)
    elif kind == 3:
        # power + exp
        p = rng.uniform(0.1, 1.2)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1) ** (-p) * np.exp(-lam * idx)
    elif kind == 4:
        # symmetric edge heavy
        p = rng.uniform(0.2, 1.5)
        left = (idx + 1) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right
    elif kind == 5:
        # half-sine shape
        a = np.sin(np.pi * (idx + 0.5) / n)
    elif kind == 6:
        # arcsine shape: 1/sqrt(i*(n-1-i)+epsilon)
        eps = 1e-3
        a = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5) + eps)
    elif kind == 7:
        # sqrt decrease: 1/sqrt(i+1)
        a = 1.0 / np.sqrt(idx + 1)
    else:
        a = np.full(n, target_sum / n)

    # add small jitter
    a += rng.normal(scale=target_sum * 0.001, size=n)
    # clip negative
    a = np.maximum(a, 0.0)
    # project onto simplex
    a = _project_to_simplex(a, target_sum)
    return a

"""
Propose a non‑negative step‑function (vector) that minimises the competition
objective.  The objective after normalising the ℓ¹‑norm to sqrt(2 n) is simply
the maximum entry of the self‑convolution a*a.  The search therefore tries to
flatten that convolution.

Algorithm
---------
1. **Diverse initialisations** – several parametric families (uniform,
   power‑law, exponential, mixed, symmetric edge‑heavy, half‑sine,
   arcsine‑like, sqrt‑decrease) are sampled and projected onto the simplex
   Σa = √(2 n).

2. **Projected sub‑gradient descent** – the smooth log‑sum‑exp surrogate of the
   maximal convolution entry is minimised.  The smoothing parameter β is
   annealed from ≈1 to 1e8, momentum is used and after each step the vector is
   projected back onto the simplex.

3. **Targeted simulated‑annealing** – mass is repeatedly transferred between
   indices (biased towards large entries) while accepting uphill moves with a
   geometric temperature schedule.  Occasional global jitter helps escape
   deep minima.

The whole loop respects the user‑supplied time budget (default 1000 s), keeps
the best sequence seen, and stops early if the target value 1.503 is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the log‑sum‑exp smooth approximation of max(a*a).

    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) ) = 2·(a ⋆ p)
    with p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentials
    p = exps / np.sum(exps)

    # gradient g = 2·(a_rev ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]         # central n entries
    return g, b


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Duchi et al. (2008) Euclidean projection onto the ℓ¹‑simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (a few diverse families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative vector on the simplex Σa = √(2 n)."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 8)          # eight families
    idx = np.arange(n, dtype=float)

    if kind == 0:                       # uniform
        a = np.full(n, target_sum / n)

    elif kind == 1:                     # decreasing power‑law
        p = rng.uniform(0.1, 1.2)
        a = (idx + 1.0) ** (-p)

    elif kind == 2:                     # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif kind == 3:                     # power‑law × exponential
        p = rng.uniform(0.1, 1.2)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 4:                     # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.5)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 5:                     # half‑sine (positive)
        a = np.sin(np.pi * (idx + 0.5) / n)

    elif kind == 6:                     # arcsine‑like 1/√(i·(n‑1‑i))
        eps = 1e-3
        a = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5) + eps)

    elif kind == 7:                     # sqrt‑decrease 1/√i
        a = 1.0 / np.sqrt(idx + 1.0)

    else:                               # fallback (should never hit)
        a = np.full(n, target_sum / n)

    # a small Gaussian jitter and projection back on the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Projected (sub)‑gradient descent on the smoothed max convolution
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp surrogate of
    max(a*a).  The schedule drives β → 1e8, uses momentum and adaptive
    step‑sizes, and re‑projects onto the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    max_beta = 1e8
    rho = 0.95                         # momentum decay
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            # step size slowly decays with iteration count
            lr = 0.8 * target_sum / (norm_m * (beta + 1.0) * np.sqrt(it + 1))
            a -= lr * momentum

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (more aggressive than before)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> np.ndarray:
    """
    Simple SA that moves a random fraction of mass from a heavy index
    to a random destination.  Acceptance follows a geometric temperature
    schedule.  The best vector seen is returned.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature proportional to current objective
    T = best_score * 0.04 + 1e-8
    cooling = 0.9995

    while time.time() < deadline:
        # pick source index biased by current mass
        probs = a / a.sum()
        i = rng.choice(n, p=probs)

        # pick a different destination uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction (up to 15 %) of the source mass
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to shake out of deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, explores several vector lengths,
    performs projected gradient descent followed by a targeted simulated‑
    annealing refinement, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # If a previous construction exists, start from it
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        obj0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_obj = obj0
        print(f"[{time.time():.1f}] Starting from height_sequence_1 (n={n0})  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Lengths to sample – a spread that balances quality and speed
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2100, 2400, 3000, 3500, 4000]

    iteration = 0
    while time.time() < deadline:
        # Pick a length (biased towards larger sizes)
        n = rng.choice(lengths)
        a_start = _random_initial_sequence(n, rng)

        remaining = deadline - time.time()
        if remaining < 5.0:
            # Not enough time for a full refinement, break out
            break

        # ------------------------------------------------------------------
        # Phase 1 – projected gradient descent (≈60 % of the remaining time)
        # ------------------------------------------------------------------
        grad_time = min(remaining * 0.55, max(1.0, remaining - 2.0))
        gd_dead = time.time() + grad_time
        a_gd = _gradient_descent_opt(a_start, gd_dead, rng)

        # ------------------------------------------------------------------
        # Phase 2 – simulated‑annealing refinement (use whatever time left)
        # ------------------------------------------------------------------
        a_ref = _simulated_annealing(a_gd, deadline, rng)

        # ------------------------------------------------------------------
        # Evaluate and possibly update the global best
        # ------------------------------------------------------------------
        obj = _max_convolution_fft(a_ref)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_ref.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (iter {iteration}) n={n}  obj={best_obj:.6f}")
            if best_obj <= 1.503:
                print(f"[{time.time():.1f}] Target reached – stopping early.")
                break

        iteration += 1

    # ------------------------------------------------------------------
    # Final pass: polish the current best with a longer gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_dead = deadline
        a_polish = _gradient_descent_opt(best_seq, final_dead, rng)
        a_polish = _simulated_annealing(a_polish, deadline, rng)
        final_obj = _max_convolution_fft(a_polish)
        if final_obj < best_obj:
            best_obj = final_obj
            best_seq = a_polish.copy()
            print(f"[{time.time():.1f}] FINAL polish improved to {best_obj:.6f}")

    # Trim trailing near‑zero entries for a neat output
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check with the official evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max conv) = {best_obj:.6f} "
          f"length = {best_seq.size if best_seq is not None else 'N/A'}")
    return best_seq.tolist()
