def _spike_initial_sequence(n, rng):

    total = np.sqrt(2.0 * n)
    K = rng.integers(20, min(150, n//10))   # at most n//10
    positions = rng.choice(n, size=K, replace=False)

    jitter = np.maximum(jitter, -a[positions]) # ensure nonnegative
    a[positions] += jitter
    # re-normalize to target sum
    a = _project_to_simplex(a, total)

def _adam_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0
    max_beta = 1e5

"""
Propose a non‑negative step‑function (list of heights) that minimises
evaluate_sequence().

Algorithm
---------
1.  For several vector lengths (1200 … 8000) generate a diverse pool of
    random initial vectors:
        * power‑law, exponential, Dirichlet, symmetric‑edge, sinusoid,
          parabola and a “spike” construction (few large Dirac spikes).
    All vectors are projected onto the ℓ¹‑simplex
          Σ a_i = sqrt(2 n).

2.  Keep the best few seeds (by the fast FFT max‑convolution value).

3.  For each elite seed run a two‑phase local optimiser
        * Adam‑style projected gradient descent on the smooth log‑sum‑exp
          surrogate of the maximum convolution entry (β is slowly annealed).
        * A short simulated‑annealing (SA) refinement that moves small
          fractions of mass between indices and occasionally jitters the
          whole vector.

   Each phase respects the remaining time budget.

4.  The best vector found over all lengths is trimmed (trailing near‑zero
   entries removed) and returned as a plain Python list.

All heavy work is performed with FFT‑based convolutions (O(n log n)),
simplex projection (Euclidean) and does not use any file or network I/O.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a ------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentials
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]         # central n values
    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smoothed max
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max convolution.
    The iterate is projected onto the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    beta = 1.0                     # smoothing parameter for the surrogate
    max_beta = 1e5
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – adapt to the norm of the vector
        step = (target_sum * 0.4) / (beta + 1.0) / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # anneal the smoothness parameter
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that moves a random fraction of mass between two indices.
    Acceptance of uphill moves follows exp(−Δ/T). Temperature decays
    multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-8        # initial temperature
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick donor index with non‑zero mass
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (≤15 %) of donor's mass
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best

# Random initialisers (including a spike family)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a diverse random non‑negative vector on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if kind == 0:               # power‑law decreasing
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:             # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:             # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:             # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    elif kind == 4:             # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 5:             # half‑sine shape
        a = np.sin(np.pi * idx / (n - 1))

    elif kind == 6:             # parabola (peak in the middle)
        a = 1.0 - ((idx - (n - 1) / 2.0) / ((n - 1) / 2.0)) ** 2
        a = np.maximum(a, 0.0)

    else:                       # uniform with tiny jitter
        a = np.full(n, target_sum / n)

    # small Gaussian tweak and projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


def _spike_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Sparse “spike” vector: a few Dirac‑like masses placed at random, then
    projected onto the simplex.  The number of spikes K is chosen between
    20 and min(150, n//10).
    """
    target_sum = np.sqrt(2.0 * n)
    K = int(rng.integers(20, min(150, max(2, n // 10) + 1)))
    positions = rng.choice(n, size=K, replace=False)
    weight = target_sum / K
    a = np.zeros(n)
    a[positions] = weight

    # tiny random perturbation of the spike weights (3 % rms)
    jitter = rng.normal(scale=weight * 0.03, size=K)
    jitter = np.clip(jitter, -a[positions], None)   # keep non‑negative
    a[positions] += jitter

    a = _project_to_simplex(a, target_sum)
    return a

def _optimise_candidate(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Run Adam‑style projected gradient descent followed by simulated‑annealing.
    The function respects the supplied deadline and returns the refined
    vector.
    """
    now = time.time()
    remaining = max(0.0, deadline - now)
    if remaining < 5.0:
        return a0

    # allocate 60 % of the remaining budget to Adam, the rest to SA
    adam_time = remaining * 0.60
    sa_time = remaining - adam_time - 0.1   # tiny safety margin

    a = _adam_opt(a0, now + adam_time)

    if sa_time > 2.0:
        a = _simulated_annealing(a, now + adam_time + sa_time)

    return a
