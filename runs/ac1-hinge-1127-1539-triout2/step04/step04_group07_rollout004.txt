def _local_optimize_subgrad(a0, deadline, rng):

    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    lr = 0.5
    lr_decay = 0.998
    swap_prob = 0.1
    target_prob = 0.2
    eps_target = 0.05

    it = 0
    last_improve = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()

        # subgradient step
        # Determine active indices within tolerance
        # Use tolerance relative
        tol = 1e-12  # maybe considered
        active_idxs = np.where(conv >= cur_max - 1e-12)[0]  # exact
        if active_idxs.size == 0:
            active_idxs = np.array([np.argmax(conv)])
        # weight vector
        w = np.zeros_like(conv)
        w[active_idxs] = 1.0 / active_idxs.size
        grad = _conv_with_weights(a, w)
        a = a - lr * grad
        a = _project_simplex(a, S)

        # evaluate new max
        conv = _full_convolution(a)
        cur_max = conv.max()

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # random swap
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = a[i] * (rng.random() * 0.05)  # fraction up to 5%
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                new_max = np.max(_full_convolution(cand))
                # accept if better or with probability
                if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(cur_max*0.02, 1e-12)):
                    a = cand
                    cur_max = new_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # targeted move similar to original
        if rng.random() < target_prob:
            # take k_max
            k_max = int(np.argmax(conv))
            # indexes i_range where i in [max(0, k_max - (n-1)), min(k_max, n-1)]
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]
            src_idx_rel = np.argmax(contributions)
            tgt_idx_rel = np.argmin(contributions)
            i_src = i_range[src_idx_rel]
            i_tgt = i_range[tgt_idx_rel]
            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max*0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        lr *= lr_decay

            # maybe increase lr a bit to escape
            lr = min(lr*1.1, 1.0)  # or break
            if lr < 1e-6:

"""
Propose a non‑negative step function that minimises
evaluate_sequence().  The algorithm mixes several ideas
that go beyond the reference implementation:

*   Warm‑starts from a variety of parametric families
    (exponential, power‑law, Dirichlet‑random) and from the
    best solution found so far (with a tiny jitter).

*   A fast projected sub‑gradient method that directly
    targets the true ∞‑norm.  For the current maximal
    convolution entries we build a weight vector w that is
    1 on the active indices and 0 elsewhere; the gradient
    of the soft‑max is replaced by the exact sub‑gradient
    (average over the active set).  Each step is projected
    onto the ℓ₁‑simplex Σa = √(2 n).

*   Occasional stochastic “mass‑swap’’ moves (to escape
    local minima) and cheap targeted moves that transfer
    mass from the largest contributing pair to the smallest
    one for the current maximal convolution index.

*   A simple annealing schedule for the learning rate and
    aggressive early‑stop when no improvement has been seen
    for many iterations.

The whole search runs for the allotted time (default 1000 s),
keeping the best sequence encountered.  At the end the
sequence is returned as a plain list of floats and its
external score is printed.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
# Scipy is available in the environment and provides a fast LP solver,
# but the core optimisation uses only numpy/FFT for speed.
# from scipy.optimize import linprog  # (kept as a reference, not used)

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = rfft(a, n=fft_len)
    conv = irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·(a ⋆ w) where ⋆ denotes correlation.
    a has length n, w has length 2·n‑1.
    Result is a vector of length n.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = rfft(a, n=fft_len)
    Fw = rfft(w_rev, n=fft_len)
    full = irfft(Fa * Fw, n=fft_len)[:conv_len]
    # indices that give Σ_i w_{i+j}·a_i for each j
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Families for warm‑starts
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """Decreasing exponential of length n; k_factor controls the decay."""
    S = np.sqrt(2.0 * n)
    lam = k_factor / n                     # λ = k_factor / n
    q = np.exp(-lam)                       # q = e^{-λ}
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)   # = 1 - q**n
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)
    return a


def _powerlaw_sequence(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-exponent} (monotonically decreasing)."""
    S = np.sqrt(2.0 * n)
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-exponent)
    a = _project_simplex(a, S)
    return a


def _random_dirichlet_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex via Gamma(1) (i.e. Dirichlet)."""
    S = np.sqrt(2.0 * n)
    a = rng.gamma(shape=1.0, size=n)
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Local optimiser – sub‑gradient + swaps
# ----------------------------------------------------------------------


def _local_optimize_subgrad(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using a pure sub‑gradient method on the
    true max‑convolution, together with random swaps and a cheap
    targeted move.  Returns the best sequence seen and its
    (exact) maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (chosen after a few manual experiments)
    lr = 0.5
    lr_decay = 0.9995
    swap_prob = 0.12
    swap_frac = 0.05
    target_prob = 0.18
    eps_target = 0.04        # fraction of source mass moved in a targeted move
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- evaluate current convolution ----
        conv = _full_convolution(a)
        cur_max = conv.max()

        # ---- sub‑gradient step on the exact max ----
        # active set: indices where the max is attained (within a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= cur_max - tol)[0]
        if active.size == 0:                     # safety fallback
            active = np.array([np.argmax(conv)])

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size             # uniform weights over the active set
        grad = _conv_with_weights(a, w)
        a = a - lr * grad
        a = _project_simplex(a, S)

        # ---- record improvement ----
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # ---- occasional random mass‑swap (simulated‑annealing flavour) ----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis‑type acceptance
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                             max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- cheap targeted move aimed at the current maximal index ----
        if rng.random() < target_prob:
            k_max = int(np.argmax(conv))

            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            src_rel = np.argmax(contributions)   # largest contribution
            tgt_rel = np.argmin(contributions)   # smallest contribution
            i_src = i_range[src_rel]
            i_tgt = i_range[tgt_rel]

            if a[i_src] > 0.0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                             max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- learning‑rate schedule ----
        lr = max(lr * lr_decay, 1e-6)

        # ---- early exit if stagnated for a long while ----
        if it - last_improve > 8000:
            break

    return best_a, best_score
