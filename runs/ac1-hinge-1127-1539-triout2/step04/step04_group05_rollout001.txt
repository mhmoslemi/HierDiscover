"""
Propose a step‑function (non‑negative sequence) minimizing the
self‑convolution L∞‑norm under a fixed ℓ₁‑norm.
The algorithm mixes several ideas:

*  Simple feasible initialisers (Dirichlet, exponential, piecewise)
   that are projected onto the simplex {x ≥ 0, Σx = √(2·n)}.
*  A fast FFT‑based convolution to evaluate the objective.
*  A hybrid local optimiser:
      – a smooth soft‑max approximation of the max‑convolution,
        with back‑tracking line‑search;
      – an exact sub‑gradient step on the active convolution entries;
      – occasional random mass‑swap moves (simulated‑annealing style);
      – Gaussian jitter.
  All updates are projected back onto the simplex.
*  A multi‑phase search:
      1.  Refine a previously discovered sequence (if any).
      2.  Run many independent restarts with varied lengths (up to 50 000).
      3.  Use the best discovered sequence for a final long‑run fine‑tune,
          optionally enlarging its length.

The routine respects a 1000 s time budget, prints progress,
stores the best sequence in the global ``height_sequence_1`` and finally
returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass S."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence scaling (linear interpolation)
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.astype(float).copy()
    if old_n == 1:
        return np.full(new_n, seq[0], dtype=float)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq).astype(float)


# ----------------------------------------------------------------------
# Feasible initialiser
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with Σ = √(2·n)."""
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size > 0:
        a = _scale_sequence(base_seq, n)
        a += rng.normal(scale=max(a.mean(), 1e-12) * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        # random shape – pick one of three simple families
        shape = rng.integers(0, 3)
        if shape == 0:                     # plain Dirichlet(1)
            a = rng.random(n)
        elif shape == 1:                   # exponential decay
            r = rng.uniform(0.96, 0.999)
            a = r ** np.arange(n, dtype=float)
        else:                               # piecewise two‑level
            half = n // 2
            a = np.concatenate([rng.random(half) * 0.5,
                                rng.random(n - half) * 1.5])

    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    """
    Returns the sub‑gradient of max(a*a) and the corresponding maximal entry.
    ``eps`` treats entries within max‑eps as active.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)

    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)          # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]            # slice length n
    return g, max_b

        phase_deadline = min(time.time() + 180.0, end_time)   # up to 3 min
        seq_opt, score_opt = _local_optimize(a0, phase_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            best_n = n0
            print(f"[phase 1] refined length={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many independent restarts with varied lengths
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempts += 1

        # Choose a problem size. Larger sizes give more degrees of freedom
        # but cost more time; we mix a few ranges.
        rnd = rng.random()
        if rnd < 0.55:
            n = rng.integers(800, 4000)          # moderate
        elif rnd < 0.85:
            n = rng.integers(4000, 15000)        # larger
        else:
            n = rng.integers(15000, 50000)       # very large (capped)

        a0 = _initial_sequence(base_seq, n, rng)

        # give each restart a short time slice (≈8 s) while respecting the
        # global deadline.
        slice_deadline = min(end_time, time.time() + 8.0)
        seq_opt, score_opt = _local_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            best_n = n
            print(f"[global] attempt={attempts} n={n} new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final long‑run fine‑tune on the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time:
        # Occasionally enlarge the best sequence to give the optimiser
        # more room to spread the mass.
        if rng.random() < 0.25:
            new_n = int(best_seq.shape[0] * 1.5)
            new_n = min(new_n, 50000)
            if new_n != best_seq.shape[0]:
                a0 = _scale_sequence(best_seq, new_n)
                a0 = _project_to_simplex(a0, np.sqrt(2.0 * new_n))
            else:
                a0 = best_seq.copy()
        else:
            a0 = best_seq.copy()

        # Use the remaining budget for a thorough optimisation.
        final_deadline = end_time
        seq_opt, score_opt = _local_optimize(a0, final_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final‑tune] length={best_n} max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the black‑box function
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                      # safeguard
    else:
        final_seq = best_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] external score = {external_score:.6f} "
              f"(internal max = {best_score:.6f}) length = {len(final_seq)}")
        # store for the next run
        globals()['height_sequence_1'] = final_seq

    return final_seq
