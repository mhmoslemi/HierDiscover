"""
Step‑function optimiser (v2).

The optimiser builds on the previous gradient‑descent + SA pipeline,
but adds a dedicated “max‑shave’’ phase that performs a
sub‑gradient descent on the *exact* ℓ∞‑norm of the self‑convolution.
A higher final inverse‑temperature (β) and a more aggressive learning‑rate
schedule are also used.  The search explores several lengths
(2000 … 10 000) and keeps the best sequence found within the
1000 s wall‑clock budget.

The entry point is ``propose_candidate`` – the only name that will be
called by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT‑based)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp) – used in the long GD phase
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = 2·(a ⋆ p) ,   p_k ∝ exp(β·b_k)   (soft‑max probabilities)
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # softmax probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar = np.fft.rfft(a_rev, n=size2)
    Pr = np.fft.rfft(p, n=size2)
    corr = np.fft.irfft(Ar * Pr, n=size2)[:conv_len]
    g = 2.0 * corr[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Uses momentum, an adaptive step‑size and a slowly annealed β.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 65536.0                # more aggressive final temperature
    beta_factor = 1.5
    step_base = target_sum * 0.25     # a bit larger initial step
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *exact* maximum (max‑shave phase)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Compute a (sub‑)gradient of the ℓ∞‑norm of the self‑convolution
    at the current maximum.  The gradient points *up* the max; we will
    move in the opposite direction.
    """
    n = a.size
    conv = np.convolve(a, a)
    m = conv.max()
    # indices achieving (or almost) the maximum
    eps = 1e-14
    idx = np.flatnonzero(conv >= m - eps)
    p = np.zeros_like(conv)
    if idx.size == 0:               # should not happen
        p[conv.argmax()] = 1.0
    else:
        p[idx] = 1.0 / float(idx.size)   # uniform weight over all ties

    # correlation a_rev ⋆ p  (same pattern as in the soft‑max case)
    a_rev = a[::-1]
    size = _next_pow2(len(a_rev) + len(p) - 1)
    Ar = np.fft.rfft(a_rev, n=size)
    Pr = np.fft.rfft(p, n=size)
    corr = np.fft.irfft(Ar * Pr, n=size)[: len(a_rev) + len(p) - 1]

    grad = 2.0 * corr[n - 1: n - 1 + n]
    return grad


def _max_shave_opt(a0: np.ndarray, deadline: float, rng) -> np.ndarray:
    """
    Fine‑tune a sequence by sub‑gradient steps that directly target the
    current peak of the auto‑convolution.  The step size is adapted
    whenever no improvement is observed for a few iterations.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sum(a)               # keep the exact ℓ¹‑norm

    # initial learning rate – schedule will shrink it if needed
    lr = target_sum * 0.05
    no_improve = 0
    best_val = _max_convolution_fft(a)

    while time.time() < deadline:
        g = _subgradient_max(a)
        # move opposite to the gradient (we want to *reduce* the max)
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        val = _max_convolution_fft(a_new)
        if val < best_val:
            a = a_new
            best_val = val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:                     # patience
                lr *= 0.5
                no_improve = 0
                if lr < 1e-12:
                    break

        # occasional tiny jitter to shake out shallow local minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very simple SA that moves a fraction of mass between two indices."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Draw a random non‑negative sequence on the simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # almost uniform + jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    The routine combines:
        • many random restarts + long projected‑gradient descent
        • a dedicated sub‑gradient “max‑shave’’ refinement
        • a final simulated‑annealing polishing
    and returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed with a previously discovered construction, if present
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0})"
              f" score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a balance between flexibility and speed
    # ------------------------------------------------------------------
    lengths = [2000, 3500, 5000, 7500, 10000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Phase A – cheap random restarts + short GD
        # ------------------------------------------------------------------
        phase_end = min(deadline,
                        time.time() + min(25.0, (deadline - time.time()) / 5.0))
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)

            # a brief GD run (≈ 3 s) to get into a decent basin
            sub_dead = min(deadline, time.time() + 3.0)
            a_gd = _gradient_descent_opt(a0, sub_dead)

            sc = _max_convolution_fft(a_gd)
            if sc < best_score:
                best_score = sc
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n}"
                      f" score={sc:.6f}")

        # ------------------------------------------------------------------
        # Phase B – deep GD refinement on the current best of this length
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            sub_dead = min(deadline, time.time() + min(180.0,
                                                       deadline - time.time()))
            a_ref = _gradient_descent_opt(best_seq, sub_dead)
            sc = _max_convolution_fft(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined best (length {n}) score={sc:.6f}")

        # ------------------------------------------------------------------
        # Phase C – max‑shave (sub‑gradient) fine tuning
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            sub_dead = min(deadline, time.time() + min(60.0,
                                                       deadline - time.time()))
            a_shave = _max_shave_opt(best_seq, sub_dead, rng)
            sc = _max_convolution_fft(a_shave)
            if sc < best_score:
                best_score = sc
                best_seq = a_shave.copy()
                print(f"[{time.time():.1f}] Max‑shave improvement n={n}"
                      f" score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time remains
    # ------------------------------------------------------------------
    if best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 5.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _max_convolution_fft(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up, evaluation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform sequence (should never be needed)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Log the official objective
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f}"
              f" (length={best_seq.size})")
    except Exception as exc:          # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
