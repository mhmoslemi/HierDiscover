def _subgradient_descent(a0, deadline, target_sum):
    a = a0.copy()
    n = a.shape[0]
    rho = 0.9
    momentum = np.zeros_like(a)
    it = 0

        b = _full_convolution_fft(a)

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Length sweep** – try several vector lengths (1500 … 5000).  For each
   length we perform a few independent restarts.

2. **Random symmetric initialisation** – draw a Dirichlet vector with a
   randomly chosen concentration (0.05 … 0.8) to obtain a highly
   heterogeneous shape, mirror it to enforce symmetry and project onto the
   ℓ¹‑simplex Σa = √(2 n).

3. **Projected sub‑gradient descent on the true ℓ∞‑norm** – the objective
   is the maximal entry of the self‑convolution.  At each iteration we
   compute the convolution by FFT, locate the index *k* of the maximal
   entry and use the exact sub‑gradient
        g_i = 2·a_{k‑i}   (if 0 ≤ k‑i < n, else 0)
   together with a momentum term.  After a step we project back onto the
   simplex and re‑symmetrise.  The step size decays with the iteration
   count.

4. **Final simulated‑annealing polishing** – when the global budget is
   almost exhausted we run a classic SA that transfers a random fraction of
   mass between two indices, accepting uphill moves with a slowly cooling
   temperature.

The whole search respects a user‑supplied time budget (default 1000 s).  
If the target value 1.503 is reached early the routine aborts
immediately and returns the current best sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s , x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the ℓ∞‑norm of the self‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray, deadline: float, target_sum: float, rng: np.random.Generator
) -> np.ndarray:
    """Minimise max(a*a) with a simple projected sub‑gradient method."""
    a = a0.copy()
    n = a.shape[0]

    # momentum parameters (exponential moving average of sub‑gradients)
    rho = 0.9
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # convolution and index of the current maximum
        b = _full_convolution_fft(a)
        k = int(np.argmax(b))                     # 0 … 2n‑2

        # exact sub‑gradient of b_k w.r.t. a
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[partner[mask]]

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # step size proportional to target_sum / norm, decays with √t
            step = 0.6 * target_sum / (norm * (np.sqrt(it + 1) + 1.0))
            a = a - step * momentum

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # enforce exact symmetry every few iterations
        if (it % 20) == 0:
            a = (a + a[::-1]) * 0.5
            a = _project_to_simplex(a, target_sum)

        # occasional tiny random jitter to escape plateaus
        if (it % 200) == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (uses the same objective)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Transfer random fractions of mass between indices, cooling slowly."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature proportional to current objective
    T = best_score * 0.015 + 1e-8
    cooling = 0.9998
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.18          # up to 18 % of the mass at i
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = cur <= best_score or (T > 0.0 and rng.random() < np.exp(-dE / T))
        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global shake
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur = _max_convolution_fft(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget,
    explores several sequence lengths, performs many restarts of a
    projected sub‑gradient descent, and finishes with a short simulated‑annealing
    polishing phase.  The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None
    best_val = np.inf
    TARGET = 1.503  # early‑exit threshold

    # ------------------------------------------------------------------
    # If a previously discovered sequence exists, start from it.
    # ------------------------------------------------------------------
    init_global = globals().get("height_sequence_1")
    if isinstance(init_global, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_global, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        val0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_val = val0
        print(f"[{time.time():.1f}] Starting from global height_sequence_1 "
              f"(n={n0})  obj={best_val:.6f}")

    # ------------------------------------------------------------------
    # Lengths to investigate.  The larger the length the more room we have
    # to flatten the convolution.
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500, 5000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # number of independent restarts for this length
        restarts = 8 if n <= 3000 else 5

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ------ random symmetric Dirichlet start -----------------
            alpha = rng.uniform(0.05, 0.8)           # low concentration => spiky
            a = rng.dirichlet(alpha=np.full(n, alpha))
            a = (a + a[::-1]) * 0.5                 # enforce symmetry
            a = a * target_sum / a.sum()            # scale to ℓ¹‑simplex

            # ------ projected sub‑gradient descent ------------------
            sub_dead = min(deadline, time.time() + 28.0)   # up to ~28 s per run
            a = _subgradient_descent(a, sub_dead, target_sum, rng)

            # ------ evaluate ----------------------------------------
            val = _max_convolution_fft(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best  n={n} run={r} "
                      f"obj={best_val:.6f}")

            if best_val <= TARGET:
                print("[*] Target reached – exiting early.")
                deadline = time.time()               # force termination of loops
                break

    # ------------------------------------------------------------------
    # Final polishing with simulated annealing (uses whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        print(f"[{time.time():.1f}] Starting final simulated‑annealing "
              f"refinement (remaining {deadline - time.time():.1f}s)")
        best_seq = _simulated_annealing(best_seq, deadline)
        best_val = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Tidy up: drop trailing near‑zero entries.
    # ------------------------------------------------------------------
    if best_seq is None:                       # should never happen
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # final sanity check with the official evaluator (optional)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max conv) = {best_val:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
