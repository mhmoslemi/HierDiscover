    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2
    best_seq = None; best_score = np.inf
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000]
    # warm starts: exponential and power-law shapes
    for n in lengths:
        for shape in ['exp', 'pow']:
            # sample several shape params
            for param in rng.uniform(low=0.5, high=3.0, size=5):
                if shape == 'exp':
                    a = _exponential_sequence(n, param)
                else:
                    a = _powerlaw_sequence(n, param)

                cur_max = np.max(_full_convolution(a))
                if cur_max < best_score:
                    best_score = cur_max; best_seq = a.copy()
    # main loop: each iteration pick n, initial a either jitter of best or new random shape
    while time.time() < deadline:
        # choose n with bias near best_seq length
        if best_seq is not None and rng.random() < 0.7:
            n = best_seq.size
        else:
            n = int(rng.integers(1000, 12001))
        # initialize a0
        if best_seq is not None and best_seq.size == n and rng.random() < 0.5:
            a0 = best_seq.copy()

def _optimize_seq(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    # hyperparameters
    lr = 0.8
    lr_decay = 0.99995
    swap_prob = 0.1
    swap_frac = 0.05
    target_prob = 0.1
    eps_target = 0.02
    smoothing_alpha = 0.5
    # schedule: start with smoothing for, say, first 2000 iterations or until alpha > 10.
    iteration = 0
    # track last improvement 
    last_improve = 0
    while time.time() < deadline:
        iteration += 1
        # compute convolution
        conv = _full_convolution(a)
        M = conv.max()
        # check best
        if M < best_score:
            best_score = M
            best_a = a.copy()
            last_improve = iteration
        # Phase: if alpha < high threshold, do smooth-weight step else subgradient step
        if smoothing_alpha < 20.0:
            a, cur_max = _smooth_step(a, lr, smoothing_alpha)
        else:
            # subgradient
            mask = (conv >= M - 1e-12)
            active = mask.sum()
            if active == 0:
                # should not happen
                w = np.zeros_like(conv)
            else:
                w = mask.astype(float) / active
            grad = _conv_with_weights(a, w)
            a = a - lr * grad
            a = _project_simplex(a, S)
            cur_max = np.max(_full_convolution(a))
        # random swap
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max*0.02,1e-12)):
                    a = cand
                    cur_max = cand_max
        # cheap targeted move: move mass from large a to small a
        if rng.random() < target_prob:
            i_max = int(np.argmax(a))
            i_min = int(np.argmin(a))
            if a[i_max] > 0 and i_max != i_min:
                delta = eps_target * a[i_max] * rng.random()
                cand = a.copy()
                cand[i_max] -= delta
                cand[i_min] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max*0.02,1e-12)):
                    a = cand
                    cur_max = cand_max
        # schedule updates
        lr *= lr_decay
        smoothing_alpha *= 1.01   # anneal temperature
        # early stop if no improvement for many iterations
        if iteration - last_improve > 20000:
            break
    return best_a, best_score

    # a_i ∝ (i+1)^(-p)

"""
A stochastic local search for the step‑function optimisation problem.

The algorithm proceeds in three stages:

1. **Warm‑start** – generate a diverse pool of candidate sequences.
   Candidates are exponential, power‑law and mixtures of two exponentials.
   Each candidate is projected onto the ℓ1‑simplex Σa = √(2·n).

2. **Local optimisation** – for a given candidate a (length n) we run a
   projected gradient descent on a smooth approximation of the ‑∞‑norm
   (soft‑max with temperature α) and, after α has grown, switch to a
   sub‑gradient descent for the true max‑convolution.  The update is
   followed by a projection onto the simplex.  Random mass‑swap moves
   (Metropolis‑style) and cheap targeted moves (mass from a large entry
   to a small entry) are interleaved to escape shallow local minima.

3. **Iterative refinement** – the whole procedure is repeated with
   different lengths (biased towards lengths that have performed well)
   until the time budget (≈1000 s) is exhausted.  The best sequence found
   is finally evaluated with the external `evaluate_sequence` black box.

The implementation uses only NumPy and the FFT for fast convolutions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_j = 2 Σ_i w_{i+j}·a_i for a∈ℝⁿ and w∈ℝ^{2n‑1}.
    This is the gradient of Σ_i w_i·(a*a)[i] w.r.t. a.
    """
    n = a.shape[0]
    # correlation a ⋆ w   (w is reversed internally)
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # slice that corresponds to Σ_i w_{i+j}·a_i
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _smooth_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the smooth approximation of the
    max‑convolution.  Returns the updated vector and the *exact* maximal
    convolution entry after the step.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    a_new = _project_simplex(a_new, np.sqrt(2.0 * n))
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Families of initial sequences
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """
    Decreasing exponential a_i ∝ q^i with q = exp(-k_factor/n),
    then projected onto the simplex Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    return _project_simplex(a, S)


def _powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """
    Power‑law a_i ∝ (i+1)^{-p}, projected onto the simplex.
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    a = a * (S / a.sum())
    return _project_simplex(a, S)


def _mix_exponential_sequence(n: int, k1: float, k2: float, w: float) -> np.ndarray:
    """
    Convex combination of two exponentials with decay factors k1 and k2.
    """
    S = np.sqrt(2.0 * n)
    lam1 = k1 / n
    lam2 = k2 / n
    q1 = np.exp(-lam1)
    q2 = np.exp(-lam2)
    a1 = (q1 ** np.arange(n))
    a2 = (q2 ** np.arange(n))
    a = w * a1 + (1.0 - w) * a2
    a = a * (S / a.sum())
    return _project_simplex(a, S)


# ----------------------------------------------------------------------
# Local optimiser (gradient + sub‑gradient + swaps)
# ----------------------------------------------------------------------


def _optimize_seq(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence:
        – start with smooth‑max gradient steps,
        – switch to a true sub‑gradient descent once the temperature α
          becomes large,
        – interleave random mass‑swap and targeted mass‑transfer moves.
    Returns the best vector encountered and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (tuned empirically)
    lr = 0.9                # initial learning rate
    lr_decay = 0.9999
    alpha = 0.4             # start temperature for smooth‑max
    alpha_mul = 1.008       # annealing speed
    swap_prob = 0.18
    swap_frac = 0.05
    target_prob = 0.25
    eps_target = 0.025
    big_move_prob = 0.07
    big_move_frac = 0.12

    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # -------------------------------------------------------------
        # 1. Gradient / sub‑gradient step
        # -------------------------------------------------------------
        conv = _full_convolution(a)
        cur_max = conv.max()

        # Update best record
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # Choose which descent direction to use
        if alpha < 15.0:                     # smooth phase
            a, cur_max = _smooth_step(a, lr, alpha)
        else:                               # sub‑gradient of the true max
            mask = (conv >= cur_max - 1e-12)
            active = mask.sum()
            if active == 0:
                w = np.zeros_like(conv)
            else:
                w = mask.astype(float) / float(active)
            grad = _conv_with_weights(a, w)
            a = a - lr * grad
            a = _project_simplex(a, S)
            cur_max = np.max(_full_convolution(a))

        # -------------------------------------------------------------
        # 2. Random mass‑swap (Metropolis acceptance)
        # -------------------------------------------------------------
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.015, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # -------------------------------------------------------------
        # 3. Targeted transfer from the largest entry to the smallest
        # -------------------------------------------------------------
        if rng.random() < target_prob:
            i_max = int(np.argmax(a))
            i_min = int(np.argmin(a))
            if i_max != i_min and a[i_max] > 0.0:
                delta = eps_target * a[i_max] * rng.random()
                cand = a.copy()
                cand[i_max] -= delta
                cand[i_min] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.015, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # -------------------------------------------------------------
        # 4. Occasional "big move" using the three largest / smallest entries
        # -------------------------------------------------------------
        if rng.random() < big_move_prob:
            # three largest indices
            large_idxs = np.argpartition(-a, kth=2)[:3]
            small_idxs = np.argpartition(a, kth=2)[:3]
            src = int(rng.choice(large_idxs))
            tgt = int(rng.choice(small_idxs))
            if src != tgt and a[src] > 0.0:
                delta = big_move_frac * a[src] * rng.random()
                cand = a.copy()
                cand[src] -= delta
                cand[tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.015, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # -------------------------------------------------------------
        # 5. Schedule updates
        # -------------------------------------------------------------
        lr *= lr_decay
        alpha *= alpha_mul

        # early stop if stagnated for a long while
        if it - last_improve > 25000:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine explores a variety of lengths and
    initial shapes, then refines promising candidates with the local
    optimiser described above.  The best sequence found before the time
    budget expires is returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Warm‑start: explore a modest grid of lengths / shapes
    # ------------------------------------------------------------------
    lengths = (800, 1200, 2000, 3000, 4000, 6000, 8000)
    n_param = 5  # number of parameter samples per family
    for n in lengths:
        if time.time() >= deadline:
            break
        # exponential family
        for k_factor in np.linspace(0.6, 2.8, n_param):
            a = _exponential_sequence(n, k_factor)
            cur = np.max(_full_convolution(a))
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[warm‑exp] n={n:4d} k={k_factor:4.2f} score={cur:.6f}")
        # power‑law family
        for p in np.linspace(0.3, 2.5, n_param):
            a = _powerlaw_sequence(n, p)
            cur = np.max(_full_convolution(a))
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[warm‑pow] n={n:4d} p={p:4.2f} score={cur:.6f}")
        # mixture of two exponentials
        for _ in range(n_param):
            k1 = rng.uniform(0.5, 2.5)
            k2 = rng.uniform(0.5, 2.5)
            w = rng.random()
            a = _mix_exponential_sequence(n, k1, k2, w)
            cur = np.max(_full_convolution(a))
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[warm‑mix] n={n:4d} k1={k1:4.2f} k2={k2:4.2f} w={w:4.2f} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2) Include the previous best solution if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                n = prev.size
                S = np.sqrt(2.0 * n)
                prev = _project_simplex(prev, S)
                cur = np.max(_full_convolution(prev))
                if cur < best_score:
                    best_score = cur
                    best_seq = prev.copy()
                    print(f"[prev] imported previous solution, n={n} score={cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Main optimisation loop – allocate time slices to candidates
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose length (biased towards the length of the current best)
        if best_seq is not None and rng.random() < 0.65:
            n = best_seq.size
        else:
            n = int(rng.integers(800, 12001))

        # Initialisation of a0
        if best_seq is not None and best_seq.size == n and rng.random() < 0.5:
            a0 = best_seq.copy()
            jitter = rng.normal(scale=0.001 * a0.mean(), size=n)
            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))
        else:
            shape_choice = rng.choice(['exp', 'pow', 'mix'])
            if shape_choice == 'exp':
                k = rng.uniform(0.5, 3.0)
                a0 = _exponential_sequence(n, k)
            elif shape_choice == 'pow':
                p = rng.uniform(0.2, 3.0)
                a0 = _powerlaw_sequence(n, p)
            else:  # mix
                k1 = rng.uniform(0.5, 3.0)
                k2 = rng.uniform(0.5, 3.0)
                w = rng.random()
                a0 = _mix_exponential_sequence(n, k1, k2, w)

        # Run the local optimiser on this candidate
        slice_deadline = min(deadline, time.time() + 28.0)
        a_opt, score_opt = _optimize_seq(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global‑best] n={n:4d} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final refinement on the best candidate (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}   length = {len(final_seq)}")
    return final_seq
