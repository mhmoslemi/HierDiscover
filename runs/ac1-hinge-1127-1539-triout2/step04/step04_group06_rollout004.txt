    The convex combination keeps the normalisation and never
    increases the maximal convolution.

The LP is solved with SciPy's HiGHS interior‑point routine.
A few inexpensive jitter moves are interleaved to escape plateaus.
The best vector obtained before the time budget expires is returned.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper numerical primitives
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) sequence of length n, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method alternates a linear‑programming
    step (maximise ℓ¹‑norm under a convolution bound) with a tiny
    stochastic jitter.  It returns the best sequence found before the
    allotted time expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # leave a safety margin

    # --------------------------------------------------------------
    # Initialise the vector (use the globally supplied best if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = seq.size
        target = np.sqrt(2.0 * n)
        if seq.sum() > 0:
            seq = seq * (target / seq.sum())
        else:
            seq = _power_law_sequence(n, 0.55, target)
        seq = _project_to_simplex(seq, target)
        f = seq
    else:
        n = 1500
        target = np.sqrt(2.0 * n)
        f = _power_law_sequence(n, 0.55, target)

    f = np.maximum(f, 0.0)
    f = _project_to_simplex(f, target)

    # current best
    best_f = f.copy()
    best_score = _max_convolution_fft(best_f)
    print(f"[{time.time():.1f}] start length={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main alternating LP loop
    # --------------------------------------------------------------
    iteration = 0
    t = 0.2                                   # mixing weight (will adapt)
    eps = 1e-12                               # avoid exact zeros

    while time.time() < deadline:
        # ----- current maximal convolution M
        M = _max_convolution_fft(f)

        # ----- build the (2n‑1)×n convolution matrix A (dense for simplicity)
        #       column j contains f shifted by j positions
        f_eps = np.where(f < eps, eps, f)      # avoid unbounded LP columns
        rows = 2 * n - 1
        A = np.empty((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f_eps

        # ----- linear program: maximise Σb  ⇔ minimise –Σb
        c = -np.ones(n, dtype=np.float64)      # objective for linprog: min c·b
        b_ub = np.full(rows, M, dtype=np.float64)

        # solve (HiGHS is the default fast method)
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # LP failed (should be rare); break out to avoid wasting time
            print(f"[{time.time():.1f}] LP failed at it {iteration}: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print(f"[{time.time():.1f}] LP returned zero sum at it {iteration}")
            break

        # ----- scale to the required ℓ¹‑norm
        g = b * (target / sum_b)

        # ----- convex combination (keeps the ℓ¹‑norm equal to target)
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = _project_to_simplex(f_new, target)

        # ----- evaluate the new maximal convolution
        M_new = _max_convolution_fft(f_new)

        # ----- accept if not worse (tiny tolerance for rounding)
        if M_new <= M + 1e-12:
            f = f_new
            # adapt step size – increase slightly on improvement
            if M_new < best_score - 1e-7:
                best_f = f.copy()
                best_score = M_new
                print(f"[{time.time():.1f}] it {iteration:04d}  improved score = {best_score:.6f}")
                t = min(t * 1.1, 0.5)
            else:
                # stagnation – shrink the move a bit
                t = max(t * 0.9, 0.01)
        else:
            # reject – make the next step more conservative
            t = max(t * 0.5, 0.01)

        # ----- occasional random jitter to escape flat regions
        if iteration % 20 == 0 and iteration > 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            f = f + jitter
            f = np.maximum(f, 0.0)
            f = _project_to_simplex(f, target)

        iteration += 1

        # safety break – do not spin forever if we are close to the deadline
        if time.time() > deadline - 1.0:
            break

    # --------------------------------------------------------------
    # Final tiny local refinement (mass‑transfer hill‑climb)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # a very lightweight version of the hill‑climb from the reference code
        while time.time() < deadline:
            conv = np.convolve(best_f, best_f)
            k_max = int(np.argmax(conv))

            # sensitivity vector for the active convolution entry
            sens = np.zeros_like(best_f)
            lo = max(0, k_max - (len(best_f) - 1))
            hi = min(k_max, len(best_f) - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            sens[idx] = best_f[k_max - idx]

            weight = sens * best_f
            if weight.max() == 0:
                break

            donor = int(np.argmax(weight))
            low_thr = np.percentile(weight, 10)
            candidates = np.where(weight <= low_thr)[0]
            if candidates.size == 0:
                break
            receiver = int(rng.choice(candidates))
            if donor == receiver or best_f[donor] <= 1e-14:
                continue

            frac = rng.random() * 0.07
            delta = best_f[donor] * frac
            best_f[donor] -= delta
            best_f[receiver] += delta

            new_score = _max_convolution_fft(best_f)
            if new_score < best_score:
                best_score = new_score
                # keep the improvement
                # (no need to print every tiny step)
            else:
                # revert
                best_f[receiver] -= delta
                best_f[donor] += delta

    # --------------------------------------------------------------
    # Trim trailing zeros and return
    # --------------------------------------------------------------
    nz = np.where(best_f > 1e-12)[0]
    if nz.size:
        best_f = best_f[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score = {best_score:.6f}, length = {len(best_f)}")
    return best_f.tolist()
