            rows.append(k)
            cols.append(i)
            data.append(f[i])

"""
Propose a step function sequence that minimizes the evaluation.
Algorithm:
- Randomly generate candidates on the ℓ₁‑simplex.
- Apply a short projected‑gradient descent on a log‑sum‑exp surrogate of the
  max entry of the self‑convolution.
- Iteratively improve the candidate with a linear‑programming step: for the
  current vector f compute M = max(f∗f) and solve
        maximise   Σ b_j
        subject to  (f∗b)_k ≤ M      (k = 0…2n‑2),   b ≥ 0.
  The optimal b (scaled to the required ℓ₁‑norm) is mixed with f.
- Finish with a quick simulated‑annealing refinement.
The LP is solved with SciPy’s high‑performance “highs” backend; a fallback to
the original gradient+SA pipeline is used when SciPy is unavailable.
The routine respects a user‑supplied time budget (default 1000 s) and returns
the best sequence found as a plain list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient‑descent phase (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a  (full convolution, length 2·n‑1)
        g = ∇_a  ((1/β)·log Σ_k exp(β·b_k))
      = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentials
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]         # central n values
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp surrogate.
    Stops when the deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9                         # momentum decay
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = 0.5 * target_sum / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # project back onto ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (slow exponential schedule)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (identical to previous version)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a random fraction of mass
    between two indices while accepting uphill moves with probability
    exp(-Δ/T).  Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or (T > 0.0 and rng.random() < np.exp(-dE / T)):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σ a_i = √(2 n).
    Several families are mixed to increase diversity, including a symmetric
    edge‑heavy shape.
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    elif kind == 4:                 # symmetric edge‑heavy (high at both ends)
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                           # uniform plus tiny jitter
        a = np.full(n, target_sum / n)

    # add a small Gaussian perturbation and re‑project
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a

    import scipy.sparse as _sp
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _build_conv_matrix(f: np.ndarray) -> "_sp.csr_matrix":
    """
    Build the matrix A ∈ ℝ^{(2n‑1)×n} such that (A @ b)_k = (f ∗ b)_k.
    Each column i of A contains the vector f_i repeated on rows i … i+n‑1.
    """
    n = f.size
    # number of non‑zero entries = n²
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp_for_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   subject to   (f ∗ b)_k ≤ M , b ≥ 0.
    Returns the optimal b (as a dense 1‑D array) or None on failure.
    """
    if not _SCIPY_AVAILABLE:
        return None

    n = f.size
    A = _build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σb = maximise Σb
    bounds = [(0.0, None)] * n

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True, 'disp': False})
    except Exception as e:  # pragma: no cover
        return None

    if not res.success:
        return None
    return np.asarray(res.x, dtype=np.float64)

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s total)
--------------------------------
1. **Random starts** – generate a handful of non‑negative vectors on the ℓ¹‑simplex
   (Σaᵢ=√(2 n)) using several families (power‑law, exponential, Dirichlet,
   symmetric edge‑heavy).  If a global sequence ``height_sequence_1`` is present it
   is also used (trimmed/padded to the current length).

2. **Projected gradient descent** – a short annealed gradient descent on the
   smooth log‑sum‑exp surrogate of the maximal entry of the self‑convolution.
   This quickly drives the shape towards a low ‑max⁡(a∗a).

3. **Iterative LP refinement** – for the current vector ``f`` compute
   ``M = max(f∗f)`` and solve the linear programme

        maximise   Σ b_j
        subject to (f∗b)_k ≤ M   (k = 0 … 2n‑2)
                    b ≥ 0 .

   The optimal ``b`` (scaled to the required ℓ¹‑norm) is mixed with ``f``
   (convex combination with coefficient *t*).  A few such iterations tend to
   flatten the convolution peak further.  The LP is solved with SciPy’s
   high‑performance “highs” backend; if SciPy is unavailable the step is
   skipped.

4. **Simulated‑annealing polishing** – a brief SA pass that transfers random
   fractions of mass between coordinates while occasionally accepting uphill
   moves.  This removes remaining small irregularities.

The routine respects the user‑supplied time budget (default 1000 s) and returns
the best sequence found as a plain Python list.  All helper functions are
top‑level, there is no file/network I/O and the code works with a single CPU.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i≥0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) utilities for gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a   (full convolution, length 2·n‑1)
        g = ∇_a ((1/β)·log Σ_k exp(β·b_k)) = 2·(a ⋆ p),
        p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # --- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # --- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentials
    p = exps / np.sum(exps)

    # --- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]         # central n values
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp surrogate.
    Stops when the deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9                         # momentum decay
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = 0.5 * target_sum / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # project back onto ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (slow exponential schedule)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated annealing (final polishing)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a random fraction of mass between two
    indices while accepting uphill moves with probability exp(-Δ/T).
    Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or (T > 0.0 and rng.random() < np.exp(-dE / T)):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the ℓ¹‑simplex Σaᵢ=√(2 n)."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    elif kind == 4:                 # symmetric edge‑heavy (high at both ends)
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                           # uniform plus tiny jitter
        a = np.full(n, target_sum / n)

    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Linear‑programming utilities (LP refinement step)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _build_conv_matrix(f: np.ndarray) -> "_sp.csr_matrix":
    """
    Build matrix A ∈ ℝ^{(2n‑1)×n} such that (A @ b)_k = (f ∗ b)_k.
    Column i contains the constant vector f[i] on rows i … i+n‑1.
    """
    n = f.size
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp_for_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise Σ b_j   subject to   (f ∗ b)_k ≤ M , b ≥ 0.
    Returns the optimal b (dense 1‑D array) or ``None`` on failure.
    """
    if not _SCIPY_AVAILABLE:
        return None

    n = f.size
    A = _build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σb = maximise Σb
    bounds = [(0.0, None)] * n

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True, 'disp': False})
    except Exception:  # pragma: no cover
        return None

    if not res.success:
        return None
    return np.asarray(res.x, dtype=np.float64)
