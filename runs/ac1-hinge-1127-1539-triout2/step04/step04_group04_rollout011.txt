      use λ = factor·e/(2·n) with factor ∈ [0.5, 2.0] (so we explore the
      theoretically relevant region).
    • All candidates are projected onto the ℓ₁‑simplex (sum = √(2n))
      and evaluated by a fast FFT‑based full convolution.
    The best candidate seen so far is retained.

3️⃣  Adam optimisation:
    Starting from the best candidate, minimise a smooth
    approximation of max(a*a) (soft‑max of the convolution entries)
    while projecting back onto the simplex after each step.
    The temperature is cooled aggressively and the learning rate
    decays slowly.

4️⃣  Optional pairwise mass‑transfer hill climbing (if time permits):
    Small amounts of mass are moved from a random index i to j;
    the move is kept when it reduces the maximal convolution value.
    This can escape shallow local minima left by Adam.

The whole routine respects the 1000 s time budget, prints progress,
stores the final sequence in the global variable `height_sequence_1`,
and returns the list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Project onto the ℓ₁‑simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum S."""
    # Fast path when all entries are non‑negative – just rescale.
    if np.min(v) >= 0.0:
        total = np.sum(v)
        if total > 0.0:
            return v * (S / total)
        # All zeros – return uniform distribution.
        return np.full_like(v, S / v.shape[0])

    # General algorithm (O(n log n)).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the τ·log∑exp(b/τ) approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Compute the gradient of the soft‑max approximation of max(a*a).
    Returns (gradient, full convolution b, current max(b)).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                     # hard max → sub‑gradient
        k = int(np.argmax(b))
        # self–correlation sub‑gradient: g_i = 2 * Σ_j a_j * 1_{i+j=k}
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal element handling (i=j) is already covered above
        return g, b, max_b

    # stable soft‑max
    diff = b - max_b                 # to avoid overflow
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)          # soft‑max weights, shape (2n‑1,)

    # compute gradient: g = 2 * (a ⋆ w) restricted to n positions
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]

    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """Adam minimisation of the soft‑max of a*a (approx. max)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                       # a little smaller than original
    t = 0

    # initialise temperature from the hard max
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = max(cur_max * 0.7, 1e-8)   # a bit higher than before

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluate new max
        b_new = _full_convolution_fft(a_new)
        cur_max_new = np.max(b_new)

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9985, 1e-9)     # a bit faster cooling
        lr *= 0.9999                     # gentle decay

        # occasional jitter to escape plateaus
        if rng.random() < 0.01:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Deterministic exponential candidate (λ ≈ e/(2n) up to a factor)
# ----------------------------------------------------------------------
def _deterministic_exponential(n: int, factor: float) -> np.ndarray:
    """Return a normalised exponential vector of length n."""
    lam = (math.e / (2.0 * n)) * factor
    idx = np.arange(n, dtype=np.float64)
    raw = np.exp(-lam * idx)
    S = math.sqrt(2.0 * n)
    # simple scaling preserves non‑negativity
    return raw * (S / raw.sum())


# ----------------------------------------------------------------------
# Random parametric distribution (extended families)
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random non‑negative vector using a variety of families."""
    # families that are useful for the problem
    families = [
        "uniform",
        "exp_opt",      # exponential with λ scaled to 1/n
        "symexp_opt",   # symmetric exponential with same scaling
        "gaussian",
        "triangular",
        "power",
        "cosine",
        "double_exp",   # sum of two exponentials with different rates
    ]
    typ = rng.choice(families)
    idx = np.arange(n, dtype=np.float64)
    a = np.empty_like(idx)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp_opt":
        lam0 = math.e / (2.0 * n)
        factor = rng.uniform(0.5, 2.0)
        lam = lam0 * factor
        a = np.exp(-lam * idx)

    elif typ == "symexp_opt":
        lam0 = math.e / (2.0 * n)
        factor = rng.uniform(0.5, 2.0)
        lam = lam0 * factor
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.03, 0.15) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "double_exp":
        lam0 = math.e / (2.0 * n)
        factor1 = rng.uniform(0.5, 2.0)
        factor2 = rng.uniform(0.5, 2.0)
        lam1 = lam0 * factor1
        lam2 = lam0 * factor2
        a = np.exp(-lam1 * idx) + np.exp(-lam2 * (n - 1 - idx))

    else:
        a = rng.random(n)

    # optional tiny jitter
    if rng.random() < 0.25:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# Simple pairwise mass‑transfer hill climbing
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Greedy hill‑climb: move tiny mass from i to j if it reduces max."""
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    cur_max = np.max(_full_convolution_fft(a))

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        # move a small fraction of a[i]
        eps = rng.uniform(0.0, a[i] * 0.03 + 1e-12)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # keep normalisation (the move conserves sum)
        b = _full_convolution_fft(a_new)
        new_max = np.max(b)
        if new_max < cur_max:
            a = a_new
            cur_max = new_max
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (height vector) with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0                     # leave a small safety margin
    deadline = time.time() + budget_s - safety

    best_seq = None
    best_max = np.inf

    # ------------------------------------------------------------------
    # Initialise from a previously stored solution (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = seq0.shape[0]
            S0 = math.sqrt(2.0 * n0)
            # simple rescaling (preserves shape)
            seq0 = seq0 * (S0 / np.sum(seq0))
            cur_max = np.max(_full_convolution_fft(seq0))
            best_seq = seq0.copy()
            best_max = cur_max
            print(f"[init] loaded previous solution: n={n0}, max={best_max:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – deterministic exponential sweep (≈ 20 % of the budget)
    # ------------------------------------------------------------------
    sweep_deadline = time.time() + (deadline - time.time()) * 0.2
    # geometric progression of lengths, up to ~200k
    n_values = [512, 1024, 2048, 4096, 8192, 16384,
                32768, 65536, 131072, 200000]
    factor_grid = [0.6, 0.8, 1.0, 1.2, 1.4]

    for n in n_values:
        if time.time() >= sweep_deadline:
            break
        for f in factor_grid:
            a = _deterministic_exponential(n, f)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_max:
                best_max = cur_max
                best_seq = a.copy()
                print(f"[sweep] n={n}, factor={f:.2f}, max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random parametric search (≈ half of the remaining budget)
    # ------------------------------------------------------------------
    search_deadline = time.time() + (deadline - time.time()) * 0.5
    min_n, max_n = 500, 200_000
    attempts = 0
    while time.time() < search_deadline:
        attempts += 1
        n = rng.integers(min_n, max_n + 1)
        raw = _random_distribution(n, rng)
        # simple scaling to the required ℓ₁‑norm
        S = math.sqrt(2.0 * n)
        a = raw * (S / raw.sum())
        cur_max = np.max(_full_convolution_fft(a))
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
            print(f"[rand] attempt={attempts} n={n} max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam refinement (uses the rest of the budget)
    # ------------------------------------------------------------------
    if best_seq is not None:
        refine_deadline = deadline
        refined_seq, refined_max = _adam_optimize(best_seq, refine_deadline, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_seq.copy()
            print(f"[adam] improved max to {best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional hill climbing (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        hill_deadline = deadline
        final_seq = _hill_climb(best_seq, hill_deadline, rng)
        final_max = np.max(_full_convolution_fft(final_seq))
        if final_max < best_max:
            best_max = final_max
            best_seq = final_seq.copy()
            print(f"[hill] improved max to {best_max:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]           # fallback (should never happen)
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_max:.6f}) length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
