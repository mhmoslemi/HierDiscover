"""
Propose a non‑negative step‑function (vector of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages and respects the 1000 s budget:

1. **Broad parametric sweep** – generate a variety of shapes (monotone,
   U‑shapes, bimodal exponentials, power laws, random Dirichlet vectors)
   and keep the best one.

2. **Smoothed Adam optimisation** – minimise a log‑sum‑exp surrogate of the
   maximum self‑convolution using the Adam optimiser with an adaptive
   smoothing parameter τ.  After each update the vector is projected onto the
   ℓ₁‑simplex {x_i≥0, Σx_i = √(2 n)}.

3. **Sub‑gradient refinement** – switch to the true nonsmooth objective,
   using the (averaged) sub‑gradient of the maximal convolution entries.
   The step size decays slowly and sporadic random pair‑transfers help to
   escape plateaus.

4. **Greedy polishing** – perform many tiny random mass‑transfers, accepting
   only improving moves (a simple hill‑climber).  The best vector found is
   returned.

All stages keep the current best sequence; intermediate progress is printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear algebra helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray,
                         target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximum entry of the full linear self‑convolution of `a`."""
    return float(conv_self_fft(a).max())

def generate_shape(n: int, target_sum: float, kind: str,
                   param: float) -> np.ndarray:
    """U‑shaped or biased shape indexed by ``kind``."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid  # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_bimodal_exponential(n: int,
                                 target_sum: float,
                                 alpha: float) -> np.ndarray:
    """Two exponential tails meeting in the centre (symmetric)."""
    left = np.exp(-alpha * np.arange(n, dtype=np.float64))
    right = left[::-1]
    raw = left + right
    return project_onto_simplex(raw, target_sum)


def generate_powerlaw(n: int,
                      target_sum: float,
                      p: float,
                      symmetric: bool = True) -> np.ndarray:
    """Power‑law decay from the left (or symmetric from both ends)."""
    base = (np.arange(1, n + 1, dtype=np.float64) ** -p)
    if symmetric:
        # mirror to get a symmetric shape
        raw = np.concatenate([base, base[::-1][1:]])[:n]
    else:
        raw = base
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def random_dirichlet_vector(n: int, target_sum: float) -> np.ndarray:
    """Sample a random point on the ℓ₁‑simplex (Dirichlet with all α=1)."""
    v = np.random.rand(n)
    return project_onto_simplex(v, target_sum)

def smoothed_gradient(a: np.ndarray,
                      tau: float) -> tuple[np.ndarray, float]:
    """
    Compute gradient of the log‑sum‑exp surrogate of max(conv) w.r.t `a`.
    Returns (gradient, current max convolution value).
    """
    conv = conv_self_fft(a)
    max_c = conv.max()
    # soft‑max weights
    w = np.exp((conv - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, max_c


def smoothed_adam(start: np.ndarray,
                  target_sum: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed surrogate.
    """
    a = start.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initial learning‑rate (roughly the size of one coefficient)
    lr = 0.5 * (target_sum / n)

    # initial smoothing – 15 % of the current max
    conv0 = conv_self_fft(a)
    tau = max(0.15 * conv0.max(), 1e-6)

    print(f"[{time.time():.2f}] Adam start – τ={tau:.3e}, lr={lr:.3e}")

    while time.time() < deadline:
        grad, cur_max = smoothed_gradient(a, tau)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # Smoothness schedule – shrink τ geometrically
        if t % 2000 == 0:
            tau = max(tau * 0.92, 1e-8)

        # Learning‑rate schedule – very gentle decay
        if t % 5000 == 0:
            lr *= 0.97

        # occasional random perturbation to escape shallow traps
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.006)

        # record improvement
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            print(f"  Adam iter {t:06d} – new best max={best_score:.6f}")

    return best_seq, best_score


def subgradient_refinement(start: np.ndarray,
                           target_sum: float,
                           best_seq: np.ndarray,
                           best_score: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution objective.
    """
    a = start.copy()
    n = a.shape[0]
    step = 0.04 * (target_sum / n)  # initial step size
    it = 0

    print(f"[{time.time():.2f}] Sub‑gradient start – step={step:.3e}")

    while time.time() < deadline:
        conv = conv_self_fft(a)
        cur_max = conv.max()
        eps = max(1e-12, 1e-9 * cur_max)
        mask = (conv >= cur_max - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0

        a_rev = a[::-1]
        L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        a -= step * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        it += 1

        # record improvement
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            print(f"  Sub‑grad iter {it:06d} – new best max={best_score:.6f}")

        # decay schedules
        if it % 500 == 0:
            step *= 0.94

        # occasional random transfer
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    return best_seq, best_score


def greedy_polish(start: np.ndarray,
                  target_sum: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Small random pair‑transfers accepted only when they improve the objective.
    """
    a = start.copy()
    n = a.shape[0]
    it = 0
    print(f"[{time.time():.2f}] Polishing start")
    while time.time() < deadline:
        cand = random_pair_transfer(a.copy(), target_sum, scale=0.0025)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand  # continue polishing from the improvement
            print(f"  Polish iter {it:06d} – improved max={best_score:.6f}")
        it += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine respects the 1000 s time budget and returns the best
    vector discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a tiny margin for the final return
    overall_deadline = start_time + budget_s - 0.25

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.asarray(base, dtype=np.float64)
        init_seq = project_onto_simplex(init_seq, math.sqrt(2.0 * n))
    else:
        n = 16384                # default length – a power of two is FFT‑friendly
        init_seq = None

    target_sum = math.sqrt(2.0 * n)      # Σ a_i = √(2 n)

    # ------------------------------------------------------------------
    # Phase 0 – parametric sweep + random seeds
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)   # uniform start
    best_score = compute_max_conv(best_seq)

    print(f"[{time.time():.2f}] Phase 0 – sweep (n={n})")
    shape_kinds = (
        "linear", "quadratic", "exponential",
        "power", "inverse",
        "linear_left", "linear_right",
        "exponential_left", "exponential_right",
    )
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),
        "quadratic": np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 12),
        "inverse": np.linspace(0.5, 5.0, 12),
        "linear_left": np.linspace(0.0, 0.9, 19),
        "linear_right": np.linspace(0.0, 0.9, 19),
        "exponential_left": np.linspace(0.0, 6.0, 31),
        "exponential_right": np.linspace(0.0, 6.0, 31),
    }

    sweep_deadline = start_time + budget_s * 0.06   # ~6 % of total budget

    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= sweep_deadline:
                break
        if time.time() >= sweep_deadline:
            break

    # extra families
    for alpha in np.linspace(0.05, 4.0, 30):
        cand = generate_bimodal_exponential(n, target_sum, alpha)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best bimodal_exp (α={alpha:.3f}) → {best_score:.6f}")
        if time.time() >= sweep_deadline:
            break

    for p in np.linspace(0.7, 3.5, 24):
        cand = generate_powerlaw(n, target_sum, p, symmetric=False)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best powerlaw (p={p:.3f}) → {best_score:.6f}")
        if time.time() >= sweep_deadline:
            break

    # random Dirichlet samples
    for _ in range(20):
        cand = random_dirichlet_vector(n, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best random Dirichlet → {best_score:.6f}")
        if time.time() >= sweep_deadline:
            break

    # incorporate provided baseline if available
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves start → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed Adam optimisation
    # ------------------------------------------------------------------
    ph1_deadline = start_time + budget_s * 0.55
    best_seq, best_score = smoothed_adam(best_seq,
                                         target_sum,
                                         best_seq,
                                         best_score,
                                         ph1_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement
    # ------------------------------------------------------------------
    ph2_deadline = start_time + budget_s * 0.90
    best_seq, best_score = subgradient_refinement(best_seq,
                                                  target_sum,
                                                  best_seq,
                                                  best_score,
                                                  ph2_deadline)

    # ------------------------------------------------------------------
    # Phase 3 – greedy polishing
    # ------------------------------------------------------------------
    best_seq, best_score = greedy_polish(best_seq,
                                         target_sum,
                                         best_seq,
                                         best_score,
                                         overall_deadline)

    print(f"[{time.time():.2f}] Returning best sequence (max_conv={best_score:.6f})")
    # Convert to plain Python list of floats
    return best_seq.tolist()
