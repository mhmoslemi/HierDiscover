"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The algorithm is a two‑phase optimizer:

1. **Adam optimisation on a smoothed max‑convolution**
   – Each iteration evaluates the smoothed maximum
     (log‑sum‑exp of the self‑convolution) and its gradient.
   – Adam’s adaptive step‑size together with a Euclidean simplex
     projection keeps the sum of the coefficients fixed (∑a = √(2 n)).
   – The smoothing parameter β is increased gradually so that the
     surrogate approaches the true max.

2. **Simulated‑annealing refinement**
   – After the Adam phase we perform a cheap SA that repeatedly
     transfers a random fraction of mass between two entries.
   – Accept‑reject follows a classic exponential schedule and a few
     global jitter moves are added to escape deep minima.

The search cycles over several vector lengths (≈1500‑3000).  For each
length a handful of diverse random starts are generated (power‑law,
exponential, Dirichlet, edge‑heavy etc.).  Each start receives a short
Adam run; the best few candidates are refined with SA.  The overall
wall‑clock budget is respected (default 1000 s).  The best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: closest power of two (for FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (using FFT)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Duchi et al. (2008) Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2· (a_rev ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stability
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a_rev ⋆ p) -----------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3·n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]  # central n entries
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed maximum of the convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   beta_start: float = 1.0,
                   beta_max: float = 1e6,
                   beta_inc: float = 1.2,
                   beta_step: int = 200) -> np.ndarray:
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.size
    S = np.sqrt(2.0 * n)

    beta = beta_start
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    # Objective evaluated via the surrogate maximum (b.max())
    g, b = _gradient_softmax(a, beta)
    best_obj = 2.0 * n * b.max() / (S * S)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while time.time() < deadline:
        # gradient of the smoothed max
        g, b = _gradient_softmax(a, beta)

        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Simple learning‑rate schedule: scale with target sum
        lr = 0.3 * S / (np.sqrt(v_hat.mean()) + eps)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, S)

        # evaluate current (surrogate) objective
        cur_obj = 2.0 * n * b.max() / (S * S)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # anneal smoothing parameter
        if t % beta_step == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refinement by random mass transfers between two indices.
    The move set is cheap and works well when the Adam phase has
    already flattened the convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = a.sum()

    # keep track of the best seen vector
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature proportional to current objective
    T = best_score * 0.02 + 1e-12
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick source (non‑zero) and destination indices
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction (≤ 20 %) of the source mass
        frac = rng.random() * 0.20
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to shake out of deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=S * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, S)
            cur = _max_convolution_fft(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Random initialisers – several diverse families
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families are mixed to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    kind = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.2)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # symmetric edge‑heavy (high at both ends)
        p = rng.uniform(0.2, 1.5)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 4:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif kind == 5:                 # binary block (first half, second half zero)
        half = n // 2
        a = np.concatenate([np.full(half, 1.0), np.zeros(n - half)])
    else:                           # uniform + tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

            # maintain elite pool
            if len(pool) < elite_per_len or cur_obj < pool[-1][0]:
                pool.append((cur_obj, cand.copy()))
                pool.sort(key=lambda x: x[0])
                pool = pool[:elite_per_len]

        # -----------------------------------------------------------------
        # Phase B – Adam optimisation of the elite seeds
        # -----------------------------------------------------------------
        for _, seed_vec in pool:
            if time.time() >= deadline:
                break

            # allocate a slice of the remaining budget for this seed
            sub_deadline = min(deadline, time.time() + 25.0)

            # Adam optimisation (smooth surrogate)
            a_adam = _adam_optimize(seed_vec, sub_deadline)

            # Simulated‑annealing refinement (use any leftover time)
            a_refine = _simulated_annealing(a_adam, deadline)

            # evaluate exact objective
            obj = 2.0 * n * _max_convolution_fft(a_refine) / (target_sum * target_sum)

            if obj < best_obj:
                best_obj = obj
                best_seq = a_refine.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 1800
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = 2.0 * n * _max_convolution_fft(best_seq) / (np.sum(best_seq) ** 2)

    # Trim trailing near‑zero entries for a cleaner output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final sanity‑check with the official evaluator (optional)
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished – best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
