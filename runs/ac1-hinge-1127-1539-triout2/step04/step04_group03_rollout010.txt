def _targeted_move(a, max_idx):
    n = a.size
    shift = max_idx - (n-1)   # compute actual shift value, can be positive or negative (0 at centre)
    if shift >= 0:
        # contributions from pairs (i,i+shift)
        idx_i = np.arange(0, n-shift)
        contrib = a[idx_i] * a[idx_i+shift]
    else:
        shift = -shift
        idx_i = np.arange(shift, n)
        contrib = a[idx_i] * a[idx_i-shift]
    # choose i proportionally to contrib (high contribution)
    probs = contrib / (contrib.sum()+1e-12)
    i = np.random.choice(idx_i, p=probs)
    # pick j random index not overlapping that shift significantly, maybe far.
    j = np.random.randint(0, n)
    # ensure i and j distinct. Transfer fraction.
    delta = a[i] * 0.1
    a[i] -= delta
    a[j] += delta
    return a
```

def _subgradient_max(a):
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_val = np.max(b)
    max_idx = np.argmax(b)
    # create mask p
    p = np.zeros(2*n-1)
    p[max_idx] = 1.0
    # compute convolution a_rev ⋆ p
    a_rev = a[::-1]
    conv_len = n + (2*n-1) - 1
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n-1 : n-1 + n]
    return max_val, max_idx, g
```

"""
Propose a non‑negative step‑function (a list of heights) that minimises
``evaluate_sequence``.

Algorithm Overview
-----------------
1. **Random initialisation** – generate a diverse set of candidates on the
   ℓ¹‑simplex Σ a_i = √(2 n) using power‑law, exponential and symmetric
   edge‑heavy families.

2. **Exact sub‑gradient descent** on the true objective
   f(a)=max_k (a∗a)[k].  The sub‑gradient is obtained by placing a unit
   mass on the active shift(s) of the convolution and convolving this mask
   with the current vector (the same trick used for the smooth surrogate
   but without the soft‑max).  After each step we project back onto the
   simplex and optionally symmetrise the vector (a←(a+a[::-1])/2).

3. **Targeted simulated‑annealing** – a Metropolis walk that preferentially
   moves mass away from the shift currently attaining the maximum.  The
   temperature schedule is much slower than in the previous code, and the
   move set contains both random transfers and “focused” transfers that
   reduce the contribution to the active shift.

4. **Final intensive SA** – once the global time budget is almost exhausted
   we run a longer SA on the best solution found.

All heavy operations (convolution, sub‑gradient) use FFTs, the code is
fully vectorised and respects a user‑supplied time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _max_convolution_fft_with_index(a: np.ndarray):
    """
    Return (max_value, max_index, full_conv) for a*a.
    max_index is the position in the full convolution array (0 … 2n‑2).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return max_val, max_idx, conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Compute the exact sub‑gradient of  f(a)=max_k (a*a)[k].

    Returns
    -------
    max_val : float
        Current value of the objective.
    max_idx : int
        Index (in the full convolution) where the maximum is attained.
    g : np.ndarray, shape (n,)
        Sub‑gradient (the same size as a).
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

    max_val = float(conv.max())
    max_idx = int(np.argmax(conv))

    # mask with a single 1 at the active shift
    mask = np.zeros(2 * n - 1, dtype=float)
    mask[max_idx] = 1.0

    # gradient = 2 * (a_rev ⋆ mask) restricted to the central n entries
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1           # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    mr_fft = np.fft.rfft(mask, n=size2)
    rev_conv = np.fft.irfft(ar_fft * mr_fft, n=size2)[:conv_len]
    g = 2.0 * rev_conv[n - 1 : n - 1 + n]   # central slice
    return max_val, max_idx, g


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Run a simple sub‑gradient descent with simplex projection."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    iteration = 0
    # initial step size – a fraction of the total mass
    step0 = 0.5 * target_sum

    while time.time() < deadline:
        max_val, max_idx, g = _subgradient_max(a)

        # step size decays slowly (1 / sqrt(iteration))
        step = step0 / (1.0 + 0.02 * iteration)

        a -= step * g
        a = np.maximum(a, 0.0)

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional symmetrisation – beneficial for the self‑convolution
        if iteration % 8 == 0:
            a = (a + a[::-1]) * 0.5
            a = _project_to_simplex(a, target_sum)

        iteration += 1

    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated annealing that preferentially tries to reduce the currently
    maximal convolution entry.  Two move types are used:

    * random mass transfer between two indices,
    * a “targeted” transfer that removes mass from an index contributing
      heavily to the active shift.
    """
    cur = a0.copy()
    n = cur.shape[0]
    cur_score = _max_convolution_fft(cur)
    best = cur.copy()
    best_score = cur_score

    # temperature proportional to current objective
    T = cur_score * 0.02 + 1e-8
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        # locate current active shift
        _, max_idx, _ = _max_convolution_fft_with_index(cur)
        shift = max_idx - (n - 1)          # signed shift

        # ---- propose a candidate ---------------------------------
        cand = cur.copy()

        if rng.random() < 0.65:           # 65 % random move
            i = rng.integers(0, n)
            if cand[i] <= 1e-14:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.random() * 0.15      # up to 15 % transfer
            delta = cand[i] * frac
            cand[i] -= delta
            cand[j] += delta
        else:                              # 35 % targeted move
            if shift >= 0:
                idx = np.arange(0, n - shift)
                prod = cand[idx] * cand[idx + shift]
            else:
                d = -shift
                idx = np.arange(d, n)
                prod = cand[idx] * cand[idx - d]

            if prod.size == 0:
                continue
            # pick a contributing index with probability ∝ product
            probs = prod / (prod.sum() + 1e-12)
            i = rng.choice(idx, p=probs)
            # choose a destination far enough (uniform)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.12          # move ~12 % of the mass
            cand[i] -= delta
            cand[j] += delta

        # ---- evaluate candidate ------------------------------------
        cand_score = _max_convolution_fft(cand)
        dE = cand_score - cur_score

        # Metropolis acceptance
        accept = False
        if cand_score <= cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{time.time():.1f}] SA new best: {best_score:.6f}")

        # temperature decay
        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=cur.sum() * 0.001, size=n)
            cur += jitter
            cur = np.maximum(cur, 0.0)
            cur = _project_to_simplex(cur, cur.sum())
            cur_score = _max_convolution_fft(cur)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                         # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                       # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                       # power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                       # Dirichlet (full simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    elif kind == 4:                       # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                                 # uniform plus jitter
        a = np.full(n, target_sum / n)

    # small Gaussian noise + projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine spends at most ``budget_s`` seconds,
    exploring several vector lengths, applying a sub‑gradient phase and a
    targeted simulated‑annealing refinement.  The best sequence found is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # Initialise from a global construction if it exists
    # ------------------------------------------------------------------
    init_global = globals().get("height_sequence_1")
    if isinstance(init_global, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_global, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        obj0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_obj = obj0
        print(f"[{time.time():.1f}] Starting from height_sequence_1 (n={n0}) obj={obj0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a moderate spread works well under the budget
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2100, 2400]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Phase A – quick random sampling (≈ 6 s per length)
        # ------------------------------------------------------------------
        phase_end = min(deadline, time.time() + 6.0)
        top_candidates = []      # keep a small elite pool

        while time.time() < phase_end:
            a_rand = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a_rand)

            if score < best_obj:
                best_obj = score
                best_seq = a_rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} obj={score:.6f}")

            # maintain elite pool (max 5 members)
            if len(top_candidates) < 5 or score < top_candidates[-1][0]:
                top_candidates.append((score, a_rand.copy()))
                top_candidates.sort(key=lambda x: x[0])
                top_candidates = top_candidates[:5]

        # ------------------------------------------------------------------
        # Phase B – sub‑gradient descent on elite seeds
        # ------------------------------------------------------------------
        for _, seed_arr in top_candidates:
            if time.time() >= deadline:
                break

            sub_dead = min(deadline, time.time() + 20.0)

            a_gd = _subgradient_descent(seed_arr, sub_dead)

            # ------------------------------------------------------------------
            # Phase C – targeted simulated‑annealing refinement
            # ------------------------------------------------------------------
            a_sa = _targeted_simulated_annealing(a_gd, deadline)

            obj = _max_convolution_fft(a_sa)
            if obj < best_obj:
                best_obj = obj
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={obj:.6f}")

        # clear pool before next length
        top_candidates.clear()

    # ------------------------------------------------------------------
    # Final intensive SA on the overall best solution (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_seq = _targeted_simulated_annealing(best_seq.copy(), deadline)
        final_obj = _max_convolution_fft(final_seq)
        if final_obj < best_obj:
            best_obj = final_obj
            best_seq = final_seq.copy()
            print(f"[{time.time():.1f}] Final SA improvement obj={final_obj:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and reporting
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]

        # optional sanity check with the official evaluator
        try:
            ev = evaluate_sequence(best_seq.tolist())
            print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
        except Exception:
            pass

    print(
        f"[{time.time():.1f}] Finished. Best objective (max conv) = {best_obj:.6f} "
        f"length = {0 if best_seq is None else best_seq.size}"
    )
    return [] if best_seq is None else best_seq.tolist()
