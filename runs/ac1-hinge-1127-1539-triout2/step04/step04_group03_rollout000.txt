"""
LP‑based iterative improvement for the step‑function objective.

The search proceeds in three stages:

1.  Generate many random non‑negative candidates on the ℓ¹‑simplex
    Σ a_i = √(2 n).  The families include power‑law, exponential,
    Dirichlet and edge‑heavy shapes.

2.  For each candidate run a small number of “LP‑refinement’’ steps.
    With the current vector f we compute M = max(f∗f).  Keeping M fixed,
    we solve the linear program

        maximise   Σ g_i
        subject to A_f·g ≤ M   (A_f encodes the convolution f∗g)
                    g_i ≥ 0

    The solution g contains (often) more total mass while still respecting
    the same max‑convolution bound.  After rescaling g to the required
    ℓ¹‑norm we move a fraction t towards it:  f ← (1−t) f + t g.
    Repeating a few times strictly reduces the objective.

3.  Keep the best vector seen across all restarts and return it as a
    plain Python list.  The routine respects the supplied time budget
    (default 1000 s) and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  The family is chosen uniformly among a few useful
    distributions.
    """
    target = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # decreasing power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                   # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif kind == 4:                   # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                             # plain uniform with tiny jitter
        a = np.full(n, target / n)

    # small Gaussian perturbation & projection to simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Build the convolution matrix A_f  (size (2n‑1) × n) such that
#   (A_f @ g)[k] = (f ∗ g)[k]
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Dense matrix for the linear map g ↦ f∗g."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # each column j contains f[0:n] shifted down by j rows
    for j in range(n):
        A[j : j + n, j] = f
    return A


# ----------------------------------------------------------------------
# LP‑refinement for a single vector
# ----------------------------------------------------------------------


def _lp_refine(f0: np.ndarray, deadline: float,
               max_iters: int = 12, t_start: float = 0.3,
               t_decay: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps on f0.
    Returns the improved vector and its max‑convolution value.
    """
    n = f0.size
    target_sum = np.sqrt(2.0 * n)
    f = f0.copy()
    best_obj = _max_convolution_fft(f)

    t = t_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # 1. compute current bound M
        M = _max_convolution_fft(f)

        # 2. build linear system A_f @ g ≤ M
        A = _build_conv_matrix(f)                      # dense (2n‑1) × n
        b = np.full(2 * n - 1, M, dtype=float)

        # 3. solve LP: maximise Σ g_i  (i.e. minimise -Σ g_i)
        c = -np.ones(n, dtype=float)                    # minimise -sum(g)
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": max(0, deadline - time.time())})

        if not res.success:
            # cannot improve – break early
            break

        g = np.maximum(res.x, 0.0)
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # 4. rescale g to the required ℓ¹‑norm
        g = g * (target_sum / sum_g)

        # 5. convex combination step
        f = (1.0 - t) * f + t * g

        # 6. project back onto the simplex (numerical drift)
        f = _project_to_simplex(f, target_sum)

        # 7. evaluate improvement
        new_obj = _max_convolution_fft(f)
        if new_obj < best_obj:
            best_obj = new_obj
        else:
            # no improvement – optionally reduce step size
            pass

        # 8. schedule
        t = max(0.05, t * t_decay)
        it += 1

    return f, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget, runs many random restarts,
    and for each candidate executes a few LP‑refinement steps.
    The best vector found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5     # small safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # If a previously discovered sequence is available, start from it
    # ------------------------------------------------------------------
    init_global = globals().get("height_sequence_1")
    if isinstance(init_global, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_global, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        obj0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_obj = obj0
        print(f"[{time.time():.1f}] Global start from height_sequence_1 "
              f"(n={n0})  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a modest spread keeps each LP cheap
    # ------------------------------------------------------------------
    lengths = [1000, 1300, 1600, 2000, 2400, 2800]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        # We keep a short pool of promising candidates for this length
        pool = []

        # ---------- Phase A: random sampling ----------
        phase_end = min(deadline, time.time() + 6.0)
        while time.time() < phase_end:
            a = _random_initial_sequence(n, rng)
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj = obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n}  obj={best_obj:.6f}")

            # keep a tiny elite list for LP refinement
            if len(pool) < 4 or obj < pool[-1][0]:
                pool.append((obj, a.copy()))
                pool.sort(key=lambda x: x[0])
                pool = pool[:4]

        # ---------- Phase B: LP‑based refinement ----------
        for _, seed_vec in pool:
            if time.time() >= deadline:
                break

            # allow up to ~20 s per elite candidate (including several LP steps)
            sub_dead = min(deadline, time.time() + 20.0)

            # LP refinements
            a_refined, obj_refined = _lp_refine(seed_vec, sub_dead,
                                                max_iters=12,
                                                t_start=0.3,
                                                t_decay=0.75)

            if obj_refined < best_obj:
                best_obj = obj_refined
                best_seq = a_refined.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (LP) n={n}  obj={best_obj:.6f}")

        # clear pool before next length
        pool.clear()

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing near‑zero entries for tidiness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional final sanity check with the official evaluator
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best max‑conv = {best_obj:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
