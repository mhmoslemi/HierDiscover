def _adam_optimize(a0, max_time, rng):
    deadline = time.time() + max_time
    a = a0.copy()
    n = len(a)
    S = sqrt(2n)
    # compute initial max
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    # set tau initial maybe cur_max * 0.5
    tau = cur_max * 0.5

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8; lr=0.5

    t=0
    best_a = a.copy()
    best_max = cur_max
    while time.time() < deadline:
        t+=1
        g,b,cur_max = _gradient_softmax(a, tau)
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat/(np.sqrt(v_hat)+eps)
        a_new = a - step
        # projection to simplex
        a_new = _project_to_simplex(a_new, S)
        # compute new max
        bnew = _full_convolution_fft(a_new)
        cur_max_new = np.max(bnew)
        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()
        a = a_new
        # schedule tau (exponential decay)
        tau = max(tau * 0.995, 1e-7)  # faster decay
        lr *= 0.9995
        # jitter occasionally
        if rng.random() < 0.01:
            noise = rng.normal(scale=a.mean()*0.02, size=n)
            a = np.clip(a + noise, 0, None)
            a = _project_to_simplex(a, S)
    return best_a, best_max

"""
Improved search for a non‑negative step‑function with a low
self‑convolution peak.

The algorithm works in three stages:

1️⃣  A broad random search over many parametric families (including several
    new families such as double‑exponential, piecewise‑constant and
    log‑normal) and a wide range of lengths (800 – 20000).  Each candidate
    is projected onto the ℓ₁‑simplex ∑a = √(2n) and its maximal
    convolution entry is evaluated via a fast real FFT.

2️⃣  Several independent runs of a projected Adam optimiser.  The optimiser
    minimises a smooth soft‑max surrogate of the ∞‑norm of a*a.
    Each run receives a fresh time budget (≈ 30 s) and starts either from
    the current best vector or from a jittered version of it.  The
    temperature of the soft‑max is cooled much faster than in the reference
    implementation (τ ← 0.995·τ) and a small Gaussian jitter is added
    occasionally to escape plateaus.

3️⃣  The best vector found is returned (and stored as `height_sequence_1`
    for reuse by later runs).  The external score printed at the end is
    exactly the maximal self‑convolution entry because the ℓ₁‑normalisation
    forces `evaluate_sequence` to equal that value.
"""

import time
import numpy as np
from typing import Tuple, List

# ----------------------------------------------------------------------
# Helper – projection onto the ℓ₁‑simplex {x ≥ 0, Σx = S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum S."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of τ·log∑exp(b/τ) where b = a*a
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Compute the gradient of the soft‑max approximation of max(b)
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                     # hard max → sub‑gradient on the active index
        k = np.argmax(b)               # index of the active entry
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal term appears once when i=j (i = k/2)
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # ----- soft‑max branch -------------------------------------------------
    # stable soft‑max computation
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)                     # shape (2n‑1,)

    # g_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # slice the part that aligns with a

    return g, b, max_b


# ----------------------------------------------------------------------
# Projected Adam optimiser (soft‑max surrogate)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    max_time: float,
    rng: np.random.Generator,
    init_tau_factor: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """
    Run Adam on the soft‑max surrogate of max(a*a) for at most max_time seconds.
    Returns the best vector seen and its maximal convolution entry.
    """
    deadline = time.time() + max_time
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.6
    t = 0

    # initialise temperature τ
    _, _, cur_max = _gradient_softmax(a, tau=0.0)   # only to obtain a scale
    tau = max(cur_max * init_tau_factor, 1e-8)

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluate new max
        b_new = _full_convolution_fft(a_new)
        cur_max_new = np.max(b_new)

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.995, 1e-8)          # faster cooling
        lr *= 0.9999

        # tiny jitter to escape shallow plateaus
        if rng.random() < 0.01:
            noise = rng.normal(scale=a.mean() * 0.02, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families (including several new ones)
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n from a randomly chosen family."""
    families = [
        "uniform", "exp", "symexp", "gaussian", "triangular",
        "power", "cosine", "doubleexp", "piecewise", "lognorm", "invpow"
    ]
    typ = rng.choice(families)
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "doubleexp":
        # mixture of a forward and a reversed exponential
        lam1 = rng.uniform(0.001, 0.15)
        lam2 = rng.uniform(0.001, 0.15)
        w = rng.random()
        a_fwd = np.exp(-lam1 * idx)
        a_rev = np.exp(-lam2 * (n - 1 - idx))
        a = w * a_fwd + (1.0 - w) * a_rev

    elif typ == "piecewise":
        # piecewise constant – random number of segments (2‑6)
        segs = rng.integers(2, 7)
        # random breakpoints (sorted)
        cuts = np.sort(rng.integers(1, n, size=segs - 1))
        lengths = np.diff(np.concatenate(([0], cuts, [n])))
        heights = rng.random(segs) * rng.uniform(0.5, 1.5)
        a = np.repeat(heights, lengths)[:n]

    elif typ == "lognorm":
        # discrete log‑normal
        mu = rng.uniform(-1.0, 2.0) * np.log(n)
        sigma = rng.uniform(0.3, 0.8)
        a = np.exp(-0.5 * ((np.log(idx + 1) - mu) / sigma) ** 2)

    elif typ == "invpow":
        p = rng.uniform(0.5, 3.0)
        a = 1.0 / ((idx + 1) ** p + 1e-6)

    else:
        a = rng.random(n)

    # small random jitter (helps diversity)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (height vector) with a low evaluation score.
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    safety = 4.0                      # small safety margin for the harness
    deadline = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # 0. Initialise from a previously stored solution (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_max = np.inf
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = seq0.shape[0]
            S0 = np.sqrt(2.0 * n0)
            seq0 = _project_to_simplex(seq0, S0)
            cur_max = np.max(_full_convolution_fft(seq0))
            best_seq = seq0.copy()
            best_max = cur_max
            print(f"[init] loaded previous solution: n={n0}, max={best_max:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Random parametric search (≈20 % of the total budget)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.20
    attempts = 0
    while time.time() < phase1_deadline:
        attempts += 1
        # length – sample more widely (up to 20000)
        n = rng.integers(800, 20001)          # inclusive lower bound, exclusive upper
        a_raw = _random_distribution(n, rng)
        S = np.sqrt(2.0 * n)
        a = _project_to_simplex(a_raw, S)

        cur_max = np.max(_full_convolution_fft(a))

        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
            print(f"[search] attempt={attempts:4d} n={n:5d} new best max={best_max:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Multiple Adam‑refinement runs (use the remaining time)
    # ------------------------------------------------------------------
    # parameters for the refinement loop
    min_run_time = 6.0                # seconds per Adam run (short runs for many restarts)
    max_run_time = 30.0               # cap – we do not let a single run dominate
    while time.time() < deadline - 1.0:
        remaining = deadline - time.time()
        # allocate a slice of the remaining time
        run_time = min(max_run_time, max(min_run_time, remaining * 0.35))
        if run_time < 3.0:   # not enough time left for a meaningful run
            break

        # initialise restart either from the current best or from a jittered version
        start_vec = best_seq.copy()
        if rng.random() < 0.4:         # sometimes start from a fresh random point
            n = start_vec.shape[0]
            a_raw = _random_distribution(n, rng)
            S = np.sqrt(2.0 * n)
            start_vec = _project_to_simplex(a_raw, S)

        # occasional stronger jitter before optimisation
        if rng.random() < 0.2:
            jitter = rng.normal(scale=start_vec.mean() * 0.05, size=start_vec.shape[0])
            start_vec = np.clip(start_vec + jitter, 0.0, None)
            start_vec = _project_to_simplex(start_vec, np.sqrt(2.0 * start_vec.shape[0]))

        refined_seq, refined_max = _adam_optimize(start_vec, run_time, rng)
        if refined_max < best_max:
            best_max = refined_max
            best_seq = refined_seq.copy()
            print(f"[refine] improved max to {best_max:.6f} (run {run_time:.1f}s)")

    # ------------------------------------------------------------------
    # 3️⃣ Final evaluation and return
    # ------------------------------------------------------------------
    if best_seq is None:                     # fallback – should never happen
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_max:.6f}) length = {len(final_seq)}"
    )

    # store for possible reuse by later calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
