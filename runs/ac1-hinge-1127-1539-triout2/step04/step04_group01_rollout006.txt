def propose_candidate(seed=42, budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    rng = np.random.default_rng(seed)

    # Determine length n: use list of possible n sorted ascending; try bigger if time left.

    candidate_lengths = [2000, 4000, 8000, 12000]

    best_seq = None
    best_score = np.inf

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        S = np.sqrt(2.0 * n)

        # Starting y0
        if best_seq is not None and best_seq.size == n:
            a0 = best_seq
            y0 = np.log(a0 + 1e-12)
        else:
            # try seed from global height_sequence_1 if appropriate size (maybe up to n)
            prior = globals().get('height_sequence_1')
            if isinstance(prior, (list, tuple, np.ndarray)):
                a_pr = np.asarray(prior, dtype=float)
                if a_pr.size <= n:
                    # pad zeros
                    a0 = np.concatenate([a_pr, np.zeros(n - a_pr.size)])
                else:
                    a0 = a_pr[:n]
                a0 = _project_to_simplex(a0, S)
                y0 = np.log(a0 + 1e-12)
            else:
                # start from uniform log
                y0 = np.zeros(n)

        # Phase ES
        # allocate time portion: maybe 0.6 of remaining time
        remaining = deadline - time.time()
        es_time = remaining * 0.6
        es_deadline = time.time() + es_time
        a_es, score_es = _evolution_strategy(n, S, y0, es_deadline, rng)

        if score_es < best_score:
            best_score = score_es
            best_seq = a_es.copy()
            print(f"[{time.time():.1f}] ES best length {n} score={score_es:.6f}")

        # Phase GD using remaining time
        remaining = deadline - time.time()
        if remaining > 5.0:
            gd_deadline = time.time() + remaining
            a_gd = _gradient_descent_opt(best_seq, gd_deadline)
            score_gd = _max_convolution_fft(a_gd)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] GD refined length {n} score={score_gd:.6f}")

    # final validation
    # cut trailing zeros
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[:nz[-1]+1]
    else:
        # fallback uniform length 1000
        n = 1000
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)

    # final evaluate
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={len(best_seq)})")
    except Exception as exc:
        print(f"Error: {exc}")

    return best_seq.tolist()

The optimizer works in two complementary phases:

1. **Global Exploration – Evolution Strategy (ES)**  
   A log‑softmax parametrisation transforms a real vector `y` into a non‑negative step‑function `a` that automatically satisfies the ℓ¹‑normalisation  
   `∑ a = √(2 n)`.  
   An (1+λ)‑ES samples many offspring around the current mean, keeps the best one and adapts the mutation strength `σ`.  
   This quickly discovers good‑looking shapes even in very high dimensions (here up to ≈ 10 000).

2. **Local Refinement – Smoothed‑Maximum Projected Gradient Descent**  
   Starting from the best ES candidate we minimise a smooth approximation of the maximal convolution entry  
   (log‑sum‑exp with annealed temperature β) while always projecting back onto the ℓ¹‑simplex.  
   A final short Simulated‑Annealing polishing step removes possible remaining plateaus.

The whole routine respects the 1000 s wall‑clock budget, prints progress and finally returns the best sequence found (as a plain `list[float]`).  

```python
"""
Step‑function optimiser – Evolution Strategy + Smoothed‑Maximum GD.

* 1+λ Evolution Strategy on the log‑softmax parametrisation of a,
  adapting the mutation strength σ on the fly.
* Follow‑up projected gradient descent on a log‑sum‑exp smooth max
  of the self‑convolution (β annealing, momentum, adaptive step).
* Optional short simulated‑annealing polishing.
* All time is shared between the phases; the routine stops at the
  supplied budget and returns the best sequence (list of non‑negative
  floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the linear convolution a*a using an FFT.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max parametrisation (log‑softmax guarantees positivity & sum)
# ----------------------------------------------------------------------


def _softmax_y_to_a(y: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Convert real vector y into a non‑negative step function a with Σa = target_sum.
    a_i = exp(y_i) / Σexp(y) * target_sum
    """
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)          # shift for numerical stability
    a = exp_y / np.sum(exp_y) * target_sum
    return a


# ----------------------------------------------------------------------
# Smoothed‑maximum gradient (log‑sum‑exp) and projected GD
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b):
        g = 2·(a ⋆ p)          with  p_k ∝ exp(β·b_k)
        b = a*a  (full linear convolution)
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities ----------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # numerical stability
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1              # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Momentum, adaptive step size and occasional jitter are used.
    The loop runs until ``deadline``.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.20
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing polishing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short SA that repeatedly transfers a random fraction of mass
    from one index to another.  Accepts uphill moves with exp(-Δ/T).
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or rng.random() < np.exp(-dE / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Evolution Strategy on the log‑softmax parametrisation
# ----------------------------------------------------------------------


def _evolution_strategy(
    n: int,
    target_sum: float,
    y0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    (1+λ) ES with adaptive σ on the log‑softmax representation.
    Returns the best sequence `a` found and its max‑convolution value.
    """
    # hyper‑parameters
    lam = max(20, int(4 + 3 * np.log(n)))          # population size
    sigma = 0.5                                     # initial log‑space std
    sigma_min, sigma_max = 1e-6, 2.0
    target_success = 0.2

    # current parent
    y_parent = y0.copy()
    a_parent = _softmax_y_to_a(y_parent, target_sum)
    parent_score = _max_convolution_fft(a_parent)

    best_a = a_parent.copy()
    best_score = parent_score

    while time.time() < deadline:
        # Sample offspring
        y_off = y_parent[None, :] + sigma * rng.normal(size=(lam, n))

        # Soft‑max → a
        y_max = np.max(y_off, axis=1, keepdims=True)
        exp_y = np.exp(y_off - y_max)
        a_off = exp_y / np.sum(exp_y, axis=1, keepdims=True) * target_sum

        # Scores
        scores = np.empty(lam, dtype=float)
        for i in range(lam):
            scores[i] = _max_convolution_fft(a_off[i])

        # Best offspring of this generation
        idx_best = int(np.argmin(scores))
        best_local_score = float(scores[idx_best])
        y_best = y_off[idx_best]
        a_best = a_off[idx_best]

        # Update global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = a_best.copy()

        # Did we improve over the parent?
        improved = best_local_score + 1e-12 < parent_score
        if improved:
            y_parent = y_best.copy()
            parent_score = best_local_score
        # else keep the old parent

        # σ adaptation (1/5‑rule style)
        if improved:
            sigma = min(sigma * 1.2, sigma_max)
        else:
            sigma = max(sigma * 0.82, sigma_min)

        # gentle decay to force eventual convergence
        sigma *= 0.99

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the given time budget,
    runs an Evolution Strategy (global search) followed by a
    projected‑gradient refinement and a final short simulated‑annealing
    polish.  The best sequence (plain Python list) is returned.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a problem size ------------------------------------------------
    # ------------------------------------------------------------------
    # Longer sequences give more flexibility but cost more.
    # We start with a medium size (8000) and, if time permits,
    # increase to a larger one (12000).
    lengths = [4000, 8000, 12000]

    best_seq = None
    best_score = np.inf

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise log‑softmax vector y0
        # --------------------------------------------------------------
        prior = globals().get("height_sequence_1")
        if isinstance(prior, (list, tuple, np.ndarray)):
            a_prior = np.asarray(prior, dtype=float)
            if a_prior.size <= n:
                a0 = np.concatenate([a_prior, np.zeros(n - a_prior.size)])
            else:
                a0 = a_prior[:n]
            a0 = _project_to_simplex(a0, target_sum)
            y0 = np.log(a0 + 1e-12)
        elif best_seq is not None and best_seq.size == n:
            y0 = np.log(best_seq + 1e-12)
        else:
            y0 = np.zeros(n)                     # uniform logits → uniform a

        # --------------------------------------------------------------
        # Phase 1 – Evolution Strategy
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 5.0:
            break
        # reserve ~60 % of the remaining budget for ES
        es_deadline = time.time() + 0.60 * time_left
        a_es, score_es = _evolution_strategy(
            n, target_sum, y0, es_deadline, rng
        )
        if score_es < best_score:
            best_score = score_es
            best_seq = a_es.copy()
            print(f"[{time.time():.1f}] ES improvement: n={n} score={score_es:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Smoothed‑max projected gradient descent
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left > 5.0:
            gd_deadline = time.time() + time_left * 0.40   # use the rest
            a_gd = _gradient_descent_opt(best_seq, gd_deadline)
            score_gd = _max_convolution_fft(a_gd)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] GD refined: n={n} score={score_gd:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Optional short SA polish (only if time still left)
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left > 3.0:
            a_sa = _simulated_annealing(best_seq, deadline)
            score_sa = _max_convolution_fft(a_sa)
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] SA polish: n={n} score={score_sa:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up tiny trailing zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform sequence (should never happen)
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Final evaluation using the official checker (for logging only)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})"
        )
    except Exception as exc:  # pragma: no cover
        print(f"Error during final evaluation: {exc}")

    return best_seq.tolist()
```
