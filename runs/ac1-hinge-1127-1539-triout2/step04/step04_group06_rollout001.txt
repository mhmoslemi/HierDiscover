def solve_lp(f, M):
    n = f.size
    # Build A_ub
    rows = []
    for i in range(2*n - 1):
        # get coefficient row for conv index i
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        row = np.zeros(n)
        # For j from lo to hi: coefficient f[i - j] for variable g_j
        # So do row[lo:hi+1] = f[i - np.arange(lo, hi+1)]
        idx = np.arange(lo, hi+1)
        row[idx] = f[i - idx]
        rows.append(row)
    A = np.vstack(rows)
    b = np.full(2*n - 1, M)
    # Solve maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # bounds (0, None)
    bounds = [(0, None)] * n
    # Use scipy.optimize.linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a non‑negative step function that minimises
``evaluate_sequence``.

Algorithm (≈ 1000 s budget)

1. Choose a modest length ``n`` (default 500) – a smaller problem
   makes the inner linear program cheap and allows many
   iterations.
2. Initialise a power‑law sequence
   a_i ∝ (i+1)^(-p)  and project it onto the ℓ¹‑simplex
   Σa_i = √(2 n) (the normalisation makes the objective equal to
   ``max(a*a)``).
3. Repeatedly:
   * compute M = max(a*a) (FFT based, O(n log n));
   * solve the linear programme

        maximise   Σ g_j
        subject to  (a * g)[k] ≤ M   for all k = 0 … 2n‑2
                    g_j ≥ 0

     – the convolution is linear in ``g``.
   * scale the LP solution to the target ℓ¹‑norm,
     mix it with the current vector (mixing weight ``t``),
     and re‑project onto the simplex.
   * keep the best vector seen (according to ``evaluate_sequence``).

   The mixing weight is gradually reduced (t → 0) to stabilise
   the process.
4. When the time budget is almost exhausted, run a short
   simulated‑annealing refinement (the same routine used in the
   reference solution) on the best vector found.
5. Trim trailing near‑zero entries and return the list.

The LP step is the core novelty: it directly maximises the
ℓ¹‑norm under the current max‑convolution bound, which provably
lowers the objective when the bound is later re‑normalised.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}
    (Duchi et al., 2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate a length‑n power‑law (i+1)^{-p} and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    total = a.sum()
    if total == 0:
        a = np.full(n, target_sum / n)
    else:
        a *= target_sum / total
    return a


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g)[k] = (f * g)[k].
    Row k contains the shifted copy of ``f`` that contributes to the k‑th
    entry of the convolution.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        A[k, idx] = f[k - idx]
    return A


def _solve_lp_max_sum_under_conv_bound(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Linear programme:
        maximise   Σ g_j
        subject to  (f * g)[k] ≤ M   for k = 0 … 2n‑2
                    g_j ≥ 0
    Returns the optimal ``g`` (or ``None`` if the solver fails).
    """
    n = f.size
    A = _build_convolution_matrix(f)          # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)         # minimise –Σ g_j
    bounds = [(0, None)] * n

    # Using HiGHS via SciPy – fast for modest sizes.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 20) -> tuple[np.ndarray, float]:
    """
    Shortened version of the SA routine from the reference code.
    It operates on a fixed‑size vector and respects the ℓ¹‑norm.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = 2.0 * n * _max_convolution_fft(a) / (cur_sum ** 2)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial full convolution (kept for the first iteration)
    conv = np.fft.irfft(np.fft.rfft(a, n=size_fft) ** 2, n=size_fft)[:2 * n - 1]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current max‑convolution location and sensitivity vector
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = np.zeros(n, dtype=np.float64)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        # propose a move
        if rng.random() < 0.70:  # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_idx)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                     # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            seg = a[start:end].copy()
            avg = seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = seg
            saved_start, saved_end = start, end

        # evaluate new state (single FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[:2 * n - 1]
        new_score = 2.0 * n * conv_new.max() / (cur_sum ** 2)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved_seg
            # conv, cur_score unchanged

        # occasional jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())
            conv = np.fft.irfft(np.fft.rfft(a, n=size_fft) ** 2, n=size_fft)[:2 * n - 1]
            cur_score = 2.0 * n * conv.max() / (cur_sum ** 2)

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence (as a list of floats) discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 1) Choose a problem size.
    #    Small enough for fast LP, large enough to give a good bound.
    # ------------------------------------------------------------------
    n = 500                     # default length; can be changed if time permits
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2) Initialise: power‑law (p≈0.55) projected onto the simplex.
    # ------------------------------------------------------------------
    seq = _power_law_sequence(n, 0.55, target_sum)
    seq = _project_to_simplex(seq, target_sum)

    # ------------------------------------------------------------------
    # 3) Main LP‑based improvement loop.
    # ------------------------------------------------------------------
    t_mix = 0.8               # initial mixing weight
    iteration = 0
    while time.time() < deadline and iteration < 2000:
        # Current max‑convolution value
        M = _max_convolution_fft(seq)

        # Solve LP: maximise ℓ¹‑norm under the bound M
        g0 = _solve_lp_max_sum_under_conv_bound(seq, M)
        if g0 is None:
            # LP failed – abort early
            break

        # Scale LP solution to the required ℓ¹‑norm
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        scale = target_sum / sum_g0
        g = g0 * scale

        # Mix the new candidate with the current one
        seq = (1.0 - t_mix) * seq + t_mix * g
        seq = _project_to_simplex(seq, target_sum)   # ensure exact normalisation

        # Evaluate (fast local version is the same as the benchmark after normalisation)
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] LP iter={iteration} t={t_mix:.3f} score={best_score:.6f}")

        # Slowly decay the mixing weight to stabilise the process
        if iteration % 200 == 0 and iteration > 0:
            t_mix = max(0.1, t_mix * 0.85)

        iteration += 1

    # ------------------------------------------------------------------
    # 4) If time permits, run a short simulated‑annealing refinement.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # Allocate up to the remaining time (but cap it to 30 s for safety)
        sa_deadline = time.time() + min(30.0, remaining)
        sa_seq, sa_score = _simulated_annealing(best_seq, sa_deadline)
        if sa_score < best_score:
            best_seq = sa_seq
            best_score = sa_score
            print(f"[{time.time():.1f}] SA refinement improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final clean‑up: trim trailing near‑zero entries.
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=np.float64)

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
