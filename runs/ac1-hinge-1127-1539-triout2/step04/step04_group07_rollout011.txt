def _intensive_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = np.max(_full_convolution(a))
    it = 0
    start_lr = 1.0
    lr = start_lr
    lr_decay = 0.9999
    start_alpha = 0.1
    alpha = start_alpha
    alpha_mul = 1.005
    swap_prob = 0.2
    swap_frac = 0.05
    target_prob = 0.3
    eps_target = 0.05
    last_improve = 0
    while time.time() < deadline:
        it += 1
        # gradient step
        a, cur_max = _gradient_step_custom(a, lr, alpha)
        # optional symmetrization every few steps
        if it % 10 == 0:
            a = _symmetrize(a)
            cur_max = np.max(_full_convolution(a))
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            last_improve = it
        # random swap
        if rng.random() < swap_prob:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_val = np.max(_full_convolution(cand))
                # accept if better or with probability
                if cand_val <= cur_max or rng.random() < np.exp(-(cand_val - cur_max) / max(cur_max*0.02, 1e-12)):
                    a = cand
                    cur_max = cand_val
                    if cur_max < best_val:
                        best_val = cur_max
                        best_a = a.copy()
                        last_improve = it
        # targeted move on max conv index
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]
            src_idx = i_range[np.argmax(contributions)]
            tgt_idx = i_range[np.argmin(contributions)]
            if a[src_idx] > 0 and src_idx != tgt_idx:
                delta = eps_target * a[src_idx] * rng.random()
                cand = a.copy()
                cand[src_idx] -= delta
                cand[tgt_idx] += delta
                cand = _project_simplex(cand, S)
                cand_val = np.max(_full_convolution(cand))
                if cand_val <= cur_max or rng.random() < np.exp(-(cand_val - cur_max) / max(cur_max*0.02, 1e-12)):
                    a = cand
                    cur_max = cand_val

                        best_val = cur_max
                        best_a = a.copy()
                        last_improve = it

        lr = max(lr * lr_decay, 1e-5)
        alpha = alpha * alpha_mul
        # early stop
        if it - last_improve > 10000:
            break

"""
Improved search for a non‑negative step function that minimises
evaluate_sequence().  The algorithm expands the initialisation pool
(beyond pure exponentials) with power‑law, linear, cosine‑window and
Dirichlet‑drawn vectors, then runs a robust local optimiser that
combines a smoothed‑max gradient step (with an annealing schedule
for the temperature), periodic symmetrisation, random mass swaps and
targeted moves that explicitly reduce the current maximal convolution
entry.  The best sequence found before the time budget expires is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g = 2 * (a * w) where * denotes the correlation.
    a has length n, w has length 2·n‑1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights from a convolution vector."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_step_custom(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the (soft‑)max‑norm.
    If alpha > 0 soft‑max is used, otherwise the pure max‑indicator.
    Returns the updated sequence and its exact maximal convolution entry.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    conv = _full_convolution(a)
    if alpha > 0:
        w = _smooth_weights(conv, alpha)
    else:
        # pure max‑indicator (sub‑gradient)
        max_val = conv.max()
        w = (conv == max_val).astype(float)
        w /= w.sum()

    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    a_new = _project_simplex(a_new, S)
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce (optional) mirror symmetry and re‑project onto the simplex."""
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    a_sym = (a + a[::-1]) / 2.0
    return _project_simplex(a_sym, S)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """
    Decreasing exponential a_i ∝ q^i with q = exp(-k_factor / n),
    projected onto the ℓ1‑simplex Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    # exact normalisation factor (avoids loss of precision)
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    return _project_simplex(a, S)


def _powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), projected onto the simplex."""
    idx = np.arange(n) + 1.0
    raw = idx ** (-p)
    S = np.sqrt(2.0 * n)
    a = raw / raw.sum() * S
    return _project_simplex(a, S)


def _linear_sequence(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = n-i, simplex‑projected."""
    raw = np.arange(n, 0, -1, dtype=float)
    S = np.sqrt(2.0 * n)
    a = raw / raw.sum() * S
    return _project_simplex(a, S)


def _cosine_sequence(n: int) -> np.ndarray:
    """Half‑period cosine window, simplex‑projected."""
    i = np.arange(n, dtype=float)
    raw = np.sin(np.pi * (i + 0.5) / n)
    S = np.sqrt(2.0 * n)
    a = raw / raw.sum() * S
    return _project_simplex(a, S)


def _random_dirichlet_sequence(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet(α,…,α) draw, simplex‑projected."""
    g = rng.gamma(alpha, 1.0, size=n)
    raw = g
    S = np.sqrt(2.0 * n)
    a = raw / raw.sum() * S
    return _project_simplex(a, S)


# ----------------------------------------------------------------------
# Local optimiser (intensive version)
# ----------------------------------------------------------------------


def _intensive_optimize(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using a schedule of smoothed‑max gradient steps,
    periodic symmetrisation, random mass swaps and cheap targeted moves.
    Returns the best sequence seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    it = 0
    # schedule parameters
    lr = 1.0
    lr_decay = 0.9999
    alpha = 0.1          # start very smooth
    alpha_mul = 1.005    # slowly sharpen
    swap_prob = 0.20
    swap_frac = 0.05
    target_prob = 0.30
    eps_target = 0.05
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) gradient step on the (soft) max‑norm
        # --------------------------------------------------------------
        a, cur_max = _gradient_step_custom(a, lr, alpha)

        # occasional symmetrisation (helps flatten the autocorrelation)
        if it % 10 == 0:
            a = _symmetrize(a)
            cur_max = np.max(_full_convolution(a))

        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            last_improve = it

        # --------------------------------------------------------------
        # 2) random mass‑swap (exploration)
        # --------------------------------------------------------------
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_val:
                        best_val = cur_max
                        best_a = a.copy()
                        last_improve = it

        # --------------------------------------------------------------
        # 3) cheap targeted move that reduces the current maximal entry
        # --------------------------------------------------------------
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            # indices that contribute to conv[k_max]
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contrib = a[i_range] * a[j_range]

            src = i_range[np.argmax(contrib)]
            tgt = i_range[np.argmin(contrib)]

            if a[src] > 0 and src != tgt:
                delta = eps_target * a[src] * rng.random()
                cand = a.copy()
                cand[src] -= delta
                cand[tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_val:
                        best_val = cur_max
                        best_a = a.copy()
                        last_improve = it

        # --------------------------------------------------------------
        # 4) schedule updates
        # --------------------------------------------------------------
        lr = max(lr * lr_decay, 1e-5)
        alpha *= alpha_mul

        # --------------------------------------------------------------
        # early stop if no improvement for a long while
        # --------------------------------------------------------------
        if it - last_improve > 10000:
            break

    return best_a, best_val

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises evaluate_sequence().
    The routine explores a variety of initial families (exponential,
    power‑law, linear, cosine, Dirichlet) and then refines promising candidates
    with an intensive local optimiser.  The best sequence discovered before the
    time budget expires is returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) quick sweep over a diverse family of parametrised shapes
    # ------------------------------------------------------------------
    lengths = [500, 800, 1000, 1500, 2000, 2500, 3000, 3500, 4000, 5000]

    # exponential family
    for n in lengths:
        if time.time() >= deadline:
            break
        for k in np.linspace(0.5, 5.0, 9):
            a = _exponential_sequence(n, k)
            val = np.max(_full_convolution(a))
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[init] exp n={n:4d} k={k:4.2f} val={val:.6f}")
            if time.time() >= deadline:
                break

    # power‑law family
    for n in lengths:
        if time.time() >= deadline:
            break
        for p in np.linspace(0.5, 3.0, 6):
            a = _powerlaw_sequence(n, p)
            val = np.max(_full_convolution(a))
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[init] pow n={n:4d} p={p:4.2f} val={val:.6f}")
            if time.time() >= deadline:
                break

    # linear decreasing
    for n in lengths:
        if time.time() >= deadline:
            break
        a = _linear_sequence(n)
        val = np.max(_full_convolution(a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[init] lin n={n:4d} val={val:.6f}")

    # cosine window
    for n in lengths:
        if time.time() >= deadline:
            break
        a = _cosine_sequence(n)
        val = np.max(_full_convolution(a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[init] cos n={n:4d} val={val:.6f}")

    # random Dirichlet draws
    for _ in range(30):
        if time.time() >= deadline:
            break
        n = rng.integers(800, 5001)
        alpha = rng.uniform(0.2, 5.0)
        a = _random_dirichlet_sequence(n, alpha, rng)
        val = np.max(_full_convolution(a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[init] dir n={n:4d} α={alpha:4.2f} val={val:.6f}")

    # ------------------------------------------------------------------
    # 2) maybe use previous best sequence offered by the environment
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                n = prev.size
                S = np.sqrt(2.0 * n)
                prev = _project_simplex(prev, S)
                val = np.max(_full_convolution(prev))
                if val < best_val:
                    best_val = val
                    best_seq = prev.copy()
                    print(f"[prev] using height_sequence_1 val={val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) main iterative improvement phase
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (bias towards the currently best length)
        if best_seq is not None and rng.random() < 0.6:
            n = best_seq.size
        else:
            n = rng.integers(800, 5001)

        # initialise either from the current best (jittered) or fresh
        if best_seq is not None and best_seq.size == n and rng.random() < 0.5:
            a0 = best_seq.copy()
            jitter = rng.normal(scale=0.001 * a0.mean(), size=n)
            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))
        else:
            # sample one of the families at random
            method = rng.choice(['exp', 'pow', 'lin', 'cos', 'dir'])
            if method == 'exp':
                k = rng.uniform(0.5, 5.0)
                a0 = _exponential_sequence(n, k)
            elif method == 'pow':
                p = rng.uniform(0.5, 3.0)
                a0 = _powerlaw_sequence(n, p)
            elif method == 'lin':
                a0 = _linear_sequence(n)
            elif method == 'cos':
                a0 = _cosine_sequence(n)
            else:  # Dirichlet
                alpha = rng.uniform(0.2, 5.0)
                a0 = _random_dirichlet_sequence(n, alpha, rng)

        # run a focused local optimisation slice (≈30 s)
        slice_deadline = min(deadline, time.time() + 30.0)
        a_opt, val_opt = _intensive_optimize(a0, slice_deadline, rng)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[global] new best n={n:4d} val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) final evaluation via the external black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a uniform sequence
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
