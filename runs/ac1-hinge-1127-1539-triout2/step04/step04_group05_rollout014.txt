def _adam_local_optimize(a0, deadline, rng):

    n = len(a)

    b = _full_convolution_fft(a)

    t = 0

    while time.time() < deadline:

        # Pick update method

def _softmax_gradient(a, tau):
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)  # shape 2n-1
    # compute correlation a_rev * w . Actually conv of w with a_rev.
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    W = np.fft.rfft(w, n=f_len)
    A_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(W * A_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n-1:n-1+n]
    return g, max_b
```

"""
Improved heuristic for minimizing the self‑convolution L∞‑norm.
The optimizer works on the smooth log‑sum‑exp approximation of the
maximum, using an Adam‑style projected gradient descent together with
sub‑gradient steps, random mass‑swap moves and occasional Gaussian jitter.
Starting points are diverse (random, sparse, semi‑circular, bell‑shaped)
and the algorithm performs several restarts with varied lengths.
The best sequence found is stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum x = S}."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - S
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full, length 2·n‑1)."""
    return _fft_convolve(a, a)


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of seq to length new_n."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


# --------------------------------------------------------------
# Gradient / sub‑gradient utilities
# --------------------------------------------------------------
def _softmax_weights_and_grad(a: np.ndarray, tau: float):
    """
    Returns (w, g, max_b) where
        w – soft‑max weights over the convolution entries,
        g – gradient of τ·log‑sum‑exp of the convolution,
        max_b – current maximum entry of a*a.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau <= 1e-12:
        # Hard max – return uniform weights (gradient will be the sub‑gradient)
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        diff = b - max_b                     # ≤ 0
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)              # shape (2n‑1,)

    # correlation a_rev ⋆ w
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    W = np.fft.rfft(w, n=f_len)
    A_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(W * A_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]          # length n
    return w, g, max_b


def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    """
    Sub‑gradient of max(a*a) and the maximal entry.
    All entries within eps of the max are treated as active.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)

    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)          # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# --------------------------------------------------------------
# Initial sequence generator
# --------------------------------------------------------------
def _generate_initial(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a feasible (non‑negative, ℓ₁ = √(2n)) start vector."""
    S = math.sqrt(2.0 * n)

    # choose a random “style”
    style = rng.choice(['random', 'sparse', 'semicircle', 'bell'])
    if style == 'sparse':
        K = max(int(2 * math.sqrt(n)), 5)
        idx = rng.choice(n, size=K, replace=False)
        a = np.zeros(n)
        # give the selected positions random positive weights
        a[idx] = rng.random(K) + 0.5
    elif style == 'semicircle':
        x = np.linspace(-1.0, 1.0, n)
        a = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    elif style == 'bell':
        x = np.linspace(0.0, 1.0, n)
        sigma = 0.2
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    else:                                   # uniform Dirichlet draw
        a = rng.random(n)

    # optionally blend a bit of the previous best construction
    if base_seq is not None and base_seq.size > 0:
        a = a + 0.2 * _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)

    # project onto the simplex
    a = _project_to_simplex(a, S)

    # occasional symmetrisation (helps avoid asymmetric artefacts)
    if rng.random() < 0.5:
        a = (a + a[::-1]) / 2.0
        a = _project_to_simplex(a, S)

    return a


# --------------------------------------------------------------
# Adam‑style local optimiser
# --------------------------------------------------------------
def _adam_local_optimize(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the τ‑soft‑max approximation,
    interleaved with exact sub‑gradient steps, random swaps and jitter.
    Returns the best vector seen and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # initialise bookkeeping
    b = _full_convolution_fft(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.7

    # schedules
    tau = cur_max
    tau_decay = 0.9996
    temperature = cur_max * 0.15
    temp_decay = 0.9995
    move_prob = 0.08          # fraction of mass moved in a swap
    jitter_prob = 0.02
    jitter_scale = 0.02

    it = 0
    while time.time() < deadline:
        it += 1
        # ---------------- 1) pick an update type ----------------
        r = rng.random()
        if r < 0.55:
            # ---- smooth soft‑max Adam step ----
            _, grad, max_b = _softmax_weights_and_grad(a, max(tau, 1e-12))
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_candidate = a - step
            a = _project_to_simplex(a_candidate, S)

            # optional symmetric enforcement every few iterations
            if it % 250 == 0:
                a = (a + a[::-1]) / 2.0
                a = _project_to_simplex(a, S)

            cur_max = _full_convolution_fft(a).max()
            tau *= tau_decay

        elif r < 0.80:
            # ---- exact sub‑gradient step with tiny line‑search ----
            g, max_b = _subgradient_max(a)
            improved = False
            for factor in (0.5, 1.0, 1.5):
                a_cand = a - base_lr * factor * g
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max - 1e-9:
                    a = a_cand
                    cur_max = max_cand
                    improved = True
                    break
            if not improved:
                # simulated‑annealing acceptance
                a_cand = a - base_lr * g
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max or rng.random() < math.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                    a = a_cand
                    cur_max = max_cand
                temperature *= temp_decay

        elif r < 0.95:
            # ---- random mass‑swap move (simulated annealing) ----
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                delta = move_prob * a[i]
                a_cand = a.copy()
                a_cand[i] -= delta
                a_cand[j] += delta
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max or rng.random() < math.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                    a = a_cand
                    cur_max = max_cand
                temperature *= temp_decay

        else:
            # ---- Gaussian jitter ----
            if rng.random() < jitter_prob:
                noise = rng.normal(scale=jitter_scale * a.mean(), size=n)
                a_cand = np.clip(a + noise, 0.0, None)
                a_cand = _project_to_simplex(a_cand, S)
                cur_max = _full_convolution_fft(a_cand).max()
                a = a_cand

        # ---------------- 2) bookkeeping ----------------
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # occasional tiny decay of learning rate
        base_lr *= 0.9999

    return best_a, best_max

# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Time‑limited search for a non‑negative step sequence with minimal
    self‑convolution L∞‑norm.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0                         # spare a few seconds for final I/O
    end_time = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – polish the existing construction (if any)
    # ------------------------------------------------------------------
    if base_seq is not None and base_seq.size > 0:
        n0 = int(base_seq.shape[0])
        a0 = _project_to_simplex(base_seq, math.sqrt(2.0 * n0))
        phase_deadline = min(time.time() + 150.0, end_time)  # up to 2½ min
        seq_opt, score_opt = _adam_local_optimize(a0, phase_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[phase 1] refined length={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many independent restarts with varied lengths
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # pick a problem size (moderate ≈ 3000‑8000, occasionally larger)
        if rng.random() < 0.7:
            n = rng.integers(3000, 8000)
        else:
            n = rng.integers(8000, 18000)

        a0 = _generate_initial(base_seq, n, rng)

        # allocate a time slice for this restart (≈10 s, but never exceed deadline)
        slice_deadline = min(time.time() + 12.0, end_time)

        seq_opt, score_opt = _adam_local_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final refinement on the best found sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 10.0:
            # upscale to a slightly larger length to exploit more degrees of freedom
            target_n = min(25000, int(len(best_seq) * 1.4) + 1)
            a0 = _scale_sequence(best_seq, target_n)
            a0 = _project_to_simplex(a0, math.sqrt(2.0 * target_n))
            final_deadline = end_time
            seq_opt, score_opt = _adam_local_optimize(a0, final_deadline, rng)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = seq_opt.copy()
                print(f"[final‑refine] upscaled length={target_n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the public black‑box routine
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                     # fallback – should never happen
    else:
        final_seq = best_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] external score = {external_score:.6f} "
              f"(internal max = {best_score:.6f}) length = {len(final_seq)}")

    # expose the best construction for the next run
    globals()['height_sequence_1'] = final_seq
    return final_seq
