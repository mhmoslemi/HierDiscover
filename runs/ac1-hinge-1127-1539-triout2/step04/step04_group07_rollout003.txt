def _full_convolution(a):
    n = a.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv

def _local_optimize_subgrad(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    conv = _full_convolution(a)
    cur_val = np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    it = 0
    last_improve_it = 0
    base_lr = 1.0
    target_move_prob = 0.5
    random_move_prob = 0.1
    swap_frac = 0.05
    while time.time() < deadline:
        it += 1
        # recompute conv and max if needed? We already have conv. We'll use cur_val from previous iteration.
        # Compute gradient of current max.
        k_max = int(np.argmax(conv))
        # compute gradient
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        i_range = np.arange(lo, hi+1)
        j_range = k_max - i_range
        grad = np.zeros_like(a)
        grad[i_range] = 2.0 * a[j_range]

        # step size
        lr = base_lr / np.sqrt(it)

        # Euclidean gradient step
        a_candidate = a - lr * grad
        a_candidate = _project_simplex(a_candidate, S)
        conv_candidate = _full_convolution(a_candidate)
        val_candidate = np.max(conv_candidate)
        if val_candidate < cur_val:
            a = a_candidate
            conv = conv_candidate
            cur_val = val_candidate
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                last_improve_it = it

        # random swap move
        if rng.random() < random_move_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                a_swap = a.copy()
                a_swap[i] -= delta
                a_swap[j] += delta
                # a_swap remains sum unchanged
                conv_swap = _full_convolution(a_swap)
                val_swap = np.max(conv_swap)
                if val_swap < cur_val:
                    a = a_swap
                    conv = conv_swap
                    cur_val = val_swap
                    if cur_val < best_val:
                        best_val = cur_val
                        best_a = a.copy()
                        last_improve_it = it

        # targeted move
        if rng.random() < target_move_prob:
            # contributions for k_max
            contrib = a[i_range] * a[j_range]
            src_idx = i_range[np.argmax(contrib)]
            tgt_idx = i_range[np.argmin(contrib)]
            if a[src_idx] > 0 and src_idx != tgt_idx:
                delta = swap_frac * a[src_idx] * rng.random()
                a_target = a.copy()
                a_target[src_idx] -= delta

                conv_target = _full_convolution(a_target)
                val_target = np.max(conv_target)

                    a = a_target
                    conv = conv_target
                    cur_val = val_target
                    if cur_val < best_val:
                        best_val = cur_val
                        best_a = a.copy()
                        last_improve_it = it

        # early stop if no improvement for many iter
        if it - last_improve_it > 5000:
            break

    return best_a, best_val

"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.
The algorithm combines:
* simplex projection (L1‑normalisation to Σa = √(2·n));
* Euclidean sub‑gradient descent on the ℓ∞‑norm of the self‑convolution;
* occasional random mass swaps and targeted “flattening’’ moves that shift
  mass from large contributions to small ones.
Multiple restarts with different lengths and initialisations are performed
until the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Exponential warm‑start (same as the reference code)
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """Decreasing exponential sequence of length n, projected onto the simplex."""
    S = np.sqrt(2.0 * n)
    lam = k_factor / n                # λ = k_factor / n
    q = np.exp(-lam)                  # q = e^{-λ}
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)   # = 1 - q**n
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Local optimiser – sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _local_optimize_subgrad(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the maximum entry of the self‑convolution of ``a0``.
    Returns the best sequence seen (projected onto the ℓ1‑simplex) and its
    corresponding max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # current state
    conv = _full_convolution(a)
    cur_val = np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    # hyper‑parameters
    base_lr = 1.0               # learning‑rate scaling for Euclidean steps
    random_move_prob = 0.12     # probability of a random mass swap per iteration
    target_move_prob = 0.55     # probability of a targeted flattening move
    swap_frac = 0.05            # fraction of source mass moved in swaps
    max_no_improve = 6000       # early‑stop if no improvement for many steps

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # ---- 1. sub‑gradient step for the current maximal entry ----
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        i_range = np.arange(lo, hi + 1)
        j_range = k_max - i_range

        grad = np.zeros_like(a)
        grad[i_range] = 2.0 * a[j_range]   # d(conv_k)/da_i = 2*a_{k-i}

        lr = base_lr / np.sqrt(it)        # diminishing step size
        a_candidate = a - lr * grad
        a_candidate = _project_simplex(a_candidate, S)

        conv_candidate = _full_convolution(a_candidate)
        cand_val = np.max(conv_candidate)

        if cand_val < cur_val:
            a = a_candidate
            conv = conv_candidate
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                last_improve = it

        # ---- 2. occasional random mass‑swap (exploration) ----
        if rng.random() < random_move_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                a_swap = a.copy()
                a_swap[i] -= delta
                a_swap[j] += delta
                # projection is not required – the sum is unchanged and
                # we never create negativity because delta ≤ a[i]
                conv_swap = _full_convolution(a_swap)
                swap_val = np.max(conv_swap)
                if swap_val < cur_val:
                    a = a_swap
                    conv = conv_swap
                    cur_val = swap_val
                    if cur_val < best_val:
                        best_val = cur_val
                        best_a = a.copy()
                        last_improve = it

        # ---- 3. targeted flattening move (reduces the dominant contribution) ----
        if rng.random() < target_move_prob:
            # contributions of the current maximal index
            contrib = a[i_range] * a[j_range]
            src_idx = i_range[np.argmax(contrib)]
            tgt_idx = i_range[np.argmin(contrib)]
            if src_idx != tgt_idx and a[src_idx] > 0.0:
                delta = swap_frac * a[src_idx] * rng.random()
                a_target = a.copy()
                a_target[src_idx] -= delta
                a_target[tgt_idx] += delta
                conv_target = _full_convolution(a_target)
                target_val = np.max(conv_target)
                if target_val < cur_val:
                    a = a_target
                    conv = conv_target
                    cur_val = target_val
                    if cur_val < best_val:
                        best_val = cur_val
                        best_a = a.copy()
                        last_improve = it

        # ---- 4. early termination if stagnated too long ----
        if it - last_improve > max_no_improve:
            break

    return best_a, best_val
