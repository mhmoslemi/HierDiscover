from scipy.optimize import linprog
import scipy.sparse as sp

def lp_optimize(a, M):
    n = len(a)
    # Build the constraint matrix: each row k (0 to 2n-2)
    # For each k, columns i where 0 <= k-i < n == i in [max(0,k-n+1), min(k,n-1)]
    # The coefficient is a[i]
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(k, n-1)
        # columns i from lo..hi
        cols.extend(range(lo, hi+1))
        rows.extend([k]*(hi-lo+1))
        data.extend([a[i] for i in range(lo, hi+1)])
    A_ineq = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # objective: minimize -sum(g)
    c = -np.ones(n)
    # bounds: (0, None) for each var
    bounds = [(0, None)]*n
    # inequality: A_ineq @ g <= M
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A_ineq, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
```

"""
author:   OpenAI ChatGPT
date:     2026-11-27
purpose:  Find a non‑negative step‑function (height sequence) that
          minimises the benchmark score defined by
          ``evaluate_sequence``.  The method combines three
          optimisation stages:

          1.  Initialise several candidates (global best, power‑law,
              Dirichlet random).  All candidates are projected onto the
              ℓ¹‑simplex Σa = √(2·n).

          2.  Projected gradient descent on a soft‑max approximation of
              the convolution maximum.  The “temperature’’ β is increased
              up to 1024, giving a very sharp approximation of the true
              max‑operator.

          3.  Simulated‑annealing refinement.  Moves consist of mass
              transfer from a high‑sensitivity index to a low‑sensitivity
              one, short‑window smoothing, and occasional random swaps.
              A warm start (high temperature 0.5) and a slowly decaying
              move‑size schedule allow a thorough exploration of the
              search space.

          The optimiser runs for the whole allotted time (default 1000 s)
          and returns the best sequence discovered.
"""

import time
import numpy as np
import scipy.optimize  # only imported for the optional LP step (not used here)
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (max convolution) for a normalised sum."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex  {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient of the soft‑max w.r.t. a
    a_rev = a[::-1]
    size2 = 1 << ((2 * n - 1).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation of the max.
    The temperature parameter ``beta`` is increased up to 1024.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1024.0            # higher than the original 256
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # keep feasibility
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement: β={beta:.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.40,
                         max_frac_end: float = 0.005,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves:
      * donor→receiver mass transfer (≈70 % of the iterations);
      * local smoothing (replace a short window by its average);
      * random index swap (≈10 % of the iterations) to diversify the search.
    Temperature decays exponentially, as does the allowed move size.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)                     # current convolution

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current statistics
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity of each coordinate for the active (maximum) entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.70:                      # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:                    # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        else:                                         # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ------------------------------------------------------------------
        # Evaluate the new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _run_one_instance(n: int, init_mode: str, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a single instance of length ``n``.
    ``init_mode`` can be 'global', 'power', or 'random'.
    Returns the best sequence found and its score.
    """
    target = np.sqrt(2.0 * n)

    if init_mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    elif init_mode == "power":
        exp = 0.55                     # a reasonable default exponent
        seq = _power_law_sequence(n, exp, target)
    elif init_mode == "random":
        rng = np.random.default_rng()
        seq = rng.random(n)
        seq = seq / seq.sum() * target
    else:
        seq = _power_law_sequence(n, 0.55, target)

    # Make sure we sit on the simplex
    seq = _project_to_simplex(seq, target)

    # -------- gradient descent stage ------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        gd_deadline = time.time() + min(30.0, remaining * 0.30)
        seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)
    else:
        gd_score = _score(seq)

    # -------- simulated annealing stage ---------------------------------
    cur_time = time.time()
    if cur_time < deadline:
        seq, sa_score = _simulated_annealing(seq, deadline)
        final_score = min(gd_score, sa_score)
    else:
        final_score = gd_score

    return seq, final_score
