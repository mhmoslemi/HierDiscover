```python
"""
Algorithm: Hybrid search for step-function minimizing the self‑convolution
max‑norm (the objective used by ``evaluate_sequence``).

* Start from a random Dirichlet draw (or re‑use the previously discovered
  ``height_sequence_1``) on the L1‑simplex Σa = √(2·n).
* Run a smooth SGD phase with a soft‑max approximation of the max‑norm,
  followed by a sub‑gradient phase focusing on the worst convolution entry.
  This quickly finds a decent candidate (≈1.5).
* Refine the candidate with an *alternating linear‑program* step:
  – For the current vector f compute M = max(f∗f).
  – Solve the LP   maximise Σ b_j   subject to   b_j≥0  and  (f∗b)_k ≤ M.
    The LP finds a complementary vector b that can carry additional mass
    without increasing the worst convolution value.
  – Scale b to the simplex, mix it with f (convex combination) and keep the
    mixture if the max‑convolution decreases.
  Several such LP refinements drive the objective close to the theoretical
  lower bound 1+1/(2n‑1) (≈1.0 for large n).  The whole procedure respects the
  1000 s budget and returns the best sequence found.
"""

import time
import numpy as np
import math

# optional SciPy – used only for the LP refinement
try:
    import scipy.optimize
    import scipy.sparse
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j) (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generator
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If ``base_seq`` is supplied and long enough, its leading part (with tiny jitter)
    is used; otherwise a Dirichlet/Uniform sample is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)               # Dirichlet(1) – uniform on simplex.

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Smooth optimisation phase (soft‑max + SGD)
# ----------------------------------------------------------------------


def _optimize_smooth(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Gradient descent on the smooth max‑norm (soft‑max approximation)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (tuned for stability)
    lr = 0.4
    lr_decay = 0.9995
    alpha = 0.5
    alpha_mul = 1.006
    swap_prob = 0.08
    swap_frac = 0.025

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        a = a - lr * grad
        a = _project_simplex(a, S)

        # occasional random mass‑swap to escape plateaus
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand
                    conv = _full_convolution(a)

        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # schedule updates
        lr *= lr_decay
        alpha = min(alpha * alpha_mul, 30.0)

        # crude early stop
        if it % 5000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation phase (focus on the worst shift)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst entry)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: reversed slice of a aligned with the worst shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (smooth → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth phase for ~60 % of the time budget, then finish
    with the sub‑gradient phase.  Return the best intermediate result.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_smooth(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating LP)
# ----------------------------------------------------------------------


def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise Σ b_j   subject to   b_j ≥ 0  and (f ∗ b)_k ≤ M
    for every shift k.  Returns the optimal vector b (numpy array) or
    None if the solver fails.
    """
    if not _HAS_SCIPY:  # pragma: no cover
        return None

    n = f.shape[0]
    # Build sparse matrix A where A[k, j] = f[k-j] if 0≤k-j<n else 0
    rows = []
    cols = []
    data = []

    # Iterate over i (index in f) and add its contribution to all relevant rows
    for i in range(n):
        fi = f[i]
        # For a given i, the term f[i] * b[j] appears in shift k = i + j
        # where j ranges 0..n-1. Therefore rows = i + j, cols = j.
        j_start = 0
        j_end = n - 1
        # Rather than inner loop over j, we fill using vectorised pattern
        # but a pure Python loop is acceptable for n ≤ 2500.
        for j in range(n):
            rows.append(i + j)
            cols.append(j)
            data.append(fi)

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    c = -np.ones(n)                     # maximise sum → minimise -sum
    bounds = [(0, None)] * n

    # Use the high‑performance HiGHS interior‑point solver
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return None
    return res.x


def _refine_with_lp(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Apply a few alternating‑LP refinements to a candidate vector ``a``.
    Returns the improved vector and its max‑convolution value.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    while time.time() < deadline:
        conv = _full_convolution(best_a)
        M = np.max(conv)

        b = _solve_lp_max_sum(best_a, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Scale b back to the simplex Σb = √(2·n)
        scale = np.sqrt(2.0 * n) / sum_b
        g = b * scale

        # Try several mixing coefficients; keep whichever improves the max.
        improved = False
        for t in (0.25, 0.5, 0.75):
            cand = (1 - t) * best_a + t * g
            cand_max = np.max(_full_convolution(cand))
            if cand_max + 1e-12 < best_val:
                best_a = cand
                best_val = cand_max
                improved = True
                break
        if not improved:
            # No improvement – stop the LP loop
            break

    return best_a, best_val

"""
Hybrid optimisation for the self‑convolution max‑norm.

The algorithm works in three stages:

1. **Initialisation** – start from a random Dirichlet vector (or reuse the
   previously discovered ``height_sequence_1``) that lies on the L¹‑simplex
   Σa = √(2·n).

2. **Gradient descent** – a smooth SGD phase with a soft‑max approximation
   of the max‑norm followed by a sub‑gradient phase that directly attacks
   the worst convolution entry.  This quickly yields a respectable candidate
   (≈ 1.5).

3. **Alternating LP refinement** – for the current vector *f* compute
   its maximal convolution value *M*.  Solve the linear programme

        maximise   Σ b_j
        subject to  b_j ≥ 0
                    (f ∗ b)_k ≤ M   for every shift k

   The LP finds a complementary vector *b* that can carry additional mass
   without raising the worst convolution value.  After scaling *b* to the
   simplex we mix it with *f*.  Repeating this step drives the objective
   close to its theoretical lower bound 1 + 1/(2n‑1) (≈ 1.0 for large *n*).

The whole routine respects the 1000 s budget and returns the best sequence
found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – required only for the LP refinement stage
# ----------------------------------------------------------------------
try:                                   # pragma: no cover
    import scipy.optimize
    import scipy.sparse
    _HAS_SCIPY = True
except Exception:                     # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project *v* onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1       # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)       # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j) (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generator
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length *n* with L¹‑norm √(2·n).
    If ``base_seq`` is supplied and long enough, its leading part (with tiny
    jitter) is used; otherwise a Dirichlet/Uniform sample is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)               # Dirichlet(1) – uniform on simplex.

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Smooth optimisation phase (soft‑max + SGD)
# ----------------------------------------------------------------------
def _optimize_smooth(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Gradient descent on the smooth max‑norm (soft‑max approximation)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (tuned for stability)
    lr = 0.4
    lr_decay = 0.9995
    alpha = 0.5
    alpha_mul = 1.006
    swap_prob = 0.08
    swap_frac = 0.025

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        a = a - lr * grad
        a = _project_simplex(a, S)

        # occasional random mass‑swap to escape plateaus
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand
                    conv = _full_convolution(a)

        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        lr *= lr_decay
        alpha = min(alpha * alpha_mul, 30.0)

        if it % 5000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation phase (focus on the worst shift)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst entry)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: reversed slice of a aligned with the worst shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (smooth → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth phase for ~60 % of the time budget,
    then finish with the sub‑gradient phase.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_smooth(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating LP)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve
        maximise Σ b_j
        subject to b_j ≥ 0 and (f ∗ b)_k ≤ M   for every shift k.
    Returns the optimal vector *b* (numpy array) or None on failure.
    """
    if not _HAS_SCIPY:                          # pragma: no cover
        return None

    n = f.shape[0]

    # Build the sparse constraint matrix A where
    #   A[k, j] = f[k - j]  if 0 ≤ k‑j < n, otherwise 0.
    rows = []
    cols = []
    data = []

    idx = np.arange(n, dtype=int)
    for i in range(n):
        rows.extend((i + idx).tolist())
        cols.extend(idx.tolist())
        data.extend([f[i]] * n)

    A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    c = -np.ones(n)                     # maximise sum → minimise -sum
    bounds = [(0, None)] * n

    # HiGHS is very fast for dense‑ish problems; we give it a short time limit.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True,
                                          'time_limit': 30})
    if not res.success:
        return None
    return res.x


def _refine_with_lp(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Apply a few alternating‑LP refinements to *a*.
    Returns the improved vector and its maximal convolution value.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    while time.time() < deadline:
        conv = _full_convolution(best_a)
        M = np.max(conv)

        b = _solve_lp_max_sum(best_a, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Scale back to the simplex Σb = √(2·n)
        scale = np.sqrt(2.0 * n) / sum_b
        g = b * scale

        improved = False
        # Try a handful of mixing coefficients and keep the first improvement.
        for t in (0.25, 0.5, 0.75):
            cand = (1 - t) * best_a + t * g
            cand_max = np.max(_full_convolution(cand))
            if cand_max + 1e-12 < best_val:
                best_a = cand
                best_val = cand_max
                improved = True
                break

        if not improved:
            break                          # no further progress

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    The search consists of three stages:

      1. random initialisation (or reuse ``height_sequence_1``);
      2. smooth + sub‑gradient optimisation;
      3. alternating‑LP refinement.

    The function respects the supplied time budget and finally returns the
    best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for the final check

    # ------------------------------------------------------------------
    # Load any previously discovered construction
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    last_improve = start_time
    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # --------------------------------------------------------------
        # Choose a problem size.  Keep the current size if it is improving,
        # otherwise restart with a fresh length.
        # --------------------------------------------------------------
        if best_overall_seq is None or (time.time() - last_improve > 20):
            n = rng.integers(1500, 3000)            # sweet‑spot for LP stage
            a0 = _initial_sequence(base_seq, n, rng)
        else:
            a0 = best_overall_seq.copy()
            n = a0.shape[0]

        # Allocate a slice of the remaining budget for this attempt
        slice_deadline = min(time.time() + 30.0, deadline)

        # --------------------------------------------------------------
        # Stage 1 / 2 : smooth → sub‑gradient optimisation
        # --------------------------------------------------------------
        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        # --------------------------------------------------------------
        # Stage 3 : LP‑based refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            lp_deadline = min(time.time() + 15.0, deadline)
            a_opt, score = _refine_with_lp(a_opt, lp_deadline, rng)

        # Record the best candidate seen so far
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} "
                  f"new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
