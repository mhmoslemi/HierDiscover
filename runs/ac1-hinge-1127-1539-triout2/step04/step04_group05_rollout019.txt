"""
Generate a monotone decreasing power‑law step sequence
    a_i ∝ (n‑i)^p ,   i = 0 … n‑1
and scale it to the required ℓ₁‑norm  √(2·n).

For the shape f(x) = (p+1)(1‑x)^p on [0,1] the self‑convolution satisfies

    max_k (f*f)(k/n) = 2·(p+1)²·B(p+1,p+1),

where B is the Beta function.  The corresponding discrete bound is
approximately this constant and can be made arbitrarily small by
increasing p.  The only competing term is the very first
convolution entry a₀² ≈ 2·(p+1)² / n, which becomes negligible for
moderate n (here n≈8000).

The routine tests a few small exponents (p=1…8), evaluates the
objective with the provided `evaluate_sequence` black‑box and returns
the best sequence found.  All numbers are non‑negative, well below the
1000‑clip, and the ℓ₁‑norm is exactly √(2·n), so the returned score is
the maximal self‑convolution entry.
"""

import numpy as np

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Build a decreasing power‑law sequence a_i = (n‑i)^p, normalise it,
    evaluate a few exponents and return the best one.
    """
    # length – large enough that the a₀² term is tiny, but still cheap to evaluate
    n = 8000                     # can be increased if time permits
    S = np.sqrt(2.0 * n)         # required ℓ₁‑norm

    best_seq = None
    best_score = float('inf')
    best_p = None

    # try several exponents; higher p gives lower theoretical bound
    for p in range(1, 9):        # p = 1 … 8
        # raw decreasing powers: n, n-1, …, 1  raised to p
        raw = np.arange(n, 0, -1, dtype=np.float64) ** p
        scale = S / raw.sum()   # scale so that sum = √(2·n)
        seq = raw * scale

        # evaluate the candidate with the supplied black‑box
        score = evaluate_sequence(seq.tolist())
        print(f"p={p:2d} → external score = {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_p = p

    # store the best construction globally (useful for later runs)
    globals()['height_sequence_1'] = best_seq.tolist()
    print(f"[final] chosen p={best_p}, n={n}, score={best_score:.6f}")

    return best_seq.tolist()
