"""
Hybrid search for a step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages and respects the 1000 s budget:

1. **Wide parametric sweep** – a larger collection of symmetric
   “U‑shaped’’ families (including new “ends‑biased’’ shapes) and
   a few random constructions are projected onto the ℓ₁‑simplex.
   The best candidate is kept.

2. **Smoothed gradient descent** – minimises a log‑sum‑exp
   surrogate of the maximum using momentum.  The smoothing
   parameter τ is reduced geometrically; the step size adapts
   with a small decay.

3. **Sub‑gradient refinement** – switches to the true nonsmooth
   objective.  The sub‑gradient is taken as the average of the
   gradients of all (approximately) maximal lags.  A simple
   decreasing step‑size schedule is used.

4. **Simulated‑annealing polishing** – tiny random pair‑mass
   transfers are accepted with a Metropolis criterion whose
   temperature decays to zero.  This final burst often squeezes
   out the last few hundredths of the bound.

The routine returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another
    and re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str,
                  param: float, rng: np.random.Generator = None) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or end‑biased sequence.
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    if rng is None:
        rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    # distance to centre   ∈ [0,1]
    d_center = np.abs(idx - mid) / mid
    # distance to the nearest end ∈ [0,1] (0 at the ends, 1 in the middle)
    d_end = np.minimum(idx, n - 1 - idx) / mid

    # ----- families ----------------------------------------------------
    if kind == "linear":
        raw = 1.0 + param * d_center
    elif kind == "quadratic":
        raw = 1.0 + param * d_center ** 2
    elif kind == "exponential":
        raw = np.exp(param * d_center)
    elif kind == "power":
        raw = d_center ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d_center ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))

    # ----- new end‑biased families --------------------------------------
    elif kind == "ends_linear":
        # high at the ends, linear decay to the centre
        raw = 1.0 - param * d_end
    elif kind == "ends_power":
        raw = (1.0 - d_end) ** param
    elif kind == "ends_exponential":
        raw = np.exp(-param * d_end)
    elif kind == "ends_inverse":
        raw = 1.0 / (1e-4 + d_end ** param)

    # ----- random distance‑sorted construction ---------------------------
    elif kind == "sorted_by_distance":
        # draw random values and place the biggest at the ends
        vals = rng.random(n)
        order = np.argsort(d_end)                # ends first
        raw = np.empty_like(vals)
        raw[order] = np.sort(vals)[::-1]        # descending into ends‑first order
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution of `a`."""
    return float(conv_self_fft(a).max())

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

        "power": np.linspace(0.5, 5.0, 12),

        "exponential_left": np.linspace(0.0, 6.0, 31),
        "exponential_right": np.linspace(0.0, 6.0, 31),
        "ends_linear": np.linspace(0.0, 0.9, 19),
        "ends_power": np.linspace(0.5, 5.0, 12),

                score = compute_max_conv(cand)
                if score < best_score:

                    best_seq = cand.copy()
                    print(f"  New best {kind} (random) → {best_score:.6f}")
                if time.time() >= deadline:
                    print("Time exhausted during sweep.")
                    return best_seq.tolist()
            continue

        grid = param_grids.get(kind, np.linspace(0.0, 0.0, 1))
        for p in grid:
            cand = generate_shape(n, target_sum, kind, p, rng=rng)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # a few end‑biased random vectors (concentration towards the ends)
    for _ in range(12):
        d_center = np.abs(np.arange(n) - (n - 1) / 2.0) / ((n - 1) / 2.0)
        bias = 1.0 - d_center          # larger near the ends

        cand = project_onto_simplex(cand_raw, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:

            best_seq = cand.copy()

            return best_seq.tolist()

        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves start → {best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    smoothed_deadline = start_time + budget_s * 0.55
    a = best_seq.copy()

    eta = 0.5 * (target_sum / n)        # initial step size (a bit smaller than before)
    beta = 0.92                         # momentum factor
    tau_factor = 0.12
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Phase 1 – smoothed gradient descent")
    while time.time() < smoothed_deadline:
        conv = conv_self_fft(a)

        w = np.exp((conv - max_conv) / tau)

            w /= w_sum

    subgrad_deadline = start_time + budget_s * 0.90

        a_rev = a[::-1]

        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        a -= step * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        sub_iter += 1
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Sub‑grad iter {sub_iter:06d} – improved to {best_score:.6f}")

        if sub_iter % 600 == 0:
            step *= 0.94

        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.006)

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – polishing (simulated annealing)")
    current = best_seq.copy()
    temp_initial = 0.015 * best_score          # temperature in objective units
    temp_final = 1e-6 * best_score
    scale_initial = 0.004 * (target_sum / n)   # big jitter at start of polishing
    scale_final = 0.0004 * (target_sum / n)    # very fine at the end

    while time.time() < deadline:
        # linear schedule based on elapsed fraction
        elapsed = time.time() - start_time
        fraction = (elapsed) / (deadline - start_time)
        temp = temp_initial * (temp_final / temp_initial) ** fraction
        scale = scale_initial * (1.0 - fraction) + scale_final * fraction

        cand = random_pair_transfer(current.copy(), target_sum, scale=scale)
        cand_score = compute_max_conv(cand)

        if cand_score < best_score - 1e-12:

            best_seq = cand.copy()
            current = cand.copy()
            print(f"Polish improvement → {best_score:.6f}")

            prob = math.exp(-(cand_score - best_score) / (temp + 1e-18))
            if np.random.rand() < prob:
                current = cand.copy()

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
