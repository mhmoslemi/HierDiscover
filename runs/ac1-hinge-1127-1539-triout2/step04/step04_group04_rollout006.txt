"""
Search for a step‑function with a small self‑convolution peak.

Algorithm
---------
1. Load the previous best sequence (if any).
2. Phase 1 – Random sampling.
   * Lengths from 200 up to 12 000.
   * Shapes: exponential, symmetric‑exponential, Gaussian,
     triangular, power‑law, cosine, Dirichlet and a *symmetric Beta*
     family (a_i ∝ x^{α‑1}(1‑x)^{α‑1}), the latter being new.
   * Each raw shape is normalised to the ℓ₁‑simplex
     Σa = √(2 n) (any scaling works because the objective is 0‑homogeneous).
   * Keep the vector with the smallest internal objective
     2·n·max(a∗a)/(Σa)².
3. Phase 2 – Adam optimisation.
   * Minimise a smooth soft‑max approximation of max(a∗a).
   * After every step the vector is projected back onto the simplex.
   * Learning‑rate and soft‑max temperature are annealed.
4. Phase 3 – Tiny random hill‑climbing moves (move a small amount of mass
   from one index to another) until the time budget expires.
5. Return the best vector found; store it in the global
   `height_sequence_1` for later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the ℓ₁‑simplex {x ≥ 0, Σx = S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum S."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(a*a)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Compute gradient of τ·log∑exp(b/τ) where b = a*a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    # Hard max → sub‑gradient
    if tau < 1e-12:
        k = np.argmax(b)                     # index of the current maximum
        g = np.zeros_like(a)
        # contribution of pairs (i, j) with i+j = k
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal term counted only once
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # Soft‑max weights w_k ∝ exp(b_k / τ) (stable version)
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)                    # shape (2n‑1,)

    # Compute g = 2·a ⋆ w   (cross‑correlation)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]           # keep length n
    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Minimise the (soft) max of the self‑convolution using Adam.
    Returns the best vector encountered and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6
    t = 0

    # initialise temperature from a hard‑max scale
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-12

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, b, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluate
        max_new = np.max(_full_convolution_fft(a_new))
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()

        a = a_new

        # annealing
        tau = max(tau * 0.9995, 1e-9)
        lr *= 0.9999

        # occasional tiny jitter to escape plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# ``_random_distribution`` – same families as the previous run
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular",
         "power", "cosine", "dirichlet"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "dirichlet":
        # concentration < 1 favours extreme entries → good for flattening
        conc = rng.uniform(0.2, 0.9)
        a = rng.gamma(conc, 1.0, size=n)

    else:
        a = rng.random(n)

    # small random jitter
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# Beta family (symmetric) – new ingredient
# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1} (mid‑point rule)."""
    # use mid‑point discretisation to avoid exact zeros at the boundaries
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n                     # in (0,1)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    # Guard against possible NaNs for extreme α
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    return a
