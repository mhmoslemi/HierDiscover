"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.
---------------------------------------------------------------------

The objective of ``evaluate_sequence`` (after normalising the sum to
√(2·n)) is exactly the infinity‑norm of the auto‑convolution
``a*a``.  For a non‑negative vector the quantity is convex but
non‑smooth.  An analytically tractable family of sequences is the
linear‑decreasing shape

        a_i ∝ (n‑i) ,  i = 0,…,n‑1 .

A short calculation shows that, after normalising the ℓ₁‑norm to
√(2·n), the maximal convolution entry tends to 4/3 ≈ 1.333 for large
n – already well below the target 1.503.  The algorithm therefore
tries a handful of lengths, builds the linear shape, projects it onto
the simplex (to guarantee the exact required sum), and keeps the best
one.  If time permits a very cheap hill‑climbing refinement with random
mass‑swaps is performed.

The whole routine respects the 1000 s time budget, prints progress,
and finally returns the best sequence as a plain list of ``float``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helper: Euclidean projection onto the ℓ₁‑simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Simple hill‑climbing refinements (optional)
# ----------------------------------------------------------------------


def _refine_sequence(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap random mass‑swap hill climbing."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(best_a))

    # fraction of source mass moved in a trial (tuned for fast progress)
    swap_frac = 0.02

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        # move a random fraction of a[i] from i to j
        delta = swap_frac * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_a = cand
            best_score = cand_score
            a = cand   # continue from the improved point
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The search is essentially analytical:
    a linear decreasing profile a_i ∝ (n‑i) yields a theoretical bound
    of 4/3 ≈ 1.333.  Several lengths are examined, the best is kept,
    and an optional brief local refinement is performed.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) try the analytically motivated linear decreasing shape for a few
    #    candidate lengths
    # ------------------------------------------------------------------
    candidate_lengths = (800, 1200, 2000, 3000, 4000, 5000)
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        # raw decreasing integer vector n, n‑1, … , 1
        raw = np.arange(n, 0, -1, dtype=float)
        S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm
        factor = S / raw.sum()                  # scaling to the simplex
        a = raw * factor
        # the projection is inexpensive and guarantees the exact sum
        a = _project_simplex(a, S)
        cur_score = float(np.max(_full_convolution(a)))   # = objective
        print(f"[linear] n={n:4d} score={cur_score:.6f}")
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 2) optional tiny refinement if we still have time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # give the hill‑climber about 20 s (or the remaining time)
        slice_deadline = min(deadline, time.time() + 20.0)
        a_refined, refined_score = _refine_sequence(best_seq, slice_deadline, rng)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_refined.copy()
            print(f"[refine] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) fallback (should never happen) – uniform sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # ------------------------------------------------------------------
    # 4) final external evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
