"""
Proposal: Gradient–Adam optimiser with monotone projection.
Searches over a range of lengths and initial shapes.
"""
import time
import numpy as np

# Simplex projection, FFT utilities, smooth weights as earlier (copy definitions)
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of log‑sum‑exp of the convolution."""
    n = a.shape[0]
    # reverse w for convolution with a
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)  # pick the slice giving gradient
    grad = 2.0 * full[idx]
    return grad

def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:

    scaled = alpha * conv
    m = np.max(scaled)
    exp_shifted = np.exp(scaled - m)
    return exp_shifted / exp_shifted.sum()

def _monotonize_desc(a: np.ndarray) -> np.ndarray:
    """Return a copy sorted in decreasing order (preserves sum)."""
    return np.sort(a)[::-1]

    S = np.sqrt(2.0 * n)

        a = np.clip(a, 0.0, None)

        mode = rng.integers(0, 4)
        if mode == 0:

        total = a.sum()
    a = a / total * S

    a = _monotonize_desc(a)
    return a

    # Adam hyper‑parameters
    lr = 0.3
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    v = np.zeros_like(a)

        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a, S)
        a = _monotonize_desc(a)

        cur = np.max(conv)
        if cur < best_score:

            best_a = a.copy()

        alpha = min(alpha * alpha_mul, 30.0)

            j = rng.integers(0, n)

                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = _monotonize_desc(cand)
                cand_max = np.max(_full_convolution(cand))

        a = _project_simplex(a, S)
        a = _monotonize_desc(a)

            best_score = cur
            best_a = a.copy()

Search for a step‑function that minimises ``evaluate_sequence``.

The optimisation proceeds in two phases:

* **Smooth phase** – minimise a log‑sum‑exp approximation of the
  max‑norm using the Adam optimiser.  The gradient of the smooth
  objective is obtained by a weighted convolution (FFT).  After each
  update the vector is projected onto the simplex `{x≥0, Σx=√(2·n)}`
  and then sorted in decreasing order (monotone projection).  The
  temperature of the soft‑max is increased gradually.

* **Sub‑gradient phase** – directly reduces the currently maximal
  convolution entry.  The sub‑gradient is the reversed copy of the
  vector aligned with the worst shift.  The same simplex‑plus‑monotone
  projection is applied.

The outer loop repeatedly restarts with diverse initial shapes
(linear, exponential, power‑law, random Dirichlet) and with

it becomes the new global best.  The whole procedure runs for the

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp of the convolution.
    g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad
