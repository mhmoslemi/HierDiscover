"""
Improved multiscale optimizer for the L∞‑norm of a non‑negative step function.
The algorithm works in three phases:

1.  If a previous construction is available (global ``height_sequence_1``)
    it is polished with a dedicated optimiser.
2.  A multiscale schedule enlarges the sequence gradually:
    – start from a moderate size,
    – repeatedly interpolate the current best shape to a larger length,
    – run a local optimiser (additive + multiplicative updates, simulated‑annealing
      moves) for a short budget.
    This helps the optimiser escape local minima that are hard to leave at a
    single scale.
3.  The remaining time is spent on many independent random restarts with
    varied lengths.  Each restart uses the same hybrid optimiser.

All updates are projected onto the simplex
    {x ≥ 0, ∑x = √(2·n)}.
The internal objective is simply ``max(a*a)`` because the external
``evaluate_sequence`` returns 2·n·max_b/∑a² = max_b after the normalisation.
The code uses FFT‑based convolutions (≈O(n log n)) and works with only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: nearest power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the simplex with total mass S."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y using a real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Linear scaling / interpolation of a sequence to a new length
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Sub‑gradient of the ∞‑norm of a*a (exact)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    """
    Compute a sub‑gradient of max(a*a) and the maximal convolution entry.
    ``eps`` allows a tolerance for the active set (entries within max‑eps are active).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)
    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)               # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]                  # slice of length n
    return g, max_b


# ----------------------------------------------------------------------
# Heuristic local optimiser (one restart)
# ----------------------------------------------------------------------
def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hybrid optimiser mixing additive/sub‑gradient steps, multiplicative
    (exponential‑weights) updates, simulated‑annealing swaps and jitter.
    Returns the best vector seen and its maximal self‑convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # bookkeeping
    cur_max = _max_convolution(a)
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters (tuned empirically)
    lr = 0.6                     # base step size
    lr_decay = 0.9995
    temperature = cur_max * 0.12
    temp_decay = 0.9996
    swap_prob = 0.07
    jitter_prob = 0.02
    jitter_scale = 0.02
    eps_fraction = 1e-3          # active‑set tolerance as fraction of cur_max

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) compute sub‑gradient of the current max entry
        # --------------------------------------------------------------
        eps = cur_max * eps_fraction
        g, cur_max = _subgradient_max(a, eps=eps)

        # --------------------------------------------------------------
        # 2) pick an update type
        # --------------------------------------------------------------
        r = rng.random()

        if r < 0.55:
            # ---- additive (projected gradient) step ----
            a_cand = a - lr * g
            a_cand = _project_to_simplex(a_cand, S)

        elif r < 0.85:
            # ---- multiplicative (exponential‑weights) step ----
            factor = np.exp(-lr * g)            # element‑wise
            a_cand = a * factor
            total = a_cand.sum()
            if total > 0:
                a_cand = a_cand * (S / total)
            else:
                a_cand = a.copy()

        elif r < 0.95:
            # ---- random mass swap (simulated annealing) ----
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = 0.03 * a[i]          # move a small fraction
                    a_cand = a.copy()
                    a_cand[i] -= delta
                    a_cand[j] += delta
                    a_cand = _project_to_simplex(a_cand, S)
                else:
                    a_cand = a.copy()
            else:
                a_cand = a.copy()

        else:
            # ---- Gaussian jitter ----
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a_cand = np.clip(a + noise, 0.0, None)
            a_cand = _project_to_simplex(a_cand, S)

        # --------------------------------------------------------------
        # 3) acceptance (simulated annealing)
        # --------------------------------------------------------------
        max_cand = _max_convolution(a_cand)
        accept = False
        if max_cand < cur_max:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            if rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                accept = True

        if accept:
            a = a_cand
            cur_max = max_cand
            temperature *= temp_decay

        # --------------------------------------------------------------
        # 4) bookkeeping
        # --------------------------------------------------------------
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # decay of the learning rate
        lr *= lr_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# Initial feasible vector (ℓ₁‑norm = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with sum √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size > 0:
        a = _scale_sequence(base_seq, n).astype(float)
        a += rng.normal(scale=max(a.mean(), 1e-12) * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                     # Dirichlet(1) – uniform on simplex before projection
    a = _project_to_simplex(a, S)
    return a

    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously stored construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – polish the existing construction (if available)
    # ------------------------------------------------------------------
    if base_seq is not None and base_seq.size > 0:
        n0 = int(base_seq.shape[0])
        a0 = _project_to_simplex(base_seq, np.sqrt(2.0 * n0))
        phase_deadline = min(time.time() + 180.0, end_time)   # up to 3 min
        seq_opt, score_opt = _local_optimize(a0, phase_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[phase 1] refined base length={n0} max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multiscale refinement (gradually increase the length)
    # ------------------------------------------------------------------
    # Determine a starting point for the scaling chain
    if best_seq is None:
        cur_n = 256
        cur_seq = _initial_sequence(None, cur_n, rng)
        # quick polishing of the small start
        tmp_dead = min(time.time() + 30.0, end_time)
        cur_seq, cur_max = _local_optimize(cur_seq, tmp_dead, rng)
        best_seq, best_score = cur_seq.copy(), cur_max
        print(f"[scale‑init] start length={cur_n} max={best_score:.6f}")
    else:
        cur_seq = best_seq.copy()
        cur_n = cur_seq.shape[0]

    max_n = 20000                         # upper bound for the multiscale ladder
    scaling_factors = [1.5, 1.7, 2.0]     # how fast we grow the size
    while True:
        if time.time() >= end_time:
            break
        # propose next size
        next_n = int(min(max_n, cur_n * scaling_factors[0]))
        if next_n <= cur_n:
            break
        # initialise at new size by interpolation
        a0 = _scale_sequence(cur_seq, next_n)
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * next_n))
        # allocate a modest slice of the remaining budget
        remaining = end_time - time.time()
        slice_deadline = time.time() + min(30.0, max(5.0, remaining * 0.15))
        seq_opt, score_opt = _local_optimize(a0, slice_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[scale] up to n={next_n} new best max={best_score:.6f}")
        # prepare for possible further scaling
        cur_seq = best_seq.copy()
        cur_n = next_n
        # rotate scaling factors to avoid too aggressive jumps
        scaling_factors = scaling_factors[1:] + scaling_factors[:1]

    # ------------------------------------------------------------------
    # Phase 3 – many independent random restarts with varied lengths
    # ------------------------------------------------------------------
    attempt = 0
    while time.time() < end_time:
        attempt += 1
        # pick a length – a mix of moderate and larger sizes
        if rng.random() < 0.6:
            n = rng.integers(800, 6000)          # moderate
        else:
            n = rng.integers(6000, 25000)        # larger (still fast enough)

        a0 = _initial_sequence(best_seq, n, rng)
        # allocate a short slice (≈8 s) but respect the global deadline
        slice_deadline = min(time.time() + 8.0, end_time)
        seq_opt, score_opt = _local_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[restart] attempt={attempt} n={n} new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the black‑box function
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                     # fallback
    else:
        final_seq = best_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] external score = {external_score:.6f} "
              f"(internal max = {best_score:.6f}) length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
