"""
Propose a non‑negative step‑function that minimises the maximum entry of its
self‑convolution (the value returned by ``evaluate_sequence``).

Algorithm
---------

1. **Diverse random starts** – power‑law, exponential, Dirichlet and
   edge‑heavy families are sampled on the ℓ¹‑simplex
   Σaᵢ = √(2·n).

2. **Adam‑style projected gradient descent** on a log‑sum‑exp (soft‑max)
   surrogate of the maximal convolution entry.  The smoothing parameter β is
   annealed from 1 to 1e⁶ while Adam adapts the step sizes.  After each step
   the iterate is projected back onto the simplex.

3. **Linear‑programming refinement** – with the current maximum M = max(a∗a)
   we solve a small LP:

        maximise   Σ bᵢ
        subject to  a∗b ≤ M ,   b ≥ 0

   (scipy.optimize.linprog, “highs”).  The solution b is rescaled to the
   required ℓ¹‑norm and mixed with the current iterate (τ≈0.1) to obtain a
   strictly better point.

4. **Simulated‑annealing polishing** – random mass transfers between two
   indices are performed; moves that lower the maximal convolution are always
   accepted, uphill moves are accepted with a temperature schedule.
   The best sequence seen is kept.

All phases respect the global deadline (default 1000 s).  The best sequence
found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum of the convolution (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable
    p = exps / np.sum(exps)

    # gradient = 2·(a_rev ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]         # central n entries
    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smooth surrogate
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(a*a).
    Operates entirely on the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    beta = 1.0
    max_beta = 1e6

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        it += 1

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base learning rate scaled to the simplex size
        base_lr = 0.4 * target_sum / (np.linalg.norm(g) + 1e-12)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # anneal β slowly
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

    return a

def _lp_refine(a: np.ndarray, target_sum: float):
    """
    Solve   maximise Σ bᵢ   subject to   a∗b ≤ M  and b ≥ 0,
    where M = max(a∗a).  The solution is scaled to have ℓ¹‑norm = target_sum.
    Returns the scaled vector or ``None`` on failure.
    """
    if not _has_scipy:
        return None

    n = a.size
    M = _max_convolution_fft(a)

    # ----- build sparse constraint matrix (2n‑1) × n  -----------------
    # rows for entry k = i + j (i : index of a, j : index of b)
    i = np.arange(n)[:, None]          # (n,1)
    j = np.arange(n)[None, :]          # (1,n)
    rows = (i + j).ravel()             # length n², values 0 … 2n‑2
    cols = np.repeat(np.arange(n), n)  # column index of b
    data = np.repeat(a, n)             # constant a_i on each diagonal

    A_ub = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    # ----- linear programme (maximise Σb) --------------------------------
    c = -np.ones(n)                     # minimise –Σb   ≡ maximise Σb
    bounds = [(0, None)] * n

    try:
        res = _opt.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
        )
    except Exception:  # pragma: no cover
        return None

    if not res.success:
        return None

    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return None

    scale = target_sum / sum_b
    g = b_opt * scale
    return g


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Random mass‑transfer SA.  The current best (lowest max‑convolution) is
    remembered; swaps that improve it are always accepted,
    uphill moves are accepted with probability exp(-Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature from current objective
    T = best_score * 0.02 + 1e-12
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick donor i (must have some mass) and recipient j ≠ i
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 15 %) from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional small global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score, best = cur_score, a.copy()

    return best


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families are mixed to increase diversity,
    including a symmetric edge‑heavy shape.
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    elif kind == 4:                 # symmetric edge‑heavy (high at both ends)
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                           # uniform plus tiny jitter
        a = np.full(n, target_sum / n)

    # small Gaussian perturbation + re‑projection
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    competition objective.  The routine respects the supplied time budget
    (seconds) and returns the best sequence it found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5          # tiny safety margin
    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # If a global construction is present, start from it
    # ------------------------------------------------------------------
    global height_sequence_1
    if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
        a0 = np.asarray(height_sequence_1, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_obj = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Starting from global height_sequence_1 "
              f"(n={n0})  max_conv={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore (moderate values keep FFT cheap)
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1500, 1800, 2100, 2500]

    # keep a tiny pool of promising seeds for deeper work
    elite_per_len = 4

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Phase A – quick random sampling
        # --------------------------------------------------------------
        elite = []   # list of (score, vector)
        phase_end = min(deadline, time.time() + 6.0)  # ~6 s per length
        while time.time() < phase_end:
            a_rand = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a_rand)
            if score < best_obj:
                best_obj, best_seq = score, a_rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n}  max_conv={best_obj:.6f}")

            # maintain elite pool
            elite.append((score, a_rand))
            elite.sort(key=lambda x: x[0])
            if len(elite) > elite_per_len:
                elite.pop()   # discard worst

        # --------------------------------------------------------------
        # Phase B – deterministic refinement of elite seeds
        # --------------------------------------------------------------
        for _, seed_vec in elite:
            if time.time() >= deadline:
                break

            # allocate a slice of the remaining budget for this candidate
            sub_dead = min(deadline, time.time() + 30.0)   # 30 s max

            # ---- Adam projected gradient descent -----------------------
            a_gd = _adam_opt(seed_vec, sub_dead)

            # ---- LP refinement (if available and time permits) ---------
            if _has_scipy and time.time() < deadline - 5.0:
                g = _lp_refine(a_gd, target_sum)
                if g is not None:
                    # gentle mixing; τ small to keep the bound safe
                    tau = 0.12
                    a_mix = (1.0 - tau) * a_gd + tau * g
                    a_mix = _project_to_simplex(a_mix, target_sum)
                    a_gd = a_mix

            # ---- Simulated‑annealing polishing -------------------------
            a_final = _simulated_annealing(a_gd, deadline)

            final_score = _max_convolution_fft(a_final)
            if final_score < best_obj:
                best_obj, best_seq = final_score, a_final.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n}  max_conv={best_obj:.6f}")

        # clear elite before next length
        elite.clear()

    # ------------------------------------------------------------------
    # Fallback – uniform (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zero entries for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional final sanity check with the official evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {val:.6f}")
    except Exception:  # the grader will call evaluate_sequence itself
        pass

    print(f"[{time.time():.1f}] Finished. Best max‑convolution = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
