def _adam_descent(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    size_fft = 1 << ((2 * n - 1).bit_length())
    # hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # step size base: maybe 0.5 * target_sum
    lr0 = target_sum * 0.05
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    # compute initial max conv
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
    max_b = conv.max()
    best_score = float(2 * n * max_b / (target_sum ** 2))
    # store best
    while time.time() < deadline:
        # compute conv and max index each iteration
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))
        # compute gradient (subgradient of max conv)
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[max_idx - idx]

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # Learning rate schedule decaying with sqrt(t)
        lr = lr0 / (np.sqrt(t) + 1.0)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # evaluate score (max conv)
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b = conv.max()
        cur_score = float(2 * n * max_b / (target_sum ** 2))

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # printing periodic improvements
            if t % 500 == 0:
                print(f"[{time.time():.1f}] Adam iter {t}, new best {best_score:.6f}")

        # occasional jitter for escape
        if t % 4000 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum*0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

"""
Propose a high‑quality step‑function.

The algorithm works in three phases:

1. **Coarse power‑law sweep** – for a few candidate lengths a short grid
   of exponents  p∈[0.4,1.0] is scanned and the best power‑law (a_i∝(i+1)⁻ᵖ)
   is kept as a deterministic start.

2. **Projected Adam sub‑gradient descent** – using the exact sub‑gradient
   of the current maximal convolution entry (the true objective), the
   vector is updated with Adam’s adaptive moment estimator and projected
   onto the ℓ¹‑simplex Σ a_i = √(2·n).  The step‑size decays with 1/√t
   and occasional small Gaussian jitter is added to escape flat regions.
   The best sequence seen during the run is retained.

3. **Final simulated‑annealing refinement** – a short SA pass (the one used
   in the reference implementation) is applied to the best sequence from
   phase 2 to fine‑tune it.

All moves respect non‑negativity and the ℓ¹‑norm constraint, and the
routine stops automatically when the time budget is exhausted.
"""

import time
import numpy as np


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a computed by FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (FFT version)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s} (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and scale to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _adam_descent(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the true objective (max of a*a).
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    n = a.size
    size_fft = 1 << ((2 * n - 1).bit_length())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = target_sum * 0.05  # base learning‑rate (≈ 2–3 for n≈1500)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # initial score (the objective equals max_b because Σa = √(2n))
    conv = _conv_fft(a)
    best_score = float(2.0 * n * conv.max() / (target_sum ** 2))
    best_a = a.copy()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # compute convolution and locate the maximal entry
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active max entry (g_i = 2·a_{k_max-i} where defined)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / (np.sqrt(t) + 1.0)          # simple decay schedule
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # evaluate current objective
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        cur_score = float(2.0 * n * conv.max() / (target_sum ** 2))

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if t % 500 == 0:
                print(f"[{time.time():.1f}] Adam iter {t}, new best {best_score:.6f}")

        # occasional tiny jitter to help escape plateaus
        if t % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 20) -> tuple[np.ndarray, float]:
    """
    Classic SA from the reference code – kept unchanged.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = 2.0 * n * _max_convolution_fft(a) / (cur_sum ** 2)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial convolution (kept for the first iteration)
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity for the active entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        # propose a move
        if rng.random() < 0.70:                     # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                                        # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # evaluate new state (single FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new / (cur_sum ** 2)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved_seg
                # conv unchanged

        # occasional global jitter to avoid deep traps
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = 2.0 * n * conv.max() / (cur_sum ** 2)

        it += 1

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point required by the platform.
    It explores a few problem sizes, builds a good power‑law start,
    refines it with Adam sub‑gradient descent and finishes with a short
    simulated‑annealing pass.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the external checker
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # 1) candidate problem sizes (lengths of the step‑function)
    # ------------------------------------------------------------------
    length_pool = [1024, 1500, 2048, 3000]          # a few diverse values
    best_seq = None
    best_score = np.inf

    # cache the best power‑law for each length – reused across runs
    cached_powerlaw: dict[int, tuple[np.ndarray, float]] = {}

    # ------------------------------------------------------------------
    # 2) exploration loop – run Adam on many random starts until ~80 % of time
    # ------------------------------------------------------------------
    exploration_deadline = deadline - 15.0           # reserve ~15 s for final SA
    while time.time() < exploration_deadline:
        n = int(rng.choice(length_pool))
        target = np.sqrt(2.0 * n)

        # --- obtain a deterministic power‑law start (cached) -----------------
        if n not in cached_powerlaw:
            # cheap grid search over exponents
            p_grid = np.linspace(0.40, 1.00, 31)   # step ≈0.02
            best_p_seq = None
            best_p_score = np.inf
            for p in p_grid:
                a_tmp = _power_law_sequence(n, p, target)
                score = float(2.0 * n * _max_convolution_fft(a_tmp) / (target ** 2))
                if score < best_p_score:
                    best_p_score = score
                    best_p_seq = a_tmp
            cached_powerlaw[n] = (best_p_seq, best_p_score)
        a0, _ = cached_powerlaw[n]

        # --- occasional random Dirichlet perturbation ------------------------
        if rng.random() < 0.4:
            raw = rng.exponential(scale=1.0, size=n)
            a0 = _project_to_simplex(raw, target)

        # --- run Adam for a modest slice of time ---------------------------
        remaining = exploration_deadline - time.time()
        if remaining < 5.0:
            break
        sub_deadline = time.time() + min(25.0, remaining / 2.0)
        a_opt, score_opt = _adam_descent(a0, target, sub_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] New global best: score={best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3) final refinement with simulated annealing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        sa_seq, sa_score = _simulated_annealing(best_seq, deadline)
        if sa_score < best_score:
            best_seq = sa_seq
            best_score = sa_score
            print(f"[{time.time():.1f}] After SA improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) fallback – very unlikely to be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target = np.sqrt(2.0 * n)
        best_seq = _power_law_sequence(n, 0.55, target)
        best_seq = _project_to_simplex(best_seq, target)

    # ------------------------------------------------------------------
    # 5) trim negligible trailing zeros for a tidy answer
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
