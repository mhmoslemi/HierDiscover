import cvxpy as cp

def lp_optimize(f, M):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    # compute convolution constraints:
    # For each k, sum_i f[i] * g[k - i] <= M
    # We can pre-compute using conv: slice g with appropriate zero padding.
    # Use cp.sum(cp.multiply(f, cp.hstack(...))) - but loops easiest.
    for k in range(2*n-1):
        # compute indices i where 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # corresponding g indices
        g_indices = [k - i for i in range(i_start, i_end+1)]
        # Build linear expression sum f[i] * g_{k-i}
        expr = cp.sum(cp.multiply(f[i_start:i_end+1], g[g_indices]))
        constraints.append(expr <= M)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True) # maybe faster.
    g_opt = np.array(g.value).reshape(-1)
    return g_opt
```

    n = len(f)

    # Row k corresponds to convolution index k:
    # For each column j, we need coefficient f[i] where i = k - j.

def _lp_optimize(f, M):
    n = len(f)

    for i in range(n):
        A[i:i+n, i] = f

    b = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    return res.x

"""
A fast multi‑phase optimizer for the step‑function challenge.

1.  **Parametric sweep** – generate many diverse shapes (power‑law,
    exponential edge, beta‑distribution, cosine windows, Dirichlet,
    etc.) for a given length `n`.  Each candidate is normalized to unit
    ℓ¹‑norm; the objective is `obj = 2·n·max(conv(a,a))`, which is exactly
    the value returned by ``evaluate_sequence`` after scaling to
    `sum(a)=√(2n)`.  The best few candidates are kept.

2.  **Adam‑based projected gradient descent** – starting from the elite
    candidates we minimise a smooth log‑sum‑exp surrogate of the
    convolution peak.  The surrogate is annealed (β grows) and the
    update is projected onto the ℓ¹‑simplex (Duchi et al. 2008).

3.  **Local simulated‑annealing** – a cheap SA that repeatedly moves a
    random fraction of mass between two indices while preserving the
    simplex constraints.  The temperature decays geometrically; accepted
    moves that improve the peak are kept.

The whole procedure respects a user‑supplied time budget (default
1000 s) and works for any length up to ≈10 000 without exceeding the
single‑CPU limit.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional – not used in the main loop


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution a*a using FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex (sum = s, a_i ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the self‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
      = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ------- b = a*a -------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ------- soft‑max probabilities -----------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentials
    p = exps / np.sum(exps)

    # ------- gradient g = 2·(a ⋆ p) -----------------------------------
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]       # central n values
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑style projected optimisation of the log‑sum‑exp surrogate.
    The vector is kept on the ℓ¹‑simplex Σa = 1 (the scaling is irrelevant).
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    beta = 1.0                # smoothing parameter for the surrogate
    max_beta = 1e5
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # global step size (tuned for stability)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # project back to the unit simplex
        a = _project_to_simplex(a, 1.0)

        # anneal the smoothness parameter
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        # occasional jitter to jump out of plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, 1.0)

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Local SA that moves a random fraction of mass between two indices.
    The sum and non‑negativity are preserved by construction.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature proportional to current peak
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move up to 15 % of the mass at i to j
        fraction = rng.random() * 0.15
        delta = a[i] * fraction
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        if cur_score <= best_score or rng.random() < np.exp(-(cur_score - best_score) / T):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # reject – revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very rare global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, 1.0)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Candidate generators (families of shapes)
# ----------------------------------------------------------------------


def _candidate_shapes(n: int, rng) -> list[np.ndarray]:
    """
    Produce a diverse list of unit‑sum sequences for length `n`.
    The list is intentionally modest – it is only used for an initial
    screening before the more expensive optimiser is applied.
    """
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws with several concentration parameters
    for alpha in (0.5, 1.0, 2.0, 5.0):
        a = rng.dirichlet(alpha=np.full(n, alpha))
        candidates.append(a)

    # symmetric edge‑heavy power‑law
    for p in np.arange(0.1, 2.1, 0.2):
        idx = np.arange(n, dtype=float)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right
        a /= a.sum()
        candidates.append(a)

    # exponential decay from both ends
    for lam in np.arange(0.001, 0.031, 0.004):
        idx = np.arange(n, dtype=float)
        dist = np.minimum(idx, n - 1 - idx)
        a = np.exp(-lam * dist)
        a /= a.sum()
        candidates.append(a)

    # beta‑distribution shapes (including symmetric and skewed)
    for a_param in (0.5, 1.0, 2.0, 3.0, 5.0):
        for b_param in (0.5, 1.0, 2.0, 3.0, 5.0):
            x = np.linspace(0.0, 1.0, n)
            a = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1) + 1e-12)
            a /= a.sum()
            candidates.append(a)

    # cosine window raised to a power
    for p in (0.5, 1.0, 2.0, 3.0):
        idx = np.arange(n, dtype=float)
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n - 1))) ** p
        a /= a.sum()
        candidates.append(a)

    # a few jittered uniforms (to give random low‑amplitude noise)
    for _ in range(5):
        a = np.full(n, 1.0 / n)
        a += rng.normal(scale=0.01, size=n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidates.append(a)

    return candidates

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the time budget,
    explores several vector lengths, performs a cheap parametric sweep,
    refines the best few candidates with Adam optimisation and a short
    simulated‑annealing phase, and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # If a global construction exists, start from it.
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        # Normalise to unit sum – scaling does not affect the objective.
        a0 = a0 / a0.sum()
        val0 = evaluate_sequence((a0 * np.sqrt(2.0 * a0.size)).tolist())
        best_seq = a0.copy()
        best_val = val0
        print(f"[{time.time():.1f}] Global seed used, init obj={val0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to try – covering a wide range up to ≈10 000.
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 10000]

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"[{time.time():.1f}] ==== Exploring length n={n} ====")

        # how many seconds to spend on this length (roughly equal split)
        remaining = deadline - time.time()
        length_budget = remaining / (len(lengths) - lengths.index(n) + 1e-12)
        length_deadline = time.time() + length_budget

        # --------------------------------------------------------------
        # Phase 1 – cheap screening of many parametrised candidates.
        # --------------------------------------------------------------
        cand_pool = _candidate_shapes(n, rng)
        scores = []
        for a in cand_pool:
            max_b = _max_convolution_fft(a)      # a already sums to 1
            obj = 2.0 * n * max_b                # equals evaluate_sequence after scaling
            scores.append((obj, a))
        # keep the 6 best candidates for deeper refinement
        scores.sort(key=lambda x: x[0])
        elite = scores[:6]

        # --------------------------------------------------------------
        # Phase 2 – Adam optimisation + brief SA on the elite candidates.
        # --------------------------------------------------------------
        for elite_obj, elite_a in elite:
            if time.time() >= length_deadline:
                break

            # allocate a slice of the remaining time for this candidate
            sub_deadline = min(length_deadline, time.time() + 30.0)

            # Adam optimisation on the smoothed surrogate
            a_opt = _adam_optimize(elite_a, sub_deadline)

            # short SA refinement (a few seconds, if time permits)
            if time.time() + 5.0 < sub_deadline:
                a_opt = _simulated_annealing(a_opt, time.time() + 5.0)

            # evaluate the refined candidate (still unit‑sum)
            max_b = _max_convolution_fft(a_opt)
            obj = 2.0 * n * max_b
            if obj < best_val:
                best_val = obj
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST: n={n}  obj={best_val:.6f}")

        # ------------------------------------------------------------------
        # End of this length – forget temporary data.
        # ------------------------------------------------------------------
        # (nothing to do; variables are overwritten in the next iteration)

    # ------------------------------------------------------------------
    # Prepare final sequence: scale to the required ℓ¹‑norm √(2·n).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple uniform vector (should never happen)
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    n_best = best_seq.size
    scale = np.sqrt(2.0 * n_best) / best_seq.sum()
    final_seq = (best_seq * scale).tolist()

    # Final sanity check (optional – cheap)
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[{time.time():.1f}] Final evaluate_sequence = {final_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] SEARCH FINISHED – best objective = {best_val:.6f}, length = {n_best}")
    return final_seq
