def generate_sidon_set(n, target_m=None):
    sums = set()
    positions = []
    for candidate in range(n):
        # Check if adding candidate violates distinctness with existing positions
        conflict = False
        # compute sums with each existing pos
        for p in positions:
            if candidate + p in sums:
                conflict = True
                break
        if conflict:
            continue
        # also check double sum candidate+candidate
        if candidate + candidate in sums:
            continue
        # accept
        positions.append(candidate)
        # add new sums
        sums.add(candidate + candidate)
        for p in positions[:-1]:
            sums.add(candidate + p)
        if target_m and len(positions) >= target_m:
            break
    return positions

def lp_update(a, M, target_sum):
    n = a.shape[0]
    rows = 2*n - 1
    # Build A_ub as shape (rows, n)
    # We can build using loops:
    A_ub = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices j from lo to hi inclusive: coefficient a_{k-j}
        # we can set A_ub[k, lo:hi+1] = a[k - np.arange(lo, hi+1)]
        idx = np.arange(lo, hi + 1)
        A_ub[k, lo:hi+1] = a[k - idx]
    # objective: maximize sum(g)
    c = -np.ones(n)  # minimize -sum(g)
    bounds = [(0, None)] * n
    # Solve:
    res = linprog(c, A_ub=A_ub, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    # Scale to target sum:
    sum_g = g.sum()
    if sum_g < 1e-12:
        # fallback: return original a
        return a
    g_scaled = g * (target_sum / sum_g)
    return g_scaled

"""
Propose a candidate step‑function using a two‑phase optimizer:

Phase 1 – *LP mass redistribution*: given a current sequence a,
solve the linear program
    maximize   Σ g_j
    subject to a * g ≤ M   (pointwise, where M = max(a*a))
               g ≥ 0 .
The solution pushes mass onto low‑sensitivity positions.
Mix the new vector g (scaled to the required ℓ¹‑norm) with a.
Repeat while time permits.

Phase 2 – *Simulated annealing* (as in the reference code) to clean up
any remaining peaks.  The whole routine respects non‑negativity,
clips to the simplex Σa = √(2 n), and returns the best sequence found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# optional: SciPy is used for the linear program.
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:        # pragma: no cover
    _SCIPY_AVAILABLE = False


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact evaluation of the competition objective."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:      # safeguard (should never happen)
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) sequence of length n, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _lp_optimize(a: np.ndarray, M: float, target_sum: float):
    """
    Solve   max Σ g_j  s.t.   (a * g)_k ≤ M  (for all k), g ≥ 0.
    Return g scaled to have ℓ¹‑norm = target_sum,
    or None if the LP fails.
    """
    if not _SCIPY_AVAILABLE:          # pragma: no cover
        return None
    n = a.shape[0]
    rows = 2 * n - 1
    # Build the constraint matrix A_ub such that A_ub @ g ≤ M
    A_ub = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        A_ub[k, lo:hi + 1] = a[k - idx]      # a_{k-j}
    c = -np.ones(n, dtype=np.float64)         # minimise -Σg   ⇒ maximise Σg
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    total = g.sum()
    if total <= 0.0:
        return None
    g_scaled = g * (target_sum / total)
    return g_scaled


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 20) -> tuple[np.ndarray, float]:
    """
    Classic SA refinement (identical to the reference implementation).
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = 2.0 * n * _max_convolution_fft(a) / (cur_sum ** 2)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity for the active convolution entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # ---- propose a move ----
        if rng.random() < 0.70:                 # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.integers(n) if low_idx.size == 0 else rng.choice(low_idx)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                                    # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # evaluate new state (single FFT)
        fa = rfft(a, n=size_fft)
        conv_new = irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new / (cur_sum ** 2)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved_seg

        # occasional jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = 2.0 * n * conv.max() / (cur_sum ** 2)

        it += 1

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point.  Builds a length‑1500 sequence, repeatedly
    applies the LP‑based redistribution, then finishes with simulated
    annealing.  Returns the best sequence discovered before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a safety margin

    # --------------------------------------------------------------
    # Initialise a vector (use the globally supplied best if present)
    # --------------------------------------------------------------
    n = 1500                                  # fixed length used by the benchmark
    target = np.sqrt(2.0 * n)

    if 'height_sequence_1' in globals():
        seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        if seq.size != n:
            # truncate or pad with zeros to match n
            if seq.size > n:
                seq = seq[:n]
            else:
                seq = np.pad(seq, (0, n - seq.size), constant_values=0.0)
        if seq.sum() > 0:
            seq = seq * (target / seq.sum())
        else:
            seq = _power_law_sequence(n, 0.55, target)
        a = _project_to_simplex(seq, target)
        print(f"[{time.time():.1f}] start from global seq, score={_score(a):.6f}")
    else:
        a = _power_law_sequence(n, 0.55, target)
        a = _project_to_simplex(a, target)
        print(f"[{time.time():.1f}] start from power‑law, score={_score(a):.6f}")

    best_a = a.copy()
    best_score = _score(best_a)

    # --------------------------------------------------------------
    # Phase 1 – LP‑based mass redistribution
    # --------------------------------------------------------------
    lp_iterations = 0
    while time.time() < deadline - 10.0:   # keep a few seconds for the final SA
        cur_M = _max_convolution_fft(a)
        g = _lp_optimize(a, cur_M, target)
        if g is None:
            # LP failed (unlikely); break out to SA
            print(f"[{time.time():.1f}] LP step failed, exiting LP phase")
            break

        # mixing coefficient – start aggressive, then taper off
        elapsed = time.time() - start_time
        total = deadline - start_time
        mix = 0.5 * (1.0 - elapsed / total) + 0.05   # between ~0.55 and 0.05
        a = (1.0 - mix) * a + mix * g
        a = _project_to_simplex(a, target)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {lp_iterations}: improved score={best_score:.6f}")

        lp_iterations += 1

    # --------------------------------------------------------------
    # Phase 2 – Simulated annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        sa_a, sa_score = _simulated_annealing(best_a, deadline)
        if sa_score < best_score:
            best_score = sa_score
            best_a = sa_a
            print(f"[{time.time():.1f}] SA final improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing negligible entries and return
    # --------------------------------------------------------------
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_a)}")
    return best_a.tolist()
