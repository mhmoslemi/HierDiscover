```python
def _lp_refine_once(a, t):
    n = a.shape[0]
    M = np.max(_full_convolution_fft(a))
    offsets = np.arange(n)  # 0..n-1
    # Build sparse matrix A
    A = sp.diags([a] * n, offsets, shape=(2*n-1, n), format='csr')
    b = np.full(2*n-1, M, dtype=float)
    # objective - maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # Using high accuracy might be heavy; set options to speed if needed.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return a, M
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, M
    S_target = np.sqrt(2.0 * n)
    g = g * (S_target / sum_g)
    a_new = (1 - t) * a + t * g
    # Project again (to counter rounding errors)
    a_new = _project_to_simplex(a_new, S_target)
    new_max = np.max(_full_convolution_fft(a_new))
    return a_new, new_max
```

def _hardmax_subgradient_opt(a, deadline, rng):
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    lr = 0.1
    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k = np.argmax(b)
        # compute gradient: for each i, g_i = 2*a_{k-i} if valid, else 0
        grad = np.zeros_like(a)
        # compute partner indices vectorized:
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        grad[mask] = 2.0 * a[partner[mask]]
        # also diagonal? Already included when i=partner? Should handle.
        # gradient step
        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, S)
        # maybe occasional jitter
        if rng.random() < 0.01:
            noise = rng.normal(scale=a.mean()*0.01, size=n)
            a_new = np.clip(a_new + noise, 0, None)
            a_new = _project_to_simplex(a_new, S)
        # evaluate
        new_max = np.max(_full_convolution_fft(a_new))
        if new_max < np.max(_full_convolution_fft(a)):
            a = a_new
            # reduce lr gradually

"""
Improved search for a low‑max self‑convolution step‑function.

The algorithm consists of four stages within the 1000 s budget:

1️⃣  Random parametric families (including a new “sparse” family) generate many
    candidates of varying length.  Each candidate is projected onto the ℓ₁‑simplex
    ∑a = √(2·n) so that the external evaluation equals max(a∗a).

2️⃣  For the current best candidate we run a few *linear‑programming refinements*
    (the “LP‑step” of the classic iterative scheme).  The LP maximises the
    ℓ₁‑norm of a partner vector g under the constraint (a∗g) ≤ M, where
    M = max(a∗a).  Mixing a and the normalised g with a factor t (t≈0.5)
    strictly reduces the max‑convolution entry.

3️⃣  A projected Adam optimiser minimises a smooth proxy of the ℓ∞‑norm –
    τ·log∑exp(b/τ) with b = a∗a.  The temperature τ is annealed to the
    hard‑max regime while each step is projected back onto the simplex.

4️⃣  A final short hard‑max sub‑gradient descent finishes the polishing.

All stages obey the time budget and the best sequence found is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the ℓ₁‑simplex of sum S."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Compute gradient of τ·log∑exp(b/τ) where b = a*a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                                   # hard max → sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        return g, b, max_b

    # soft‑max
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)                           # shape (2n‑1,)

    # convolution a_rev ⋆ w  (cross‑correlation)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]                   # length n

    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """Minimise the soft‑max proxy of max(a*a) using Adam."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.7               # a bit larger than before
    t = 0

    # initialise temperature
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.6 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluate
        cur_max_new = np.max(_full_convolution_fft(a_new))
        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.998, 1e-8)
        lr *= 0.9999

        # occasional jitter to escape plateaus
        if rng.random() < 0.005:
            noise = rng.normal(scale=a.mean() * 0.02, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families (including a new sparse family)
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create an unnormalised non‑negative vector using a random family."""
    families = [
        "uniform", "exp", "symexp", "gaussian",
        "triangular", "power", "cosine", "sparse"
    ]
    typ = rng.choice(families)
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "sparse":
        # choose a small number of non‑zero entries
        max_k = max(2, n // 10)
        k = rng.integers(2, max_k + 1)
        a = np.zeros(n)
        positions = rng.choice(n, size=k, replace=False)
        a[positions] = rng.random(k) + 0.1   # keep them away from zero
    else:
        a = rng.random(n)

    # optional jitter for diversity
    if rng.random() < 0.25:
        a += rng.normal(scale=a.mean() * 0.02, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# One LP refinement step (the classic “g‑step”)
# ----------------------------------------------------------------------
def _lp_refine_once(a: np.ndarray, t: float):
    """
    Perform a single linear‑programming refinement:
    maximise sum(g) subject to g≥0 and (a∗g) ≤ M,
    then mix the result back with a using factor t.
    """
    n = a.shape[0]
    M = np.max(_full_convolution_fft(a))

    # Build the convolution matrix A where (A @ g)_k = (a∗g)_k
    offsets = np.arange(n)                     # 0 … n‑1
    # each column i receives the same coefficient a[i] on a diagonal
    A = sp.diags([a] * n, offsets, shape=(2 * n - 1, n), format="csr")
    b = np.full(2 * n - 1, M, dtype=float)

    # linprog: minimise -sum(g)  -> maximise sum(g)
    c = -np.ones(n, dtype=float)
    res = opt.linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method="highs", options={"presolve": True})

    if not res.success:
        return a, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, M

    S_target = np.sqrt(2.0 * n)
    g = g * (S_target / sum_g)                # rescale to the simplex
    a_new = (1.0 - t) * a + t * g
    a_new = _project_to_simplex(a_new, S_target)
    new_max = np.max(_full_convolution_fft(a_new))
    return a_new, new_max
