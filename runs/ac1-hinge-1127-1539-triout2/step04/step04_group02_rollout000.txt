        a = best_seq.copy()
        # apply random perturbation

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1=0.9; beta2=0.999; eps=1e-8
        lr = 0.5 * (S/n)  # initial step size maybe.
        tau_factor = 0.2
        # schedule: tau reduces every few thousand steps.
        iteration = 0
        while time.time() < deadline and iteration < max_iters_per_run:
            conv = conv_self_fft(a)
            max_conv = conv.max()
            # compute smooth max gradient
            tau = max(tau_factor * max_conv, 1e-12)
            # compute weights
            w = np.exp((conv - max_conv)/tau)
            w_sum = w.sum()
            w /= w_sum
            # Compute gradient via convolution: grad = 2/tau * (w * a_rev) conv.
            # Use FFT convolution but careful.
            a_rev = a[::-1]
            L = 1 << ((len(w)+len(a_rev)-1).bit_length())
            W = np.fft.rfft(w, n=L)
            A_rev = np.fft.rfft(a_rev, n=L)
            conv_w_a_rev = np.fft.irfft(W * A_rev, n=L)[:len(w)+len(a_rev)-1]
            grad = (2.0 / tau) * conv_w_a_rev[n-1:n-1+n]
            # Adam update
            iteration += 1
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad**2)
            m_hat = m / (1-beta1**iteration)
            v_hat = v / (1-beta2**iteration)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            # enforce nonnegativity & simplex
            a = np.maximum(a, 0)
            a = project_onto_simplex(a, S)

            if max_conv < best_score - 1e-12:

"""
Hybrid optimizer for the step‑function self‑convolution bound.

The algorithm proceeds in several stages within the 1000 s budget:

1. **Initialisation** – try a variety of parametric shapes (U‑shapes,
   skewed exponentials, etc.) and a few Dirichlet‐sampled random
   vectors.  All candidates are projected onto the ℓ₁‑simplex
   Σa_i = √(2 n) (the objective equals max conv for such normalisation).

2. **Adam‑based smoothed descent** – minimise a log‑sum‑exp
   approximation of the max‑convolution.  The gradient is obtained by a
   single FFT‑based convolution.  Adam’s adaptive moments together with
   a geometric decay of the smoothing parameter τ provide fast
   convergence.  The iterate is re‑projected onto the simplex after each
   step.

3. **Sub‑gradient polishing** – switch to the true non‑smooth objective.
   The sub‑gradient is the average of gradients over all (approximately)
   maximal lags.  A diminishing step size is used.

4. **Simulated‑annealing polishing** – many tiny random pair‑mass
   transfers are accepted according to a cooling schedule, allowing the
   search to escape shallow local minima.

The best sequence discovered during the run is returned as a plain
Python list.  All helper routines are top‑level (no closures) and only
numpy, math and time are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero → give uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence.
    `kind` may be one of:
        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right, exponential_left, exponential_right.
    The parameter `param` controls the shape (exponent, rate, etc.).
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution of a."""
    return float(conv_self_fft(a).max())

                mask[conv.argmax()] = 1.0
                cnt = 1.0

            a_rev = a[::-1]
            L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
            M = np.fft.rfft(mask, n=L)
            A_rev = np.fft.rfft(a_rev, n=L)
            conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
            grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

            a -= step_sz * grad
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a, target_sum)

            sub_it += 1
            if max_conv < best_score - 1e-12:
                best_score = max_conv
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Sub‑grad {sub_it:05d} → {best_score:.6f}")

            if sub_it % 500 == 0:
                step_sz *= 0.94

    # --------------------------------------------------------------
    # Phase 3 – final simulated‑annealing polishing
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – simulated‑annealing polishing")
    temperature = 0.02 * best_score   # initial temperature
    while time.time() < deadline:
        # geometric cooling
        temperature = max(1e-5, temperature * 0.9999)
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.0018)
        cand_score = compute_max_conv(cand)
        delta = cand_score - best_score
        if delta < 0 or np.random.rand() < math.exp(-delta / temperature):
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time():.2f}] Polished → {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
