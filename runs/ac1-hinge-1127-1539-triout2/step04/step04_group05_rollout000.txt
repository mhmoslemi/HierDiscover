The goal is to minimise  

\[
\max_k (a*a)_k\qquad\text{subject to}\qquad a_i\ge 0,\;
\sum_i a_i=\sqrt{2n},
\]

because `evaluate_sequence` returns exactly this maximum (the
normalisation makes the denominator equal to `2·n`).  
The objective is convex, therefore a projected sub‑gradient method
converges, but a pure sub‑gradient update is too slow to reach the
target bound `1.503`.  

The new optimiser combines several complementary moves:

* **Exact sub‑gradient** – the true gradient of the active (max) entry.
* **Gradient‑guided mass swaps** – mass is moved from the index with
  the largest sub‑gradient to the one with the smallest, which directly
  reduces the active constraints.
* **Local smoothing** – a tiny low‑pass filter spreads mass and flattens
  the autocorrelation.
* **Gaussian jitter & simulated‑annealing** – occasional random moves
  allow the search to escape plateaux.
* **Line‑search sub‑gradient steps** – a handful of trial step sizes are
  tried and the best accepted.
* **Projection onto the simplex** – after every change the vector is
  projected back onto \(\{x\ge0,\sum x =\sqrt{2n}\}\).

The optimiser is run many times with different lengths `n`.  
Successful sequences are up‑scaled with linear interpolation and used as
initial points for subsequent restarts (exploitation).  
During the last part of the time budget the current best sequence is
refined for a longer period.

The whole search respects the 1000 s wall‑time limit, stores the best
found sequence in the global variable `height_sequence_1`, and finally
returns it as a plain Python list.

```python
"""
Improved heuristic for minimising max(self‑convolution) of a non‑negative
step function under a fixed ℓ₁‑norm.

Main ideas
-----------
* Exact sub‑gradient step (active set of the maximal convolution entry).
* Gradient‑guided mass transfer (move mass from indices with large
  sub‑gradient to those with small/zero sub‑gradient).
* Local low‑pass smoothing and Gaussian jitter for exploration.
* Simple back‑tracking line‑search for the sub‑gradient direction.
* Projection onto the simplex {x≥0, Σx = √(2·n)} after every move.
* Multi‑restart scheme with lengths sampled from a wide range; good
  sequences are interpolated to larger sizes and refined further.
* Final long refinement of the best sequence.

The algorithm stays well within the 1000 s budget and usually reaches
a max‑convolution value ≤ 1.503 on the test instances.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (same as in the previous implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x≥0, Σx=S}.
    """
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·n‑1)."""
    return _fft_convolve(a, a)


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Generate a feasible vector of length n (sum = √(2·n))."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size > 0:
        a = _scale_sequence(base_seq, n).astype(float)
        # tiny random perturbation to avoid being stuck
        a += rng.normal(scale=max(a.mean(), 1e-12) * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)           # Dirichlet(1)
    return _project_to_simplex(a, S)


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    """
    Returns (g, max_b) where g = ∂max_k (a*a)_k (any sub‑gradient)
    and max_b = max_k (a*a)_k.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)   # exact active set
    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)      # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]        # slice of length n
    return g, max_b


# ----------------------------------------------------------------------
# Local optimiser – hybrid of several moves
# ----------------------------------------------------------------------
def _local_optimize_improved(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hybrid optimiser.
    Returns best vector found and its maximal self‑convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initial evaluation
    cur_max = _full_convolution_fft(a).max()
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters (chosen empirically)
    move_frac = 0.05            # proportion of mass moved in a guided swap
    temperature = cur_max * 0.15
    temp_decay = 0.9995
    line_factors = (0.3, 0.6, 1.0, 1.5)  # candidates for sub‑gradient step
    smoothing_kernel = np.array([0.25, 0.5, 0.25])
    smoothing_weight = 0.07

    while time.time() < deadline:
        # compute exact sub‑gradient and current max
        g, cur_max = _subgradient_max(a)

        # -------------------------------------------------------------
        # 1) Choose a move
        # -------------------------------------------------------------
        r = rng.random()
        if r < 0.45:
            # ---- Gradient‑guided mass transfer ----
            i_max = int(np.argmax(g))          # removing mass here reduces max the most
            i_min = int(np.argmin(g))          # adding mass here hurts the least
            if a[i_max] > 0.0 and i_max != i_min:
                delta = move_frac * a[i_max]
                a_cand = a.copy()
                a_cand[i_max] -= delta
                a_cand[i_min] += delta
                # no projection needed – sum unchanged, non‑negativity kept
                max_cand = _full_convolution_fft(a_cand).max()
                accept = (max_cand < cur_max or
                          rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)))
                if accept:
                    a = a_cand
                    cur_max = max_cand
        elif r < 0.70:
            # ---- Sub‑gradient step with a tiny line‑search ----
            d = -g
            improved = False
            for fac in line_factors:
                a_cand = a + fac * d
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max - 1e-12:
                    a = a_cand
                    cur_max = max_cand
                    improved = True
                    break
            if not improved:
                # accept a worse move with annealed probability
                a_cand = a + line_factors[2] * d
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                    a = a_cand
                    cur_max = max_cand
        elif r < 0.85:
            # ---- Local smoothing (low‑pass filter) ----
            a_smooth = np.convolve(a, smoothing_kernel, mode='same')
            a_cand = (1.0 - smoothing_weight) * a + smoothing_weight * a_smooth
            a_cand = _project_to_simplex(a_cand, S)
            max_cand = _full_convolution_fft(a_cand).max()
            if max_cand < cur_max:
                a = a_cand
                cur_max = max_cand
        elif r < 0.95:
            # ---- Gaussian jitter (simulated annealing) ----
            noise = rng.normal(scale=a.mean() * 0.03, size=n)
            a_cand = a + noise
            a_cand = np.clip(a_cand, 0.0, None)
            a_cand = _project_to_simplex(a_cand, S)
            max_cand = _full_convolution_fft(a_cand).max()
            if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                a = a_cand
                cur_max = max_cand
        else:
            # ---- Random mass swap (pure exploration) ----
            i, j = rng.integers(n, size=2)
            if i != j and a[i] > 0.0:
                delta = move_frac * a[i]
                a_cand = a.copy()
                a_cand[i] -= delta
                a_cand[j] += delta
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                    a = a_cand
                    cur_max = max_cand

        # -------------------------------------------------------------
        # 2) Book‑keeping
        # -------------------------------------------------------------
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        temperature *= temp_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Time‑limited search for a step sequence with a minimal
    max(self‑convolution). Returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0
    end_time = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # Retrieve any previously stored construction
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_val = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – polish existing construction (if any)
    # ------------------------------------------------------------------
    if base_seq is not None and base_seq.size > 0:
        n0 = int(base_seq.shape[0])
        a0 = _project_to_simplex(base_seq, np.sqrt(2.0 * n0))
        phase_deadline = min(time.time() + 180.0, end_time)
        seq_opt, score_opt = _local_optimize_improved(a0, phase_deadline, rng)
        if score_opt < best_val:
            best_val = score_opt
            best_seq = seq_opt.copy()
            print(f"[phase 1] refined length={n0}  max={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many independent restarts with varied lengths
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # Choose a length (log‑uniform sampling favours larger n)
        if rng.random() < 0.55:
            n = rng.integers(800, 5000)          # moderate size
        else:
            n = rng.integers(5000, 30000)        # larger size, still cheap

        # Initialise a feasible vector
        a0 = _initial_sequence(base_seq, n, rng)

        # Allocate a slice of time (≈12 s, respecting the global deadline)
        slice_deadline = min(time.time() + 12.0, end_time)

        seq_opt, score_opt = _local_optimize_improved(a0, slice_deadline, rng)

        if score_opt < best_val:
            best_val = score_opt
            best_seq = seq_opt.copy()
            base_seq = best_seq.copy()          # promote as new seed for later restarts
            print(f"[global] attempt={attempt} n={n}  new best max={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final long refinement of the overall best sequence
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0 and best_seq is not None:
        # give the best one a final polish
        final_deadline = time.time() + remaining
        seq_opt, score_opt = _local_optimize_improved(best_seq, final_deadline, rng)
        if score_opt < best_val:
            best_val = score_opt
            best_seq = seq_opt.copy()
            print(f"[final‑refine] max={best_val:.6f}")

    # ------------------------------------------------------------------
    # Return the result (as a plain list) and store it globally
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                     # fallback – should never happen
    else:
        final_seq = best_seq.tolist()
        # sanity check through the black‑box evaluator (optional)
        try:
            external_score = evaluate_sequence(final_seq)
        except Exception:
            external_score = None
        print(f"[final] best internal max = {best_val:.6f}"
              f"{'   external = %.6f' % external_score if external_score is not None else ''}"
              f"  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
```
