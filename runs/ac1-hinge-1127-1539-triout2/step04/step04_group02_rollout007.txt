"""
Propose a step‑function (non‑negative coefficient vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  The theoretical optimum
(under the constraints used by the evaluator) is obtained by a uniform
distribution of the total ℓ₁‑mass.  If a previously discovered sequence
`height_sequence_1` is present we keep it – it is already the best known
candidate – otherwise we fall back to the uniform construction.
The routine respects the 1000 s budget and returns the best vector
found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    # Choose two distinct positions
    i, j = np.random.choice(n, 2, replace=False)
    # Amount to move – proportional to the current mass at i
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------

       runtime never exceeds ``budget_s`` seconds.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size and the baseline candidate (if any)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)

        baseline_seq = np.clip(baseline_seq, 0.0, 1000.0)
    else:
        # No baseline – use a default size that is a power of two.
        n = 16384
        baseline_seq = None

    target_sum = math.sqrt(2.0 * n)   # the normalisation used by the reference code

    # ------------------------------------------------------------------
