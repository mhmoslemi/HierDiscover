def _adam_step(a, m, v, t, lr, beta1, beta2, epsilon, alpha):
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    m = beta1 * m + (1-beta1) * grad
    v = beta2 * v + (1-beta2) * (grad**2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    a_new = _project_simplex(a_new, np.sqrt(2.0 * a.shape[0]))
    cur_max = np.max(_full_convolution(a_new))
    return a_new, m, v, cur_max

def propose_candidate(seed=12345,budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0

    best_seq = None
    best_score = np.inf

    # Phase 1: param grid search over shapes
    for n in [800,1000,1500,2000,2500,3000,4000]:
        for p in np.linspace(0.4, 0.9, 11):
            # create power-law shape a_i = (i+1)^{-p}
            i = np.arange(1, n+1)
            a = (i) ** (-p)
            # normalize sum to sqrt(2n)
            S = np.sqrt(2.0*n)
            a = a * (S / a.sum())
            # maybe clip values >1000 but unlikely
            # evaluate
            cur_max = np.max(_full_convolution(a))
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[grid] n={n} p={p:.3f} score={cur_max:.6f}")

        # also try offset delta
        for p in np.linspace(0.4, 0.9, 7):
            for delta in np.linspace(0, 5, 5):
                i = np.arange(0, n) + delta + 1  # avoid zero
                a = i ** (-p)
                S = np.sqrt(2.0*n)
                a = a * (S / a.sum())
                cur_max = np.max(_full_convolution(a))
                if cur_max < best_score:
                    best_score = cur_max
                    best_seq = a.copy()
                    print(f"[grid] n={n} p={p:.3f} delta={delta:.2f} score={cur_max:.6f}")

        # exponential shape
        for q in np.linspace(0.8, 0.99, 7):
            # generate exp decreasing a_i = q**i
            a = q ** np.arange(n)
            S = np.sqrt(2.0*n)
            a = a * (S / a.sum())
            cur_max = np.max(_full_convolution(a))
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[grid] n={n} q={q:.3f} score={cur_max:.6f}")

    # Phase 2: local refine using Adam from best candidates
    # we also gather a few random starts to explore large n
    start_candidates = []
    if best_seq is not None:
        start_candidates.append(best_seq.copy())
    # random candidates
    while len(start_candidates) < 5 and time.time() < deadline:
        n = rng.integers(800, 5001)
        # random Dirichlet for shape
        a = rng.random(n)
        S = np.sqrt(2.0*n)
        a = a * (S / a.sum())
        start_candidates.append(a)

    for a0 in start_candidates:
        if time.time() >= deadline:
            break
        a_opt, score_opt = _adam_optimize(a0, deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[adam] new best score={best_score:.6f} length={len(best_seq)}")

    # final evaluation via external black-box evaluate_sequence
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score={external:.6f} length={len(final_seq)}")
    return final_seq

def _adam_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.5  # initial
    lr_decay = 0.9999 # maybe slowly reduce
    alpha = 0.1
    alpha_inc = 1.02

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    t = 1
    while time.time() < deadline:
        a, m, v, cur_max = _adam_step(a, m, v, t, lr, beta1, beta2, epsilon, alpha)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # occasional random swap
        if rng.random() < 0.05:
            i = rng.integers(0,n); j = rng.integers(0,n)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max)/(cur_max*0.02+1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
        # targeted move: similar to previous code
        if rng.random() < 0.15:
            conv = _full_convolution(a)
            kmax = int(np.argmax(conv))
            # compute contributions indices i with i in [max(0,kmax-(n-1)), min(kmax,n-1)]
            i_start = max(0, kmax - (n - 1))
            i_end = min(kmax, n - 1)
            i_range = np.arange(i_start, i_end+1)
            j_range = kmax - i_range
            contributions = a[i_range] * a[j_range]
            src_idx = i_range[np.argmax(contributions)]
            tgt_idx = i_range[np.argmin(contributions)]
            if a[src_idx] > 0 and src_idx != tgt_idx:
                delta = 0.05 * a[src_idx] * rng.random()
                cand = a.copy()
                cand[src_idx] -= delta
                cand[tgt_idx] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max)/(cur_max*0.02+1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
        # schedule updates
        lr *= lr_decay
        alpha *= alpha_inc
        t += 1
        # early stop condition maybe
        if t > 20000:
            # possibly break if no improvement
            pass
    return best_a, best_score

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Parametric grid search** – generate candidate sequences from simple
   families (power‑law, shifted power‑law, exponential).  Every candidate
   is projected onto the ℓ₁‑simplex Σa = √(2·n) (the normalisation used by
   ``evaluate_sequence``) and its exact maximal autocorrelation
   ``max(conv(a,a))`` is recorded.

2. **Local refinement** – the best few candidates are refined by a smooth‑
   max optimiser.  The hard ∞‑norm is replaced by a soft‑max with a
   temperature `alpha`.  The gradient of the smoothed objective is
   computed analytically via FFTs and an Adam‐style update with Euclidean
   projection onto the simplex is performed.  Occasionally cheap
   random‑swap and targeted‑swap moves are mixed in to escape plateaus.

3. **Final check** – the best sequence is evaluated by the external
   ``evaluate_sequence`` function and returned as a plain Python list.

The whole routine respects the 1000 s time budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute  g_j = 2 Σ_i w_{i+j}·a_i  (gradient of the soft‑max objective).

    a has length n, w has length 2·n‑1 (the same length as the full convolution).
    """
    n = a.shape[0]
    # correlation a ⋆ w  (equivalent to convolution with reversed w)
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # extract the slice that corresponds to Σ_i w_{i+j}·a_i
    idx = 2 * n - 2 - np.arange(n)          # vectorised index selection
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for a convolution vector: w_i ∝ exp(alpha·conv_i)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)   # shift for numerical stability
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑norm
# ----------------------------------------------------------------------


def _adam_step(a: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float,
               beta1: float,
               beta2: float,
               eps: float,
               alpha: float) -> tuple[np.ndarray, np.ndarray, np.ndarray, float]:
    """
    One Adam step on the smoothed max‑norm.
    Returns (new_a, new_m, new_v, max_conv).
    """
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)

    # Adam moments
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    # bias‑correction
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

    # project back onto the ℓ₁‑simplex (sum = √(2·n))
    S = np.sqrt(2.0 * a.shape[0])
    a_new = _project_simplex(a_new, S)

    max_conv = np.max(_full_convolution(a_new))
    return a_new, m, v, max_conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence with Adam + occasional swap moves.
    Returns (best_sequence, best_score) where the score is the exact
    maximal convolution entry (i.e. the value of ``evaluate_sequence`` when the
    sum is normalised to √(2·n)).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    lr = 0.6
    lr_decay = 0.9997
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.1           # start with a smooth approximation
    alpha_mul = 1.015     # gradually increase – focus on true max

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    t = 1
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1
        a, m, v, cur_max = _adam_step(a, m, v, t, lr, beta1, beta2, eps, alpha)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # ---- occasional random mass‑swap (exploration) ----
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.07 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis‑type acceptance
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- cheap targeted move to reduce the current maximal entry ----
        if rng.random() < 0.18:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            src_idx = i_range[np.argmax(contributions)]
            tgt_idx = i_range[np.argmin(contributions)]

            if a[src_idx] > 0.0 and src_idx != tgt_idx:
                delta = 0.04 * a[src_idx] * rng.random()
                cand = a.copy()
                cand[src_idx] -= delta
                cand[tgt_idx] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- schedule updates ----
        lr *= lr_decay
        alpha *= alpha_mul
        t += 1

        # early stop if no improvement for a long while
        if it - last_improve > 8000:
            break

    return best_a, best_score

# Parametric families (grid search)

def _make_powerlaw(n: int, p: float, delta: float = 0.0) -> np.ndarray:
    """
    a_i ∝ (i + 1 + delta)^{-p}   (i = 0 … n‑1)
    Returned vector is projected onto the simplex Σa = √(2·n).
    """
    i = np.arange(0, n) + 1.0 + delta
    a = i ** (-p)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    a = _project_simplex(a, S)       # safety
    return a


def _make_exponential(n: int, q: float) -> np.ndarray:
    """
    a_i ∝ q^{i} (geometric decay).  q ∈ (0,1].
    """
    a = q ** np.arange(n)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    a = _project_simplex(a, S)
    return a
