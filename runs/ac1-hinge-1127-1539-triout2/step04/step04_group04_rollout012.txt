def _build_conv_matrix(f):
    n = f.shape[0]
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j+n, j] = f
    return A
```

Now _lp_step.

```
def _lp_step(f, M):
    n = f.shape[0]
    A = _build_conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(A.shape[0])]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if g.value is None:
        return None
    return np.array(g.value)
```

"""
Optimization of a non‑negative step function by a two‑phase method:

Phase 1 – Linear‑programming expansion.
    Starting from a feasible vector f (sum = √(2·n)), we solve the LP
        maximize   Σ g_i
        subject to g_i ≥ 0
                   (f * g)(k) ≤ M   for all k = 0,…,2n‑2
        where M = max_k (f*f)(k).
    The LP tries to increase the ℓ₁‑norm while keeping the convolution
    below the current worst value.  After rescaling g to the required sum
    we take a small convex combination f ← (1‑t)·f + t·g (t≈0.2).  The
    max‑convolution value never increases, and in practice it drops
    quickly.

Phase 2 – Adam optimisation (soft‑max approximation of the ∞‑norm).
    The best vector from phase 1 is refined with projected Adam updates,
    using the smooth maximum of the full convolution as loss.  This
    removes the remaining “bumps’’ and yields a very low final score.

The whole procedure respects the 1000 s time budget, falls back to the
previously‑found solution if available, and stores the result in the
global variable `height_sequence_1` for reuse.
"""

import time
import numpy as np
import cvxpy as cp


# ----------------------------------------------------------------------
#  Helpers
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full (zero‑padded) convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Gradient of τ·log∑ exp(b/τ) where b = a*a.
    Returns (grad, raw convolution b, max(b)).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                      # hard max ⇒ sub‑gradient
        k = np.argmax(b)
        # contributions a_i * a_{k-i}
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal entry (i=j) when k even
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # ---- soft‑max branch ----
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)

    # convolution of a reversed with the weight vector w
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, b, max_b


def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Adam optimiser (projected onto the ℓ₁‑simplex) that minimises the
    soft‑max approximation of max(a*a).  Returns the best vector found
    and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.6
    t = 0

    # initialise temperature from a hard‑max run
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        b_new = _full_convolution_fft(a_new)
        cur_max_new = np.max(b_new)

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        # occasional jitter
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense matrix A such that (A @ g) = f * g (full convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_expand(f: np.ndarray, M: float, t: float = 0.2):
    """
    One linear‑programming expansion step.
    Returns a new vector f_new = (1‑t)·f + t·g_scaled,
    where g is the LP solution that maximises Σg subject to (f * g) ≤ M.
    If the step does not improve the max‑convolution value, returns (f, M).
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(A.shape[0])]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # solve – ECOS works well for dense LPs of this size
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)

    if g.value is None:
        # solver failure – fallback to doing nothing
        return f, M

    g_opt = np.maximum(g.value, 0.0)
    sum_g = g_opt.sum()
    if sum_g <= 1e-12:
        return f, M

    # rescale to keep the ℓ₁‑norm fixed (S = √(2·n))
    S = np.sqrt(2.0 * n)
    g_scaled = g_opt * (S / sum_g)

    # convex combination
    f_new = (1.0 - t) * f + t * g_scaled

    # compute the new worst convolution entry
    new_M = np.max(_full_convolution_fft(f_new))

    # keep the step only if it improves the bound
    if new_M < M - 1e-9:
        return f_new, new_M
    else:
        return f, M

#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Find a step‑function (list of non‑negative heights) that minimises the
    provided `evaluate_sequence` score.  The method first tries a
    linear‑programming expansion, then refines the result with a projected
    Adam optimiser.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0                     # keep a safety margin for the harness
    deadline = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # 1) Initialise a feasible vector (use previous solution if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            start = np.asarray(globals()["height_sequence_1"], dtype=float)
            n = start.shape[0]
            S = np.sqrt(2.0 * n)
            f = _project_to_simplex(start, S)
            print(f"[init] using previous solution, n={n}")
        except Exception:
            start = None
    else:
        start = None

    if start is None:
        # random length in a comfortable range
        n = rng.integers(800, 1500)
        # start from a decaying exponential (the family that performed well)
        lam = rng.uniform(0.0015, 0.008)          # moderate decay
        idx = np.arange(n, dtype=float)
        raw = np.exp(-lam * idx)
        # occasional jitter for diversity
        if rng.random() < 0.3:
            raw += rng.normal(scale=raw.mean() * 0.01, size=n)
            raw = np.clip(raw, 0.0, None)
        S = np.sqrt(2.0 * n)
        f = _project_to_simplex(raw, S)
        print(f"[init] random start, n={n}, λ={lam:.5f}")

    # ensure exact ℓ₁‑norm
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    f = _project_to_simplex(f, S)

    # initial bound
    cur_max = np.max(_full_convolution_fft(f))
    best_f, best_max = f, cur_max
    print(f"[init] initial max = {best_max:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑programming expansion phase
    # ------------------------------------------------------------------
    iter_no = 0
    while time.time() < deadline and iter_no < 30:
        iter_no += 1
        best_f, best_max = _lp_expand(best_f, best_max, t=0.2)
        print(f"[lp {iter_no}] max = {best_max:.6f}")

        # stop early if improvements become tiny
        if iter_no > 1 and best_max > prev_max * (1 - 1e-4):
            break
        prev_max = best_max

    # ------------------------------------------------------------------
    # 3) Adam refinement (soft‑max of the convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        refined_f, refined_max = _adam_optimize(best_f, deadline, rng)
        if refined_max < best_max:
            best_f, best_max = refined_f, refined_max
            print(f"[adam] improved to {best_max:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the solution
    # ------------------------------------------------------------------
    final_seq = best_f.tolist()
    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_max:.6f}) length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
