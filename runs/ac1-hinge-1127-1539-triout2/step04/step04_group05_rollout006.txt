"""
Propose a new step‑function sequence that minimizes the evaluation score

    score = 2·n·max(convolution) / (sum(sequence))²

The algorithm is a multi‑restart projected Adam optimiser that directly
minimises a smooth approximation of the max‑convolution (soft‑max).  For
each restart the vector is projected onto the simplex
    {x ≥ 0,  Σ x = √(2·len(x))}
so the evaluation score equals the (un‑scaled) maximum convolution entry.
The optimiser works in the frequency domain (FFT) for O(n log n) speed,
uses an adaptive step‑size (Adam), gradually reduces the soft‑max
temperature τ and occasionally injects random jitter to escape plateaus.

The search proceeds in two phases:

1.  If a previous construction (global variable ``height_sequence_1``) is
    available it is polished with a longer optimisation run.
2.  Many independent restarts with varied lengths (small, medium,
    large) are performed until the time budget expires.  The best
    sequence found (according to the black‑box ``evaluate_sequence``)
    is stored back into ``height_sequence_1`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: nearest power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum S."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                    # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]            # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full linear convolution, length 2·n‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Linear interpolation to a new length (used for warm‑starting)
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to length ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Initialise a feasible vector (ℓ₁‑norm = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with sum √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size > 0:
        a = _scale_sequence(base_seq, n).astype(float)
        # add a tiny jitter before projection to break symmetry
        a += rng.normal(scale=max(a.mean(), 1e-12) * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        # Dirichlet(1) – uniform random on the simplex (before projection)
        a = rng.random(n)
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max entry of the convolution
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    """
    Return a sub‑gradient of max(a*a) and the current maximal entry.
    ``eps`` can be used to treat near‑max entries as active.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)
    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)           # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser that minimises a τ‑soft‑max approximation of max(b)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the smooth max objective.
    Returns the best vector seen and its (hard) max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.8               # base learning rate (after ADAM scaling)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    b_cur = _full_convolution_fft(a)
    tau = b_cur.max()                 # start with the current max
    tau_decay = 0.993                 # fairly slow decay

    # bookkeeping
    best_a = a.copy()
    best_max = b_cur.max()
    iteration = 0

    # pre‑compute the FFT length for the gradient convolution
    conv_len = 3 * n - 2               # length of conv(mask, a_rev)
    fft_len = _next_pow2(conv_len)

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) compute smooth‑max weights and gradient
        # --------------------------------------------------------------
        b = _full_convolution_fft(a)           # length 2n‑1
        max_b = b.max()

        # soft‑max weights (τ‑soft‑max)
        cur_tau = max(tau, 1e-12)
        diff = b - max_b
        w = np.exp(diff / cur_tau)
        w /= w.sum()

        # gradient g = 2 * (a_rev ⊛ w)
        a_rev = a[::-1]
        Fw = np.fft.rfft(w, n=fft_len)
        Fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

        # --------------------------------------------------------------
        # 2) Adam update + projection onto the simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a = _project_to_simplex(a_candidate, S)

        # --------------------------------------------------------------
        # 3) evaluate and keep the best point
        # --------------------------------------------------------------
        b_new = _full_convolution_fft(a)
        max_new = b_new.max()
        if max_new < best_max:
            best_max = max_new
            best_a = a.copy()

        # --------------------------------------------------------------
        # 4) temperature decay, occasional hard‑max sub‑gradient tweak
        # --------------------------------------------------------------
        tau = tau * tau_decay

        # Every few hundred iterations try a pure sub‑gradient step
        if iteration % 300 == 0:
            g_sub, max_sub = _subgradient_max(a, eps=1e-12)
            # a small Adam‑scaled step in the sub‑gradient direction
            a_sub = a - 0.4 * lr * g_sub
            a_sub = _project_to_simplex(a_sub, S)
            max_sub_val = _full_convolution_fft(a_sub).max()
            if max_sub_val < best_max:
                best_max = max_sub_val
                best_a = a_sub.copy()
                a = a_sub     # keep the improvement

        # --------------------------------------------------------------
        # 5) random jitter (helps escape flat plateaus)
        # --------------------------------------------------------------
        if rng.random() < 0.008:
            noise = rng.normal(scale=0.02 * a.mean(), size=n)
            a = _project_to_simplex(np.clip(a + noise, 0.0, None), S)

    # final best value (hard max)
    final_max = _full_convolution_fft(best_a).max()
    return best_a, final_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step sequence that minimises the black‑box
    evaluation function.  The routine runs for at most ``budget_s``
    seconds, stores the best sequence in the global variable
    ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously discovered construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – polish the existing construction (if available)
    # ------------------------------------------------------------------
    if base_seq is not None and base_seq.size > 0:
        n0 = int(base_seq.shape[0])
        a0 = _project_to_simplex(base_seq, np.sqrt(2.0 * n0))
        phase_deadline = min(time.time() + 150.0, end_time)   # up to 2½ min
        seq_opt, _ = _adam_optimize(a0, phase_deadline, rng)
        ext_score = evaluate_sequence(seq_opt.tolist())
        if ext_score < best_score:
            best_score = ext_score
            best_seq = seq_opt.tolist()
            print(f"[phase‑1] refined n={n0} external={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many independent restarts with varied lengths
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # pick a length: small / medium / large (biased towards medium)
        r = rng.random()
        if r < 0.55:
            n = rng.integers(800, 3000)          # small‑to‑moderate
        elif r < 0.85:
            n = rng.integers(3000, 9000)         # medium
        else:
            n = rng.integers(9000, 25000)        # large (still cheap)

        # initialise a feasible vector
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a time slice (≈10 s) respecting the global deadline
        slice_deadline = min(time.time() + 10.0, end_time)

        seq_opt, _ = _adam_optimize(a0, slice_deadline, rng)

        # external evaluation (this is cheap compared to the optimisation)
        ext_score = evaluate_sequence(seq_opt.tolist())
        if ext_score < best_score:
            best_score = ext_score
            best_seq = seq_opt.tolist()
            print(f"[global] attempt={attempt} n={n} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – store the best sequence globally
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0]                         # fallback – should never happen

    globals()['height_sequence_1'] = best_seq
    print(f"[final] best external score = {best_score:.6f} "
          f"length = {len(best_seq)}")
    return best_seq
