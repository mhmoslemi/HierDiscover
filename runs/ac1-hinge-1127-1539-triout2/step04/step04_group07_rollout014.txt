def _subgrad_optimize(a0, deadline, rng, max_iter=None):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    last_improve = 0
    while (time.time() < deadline) and (max_iter is None or it < max_iter):
        it += 1
        # compute conv and max index
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        # gradient from max entry
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        grad = _conv_with_weights(a, w)
        # step size schedule
        lr = 0.5 / np.sqrt(it)
        a_new = a - lr * grad
        a_new = _project_simplex(a_new, S)
        cur_max = np.max(_full_convolution(a_new))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a_new.copy()
            last_improve = it
        # opportunistic random swap
        if rng.random() < 0.1:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a_new[i] > 0:
                delta = a_new[i] * 0.05 * rng.random()
                cand = a_new.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a_new = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a_new.copy()
                        last_improve = it

        # cheap targeted move (similar to previous)
        if rng.random() < 0.2:
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contrib = a[i_range] * a[j_range]
            src_idx = i_range[np.argmax(contrib)]
            tgt_idx = i_range[np.argmin(contrib)]
            if a[src_idx] > 0 and src_idx != tgt_idx:
                delta = a[src_idx] * 0.04 * rng.random()
                cand = a.copy()
                cand[src_idx] -= delta
                cand[tgt_idx] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it
        else:
            a = a_new

        # break if stagnated long
        if it - last_improve > 5000:
            break
    return best_a, best_score

"""
Proposes a non‑negative step function that minimizes the given
evaluate_sequence() objective.

The method explores a compact parametric family

    a_i = C · (i + β + 1)^{-p} · q^{\,i} ,    i = 0 … n‑1

where:
* n   – length of the sequence,
* p   – power‑law exponent (≈ 0.5 gives a flat auto‑convolution),
* β   – small shift to avoid the singularity at i = –1,
* q   – exponential damping factor (0 < q ≤ 1),
* C   – scaling chosen so that Σ a_i = √(2·n).

All candidates are projected onto the simplex {x ≥ 0, Σx = √(2·n)}.
The best candidates are then refined by a sub‑gradient descent on the
exact maximum entry of the auto‑convolution (the true objective when
the sum is fixed).  Random mass‑swap moves and cheap targeted swaps
help escape shallow local minima.

The search runs for the whole time budget (≤ 1000 s) and finally
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_k w_k (a*a)[k] w.r.t. a.
    The formula is
        g_j = 2 Σ_i w_{i+j}·a_i .
    """
    n = a.shape[0]
    # correlation a ⋆ w_rev
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (max‑convolution objective)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence a0 using a sub‑gradient descent on the
    exact max‑convolution value.  Occasional random swaps and cheap
    targeted moves are performed to encourage escape from local minima.
    Returns the best sequence found and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # ---- sub‑gradient step on the current maximal entry ----
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # one‑hot weight vector for the maximal index
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        grad = _conv_with_weights(a, w)

        # step size decreasing like 1/sqrt(t)
        lr = 0.5 / np.sqrt(it)
        a_new = a - lr * grad
        a_new = _project_simplex(a_new, S)

        cur_max = np.max(_full_convolution(a_new))

        if cur_max < best_score:
            best_score = cur_max
            best_a = a_new.copy()
            last_improve = it

        # ---- occasional random mass‑swap (simulated‑annealing flavour) ----
        if rng.random() < 0.12:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * 0.05 * rng.random()
                cand = a_new.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis criterion
                if cand_max <= cur_max or rng.random() < np.exp(
                    -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)
                ):
                    a_new = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a_new.copy()
                        last_improve = it

        # ---- cheap targeted move that explicitly reduces the current maximal entry ----
        if rng.random() < 0.18:
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]
            src_idx = i_range[np.argmax(contributions)]
            tgt_idx = i_range[np.argmin(contributions)]
            if a[src_idx] > 0.0 and src_idx != tgt_idx:
                delta = a[src_idx] * 0.04 * rng.random()
                cand = a.copy()
                cand[src_idx] -= delta
                cand[tgt_idx] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(
                    -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)
                ):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it
        else:
            a = a_new

        # early stop if no improvement for a while
        if it - last_improve > 6000:
            break

    return best_a, best_score
