def _adam_optimize(a0, deadline, rng):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    # convert to probabilities (softmax)
    eps = 1e-12
    soft = a0 / S
    soft = np.clip(soft, eps, None)
    soft = soft / soft.sum()
    p = np.log(soft)  # initial logits

    # Adam parameters
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2  # base learning rate for logits
    t = 0

    # initial tau
    b = _full_convolution_fft(a0)
    tau = max(b.max(), 1.0)

    best_a = a0.copy()
    best_max = b.max()

    # loop
    while time.time() < deadline:
        t += 1
        # compute a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # compute convolution & max
        b = _full_convolution_fft(a)
        max_b = b.max()

        # update best
        if max_b < best_max:
            best_max = max_b
            best_a = a.copy()

        # compute gradient g (softmax approx)
        if tau > 1e-6:
            diff = b - max_b
            exps = np.exp(diff / tau)
            w = exps / np.sum(exps)
            # compute gradient
            a_rev = a[::-1]
            conv_len = len(w) + n - 1
            f_len = _next_pow2(conv_len)
            Fw = np.fft.rfft(w, n=f_len)
            Fa_rev = np.fft.rfft(a_rev, n=f_len)
            conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
            g = 2.0 * conv[n-1:n-1+n]
        else:
            # use hard subgradient
            g, _ = _subgradient_max(a)

        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)  # shape (n,)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_p
        v = beta2 * v + (1 - beta2) * (grad_p ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal tau
        tau *= 0.9995

        # occasional jitter to avoid stagnation
        if t % 250 == 0:
            # add gaussian jitter to logits
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_max

"""
adversarial step‑function optimizer

The optimizer looks for a non‑negative vector `a` of length `n` with
ℓ₁‑norm `S = sqrt(2·n)` that minimizes

    2·n·max(a*a) / S²

i.e. the peak of its self‑convolution under a fixed total mass.
The algorithm combines

* a smooth approximation of the ∞‑norm (soft‑max over the convolution
  entries) giving an explicit gradient,
* Adam updates on the soft‑max logits (the vector lives on the simplex by
  construction),
* a geometric cooling of the smoothing parameter `tau`,
* occasional Gaussian jitter in the log‑space to escape plateaus,
* many independent restarts with different lengths and initial shapes
  (uniform, edge‑heavy, random Dirichlet).

All operations are performed with FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s budget, keeps the best candidate seen,
stores it in the global variable ``height_sequence_1`` and returns the
corresponding list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full, linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of a τ‑soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (g, max_b) where g = ∂/∂a  τ‑soft‑max(b)
    and max_b is the current (hard) maximum of b = a*a.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # fallback: exact sub‑gradient (all active entries)
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # soft‑max weights, shape (2n‑1,)
        mask = w

    # gradient w.r.t a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimisation on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimises the soft‑max approximation of the convolution peak.
    Returns the best vector encountered and the corresponding hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits so that softmax(p)*S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits (no projection needed later)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2

    # smoothing schedule
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- a from logits -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution and gradient -------
        g, peak = _softmax_gradient(a, tau)

        # keep best (hard) peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient wrt logits -------
        mu = np.dot(a, g) / S                     # scalar
        grad_p = a * (g - mu)                     # shape (n,)

        # optional gradient clipping (helps stability on large n)
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- schedules -------
        tau *= 0.9995               # slowly tighten the soft‑max
        lr *= 0.9999                # slow decay of step size

        # occasional jitter in log‑space
        if it % 500 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises evaluate_sequence().
    The routine runs a time‑limited multi‑restart Adam optimisation,
    keeping the best vector found.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # reuse an existing construction if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Helper: build an initial feasible vector for a given length
    # ------------------------------------------------------------------
    def _initial_vector(n):
        S = np.sqrt(2.0 * n)
        shape = rng.choice(['uniform', 'edge', 'random'])
        if shape == 'uniform':
            a = np.full(n, S / n)
        elif shape == 'edge':
            # edge‑heavy: a_i ∝ (dist_to_edge+1)^(-p)
            p_exp = rng.uniform(0.7, 2.0)
            idx = np.arange(n)
            d = np.minimum(idx, n - 1 - idx) + 1          # 1‑based distance
            a = (d.astype(float) ** (-p_exp))
            a = a / a.sum() * S
        else:  # random Dirichlet
            a = rng.random(n)
            a = a / a.sum() * S
        # optionally mix with the best known construction
        if base_seq is not None and rng.random() < 0.2:
            a = _scale_sequence(base_seq, n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * S
        return a

    # ------------------------------------------------------------------
    # Main loop: independent restarts until time runs out
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1
        # pick a length (mix small‑moderate and larger sizes)
        if rng.random() < 0.6:
            n = rng.integers(1500, 8000)       # moderate
        else:
            n = rng.integers(8000, 25000)      # larger

        a_start = _initial_vector(n)

        # allocate a slice of the remaining budget (≈12 s per restart)
        slice_deadline = min(time.time() + 12.0, end_time)

        a_opt, peak = _adam_optimize(a_start, slice_deadline, rng)

        score = 2.0 * n * peak / (np.sum(a_opt) ** 2)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[restart] attempt={attempt} n={n}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
