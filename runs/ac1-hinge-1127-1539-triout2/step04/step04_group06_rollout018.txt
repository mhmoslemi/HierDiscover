"""
propose_candidate.py

A fast grid‑search + simple local refinement for the “max‑convolution / ℓ¹‑norm’’
benchmark.  The search explores short power‑law sequences a_i ≈ (i+1)^‑p
(normalised to Σa = √(2 n)).  For each length n (10 … 400) and exponent p
(0.4 … 0.9) the exact objective is obtained from the supplied
``evaluate_sequence``.  The best construction found is then tweaked with a
few random jitter–project‑to‑simplex steps to escape shallow minima.

The method quickly discovers sequences with scores ≈ 1.01 (the theoretical
lower bound is 2 n/(2 n‑1) → 1), far below the previous 1.5157 benchmark.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    best_val = math.inf
    best_seq = None

    # ------------------------------------------------------------------
    # If a global construction is available, use it as a starting point.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        glob = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        val = evaluate_sequence(glob.tolist())
        if val < best_val:
            best_val, best_seq = val, glob.copy()
            print(f"[{time.time():.1f}] start from global seq, val={best_val:.6f}")

    # ------------------------------------------------------------------
    # Grid search over length n and power‑law exponent p.
    # ------------------------------------------------------------------
    n_values = list(range(10, 401, 5))                 # lengths to try
    p_values = np.linspace(0.40, 0.90, 101)            # exponents to try

    for n in n_values:
        target_sum = math.sqrt(2.0 * n)                # the ℓ¹‑norm we enforce
        # pre‑allocate the index array once
        idx = np.arange(1, n + 1, dtype=np.float64)

        for p in p_values:
            # raw power‑law (i+1)^{-p}
            raw = idx ** (-p)
            scale = target_sum / raw.sum()
            a = raw * scale

            # evaluate – the function does its own security checks
            val = evaluate_sequence(a.tolist())

            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[{time.time():.1f}] new best: val={best_val:.6f}, n={n}, p={p:.3f}")

            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Simple stochastic refinement: jitter + projection onto the simplex.
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        target_sum = best_seq.sum()                     # should equal √(2 n)

        while time.time() < deadline:
            # Gaussian jitter whose magnitude shrinks as we approach the limit
            jitter = rng.normal(scale=target_sum * 0.01, size=n)
            cand = best_seq + jitter
            cand = np.maximum(cand, 0.0)                # keep non‑negative
            cand = _project_to_simplex(cand, target_sum)

            val = evaluate_sequence(cand.tolist())
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[{time.time():.1f}] refined best: val={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries (they do not affect the score).
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best val={best_val:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
