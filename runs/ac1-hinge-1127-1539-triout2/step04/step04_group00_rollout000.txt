"""
Propose a step‑function (non‑negative vector) that minimizes the
self‑convolution max‑norm used by ``evaluate_sequence``.

Algorithm
---------
* Work on the simplex Σ a = √(2·n).  The objective is simply the
  maximum entry of the full convolution a*a.
* For each candidate length we run a two‑phase optimiser:
    - **Phase 1 (smooth Adam)** – minimise a log‑sum‑exp
      approximation of the max using Adam.  The temperature α is
      slowly increased so that the smooth objective converges to the
      true max.
    - **Phase 2 (sub‑gradient Adam)** – focus on the currently worst
      shift (the index where the convolution attains its maximum) and
      perform Adam updates with that sub‑gradient.
* Between updates we project onto the simplex, occasionally perform
  tiny random “mass‑swap’’ moves and (optionally) enforce symmetry.
* The outer loop repeatedly restarts with different lengths (800‑8000)
  and with a resized version of the best known sequence.  The best
  candidate seen so far is kept and finally returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generator
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If ``base_seq`` is supplied and long enough, its leading part (with tiny jitter)
    is used; otherwise a Dirichlet/Uniform sample is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)               # Dirichlet(1) – uniform on simplex

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence to length ``new_n`` by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    # Normalise to unit sum before interpolation (easier)
    seq = seq / seq.sum()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_vals = np.interp(x_new, x_old, seq)
    # Re‑scale to the correct simplex sum √(2·new_n)
    S = np.sqrt(2.0 * new_n)
    new_vals = new_vals / new_vals.sum() * S
    return new_vals


# ----------------------------------------------------------------------
# Adam optimiser (smooth + sub‑gradient phases)
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that first minimises a soft‑max approximation of the max
    convolution entry, then switches to the true sub‑gradient of the max.
    Returns the best sequence found and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.2                     # base learning rate

    # smooth‑phase parameters
    alpha = 0.5                  # temperature
    alpha_mul = 1.02
    alpha_max = 80.0

    # bookkeeping
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    phase = "smooth"

    # -------------------------------------------------------------
    # Main optimisation loop
    # -------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)

        # record best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # ---------------------------------------------------------
        # Phase switching
        # ---------------------------------------------------------
        if phase == "smooth" and (alpha >= alpha_max or it > 3000):
            # after enough annealing switch to sub‑gradient
            phase = "subgrad"
            # reset Adam accumulators for the new phase
            m.fill(0.0)
            v.fill(0.0)
            it = 0  # reuse counter for sub‑grad steps

        # ---------------------------------------------------------
        # Compute gradient
        # ---------------------------------------------------------
        if phase == "smooth":
            # soft‑max weights and gradient of the smooth approximation
            w = _smooth_weights(conv, alpha)
            grad = _conv_with_weights(a, w)
        else:
            # sub‑gradient of the actual max entry
            k_max = int(np.argmax(conv))
            # Build gradient g_i = 2 * a_{k_max-i} where defined
            g = np.zeros_like(a)
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            idx_i = np.arange(i_low, i_high + 1)
            idx_j = k_max - idx_i
            g[idx_i] = a[idx_j]
            grad = 2.0 * g

        # ---------------------------------------------------------
        # Adam update
        # ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        a = _project_simplex(a, S)

        # optional symmetrisation (helps in many runs)
        if symmetrize:
            a = (a + a[::-1]) * 0.5   # preserves sum and non‑negativity

        # ---------------------------------------------------------
        # Random mass‑swap perturbation (helps escape plateaux)
        # ---------------------------------------------------------
        if rng.random() < 0.04:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Accept if improvement (or with a tiny probability)
                if cand_max < cur_max or rng.random() < 0.15:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # ---------------------------------------------------------
        # Update smooth‑phase temperature
        # ---------------------------------------------------------
        if phase == "smooth":
            alpha = min(alpha * alpha_mul, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load previous construction if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0
    last_improve_time = start_time

    # candidate lengths – we sample from a broad range; smaller n gives faster steps
    length_choices = [800, 1200, 1600, 2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000]

    while time.time() < deadline:
        attempts += 1

        # ------------------------------------------------------------------
        # Choose a length for this attempt
        # ------------------------------------------------------------------
        if best_overall_seq is None or (time.time() - last_improve_time > 25):
            # fresh start – pick a random length
            n = int(rng.choice(length_choices))
            a0 = _initial_sequence(base_seq, n, rng)
        else:
            # continue from the current best, possibly resized
            n_target = int(rng.choice(length_choices))
            if best_overall_seq.shape[0] == n_target:
                a0 = best_overall_seq.copy()
                n = n_target
            else:
                a0 = _resize_sequence(best_overall_seq, n_target)
                n = n_target

        # ------------------------------------------------------------------
        # Allocate a slice of remaining budget for this attempt
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + 12.0, deadline)

        # ------------------------------------------------------------------
        # Optimise (Adam smooth → sub‑gradient)
        # ------------------------------------------------------------------
        sym_flag = rng.random() < 0.5      # randomly decide whether to enforce symmetry
        a_opt, score = _optimize_adam(a0, slice_deadline, rng, symmetrize=sym_flag)

        # ------------------------------------------------------------------
        # Record global best
        # ------------------------------------------------------------------
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={n} new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final local refinement on the best candidate (use remaining time)
    # ------------------------------------------------------------------
    if best_overall_seq is not None and time.time() < deadline:
        final_deadline = deadline
        a_refined, final_score = _optimize_adam(best_overall_seq,
                                                final_deadline,
                                                rng,
                                                symmetrize=False)
        if final_score < best_overall_score:
            best_overall_seq = a_refined
            best_overall_score = final_score

    # ------------------------------------------------------------------
    # Convert to Python list and sanity‑check via the external evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]   # fallback (should never happen)
    else:
        final_seq = best_overall_seq.tolist()

    # External evaluation (the harness will also call evaluate_sequence)
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
