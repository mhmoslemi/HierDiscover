"""
Propose a step‑function (non‑negative coefficient vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  The algorithm works inside the
1000 s time budget and consists of four stages:

1.  Wide parametric sweep over many “U‑shaped”, “edge‑biased” and random
    Dirichlet vectors.  All vectors are projected onto the ℓ₁‑simplex with
    Σa_i = √(2 n).  The best candidate is kept.

2.  Smooth optimisation of a log‑sum‑exp surrogate of the maximum
    using an Adam‑style update.  The temperature τ is reduced
    geometrically, while the learning‑rate decays slowly.  After each
    step the vector is projected onto the simplex.

3.  Sub‑gradient descent on the true max‑convolution.  The step size is
    decreased and occasional random pair‑transfers are performed to
    escape plateaus.

4.  Simulated‑annealing polishing: cheap random pair‑transfers are
    accepted if they improve the objective or with a temperature‑controlled
    probability.

The best vector discovered during the whole run is returned as a plain
Python list.  The implementation avoids closures, uses only top‑level
functions and relies on NumPy (FFT) for fast convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT of length
    equal to the next power of two.  Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float,
                   kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or edge‑biased sequence.
    `kind` may be:

        linear, quadratic, exponential, power,
        power_edge, exponential_edge, log_edge,
        linear_left, linear_right,
        exponential_left, exponential_right.

    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "power_edge":
        # larger near the edges, tiny in the centre
        raw = (np.minimum(idx + 1, n - idx)) ** (-param) + 1e-6
    elif kind == "exponential_edge":
        raw = np.exp(-param * np.minimum(idx + 1, n - idx))
    elif kind == "log_edge":
        raw = -np.log(np.minimum(idx + 1, n - idx) / n + 1e-8) + 1e-6
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)

    print(f"[{time.time():.2f}] Phase 3 – simulated‑annealing polishing")
    temperature = 0.01 * best_score   # start temperature
    while time.time() < deadline:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.0025)
        cand_score = compute_max_conv(cand)
        delta = cand_score - best_score
        if delta < 0 or np.random.rand() < math.exp(-delta / max(temperature, 1e-12)):
            best_seq = cand
            best_score = cand_score
            print(f"Polish improved to {best_score:.6f}")
        # exponential cooling
        temperature *= 0.9995
        if temperature < 1e-8:
            temperature = 1e-8

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
