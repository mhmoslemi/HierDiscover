"""
Multi‑stage optimizer for the max‑self‑convolution objective.

1.  Load a previously found construction (global ``height_sequence_1``)
    if it exists, otherwise start from a uniform vector.
2.  Phase 0 – a quick parametric sweep (U‑shapes, skewed shapes,
    random Dirichlet vectors) to obtain a decent initial point.
3.  Phase 1 – smoothed log‑sum‑exp minimisation with an Adam‑type
    optimiser.  The smoothness parameter τ is reduced geometrically.
    The iterates are projected onto the ℓ₁‑simplex
    `{x_i≥0, Σx_i = √(2 n)}` after every update.
4.  Phase 2 – simulated‑annealing local search.  Random pair‑mass
    transfers are accepted according to a decreasing temperature.
    Every few hundred iterations a *targeted* pair transfer is tried:
    mass is moved from the index that contributes most to the current
    maximal convolution entry to the index that contributes least.
5.  Phase 3 – final polishing with tiny random transfers.

The routine respects the 1000 s budget and returns the best sequence
found (a plain ``list[float]``).  The returned vector satisfies all
constraints of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of ``a`` with itself using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def targeted_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Move mass from the index that contributes most to the current
    maximal convolution entry to the index that contributes least.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    k_max = int(np.argmax(conv))

    # partner[i] = a[k_max - i] if the index is inside the array, else 0
    partner = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1)
    partner[idx] = a[k_max - idx]

    contribution = a * partner
    i_max = int(np.argmax(contribution))
    i_min = int(np.argmin(contribution))

    delta = scale * a[i_max]
    if delta <= 0:
        return a
    a[i_max] -= delta
    a[i_min] += delta
    return project_onto_simplex(a, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())


def compute_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed maximum:
        ∇_a  τ·log(∑_k exp((c_k)/τ)),
    where c_k = (a*a)[k].
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_c = conv.max()
    # log‑sum‑exp weights
    w = np.exp((conv - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w[:] = 1.0 / w.size
    else:
        w /= w_sum

    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * full[n - 1: n - 1 + n]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    ``evaluate_sequence`` objective.  The routine respects the 1000 s
    time budget and returns the best vector discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.25   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size (use existing construction if present)
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        a = np.asarray(base, dtype=np.float64)
        a = project_onto_simplex(a, math.sqrt(2.0 * n))
    else:
        n = 16384           # default length (power of two for fast FFT)
        a = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)

    target_sum = math.sqrt(2.0 * n)   # we keep the ℓ₁‑norm fixed

    # --------------------------------------------------------------
    # Phase 0 – quick parametric sweep + few random vectors
    # --------------------------------------------------------------
    best_seq = a.copy()
    best_score = compute_max_conv(best_seq)

    shape_kinds = (
        "linear", "quadratic", "exponential",
        "power", "inverse",
        "linear_left", "linear_right",
        "exponential_left", "exponential_right",
    )
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),
        "quadratic": np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 12),
        "inverse": np.linspace(0.5, 5.0, 12),
        "linear_left": np.linspace(0.0, 0.9, 19),
        "linear_right": np.linspace(0.0, 0.9, 19),
        "exponential_left": np.linspace(0.0, 6.0, 31),
        "exponential_right": np.linspace(0.0, 6.0, 31),
    }

    phase0_deadline = start_time + budget_s * 0.07
    print(f"[{time.time():.2f}] Phase 0 – parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"   New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= phase0_deadline:
                break
        if time.time() >= phase0_deadline:
            break

    # a few random Dirichlet‑like vectors
    while time.time() < phase0_deadline:
        cand = np.random.rand(n)
        cand = project_onto_simplex(cand, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"   New best random → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation of a log‑sum‑exp surrogate
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.55
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    iteration = 0
    tau_factor = 0.20
    lr0 = 0.4 * (target_sum / n)    # initial step‑size
    print(f"[{time.time():.2f}] Phase 1 – Adam on smoothed max")
    while time.time() < phase1_deadline:
        iteration += 1
        conv = conv_self_fft(a)
        max_conv = conv.max()
        # keep track of the true objective (max‑convolution)
        if max_conv < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"  Iter {iteration:06d} – improved max to {best_score:.6f}")

        # smoothing parameter (never below a tiny epsilon)
        tau = max(tau_factor * max_conv, 1e-12)

        # gradient of the smooth surrogate
        grad = compute_gradient(a, tau)

        # Adam update (biased‑corrected moments)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # adapt a few hyper‑parameters
        if iteration % 2500 == 0:
            lr0 *= 0.94
            tau_factor = max(0.005, tau_factor * 0.94)

        # occasional exploratory moves
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.004)
        if np.random.rand() < 0.008:
            a = targeted_pair_transfer(a, target_sum, scale=0.005)

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing local search
    # --------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.90
    temperature = best_score * 0.05 + 1e-3
    temp_decay = 0.9995
    iter_sa = 0
    print(f"[{time.time():.2f}] Phase 2 – simulated annealing")
    while time.time() < phase2_deadline:
        iter_sa += 1
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.003)
        cand_score = compute_max_conv(cand)
        delta = cand_score - best_score
        if delta < -1e-12:
            best_seq = cand
            best_score = cand_score
        else:
            # Metropolis accept‑probability
            if np.random.rand() < math.exp(-delta / max(temperature, 1e-12)):
                best_seq = cand

        if iter_sa % 2000 == 0:
            temperature *= temp_decay
            # occasional targeted move to shake the solution
            best_seq = targeted_pair_transfer(best_seq, target_sum, scale=0.004)

    # --------------------------------------------------------------
    # Phase 3 – final polishing with tiny random transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – polishing")
    while time.time() < overall_deadline:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.0015)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"   Polished → {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (max‑conv={best_score:.6f})")
    return best_seq.tolist()

def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence of length n.
    `kind` may be one of:
        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right, exponential_left, exponential_right.
    The parameter `param` controls the shape (exponent, rate, etc.).
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)
