    return g

import scipy.sparse as sp

def conv_matrix(f):
    n = len(f)
    diags = []
    offsets = []
    for d in range(n):
        # create array of length n filled with f[d]
        diags.append(np.full(n, f[d]))
        offsets.append(d)
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    return A

def _lp_step_optimize(a0, deadline, rng):

    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_max = _full_convolution_fft(a).max()
    # parameters
    t = 0.3
    t_min = 1e-4
    t_decay = 0.99
    while time.time() < deadline:
        # compute M
        b = _full_convolution_fft(a)
        M = b.max()
        # Build constraint matrix
        A = _conv_matrix(a)   # shape (2n-1, n) sparse
        # Solve LP
        c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n,
                                    method='highs')
        if not res.success:
            # fallback break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            break
        # scale to target sum
        alpha = S / sum_g
        g_scaled = g * alpha

        # backtracking line-search for step size to keep max non‑increasing
        t_cur = t
        improved = False
        while t_cur >= t_min:
            a_new = (1 - t_cur) * a + t_cur * g_scaled
            a_new = _project_to_simplex(a_new, S)
            max_new = _full_convolution_fft(a_new).max()
            if max_new <= M * (1 + 1e-9):
                a = a_new
                M = max_new
                improved = True
                break
            else:
                t_cur *= 0.5
        if not improved:
            # cannot improve, break
            break
        # record best if better
        if M < best_max - 1e-12:
            best_max = M
            best_a = a.copy()
        # decay step size
        t *= t_decay
        if t < t_min:
            t = t_min
    return best_a, best_max

```python
"""...
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥0, sum x = S}."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution using real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:2 * n - 1]
    return conv

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y via FFT."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:n]
    return conv

def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix A (size (2n‑1)×n) such that A @ g = f * g."""
    n = f.shape[0]
    # each diagonal offset d (0 ≤ d < n) has constant value f[d]
    diags = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)   # 0,1,…,n‑1
    return sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')

def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of seq to length new_n."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    # use np.interp for speed
    x_old = np.linspace(0, old_n - 1, old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    return np.interp(x_new, x_old, seq)

def _lp_step_optimize(a0: np.ndarray, deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Iteratively improve a by LP‑based updates.
    Returns best vector encountered and its maximal self‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initial record
    cur_max = _full_convolution_fft(a).max()
    best_a = a.copy()
    best_max = cur_max

    # parameters
    t = 0.3                 # initial convex‑combination weight
    t_min = 1e-4
    t_decay = 0.995

    while time.time() < deadline:
        # 1. compute current bound M
        M = cur_max

        # 2. build convolution matrix for current a
        A = _conv_matrix(a)                     # shape (2n‑1, n)

        # 3. solve LP: maximize sum(g) subject to A @ g ≤ M, g ≥ 0
        c = -np.ones(n)                         # minimise -sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds,
                                    method='highs',
                                    options={'presolve': True})
        if not res.success:
            # LP failed – abort this run
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # 4. scale g to satisfy ℓ₁‑norm = S
        alpha = S / sum_g
        g_scaled = g * alpha

        # 5. back‑tracking line‑search on the convex‑combination weight
        t_cur = t
        improved = False
        while t_cur >= t_min:
            a_candidate = (1.0 - t_cur) * a + t_cur * g_scaled
            a_candidate = _project_to_simplex(a_candidate, S)
            max_candidate = _full_convolution_fft(a_candidate).max()
            # accept if we do not increase the peak (tiny tolerance allowed)
            if max_candidate <= M * (1.0 + 1e-9):
                a = a_candidate
                cur_max = max_candidate
                improved = True
                break
            else:
                t_cur *= 0.5

        if not improved:
            # cannot find a non‑increasing step – terminate
            break

        # 6. bookkeeping
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()

        # 7. decay step size for next iteration
        t *= t_decay
        if t < t_min:
            t = t_min

    return best_a, best_max

def _softmax_update(a: np.ndarray, tau: float, eta: float):

        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(w, n=f_len)

        g = 2.0 * conv[n - 1:n - 1 + n]    # slice length n
    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)
    b_new = _full_convolution_fft(a_proj)
    max_new = b_new.max()
    return a_proj, b_new, max_new

def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    n = a.shape[0]
    b = _full_convolution_fft(a)

    conv = _fft_convolve(mask, a_rev)          # length 3n‑2

"""
Propose a step‑function minimizing the black‑box evaluation.

The algorithm mixes two complementary optimisation phases:

1. **LP‑based alternating scheme**  
   For a current vector `f` we compute `M = max(f*f)`.  
   A linear programme (maximise ∑g subject to f∗g ≤ M, g≥0) returns a
   vector `g` whose ℓ₁‑norm is as large as possible while keeping the
   cross‑convolution bounded by `M`.  After scaling `g` to the required
   ℓ₁‑norm (`√(2·n)`) we move `f ← (1‑t) f + t g` (t is gradually
   reduced) and accept the step only if the maximal self‑convolution does
   not increase.  Repeating this step drives the peak of `f*f` down.

2. **Gradient‑based local refinement**  
   A hybrid optimiser (soft‑max gradient, exact sub‑gradient, simulated‑annealing
   mass swaps and Gaussian jitter) is run on the best vector found by the
   LP phase.  This cleans up remaining irregularities.

Multiple random restarts with varied lengths are performed within the
overall time budget (≈1000 s).  The best sequence discovered is returned
and stored in the global variable ``height_sequence_1`` for future runs.
"""

import time

import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = S}."""
    if S <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, S / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:2 * n - 1]
    return conv


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of x with y via FFT."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:n]
    return conv


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Sparse matrix A (size (2n‑1)×n) such that A @ g = f * g.
    Entry A[i,k] = f[i‑k] when 0 ≤ i‑k < n, else 0.
    """
    n = f.shape[0]
    # each diagonal offset d (0 ≤ d < n) contains the constant f[d]
    diags = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)      # 0,…,n‑1
    return sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, old_n - 1, old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    return np.interp(x_new, x_old, seq)

# LP‑based alternating optimisation
# ----------------------------------------------------------------------
def _lp_step_optimize(a0: np.ndarray, deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Starting from ``a0`` (already projected onto the ℓ₁‑simplex),
    repeatedly solve the LP max ∑g  s.t.  a∗g ≤ M, g ≥ 0,
    scale g to the target ℓ₁‑norm and move a ← (1‑t) a + t g.
    The step size ``t`` is back‑tracked to guarantee a non‑increase
    of the maximal self‑convolution.
    Returns the best vector seen and its maximal self‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_max = _full_convolution_fft(a).max()
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters
    t = 0.3               # initial convex‑combination weight
    t_min = 1e-4
    t_decay = 0.995

        M = cur_max

        # Build the (2n‑1)×n convolution matrix for the current a
        A = _conv_matrix(a)

        # Linear programme: maximise 1ᵀg  ⇔ minimise -1ᵀg
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds,
                                    method='highs',
                                    options={'presolve': True})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # Scale g to the required ℓ₁‑norm
        alpha = S / sum_g
        g_scaled = g * alpha

        # Back‑tracking line search on the convex combination weight
        t_cur = t
        improved = False
        while t_cur >= t_min:
            a_cand = (1.0 - t_cur) * a + t_cur * g_scaled
            a_cand = _project_to_simplex(a_cand, S)
            max_cand = _full_convolution_fft(a_cand).max()
            if max_cand <= M * (1.0 + 1e-9):
                a = a_cand
                cur_max = max_cand
                improved = True
                break
            t_cur *= 0.5

        if not improved:
            break   # cannot find a non‑increasing step – stop

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()

        t *= t_decay
        if t < t_min:
            t = t_min

    return best_a, best_max

# Gradient‑based local optimiser (taken from the previous solution)

def _softmax_update(a: np.ndarray, tau: float, eta: float):
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau <= 1e-12:
        g = np.zeros_like(a)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)                 # shape (2n‑1,)
        a_rev = a[::-1]
        conv_len = len(w) + n - 1
        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(w, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]
    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)
    b_new = _full_convolution_fft(a_proj)
    max_new = b_new.max()
    return a_proj, b_new, max_new


def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)
    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)          # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Hybrid optimiser (soft‑max, sub‑gradient, mass‑swap, jitter)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    eta = 0.7
    eta_decay = 0.9997

    temperature = cur_max * 0.12
    temp_decay = 0.9996

    jitter_prob = 0.015
    jitter_scale = 0.015

    use_softmax = True
    tau = cur_max
    tau_decay = 0.9995

    while time.time() < deadline:
        r = rng.random()
        if use_softmax and r < 0.55:
            a, b, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
            tau *= tau_decay
        elif r < 0.80:
            g, cur_max = _subgradient_max(a)
            improved = False
            for factor in (0.5, 1.0, 1.5):
                a_cand = a - eta * factor * g
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max - 1e-9:
                    a = a_cand
                    cur_max = max_cand
                    improved = True
                    break
            if not improved:
                a_cand = a - eta * g
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                    a = a_cand
                    cur_max = max_cand
                temperature *= temp_decay
        elif r < 0.95:
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = 0.04 * a[i]
                    a_cand = a.copy()
                    a_cand[i] -= delta
                    a_cand[j] += delta
                    max_cand = _full_convolution_fft(a_cand).max()
                    if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                        a = a_cand
                        cur_max = max_cand
                        temperature *= temp_decay
        else:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a_cand = np.clip(a + noise, 0.0, None)
            a_cand = _project_to_simplex(a_cand, S)
            max_cand = _full_convolution_fft(a_cand).max()
            if max_cand < cur_max:
                a = a_cand
                cur_max = max_cand

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        eta *= eta_decay
        if rng.random() < 0.01:
            use_softmax = not use_softmax

    return best_a, best_max

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step sequence that minimises the
    black‑box evaluate_sequence().  The routine first runs the LP‑based
    scheme (fast descent) and then a gradient‑based local optimiser for
    fine polishing.  Multiple random restarts with different lengths are
    performed until the time budget expires.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load a previously stored construction, if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – polish the existing construction (if any)
    # ------------------------------------------------------------------
    if base_seq is not None and base_seq.size > 0:
        n0 = int(base_seq.shape[0])
        a0 = _project_to_simplex(base_seq, np.sqrt(2.0 * n0))
        phase_deadline = min(time.time() + 120.0, end_time)   # up to 2 min
        seq_opt, score_opt = _lp_step_optimize(a0, phase_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[phase‑1‑LP] length={n0} score={best_score:.6f}")

        # optional local polishing
        polish_deadline = min(time.time() + 30.0, end_time)
        seq_opt2, score_opt2 = _local_optimize(best_seq, polish_deadline, rng)
        if score_opt2 < best_score:
            best_score = score_opt2
            best_seq = seq_opt2.copy()
            print(f"[phase‑1‑local] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many independent restarts (LP + occasional local refinement)
    # ------------------------------------------------------------------
    while time.time() < end_time - 20.0:       # keep ~20 s for final polishing
        attempt += 1

        # Choose a problem size
        if base_seq is not None and rng.random() < 0.6:
            # Vary around the known length
            n = int(base_seq.shape[0] * (0.7 + 0.6 * rng.random()))
            n = max(400, min(n, 3000))
        else:
            n = rng.integers(800, 3000)

        # Initialise a feasible vector (ℓ₁‑norm = √(2·n))
        if base_seq is not None:
            a0 = _scale_sequence(base_seq, n).astype(float)
            a0 += rng.normal(scale=max(a0.mean(), 1e-12) * 0.001, size=n)
            a0 = np.clip(a0, 0.0, None)
        else:
            a0 = rng.random(n)
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n))

        # LP optimisation for this restart
        slice_deadline = min(time.time() + 12.0, end_time - 20.0)
        seq_opt, score_opt = _lp_step_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[global‑LP] attempt={attempt} n={n} new best score={best_score:.6f}")

        # Occasionally run a short local refinement on the current best
        if rng.random() < 0.15:
            polish_deadline = min(time.time() + 6.0, end_time - 20.0)
            seq_opt2, score_opt2 = _local_optimize(best_seq, polish_deadline, rng)
            if score_opt2 < best_score:
                best_score = score_opt2
                best_seq = seq_opt2.copy()
                print(f"[extra‑local] after attempt={attempt} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_deadline = end_time
        seq_opt, score_opt = _local_optimize(best_seq, final_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[final‑local] polished score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as a Python list) and store it globally
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] external score = {external_score:.6f} "
              f"internal max = {best_score:.6f} length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
