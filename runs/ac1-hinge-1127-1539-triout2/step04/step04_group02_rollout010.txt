"""
Hybrid search for a low‑peak self‑convolution step function.

The algorithm proceeds in three stages:

1. **Diverse initialisation** – a fairly wide parametric sweep over several
   families of monotone and “U‑shaped’’ sequences (including aggressively
   biased shapes with clipping) and a handful of random Dirichlet vectors.
   The best candidate from this cheap exploration seeds the next stage.

2. **Simulated‑annealing hill climbing** – the current best vector is refined
   by repeatedly transferring a small amount of mass from a random index i to
   a random index j.  Moves that improve the peak of the self‑convolution are
   always accepted; worsening moves are accepted with a temperature‑dependent
   probability.  The temperature and the transfer scale decay linearly with
   elapsed time, while occasional jitter steps help escape plateaux.

3. **Polishing** – a short burst of a smoothed gradient‑descent phase (identical
   to the original LP‑inspired surrogate) cleans up any remaining roughness.

The routine respects the 1000 s budget, prints occasional status updates and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of `a` with itself using an FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_shape(
    n: int,
    target_sum: float,
    kind: str,
    param: float,
) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence of length n.
    `kind` may be one of:
        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right, exponential_left, exponential_right.
    The parameter `param` controls the shape (exponent, rate, etc.).
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    `evaluate_sequence` objective.  The routine respects the 1000 s time
    budget and returns the best vector discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.25

    # --------------------------------------------------------------
    # Problem size (use existing construction if present)
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")

        n = len(base)
        init_seq = np.asarray(base, dtype=np.float64)
        init_seq = project_onto_simplex(init_seq, math.sqrt(2.0 * n))
    else:
        n = 16384                     # default length (power‑of‑two)
        init_seq = None

    target_sum = math.sqrt(2.0 * n)   # Σa_i = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – broad parametric sweep + random seeds
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)
    best_score = compute_max_conv(best_seq)

    if init_seq is not None:
        sc = compute_max_conv(init_seq)
        if sc < best_score:
            best_seq = init_seq.copy()
            best_score = sc
            print(f"Baseline improved start → {best_score:.6f}")

    shape_kinds = (
        "linear",
        "quadratic",
        "exponential",
        "power",
        "inverse",
        "linear_left",
        "linear_right",
        "exponential_left",
        "exponential_right",
    )
    # expand parameter ranges, especially near the clipping regime
    param_grids = {

        "exponential": np.linspace(0.0, 6.0, 31),

    print(f"[{time.time():.2f}] Phase 0 – exhaustive sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            sc = compute_max_conv(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # a few random Dirichlet‑like vectors (beta=0.5 gives moderate sparsity)
    for _ in range(30):
        cand = np.random.gamma(shape=0.5, scale=1.0, size=n)
        cand = project_onto_simplex(cand, target_sum)
        sc = compute_max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

                best_seq = a.copy()
                print(f"Iter {iter_cnt:09d} – jitter improved to {best_score:.6f}")

    # --------------------------------------------------------------

    a = best_seq.copy()

    eta = 0.4 * (target_sum / n)
    beta = 0.85
    tau_factor = 0.12
    polish_deadline = start_time + budget_s * 0.97

    while time.time() < polish_deadline:

        max_conv = conv.max()
        tau = max(tau_factor * max_conv, 1e-12)

        # log‑sum‑exp weights
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)

        momentum = beta * momentum - eta * grad
        a = a + momentum
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        max_conv = compute_max_conv(a)
        if max_conv + 1e-12 < best_score:

            best_seq = a.copy()
            print(f"Polish iteration – improved to {best_score:.6f}")

        # decay schedule
        eta *= 0.995
        tau_factor = max(0.005, tau_factor * 0.995)

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
