"""
search for a non‑negative step function that minimises
evaluate_sequence().  The algorithm consists of three stages:

1.  **Diversified warm‑starts** – a collection of parametric families
    (exponential, power‑law, random) is sampled for many lengths.
    Each candidate is projected onto the ℓ1‑simplex Σa = √(2·n).

2.  **Projected sub‑gradient descent on the true max‑norm** – at each
    iteration the index `k_max` where the convolution attains its
    maximum is identified.  The sub‑gradient of this entry is
    `g_i = 2·a_{k_max‑i}` (zero when the index is out of range).  A
    diminishing step size `η_t = η₀/√t` is used, after which the vector
    is projected back onto the simplex.  This drives the peak of the
    convolution down directly, without any smoothing tricks.

3.  **Exploratory mass‑swap moves** – occasional random transfers of
    mass between two coordinates (Metropolis acceptance) help the
    procedure escape plateaus.

The whole search runs for the allotted budget (≈ 1000 s) and keeps the
best sequence seen.  The final sequence is returned as a plain list of
floats; its quality is printed together with the external evaluator
value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("target simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # decreasing sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFTs (length 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Parametric families for warm‑starts
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Geometric sequence a_i ∝ exp(-lam·i) projected onto the required simplex."""
    if lam <= 0:
        # fallback to uniform
        a = np.full(n, 1.0 / n)
    else:
        a = np.exp(-lam * np.arange(n))
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    return a


def _power_sequence(n: int, exponent: float, beta: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+beta)^{-exponent} (beta>0 avoids division by zero)."""
    idx = np.arange(n, dtype=float) + beta
    a = idx ** (-exponent)
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    return a


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex Σa = √(2·n) (Dirichlet with α=1)."""
    raw = rng.random(n)
    a = raw / raw.sum()
    a = a * np.sqrt(2.0 * n)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         eta0: float = 0.8,
                         swap_prob: float = 0.12,
                         swap_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly minimises the peak of
    the convolution a*a.  Returns the best vector seen and its max‐conv
    value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # ---- current max entry and its sub‑gradient ----
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        cur_max = conv[k_max]

        # compute sub‑gradient g_i = 2 * a_{k_max - i}   (if in range)
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)          # inclusive
        inds = np.arange(i_start, i_end + 1, dtype=int)
        grad[inds] = 2.0 * a[k_max - inds]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a, S)

        # evaluate after projection
        conv = _full_convolution(a)
        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # ---- occasional random mass‑swap (exploration) ----
        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis acceptance
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- early stop if stagnated ----
        if it - last_improve > 8000:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345,

    """
    Search for a non‑negative step sequence with the smallest possible
    value returned by evaluate_sequence().  The routine combines

    found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lengths = [800, 1200, 2000, 3000, 4000, 6000, 8000, 10000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # (a) exponential families – λ = c / n with c sampled
        for c in np.linspace(0.4, 3.0, 12):
            lam = c / n
            a = _exponential_sequence(n, lam)
            cur = np.max(_full_convolution(a))
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()

    # ------------------------------------------------------------------
