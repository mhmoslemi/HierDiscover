"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The search consists of three stages:

1. **Deterministic parametric sweep** – we generate power–law sequences
   a_i = (i+shift)^(-p) (and a symmetric variant) for a grid of
   (n, p, shift) and keep the few best candidates.

2. **Projected gradient descent** – each elite candidate is refined by
   the smooth log‑sum‑exp surrogate of the maximal convolution entry.
   The optimiser uses momentum and projects back onto the ℓ¹‑simplex
   Σa_i = √(2 n).

3. **Simulated annealing** – a short SA phase tweaks the refined vectors
   with random mass‑transfer moves and a multiplicative cooling schedule.

All phases respect a user‑supplied time budget (default 1000 s) and never
perform any I/O.  The final result is returned as a plain Python list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4 / (beta + 1.0),
                               size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # exponential annealing of β
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA with multiplicative cooling."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature initialised from current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move up to 15 % of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law / symmetric power‑law)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    a_i = (i + shift)^{-p},   i = 0,…,n‑1 .
    Returned vector is projected onto the ℓ¹‑simplex Σa = √(2 n).
    """
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    # final safeguard – tiny negatives due to numerical noise
    a = np.maximum(a, 0.0)
    return a


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    Symmetric version: a_i = (i+shift)^{-p} + (n-1-i+shift)^{-p}.
    Afterwards projected onto the ℓ¹‑simplex.
    """
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    ``evaluate_sequence`` objective.  The routine respects the allotted
    time budget, explores a parametric grid, refines promising candidates
    with gradient descent + SA, and finally returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5  # keep a small safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 2000, 2500, 3000, 3500, 4000]
    p_grid = np.linspace(0.45, 0.80, 15)          # exponent grid
    shift_grid = np.linspace(0.5, 5.0, 10)        # shift grid

    # keep a short list of elite seeds for later refinement
    elite = []  # list of (score, array)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        for p in p_grid:
            for shift in shift_grid:
                # plain power‑law
                a = _powerlaw_sequence(n, p, shift)
                score = _max_convolution_fft(a)
                if score < best_obj:
                    best_obj = score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best (grid) n={n} p={p:.3f} "
                          f"shift={shift:.2f}  obj={best_obj:.6f}")

                # symmetric version
                a2 = _symmetric_powerlaw_sequence(n, p, shift)
                score2 = _max_convolution_fft(a2)
                if score2 < best_obj:
                    best_obj = score2
                    best_seq = a2.copy()
                    print(f"[{time.time():.1f}] New global best (sym) n={n} p={p:.3f} "
                          f"shift={shift:.2f}  obj={best_obj:.6f}")

                # keep the top‑5 candidates (by objective) for refinement
                if len(elite) < 5 or score < elite[-1][0]:
                    elite.append((score, a))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:5]
                if len(elite) < 5 or score2 < elite[-1][0]:
                    elite.append((score2, a2))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:5]

        # optional short random sampling for this length (adds diversity)
        rnd_end = min(deadline, time.time() + 2.0)
        while time.time() < rnd_end:
            a_rand = rng.random(n)
            a_rand = a_rand / a_rand.sum() * np.sqrt(2.0 * n)
            score = _max_convolution_fft(a_rand)
            if score < best_obj:
                best_obj = score
                best_seq = a_rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n}  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        sub_dead = min(deadline, time.time() + 30.0)   # up to 30 s per candidate
        a_gd = _gradient_descent_opt(seed_seq, sub_dead)

        # use the remaining time for SA (no separate deadline needed,
        # the function itself respects the global deadline)
        a_sa = _simulated_annealing(a_gd, deadline)

        obj = _max_convolution_fft(a_sa)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n}  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity check (optional)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim negligible trailing entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
