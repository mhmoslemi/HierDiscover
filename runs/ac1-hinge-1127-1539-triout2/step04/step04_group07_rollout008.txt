def propose_candidate(seed=12345, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = np.inf

    # Precompute lengths list
    lengths = list(np.arange(2000, 16001, 2000))  # 2000,4000,...,16000

    # Stage 1: random shape sampling
    while time.time() < deadline and attempts < max_attempts:
        n = int(rng.integers(1500, 20001))  # random n
        shape_type = rng.choice(['exp','pow','stretched','combo'])
        if shape_type == 'exp':
            lam = rng.uniform(0.0005, 0.01)  # lam per step
            a = np.exp(-lam * np.arange(n))
        elif shape_type == 'pow':
            p = rng.uniform(0.0, 2.0)
            a = (np.arange(n)+1) ** (-p)
        elif shape_type == 'stretched':
            lam = rng.uniform(0.0005, 0.01)
            c = rng.uniform(1.0, 2.5)
            a = np.exp(-lam * (np.arange(n) ** c))
        else:  # combo
            lam = rng.uniform(0.0005, 0.01)
            p = rng.uniform(0.0, 2.0)
            a = np.exp(-lam * np.arange(n)) * (np.arange(n)+1) ** (-p)

        a = _project_simplex(a, np.sqrt(2.0 * n))

        # compute max_b quickly
        conv = _full_convolution(a)
        val = conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[sample] n={n}, type={shape_type}, val={val:.6f}")

        # Occasionally run local refinement on this a
        if rng.random() < 0.1: # 10% chance
            slice_deadline = min(deadline, time.time() + 10.0)
            a_opt, val_opt = _local_optimize(a, slice_deadline, rng)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[refine] n={n}, val={val_opt:.6f}")

    # Stage 2: improve best_seq via longer gradient descent
    if best_seq is not None:
        n = len(best_seq)
        # Run a dedicated refinement for remaining time
        while time.time() < deadline:
            slice_deadline = min(deadline, time.time() + 30.0)
            a_opt, val_opt = _local_optimize(best_seq, slice_deadline, rng)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[global] improved to {best_val:.6f}")

    # At end, return best_seq as list
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0*n)/n)

    final_seq = best_seq.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluate = {final_val:.6f} len={len(final_seq)}")

def _local_optimize_v2(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0*n)
    best_a = a.copy()
    conv = _full_convolution(a)
    best_score = conv.max()
    lr = 1.0
    lr_decay = 0.9995
    alpha = 1.0
    alpha_mul = 1.01
    swap_prob = 0.2
    swap_frac = 0.05
    target_prob = 0.3
    eps_target = 0.05
    last_improve = 0
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = _full_convolution(a)
        # smooth max weights
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)  # gradient of smoothed max
        a = a - lr * grad
        a = _project_simplex(a, S)
        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it
        # random swap etc similar to before
        if rng.random() < swap_prob:
            i = rng.integers(0, n); j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max-cur_max)/(cur_max*0.02+1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = a.copy()
                        last_improve = it
        if rng.random() < target_prob:
            # same targeted move as before: find largest contribution for max index
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]
            src_idx = np.argmax(contributions)
            i_src = i_range[src_idx]
            tgt_idx = np.argmin(contributions)
            i_tgt = i_range[tgt_idx]
            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max-cur_max)/(cur_max*0.02+1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it
        # schedule updates
        lr *= lr_decay
        alpha *= alpha_mul
        if it - last_improve > 10000:
            break
    return best_a, best_score

"""
Propose‑candidate optimizer for the max‑convolution objective.

The algorithm consists of three phases:

1.  Random shape sampling – generates a large variety of non‑negative
    sequences (exponential, power‑law, stretched exponential and mixtures
    thereof) for random lengths n.  Each raw vector is projected onto the
    ℓ1‑simplex Σa = √(2·n) so that evaluate_sequence() returns the exact
    maximal convolution entry.

2.  Fast local refinement – a projected gradient descent on a soft‑max
    approximation of the max‑norm together with a few cheap “mass‑swap”
    moves.  The temperature (α) is annealed upwards so that the smooth
    objective converges to the true maximum.

3.  Global polishing – the best sequence found so far is repeatedly
    re‑optimised until the time budget expires.

All helper functions are defined at top level; the entry point is
`propose_candidate`, which returns a list of floats ready for the external
evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max objective.
    Computes g_j = 2 Σ_i w_{i+j}·a_i  (indices outside 0..2n‑2 are ignored).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def _generate_raw_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an unnormalised non‑negative vector of length n.
    The distribution family is chosen at random.
    """
    i = np.arange(n, dtype=float)
    shape_type = rng.choice(["exp", "pow", "stretched", "mix"])
    if shape_type == "exp":
        # a_i = exp(-lam * i)
        lam = rng.uniform(5e-4, 2e-2)          # decay per index
        a = np.exp(-lam * i)
    elif shape_type == "pow":
        # a_i = (i+1)^{-p}
        p = rng.uniform(0.0, 2.5)
        a = (i + 1.0) ** (-p)
    elif shape_type == "stretched":
        # a_i = exp(-lam * i**c)
        lam = rng.uniform(5e-4, 2e-2)
        c = rng.uniform(1.0, 2.5)
        a = np.exp(-lam * (i ** c))
    else:  # "mix"
        lam1 = rng.uniform(5e-4, 2e-2)
        lam2 = rng.uniform(5e-4, 2e-2)
        w = rng.uniform(0.0, 1.0)
        a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    return a


# ----------------------------------------------------------------------
# Local optimiser (projected gradient + cheap moves)
# ----------------------------------------------------------------------


def _local_optimize_v2(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine `a0` with a smoothed‑max gradient descent and occasional
    random/targeted mass‑swap moves.
    Returns the best sequence seen and its (exact) maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # best seen during this run
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (tuned for the larger time‑budget)
    lr = 1.0               # learning rate
    lr_decay = 0.9995
    alpha = 1.0            # temperature of the soft‑max
    alpha_mul = 1.01       # annealing speed
    swap_prob = 0.2        # probability of a random swap each iteration
    swap_frac = 0.06       # fraction of source mass moved in a random swap
    target_prob = 0.3      # chance to try the cheap targeted move
    eps_target = 0.07      # mass fraction used in the targeted move

    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- gradient step on the smooth max ----
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)
        a = a - lr * grad
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # ---- occasional random mass‑swap (exploration) ----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis style acceptance
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- cheap targeted move aimed at the current maximal index ----
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            src_idx = np.argmax(contributions)
            i_src = i_range[src_idx]

            tgt_idx = np.argmin(contributions)
            i_tgt = i_range[tgt_idx]

            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- schedule updates ----
        lr *= lr_decay
        alpha *= alpha_mul

        # early stop if no improvement for a long time
        if it - last_improve > 8000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises evaluate_sequence().
    The routine samples many random shapes, projects them onto the
    ℓ1‑simplex, retains the best, and repeatedly refines it with a
    projected‑gradient optimiser.  The best sequence found before the
    time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.5   # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 0) keep a previous solution if available (height_sequence_1)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                n = prev.size
                S = np.sqrt(2.0 * n)
                prev = _project_simplex(prev, S)
                val = np.max(_full_convolution(prev))
                if val < best_val:
                    best_val = val
                    best_seq = prev.copy()
                    print(f"[prev] loaded height_sequence_1, n={n}, val={val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) random shape sampling phase (fast)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length – longer lengths give a lower theoretical floor
        n = int(rng.integers(800, 20001))            # 800 … 20000
        raw = _generate_raw_shape(n, rng)
        a = _project_simplex(raw, np.sqrt(2.0 * n))
        val = np.max(_full_convolution(a))

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[sample] n={n:5d} val={val:.6f}")

        # occasionally refine a promising candidate
        if rng.random() < 0.08:
            slice_deadline = min(deadline, time.time() + 12.0)
            a_opt, val_opt = _local_optimize_v2(a, slice_deadline, rng)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[refine] n={n:5d} improved to {val_opt:.6f}")

        # keep sampling while we still have time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) global polishing – re‑optimise the current best until out of time
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.time() < deadline:
            slice_deadline = min(deadline, time.time() + 30.0)
            a_opt, val_opt = _local_optimize_v2(best_seq, slice_deadline, rng)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[global] new best = {val_opt:.6f} (len={len(best_seq)})")
            else:
                # no improvement – break early if we are close to the deadline
                if time.time() + 5.0 >= deadline:
                    break

    # ------------------------------------------------------------------
    # 3) fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
