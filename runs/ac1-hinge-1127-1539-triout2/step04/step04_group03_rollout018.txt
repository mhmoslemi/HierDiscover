def _lp_optimize(f, M, sum_target):
    n = len(f)
    # Build A_ub as sparse matrix (2n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # indices j where g_j appears: j = i, f_{k - i} corresponds.
        # Actually A_{k,j} = f_{k - j} if 0<=k - j< n -> j in [k-(n-1), k]
        j_start = max(0, k-(n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Objective: maximize sum(g) -> minimize -1^T g
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    g0 = res.x
    # Scale to sum_target
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return None
    scale = sum_target / sum_g0
    return g0 * scale
```

"""
Simple yet powerful optimisation for the “max‑convolution / ℓ₁²” problem.

The core idea is the classic LP‑based improvement described in the
statement: for a current step‑function f with
M = max(f∗f) we solve

    maximise   Σ g_i
    subject to (f∗g)_k ≤ M   (for every k)
               g_i ≥ 0

The solution g has a larger ℓ₁‑mass than f while still respecting the
same upper bound M on the convolution with f.  After normalising g to
the required ℓ₁‑norm we replace f by a small convex combination
(1−t) f + t g (t≈0.1).  A theorem guarantees that the new max‑convolution
does not increase, and in practice it strictly decreases.
We start from a few random shapes, optionally polish them with a
projected‑gradient step on a smooth surrogate of the max‑operator,
and then run the LP‑based refinement until the time budget expires.

The implementation uses only NumPy, SciPy (linprog with the HiGHS
solver) and the already‑provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full, length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Duchi et al. 2008 projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (a few families for diversity)
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:  # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:  # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:  # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:  # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif kind == 4:  # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:  # uniform plus tiny jitter
        a = np.full(n, target / n)

    # small Gaussian jitter and re‑project
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns the gradient of  (1/β)·log Σ exp(β·b_k)  w.r.t. a,
    where b = a∗a (full convolution).  The gradient is
        g = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a  (full convolution)
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities on the convolution entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentials
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]         # central n entries
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed surrogate
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp surrogate.
    Stops when the provided time budget expires.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = 0.5 * target_sum / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # project back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (slow exponential schedule)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step” of the paper)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """
    One LP improvement step:
        maximise Σ g_i
        subject to (f∗g)_k ≤ M   for all k
                   g_i ≥ 0
    The returned vector is rescaled to have ℓ¹‑norm = target_sum.
    """
    n = f.size

    # Build the convolution matrix A where A[k, j] = f_{k-j} (0 if out‑of‑range)
    # A can be constructed efficiently with scipy.sparse.diags.
    try:
        from scipy import sparse
        from scipy.optimize import linprog
    except Exception as e:
        raise RuntimeError("SciPy is required for the LP step.") from e

    offsets = np.arange(n)                 # 0 … n‑1
    # data matrix: each row of the diagonal data is the vector f
    data = np.tile(f, (n, 1))              # shape (n, n)
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

    c = -np.ones(n)                        # maximise Σg  → minimise −Σg
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12))  # slight tolerance
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})

    if not res.success:
        # fall back to the current vector (no improvement possible)
        return None

    g0 = res.x
    s0 = g0.sum()
    if s0 <= 0:
        return None
    scale = target_sum / s0
    return g0 * scale


# ----------------------------------------------------------------------
# Mixing step – try a few values of t and keep the best improvement
# ----------------------------------------------------------------------
def _mix_with_g(f: np.ndarray, g: np.ndarray, M_current: float):
    """Return (f_new, M_new) after a small convex combination with g."""
    best_f = f
    best_M = M_current
    # try several mixing factors; keep t small to guarantee descent
    for t in (0.05, 0.1, 0.15, 0.2):
        cand = (1.0 - t) * f + t * g
        M_cand = _max_convolution_fft(cand)
        if M_cand < best_M - 1e-8:
            best_f, best_M = cand, M_cand
    return best_f, best_M

                print(f"[{time.time():.1f}] NEW GLOBAL BEST (n={n})  val={best_val:.6f}")

    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = (np.full(n, np.sqrt(2.0 * n) / n)).tolist()
        best_val = evaluate_sequence(best_seq)

    # Trim trailing near‑zero entries for neatness
    arr = np.asarray(best_seq)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        best_seq = arr[: nz[-1] + 1].tolist()

    # Final sanity check (optional)
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] Final evaluate_sequence = {final_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best value = {best_val:.6f} "
          f"length = {len(best_seq)}")
    return best_seq
