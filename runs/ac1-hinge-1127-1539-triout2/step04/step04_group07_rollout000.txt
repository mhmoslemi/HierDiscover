def _local_optimize_with_lp(a0, deadline, rng, lp_interval=150, t_lp=0.2):

    a = a0.copy()

    n = a.shape[0]; S = np.sqrt(2*n)

    best_score = np.max(_full_convolution(a))

    best_a = a.copy()

    lr = 0.5; lr_decay = 0.9999; alpha = 0.5; alpha_mul = 1.0015

    swap_prob = 0.15; swap_frac = 0.04; target_prob = 0.2; eps_target = 0.03

    iteration = 0; last_improve =0

    while time.time() < deadline:

        iteration +=1

        # gradient step
        a, cur_max = _gradient_step(a, lr, alpha)

        if cur_max < best_score:
            best_score = cur_max; best_a = a.copy(); last_improve = iteration

        # random swap
        if rng.random()<swap_prob:
            i = rng.integers(0,n); j = rng.integers(0,n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(- (cand_max - cur_max) / max(cur_max*0.02, 1e-12)):
                    a = cand; cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max; best_a = a.copy(); last_improve = iteration

        # targeted move
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n-1))
            i_end = min(k_max, n-1)
            i_range = np.arange(i_start, i_end+1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]
            src_idx = np.argmax(contributions)
            i_src = i_range[src_idx]
            tgt_idx = np.argmin(contributions)
            i_tgt = i_range[tgt_idx]
            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.
Algorithm
---------

* **Warm‑starts** – generate a variety of initial candidates:
  - exponential decays,
  - Dirichlet (Gamma) samples with a sparse shape.
  All candidates are projected onto the ℓ¹‑simplex
  ``Σ a_i = √(2·n)`` (the normalisation used by the evaluator).

* **Local optimiser** – a projected gradient descent on a smooth
  approximation of the max‑norm of the autocorrelation, together with
  cheap random‐swap and targeted mass‑transfer moves (the same ideas as
  the reference code, but with tuned hyper‑parameters).

* **Linear‑programming boost** – every few hundred gradient steps we solve
  a linear program

      maximise   Σ g_i
      subject to f * g ≤ M          (elementwise)
                 g_i ≥ 0

  where ``f`` is the current sequence, ``M`` its present maximal
  convolution value and ``*`` denotes the (linear) convolution.
  The LP pushes mass into positions where ``f`` is small, thus
  potentially lowering the maximal autocorrelation when we mix the
  resulting ``g`` back into ``f``.

* **Iterated search** – keep the best sequence found so far, repeatedly
  restart from it (with jitter) or from a fresh random candidate,
  run the optimiser for a time slice (≈30 s or the remaining budget),
  and accept any improvement.  The whole process respects the 1000 s
  budget.

The routine returns the best sequence as a plain ``list[float]``.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute gradient of the smoothed max:
        g_j = 2 Σ_i w_{i+j}·a_i .
    ``a`` has length n, ``w`` has length 2·n‑1 (weights that sum to 1).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3·n‑2
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights from a convolution vector."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the smoothed max‑norm.
    Returns the updated sequence and the *exact* maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Warm‑starts
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """Decreasing exponential of length n with decay controlled by k_factor."""
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)          # = 1 - q**n
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)
    return a


def _dirichlet_sequence(n: int, rng: np.random.Generator, shape: float = 0.3) -> np.ndarray:
    """Sparse Dirichlet‐like sample (Gamma with shape < 1)."""
    # Gamma with shape <1 gives heavy‑tailed, sparse vectors
    x = rng.gamma(shape, 1.0, size=n)
    a = _project_simplex(x, np.sqrt(2.0 * n))
    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement
# ----------------------------------------------------------------------


def _lp_maximize_sum_under_conv(f: np.ndarray, M: float, rng: np.random.Generator) -> tuple[np.ndarray | None, float]:
    """
    Solve   maximise Σ g_i  subject to  f * g ≤ M , g ≥ 0.
    Returns the optimal g and its sum (or (None,0) on failure).
    """
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, value=f)
    conv = cp.conv(f_param, g)                     # length 2·n‑1
    constraints = [conv <= M]                     # elementwise
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-6, reltol=1e-6)
    except Exception:  # pragma: no cover
        return None, 0.0
    if g.value is None:
        return None, 0.0
    g_opt = np.maximum(g.value, 0.0)
    sum_g = float(g_opt.sum())
    return g_opt, sum_g


# ----------------------------------------------------------------------
# Local optimiser (gradient + swaps + LP boost)
# ----------------------------------------------------------------------


def _local_optimize_with_lp(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            lp_interval: int = 200,
                            t_lp: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using:
    * projected gradient steps on a smooth approximation of max ‖a*a‖∞,
    * random mass‑swap moves,
    * cheap targeted moves that reduce the current maximal convolution entry,
    * and an occasional linear‑programming boost (see `_lp_maximize_sum_under_conv`).

    Returns the best sequence encountered and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (empirically tuned)
    lr = 0.5
    lr_decay = 0.9999
    alpha = 0.5
    alpha_mul = 1.003
    swap_prob = 0.18
    swap_frac = 0.05
    target_prob = 0.22
    eps_target = 0.035

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # ---- gradient descent step ----
        a, cur_max = _gradient_step(a, lr, alpha)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # ---- random mass‑swap (explorative) ----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- cheap targeted move aimed at the current max index ----
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            src_idx = np.argmax(contributions)
            i_src = i_range[src_idx]
            tgt_idx = np.argmin(contributions)
            i_tgt = i_range[tgt_idx]

            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- linear‑programming boost ----
        if it % lp_interval == 0:
            M = cur_max
            g_opt, sum_g = _lp_maximize_sum_under_conv(a, M, rng)
            if g_opt is not None and sum_g > 0:
                # scale to keep the norm √(2n)
                scale = S / sum_g
                g_scaled = g_opt * scale
                # convex combination (small step towards g)
                cand = (1.0 - t_lp) * a + t_lp * g_scaled
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- schedule updates ----
        lr *= lr_decay
        alpha *= alpha_mul

        # early stop if stagnated for a long time
        if it - last_improve > 5000:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0) If a previous solution exists, use it as a starting point
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                n = prev.size
                S = np.sqrt(2.0 * n)
                prev = _project_simplex(prev, S)
                cur = np.max(_full_convolution(prev))
                best_seq = prev.copy()
                best_score = cur
                print(f"[prev] loaded height_sequence_1, length={n}, score={cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) quick warm‑starts (exponential & Dirichlet)
    # ------------------------------------------------------------------
    for n in (800, 1200, 2000, 3000, 4000):
        if time.time() >= deadline:
            break
        for k_factor in np.linspace(0.6, 2.5, 9):
            a = _exponential_sequence(n, k_factor)
            cur = np.max(_full_convolution(a))
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[warm‑exp] n={n:4d} k={k_factor:4.2f} score={cur:.6f}")

        # a few Dirichlet samples (sparse)
        for _ in range(3):
            a = _dirichlet_sequence(n, rng, shape=0.4)
            cur = np.max(_full_convolution(a))
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[warm‑dir] n={n:4d} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2) main optimisation loop – run time‑sliced local searches
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length: usually stay around current best length,

        if best_seq is not None and rng.random() < 0.6:

            else:

        slice_deadline = min(deadline, time.time() + 30.0)

        a_opt, score_opt = _local_optimize_with_lp(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global] new best n={n:4d} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final sanity check via the external evaluator
    # ------------------------------------------------------------------
    if best_seq is None:

        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
