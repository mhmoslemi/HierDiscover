"""
Propose a non‑negative step‑function (vector of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation function is homogeneous, therefore we fix the ℓ₁‑norm
to Σa_i = √(2 n).  The algorithm consists of three stages:

1.  **Broad initial sweep** – generate many parametric “U‑shaped”,
    cosine‑valley, Gaussian‑valley and random Dirichlet vectors,
    project each onto the simplex and keep the best (smallest maximal
    self‑convolution).

2.  **Smoothed Adam descent** – minimise a log‑sum‑exp (soft‑max)
    surrogate of the maximal convolution.  The gradient is obtained
    by FFT‑based convolution, Adam‑style moments are kept, and after
    each step the iterate is projected back onto the simplex.
    The temperature τ = τ_factor·max_conv is reduced geometrically,
    and occasional random pair‑transfers and jitter are applied to
    escape shallow plateaus.

3.  **Sub‑gradient refinement + polishing** – switch to the true
    non‑smooth objective (the maximum), use an averaged sub‑gradient
    of the (approximately) maximal lags and a decaying step size,
    interleaved with cheap random pair‑transfers.  A final short
    hill‑climbing phase tries tiny random moves and accepts any
    improvement.

The routine respects the 1000 s time budget, prints progress, and
returns the best vector found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float,
                   kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or valley sequence of length n.
    `kind` may be one of:
        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right, exponential_left, exponential_right,
        cosine, gaussian.
    `param` controls the shape (exponent, rate, amplitude, sigma, …).
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    elif kind == "cosine":
        # valley in the centre: raw = 1 + param * cos(pi * i/(n-1))
        raw = 1.0 + param * np.cos(np.pi * idx / (n - 1))
    elif kind == "gaussian":
        # Gaussian valley centred in the middle; param is sigma as a *fraction* of n
        sigma = max(param, 1e-6) * n
        raw = np.exp(-((idx - mid) ** 2) / (2 * sigma ** 2))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximum value of the self‑convolution of a (a≥0)."""
    return float(conv_self_fft(a).max())

        grad = (2.0 / cnt) * grad_full[n - 1 : n - 1 + n]

        a = a - step_size * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Sub‑grad iter {sub_iter:06d} – improved to {best_score:.6f}")

        if sub_iter % 500 == 0:
            step_size *= 0.94

        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    # --------------------------------------------------------------
    # Phase 3 – final polishing (tiny random pair moves)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – polishing")
    while time.time() < deadline:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.0025)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
