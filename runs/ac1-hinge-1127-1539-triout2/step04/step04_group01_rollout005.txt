"""
LP‑guided step‑function optimizer.

The algorithm alternates between:

1.  Solving a linear program that, for the current shape `f`,
    finds a direction `g` with maximal ℓ¹‑mass under the
    constraint  `f*g ≤ max(f*f)`.  (All convolutions are linear in the
    unknown `g`.)

2.  Updating `f ← (1‑t)·f + t·g` with a line‑search on `t`
    that ensures a strict decrease of the objective
    `max(conv(f,f))`.

The LP is solved with SciPy’s high‑performance “highs” backend.
When SciPy is unavailable the routine falls back to the
projected‑gradient + simulated‑annealing phase used in the
previous state‑of‑the‑art code.

The search respects a 1000 s wall‑clock budget, starts from the
previous best construction if available, and finishes with a short
gradient‑descent polishing and a simulated‑annealing run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation (used when no previous construction is found)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )
    else:                               # uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Gradient‑descent + simulated‑annealing (fallback / final polishing)
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the smooth max."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.20
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        # ---- smooth‑max gradient ------------------------------------
        n = a.size
        size1 = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(a, n=size1)
        b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        conv_len = 3 * n - 2
        size2 = _next_pow2(conv_len)
        ar_fft = np.fft.rfft(a_rev, n=size2)
        pr_fft = np.fft.rfft(p, n=size2)
        conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # ---- momentum step -----------------------------------------
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Simple SA that transfers a random fraction of mass between two indices."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < math.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Helpers for the LP phase
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that
        A @ g = conv(f, g)   (linear convolution, no padding).
    The entry A[k, i] = f[i] if 0 ≤ k‑i < n, else 0.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=f.dtype)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _interpolate_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a vector to a new length."""
    old_n = v.size
    if old_n == new_n:
        return v.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, v)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine first attempts a
    linear‑programming driven improvement loop (if SciPy is
    available) and finally polishes the best candidate with a
    projected‑gradient descent followed by simulated annealing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose an initial length.  2000 works well inside the time budget.
    # ------------------------------------------------------------------
    n = 2000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise from the previous construction if possible.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        # pad / truncate to the desired length
        if a0.size < n:
            a0 = np.concatenate([a0, np.zeros(n - a0.size)])
        elif a0.size > n:
            a0 = a0[:n]
        a0 = _project_to_simplex(a0, target_sum)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (len={a0.size})")
    else:
        a0 = _random_initial_sequence(n, rng)

    best_seq = a0.copy()
    best_M = _max_convolution_fft(best_seq)
    print(f"[{time.time():.1f}] Initial objective = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Try the LP‑based improvement loop (uses SciPy if available).
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        SCIPY_AVAILABLE = True
    except Exception:  # pragma: no cover
        SCIPY_AVAILABLE = False
        print("[!] SciPy not available – skipping LP phase")

    if SCIPY_AVAILABLE:
        iteration = 0
        while time.time() < deadline - 5.0:
            iteration += 1
            f = best_seq.copy()
            M = _max_convolution_fft(f)

            # Build A (conv matrix) for the current f
            A = _build_conv_matrix(f)

            # Linear program: maximize sum(g)  ⇔  minimize –sum(g)
            c = -np.ones(n)
            b_ub = np.full(A.shape[0], M)
            bounds = [(0.0, None)] * n

            # Solve
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                print(f"[{time.time():.1f}] LP failure at iter {iteration}: {res.message}")
                break

            g = res.x
            sum_g = g.sum()
            if sum_g <= 0.0:
                print(f"[{time.time():.1f}] LP returned zero mass – exit")
                break

            # Rescale g to the correct ℓ¹‑norm
            g = g * (target_sum / sum_g)

            # ----- line‑search on the mixing coefficient t
            improved = False
            for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):
                candidate = (1.0 - t) * f + t * g
                candidate = np.maximum(candidate, 0.0)
                candidate = _project_to_simplex(candidate, target_sum)
                M_new = _max_convolution_fft(candidate)
                if M_new < best_M - 1e-9:
                    best_M = M_new
                    best_seq = candidate.copy()
                    print(
                        f"[{time.time():.1f}] LP iter {iteration} t={t:.4f} → M={best_M:.6f}"
                    )
                    improved = True
                    break

            if not improved:
                # No improvement – add a tiny random jitter and continue
                jitter = rng.normal(scale=target_sum * 5e-4, size=n)
                best_seq = best_seq + jitter
                best_seq = np.maximum(best_seq, 0.0)
                best_seq = _project_to_simplex(best_seq, target_sum)
                best_M = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Final refinement: gradient descent + simulated annealing.
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 8.0:
        # allocate ~60 % to GD, 40 % to SA
        gd_deadline = time.time() + time_left * 0.60
        seq_gd = _gradient_descent_opt(best_seq.copy(), gd_deadline)
        M_gd = _max_convolution_fft(seq_gd)
        if M_gd < best_M:
            best_M = M_gd
            best_seq = seq_gd.copy()
            print(f"[{time.time():.1f}] GD improvement → M={best_M:.6f}")

        sa_deadline = time.time() + time_left * 0.40
        seq_sa = _simulated_annealing(best_seq.copy(), sa_deadline)
        M_sa = _max_convolution_fft(seq_sa)
        if M_sa < best_M:
            best_M = M_sa
            best_seq = seq_sa.copy()
            print(f"[{time.time():.1f}] SA improvement → M={best_M:.6f}")

    # ------------------------------------------------------------------
    # Trim insignificant trailing zeros for a tidy output.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the final objective as seen by the official evaluator.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
