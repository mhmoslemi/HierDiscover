def solve_lp(f, M):
    n = len(f)
    m = 2*n - 1
    A_ub = np.zeros((m, n))
    for i in range(n):
        A_ub[i:i+n, i] = f
    b_ub = np.full(m, M, dtype=float)
    bounds = [(0, None)] * n
    c = -np.ones(n) # minimize -sum => maximize sum
    result = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not result.success:
        # fallback maybe return zero vector?
        return np.zeros(n)
    return result.x

def solve_lp(f, M):
    n = len(f)
    m = 2*n - 1
    A = np.zeros((m, n))
    for i in range(n):
        A[i:i+n, i] = f
    b = np.full(m, M)
    bounds = [(0, None)] * n
    c = -np.ones(n)
    # use high performance method = 'highs' (works with SciPy >=1.6)
    result = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        # possibly fallback to simpler solution: keep previous?
        return None
    return result.x

def solve_lp(f_vec, M):
    import numpy as np, scipy.optimize
    n = len(f_vec)
    m = 2*n - 1
    # Build convolution matrix A such that A @ g = f * g at each index k.
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f_vec
    b = np.full(m, M, dtype=np.float64)
    bounds = [(0, None)] * n
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    # Solve using HiGHS
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return res.x

"""
Propose a step‑function sequence that minimises the benchmark objective.

The algorithm combines three ideas:

1. **Quick hyper‑parameter sweep** – try several lengths `n` and power‑law
   exponents `p` to obtain a good starting point (small cost, cheap FFT).

2. **Iterative linear‑programming refinement** – for the current sequence
   `a` we solve a linear program that maximises the ℓ¹‑norm of a new
   sequence `g` under the constraints `a * g ≤ M` (where `M` is the current
   maximal convolution entry).  The solution is re‑scaled to the required
   ℓ¹‑norm and mixed with `a`.  A back‑tracking line search guarantees a
   non‑increase of the maximum convolution.  Repeating this step drives
   the max down substantially.

3. **Simulated annealing (SA) finish** – after the LP phase the remaining
   time is spent on a classic SA that moves mass between coordinates and
   locally smoothes short windows.  This fine‑tunes the sequence.

All operations respect non‑negativity and the ℓ¹‑norm
`Σ a_i = sqrt(2·n)`.  The final list of floats is returned.
"""

import time
import math
import numpy as np

# Optional import – if SciPy is unavailable we fall back to SA only.
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (used by SA)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _solve_lp_max_sum(f_vec: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_i   s.t.   f_vec * g ≤ M  (linear convolution bound),
    g_i ≥ 0.  Returns the optimal g or None on failure.
    """
    if linprog is None:                     # pragma: no cover
        return None

    n = f_vec.shape[0]
    m = 2 * n - 1

    # Build dense convolution matrix A where (A @ g)[k] = (f_vec * g)[k]
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f_vec

    b = np.full(m, M, dtype=np.float64)
    bounds = [(0, None)] * n
    c = -np.ones(n, dtype=np.float64)       # minimise -Σg_i  → maximise Σg_i

    # HiGHS solver (fast and can handle moderate‑size LPs)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return res.x


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 20) -> tuple[np.ndarray, float]:
    """
    Classic SA that moves mass between coordinates or locally smooths a short
    window.  Accepts moves that lower the maximal convolution and otherwise
    with Metropolis probability.  Operates until `deadline` is reached.
    Returns the best sequence seen and its max‑convolution value.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = _max_convolution_fft(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial convolution (kept for the first iteration)
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current active index
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity (weight the contribution of each coordinate to the active entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # ---- propose a move ----
        if rng.random() < 0.70:          # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                            # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # ---- evaluate new state (single FFT) ----
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b_new = conv_new.max()

        # ---- Metropolis acceptance ----
        accept = False
        if max_b_new < cur_score:
            accept = True
        else:
            prob = math.exp(-(max_b_new - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = max_b_new
            conv = conv_new
            if max_b_new < best_score:
                best_score = max_b_new
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved_seg
            # conv stays unchanged

        # occasional jitter to escape a deep basin
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())  # re‑normalise to keep ℓ¹‑norm
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The method consists of:
      * a brief sweep over lengths / power‑law exponents,
      * an LP‑based refinement loop that never increases the max‑convolution,
      * a final simulated‑annealing pass for fine tuning.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Quick hyper‑parameter sweep to obtain a good start.
    # ------------------------------------------------------------------
    best_start = None
    best_M = math.inf
    # try a few lengths and exponents; all evaluations are cheap (FFT)
    for n in (1200, 1500, 1800, 2000):
        target = math.sqrt(2.0 * n)
        for p in (0.35, 0.45, 0.55, 0.65, 0.75):
            a = _power_law_sequence(n, p, target)
            M = _max_convolution_fft(a)
            if M < best_M:
                best_M = M
                best_start = a.copy()
    # If a global construction exists, also consider it
    if 'height_sequence_1' in globals():
        hs = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n = hs.size
        target = math.sqrt(2.0 * n)
        if hs.sum() > 0:
            hs = hs * (target / hs.sum())
        else:
            hs = _power_law_sequence(n, 0.55, target)
        hs = _project_to_simplex(hs, target)
        M = _max_convolution_fft(hs)
        if M < best_M:
            best_M = M
            best_start = hs.copy()

    if best_start is None:                     # pragma: no cover
        raise RuntimeError("Failed to initialise a start sequence.")

    a = best_start
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    print(f"[{time.time():.1f}] initial length={n}, M={best_M:.6f}")

    best_a = a.copy()
    best_M = _max_convolution_fft(best_a)

    # ------------------------------------------------------------------
    # 2.  Iterative LP refinement (main workhorse).
    # ------------------------------------------------------------------
    # We keep a modest number of iterations; each LP is the dominant cost.
    max_lp_iters = 12
    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_lp_iters:
        M = _max_convolution_fft(a)

        # Solve LP to obtain a candidate g that respects the current bound M.
        g0 = _solve_lp_max_sum(a, M)
        if g0 is None:
            # fallback: break out of LP loop and continue with SA only
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        # Scale to the required ℓ¹‑norm
        g = g0 * (target_sum / sum_g0)

        # Back‑tracking line search to ensure non‑increase of the max‑convolution
        t = 0.5
        while t > 1e-4:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            M_new = _max_convolution_fft(a_candidate)
            if M_new <= M * (1.0 + 1e-12):
                a = a_candidate
                M = M_new
                break
            t *= 0.5
        # if line search fails we keep the old a (no update)

        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {iter_cnt} improved M={best_M:.6f}")

        # occasional jitter to avoid stagnation
        if iter_cnt % 4 == 0 and iter_cnt > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        iter_cnt += 1

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing fine‑tuning (uses the remaining time budget).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        sa_seq, sa_M = _simulated_annealing(best_a, time.time() + remaining)
        if sa_M < best_M:
            best_a = sa_seq
            best_M = sa_M
            print(f"[{time.time():.1f}] final SA improved M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 4.  Trim trailing near‑zero entries and return.
    # ------------------------------------------------------------------
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]
    else:
        best_a = np.array([target_sum])

    print(f"[{time.time():.1f}] finished: best max‑conv={best_M:.6f}, length={len(best_a)}")
    return best_a.tolist()
