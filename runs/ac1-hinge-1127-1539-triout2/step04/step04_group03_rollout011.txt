"""
A fast multi‑phase optimiser for the step‑function problem.

*Phase 1* – diverse random starts on the ℓ¹‑simplex (sum = √(2 n)).
*Phase 2* – sub‑gradient descent on the true max‑convolution (not a smooth
            surrogate).  The sub‑gradient of maxₖ (a⋆a)ₖ is
            g_i = 2·a_{k*‑i} for the lag k* attaining the maximum.
            A momentum term and a diminishing step‑size are used and the
            iterate is projected back onto the simplex after every step.
*Phase 3* – recombination of the best candidates (random convex mixes) and a
            simulated‑annealing fine‑tuner that moves a random fraction of
            mass between two positions while accepting uphill moves with a
            slowly cooling temperature.

All work respects the supplied time budget (default 1000 s) and never
touches the file system.  The returned list is the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a⋆a using an FFT (linear convolution)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Duchi‑et‑al. (2008) Euclidean projection onto the ℓ¹‑simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """
    Produce a random non‑negative vector on the simplex Σx = target_sum.
    A mixture of power‑law, exponential, Dirichlet and edge‑heavy patterns
    is used for diversity.
    """
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                       # decreasing power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                     # pure exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                     # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                     # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                               # uniform with tiny jitter
        a = np.full(n, 1.0)

    # small Gaussian perturbation and project back onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iters: int = 300) -> np.ndarray:
    """
    Projected sub‑gradient descent for   min   maxₖ (a⋆a)ₖ
    subject to a≥0, Σa = target_sum.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    # momentum parameters
    mom = np.zeros_like(a)
    rho = 0.9
    base_step = target_sum * 0.4

    it = 0
    while time.time() < deadline and it < max_iters:
        # full convolution and the maximal index set
        b = _full_convolution_fft(a)
        max_val = b.max()
        max_idxs = np.where(np.isclose(b, max_val, atol=1e-12))[0]

        # sub‑gradient = sum over all maximal lags
        g = np.zeros_like(a)
        for k in max_idxs:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            idx = np.arange(start, end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # momentum update
        mom = rho * mom + (1.0 - rho) * g
        norm = np.linalg.norm(mom) + 1e-12
        lr = base_step / norm
        a -= lr * mom

        # projection onto simplex (keeps sum = target_sum)
        a = _project_to_simplex(a, target_sum)

        # keep best ever seen
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if it % 50 == 0 and rng.random() < 0.1:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a random fraction of mass between
    two indices.  The temperature is slowly annealed.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9992
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move at most 12 % of the mass at i to j
        frac = rng.random() * 0.12
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Mixing of top candidates (random convex combinations)
# ----------------------------------------------------------------------


def _mix_top_candidates(candidates, target_sum, rng):
    """
    Produce a new candidate by taking a random convex combination of the
    supplied candidates and projecting onto the simplex.
    """
    k = len(candidates)
    if k == 0:
        return None
    weights = rng.dirichlet(alpha=np.ones(k))
    mixed = np.sum([w * v for w, (_, v) in zip(weights, candidates)], axis=0)
    mixed = _project_to_simplex(mixed, target_sum)
    return mixed

            mixed = _mix_top_candidates(top_pool, target_sum, rng)
            if mixed is not None:
                # a short SA run
                a_sa = _simulated_annealing(mixed, deadline)
                obj = _max_convolution_fft(a_sa)
                if obj < best_obj:
                    best_obj = obj
                    best_seq = a_sa.copy()
                    print(f"[{time.time():.1f}] NEW GLOBAL BEST (mix+SA) n={n}  obj={obj:.6f}")

        # reset pool before next length
        top_pool.clear()

    # ------------------------------------------------------------------
    # Final global fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # a last quick SA on the best sequence found so far
        final_seq = _simulated_annealing(best_seq, deadline)
        final_obj = _max_convolution_fft(final_seq)
        if final_obj < best_obj:
            best_obj = final_obj
            best_seq = final_seq.copy()
            print(f"[{time.time():.1f}] FINAL SA improvement: obj={final_obj:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check with the official evaluator (cheap)
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished – best max‑conv = {best_obj:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
