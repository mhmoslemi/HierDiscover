"""
Propose a candidate step function for the max‑convolution / ℓ¹‑norm benchmark.

Strategy
--------
* Try several problem sizes (the length of the step vector).  Larger
  lengths give more freedom to spread the ℓ¹‑mass and can lower the
  maximum convolution entry.
* For each length:
    – Initialise with a simple power‑law and project onto the simplex
      Σa = √(2·n) (the ℓ¹‑norm required by the evaluation function).
    – Run a brief soft‑max gradient descent (annealed β) to obtain a
      refined starting point.
    – Perform a simulated‑annealing (SA) phase that repeatedly moves a
      small amount of mass from a high‑sensitivity coordinate to a
      low‑sensitivity one or smooths a short window.  The SA uses an
      exponential temperature schedule and keeps the ℓ¹‑norm exactly
      constant.
    – Finish with a deterministic hill‑climb that greedily transfers
      mass based on the current sensitivity.
* Keep the best solution seen over all lengths and return it as a plain
  list of floats.  The total running time never exceeds the given budget
  (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a computed via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real part only)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective (max entry of a*a) when Σa = √(2·n)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return a (i+1)^(-p) sequence of length n, rescaled to sum to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient descent (annealed β)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (g, b) where g is the gradient w.r.t. a and b the exact
    convolution values.
    """
    n = a.shape[0]
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # Compute gradient: 2 * (a_rev * p) (linear convolution)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2 (covers all needed indices)
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max surrogate.
    β is increased exponentially (1 → 256).  The step size adapts to the
    gradient norm.  The iterate is periodically jittered to escape plateaus.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 256.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD   iter={it} β={beta:.2f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing based on current sensitivity
# ----------------------------------------------------------------------


def _guided_hill_climb(a0: np.ndarray, deadline: float):
    """
    Deterministic hill‑climbing: move a fraction of mass from the
    largest‑sensitivity coordinate to a low‑sensitivity one.
    The sensitivity is a_i * a_{k*‑i} where k* is the index attaining the
    current max convolution entry.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07

    while time.time() < deadline:
        conv = np.convolve(a, a)                # O(n) for moderate n
        k_max = int(np.argmax(conv))

        # Compute sensitivity a_i * a_{k_max-i} (only where indices overlap)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        sens[idx] = a[k_max - idx]

        weight = sens * a
        if weight.max() == 0:
            break
        donor = int(np.argmax(weight))

        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))

        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] HC   improved={best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 20) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves: (i) transfer a fraction of mass from a donor to a receiver,
           (ii) replace a short window by its average (smoothing).
    Acceptance follows a Metropolis rule with an exponentially cooling
    temperature schedule.  The ℓ¹‑norm stays constant.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = 2.0 * n * _max_convolution_fft(a) / (cur_sum ** 2)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # initial convolution for the first iteration
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # progress‐dependent parameters
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # ----- current state data -----
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity for the active convolution entry (same as in hill‑climb)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # ----- propose a move -----
        if rng.random() < 0.70:          # mass transfer (most common)
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.integers(n) if low_idx.size == 0 else rng.choice(low_idx)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            # apply move
            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                            # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # ----- evaluate new state (single FFT) -----
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new / (cur_sum ** 2)

        # ----- Metropolis acceptance -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA   new best={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:  # smooth
                a[saved_start:saved_end] = saved_seg
            # conv and cur_score stay unchanged

        # occasional jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = 2.0 * n * conv.max() / (cur_sum ** 2)

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **_ignore) -> list[float]:
    """
    Search for a non‑negative step function that minimizes ``evaluate_sequence``.
    The function tries several lengths, each time performing:
        1. power‑law initialization,
        2. short soft‑max gradient descent,
        3. simulated‑annealing,
        4. deterministic hill‑climb.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Determine candidate lengths
    # ------------------------------------------------------------------
    base_n = None
    if 'height_sequence_1' in globals():
        base_n = len(globals()['height_sequence_1'])
        print(f"[{time.time():.1f}] global sequence length = {base_n}")
    else:
        base_n = 1500
        print(f"[{time.time():.1f}] no global sequence, using default length {base_n}")

    # Build a list of increasing lengths (up to a reasonable bound)
    lengths = [base_n]
    multipliers = [1.3, 1.6, 2.0, 2.5]   # try a few larger problems
    for m in multipliers:
        n = int(base_n * m)
        # cap n to avoid excessive memory/computation
        if n > 8000:
            n = 8000
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    print(f"[{time.time():.1f}] candidate lengths: {lengths}")

    best_seq = None
    best_score = float('inf')

    for n in lengths:
        remaining = deadline - time.time()
        if remaining < 30.0:
            print(f"[{time.time():.1f}] time budget low, stop.")
            break

        # Allocate a chunk of time for this length (max 300 s)
        sub_deadline = time.time() + min(remaining * 0.5, 300.0)

        target_sum = np.sqrt(2.0 * n)
        seq = _power_law_sequence(n, 0.55, target_sum)
        seq = _project_to_simplex(seq, target_sum)

        # ------------------------------------------------------------------
        # Gradient‑descent warm‑start (if we have at least ~40 s left)
        # ------------------------------------------------------------------
        if time.time() < sub_deadline - 30.0:
            gd_deadline = time.time() + min(20.0, (sub_deadline - time.time()) * 0.3)
            seq_gd, gd_score = _annealed_gradient_descent(seq, gd_deadline)
            if gd_score < best_score:
                best_score = gd_score
                best_seq = seq_gd.copy()
                print(f"[{time.time():.1f}] length {n} GD improved best to {best_score:.6f}")

        # ------------------------------------------------------------------
        # Simulated annealing (core refinement)
        # ------------------------------------------------------------------
        if time.time() < sub_deadline:
            sa_seq, sa_score = _simulated_annealing(seq, sub_deadline)
            if sa_score < best_score:
                best_score = sa_score
                best_seq = sa_seq.copy()
                print(f"[{time.time():.1f}] length {n} SA improved best to {best_score:.6f}")

        # ------------------------------------------------------------------
        # Final deterministic hill‑climb (short refinement)
        # ------------------------------------------------------------------
        if time.time() < sub_deadline:
            hc_deadline = sub_deadline
            hc_seq, hc_score = _guided_hill_climb(best_seq.copy(), hc_deadline)
            if hc_score < best_score:
                best_score = hc_score
                best_seq = hc_seq.copy()
                print(f"[{time.time():.1f}] length {n} HC improved best to {best_score:.6f}")

        print(f"[{time.time():.1f}] after length {n}, current global best={best_score:.6f}")

        # Early exit if we already reached the target bound
        if best_score <= 1.503:
            print(f"[{time.time():.1f}] target bound reached; stop early.")
            break

    # ------------------------------------------------------------------
    # Trim trailing zeros to keep output compact
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.array([], dtype=np.float64)
    else:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = best_seq[:1]   # keep at least a single zero

    final_score = _score(best_seq)
    print(f"[{time.time():.1f}] final best score={final_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
