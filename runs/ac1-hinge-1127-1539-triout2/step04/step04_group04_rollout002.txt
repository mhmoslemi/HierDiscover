def _lp_refine(f0: np.ndarray, deadline: float, rng: np.random.Generator):
    f = f0.copy()
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    best_f = f.copy()
    best_max = np.max(_full_convolution_fft(f))
    # Precompute possible maximal iterations based on time budget.
    iter_cnt = 0
    t = 0.2   # mixture parameter
    Bmax = 1e5  # upper bound for b entries
    while time.time() < deadline:
        iter_cnt += 1
        # Compute current max
        cur_max = np.max(_full_convolution_fft(f))
        # Build constraint matrix A
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f[i]
        # Solve LP
        try:
            b = cp.Variable(n, nonneg=True)
            constraints = [A @ b <= cur_max]
            # add upper bound to avoid unbounded large variables
            constraints.append(b <= Bmax)
            prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-7, reltol=1e-7)
        except Exception as e:
            print(f"[lp] error solving LP: {e}")
            break
        if b.value is None:
            print("[lp] solver failed to produce solution.")
            break
        b_val = np.array(b.value, dtype=np.float64)
        # Scale to sum S
        sum_b = b_val.sum()
        if sum_b < 1e-12:
            break
        scaling = S / sum_b
        g = b_val * scaling
        # Convex combination
        f_new = (1 - t) * f + t * g
        # Project onto simplex to maintain sum (tiny error)
        f_new = _project_to_simplex(f_new, S)
        new_max = np.max(_full_convolution_fft(f_new))
        if new_max < best_max - 1e-8:
            best_max = new_max
            best_f = f_new.copy()
            f = f_new
            print(f"[lp] iter {iter_cnt}: improved max to {best_max:.6f}")
        else:
            # If no improvement, reduce step size t
            t *= 0.5
            if t < 1e-4:
                # cannot improve further
                break
    return best_f, best_max
```

"""
Step‑function optimizer.

The optimizer works in three stages:

1️⃣  Random sampling of parametric families (exponential, Gaussian, triangular,
    cosine, power‑law, etc.).  Each candidate vector is projected onto the
    ℓ₁‑simplex Σa = √(2·n) (the normalisation used by `evaluate_sequence`),
    its self‑convolution is evaluated and the best candidate is retained.

2️⃣  Iterative linear‑program (LP) refinement.
    For the current vector f we compute M = max(f∗f) and solve
        maximise    Σ b_j
        subject to  (f∗b)_k ≤ M   for all k,
                    b_j ≥ 0 .
    The solution b is rescaled to the simplex, mixed with f
        f ← (1‑t)·f + t·g   (g is the rescaled b),
    and the loop repeats while the bound improves or time runs out.
    This is the classic “alternating LP” method described in the prompt
    and drives the maximal autocorrelation down monotonically.

3️⃣  A short Adam optimisation (soft‑max approximation of the ℓ∞‑norm) is
    applied as a final polishing step.

The final sequence is stored in the global variable `height_sequence_1`
and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional import of cvxpy – LP based refinement
# ----------------------------------------------------------------------

    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the ℓ₁‑simplex of sum S."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):  # fallback – uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a∗a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Compute gradient of τ·log∑exp(b/τ) where b = a∗a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard max ⇒ sub‑gradient (exact max)
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal contribution when i=j (only once)
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights w ∝ exp(b/τ)
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)          # shape (2n‑1,)

    # compute gradient = 2·a ⋆ w
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # length n

    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection (used as a polishing step)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Minimise the soft‑max approximation of max(a∗a) using Adam.
    Returns the best vector found and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6                      # initial learning rate
    t = 0

    # initialise temperature for soft‑max
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        cur_max_new = np.max(_full_convolution_fft(a_new))

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)           # cooling
        lr *= 0.9999                           # decay

        # occasional tiny jitter to escape plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector from a random parametric family."""
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular",
         "power", "cosine"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        # raised cosine – non‑negative on the interval
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    else:
        a = rng.random(n)

    # occasional jitter
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a


# ----------------------------------------------------------------------
# LP based refinement (core of the iterative improvement)
# ----------------------------------------------------------------------
def _lp_refine(f0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Perform the alternating LP refinement described in the prompt.
    Returns the refined vector and its maximal convolution entry.
    """
    if not _HAS_CVX:
        return f0, np.max(_full_convolution_fft(f0))

    f = f0.copy()
    n = f.shape[0]
    S = np.sqrt(2.0 * n)

    best_f = f.copy()
    best_max = np.max(_full_convolution_fft(f))
    t_mix = 0.2         # initial mixing weight
    Bmax = 1e6          # safety bound for b_j

    iter_cnt = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 2.0:   # not enough time for another LP solve
            break

        cur_max = np.max(_full_convolution_fft(f))

        # Build the convolution matrix A where (A @ b)[k] = (f∗b)[k]
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f[i]

        # Linear programme: maximise sum(b) s.t. A·b ≤ cur_max, b ≥ 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= cur_max, b_var <= Bmax]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, abstol=1e-7, reltol=1e-7)
        except Exception as e:   # pragma: no cover
            print(f"[lp] solver exception: {e}")
            break

        if b_var.value is None:
            # solver failed – abort refinement
            break

        b_val = np.array(b_var.value, dtype=np.float64)
        sum_b = b_val.sum()
        if sum_b <= 0:
            break

        # Rescale to the simplex Σg = √(2n)
        scaling = S / sum_b
        g = b_val * scaling

        # Convex mixture with the current vector
        f_new = (1.0 - t_mix) * f + t_mix * g
        f_new = _project_to_simplex(f_new, S)

        new_max = np.max(_full_convolution_fft(f_new))

        iter_cnt += 1
        if new_max < best_max - 1e-9:
            best_max = new_max
            best_f = f_new.copy()
            f = f_new
            print(f"[lp] iter {iter_cnt}: improved max to {best_max:.6f}")
        else:
            # no improvement – shrink the mixing weight
            t_mix *= 0.5
            if t_mix < 1e-4:
                break

    return best_f, best_max
