def _lp_step_optimize(f, target_sum, max_iter=10, time_deadline=None):
   n = len(f)
   best_f = f.copy()
   best_M = _full_convolution_fft(f).max()
   for iteration in range(max_iter):
       # check time
       if time_deadline is not None and time.time()>time_deadline:
           break
       M = best_M
       # Build A matrix
       A = np.zeros((2*n-1, n), dtype=np.float64)
       # fill columns
       for j in range(n):
           A[j:j+n, j] = f
       # solve LP
       c = -np.ones(n) # maximize sum, minimize negative
       res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
       if not res.success:
           break
       g = res.x
       sum_g = g.sum()
       if sum_g <= 0:
           break

def _lp_alternating_optimize(f0, max_iter, deadline, rng):
    f = f0.copy()
    n = len(f)
    target_sum = np.sqrt(2.0 * n)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute M = max self-convolution
        b = _full_convolution_fft(f)
        M = b.max()
        # Build A matrix if M >0 else break
        # A shape (2n-1, n)
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = f
        # Solve LP
        c = -np.ones(n)   # maximize sum(g)
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        scale = target_sum / sum_g
        g_scaled = g * scale
        # clip to 1000 to avoid overflow, though not needed
        g_scaled = np.clip(g_scaled, 0.0, 1000.0)
        # project back to simplex (for safety)
        g_proj = _project_to_simplex(g_scaled, target_sum)
        # Update f
        f_new = g_proj  # replace
        # compute new max
        M_new = _full_convolution_fft(f_new).max()
        if M_new < M - 1e-9:
            f = f_new
        else:
            # If no improvement, maybe try mixing
            # fallback: gradient step from earlier algorithm
            # partly break out
            break
    # final max
    final_max = _full_convolution_fft(f).max()
    return f, final_max

"""
Optimisation via alternating linear programming + simplex projection.

The algorithm proceeds as follows:

1. Try a handful of deterministic window shapes (triangular, Hann,
   sine‑squared) for several lengths and keep the best one.

2. Repeatedly draw a random feasible vector (non‑negative, ℓ₁‑norm
   √(2·n)) for a randomly chosen length n (600 – 2500).  For each
   start we run a short *alternating* optimisation:
      – compute M = max (f * f),
      – solve the linear programme  max Σ g_j  subject to  f * g ≤ M,
        g ≥ 0,
      – scale g to the required ℓ₁‑norm,
      – replace f by a convex combination (1‑t)·f + t·g (projected onto the simplex),
        with a slowly decreasing mixing factor t.
   The LP is tiny (≤ 4000 constraints, ≤ 2500 variables) and solved
   with SciPy's HiGHS solver, so many restarts can be performed within
   the time budget.

3. After the LP phase a few smooth soft‑max gradient steps are applied
   (the same update used in the original code) to polish the solution.

The best sequence found before the deadline is returned and stored in
the global variable ``height_sequence_1`` for future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (required for the LP step)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None

# ----------------------------------------------------------------------
# Utility: nearest power of two (used by FFT‑based convolutions)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT‑based full convolution of two real vectors
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y using a real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the simplex with total mass S."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort in descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]               # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear scaling / interpolation of a sequence to a new length
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Deterministic window shapes (triangular, Hann, sine‑squared)
# ----------------------------------------------------------------------
def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular vector: min(i+1, n-i)."""
    idx = np.arange(n)
    return np.minimum(idx + 1, n - idx).astype(float)


def _hann_window(n: int) -> np.ndarray:
    """Standard Hann window, zero at the boundaries."""
    i = np.arange(n)
    return 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1))).astype(float)


def _cosine_bell(n: int) -> np.ndarray:
    """Sine‑squared bell (zero at both ends, smooth interior)."""
    i = np.arange(n)
    return (np.sin(np.pi * i / (n - 1)) ** 2).astype(float)


# ----------------------------------------------------------------------
# Soft‑max (smooth) gradient of the ∞‑norm of a*a (used for polishing)
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float):
    """
    Perform one additive step using a τ‑soft‑max approximation of
    max(b), where b = a*a.
    Returns the projected vector, the new convolution and its maximum entry.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau <= 1e-12:  # hard max → fallback to zero sub‑gradient
        g = np.zeros_like(a)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)                     # (2n‑1,) soft‑max weights

        # cross‑correlation a ⋆ w = Σ_j a_j w_{i+j}
        a_rev = a[::-1]
        conv_len = len(w) + n - 1
        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(w, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    b_new = _full_convolution_fft(a_proj)
    max_new = b_new.max()
    return a_proj, b_new, max_new


# ----------------------------------------------------------------------
# Alternating linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_alternating_optimize(f0: np.ndarray,
                             max_iter: int,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Starting from f0 (already on the simplex), run a short alternating
    scheme:
        * compute M = max(f*f)
        * solve   max Σ g_j   s.t.  f*g ≤ M,  g ≥ 0
        * scale g to the target ℓ₁‑norm, mix with f,
          project back onto the simplex and keep the mixture if it
          strictly lowers the maximal self‑convolution.
    The function returns the best vector encountered and its maximal
    self‑convolution entry.
    """
    if linprog is None:      # pragma: no cover
        # SciPy not available – fall back to the input vector unchanged.
        return f0, _full_convolution_fft(f0).max()

    f = f0.copy()
    n = len(f)
    target_sum = np.sqrt(2.0 * n)

    best_f = f.copy()
    best_M = _full_convolution_fft(f).max()

    # mixing coefficient – starts moderate and is adapted automatically
    t = 0.25

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # 1) current bound M
        M = _full_convolution_fft(f).max()

        # 2) build the (2n‑1)×n convolution matrix A where A·g = f*g
        #    (each column j contains f shifted down by j positions)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        # 3) LP: maximise sum(g)  → minimise -1ᵀg
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=(0, None), method='highs', options={'presolve': True})
        if not res.success:
            # if the LP fails, abort the alternating loop
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # 4) scale g to the required ℓ₁‑norm and clip to the evaluator’s limits
        scale = target_sum / sum_g
        g_scaled = np.clip(g * scale, 0.0, 1000.0)

        # 5) convex combination with the current iterate & project back
        f_candidate = (1.0 - t) * f + t * g_scaled
        f_candidate = _project_to_simplex(f_candidate, target_sum)

        M_candidate = _full_convolution_fft(f_candidate).max()

        # 6) accept if we strictly improve the bound
        if M_candidate < best_M - 1e-12:
            best_M = M_candidate
            best_f = f_candidate.copy()
            f = f_candidate
            # become a bit more aggressive after a successful step
            t = min(t * 1.2, 0.5)
        else:
            # otherwise back‑off the mixing weight
            t *= 0.5
            if t < 1e-4:
                break

    return best_f, best_M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step sequence minimising the evaluation
    function.  The search proceeds in three stages:

    1. deterministic window shapes (triangular, Hann, sine‑squared) for a
       handful of lengths,
    2. many random restarts of an alternating LP+simplex refinement,
    3. a few soft‑max polishing steps.

    The best sequence found before the deadline expires is returned.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously stored construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Stage 1 – deterministic windows for a few lengths
    # ------------------------------------------------------------------
    candidate_lengths = [256, 512, 768, 1024, 1280, 1536, 1792, 2048]
    for n in candidate_lengths:
        S = np.sqrt(2.0 * n)

        for shape_fun in (_triangular_shape, _hann_window, _cosine_bell):
            seq = shape_fun(n).astype(float)
            # avoid all‑zero shape (should never happen) and scale to the required ℓ₁‑norm
            if seq.sum() == 0:
                continue
            seq = seq * (S / seq.sum())
            cur_max = _full_convolution_fft(seq).max()
            if cur_max < best_score:
                best_score = cur_max
                best_seq = seq.copy()
                print(f"[init] n={n:4d} shape={shape_fun.__name__:13s} max={cur_max:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – random restarts with alternating LP refinement
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # pick a length (moderately sized – stays comfortably within LP limits)
        n = int(rng.integers(600, 2500))
        S = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # initialise a feasible vector on the simplex
        # --------------------------------------------------------------
        if base_seq is not None and base_seq.size > 0:
            f0 = _scale_sequence(base_seq, n).astype(float)
            f0 += rng.normal(scale=max(f0.mean(), 1e-12) * 0.001, size=n)
            f0 = np.clip(f0, 0.0, None)
        else:
            f0 = rng.random(n)
        f0 = _project_to_simplex(f0, S)

        # --------------------------------------------------------------
        # allocate a time slice for this restart (≈8 s) and run the LP loop
        # --------------------------------------------------------------
        slice_deadline = min(time.time() + 8.0, end_time)
        f_opt, max_opt = _lp_alternating_optimize(f0, max_iter=8,
                                                  deadline=slice_deadline)

        # --------------------------------------------------------------
        # a tiny polishing phase using the soft‑max update (3 steps max)
        # --------------------------------------------------------------
        polishing_deadline = min(time.time() + 2.0, end_time)
        τ = max_opt
        η = 0.7
        for _ in range(3):
            if time.time() > polishing_deadline:
                break
            f_opt, _, τ = _softmax_update(f_opt, max(tau := τ, 1e-12), η)
            τ *= 0.95

        final_max = _full_convolution_fft(f_opt).max()
        if final_max < best_score:
            best_score = final_max
            best_seq = f_opt.copy()
            print(f"[lp] attempt={attempt:3d} n={n:4d} new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the black‑box function
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                     # fallback – should never happen
    else:
        final_seq = best_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_score:.6f}) length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
