"""
Propose a step‐function (list of non‑negative heights) that minimises the
provided evaluate_sequence() function.

Algorithm
---------

1. **Warm‑start** – generate a few exponential‑decay sequences of several
   lengths and keep the best (according to the exact objective
   2·n·max(a*a)/sum(a)^2).  If a previous solution (height_sequence_1) exists
   it is also considered.

2. **Linear‑programming refinement** – for the current sequence f (sum = √(2n))
   solve the LP

        minimise   M
        s.t.       f * g  ≤  M            (2n‑1 inequalities)
                  Σ g    =  √(2n)
                  g ≥ 0

   The LP variables are g (the new candidate) and the scalar M.
   The optimal M is never larger than the present max‑convolution of f,
   therefore the LP always yields a feasible improvement direction.

   After solving, we mix the old and the new vector:
        f←(1‑t)·f + t·g  with a line‑search on t∈{0.5,0.25,…}.
   The mixing step is guaranteed to reduce the maximal convolution entry.
   The process is repeated a limited number of times (or until the time
   budget is exhausted).

3. **Local polishing** – a short projected‑gradient + random‑swap routine
   (the same as in the reference code) is run on the best vector obtained
   from step 2 for the remaining time.

All vectors are kept normalised to the ℓ1‑simplex Σa = √(2·n) which makes the
objective scale‑invariant; the final sequence is returned as a plain Python
list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max objective.
    a has length n, w has length 2·n‑1 and sums to 1.
    Returns 2·Σ_i w_{i+j}·a_i for each j.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers (used by the local optimiser)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i ∝ exp(α·conv_i)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """One projected‑gradient step on the smoothed max‑norm."""
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Warm‑start: exponential decay sequence
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """
    Decreasing exponential sequence of length n.
    k_factor controls the decay (larger → faster decay).
    Result is projected onto the simplex Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - q ** n
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Linear programming helpers
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × (n+1) constraint matrix for the LP:
        (f * g)[k] ≤ M   ⇔   Σ_j f_{k‑j} g_j – M ≤ 0.
    The last column corresponds to the variable M (with coefficient –1).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n + 1), dtype=np.float64)
    # Each column j receives the vector f shifted down by j rows.
    for j in range(n):
        A[j:j + n, j] = f
    A[:, n] = -1.0          # –M term
    return A


def _lp_optimize_step(f: np.ndarray, S: float):
    """
    Solve the LP described in the docstring.
    Returns the optimal g (length n) and the optimal bound M.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A_ub = _build_A_ub(f)                # dense matrix (rows × (n+1))
    b_ub = np.zeros(rows, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0                    # Σ g = S
    b_eq = np.array([S], dtype=np.float64)

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0                           # minimise M

    bounds = [(0, None)] * (n + 1)        # g ≥ 0, M ≥ 0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    x = res.x
    g = x[:n]
    M_opt = x[-1]
    return g, M_opt


def _mix_step(f, g, S, cur_max, deadline):
    """
    Convexly combine f and g.  Perform a short line‑search on t∈{0.5,0.25,…}
    and accept the first t that yields a strictly smaller maximal convolution.
    Returns the new vector, its new maximal convolution and the full convolution.
    """
    t = 0.5
    while t > 1e-4 and time.time() < deadline:
        new_f = (1.0 - t) * f + t * g
        # sum stays equal to S automatically
        conv = _full_convolution(new_f)
        new_max = np.max(conv)
        if new_max < cur_max:
            return new_f, new_max, conv
        t *= 0.5
    return f, cur_max, None


def _lp_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Repeatedly apply the LP step and a mixing move.
    Stops when the deadline is reached or after a modest number of iterations.
    Returns the best vector found and its objective value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    conv = _full_convolution(a)
    cur_max = np.max(conv)
    best_a = a.copy()
    best_obj = 2.0 * n * cur_max / (S ** 2)

    it = 0
    stagn = 0
    while time.time() < deadline and it < 60:
        it += 1
        # 1) LP – minimise bound M for the *current* f
        try:
            g, M_lp = _lp_optimize_step(a, S)
        except Exception as e:
            print(f"[LP] iteration {it}: LP error – {e}")
            break

        # 2) Mix f and g (line search on t)
        a, cur_max, new_conv = _mix_step(a, g, S, cur_max, deadline)

        # 3) Update best record
        obj = 2.0 * n * cur_max / (S ** 2)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[LP] iter {it:3d}  obj={obj:.6f}")
            stagn = 0
        else:
            stagn += 1

        # occasional jitter if we appear stuck
        if stagn >= 5:
            jitter = rng.normal(scale=0.001 * a.mean(), size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = a * (S / a.sum())
            conv = _full_convolution(a)
            cur_max = np.max(conv)
            stagn = 0

    return best_a, best_obj

def _smooth_weights_lp(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Alias for compatibility with the local optimiser."""
    return _smooth_weights(conv, alpha)


def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fine‑tuning using projected‑gradient descent on a smoothed max‑norm
    together with cheap random/targeted mass‑swap moves.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (empirically chosen)
    lr = 0.5
    lr_decay = 0.9999
    alpha = 0.5
    alpha_mul = 1.005
    swap_prob = 0.15
    swap_frac = 0.04
    target_prob = 0.2
    eps_target = 0.03

    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # gradient step
        a, cur_max = _gradient_step(a, lr, alpha)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # random mass‑swap
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                             max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # cheap targeted move
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            src_idx = np.argmax(contributions)
            i_src = i_range[src_idx]
            tgt_idx = np.argmin(contributions)
            i_tgt = i_range[tgt_idx]

            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                             max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # schedule updates
        lr *= lr_decay
        alpha *= alpha_mul

        if it - last_improve > 5000:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function minimising evaluate_sequence().
    The routine consists of a warm‑start, a linear‑programming refinement,
    and a final projected‑gradient polishing phase.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Warm‑start: a few exponential candidates
    # ------------------------------------------------------------------
    for n in (1500, 2000, 2500):
        if time.time() >= deadline:
            break
        for k_factor in np.linspace(0.6, 2.5, 6):
            a = _exponential_sequence(n, k_factor)
            cur_score = 2.0 * n * np.max(_full_convolution(a)) / (np.sqrt(2.0 * n) ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[warm] n={n:4d} k={k_factor:4.2f} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Use previous solution if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                n = prev.size
                S = np.sqrt(2.0 * n)
                prev = prev * (S / prev.sum())
                cur_score = 2.0 * n * np.max(_full_convolution(prev)) / (S ** 2)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = prev.copy()
                    print(f"[prev] using height_sequence_1 score={cur_score:.6f}")
        except Exception:
            pass

    if best_seq is None:
        # fallback: uniform sequence of length 2000
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # ------------------------------------------------------------------
    # 3) LP‑based refinement (major portion of the budget)
    # ------------------------------------------------------------------
    lp_deadline = min(deadline, time.time() + budget_s * 0.6)
    try:
        a_lp, score_lp = _lp_optimize(best_seq, lp_deadline, rng)
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[lp] improved to score={score_lp:.6f}")
    except Exception as e:
        print("[lp] optimisation error:", e)

    # ------------------------------------------------------------------
    # 4) Local gradient‑based polishing for the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_loc, max_loc = _local_optimize(best_seq.copy(), deadline, rng)
        obj_loc = 2.0 * len(a_loc) * max_loc / (np.sqrt(2.0 * len(a_loc)) ** 2)
        if obj_loc < best_score:
            best_score = obj_loc
            best_seq = a_loc.copy()
            print(f"[local] polished best score={obj_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Final sanity check via the external evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
