"""
Improved search for low‑evaluation step‑function.

Strategy
---------
1. Randomly draw many candidate height vectors from a rich catalogue of
   parametric families (including dense families, sparse Dirichlet‑
   style vectors and mixtures).  The vector length `n` is drawn
   logarithmically from 800 up to 20000, allowing larger sequences
   to explore the theoretical lower bound 2·n/(2n‑1)≈1.
2. Each candidate is normalised to a fixed ℓ₁‑norm
   `sqrt(2·n)` (the objective is scale‑invariant) and its
   convolution peak `max(a*a)` is computed with an FFT‑based
   convolution.
3. The best `k` candidates seen so far are stored.  When a new
   candidate beats the current best by a modest margin, an
   Adam optimiser with a soft‑max approximation of the ∞‑norm is
   launched to refine it.  The optimiser works on the simplex
   (projection after every step) and slowly cools the temperature.
4. The search runs until the time budget (≈1000 s) is exhausted.
   The best sequence found is returned and cached in the global
   variable `height_sequence_1` for subsequent runs.

All helper functions are top‑level (no closures) and use only the
standard library plus NumPy.  The code respects the 1‑CPU limit
and prints periodic progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the ℓ₁‑simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum S."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution using a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (full convolution) computed by FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient (or sub‑gradient) of the soft‑max approximation of max
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Returns (g, b, max_b) where
      g      – gradient of τ·log Σ exp(b/τ) w.r.t a   (or a sub‑gradient for τ≈0)
      b      – the full convolution a*a
      max_b  – max(b)
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard max -> sub‑gradient
        k = np.argmax(b)                # index of the maximum entry
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal term (i=j) appears only once
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights w_i ∝ exp(b_i / τ)
    diff = b - max_b               # for numerical stability
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)         # shape (2n‑1,)

    # compute g = 2·a ⋆ w   (cross‑correlation)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]

    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   lr: float = 0.8, beta1: float = 0.9,
                   beta2: float = 0.999, eps: float = 1e-8,
                   tau_factor: float = 0.5,
                   decay_lr: float = 0.9999,
                   decay_tau: float = 0.9995,
                   jitter_prob: float = 0.008,
                   jitter_scale: float = 0.015,
                   rng: np.random.Generator = None):
    """
    Minimise the soft‑max approximation of max(a*a) with Adam.
    Returns the best vector encountered and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # initialise temperature
    _, _, cur_max = _gradient_softmax(a, tau=0.0)   # hard max just for scaling
    tau = max(cur_max * tau_factor, 1e-12)

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, b, cur_max = _gradient_softmax(a, tau)

        # Adam update before projection
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluation
        b_new = _full_convolution_fft(a_new)
        cur_max_new = np.max(b_new)

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * decay_tau, 1e-8)
        lr *= decay_lr

        # occasional jitter
        if jitter_prob and rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families (dense + sparse)
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create an unnormalised non‑negative vector from a random family."""

    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular",
         "power", "cosine", "sparse", "bimodal"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        dec = inc[::-1] if n % 2 == 0 else inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "sparse":
        # random support of size roughly sqrt(2n)
        m = max(2, int(rng.uniform(0.5, 2.0) * np.sqrt(2.0 * n)))
        indices = rng.choice(n, size=m, replace=False)
        weights = rng.random(m) + 0.1   # avoid exact zeros
        a = np.zeros(n)
        a[indices] = weights

    elif typ == "bimodal":
        # two Gaussian bumps at random locations
        sigma = rng.uniform(0.02, 0.07) * n
        centre1 = rng.uniform(0.2, 0.4) * n
        centre2 = rng.uniform(0.6, 0.8) * n
        a = np.exp(-0.5 * ((idx - centre1) / sigma) ** 2)
        a += np.exp(-0.5 * ((idx - centre2) / sigma) ** 2)

    else:
        a = rng.random(n)

    # small random jitter (helps diversity)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a

"""
Improved search for a low‑score step‑function.

Strategy
--------
1. Randomly generate many candidates from a rich set of parametric
   families (dense, sparse‑Dirichlet, bimodal Gaussians, …).  The
   length `n` is drawn log‑uniformly between 800 and 20 000, giving the
   optimiser room to approach the theoretical lower bound
   `max(a*a) ≈ 1`.

2. Every candidate is normalised to ℓ₁‑norm `√(2·n)` – after this
   normalisation the external evaluation equals the maximal
   convolution entry, i.e. `evaluate_sequence(a) = max(a*a)`.  This makes
   the objective scale‑invariant and allows direct minimisation of the
   ℓ∞‑norm of the self‑convolution.

3. The best few candidates are refined with an Adam optimiser that
   minimises a soft‑max approximation of the ℓ∞‑norm while staying on
   the simplex.  The temperature is cooled, the learning rate decayed,
   and occasional Gaussian jitter is added to escape plateaus.

4. The search runs until the time budget (~1000 s) is exhausted.
   The best sequence found is returned and cached in the global variable
   `height_sequence_1` for reuse by later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the ℓ₁‑simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """ Euclidean projection of v onto the simplex of total mass S. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution a*a using a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient (or sub‑gradient) of the soft‑max approximation of max(a*a)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Returns (g, b, max_b) where
        g      – gradient of τ·log Σ exp(b/τ) w.r.t. a (sub‑gradient if τ≈0)
        b      – the full convolution a*a
        max_b  – max(b)
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                     # hard max → sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal term (i=j) appears only once
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)

    # g = 2·a ⋆ w   (cross‑correlation)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]

    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   lr: float = 0.8, beta1: float = 0.9,
                   beta2: float = 0.999, eps: float = 1e-8,
                   tau_factor: float = 0.5,
                   decay_lr: float = 0.9999,
                   decay_tau: float = 0.9995,
                   jitter_prob: float = 0.008,
                   jitter_scale: float = 0.015,
                   rng: np.random.Generator = None):
    """
    Adam optimisation of the soft‑max approximation of max(a*a).
    Returns the best vector seen and its max‑convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # initialise temperature from a hard‑max evaluation
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = max(cur_max * tau_factor, 1e-12)

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        g, b, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # evaluate new max
        cur_max_new = np.max(_full_convolution_fft(a_new))

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedule updates
        tau = max(tau * decay_tau, 1e-8)
        lr *= decay_lr

        # occasional jitter to escape plateaus
        if jitter_prob and rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Random families (dense + sparse)
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create an unnormalised non‑negative vector from a random family."""
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular",
         "power", "cosine", "sparse", "bimodal"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        dec = inc[::-1] if n % 2 == 0 else inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "sparse":
        # support size ≈ √(2n) with a random factor
        m = max(2, int(rng.uniform(0.5, 2.0) * np.sqrt(2.0 * n)))
        indices = rng.choice(n, size=m, replace=False)
        weights = rng.random(m) + 0.1
        a = np.zeros(n)
        a[indices] = weights

    elif typ == "bimodal":
        sigma = rng.uniform(0.02, 0.07) * n
        centre1 = rng.uniform(0.2, 0.4) * n
        centre2 = rng.uniform(0.6, 0.8) * n
        a = np.exp(-0.5 * ((idx - centre1) / sigma) ** 2)
        a += np.exp(-0.5 * ((idx - centre2) / sigma) ** 2)

    else:
        a = rng.random(n)

    # small jitter for diversity
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a

    # ------------------------------------------------------------------

        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external_score:.6f} "
        f"(internal max = {best_max:.6f}) length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
