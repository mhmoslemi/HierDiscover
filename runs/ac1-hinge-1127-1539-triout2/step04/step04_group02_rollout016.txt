"""
Propose a step‑function (non‑negative coefficient vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

by a three‑stage heuristic:

1.  **Broad parametric sweep** – a collection of symmetric “Beta‑shaped’’
    vectors  a_i ∝ (i+½)^{α‑1} (n‑i‑½)^{β‑1}  (α,β∈(0,1] ), together with the
    previously used linear / exponential families and many random Dirichlet
    vectors.  The best candidate of this fast scan is kept.

2.  **Smooth + sub‑gradient descent** – minimise a log‑sum‑exp surrogate of
    the maximum using momentum (Adam‑style) and finally the true nonsmooth
    sub‑gradient.  The iterates are always projected onto the ℓ₁‑simplex
    {x≥0, Σx = √(2 n)}.

3.  **Simulated‑annealing polishing** – many tiny random pair‑mass
    transfers are accepted according to a temperature schedule; the last
    few seconds perform a pure improvement‑only scan.

All phases respect the 1000 s budget and return the best vector found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a fraction of the mass from a random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence.
    Supported ``kind`` values are the same as in the reference code.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_beta_shape(n: int, target_sum: float, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta‑type shape:
        a_i ∝ (i+½)^{α‑1} (n‑i‑½)^{β‑1}
    All α,β>0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    raw = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_inverse_power(n: int, target_sum: float, p: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)*(n‑i‑½))^{-p}
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    raw = ((i * (n - i)) ** (-p))
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    `evaluate_sequence` objective.  The routine respects the 1000 s time
    budget and returns the best vector discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    # --------------------------------------------------------------
    # Problem size – use a supplied baseline if present
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.asarray(base, dtype=np.float64)
        init_seq = project_onto_simplex(init_seq, math.sqrt(2.0 * n))
    else:
        n = 16384                # default power‑of‑two length
        init_seq = None

    target_sum = math.sqrt(2.0 * n)   # Σa_i = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – wide parametric sweep (including new Beta shapes)
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)   # uniform start
    best_score = compute_max_conv(best_seq)

    print(f"[{time.time():.2f}] Phase 0 – parametric sweep (n={n})")

    # ----- predefined families -------------------------------------------------
    shape_kinds = (
        "linear", "quadratic", "exponential",
        "power", "inverse",
        "linear_left", "linear_right",
        "exponential_left", "exponential_right",
    )
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),
        "quadratic": np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 12),
        "inverse": np.linspace(0.5, 5.0, 12),
        "linear_left": np.linspace(0.0, 0.9, 19),
        "linear_right": np.linspace(0.0, 0.9, 19),
        "exponential_left": np.linspace(0.0, 6.0, 31),
        "exponential_right": np.linspace(0.0, 6.0, 31),
    }

    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind:15s} (p={p:5.2f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # ----- new Beta‑family -------------------------------------------------------
    alphas = np.linspace(0.2, 1.2, 11)          # 0.2 … 1.2
    for a in alphas:
        cand = generate_beta_shape(n, target_sum, a, a)   # symmetric
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best beta (α=β={a:4.2f}) → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during beta sweep.")
            return best_seq.tolist()

    # ----- inverse‑power family ---------------------------------------------------
    p_vals = np.linspace(0.05, 0.45, 9)        # 0.05 … 0.45
    for p in p_vals:
        cand = generate_inverse_power(n, target_sum, p)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best inv‑pow (p={p:4.2f}) → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during inverse‑power sweep.")
            return best_seq.tolist()

    # ----- a few random Dirichlet‑like vectors ------------------------------------
    for _ in range(30):
        cand = np.random.rand(n)
        cand = project_onto_simplex(cand, target_sum)
        score = compute_max_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best random init → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during random init.")
            return best_seq.tolist()

    # ----- consider supplied baseline --------------------------------------------
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves start → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed gradient descent (Adam‑style)
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.55
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.5 * (target_sum / n)   # base learning rate
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Phase 1 – smoothed gradient descent")
    while time.time() < phase1_deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()
        tau = max(0.12 * max_conv, 1e-12)          # smoothing scale

        # log‑sum‑exp weights (softmax)
        w = np.exp((conv - max_conv) / tau)
        w /= w.sum()

        # gradient of the smoothed max (see reference implementation)
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # occasional jitter to escape flat spots
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # keep best
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration:07d} – improved to {best_score:.6f}")
        else:
            no_improve += 1

        # learning‑rate decay
        if iteration % 3000 == 0:
            lr *= 0.92

        if no_improve >= 5000:
            jitter = np.random.randn(n) * 0.0003
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:07d} – jitter applied")

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement
    # --------------------------------------------------------------
    subgrad_deadline = start_time + budget_s * 0.90
    step = 0.04 * (target_sum / n)
    sub_iter = 0
    print(f"[{time.time():.2f}] Phase 2 – sub‑gradient refinement")
    while time.time() < subgrad_deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()

        eps_loc = max(1e-12, 1e-9 * max_conv)
        mask = (conv >= max_conv - eps_loc).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0

        a_rev = a[::-1]
        L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        a -= step * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        sub_iter += 1
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Sub‑grad iter {sub_iter:06d} – improved to {best_score:.6f}")

        if sub_iter % 500 == 0:
            step *= 0.94

        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – simulated‑annealing polishing")
    anneal_start = time.time()
    # linear temperature schedule
    T0, Tf = 5e-3, 1e-6
    total_anneal = max(1.0, deadline - time.time())
    cur = best_seq.copy()
    cur_score = best_score

    while time.time() < deadline:
        # temperature decays linearly with elapsed annealing time
        elapsed = time.time() - anneal_start
        T = T0 + (Tf - T0) * (elapsed / total_anneal)

        # propose a move
        scale = 0.004 + 0.01 * (T / T0)   # larger steps early, smaller later
        cand = random_pair_transfer(cur.copy(), target_sum, scale=scale)
        cand_score = compute_max_conv(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < math.exp((cur_score - cand_score) / max(T, 1e-12)):
            cur, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"SA improved to {best_score:.6f}")

        # occasional pure improvement scan (no‑temperature move)
        if np.random.rand() < 0.005:
            polish = random_pair_transfer(best_seq.copy(), target_sum, scale=0.0025)
            p_score = compute_max_conv(polish)
            if p_score < best_score:
                best_score = p_score
                best_seq = polish.copy()
                cur, cur_score = best_seq.copy(), best_score
                print(f"Polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
