"""
A fast, multi‑start projected‑gradient + simulated‑annealing optimizer.

The objective of evaluate_sequence() is
    2·n·max(conv) / (sum(a))² .
Because the formula is homogeneous we may fix sum(a)=√(2 n); then
the evaluator simply returns max(conv).  The optimizer works on this
simplified objective:

* vectors are kept on the ℓ₁‑simplex Σa = √(2 n) (Euclidean projection).
* a smooth‑max surrogate of max(conv) provides a gradient that is
  computed by FFT‑based convolution.
* sub‑gradient steps on the true max (the anti‑diagonal of the outer‑product)
  are interleaved with the smooth‑max steps to avoid stagnation.
* random “mass‑swap’’ moves and cheap targeted moves (moving mass from the
  largest contributing pair for the current maximal entry toward the
  smallest) act as a simulated‑annealing layer.
* The algorithm repeatedly restarts from several simple parametric families
  (exponential, linear, random Dirichlet) and keeps the overall best
  sequence found before the time budget expires.

Running time ~1 000 s, the routine returns the best sequence as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex target sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i ∝ exp(alpha·conv_i)."""
    scaled = alpha * conv
    m = np.max(scaled)
    exp_shift = np.exp(scaled - m)
    w = exp_shift / exp_shift.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇ of Σ w_k·(a*a)[k]; uses correlation via FFT."""
    n = a.shape[0]
    # reverse w because correlation = a ⋆ w
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # pick the right slice (see _gradient_step implementation)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the true max (exact)
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray):
    """Return the sub‑gradient of max_k (a*a)[k] and the full convolution."""
    conv = _full_convolution(a)
    k = int(np.argmax(conv))
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1)
    # off‑diagonal contributions appear twice → factor 2
    grad[idx] = 2.0 * a[k - idx]
    # diagonal term (i=j) should be counted only once
    if k % 2 == 0:
        mid = k // 2
        if i_start <= mid <= i_end:
            grad[mid] -= a[mid]          # correct from 2·a[mid] to a[mid]
    return grad, conv


# ----------------------------------------------------------------------
# One gradient step on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_gradient_step(a: np.ndarray, lr: float, alpha: float):
    """Projected gradient descent step on the smoothed max‑norm."""
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Simple parametric families for warm‑starts
# ----------------------------------------------------------------------
def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """Decreasing exponential a_i ∝ q^i with decay factor k_factor/n."""
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    # exact normalisation for the infinite series, then truncated
    one_minus_qn = 1.0 - np.exp(-k_factor)          # = 1 - q**n
    c = S * (1.0 - q) / one_minus_qn
    a = c * (q ** np.arange(n))
    a = _project_simplex(a, S)
    return a


def _linear_decrease(n: int) -> np.ndarray:
    """Straight line from 1 to 0, then projected."""
    S = np.sqrt(2.0 * n)
    a = np.linspace(1.0, 0.0, n)
    a = _project_simplex(a, S)
    return a


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex Σa = √(2·n)."""
    S = np.sqrt(2.0 * n)
    a = rng.random(n)
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Local optimiser – combination of smooth‑max, exact max, swaps
# ----------------------------------------------------------------------
def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate vector a0.  The function returns the best vector
    seen and its max‑convolution value (which equals the evaluator's
    output when Σa = √(2·n)).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # keep the best ever seen
    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    # hyper‑parameters (tuned empirically)
    lr = 0.6
    lr_decay = 0.9997
    alpha = 0.4          # temperature of the smooth‑max
    alpha_mul = 1.008
    swap_prob = 0.22
    swap_frac = 0.06
    target_prob = 0.25
    eps_target = 0.05

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 1) alternate between smooth‑max gradient and exact max sub‑gradient
        # ------------------------------------------------------------------
        if rng.random() < 0.5:
            # smooth‑max step
            a, cur_max = _smooth_gradient_step(a, lr, alpha)
        else:
            # exact max sub‑gradient step
            subgrad, conv = _max_subgradient(a)
            a_new = a - lr * subgrad
            a_new = _project_simplex(a_new, S)
            a = a_new
            cur_max = np.max(conv)

        # ------------------------------------------------------------------
        # 2) accept if we improved the best value
        # ------------------------------------------------------------------
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            last_improve = it

        # ------------------------------------------------------------------
        # 3) random mass‑swap (exploratory)
        # ------------------------------------------------------------------
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis acceptance
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_val:
                        best_val = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ------------------------------------------------------------------
        # 4) cheap targeted move that reduces the current maximal entry
        # ------------------------------------------------------------------
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            src_local = np.argmax(contributions)
            tgt_local = np.argmin(contributions)
            i_src = i_range[src_local]
            i_tgt = i_range[tgt_local]

            if a[i_src] > 0.0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_val:
                        best_val = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ------------------------------------------------------------------
        # 5) schedule updates
        # ------------------------------------------------------------------
        lr *= lr_decay
        alpha *= alpha_mul

        # early stop if no improvement for a while
        if it - last_improve > 8000:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises evaluate_sequence().
    The routine mixes several warm‑starts, a robust projected‑gradient
    optimiser with sub‑gradient steps, random mass swaps and targeted moves.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.5   # safety margin

    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------------
    # 0) initialise from the previously submitted sequence, if any
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                n0 = prev.size
                S0 = np.sqrt(2.0 * n0)
                prev = _project_simplex(prev, S0)
                cur = np.max(_full_convolution(prev))
                best_seq = prev.copy()
                best_val = cur
                print(f"[init] using previous solution, n={n0}, val={cur:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1) warm‑start sweep (very cheap)
    # --------------------------------------------------------------
    for n in (800, 1200, 2000, 3000, 4000):
        if time.time() >= deadline:
            break
        for kf in np.linspace(0.6, 2.5, 8):
            a = _exponential_sequence(n, kf)
            val = np.max(_full_convolution(a))
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[warm] n={n:4d} k={kf:4.2f} val={val:.6f}")

    # --------------------------------------------------------------
    # 2) main stochastic optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # choose a length – bias toward the current best length
        if best_seq is not None and rng.random() < 0.55:
            n = len(best_seq)
        else:
            n = rng.integers(800, 8000)

        # pick an initialisation method
        init_mode = rng.choice(['exponential', 'linear', 'random', 'uniform'])
        if init_mode == 'exponential':
            kf = rng.uniform(0.4, 2.5)
            a0 = _exponential_sequence(n, kf)
        elif init_mode == 'linear':
            a0 = _linear_decrease(n)
        elif init_mode == 'random':
            a0 = _random_dirichlet(n, rng)
        else:  # uniform
            S = np.sqrt(2.0 * n)
            a0 = np.full(n, S / n)

        # occasional jitter if we start from the current best
        if best_seq is not None and best_seq.size == n and rng.random() < 0.5:
            jitter = rng.normal(scale=0.001 * a0.mean(), size=n)
            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))

        # allocate a time slice (≈ 25 s) for local optimisation
        slice_deadline = min(deadline, time.time() + 25.0)
        a_opt, val_opt = _local_optimize(a0, slice_deadline, rng)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[global] new best n={n:4d} val={best_val:.6f}")

    # --------------------------------------------------------------
    # 3) final refinement on the best candidate (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0 and best_seq is not None:
        # use a very low learning rate and a longer run to polish
        final_deadline = deadline
        best_seq, best_val = _local_optimize(best_seq, final_deadline, rng)
        print(f"[refine] final refined val={best_val:.6f}")

    # --------------------------------------------------------------
    # 4) final sanity check via the external evaluator
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: a simple uniform vector of length 1000
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}   length = {len(final_seq)}")
    return final_seq
