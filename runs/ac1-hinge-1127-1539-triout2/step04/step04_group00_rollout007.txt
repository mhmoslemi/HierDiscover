def _optimize_adam(a0, deadline, rng):
   a = a0.copy()
   n = len(a)
   S = np.sqrt(2.0 * n)
   # Adam hyperparameters: lr, beta1, beta2, eps
   lr = 0.1
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8

   m = np.zeros_like(a)
   v = np.zeros_like(a)
   t = 0
   best_a = a.copy()
   # compute initial max
   best_score = np.max(_full_convolution(a))

   alpha = 1.0  # start temperature
   while time.time() < deadline:
        t += 1
        # compute gradient for current alpha
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)
        # Adam update:
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)
        # Evaluate max again
        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # schedule alpha
        alpha = min(alpha * 1.01, 200.0)  # increase temperature gradually

        # random mass swap occasionally
        if rng.random() < 0.04:
            # pick two indices
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.01  # small
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
        # early break
        if t % 5000 == 0 and best_score == cur_max:
            break

   return best_a, best_score

* Work on the L¹‑simplex Σ a_i = √(2·n).  The objective is exactly the max‑entry
  of the full convolution a * a, therefore we minimise that value.
* Two‑phase optimisation for each trial:
    1. **Adam** on a smooth soft‑max approximation of the max‑norm.
       The gradient is obtained via a weighted convolution (FFT).  After each
       Adam step the vector is projected back onto the simplex.
    2. **Projected sub‑gradient** that directly attacks the current worst
       convolution entry.
* The outer loop performs many short trials (≈15 s each) with different
  problem sizes.  Successful trials are up‑sampled to larger ``n`` to give
  the optimiser more degrees of freedom.
* Random “mass‑swap’’ and jitter moves are injected occasionally to escape
  plateaus.
* The best sequence seen over the whole budget is finally evaluated with the
  supplied ``evaluate_sequence`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimisation on the smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max approximation of max(conv). Returns best vector."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.3
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5          # temperature – starts smooth
    while time.time() < deadline:
        t += 1
        # ---- gradient of smooth max ----
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the simplex ----
        a = _project_simplex(a, S)

        # ---- occasional random mass‑swap (exploration) ----
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- temperature schedule (make the soft‑max sharper) ----
        alpha = min(alpha * 1.02, 250.0)

        # cheap early‑stop test
        if t % 5000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst shift."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.25  # initial step size constant
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # construct sub‑gradient: g_i = a_{k_max-i} (zero outside overlap)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early‑stop
        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run Adam for ≈60 % of the allocated time, then finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, sc_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine mixes Adam‑based smooth optimisation,
    sub‑gradient refinement, random restarts and progressive up‑sampling.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main optimisation loop (many short trials)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # decide on problem size
        if best_seq is None or (time.time() - last_improve > 40):
            # fresh random start, moderate size
            n = int(rng.integers(800, 3000))
            a0 = rng.random(n)
        else:
            # sometimes try a larger version of the current best
            if rng.random() < 0.25 and len(best_seq) < 8000:
                n = min(int(len(best_seq) * 1.5), 8000)
                a0 = _upsample(best_seq, n)
            else:
                n = len(best_seq)
                a0 = best_seq.copy()

            # add a tiny jitter to avoid stagnation
            jitter = rng.normal(scale=a0.mean() * 1e-4, size=n)
            a0 = a0 + jitter
            a0 = np.clip(a0, 0.0, None)

        # projection onto the correct simplex (Σa = √(2·n))
        S = np.sqrt(2.0 * n)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, 1.0)
            a0_sum = a0.sum()
        a0 = a0 / a0_sum * S

        # allocate a time slice for this trial (≈15 s)
        slice_deadline = min(time.time() + 15.0, deadline)

        # run the combined optimiser
        a_opt, score_opt = _optimize_combined_adam(a0, slice_deadline, rng)

        # keep global best
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
