def _lp_refine(a, time_limit):
    deadline = time.time() + time_limit
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    t = 0.2
    while time.time() < deadline:
        M = _max_convolution_fft(a)
        # Build A matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        # fill
        for k in range(2*n-1):
            i_min = max(0, k - (n-1))
            i_max = min(k, n-1)
            j = np.arange(i_min, i_max+1)
            A[k, i_min:i_max+1] = a[k - j]  # because a[k-j]
        # LP solve
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= target_sum * (1 + 1e-12):
            # can't increase sum under same bound, stop
            break
        # scale to target sum
        g_scaled = g * (target_sum / sum_g)
        M_g = _max_convolution_fft(g_scaled)

        if M_g < M:
            a = g_scaled
            continue
        # else mix
        a = (1 - t) * a + t * g_scaled
        # enforce simplex (should remain sum = target_sum but floating errors)
        a = _project_to_simplex(a, target_sum)
        # maybe adjust t gradually
        t = min(0.5, t * 1.05)   # increase
        # stop condition if no improvement after few iterations
        new_M = _max_convolution_fft(a)
        if new_M >= M - 1e-8:
            break
    return a

def _lp_refine(a0: np.ndarray, remaining_time: float) -> np.ndarray:
    deadline = time.time() + remaining_time
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    # Step size mixing parameter
    t = 0.3
    # limit iterations
    iter_max = 15
    for iter_i in range(iter_max):
        if time.time() >= deadline:
            break
        M = _max_convolution_fft(a)
        # Build constraint matrix
        A = _build_constraint_matrix(a)
        # solve LP
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        # bounds
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            # fallback to break
            break
        if not res.success:
            # maybe try different tolerance
            break
        g = np.maximum(res.x, 0.0)
        sum_g = g.sum()
        if sum_g <= target_sum * (1 + 1e-12):
            # No increase possible under current M
            break

"""
Minimise the “max‑convolution” objective using an LP‑based
iterative improvement followed by a short simulated‑annealing polishing.
The algorithm works on a simplex of fixed ℓ¹‑norm
    Σ a_i = √(2 n)  (n = length of the vector).
For a given a we solve a linear programme

    maximise Σ g_i
    subject to a ⋆ g ≤ M   (M = max(a⋆a)),   g ≥ 0

where a⋆g denotes the linear convolution.
If the feasible total mass Σ g_i is larger than Σ a_i we can move part
of that mass into a (by a convex combination) and strictly reduce M.
Repeating this step drives the maximal convolution entry down.
Finally a tiny simulated‑annealing (SA) phase cleans up any remaining
irregularities.

The code respects a user‑supplied time budget (default 1000 s) and never
does file/network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
    _SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    linprog = None
    toeplitz = None
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex Σx = s, x_i ≥ 0 (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers – a light‑weight mix of families
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families (power‑law, exponential, Dirichlet,
    edge‑heavy, uniform) are mixed to increase diversity.
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                           # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                         # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                         # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                         # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    elif kind == 4:                         # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                                   # plain uniform + jitter
        a = np.full(n, target_sum / n)

    # small Gaussian perturbation and re‑projection
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Build the linear‑program constraint matrix for a given a:
#   (a⋆g)_k = Σ_i a_i g_{k-i}  ≤  M   for all k=0..2n‑2
# ----------------------------------------------------------------------
def _build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix A with A[k,i] = a_{k‑i} (or 0 if out of range)."""
    n = a.shape[0]
    if toeplitz is not None:               # fast construction via SciPy if possible
        col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
        row = np.concatenate([[a[0]], np.zeros(n - 1, dtype=a.dtype)])
        return toeplitz(col, row)
    # fallback dense construction (pure NumPy)
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=a.dtype)
    for k in range(K):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        j = np.arange(i_min, i_max + 1)
        A[k, i_min:i_max + 1] = a[k - j]
    return A


# ----------------------------------------------------------------------
# Linear‑program based refinement of a candidate vector
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Iteratively improve a0 using the LP described in the paper
    “maximise Σ g_i subject to a⋆g ≤ M”.  The loop stops when
    (i) time runs out, (ii) the bound M cannot be lowered further,
    or (iii) the LP cannot increase the feasible total mass.
    """
    if not _SCIPY_AVAILABLE:                # pragma: no cover
        return a0.copy()

    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    t = 0.3                                 # mixing parameter
    # limit the number of LP iterations to keep the run time bounded
    max_iters = 12

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution_fft(a)
        # build constraint matrix A (shape (2n‑1) × n)
        A = _build_constraint_matrix(a)

        # linear programme: maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:                  # pragma: no cover
            break

        if not res.success:
            break

        g = np.maximum(res.x, 0.0)          # numerical safety
        sum_g = g.sum()

        # If the LP cannot increase the total mass under the same bound,
        # we are already at (or very close to) the local optimum.
        if sum_g <= target_sum * (1.0 + 1e-12):
            break

        # Scale g to have the same ℓ¹‑norm as a (required for convex mixing)
        g_scaled = g * (target_sum / sum_g)

        M_g = _max_convolution_fft(g_scaled)

        # If the scaled vector already improves the bound, accept it directly.
        if M_g < M - 1e-12:
            a = g_scaled
            # after a direct replacement the bound dropped – we can try again.
            continue

        # Otherwise perform a convex update and re‑project onto the simplex.
        a = (1.0 - t) * a + t * g_scaled
        a = _project_to_simplex(a, target_sum)

        # adapt mixing coefficient (slowly become more aggressive)
        t = min(0.7, t * 1.07)

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (identical to the previous version)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Fine‑tune a candidate using a fast SA that moves random mass fractions."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter (helps escape deep traps)
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best

# Main entry point – the function called by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    the competition objective.  The routine uses:

    1.   A small collection of diverse random starts (projected onto the
         ℓ¹‑simplex Σa = √(2 n)).
    2.   LP‑based iterative improvement that monotonically reduces the
         maximal convolution entry.
    3.   A brief simulated‑annealing polishing phase.

    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is available, start from it.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        a0 = np.asarray(prev, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        obj0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_obj = obj0
        print(f"[{time.time():.1f}] Init from global height_sequence_1 "
              f"(n={n0})  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a modest spread keeps the LP solves cheap.
    # ------------------------------------------------------------------
    lengths = [800, 1000, 1200, 1500, 1800, 2000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        # a short pool of promising seeds for this length
        seed_pool = []

        # --------------------------------------------------------------
        # Phase A – generate a few diverse starts and keep the best few.
        # --------------------------------------------------------------
        phase_a_end = min(deadline, time.time() + 6.0)
        while time.time() < phase_a_end:
            a_rand = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a_rand)

            if score < best_obj:

            # keep up to 5 elite seeds

        # --------------------------------------------------------------
        # Phase B – LP refinement of the elite seeds + brief SA polish.
        # --------------------------------------------------------------
        for _, a_seed in seed_pool:
            if time.time() >= deadline:

            # allocate a slice of the remaining budget to this candidate

            # 1) LP‑based improvement
            a_lp = _lp_refine(a_seed, sub_dead)

            # 2) short SA polishing (use whatever time is left)
            a_sa = _simulated_annealing(a_lp, deadline)

            obj = _max_convolution_fft(a_sa)
            if obj < best_obj:
                best_obj = obj
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n}  obj={best_obj:.6f}")

        # clear pool for the next length
        seed_pool.clear()

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Remove trailing negligible entries for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final sanity check with the official evaluator (cheap)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {final_val:.6f}")
    except Exception:                     # pragma: no cover
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max conv) = {best_obj:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
