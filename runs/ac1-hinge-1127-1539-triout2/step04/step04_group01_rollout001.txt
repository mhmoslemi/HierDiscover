"""
Step‑function optimizer with a linear‑programming refinement.

The search proceeds in four stages:

1. **Random warm‑starts** – draw many points on the ℓ¹‑simplex
   (different families such as power‑law, exponential, Dirichlet, …).

2. **Projected gradient descent** on a smooth approximation of the
   max‑convolution (log‑sum‑exp).  This rapidly lowers the peak
   autocorrelation.

3. **LP refinement** – given the current sequence `a`, solve the linear
   program

        maximise   ∑ g_i
        subject to  g_i ≥ 0
                    (a * g)[k] ≤ M   for all k,
        where M = max(a * a).

   The optimal `g` has a larger ℓ¹‑norm than `a`.  After scaling `g`
   to the required sum and mixing it with `a` the maximal convolution
   usually drops.  A few iterations of this step are cheap (the LP
   has O(n²) data but only O(n) variables).

4. **Optional simulated‑annealing polishing** – a cheap mass‑transfer
   SA finishes the remaining time budget.

All sequences are projected onto the simplex
    {x | x_i ≥ 0, Σ x_i = √(2 n)}
so the official evaluator `evaluate_sequence` simply returns the
maximum entry of the self‑convolution.  The routine respects the
1000 s wall‑clock limit and returns the best list it found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation: max(conv(a,a))."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) and projected gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)      where   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities p
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smoothed maximum.
    Uses momentum, an adaptive step size and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.20
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisation on the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the simplex
    Σ a = √(2 n).  Several families (power‑law, exponential,
    mixed, Dirichlet, piecewise exponential) are sampled.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # uniform plus tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a random fraction of mass
    from one index to another.  Accepts uphill moves with probability
    exp(-Δ/T) where the temperature decays geometrically.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < math.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

except Exception:            # pragma: no cover

def _lp_refine(a: np.ndarray, deadline: float, max_iters: int = 5) -> np.ndarray:
    """
    Linear‑programming refinement.
    At each iteration we solve:
        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (a * g)[k] ≤ M  ∀k,
    where M = max(a * a).  The solution g is scaled to the required
    ℓ¹‑norm and mixed (50 %–50 %) with the current a.
    """
    if not _SCIPY_AVAILABLE:
        print("SciPy not available – skipping LP refinement.")
        return a

    n = a.size
    target_sum = math.sqrt(2.0 * n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution_fft(a)

        # build constraint matrix A where A[k, j] = a_{k-j} if 0≤k-j<n else 0
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        for j in range(n):
            A[j:j + n, j] = a

        # LP: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        b_ub = M * np.ones(rows)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # scale to the required ℓ¹‑norm
        scale = target_sum / sum_g
        g_scaled = g * scale

        # convex combination and projection
        a = 0.5 * a + 0.5 * g_scaled
        a = _project_to_simplex(a, target_sum)

        new_M = _max_convolution_fft(a)
        print(f"[{time.time():.1f}] LP iter {it}: max conv = {new_M:.6f}")

        # stop early if improvement stalls
        if new_M >= M * 0.9995:
            break

    return a

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously found construction, if present.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, math.sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a moderate spread around the known good size.
    # ------------------------------------------------------------------
    lengths = [1200, 1600, 2000, 2400, 3000]

    # main search loop over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Phase A – many cheap random restarts with a short GD refinement
        # --------------------------------------------------------------
        phase_end = min(deadline,
                        time.time() + min(30.0,
                                          (deadline - time.time()) / 4.0))
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)

            sub_dead = min(deadline, time.time() + 3.5)
            a_gd = _gradient_descent_opt(a0, sub_dead)

            sc = _score(a_gd)
            if sc < best_score:
                best_score = sc
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase B – deep refinement of the best candidate for this n
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            sub_dead = min(deadline,
                           time.time() + min(120.0, deadline - time.time()))
            a_ref = _gradient_descent_opt(best_seq, sub_dead)
            sc = _score(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined best (length {n}) score={sc:.6f}")

            # LP refinement (if we still have time)
            sub_dead = min(deadline,
                           time.time() + min(30.0, deadline - time.time()))
            a_lp = _lp_refine(best_seq, sub_dead, max_iters=6)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
                print(f"[{time.time():.1f}] LP refined best (length {n}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time is left
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 8.0:
        a_final = _simulated_annealing(best_seq, deadline)
        sc = _score(a_final)
        if sc < best_score:
            best_score = sc
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for a tidy output
    # ------------------------------------------------------------------
    if best_seq is None:                     # fallback – should never happen
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Report the final objective using the official evaluator
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:                # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
