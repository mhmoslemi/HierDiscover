    n = len(a)
    best_M = _max_convolution_fft(a)
    t = 0.3
    max_iters = 100

"""
Propose a step‑function (non‑negative sequence) that minimises the
competition objective.

The search consists of three layers:

1. **Diverse random starts** – generate candidates on the ℓ¹‑simplex
   (Σ a = √(2 n)) using several families (power‑law, exponential,
   Dirichlet, edge‑heavy, etc.).

2. **Linear‑programming refinement** – for a current sequence *f* we
   solve the LP  

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k,
                    g_j ≥ 0,

   where *M* = max_k (f*f)_k.  The LP pushes mass to positions where the
   convolution bound is not active.  The resulting *g* is blended with
   *f* (convex combination) and projected back onto the ℓ¹‑simplex.
   Repeating this step (with a slowly decreasing step‑size) monotonically
   reduces the maximum self‑convolution.

3. **Local polishing** – a fast simulated‑annealing routine makes small
   random mass‑transfers between indices, accepting uphill moves with a
   decaying temperature.  This removes the last irregularities.

All heavy work is done with NumPy and SciPy (the “highs” LP solver).
The routine respects a user supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities (linear convolution via FFT)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (several families) – always projected onto the simplex
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Create a non‑negative sequence on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:               # pure decreasing power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:             # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:             # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:             # Dirichlet (uniform simplexes)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif kind == 4:             # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                       # flat plus tiny jitter
        a = np.full(n, target / n)

    # small Gaussian perturbation and final projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Linear‑programming based refinement
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the (2·n‑1) × n matrix A such that A @ g = f * g (linear convolution).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f shifted by i positions
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _lp_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_outer_iters: int = 40,
    start_step: float = 0.5,
    min_step: float = 0.02,
) -> np.ndarray:
    """
    Iteratively improve `a0` by solving a linear programme and mixing.
    The LP maximises Σ g_j under the convolution bound derived from the
    current maximum self‑convolution M = max (a * a).  The mixing weight `t`
    is reduced when a step does not improve the objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_M = _max_convolution_fft(a)
    t = start_step
    it = 0
    while time.time() < deadline and it < max_outer_iters:
        M = best_M
        # Linear programme: maximise Σ g_j  s.t. A @ g <= M, g >= 0
        A = _build_convolution_matrix(a)
        c = -np.ones(n)                     # maximise → minimise -sum
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        # Solve with HiGHS (fast interior‑point / simplex)
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
        if not res.success:
            # LP failed – fall back to gradient descent later
            break

        g = res.x
        # Normalise g to the required ℓ¹‑norm (convex combination will keep the sum)
        g = _project_to_simplex(g, target_sum)

        # Convex combination and re‑project (keeps Σa = target_sum)
        a_candidate = (1.0 - t) * a + t * g
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        new_M = _max_convolution_fft(a_candidate)

        # Accept if we made progress, otherwise shrink the step size
        if new_M < best_M - 1e-7:
            a = a_candidate
            best_M = new_M
            # after a good move we can be a bit more aggressive
            t = min(t * 1.08, start_step)
        else:
            t = max(t * 0.7, min_step)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (same as in the previous solution)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a random fraction of mass
    between two indices while accepting uphill moves with probability
    exp(-Δ/T).  Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    competition objective.  The routine runs for at most ``budget_s`` seconds,
    exploring several vector lengths, generating diverse random starts,
    applying a linear‑programming refinement, and finishing with a short
    simulated‑annealing polishing.  The best sequence found is returned.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin
    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # If a global construction is already present, use it as a seed
    # ------------------------------------------------------------------
    init_global = globals().get("height_sequence_1")
    if isinstance(init_global, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_global, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        obj0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_obj = obj0
        print(f"[{time.time():.1f}] Seeded with global height_sequence_1 (n={n0}) obj={obj0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – a modest spread keeps the LP sizes tractable
    # ------------------------------------------------------------------
    lengths = [800, 1000, 1200, 1500, 1800, 2000]

    # we keep a small pool of promising seeds per length
    pool_size = 5

    for n in lengths:
        if time.time() >= deadline:
            break

        target = np.sqrt(2.0 * n)
        pool = []   # list of (score, array)

        # --------------------------------------------------------------
        # Phase 1 – random sampling (≈ 6 s per length)
        # --------------------------------------------------------------
        phase_end = min(deadline, time.time() + 6.0)
        while time.time() < phase_end:
            a_rand = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a_rand)
            if score < best_obj:
                best_obj = score
                best_seq = a_rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n}  obj={best_obj:.6f}")

            # maintain a short elite pool
            if len(pool) < pool_size or score < pool[-1][0]:
                pool.append((score, a_rand.copy()))
                pool.sort(key=lambda x: x[0])
                pool = pool[:pool_size]

        # --------------------------------------------------------------
        # Phase 2 – LP refinement + SA polishing for each elite seed
        # --------------------------------------------------------------
        for _, seed_seq in pool:
            if time.time() >= deadline:
                break

            # allocate a slice of the remaining budget for this candidate
            sub_dead = min(deadline, time.time() + 30.0)

            # ----- LP refinement -------------------------------------------------
            a_lp = _lp_refine(seed_seq, target, sub_dead)

            # ----- Simulated annealing polishing ----------------------------------
            a_sa = _simulated_annealing(a_lp, deadline)

            final_score = _max_convolution_fft(a_sa)
            if final_score < best_obj:
                best_obj = final_score
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n}  obj={best_obj:.6f}")

        # clear the pool before the next length
        pool.clear()

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing near‑zero entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # One final sanity call (cheap)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {final_val:.6f}")
    except Exception:
        pass

    print(
        f"[{time.time():.1f}] Finished. Best objective (max conv) = {best_obj:.6f} "
        f"length = {best_seq.size}"
    )
    return best_seq.tolist()
