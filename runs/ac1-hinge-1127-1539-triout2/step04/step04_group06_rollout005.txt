def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact evaluation of the competition objective for a normalised sum."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient (w.r.t. a) of the soft‑max approximation
          (1/β)·log Σ exp(β·(a*a)[k])
    and the raw convolution values.
    """
    n = a.shape[0]
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max probabilities
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a_rev * p)  (convolution with reversal)
    a_rev = a[::-1]
    size2 = 1 << ((2 * n - 1).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # keep only the n relevant entries
    return grad, b


# ----------------------------------------------------------------------
# Optimiser – Adam in log‑space on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_log_softmax_opt(a0: np.ndarray, deadline: float,
                          beta_start: float = 1.0,
                          beta_max: float = 1024.0,
                          beta_factor: float = 1.03,
                          lr: float = 0.08,
                          eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on log‑variables.
    Returns the best sequence (projected onto the simplex) seen before *deadline*.
    """
    n = a0.size
    target = np.sqrt(2.0 * n)

    # initialise log‑variables
    x = np.log(np.maximum(a0, eps))

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    t = 0

    best_a = a0.copy()
    best_val = _score(best_a)

    beta = beta_start

    while time.time() < deadline:
        # reconstruct a and re‑project to keep the ℓ¹‑norm exact
        a = np.exp(x)
        a = _project_to_simplex(a, target)

        # soft‑max gradient w.r.t. a
        g_a, _ = _gradient_softmax(a, beta)

        # chain rule: ∂/∂x = (∂/∂a)·a
        g_x = g_a * a

        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam update in log‑space
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase the soft‑max sharpening
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # evaluate the *real* objective (no smoothing)
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (same spirit as the reference)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.03,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 24) -> tuple[np.ndarray, float]:
    """
    Lightweight SA that moves mass between indices and occasionally smooths
    short windows.  The ℓ¹‑norm is kept constant by construction.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # keep current convolution for cheap reuse
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time + 1e-12)
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current worst convolution entry
        k_max = int(np.argmax(conv))
        # sensitivity vector : a[k_max-i] where defined
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        # ---- propose a move -------------------------------------------------
        if rng.random() < 0.70:                     # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_idx)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                                        # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # ---- evaluate new state (single FFT) --------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)           # cheap because sum unchanged

        # ---- Metropolis acceptance -------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:
                a[saved_start:saved_end] = saved_seg
            # conv unchanged

        # occasional jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine combines several random restarts of a log‑space Adam optimiser
    with a final simulated‑annealing polish.  It respects the time budget
    (leaves a small safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Initialise from the global best if it exists, otherwise a power law.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n0 = init_seq.size
        target = np.sqrt(2.0 * n0)
        if init_seq.sum() > 0:
            init_seq = init_seq * (target / init_seq.sum())
        else:
            init_seq = _power_law_sequence(n0, 0.55, target)
        init_seq = _project_to_simplex(init_seq, target)
        best_seq = init_seq.copy()
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] start from global seq, score={best_score:.6f}")
    else:
        n0 = 1500
        target = np.sqrt(2.0 * n0)
        best_seq = _power_law_sequence(n0, 0.55, target)
        best_seq = _project_to_simplex(best_seq, target)
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] start from power‑law, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – random restarts of Adam + short SA.
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:          # keep ~30 s for final polish
        # pick a length (keep it reasonable for the FFT size)
        n = rng.integers(1000, 2501)

        # random power‑law exponent and a little noise
        p = rng.uniform(0.35, 0.90)
        target = np.sqrt(2.0 * n)
        seq = _power_law_sequence(n, p, target)

        # add a tiny Gaussian perturbation and re‑project
        seq += rng.normal(scale=seq.mean() * 0.02, size=n)
        seq = np.maximum(seq, 0.0)
        seq = _project_to_simplex(seq, target)

        # --------------------------------------------------------------
        # Adam optimisation (log‑space) – allocate up to 35 s per restart
        # --------------------------------------------------------------
        remain = deadline - time.time()
        if remain <= 0:
            break
        sub_deadline = time.time() + min(35.0, remain / 2.0)

        seq_adam, score_adam = _adam_log_softmax_opt(seq, sub_deadline)
        if score_adam < best_score:
            best_seq, best_score = seq_adam, score_adam
            print(f"[{time.time():.1f}] NEW GLOBAL best after Adam = {best_score:.6f}")

        # --------------------------------------------------------------
        # Quick SA polish on the result of Adam – use remaining slice
        # --------------------------------------------------------------
        remain = deadline - time.time()
        if remain <= 5.0:
            break
        sa_deadline = time.time() + min(15.0, remain - 5.0)
        seq_sa, score_sa = _simulated_annealing(seq_adam, sa_deadline)
        if score_sa < best_score:
            best_seq, best_score = seq_sa, score_sa
            print(f"[{time.time():.1f}] NEW GLOBAL best after SA = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polish on the absolute best sequence with the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _simulated_annealing(best_seq,
                                                      deadline)
        if final_score < best_score:
            best_seq, best_score = final_seq, final_score
            print(f"[{time.time():.1f}] FINAL polish improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros (purely cosmetic)
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] DONE – best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
