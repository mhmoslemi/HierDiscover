"""
propose_candidate.py

A population‑based optimiser for the max‑convolution / ℓ¹‑norm benchmark.

Algorithm
---------
1. **Simplex normalisation** – all candidates are projected onto the
   simplex Σa = √(2·n) (the normalisation used by the benchmark).

2. **Differential Evolution (DE)** – a small population (default 30)
   evolves with the classic “DE/rand/1/bin” scheme.
   *Mutation* :  v = a_r1 + F·(a_r2 − a_r3)  (F≈0.8)  
   *Crossover*: binomial with rate Cr≈0.9.  
   After crossover the candidate is clipped to non‑negative values and
   projected back to the simplex.

   The fitness is the exact benchmark score (computed with an FFT‑based
   convolution, i.e. 2·n·max(a∗a) / (∑a)²).  Whenever a trial is better
   than its parent it replaces it, and the global best is memorised.

   To keep diversity we periodically replace the worst half of the
   population by fresh random Dirichlet samples.

3. **Local mass‑transfer hill‑climb** – after DE finishes, a fast
   deterministic refinement is applied.  The current convolution peak
   is located; the contribution of each coordinate to that peak is
   computed (the “sensitivity”).  A small fraction of mass is moved
   from a high‑sensitivity donor to a low‑sensitivity receiver.  Moves
   are accepted only if they improve the score.

4. **Result post‑processing** – trailing entries that are (near) zero
   are stripped to keep the representation compact.

The whole search runs within the supplied time budget (default 1000 s)
and returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level to satisfy the execution rules.
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray, size_fft: int) -> float:
    """Maximum entry of the linear convolution a*a using an FFT of given size."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * a.size - 1]
    return float(conv.max())


def _score_fft(a: np.ndarray, size_fft: int) -> float:
    """Exact benchmark score for a normalised vector (no clipping needed)."""
    n = a.size
    max_b = _max_convolution_fft(a, size_fft)
    s = a.sum()
    # safeguard – the optimiser never creates a zero‑sum vector
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _random_simplex_vector(dim: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using a Gamma (Dirichlet) distribution."""
    x = rng.gamma(1.0, size=dim)
    x = x / x.sum() * target_sum
    return x


def _de_optimize(
    n: int,
    target_sum: float,
    size_fft: int,
    deadline: float,
    rng: np.random.Generator,
    init_pop: np.ndarray | None = None,
    pop_size: int = 30,
    F: float = 0.8,
    Cr: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on the simplex.
    Returns the best vector and its score.
    """
    # ------------------------------------------------------------------
    # initialise population
    # ------------------------------------------------------------------
    if init_pop is not None and init_pop.shape == (pop_size, n):
        pop = init_pop.copy()
    else:
        pop = np.empty((pop_size, n), dtype=np.float64)
        # first individual: if a global construction exists we use it,
        # otherwise a random one.
        if init_pop is not None and init_pop.shape == (1, n):
            pop[0] = init_pop[0].copy()
        else:
            pop[0] = _random_simplex_vector(n, target_sum, rng)
        for i in range(1, pop_size):
            pop[i] = _random_simplex_vector(n, target_sum, rng)

    # evaluate initial population
    scores = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        scores[i] = _score_fft(pop[i], size_fft)

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    generation = 0
    # ------------------------------------------------------------------
    # main DE loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        generation += 1
        # occasional diversity injection
        if generation % 50 == 0:
            # replace the worst half by fresh random individuals
            order = np.argsort(scores)
            worst = order[pop_size // 2 :]
            for i in worst:
                pop[i] = _random_simplex_vector(n, target_sum, rng)
                scores[i] = _score_fft(pop[i], size_fft)
                if scores[i] < best_score:
                    best_score = scores[i]
                    best_vec = pop[i].copy()
            continue

        for i in range(pop_size):
            if time.time() >= deadline:
                break

            # select three distinct indices different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

            # mutation
            mutant = pop[r1] + F * (pop[r2] - pop[r3])

            # crossover (binomial)
            cross_mask = rng.random(n) < Cr
            trial = np.where(cross_mask, mutant, pop[i])

            # enforce non‑negativity and simplex
            trial = np.maximum(trial, 0.0)
            trial = _project_to_simplex(trial, target_sum)

            # evaluate
            trial_score = _score_fft(trial, size_fft)

            # selection
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(f"[{time.time():.1f}] DE improved to {best_score:.6f}")

    return best_vec, best_score


def _hill_climb(
    a0: np.ndarray,
    size_fft: int,
    deadline: float,
    rng: np.random.Generator,
    max_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Deterministic local search that repeatedly moves a small fraction
    of mass from a high‑sensitivity donor to a low‑sensitivity receiver.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score_fft(a, size_fft)
    best_vec = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        # full convolution (FFT – cheap)
        conv = _max_convolution_fft(a, size_fft)  # only need the max value
        # locate the index of the current maximum (need the full array)
        fa = np.fft.rfft(a, n=size_fft)
        full_conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        k_max = int(np.argmax(full_conv))

        # sensitivity of each coefficient to the current maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = a[k_max - idx]                     # a_{k*‑i}
        weight = a * sens                         # contribution a_i·a_{k*‑i}

        if weight.max() == 0.0:
            break

        donor = int(np.argmax(weight))

        # choose a receiver among low‑weight positions (10‑percentile)
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = int(rng.choice(candidates))

        if donor == receiver or a[donor] <= 0.0:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        if delta <= 0.0:
            continue

        # perform the transfer (the ℓ¹‑norm stays unchanged)
        a[donor] -= delta
        a[receiver] += delta

        new_score = _score_fft(a, size_fft)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.1f}] HC new best = {best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta

    return best_vec, best_score

        base_seq = None
        n = 2000

    target_sum = np.sqrt(2.0 * n)
    size_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Initialise the first population member from the global best if possible.
    # ------------------------------------------------------------------
    init_pop = None
    if base_seq is not None:
        seq = np.zeros(n, dtype=np.float64)
        seq[: base_seq.size] = base_seq
        seq = _project_to_simplex(seq, target_sum)
        init_pop = np.empty((1, n), dtype=np.float64)
        init_pop[0] = seq

    # ------------------------------------------------------------------
    # DE stage (majority of the time budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    de_deadline = time.time() + max(30.0, remaining * 0.85)   # leave a slice for HC
    best_vec, best_score = _de_optimize(
        n=n,
        target_sum=target_sum,
        size_fft=size_fft,
        deadline=de_deadline,
        rng=rng,
        init_pop=init_pop,
        pop_size=30,
        F=0.8,
        Cr=0.9,
    )
    print(f"[{time.time():.1f}] DE finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Local hill‑climb refinement (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hc_deadline = deadline
        best_vec, best_score = _hill_climb(
            best_vec, size_fft=size_fft, deadline=hc_deadline, rng=rng, max_frac=0.03
        )
        print(f"[{time.time():.1f}] HC finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for a compact representation.
    # ------------------------------------------------------------------
    nz = np.where(best_vec > 1e-12)[0]
    if nz.size:
        best_vec = best_vec[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_vec)}")
    return best_vec.tolist()
