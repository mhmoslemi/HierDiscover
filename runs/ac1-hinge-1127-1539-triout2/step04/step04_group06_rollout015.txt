"""
A multi‑restart optimiser for the max‑convolution / ℓ¹‑norm benchmark.

Strategy
~~~~~~~~
* Randomly generate many power‑law initial sequences of various lengths
  and exponents.
* Refine each sequence with a soft‑max gradient descent (annealed
  β) that directly minimises a smooth approximation of the max‑convolution.
* Follow up with a short simulated‑annealing (SA) phase that moves mass
  between “high‑weight” and “low‑weight” positions and performs local
  smoothing.
* Keep the best sequence seen over the whole time budget (≤ 1000 s).

The objective after normalising the ℓ¹‑norm to √(2 n) is simply the
maximum entry of the self‑convolution, so the code works with the fast
FFT‑based convolution and a cheap internal score function.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog   # optional – not used in the final loop

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of the competition objective for a normalised sum.
    There is no security‑check overhead here – only the mathematical part.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:          # should never happen for our normalised vectors
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max of the max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv_vector).
    """
    n = a.shape[0]
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)                     # soft‑max probabilities

    # gradient = 2 * (a ⊛ p)   (where ⊛ is the linear convolution)
    # we compute it via FFT as well.
    a_rev = a[::-1]
    size2 = 1 << ((2 * n - 1).bit_length())
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    grad_full = np.fft.irfft(a_rev_fft * p_fft, n=size2)[:2 * n - 1]
    grad = 2.0 * grad_full[n - 1: n - 1 + n]    # keep only n components
    return grad, conv


def _annealed_gradient_descent(a0: np.ndarray, deadline: float,
                               max_beta: float = 512.0):
    """
    Projected gradient descent on the soft‑max approximation of max‑convolution.
    The algorithm follows the scheme used in the reference implementation,
    but runs for the whole allocated time slice.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # projection onto the simplex (keeps sum == target_sum)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow minima
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD   new best {best_val:.6f} beta={beta:.2f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.02,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.01,
                         max_window: int = 20) -> tuple[np.ndarray, float]:
    """
    Refines a candidate with a classic SA scheme.
    * donor→receiver mass transfer (probability 0.7)
    * local smoothing (probability 0.3)
    Acceptance follows a Metropolis rule with exponential cooling.
    """
    a = a0.copy()
    n = a.size
    cur_sum = a.sum()
    cur_score = 2.0 * n * _max_convolution_fft(a) / (cur_sum ** 2)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _max_convolution_fft(a)  # we only need the max value for the start

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # --- current state -------------------------------------------------
        conv_full = np.fft.irfft(np.fft.rfft(a, n=size_fft) *
                                 np.fft.rfft(a, n=size_fft), n=size_fft)
        conv_full = conv_full[: 2 * n - 1]
        max_b = conv_full.max()
        k_max = int(np.argmax(conv_full))

        # sensitivity vector for the active index k_max
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # --- propose a move ------------------------------------------------
        if rng.random() < 0.70:          # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low_idx)

            if donor == receiver:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            move_type = "transfer"
        else:                            # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start < 2:
                continue
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved_seg = old_seg
            saved_start, saved_end = start, end

        # --- evaluate new state -------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new / (cur_sum ** 2)

        # --- Metropolis acceptance -----------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA   new best {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            else:  # smooth
                a[saved_start:saved_end] = saved_seg

        # occasional global jitter (helps escape deep minima)
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_sum / a.sum())
            # recompute convolution after a big perturbation
            fa = np.fft.rfft(a, n=size_fft)
            conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
            cur_score = 2.0 * n * conv.max() / (cur_sum ** 2)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm performs many independent restarts, each consisting of:
      1. Random power‑law initialisation (random length & exponent).
      2. Soft‑max gradient descent (annealed β).
      3. Simulated‑annealing refinement.
    The best sequence discovered before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # If a globally supplied construction exists, use it as a first seed.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n0 = seq.size
        target = math.sqrt(2.0 * n0)
        if seq.sum() > 0:
            seq = seq * (target / seq.sum())
        else:
            seq = _power_law_sequence(n0, 0.55, target)
        seq = _project_to_simplex(seq, target)

        # a quick refinement (short GD + SA) to make sure the global seed is
        # at least as good as the baseline we have already.
        sub_deadline = time.time() + min(30.0, deadline - time.time())
        seq, _ = _annealed_gradient_descent(seq, sub_deadline, max_beta=256.0)
        if time.time() < deadline:
            seq, _ = _simulated_annealing(seq, deadline)
        cur_score = _score(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] baseline from global, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – many independent restarts.
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0:      # keep a small safety margin
        restart += 1
        time_left = deadline - time.time()

        # --- random problem size and power‑law exponent -----------------
        n = int(rng.integers(800, 2500))          # length of the step function
        target_sum = math.sqrt(2.0 * n)
        p = float(rng.uniform(0.45, 0.60))        # exponent of the power law

        a = _power_law_sequence(n, p, target_sum)

        # small random jitter before the optimisation phases
        jitter = rng.normal(scale=target_sum * 0.001, size=n)
        a = np.maximum(a + jitter, 0.0)
        a = _project_to_simplex(a, target_sum)

        # -------------------------------------------------------------
        # Phase 1 – soft‑max gradient descent (takes at most ~30 % of the
        # remaining budget for this restart)
        # -------------------------------------------------------------
        gd_budget = min(15.0, 0.3 * time_left)
        gd_deadline = time.time() + gd_budget
        a, _ = _annealed_gradient_descent(a, gd_deadline, max_beta=512.0)

        # -------------------------------------------------------------
        # Phase 2 – simulated annealing for the rest of the restart budget
        # -------------------------------------------------------------
        if time.time() < deadline:
            a, _ = _simulated_annealing(a, deadline)

        # -------------------------------------------------------------
        # Evaluate & possibly update the global best
        # -------------------------------------------------------------
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] restart {restart} – new global best "
                  f"{best_score:.6f} (n={n}, p={p:.3f})")

    # ------------------------------------------------------------------
    # Final tidy‑up: trim trailing zeros (they only waste length).
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = [0.0]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, "
          f"length={len(best_seq)}")
    return best_seq.tolist()
