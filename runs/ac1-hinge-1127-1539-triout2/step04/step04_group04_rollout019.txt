"""
Improved step‑function search.

The core idea is very simple:
1️⃣  Generate many candidate height vectors *without* the aggressive
    simplex‐projection used before (which zeroes out the tiny tail
    entries).  Instead we just scale the raw shape to the required ℓ₁‑norm
    √(2·n).  Families include pure exponentials, mixtures of exponentials,
    power‑law tails, Dirichlet draws and simple cosine/triangular shapes.
2️⃣  Keep the best candidate seen (lowest convolution maximum).
3️⃣  Refine that candidate with an Adam optimiser that minimises a
    soft‑max approximation of the ∞‑norm of the self‑convolution.
    After each Adam step we clip negatives and *renormalise* the vector
    by a simple scaling (preserves every non‑zero entry, unlike the
    previous projection).  This allows the optimiser to explore shapes
    that the earlier projection would discard.
The combination of richer random families and a less destructive
projection yields a lower final score (target < 1.503)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple scaling onto the ℓ₁‑simplex {x ≥ 0, sum = S}
# ----------------------------------------------------------------------
def _scale_to_sum(v: np.ndarray, S: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm S."""
    total = v.sum()
    if total == 0:
        # fallback to a uniform vector
        return np.full_like(v, S / v.shape[0])
    return v * (S / total)


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float):
    """
    Compute gradient of τ·log∑exp(b/τ) where b = a*a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:                     # hard max → sub‑gradient
        k = np.argmax(b)               # index of the maximal entry
        g = np.zeros_like(a)
        # a_i participates if i and k-i are valid indices
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] = 2.0 * a[partner[valid]]
        return g, b, max_b

    # ----- smooth (soft‑max) case ------------------------------------
    # stable soft‑max
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)          # shape (2n‑1,)

    # compute gradient = 2 * a  ⋆  w   (cross‑correlation)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # centre the result
    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with *scaling*‑projection (preserves small tails)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Minimise the soft‑max approximation of max(a*a) using Adam.
    Returns the best vector and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6
    t = 0

    # initialise temperature from a hard max
    _, _, cur_max = _gradient_softmax(a, tau=0.0)
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        # gradient w.r.t. soft‑max (smooth) objective
        g, _, cur_max = _gradient_softmax(a, tau)

        # Adam pre‑projection step
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)          # enforce non‑negativity
        a_new = _scale_to_sum(a_new, S)         # renormalise to the ℓ₁‑simplex

        # evaluate
        cur_max_new = np.max(_full_convolution_fft(a_new))

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # schedules
        tau = max(tau * 0.9995, 1e-7)
        lr *= 0.9999

        # occasional jitter to escape plateaus
        if rng.random() < 0.007:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _scale_to_sum(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Random parametric families (no projection – only scaling)
# ----------------------------------------------------------------------
def _random_family(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a raw non‑negative vector from a randomly chosen parametric family.
    The caller must scale the result to the required ℓ₁‑norm.
    """
    idx = np.arange(n, dtype=float)
    typ = rng.choice(
        ["exp", "exp_mix", "power", "dirichlet", "cosine", "triangular", "quadratic"]
    )

    if typ == "exp":
        lam = rng.uniform(0.001, 0.02)          # slower decay → more mass
        a = np.exp(-lam * idx)

    elif typ == "exp_mix":
        # mixture of a few exponentials
        K = rng.integers(2, 5)                 # 2‑4 components
        a = np.zeros_like(idx)
        for _ in range(K):
            lam = rng.uniform(0.001, 0.02)
            w = rng.random() + 0.1
            a += w * np.exp(-lam * idx)

    elif typ == "power":
        p = rng.uniform(0.0, 1.5)               # 0 → almost uniform, 1.5 → steep
        a = (idx + 1.0) ** (-p)

    elif typ == "dirichlet":
        # concentration governs sparsity/flatness
        concentration = rng.uniform(0.05, 5.0)
        a = rng.dirichlet(alpha=np.full(n, concentration))

    elif typ == "cosine":
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        dec = inc[::-1] if n % 2 == 0 else inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "quadratic":
        # parabola opening downwards, zero at the ends
        a = 1.0 - ((idx - (n - 1) / 2.0) / (n / 2.0)) ** 2
        a = np.clip(a, 0.0, None)

    else:                                     # very unlikely fallback
        a = rng.random(n)

    # a tiny jitter helps diversity; it is applied *before* scaling
    if rng.random() < 0.35:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a
