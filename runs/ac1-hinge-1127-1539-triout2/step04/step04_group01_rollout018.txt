"""
Simple‑search optimiser for the “max‑self‑convolution” problem.

The programme scans a family of power‑law step functions
        a_i ∝ (i+1)^{-p}
for several lengths n and exponents p.  Each candidate is
scaled to satisfy the ℓ¹‑normalisation ∑a_i = √(2 n) that the
official evaluator assumes.  The objective
        2 n·max(a*a) / (∑a)²
coincides with max(a*a) after the normalisation, so we only need
the maximum entry of the linear convolution.  That maximum is
computed with an FFT (O(n log n)) which makes the grid search
very fast.

The search proceeds as follows

1.  A coarse grid of lengths (200 … 2000) and exponents p∈[0.55,0.75]
    is examined.  The best (n,p) pair is remembered.
2.  Around the best pair a finer grid is explored.
3.  If time permits a short simulated‑annealing phase moves a little
    mass between two positions in order to escape possible plateaus.
4.  The final sequence is returned as a plain Python list.  A single
    call to ``evaluate_sequence`` is performed only for logging the
    official score.

The whole routine respects the supplied time budget (default 1000 s)
and never uses more than one CPU core.
"""

import time
import numpy as np
from scipy.special import beta  # beta(x,y) = Γ(x)Γ(y)/Γ(x+y)


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return max entry of the linear convolution a*a using the FFT.
    a is assumed to be one‑dimensional and real.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Candidate generation (power‑law family)
# ----------------------------------------------------------------------


def _make_power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    a_i = C·(i+1)^{-p},  i = 0 … n‑1,
    where the constant C forces   Σ a_i = √(2 n).
    """
    idx = np.arange(1, n + 1, dtype=float)               # 1 … n
    a = idx ** (-p)
    target = np.sqrt(2.0 * n)
    a *= target / a.sum()
    return a


# ----------------------------------------------------------------------
# Very small simulated annealing to polish a candidate
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Mass‑transfer SA; returns the best sequence seen before deadline."""
    a = seq.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(a)

    # temperature – a tiny fraction of the current score
    T = best_score * 0.01 + 1e-12
    decay = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction of the mass from i to j
        frac = rng.random() * 0.12
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - best_score
        accept = dE <= 0.0 or rng.random() < np.exp(-dE / T)

        if accept:
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement → {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= decay

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The strategy is a dense grid search over the
    two‑parameter family a_i ∝ (i+1)^(-p) followed by a brief
    simulated‑annealing polish.  The function respects the given time
    budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) coarse grid search
    # ------------------------------------------------------------------
    lengths = [200, 400, 600, 800, 1000, 1500, 2000]
    p_grid_coarse = np.linspace(0.55, 0.75, 41)   # step ≈0.005

    for n in lengths:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        for p in p_grid_coarse:
            # early‑exit if we run out of time
            if time.time() >= deadline:
                break

            a = _make_power_law_sequence(n, p)
            max_conv = _max_convolution_fft(a)
            # after the normalisation the denominator is exactly 2 n,
            # therefore the objective equals max_conv
            val = max_conv

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New best  n={n:4d} p={p:.4f}  val={val:.6f}")

    # ------------------------------------------------------------------
    # 2) fine search around the best (n,p) found so far
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_n = best_seq.size
        # we do not know the exact p that produced best_seq – recover it
        # from the ratio of a[0] to the ℓ¹ norm (it equals C)
        # and from the analytic form a_i ∝ (i+1)^(-p); we simply
        # re‑search a narrow band of exponents.
        p_fine = np.linspace(max(0.52, 0.55), 0.78, 81)

        for p in p_fine:
            if time.time() >= deadline:
                break
            a = _make_power_law_sequence(best_n, p)
            max_conv = _max_convolution_fft(a)
            val = max_conv
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Fine‑grid improvement n={best_n} p={p:.4f} val={val:.6f}")

    # ------------------------------------------------------------------
    # 3) optional simulated‑annealing polishing (time‑budget aware)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_final = _anneal(best_seq, deadline)
        val = _max_convolution_fft(a_final)
        if val < best_val:
            best_val = val
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final SA polish → {val:.6f}")

    # ------------------------------------------------------------------
    # 4) final sanity checks and logging with the official evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence (should never be needed)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    # remove trailing zeros (only for neatness)
    nz = np.where(best_seq > 1e-14)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # official score (uses the provided evaluate_sequence implementation)
    try:
        official = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {official:.6f} (len={best_seq.size})")
    except Exception as exc:               # pragma: no cover
        print(f"Error during official evaluation: {exc}")

    return best_seq.tolist()
