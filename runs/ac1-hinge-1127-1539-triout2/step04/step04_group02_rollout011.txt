"""
Propose a non‑negative step‑function (height vector) that minimises

        2 * n * max(conv(a, a)) / sum(a)**2

where `conv` is the full linear self‑convolution.  

The algorithm works in three logical layers:

1. **Broad initial sweep** – a few parametric families (U‑shapes,
   skewed exponentials, cosine, power, inverse) and a set of random Dirichlet
   vectors.  All candidates are projected onto the ℓ₁‑simplex
   {x ≥ 0, Σx = √(2 n)} and the best one is kept.

2. **Multiple local optimisations** – from the current best vector we launch a
   large number of short optimisation “restarts”.  Each restart consists of  

   * a *smoothed* gradient descent on the log‑sum‑exp surrogate of the
     maximum (momentum + simple ℓ₁‑simplex renormalisation), followed by  

   * a short *sub‑gradient* phase on the true ℓ∞‑objective, and finally a
     tiny polishing step made of random pair‑mass transfers.

   The restarts explore many basins of attraction; whenever a lower
   maximum‑convolution is discovered the global best is updated.

3. **Final polishing** – a last burst of random pair‑transfers keeps the
   timer until the very end.

All heavy work is performed with FFT‑based convolutions (O(n log n)) and all
updates keep the vector on the ℓ₁‑simplex, ensuring a valid input for
`evaluate_sequence`.  The total running time is limited to the supplied
budget (default 1000 s).
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of `a` with itself using FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def simple_rescale_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Clip negatives, then renormalise to have ℓ₁‑norm equal to `target`.
    This is a cheap alternative to the Euclidean projection and suffices
    for the optimisation loops.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0:
        return np.full_like(v, target / v.size)
    return v * (target / total)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Move a fraction `scale` of the mass from a random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta > 0:
        a[i] -= delta
        a[j] += delta
    return simple_rescale_to_simplex(a, target_sum)


def generate_shape(
    n: int,
    target_sum: float,
    kind: str,
    param: float,
) -> np.ndarray:
    """Create a parametric shape, then project onto the ℓ₁‑simplex."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    elif kind == "cosine":
        raw = 1.0 + param * np.cos(np.pi * idx / (n - 1))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return simple_rescale_to_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the full self‑convolution of `a`."""
    return float(conv_self_fft(a).max())

# Optimisation primitives

def smooth_gradient_descent(
    a0: np.ndarray,
    target_sum: float,
    time_deadline: float,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Log‑sum‑exp smoothed gradient descent with momentum.
    Returns the best vector seen and its maximal convolution.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    n = a.shape[0]
    eta = 0.6 * (target_sum / n)          # initial step size
    beta = 0.9
    tau_factor = 0.20                     # τ = τ_factor * max_conv
    best_vec = a.copy()
    best_score = compute_max_conv(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = conv_self_fft(a)
        max_conv = conv.max()
        tau = max(tau_factor * max_conv, 1e-12)

        # soft‑max weights for the log‑sum‑exp surrogate
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient = (2/τ) * conv(w, a_rev)   (central n entries)
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

        # momentum update
        momentum = beta * momentum - eta * grad
        a = a + momentum
        a = simple_rescale_to_simplex(a, target_sum)

        # occasionally inject a tiny jitter to avoid stagnation
        if it % 1500 == 0:
            eta *= 0.92
            tau_factor = max(0.001, tau_factor * 0.92)

        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.006)

        # keep the best vector seen so far
        if max_conv < best_score:
            best_score = max_conv
            best_vec = a.copy()
            # print only occasionally to avoid clutter
            print(f"[{time.time():.2f}]   smooth – iter {it:05d} → {best_score:.6f}")

    return best_vec, best_score


def subgradient_refinement(
    a0: np.ndarray,
    target_sum: float,
    time_deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution objective.
    Returns the best vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    step = 0.04 * (target_sum / n)        # initial step size
    best_vec = a.copy()
    best_score = compute_max_conv(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = conv_self_fft(a)
        max_conv = conv.max()

        # mask of (approximately) maximal entries
        eps = max(1e-12, 1e-9 * max_conv)
        mask = (conv >= max_conv - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0

        a_rev = a[::-1]
        L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        a = a - step * grad
        a = simple_rescale_to_simplex(a, target_sum)

        if max_conv < best_score:
            best_score = max_conv
            best_vec = a.copy()
            print(f"[{time.time():.2f}]   sub‑grad – iter {it:05d} → {best_score:.6f}")

        if it % 500 == 0:
            step *= 0.94

        if np.random.rand() < 0.018:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    return best_vec, best_score


def polish_sequence(
    seq: np.ndarray,
    target_sum: float,
    time_deadline: float,
    trial_scale: float = 0.0025,
) -> tuple[np.ndarray, float]:
    """Final burst of tiny random pair‑transfers."""
    best = seq.copy()
    best_score = compute_max_conv(best)

    while time.time() < time_deadline:
        cand = random_pair_transfer(best.copy(), target_sum, scale=trial_scale)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            print(f"[{time.time():.2f}]   polish → {best_score:.6f}")

    return best, best_score

        # 2) sub‑gradient refinement (uses remaining time of the restart)
        a_refine, score_refine = subgradient_refinement(
            a_smooth, target_sum, sub_deadline, max_iters=3000
        )
        # keep the best of the two phases
        if score_refine < best_score:
            best_score = score_refine
            best_seq = a_refine.copy()
            print(f"[{time.time():.2f}] Restart {restart_counter:03d} → new global best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final polishing
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – final polishing")
    best_seq, best_score = polish_sequence(
        best_seq, target_sum, deadline, trial_scale=0.0015
    )

    print(f"[{time.time():.2f}] Returning best sequence (score={best_score:.6f})")
    return best_seq.tolist()
