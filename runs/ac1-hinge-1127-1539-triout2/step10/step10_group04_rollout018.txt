"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Scan several simple parametric families (exponential, power‑law,
   beta‑shaped, piecewise) and keep the best few candidates.
2. Refine the best candidate with a *projected sub‑gradient descent* on
   the exact objective  f(a)=max_k (a∗a)_k .
   The sub‑gradient is
        g = 2·conv(w , a_rev)   (slice (n‑1)…(2n‑2)),
   where w is the uniform weight on all indices attaining the current
   maximum of the convolution.  After each step the vector is projected
   onto the ℓ¹‑simplex Σa=S (S=√(2·n)).
   A tiny symmetrisation step ``a←(a+a[::-1])/2`` helps to keep the
   solution balanced.
3. A quick deterministic hill‑climb (the same routine that moves mass
   from the worst‑contributing source towards a favourable destination)
   is applied to the exact convolution to polish the result.
4. If time permits a short Adam optimisation on the smooth
   log‑sum‑exp surrogate is performed as a final polish.
5. The whole search respects the global 1000 s wall‑clock budget and
   returns the best sequence found.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast)."""
    b = _full_convolution(a)
    return float(b.max())

def _beta_family(n: int, rng) -> np.ndarray:
    """U‑shaped beta‑distribution on the simplex."""
    # parameters α,β∈[-2,2]; add 1 to avoid negative powers
    alpha = rng.uniform(-2.0, 2.0) + 1.0
    beta = rng.uniform(-2.0, 2.0) + 1.0
    x = np.linspace(0, 1, n, dtype=float)
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1) + 1e-30)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = _project_to_simplex(a, math.sqrt(2.0 * n))
    return a


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a vector on the simplex with a few random families."""
    families = [
        lambda: (np.exp(-rng.exponential(scale=0.015) * np.arange(n))),
        lambda: (np.arange(1, n + 1, dtype=float) ** (-rng.uniform(0.1, 1.5))),
        lambda: (np.arange(1, n + 1, dtype=float) ** (-rng.uniform(0.1, 1.5))
                * np.exp(-rng.exponential(scale=0.015) * np.arange(n))),
        lambda: _beta_family(n, rng),
        lambda: rng.dirichlet(alpha=np.ones(n))
    ]
    idx = rng.integers(len(families))
    a = families[idx]()
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, math.sqrt(2.0 * n))
    return a


def _family_candidates(n: int, rng, top_k: int = 8):
    """Generate a few cheap candidates from simple families."""
    target = math.sqrt(2.0 * n)
    cand = []

    # exponential
    for _ in range(30):
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = _project_to_simplex(a, target)
        cand.append((_max_convolution_fft(a), a))

    # power‑law
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
        a = _project_to_simplex(a, target)
        cand.append((_max_convolution_fft(a), a))

    # mixed
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p) * np.exp(-lam * np.arange(n))
        a = _project_to_simplex(a, target)
        cand.append((_max_convolution_fft(a), a))

    # beta‑shaped
    for _ in range(20):
        a = _beta_family(n, rng)
        cand.append((_max_convolution_fft(a), a))

    # Dirichlet (almost uniform)
    for _ in range(20):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        a = _project_to_simplex(a, target)
        cand.append((_max_convolution_fft(a), a))

    cand.sort(key=itemgetter(0))
    return cand[:top_k]


# ----------------------------------------------------------------------
# Sub‑gradient optimisation (exact max)
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=max_k (a∗a)_k .
    A tiny symmetrisation step keeps the solution balanced.
    """
    a = a0.copy()
    n = a.size
    t = 0
    # base learning‑rate – empirically 0.4·S works well
    base_lr = 0.4 * target_sum

    while time.time() < deadline:
        t += 1
        # full convolution and current maximum
        b = _full_convolution(a)
        max_b = b.max()
        # active set (indices attaining the max, up to tolerance)
        tol = 1e-12
        active = (b >= max_b - tol).astype(float)
        cnt = active.sum()
        if cnt == 0:
            break
        w = active / cnt                       # uniform weight on active set

        # gradient = 2·conv(w , a_rev)[n-1 : n-1+n]
        a_rev = a[::-1]
        size = _next_pow2(w.size + a_rev.size - 1)
        w_fft = np.fft.rfft(w, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        conv = np.fft.irfft(w_fft * rev_fft, n=size)[: w.size + a_rev.size - 1]
        g = 2.0 * conv[(n - 1):(n - 1 + n)]

        # step size decays like 1/sqrt(t)
        lr = base_lr / math.sqrt(t)
        a -= lr * g

        # keep non‑negativity and project onto ℓ¹‑simplex
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # optional tiny symmetrisation (helps because the problem is
        # symmetric w.r.t. reversal)
        if t % 9 == 0:
            a = (a + a[::-1]) * 0.5
            a = _project_to_simplex(a, target_sum)

    return a

def _hill_climb_deterministic(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the index that contributes most
    to the current worst convolution entry towards the index that hurts it
    the least.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    b = _full_convolution(a)
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))
        # indices that affect b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # pick destination that minimises the counterpart (the term a[M‑j])
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1, dtype=int)]
        order = np.argsort(counterpart[i_low:i_high + 1])
        j = None
        for cand in (i_low + order):
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.7   # a fairly large fraction
        if delta <= 0.0:
            break

        # update convolution (linear O(n) update)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try halving delta a few times
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                bt = b.copy()
                bt[i:i + n] -= 2.0 * delta2 * a
                bt[j:j + n] += 2.0 * delta2 * a
                if k2i < bt.size:
                    bt[k2i] += delta2 * delta2
                if k2j < bt.size:
                    bt[k2j] += delta2 * delta2
                if kij < bt.size:
                    bt[kij] -= 2.0 * delta2 * delta2
                if bt.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = bt
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # escape plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score

def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, then projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new, math.sqrt(2.0 * new_n))
    return a_new

                best_seq = a_adam.copy()
                best_score = sc_adam
                print(f"[{time.time():.1f}] Adam post‑HC improvement (n={n}) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        S = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5
        best_seq = _subgradient_optimize(best_seq, S, final_deadline)
        # final hill‑climb
        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(best_seq, deadline - 0.2)
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and reporting
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    # drop trailing near‑zeros for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # official evaluation
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
