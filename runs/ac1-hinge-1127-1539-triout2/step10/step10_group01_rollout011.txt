def _adam_optimize(a0, target_sum, deadline, max_beta=1e6, step_factor=0.7, gamma=0.99):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 150
    rng = np.random.default_rng()
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)
        # jitter occasionally
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it} β={beta:.1f} score={best_val:.6f}")
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta*1.5, max_beta) # slower growth
        it += 1
    return best_a, best_val
```

"""
Improved step‑function optimizer.

The algorithm follows three phases:

1. **Broad parametric seeding** – generate a diverse set of candidate
   sequences from several families (power‑law, power‑law with exponential
   tail, piecewise‑geometric, uniform).  Keep the best few seeds.

2. **Adam‑style projected gradient descent** – repeatedly optimise each
   seed on the smooth soft‑max approximation of the max‑convolution.
   The optimiser runs until the allotted budget expires, always keeping
   the best sequence seen.

3. **Deterministic mass‑transfer refinement** – after the gradient phase
   we greedily move mass from positions that most contribute to the
   current maximal convolution entry to positions that contribute least,
   directly flattening the convolution peak.  A final short Adam run
   polishes the result.

All operations respect the simplex constraint Σa = √(2 n) and a ≥ 0,
so the objective is scale‑invariant.  The routine obeys the provided
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective (fast version)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective 2·n·max(a*a) / (Σa)².
    Assumes a already satisfies the non‑negativity constraint.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = S}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0, Σx_i = S} (Euclidean)."""
    n = v.shape[0]
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the initial seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _piecewise_geo_sequence(
    n: int, groups: int, decay: float, target_sum: float
) -> np.ndarray:
    """Piecewise constant geometric decay (groups blocks, factor `decay`)."""
    if groups <= 0:
        groups = 1
    block_len = int(np.ceil(n / groups))
    heights = np.array([decay ** i for i in range(groups)], dtype=np.float64)
    seq = np.repeat(heights, block_len)[:n]
    seq *= target_sum / seq.sum()
    return seq


# ----------------------------------------------------------------------
# Soft‑max approximation of max‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of  log‑sum‑exp (soft‑max) approximation of max(conv).
    Returns (grad, full_conv) where full_conv = a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # FFT of a
    a_fft = np.fft.rfft(a, n=size)

    # full convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights over the convolution entries
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)  # p sums to 1, length 2n‑1

    # gradient w.r.t a: 2 * (a_rev * p)_(i) where * denotes convolution
    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    grad_full = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]

    # Only entries that correspond to original a’s indices are needed
    grad = 2.0 * grad_full[n - 1: n - 1 + n]  # shape (n,)
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smooth objective
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_beta: float = 1e6,
    step_factor: float = 0.7,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex.
    The loss is the soft‑max approximation of the max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0                     # start with a mild smoothness
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-12
    it = 0
    beta_inc_iter = 150           # raise beta more often

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1.0 - gamma) * grad
        v = gamma * v + (1.0 - gamma) * (grad * grad)

        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # anneal β (soft‑max becomes closer to true max)
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer phase (greedy peak flattening)
# ----------------------------------------------------------------------


def _mass_transfer_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    tol: float = 1e-15,
) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly reduce the current maximal convolution entry.
    Transfers mass from the most harmful to the least harmful position
    so that the peak is lowered towards the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices achieving (approximately) the maximal value
        idx_max = np.where(conv >= max_val - 1e-14)[0]
        k = idx_max[0]  # pick one of them

        # range of i that contributes to conv[k]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weights w_i = a_i + a_{k-i}
        w = a[idx] + a[k - idx]

        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contributions of donor / receiver to the peak
        contrib_d = a[k - donor]
        contrib_r = a[k - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            break  # cannot improve this peak any further

        # second largest convolution value (excluding every max entry)
        mask = np.ones_like(conv, dtype=bool)
        mask[idx_max] = False
        if np.any(mask):
            second_max = conv[mask].max()
        else:
            break

        # required reduction to bring the peak down to the second max
        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)

        if max_delta <= tol:
            break

        # perform the transfer
        a[donor] -= max_delta
        a[receiver] += max_delta
        # sum is preserved automatically, non‑negativity ensured by the bound

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Mass‑transfer it={int(max_val)} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Initial candidate generation (covers a broad family of shapes)
# ----------------------------------------------------------------------


def _initial_candidates(
    n: int, target_sum: float
) -> tuple[list[np.ndarray], list[float]]:
    """Return a handful of diverse seeds and their scores."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, target_sum / n))

    # 2) pure power‑law grid
    for p in np.linspace(0.1, 1.3, 13):
        cand.append(_power_law_sequence(n, float(p), target_sum))

    # 3) power‑law + exponential tail grid (broader ranges)
    for p in np.linspace(0.1, 1.3, 9):
        for q in np.linspace(0.0, 8.0, 9):
            cand.append(_exp_power_law_sequence(n, float(p), float(q), target_sum))

    # 4) piecewise geometric blocks
    for groups in (5, 10, 20):
        for decay in np.linspace(0.6, 0.95, 8):
            cand.append(_piecewise_geo_sequence(n, groups, float(decay), target_sum))

    # 5) previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        cand.append(prev)

    # evaluate
    scores = [_score(c) for c in cand]

    # keep the top 5 distinct seeds
    idx_sorted = np.argsort(scores)
    top_seq = [cand[i].copy() for i in idx_sorted[:5]]
    top_score = [scores[i] for i in idx_sorted[:5]]
    print(f"[{time.time():.2f}] Initial candidates best score = {top_score[0]:.6f}")
    return top_seq, top_score
