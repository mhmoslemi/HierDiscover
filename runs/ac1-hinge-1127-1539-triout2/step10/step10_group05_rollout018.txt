```python
        a = np.exp(-lam * np.arange(n))
    elif family == 3: # mixed
        p = rng.uniform(0.2,2.0)

        idx = np.arange(1, n+1)
        a = idx ** (-p) * np.exp(-lam * np.arange(n))
```

assistantfinal**Solution Overview**

The score returned by `evaluate_sequence` is  

\[
\text{score}= \frac{2\,n\;\max (a*a)}{(\sum a)^2}.
\]

Because the expression is homogeneous we may fix the ℓ¹‑norm of the
sequence to  

\[
\sum a_i = \sqrt{2\,n},
\]

so that the score is simply the maximal entry of the (linear) convolution
`a*a`.  
The problem is therefore:

*choose a non‑negative vector `a` (length `n`) with a given ℓ¹‑norm so that
the maximal anti‑diagonal sum of the outer product `a aᵀ` is as small as
possible.*

The code below searches this space in three stages:

1. **Large random‑parameter sweep** – many families (power law, exponential,
   mixed, piece‑wise constant, Gaussian bumps, Dirichlet) are sampled.
   The fast FFT based routine `_max_convolution_fft` gives the exact
   maximal convolution entry.  The best candidate is kept.

2. **Projected Adam on a smooth approximation** – the nonsmooth `max`
   is replaced by a log‑sum‑exp surrogate.  Adam with Euclidean projection
   onto the simplex (`Σa = √(2 n)`) is run from each promising start.
   The smoothing parameter `β` is increased gradually, which yields a
   reliable descent even in the highly non‑convex landscape.

3. **Exact deterministic hill‑climb + short annealing** – starting from the
   best Adam‑solution we repeatedly move a small amount of mass from the
   entry that contributes most to the current worst convolution value to
   the entry that reduces it the most.  When no monotone improvement is
   possible a brief simulated‑annealing phase is performed to escape local
   minima.

All stages respect the global wall‑clock budget (default 1000 s).  The
final sequence is trimmed of trailing zeros and returned as a plain list.

```python
"""
Propose a non‑negative step function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm proceeds in three phases:

1.  Random parameter sweep over many families (power‑law, exponential,
    mixed, piece‑wise constant, Gaussian bumps, Dirichlet) – cheap
    FFT‑based evaluation keeps the best few candidates.

2.  Projected Adam optimisation of a smooth log‑sum‑exp surrogate of the
    max‑convolution.  The iterates are projected onto the ℓ¹‑simplex
    Σa = √(2 n).  The smoothing parameter β is increased during the run.

3.  Deterministic hill‑climbing (exact max‑convolution updates) followed
    by a short simulated‑annealing refinement.

The whole search respects a wall‑clock budget (default 1000 s) and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with  g = 2·(a ⋆ p)  and  p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)

    # b = a * a  (full convolution)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution of a_rev and p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]

    # gradient (only the central n entries are needed)
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate scales with the target ℓ¹‑norm
    base_lr = 0.12 * target_sum
    beta = 2.0               # initial smoothing
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # tiny stochastic jitter every few hundred steps
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing slowly
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initial families
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)

    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                           # Dirichlet (uniform)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 1:                         # Power‑law
        p = rng.uniform(0.1, 3.0)
        a = (idx + 1.0) ** (-p)

    elif family == 2:                         # Exponential decay
        lam = rng.exponential(scale=0.007) + 0.0005
        a = np.exp(-lam * idx)

    elif family == 3:                         # Mixed power‑law × exponential
        p = rng.uniform(0.2, 2.5)
        lam = rng.exponential(scale=0.007) + 0.0005
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 4:                         # Piece‑wise constant (3‑6 blocks)
        K = rng.integers(3, 7)
        cuts = np.sort(rng.integers(1, n, size=K - 1))
        heights = rng.random(K) + 0.1
        a = np.empty(n)
        start = 0
        for end, h in zip(np.append(cuts, n), heights):
            a[start:end] = h
            start = end

    elif family == 5:                         # Sum of a few Gaussian bumps
        a = np.zeros(n)
        nb = rng.integers(2, 5)
        for _ in range(nb):
            centre = rng.uniform(0, n - 1)
            width = rng.exponential(scale=n * 0.06) + 1.0
            amp = rng.random()
            a += amp * np.exp(-0.5 * ((idx - centre) / width) ** 2)

    else:                                      # Small jitter around uniform
        a = np.full(n, target / n)

    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Shape that is large at the two edges and decays towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _smooth_gaussian_initial(n: int, rng) -> np.ndarray:
    """Broad Gaussian‑like bump."""
    target = math.sqrt(2.0 * n)
    centre = rng.uniform(0, n - 1)
    sigma = rng.exponential(scale=n * 0.15) + 1.0
    a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart (potential destination)
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # choose the smallest counterpart (largest improvement)
        j = None
        order = np.argsort(counterpart[j_range])
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.75
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # stochastic escape
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing (O(n) per move).  Returns the best vector
    and the best maximal convolution entry seen before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.4
    decay = 0.9997

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.08, 0.35)
        if delta <= 0.0:
            continue

        # O(n) convolution update
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-14:
            temperature = cur_max * 0.4

        # occasional re‑heat
        if rng.random() < 0.001:
            temperature = cur_max * 0.4

    return best_a, best_score


# ----------------------------------------------------------------------
# Small parametric sweep (cheap, but often finds a good basin)
# ----------------------------------------------------------------------


def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan a few one‑parameter families (power‑law, exponential,
    edge‑heavy, mixed) and return the best vector found (according to the
    exact max‑convolution).
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law exponents
    for p in np.linspace(0.1, 3.0, 60):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.0003, 0.025, 60):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy
    for lam in np.linspace(0.001, 0.025, 40):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 2.0, 25):
        for lam in np.linspace(0.001, 0.015, 12):
            a = (idx ** (-p)) * np.exp(-lam * lin)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Seed from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded (n={n0}) → score={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # ------------------------------------------------------------------
    # 1)  Choose a working length.
    #      Larger n gives a better theoretical limit but costs a bit more.
    # ------------------------------------------------------------------
    # Use n = 4000 if we have enough time, otherwise fall back to 2500.
    remaining_now = deadline - time.time()
    n = 4000 if remaining_now > 120 else 2500
    target = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1a)  Parameter sweep (very cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_sweep, score_sweep = _parameter_sweep(n, rng)
        if a_sweep is not None and score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1b)  Random families + Adam restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        # decide how much time to give this restart
        time_left = deadline - time.time()
        per_restart = min(25.0, time_left * 0.6)   # never exceed the remaining budget
        if per_restart < 5.0:
            break

        # first restart uses the current best, later ones are fresh random starts
        if best_seq is not None and rng.random() < 0.3:
            a0 = best_seq.copy()
        else:
            # pick a family at random
            if rng.random() < 0.45:
                a0 = _random_initial_sequence(n, rng)
            elif rng.random() < 0.7:
                a0 = _edge_high_initial_sequence(n, rng)
            else:
                a0 = _smooth_gaussian_initial(n, rng)

        a_opt = _adam_optimize(a0, target, time.time() + per_restart)
        s = _max_convolution_fft(a_opt)
        if s < best_score - 1e-12:
            best_seq = a_opt.copy()
            best_score = s
            print(f"[{time.time():.1f}] Adam restart → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Deterministic hill‑climb (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_cur = _project_to_simplex(best_seq, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur, deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Hill‑climb → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Simulated‑annealing refinement (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        a_cur = _project_to_simplex(best_seq, target)
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Final polishing: a short Adam run + hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_polish = _adam_optimize(best_seq, target, deadline - 1.5)
        b_polish = _full_convolution(a_polish)
        a_fin, s_fin = _hill_climb_deterministic(a_polish, b_polish,
                                                 deadline - 0.5)
        if s_fin < best_score - 1e-12:
            best_seq = a_fin.copy()
            best_score = s_fin
            print(f"[{time.time():.1f}] Final polish → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5)  Trim trailing near‑zeros and report official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never occur)
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
```
