def _subgradient_descent(a0, deadline, target_sum):
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute conv
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax = int(np.argmax(conv))
        # compute gradient
        n = a.shape[0]
        grad = np.zeros_like(a)
        # valid indices: i s.t. 0 <= kmax - i < n
        idx = np.arange(n)
        mask = (kmax - idx >= 0) & (kmax - idx < n)
        grad[mask] = 2 * a[kmax - idx[mask]]
        # momentum
        momentum = 0.9 * momentum + grad
        # step size
        norm = np.linalg.norm(momentum)
        if norm > 0:
            step = target_sum * 0.5 / (norm * (it + 1)**0.5)
            a -= step * momentum
        # project and reorder
        a = _project_to_simplex(a, target_sum)
        a = _alternating_order(a)  # reordering to spread large values
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"Iter {it} score {best_val}")
        # random jitter occasionally
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            a = _alternating_order(a)
        it += 1
    return best, best_val

"""
Propose a low‑peak self‑convolution sequence.

The algorithm follows three phases:

1. **Exploration** – many short runs of a simple sub‑gradient descent
   that directly minimises the maximal entry of the convolution.
   Each run starts from a random power‑law / exponential / Dirichlet
   seed (or a jitter of the current best).  After each step we project
   back onto the ℓ¹‑simplex (Σa = sqrt(2·n)) and keep the best vector.

2. **Upscaling** – the best low‑dimensional solution is linearly
   interpolated to a larger length (up to ~15000).  The values are
   reordered by the alternating “large‑small” pattern, which is known
   to reduce the inner products a_i·a_{n‑1‑i} that dominate the centre
   of the convolution.

3. **Polishing** – a longer sub‑gradient descent run (with momentum)
   and a final deterministic mass‑transfer refinement are performed on
   the large problem.  Throughout the search we respect non‑negativity,
   the ℓ¹‑norm constraint and the 1000 s time budget.

The returned list contains the non‑negative heights and is stored in
the global variable `height_sequence_1` for reuse in later invocations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT based convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real transform)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a using FFT."""
    conv = _conv_fft(a)
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Objective value: 2·n·max(a*a) / (Σa)².
    For a normalized sum Σa = sqrt(2·n) this reduces to max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – needed after every update
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Generate exp(-lam * i) and renormalise."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, alpha: float, target_sum: float,
                        rng: np.random.Generator) -> np.ndarray:
    """Generate a Dirichlet‑like random vector (Gamma(alpha) i.i.d)."""
    a = rng.gamma(alpha, 1.0, size=n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Alternating large‑small reordering (helps to pair large with small)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest value leftmost, second largest rightmost, third next‑left, …."""
    sorted_vals = np.sort(seq)[::-1]   # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _gradient_maxconv(a: np.ndarray, kmax: int) -> np.ndarray:
    """
    Sub‑gradient of max_k conv_k(a) at the index kmax that attains the maximum.
    For the convolution, the derivative w.r.t. a_j is 2 * a_{kmax‑j}
    when the index is in range; otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    mask = (kmax - idx >= 0) & (kmax - idx < n)
    grad[mask] = 2.0 * a[kmax - idx[mask]]
    return grad


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         momentum_coeff: float = 0.99,
                         base_lr_factor: float = 0.5,
                         jitter_every: int = 0) -> tuple[np.ndarray, float]:
    """
    Simple momentum sub‑gradient descent on the true max‑convolution objective.
    The step size is scaled by the ℓ²‑norm of the momentum and decays with
    sqrt(iteration) to guarantee convergence.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = float(conv.max())
        kmax = int(np.argmax(conv))

        # sub‑gradient at the maximal entry
        g = _gradient_maxconv(a, kmax)

        # momentum update
        momentum = momentum_coeff * momentum + g

        # adaptive step length
        norm = np.linalg.norm(momentum)
        if norm > 0:
            # step shrinks like 1/sqrt(iter)
            lr = (base_lr_factor * target_sum) / (norm * (1.0 + np.sqrt(it)))
            a -= lr * momentum

        # projection back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # optional jitter to escape flat zones
        if jitter_every and it % jitter_every == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 1e-3, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (kept from the previous code)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves of a small amount of mass from a “donor’’ (large contribution
    to the current maximal convolution entry) to a “receiver’’ (small
    contribution).  Moves are accepted with a Metropolis rule whose temperature
    decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight contributed by each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low self‑convolution peak.
    The routine consists of exploration (short sub‑gradient runs),
    up‑scaling, final polishing and a short deterministic refinement.
    """
    start_time = time.time()
    # safety margin to avoid overrunning the judge
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a size for the exploratory phase (larger ⇒ more freedom)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.80, 9):

    for lam in np.linspace(0.001, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, lam, target_expl))

    # Dirichlet / random‑Gamma
    for a in [0.2, 0.5, 1.0, 2.0]:
        init_pool.append(_dirichlet_sequence(n_expl, a, target_expl, rng))

    # alternating ordered versions of the above (often better)
    alt_versions = [_alternating_order(seq) for seq in list(init_pool)]
    init_pool.extend(alt_versions)

    # previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

        jitter = rng.normal(scale=target_expl * 0.01, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)

        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Exploration: many short sub‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # allocate a short time slice for this run (≈15 s)
        run_dead = min(time.time() + 15.0, deadline - 5.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _subgradient_descent(
            seed,
            run_dead,
            target_expl,
            momentum_coeff=0.98,
            base_lr_factor=0.6,
            jitter_every=0,          # keep it deterministic inside a short run
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional very brief deterministic refinement on the small scale
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.15
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upscaling to the final problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: a simple power‑law
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining_time = deadline - time.time()
    if remaining_time > 30.0:
        n_final = 15000
    elif remaining_time > 15.0:
        n_final = 12000
    else:
        n_final = n_expl                     # no room left for up‑scaling

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Apply alternating order which is cheap and frequently beneficial
    seq_big = _alternating_order(seq_big)

    best_seq = seq_big
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the large scale (sub‑gradient descent)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_final, val_final = _subgradient_descent(
            best_seq,
            deadline,
            target_final,
            momentum_coeff=0.97,
            base_lr_factor=0.5,
            jitter_every=500,          # occasional jitter helps escape plateaus
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Tiny deterministic mass‑transfer pass (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 1.0)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Log the final evaluated score (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
