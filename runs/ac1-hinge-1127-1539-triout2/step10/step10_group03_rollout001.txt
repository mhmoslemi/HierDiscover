"""
Hybrid optimizer with a linear‑programming (LP) refinement.

Idea
----
1. Work first on a modest length (≈2000).  The LP step solves

       max   Σ g_i
       s.t.  (f * g)(k) ≤ M   for all k,
             g_i ≥ 0,

   where M = max_{k} (f * f)(k).  This is a linear program
   (variables g_i, constraints are linear in g).  After solving,
   g is rescaled to the required ℓ¹‑norm and mixed with the current
   sequence f (small step size t).  Repeating a few LP iterations
   provably does not increase M and empirically reduces the maximal
   convolution entry.

2. Between LP steps a short projected‑gradient descent (soft‑max
   approximation of the max‑convolution) is used for fine tuning.

3. The best small‑scale sequence is interpolated to a larger length
   (up to 12000 depending on the remaining budget) and polished with
   a final high‑temperature gradient descent.

All sequences are projected onto the ℓ¹‑simplex Σ a_i = √(2 n) and
clipped to non‑negative values, satisfying the checks performed by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence helpers
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, smallest, 2nd‑largest, 2nd‑smallest, …."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this equals max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (for PGD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return (grad, conv) where grad is the gradient of the soft‑max approximation."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent using a soft‑max surrogate of the max.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (SciPy)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray):
    """
    Build the sparse matrix A (size (2n‑1) × n) such that
    (A @ g)[k] = (f * g)[k] for linear convolution.
    """
    from scipy.sparse import diags as spdiags

    n = f.shape[0]
    # every column i contributes f[i] on the diagonal with offset i
    diags_data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = spdiags(diags_data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def _lp_step(f: np.ndarray, target_sum: float):
    """
    Solve the LP:
        max Σ g_i
        s.t. (f * g)[k] ≤ M  ∀k,   g_i ≥ 0
    where M = max(f * f).  Return the scaled g (Σ g = target_sum)
    together with M.  If solving fails, return (None, M).
    """
    from scipy.optimize import linprog

    n = f.shape[0]
    M = _max_convolution_fft(f)

    # Build constraint matrix A @ g ≤ M
    A = _build_conv_matrix(f)

    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # minimize -Σg  → maximize Σg
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception as e:      # pragma: no cover
        print("linprog exception:", e)
        return None, M

    if not res.success:        # pragma: no cover
        print("LP failed:", res.message)
        return None, M

    g0 = res.x
    sum_g = g0.sum()
    if sum_g <= 0:
        return None, M

    g = g0 * (target_sum / sum_g)   # rescale to required ℓ¹‑norm
    return g, M
