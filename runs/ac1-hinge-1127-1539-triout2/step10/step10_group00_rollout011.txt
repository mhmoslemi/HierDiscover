def _mass_transfer_refine_stronger(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    conv = _conv_fft(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        weight = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(weight)]
        # choose receiver from low-weight tail
        low_thr = np.percentile(weight, 5)  # lower 5% threshold more aggressive
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)
        if donor == receiver:
            receiver = (receiver + 1) % n
        max_frac = 0.12   # larger transfer fraction
        delta = a[donor] * rng.random() * max_frac
        a[donor] -= delta
        a[receiver] += delta
        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # print improvement
            print(f"[{time.time():.2f}] Strong transfer improve {best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)
        # renorm
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            a = a * (target_sum / cur_sum)
    return best_a, best_val

"""
Improved step‑function search for the problem

    minimize  2·n·max(a*a) / (∑a)²
    subject to a_i ≥ 0 , a ∈ ℝⁿ

The code consists of a few independent stages that together explore the
search space more aggressively than the reference implementation:

1. **Parametric warm‑start**
   – a three‑parameter family a_i = (i+δ)^(-p)·exp(-λ·i/n)
   – hill‑climbing on (p,λ,δ) using a reduced problem size.
2. **Exponential (mirror) descent** on the smooth soft‑max surrogate
   with a schedule of increasing β.
3. **Exact sub‑gradient descent** on the true maximal convolution entry
   (the non‑smooth objective).
4. **Aggressive deterministic mass‑transfer** that moves a larger
   fraction of mass from the most “dangerous’’ index to a low‑contribution
   index.
5. **Strong simulated‑annealing** (higher initial temperature,
   larger transfers, occasional jitter) for final fine‑tuning.
6. Trim trailing zeros, store the result for warm‑starts and return it.

All stages respect the overall time budget (default 1000 s) and use
only NumPy (no external solvers).  The objective is invariant under
global scaling, therefore every stage renormalises the sequence to the
convenient ℓ¹‑norm √(2·n).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution / scoring utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Length of the next power‑of‑2 ≥ 2·n‑1 (FFT padding)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using real FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective  2·n·max(a*a) / (sum(a)²).  The ℓ¹‑norm is assumed >0."""
    n = a.size
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to length new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _parametric_seq(
    n: int, p: float, lam: float, delta: float, target_sum: float
) -> np.ndarray:
    """Three‑parameter family: (i+δ)^(-p)·exp(-λ·i/n), renormalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p) * np.exp(-lam * i / n)
    a = a * (target_sum / a.sum())
    return a


# ----------------------------------------------------------------------
# Stage 2 – exponential (mirror) descent on smoothed max
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, full_conv) where conv = a*a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    probs = exps / exps.sum()

    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    prob_fft = np.fft.rfft(probs, n=size)
    conv_rev = np.fft.irfft(rev_fft * prob_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror‑descent with multiplicative updates on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1e5
    base_eta = 0.07
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)          # multiplicative update
        a = a * (target_sum / a.sum())    # renormalise

        # occasional small jitter to avoid plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it} β={beta:.1f} → {best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val

def _subgradient_max_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of max entry (2·a_{k‑i})
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.size - 1))
        hi = min(k_max, a.size - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step that never makes a component negative
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.9 * safe_eta, 0.5 / math.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} → {best_val:.6f}")

        # occasional jitter for robustness
        if it % 600 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best, best_val

    """Move a larger fraction of mass from high‑contributing to low‑contributing indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(conv.argmax())

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        weight = a[idx] * a[k_max - idx]

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 5)      # 5 % tail
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            receiver = (receiver + 1) % n

        max_frac = 0.12                         # larger moves than before
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:

        a[receiver] += delta

            print(f"[{time.time():.2f}] Strong transfer improve → {best_val:.6f}")

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    # Stage B – exponential (mirror) descent on a smooth surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
