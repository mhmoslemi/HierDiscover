def _max_subgradient_descent(a0, deadline, target_sum):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        grad = _gradient_max(a)
        # momentum update
        momentum = 0.9 * momentum + grad
        # step size schedule
        lr = target_sum * 0.5 / (np.linalg.norm(momentum)*(it+1)**0.5 + 1e-12)
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # occasional random jitter to escape
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum*0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        it += 1
    return best_a, best_val

    best_a = a.copy()

    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        grad = _gradient_max_subgrad(a)
        momentum = momentum_decay * momentum + grad
        # dynamic step size: lr = lr0 * target_sum / (np.linalg.norm(momentum) * (it+1)**0.5 + 1e-12)
        norm = np.linalg.norm(momentum)
        if norm > 0:

            a -= lr * momentum
            a = _project_to_simplex(a, target_sum)

                print(f"[{time.time():.2f}] MaxGD it={it} score={best_score:.6f}")

            a += jitter
            a = _project_to_simplex(a, target_sum)

"""
Hybrid search for a low self‑convolution bound.

The algorithm consists of three phases:

1.  Exploration on a medium size (≈5‑6k).  Many short projected‑gradient
    runs on a soft‑max approximation of the convolution peak are launched
    from a diverse pool (power‑law, exponential, random Dirichlet and the
    previous best).  The best sequence found is kept.

2.  Upscaling.  The best medium‑size solution is interpolated to a larger
    length (up to 12 000 when time permits) and projected onto the ℓ¹‑simplex.
    Several simple reorderings (alternating large/small, reversal, random
    permutations) are tried and the best is selected.

3.  Fine polishing on the large problem.  A projected gradient descent that
    uses the exact sub‑gradient of the max‑convolution (instead of the
    soft‑max) is run for the remaining time, followed by a short deterministic
    mass‑transfer refinement.  The final sequence is trimmed of trailing zeros
    and returned.

The implementation respects all security checks of `evaluate_sequence`,
uses only NumPy and the standard library, and always finishes before the
budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² ."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Re‑ordering helper
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest value left, second largest right, etc."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (used in the exploratory phase)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      max_beta: float = 1e5,
                                      step_factor: float = 0.7,
                                      gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact max‑subgradient utilities (used for fine polishing)
# ----------------------------------------------------------------------


def _gradient_max_subgrad(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution w.r.t. `a`.
    Uses a uniform distribution over all indices attaining the maximum.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    mask = (b >= max_b - eps)
    cnt = mask.sum()
    if cnt == 0:
        p = np.zeros_like(b)
    else:
        p = mask.astype(np.float64) / cnt

    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad


def _gd_max_subgrad(a0: np.ndarray,
                    deadline: float,
                    target_sum: float,
                    lr0: float = 0.5,
                    momentum_decay: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent using the exact max‑subgradient.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        grad = _gradient_max_subgrad(a)
        momentum = momentum_decay * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            # diminishing step size
            lr = lr0 * target_sum / (norm * (it + 1) ** 0.5 + 1e-12)
            a -= lr * momentum
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] MaxGD it={it:5d} score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (used at all scales)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves from a “donor’’ (large contribution to the current peak)
    to a “receiver’’ (small contribution).  Accepted with a Metropolis rule
    whose temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time + 1e-12)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.

    The routine first explores many random starts at a moderate length,
    then up‑scales the best solution to a larger length (up to 12 000
    when time permits) and finishes with a high‑temperature exact‑max
    gradient descent followed by a deterministic refinement.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Exploration on a moderate size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 500:
        n_expl = 5000
    else:
        n_expl = 4000
    target_expl = math.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()
    init_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.75, 7):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.002, 0.012, 5):
        a = np.exp(-lam * np.arange(n_expl, dtype=float))
        a *= target_expl / a.sum()
        init_pool.append(a)

    # random Dirichlet‑like points
    for _ in range(5):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] > 0:
            init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to increase diversity
    augmented = []
    for seq in init_pool:
        augmented.append(seq)
        jitter = rng.normal(scale=target_expl * 0.01, size=n_expl)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        augmented.append(seq_j)
    init_pool = augmented

    best_seq = None
    best_score = np.inf

    # allocate about 35 % of the budget to the exploratory phase,
    # leaving a generous margin for the large‑scale polishing.
    explore_deadline = min(deadline - 60.0,
                           start_time + (deadline - start_time) * 0.35)

    while time.time() < explore_deadline:
        # pick a seed
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = init_pool[rng.integers(len(init_pool))].copy()

        run_deadline = min(time.time() + 12.0, explore_deadline)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed,
            run_deadline,
            target_expl,
            max_beta=3e5,
            step_factor=0.8,
            gamma=0.98,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Optional short deterministic refinement at the small scale
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        mt_dead = time.time() + min(12.0,
                                    (deadline - time.time()) * 0.1)
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Small MT improvement best={best_score:.6f}")

    # If nothing succeeded, fall back to a simple power‑law.
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 3. Upscaling to the final problem size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        n_final = 12000
    elif remaining > 70.0:
        n_final = 9000
    elif remaining > 40.0:
        n_final = 7000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # generate a few reorderings and keep the best one
    candidates = [seq_big, _alternating_order(seq_big), seq_big[::-1]]
    for _ in range(2):
        perm = rng.permutation(n_final)
        candidates.append(seq_big[perm])

    best_seq = None
    best_score = np.inf
    # leave ~15 s for the final deterministic pass
    final_deadline = deadline - 15.0
    per_candidate = max(5.0, (final_deadline - time.time()) / len(candidates))

    for cand in candidates:
        cand = _project_to_simplex(cand, target_final)
        run_dead = min(final_deadline, time.time() + per_candidate)
        a_opt, val_opt = _gd_max_subgrad(cand, run_dead, target_final,
                                         lr0=0.6, momentum_decay=0.9)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Max‑GD candidate best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_dead = time.time() + min(12.0, deadline - time.time() - 2.0)
        a_final, val_final = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.04,
        )
        if val_final < best_score:
            best_score, best_seq = val_final, a_final.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for possible later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
