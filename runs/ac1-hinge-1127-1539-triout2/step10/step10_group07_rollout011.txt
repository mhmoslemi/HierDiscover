# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a * a)[k] (full convolution)."""
    return _conv_fft(a).max()


def _score(a: np.ndarray) -> float:
    """
    Benchmark score used internally.
    The ℓ¹‑norm is fixed to 1, therefore the score equals 2·n·max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Temperature‑annealed soft‑max gradient descent
# ----------------------------------------------------------------------


def _softmax_descent(a0: np.ndarray,
                     target_sum: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise max(a*a) using a smoothed max (soft‑max) objective.
    Returns the best sequence found and the corresponding max(a*a).
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_maxb = _max_convolution(a)

    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    base_step = target_sum * 0.5   # initial step magnitude

    while time.time() < deadline:
        # --------------------------------------------------------------
        # full convolution and temperature schedule
        # --------------------------------------------------------------
        b = _conv_fft(a)                # length 2n‑1
        max_b = b.max()
        # exponential cooling: T decays from max_b to ~1e‑4·max_b
        T = max_b * max(1e-4, np.exp(-it / 2000.0))

        # --------------------------------------------------------------
        # soft‑max weights (probability distribution over shift indices)
        # --------------------------------------------------------------
        # avoid overflow/underflow by shifting with max_b
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum == 0.0:                # numerically unlikely, fall back to mask
            w = (b == max_b).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --------------------------------------------------------------
        # gradient of the smoothed objective w.r.t. a
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)           # length 3n‑2
        L = len(w) - 1                                        # = 2n‑2
        g = conv_full[L - np.arange(n)]

        # --------------------------------------------------------------
        # projected gradient step
        # --------------------------------------------------------------
        step = base_step / np.sqrt(it + 1.0)
        gn = np.linalg.norm(g)
        if gn > 0:
            a -= step * g / gn
        a = _project_to_simplex(a, target_sum)

        # --------------------------------------------------------------
        # jitter occasionally to escape flat regions
        # --------------------------------------------------------------
        if it % 1000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # --------------------------------------------------------------
        # bookkeeping
        # --------------------------------------------------------------
        if max_b < best_maxb:
            best_maxb = max_b
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Soft‑max it={it:6d} maxB={best_maxb:.6e}"
            )
        it += 1

    return best_a, best_maxb


# ----------------------------------------------------------------------
# Exact mass‑transfer simulated annealing refinement
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing that transfers a tiny amount of
    mass from a “bad’’ index to a “good’’ one, updating the convolution in
    O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_maxb = _max_convolution(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_maxb * 0.04                # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # indices that currently achieve (or are close to) the maximum
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, k_max-i) to the current peak
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # --------------------------------------------------------------
        # pick a receiver among indices that contribute little to the peak
        # --------------------------------------------------------------
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)   # 30‑th percentile as “low’’
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # update the convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: -2*delta*a[donor] adds to all positions where donor participates,
        #               +2*delta*a[receiver] does the opposite
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (donor‑donor, receiver‑receiver and cross term)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_maxb = b_new.max()

        # --------------------------------------------------------------
        # simulated‑annealing acceptance test
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_maxb < best_maxb:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_maxb - best_maxb) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_maxb < best_maxb:
                best_maxb = new_maxb
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved maxB={best_maxb:.6e}")

        # --------------------------------------------------------------
        # occasional full recomputation to avoid drift due to rounding
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_maxb = _max_convolution(a)
            if cur_maxb < best_maxb:
                best_maxb = cur_maxb
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved maxB={best_maxb:.6e}")
            last_recalc = time.time()

        # --------------------------------------------------------------
        # tiny jitter (kept cheap) every few seconds
        # --------------------------------------------------------------
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_maxb = _max_convolution(a)
                if cur_maxb < best_maxb:
                    best_maxb = cur_maxb
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved maxB={best_maxb:.6e}")

    return best_a, best_maxb


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the supplied time budget (seconds) and returns the
    best sequence found as a list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size selection (larger budget → larger n)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = 1.0                           # we work on the simplex Σa = 1

    # --------------------------------------------------------------
    # Build initial candidates
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    init_candidates = []

    # warm‑start, if available
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init_candidates.append(warm)

    # a deterministic power‑law baseline (helps when no warm start exists)
    power = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
    init_candidates.append(power)

    # a few random Dirichlet‑type vectors (sparse)
    for _ in range(3):
        rnd = rng.gamma(shape=0.3, scale=1.0, size=n)   # shape < 1 → sparse
        init_candidates.append(rnd)

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate roughly 60 % of the time for soft‑max descent + restarts,
    # the remaining time for the final refinement.
    softmax_deadline = start + 0.60 * (deadline - start)

    while time.time() < softmax_deadline:
        # pick a candidate, jitter it a little and project onto the simplex
        base = init_candidates[rng.integers(len(init_candidates))]
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        a0 = np.maximum(base + jitter, 0.0)
        a0 = _project_to_simplex(a0, target_sum)

        # give each run a modest time slice (10 s or the remaining budget)
        run_deadline = min(softmax_deadline,
                           time.time() + min(12.0, softmax_deadline - time.time()))
        a_opt, maxb_opt = _softmax_descent(a0, target_sum, run_deadline)

        cur_score = 2.0 * n * maxb_opt
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New best (softmax) score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final local refinement (uses the remaining time)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ref, maxb_ref = _local_refine(best_seq, target_sum, deadline)
        ref_score = 2.0 * n * maxb_ref
        if ref_score < best_score:
            best_score = ref_score
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up: strip trailing zeros and evaluate with the official helper
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
