def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

    """Objective value (equal to max convolution when Σa=√(2n))."""
    return _max_convolution_fft(a)

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)^(-p) normalised to target_sum."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Exponential decay a_i = exp(-lam * i) normalised to target_sum."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a *= target_sum / a.sum()
    return a


def _random_dirichlet_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex using a Dirichlet(α=1)."""
    x = rng.gamma(shape=1.0, scale=1.0, size=n)
    a = x / x.sum() * target_sum
    return a


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _softmax_gradient_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float,
                              beta_start: float = 0.5,
                              beta_max: float = 3e5,
                              step_factor: float = 0.6,
                              gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    beta = beta_start
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        if beta < beta_max and it % 400 == 0:
            beta = min(beta * 2.0, beta_max)
        it += 1
    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution, Euclidean simplex projection."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        # compute sub‑gradient
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1
    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                target_sum: float,
                                deadline: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves trying to lower the current peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering (plus occasional block moves)."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()
    start_temp = 0.3
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        # pick either a simple swap or a block reversal
        if rng.random() < 0.8:
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        else:
            # block reversal: pick a start and length
            start = rng.integers(0, n)
            length = rng.integers(2, min(30, n - start))
            cur_seq[start:start+length] = cur_seq[start:start+length][::-1]

        new_score = _score(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
        else:
            # revert move
            if rng.random() < 0.8:
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            else:
                cur_seq[start:start+length] = cur_seq[start:start+length][::-1]

        # occasional full reshuffle to escape deep minima
        if ratio > 0.9 and rng.random() < 0.02:
            rng.shuffle(cur_seq)
            cur_score = _score(cur_seq)

    return best_seq, best_score

# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.95,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution with simplex projection."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the max entry
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                target_sum: float,
                                deadline: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves that try to lower the current convolution peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering (sometimes reverses small blocks)."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()
    start_temp = 0.3
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        # 80 % simple swap, 20 % short block reversal
        if rng.random() < 0.8:
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        else:
            start = rng.integers(0, n - 2)
            length = rng.integers(2, min(30, n - start))
            cur_seq[start:start + length] = cur_seq[start:start + length][::-1]

        new_score = _score(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
        else:
            # revert the move
            if rng.random() < 0.8:
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            else:
                cur_seq[start:start + length] = cur_seq[start:start + length][::-1]

        # occasional full reshuffle to escape deep minima
        if ratio > 0.9 and rng.random() < 0.02:
            rng.shuffle(cur_seq)
            cur_score = _score(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with low benchmark score.

    The routine builds several initial shapes, refines them with a
    smooth‑max gradient descent, runs a true sub‑gradient descent,
    applies a deterministic mass‑transfer phase and finally uses a
    swap‑based simulated‑annealing on the ordering.  Time is split
    adaptively; with a full 1000 s budget the length is 25000.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 25000
    elif budget_s >= 600:
        n = 20000
    else:
        n = 14000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    init_seqs = []

    # power‑law templates
    for p in (0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # exponential templates
    for lam in (0.0005, 0.001, 0.002, 0.005):
        init_seqs.append(_exp_sequence(n, lam, target_sum))

    # random Dirichlet point
    init_seqs.append(_random_dirichlet_sequence(n, target_sum, rng))

    # warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
            init_seqs.append(_project_to_simplex(resized, target_sum))
        else:
            init_seqs.append(_project_to_simplex(prev, target_sum))

    # jittered copies of each template
    for seq in init_seqs[:]:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        init_seqs.append(seq_j)

    # pick the best starter
    best_seq = None
    best_score = np.inf
    for seq in init_seqs:
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Soft‑max gradient descent (smooth surrogate)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        gd_deadline = now + (deadline - now) * 0.12
        best_seq, best_score = _softmax_gradient_descent(
            best_seq, target_sum, gd_deadline,
            beta_start=0.5, beta_max=2e5, step_factor=0.7, gamma=0.99
        )
        print(f"[{time.time():.2f}] After soft‑max GD score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        sg_deadline = now + (deadline - now) * 0.30
        best_seq, best_score = _subgradient_descent(
            best_seq, target_sum, sg_deadline,
            step_factor=0.6, gamma=0.95,
            jitter_interval=400, jitter_scale=5e-4
        )
        print(f"[{time.time():.2f}] After sub‑gradient score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        mt_deadline = now + (deadline - now) * 0.20
        best_seq, best_score = _deterministic_mass_transfer(
            best_seq, target_sum, mt_deadline,
            start_temp=0.6, end_temp=5e-5, low_frac=0.04
        )
        print(f"[{time.time():.2f}] After mass‑transfer score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Re‑ordering heuristics
    # ------------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Permutation annealing
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        perm_deadline = now + (deadline - now) * 0.25
        best_seq, best_score = _permutation_annealing(
            best_seq, target_sum, perm_deadline, init_score=best_score
        )
        print(f"[{time.time():.2f}] After permutation annealing score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – Final polishing (high‑β soft‑max GD)
    # ------------------------------------------------------------------
    now = time.time()
    if now + 2.0 < deadline:
        best_seq, best_score = _softmax_gradient_descent(
            best_seq, target_sum, deadline,
            beta_start=5.0, beta_max=5e5, step_factor=0.5, gamma=0.995
        )
        print(f"[{time.time():.2f}] Final polish score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future invocations
    globals()["height_sequence_1"] = final_seq

    return final_seq
