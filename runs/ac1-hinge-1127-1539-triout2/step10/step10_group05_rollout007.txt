"""
Propose a non‑negative step function with a low value of
evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2).

The search runs a multi‑stage pipeline for several lengths
(2000 → 4000).  For each length we:

1.  Perform an extensive cheap parametric sweep (power‑law,
    exponentials, edge‑heavy, double‑exponential, mixed) using a
    fast FFT based max‑convolution.
2.  Run many short Adam restarts on a smooth log‑sum‑exp surrogate of
    the max‑convolution, always projecting onto the ℓ¹‑simplex
    Σa = √(2 n).
3.  Apply a deterministic hill‑climb that moves mass from the worst
    convolution entry to the best destination (exact max‑convolution
    updates in O(n)).
4.  Refine with a simulated‑annealing phase that performs random mass
    transfers.
5.  A final polishing step (long Adam + hill‑climb).

All phases respect the wall‑clock budget (default 1000 s).  The best
sequence found over all lengths is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p (the gradient)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning rate loosely scaled with the target ℓ¹‑norm
    base_lr = 0.12 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 220 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing temperature gradually
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Random initial families
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.1, 1.6)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.012)
        a = np.exp(-lam * idx)

    elif family == 2:                   # edge‑heavy exponential
        lam = rng.exponential(scale=0.01)
        d = np.minimum(idx, n - 1 - idx)
        a = np.exp(-lam * d)

    elif family == 3:                   # double‑exponential (both edges)
        lam1 = rng.exponential(scale=0.015)
        lam2 = rng.exponential(scale=0.015)
        a = np.exp(-lam1 * idx) + np.exp(-lam2 * (n - 1 - idx))

    elif family == 4:                   # mixed power‑law × exponential
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 5:                   # piecewise linear edge‑heavy
        split = rng.integers(int(0.2 * n), int(0.8 * n))
        left = np.linspace(1.0, 0.1, split)
        right = np.linspace(0.1, 1.0, n - split)
        a = np.concatenate([left, right])

    else:                               # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Shape that is large near the two edges and decays towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.018)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.75
        if delta <= 0.0:
            break

        # fast update of convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # random jitter to shake the state
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing (O(n) per move).
    Returns the best vector and its maximal convolution entry seen.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.6
    decay = 0.9997

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.12, 0.45)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.6

        # occasional re‑heat
        if rng.random() < 0.001:
            temperature = cur_max * 0.6

    return best_a, best_score


# ----------------------------------------------------------------------
# Extended cheap parametric sweep
# ----------------------------------------------------------------------


def _parameter_sweep_extended(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan a richer collection of one‑parameter families and keep the best
    (according to the exact max‑convolution).
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    lin = np.arange(n, dtype=float)
    idx = np.arange(1, n + 1, dtype=float)

    # power‑law
    for p in np.linspace(0.05, 2.5, 45):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponential
    for lam in np.linspace(0.0005, 0.025, 45):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy exponential (distance to nearest edge)
    for lam in np.linspace(0.001, 0.045, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # double exponential (two independent decays)
    for lam1 in np.linspace(0.001, 0.025, 12):
        for lam2 in np.linspace(0.001, 0.025, 12):
            a = np.exp(-lam1 * lin) + np.exp(-lam2 * (n - 1 - lin))
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.2, 1.8, 18):
        for lam in np.linspace(0.001, 0.015, 12):
            a = (idx) ** (-p) * np.exp(-lam * lin)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # a few Dirichlet samples for completeness
    for _ in range(8):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Per‑length optimisation driver
# ----------------------------------------------------------------------


def _optimize_length(n: int, rng, global_deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the full pipeline for a fixed length ``n`` and return the best
    vector and its max‑convolution value (the objective, because the
    ℓ¹‑norm is fixed to √(2 n)).
    """
    target = math.sqrt(2.0 * n)
    best_seq = None
    best_score = np.inf

    # ---- 0) seed from a previous global solution, if available ----
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            prev = prev[: np.where(prev > 1e-12)[0][-1] + 1]    # trim trailing zeros
            # upsample / downsample to current length
            prev = _upsample_sequence(prev, n)
            prev = _project_to_simplex(prev, target)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded length {n} → score={best_score:.6f}")
        except Exception as e:
            print(f"[{time.time():.1f}] Seed error (len {n}):", e)

    # ---- 1) cheap parametric sweep ----
    if time.time() < global_deadline - 2.0:
        a_sweep, s_sweep = _parameter_sweep_extended(n, rng)
        if s_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = s_sweep
            print(f"[{time.time():.1f}] Sweep (n={n}) → score={best_score:.6f}")

    # ---- 2) many short Adam restarts ----
    while True:
        remaining = global_deadline - time.time()
        if remaining < 6.0:
            break
        # allocate roughly 6 s per restart; make at most 8 restarts
        n_restarts = min(8, max(1, int(remaining // 6.0)))
        per_restart = remaining / n_restarts
        for r in range(n_restarts):
            if time.time() >= global_deadline - 0.5:
                break
            # initial point
            if r == 0 and best_seq is not None:
                a0 = best_seq.copy()
            else:
                if rng.random() < 0.55:
                    a0 = _random_initial_sequence(n, rng)
                else:
                    a0 = _edge_high_initial_sequence(n, rng)

            stop = min(time.time() + per_restart * 0.85, global_deadline - 0.5)
            a_opt = _adam_optimize(a0, target, stop)
            s = _max_convolution_fft(a_opt)
            if s < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = s
                print(f"[{time.time():.1f}] Adam restart {r+1}/{n_restarts} (n={n}) → {best_score:.6f}")

    # ---- 3) deterministic hill‑climb ----
    if time.time() < global_deadline - 2.0:
        a_cur = _project_to_simplex(best_seq, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur,
                                                deadline=global_deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Hill‑climb (n={n}) → {best_score:.6f}")

    # ---- 4) simulated annealing ----
    if time.time() < global_deadline - 1.0:
        a_cur = _project_to_simplex(best_seq, target)
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur,
                                        deadline=global_deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal (n={n}) → {best_score:.6f}")

    # ---- 5) final polish (long Adam + hill‑climb) ----
    if time.time() < global_deadline - 2.0:
        a_polish = _adam_optimize(best_seq, target,
                                  deadline=global_deadline - 1.5)
        b_polish = _full_convolution(a_polish)
        a_final, s_final = _hill_climb_deterministic(a_polish, b_polish,
                                                     deadline=global_deadline - 0.5)
        if s_final < best_score - 1e-12:
            best_seq = a_final.copy()
            best_score = s_final
            print(f"[{time.time():.1f}] Final polish (n={n}) → {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimizes
    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_overall = None
    best_score = np.inf

    # try two lengths – 2000 (fast) and 4000 (more degrees of freedom)
    for n in (2000, 4000):
        if time.time() >= deadline - 2.0:
            break
        seq, score = _optimize_length(n, rng, deadline)
        if score < best_score - 1e-12:
            best_overall = seq.copy()
            best_score = score
            print(f"[{time.time():.1f}] New global best (n={n}) → {best_score:.6f}")

    if best_overall is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        target = math.sqrt(2.0 * n)
        best_overall = np.full(n, target / n, dtype=float)

    # Trim trailing near‑zeros (purely cosmetic)
    nz = np.where(best_overall > 1e-12)[0]
    if nz.size:
        best_overall = best_overall[: nz[-1] + 1]
    else:
        best_overall = best_overall[:1]

    # final official evaluation (for logging only)
    try:
        val = evaluate_sequence(best_overall.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_overall.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_overall.tolist()
