"""
Propose a step‑function (non‑negative sequence) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm is a multi‑phase heuristic:

1. **Length & shape sweep** – try a handful of lengths (800…4000) and
   several parametric families (power‑law, exponential, edge‑heavy,
   Gaussian, mixed).  The best candidate (according to the exact
   max‑convolution) seeds the later stages.

2. **Projected Adam restarts** – run many short Adam optimisations on the
   smooth log‑sum‑exp surrogate of the max convolution while projecting
   onto the ℓ¹‑simplex Σa = √(2 n).  Between restarts we jitter the
   current best to escape shallow basins.

3. **Sub‑gradient descent** – after the Adam phase we switch to a true
   sub‑gradient of the max‑convolution (only the worst index contributes)
   and perform a simple decaying‑step projected gradient descent.

4. **Deterministic hill‑climb** – a greedy mass‑transfer that always
   attacks the current worst convolution entry.

5. **Simulated‑annealing refinement** – random mass transfers with a
   temperature schedule for fine polishing.

All phases respect a wall‑clock budget (default 1000 s).  The final
sequence is truncated to remove trailing near‑zeros before being
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) utilities (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p (both length >= n)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the log‑sum‑exp surrogate of max(conv).
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initial families (including a new Gaussian family)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Large near the edges, decaying towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _gaussian_initial_sequence(n: int, rng) -> np.ndarray:
    """Bell‑shaped sequence (Gaussian)."""
    target = math.sqrt(2.0 * n)
    mu = (n - 1) / 2.0
    sigma_frac = rng.uniform(0.08, 0.35)   # relative to n
    sigma = sigma_frac * n
    idx = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Small parametric sweep for a fixed length
# ----------------------------------------------------------------------


def _quick_parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Coarse sweep over a few one‑parameter families for length n.
    Returns best vector and its maximal convolution value.
    """
    target = math.sqrt(2.0 * n)
    best_seq = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)     # 1‑based for power‑law
    lin = np.arange(n, dtype=float)

    # power‑law
    for p in np.linspace(0.2, 2.0, 12):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_seq = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.001, 0.03, 12):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_seq = s, a.copy()

    # edge‑heavy
    for lam in np.linspace(0.001, 0.03, 10):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_seq = s, a.copy()

    # Gaussian bump
    for sigma_frac in np.linspace(0.08, 0.3, 8):
        sigma = sigma_frac * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((lin - mu) / sigma) ** 2)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_seq = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 6):
        for lam in np.linspace(0.001, 0.015, 5):
            a = (idx ** (-p)) * np.exp(-lam * lin)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_seq = s, a.copy()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max) – unchanged from the original
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # line‑search half‑step
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement – unchanged
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing (O(n) per move)."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9999

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.5)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5

        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (simple version)
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray, target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector seen and its maximal convolution.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    lr = 0.2 * target_sum          # initial step size
    step_decay = 0.98
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # locate the worst convolution entry
        b = _full_convolution(a)
        M = int(np.argmax(b))

        # sub‑gradient of max entry
        idx = np.arange(n, dtype=int)
        j = M - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # gradient step + projection
        a -= lr * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # decay step size occasionally
        if it % 200 == 0:
            lr *= step_decay

        # occasional exact evaluation
        if it % 400 == 0:
            score = _max_convolution_fft(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad iter {it} → score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimizes ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 0)  Seed from a previous run if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded (n={n0}) → score={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # --------------------------------------------------------------
    # 1)  Length sweep with cheap parametric families
    # --------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400, 3000, 3600, 4000]
    sweep_deadline = deadline - 20.0    # reserve at least 20 s for later phases
    for n in lengths:
        if time.time() > sweep_deadline:
            break
        cand_seq, cand_score = _quick_parameter_sweep(n, rng)
        if cand_score < best_score - 1e-12:
            best_seq = cand_seq.copy()
            best_score = cand_score
            print(f"[{time.time():.1f}] Length {n} sweep → score={best_score:.6f}")

    # If for some reason we never obtained a sequence, fall back to uniform.
    if best_seq is None:
        n = 2000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = _max_convolution_fft(best_seq)

    n = best_seq.size
    target = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 2)  Projected Adam restarts (short bursts) + occasional hill‑climb
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # each restart gets roughly 7–8 s (leaving ~5 s for final phases)
        per_restart = min(8.0, remaining - 5.0)
        if per_restart <= 1.0:
            break

        # initialise a candidate
        if rng.random() < 0.5 and best_seq is not None:
            a0 = best_seq.copy()
            # small jitter to diversify the restart
            a0 += rng.normal(scale=target * 0.005, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target)
        else:
            # draw from a random family
            choice = rng.integers(0, 3)
            if choice == 0:
                a0 = _random_initial_sequence(n, rng)
            elif choice == 1:
                a0 = _edge_high_initial_sequence(n, rng)
            else:
                a0 = _gaussian_initial_sequence(n, rng)

        # Adam optimisation (≈80 % of allocated time)
        adam_deadline = time.time() + per_restart * 0.8
        a_opt = _adam_optimize(a0, target, adam_deadline)

        # evaluate
        score_opt = _max_convolution_fft(a_opt)
        if score_opt < best_score - 1e-12:
            best_seq, best_score = a_opt.copy(), score_opt
            print(f"[{time.time():.1f}] Adam restart → score={best_score:.6f}")

        # brief hill‑climb on the new candidate (≈20 % of time)
        if time.time() + 1.2 < deadline:
            b_opt = _full_convolution(best_seq)
            hc_deadline = time.time() + 1.2
            a_hc, s_hc = _hill_climb_deterministic(best_seq.copy(),
                                                   b_opt,
                                                   hc_deadline)
            if s_hc < best_score - 1e-12:
                best_seq, best_score = a_hc.copy(), s_hc
                print(f"[{time.time():.1f}] Hill‑climb post‑Adam → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3)  Sub‑gradient descent on the true max (medium‑term refinement)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.4
        a_sub, s_sub = _subgradient_optimize(best_seq.copy(),
                                             target,
                                             sub_deadline)
        if s_sub < best_score - 1e-12:
            best_seq, best_score = a_sub.copy(), s_sub
            print(f"[{time.time():.1f}] Sub‑gradient → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4)  Deterministic hill‑climb (longer) and simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        hc_deadline = time.time() + (deadline - time.time()) * 0.4
        a_hc, s_hc = _hill_climb_deterministic(best_seq.copy(),
                                               _full_convolution(best_seq),
                                               hc_deadline)
        if s_hc < best_score - 1e-12:
            best_seq, best_score = a_hc.copy(), s_hc
            print(f"[{time.time():.1f}] Hill‑climb final → score={best_score:.6f}")

    if time.time() < deadline - 1.0:
        anneal_deadline = deadline - 0.5
        a_ann, s_ann = _anneal_optimize(best_seq.copy(),
                                         _full_convolution(best_seq),
                                         anneal_deadline)
        if s_ann < best_score - 1e-12:
            best_seq, best_score = a_ann.copy(), s_ann
            print(f"[{time.time():.1f}] Anneal final → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5)  Clean‑up: drop trailing near‑zeros
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 6)  Report official evaluation (optional, for diagnostics)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
