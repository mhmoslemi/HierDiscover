"""
propose_candidate – heuristic optimizer for the step‑function benchmark.

The algorithm consists of several stages that respect a global time budget:

1. **Parametric pre‑search** – a lightweight grid over a three‑parameter
   exponential‑power‑law family (p, q, r).  The best parameters are kept.
2. **Local parameter hill‑climbing** – random perturbations of (p,q,r)
   guided by the objective.
3. **Projected Adam‑style gradient descent** (soft‑max approximation of the
   max‑convolution) with several restarts.
4. **Exact deterministic mass‑transfer** – greedy moves that flatten the
   current peak of the convolution by transferring the exact amount of mass
   required to make the highest entry equal to the second‑largest one.
5. **Final polishing** – a longer Adam run with a very high temperature.

All updates are projected onto the simplex `{x_i ≥ 0, Σ x_i = √(2 n)}` so that
the objective is invariant to scaling.  The best sequence found before the
deadline is returned (trailing zeros are trimmed).  Progress and intermediate
scores are printed for monitoring.

The implementation is completely self‑contained; `evaluate_sequence`
is assumed to be available in the execution environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _shape_sequence(n: int,
                    p: float,
                    q: float,
                    r: float,
                    target_sum: float) -> np.ndarray:
    """
    a_i = exp( - ( p*log(i+1) + q*i/n + r*(i/n)^2 ) )
    Normalised to have Σ a_i = target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    log_a = -(p * np.log(i + 1.0) + q * i / n + r * (i / n) ** 2)
    a = np.exp(log_a)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a *= target_sum / total
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution (used by Adam optimizer)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth max (soft‑max) of the convolution.
    Returns (grad, conv) where conv = a*a (full length 2n‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    P = exps / np.sum(exps)                     # soft‑max weights

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(P, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]     # only the n components that affect a
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smooth max
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimization of the soft‑max approximation.
    Returns best sequence and its exact score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiple short Adam runs with random jitter; keep the best result.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float) -> tuple[np.ndarray, float]:
    """
    Greedy operation that transfers just enough mass from the most harmful
    position to the least harmful one to lower the current peak to (≈) the
    second‑largest convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # pick the first maximal index
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight = a_i + a_{k0-i}
        w = a[idx] + a[k0 - idx]

        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contribution to the peak from the donor / receiver side
        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            # try another maximal index if possible
            if len(idx_max) > 1:
                k0 = idx_max[1]
                lo = max(0, k0 - (n - 1))
                hi = min(k0, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                w = a[idx] + a[k0 - idx]
                donor = idx[np.argmax(w)]
                receiver = idx[np.argmin(w)]
                contrib_d = a[k0 - donor]
                contrib_r = a[k0 - receiver]
                denom = 2.0 * (contrib_d - contrib_r)
                if denom <= 0.0:
                    break
            else:
                break

        # second largest convolution entry (excluding current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # how much we need to lower the peak
        delta_needed = (max_val - second_max) / denom

        # move as much as possible, but never make the donor negative
        max_delta = min(a[donor], delta_needed * 0.99)   # 0.99 for safety

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style moves: take a small amount from a
    relatively large entry and give it to a relatively small entry.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found (trailing zeros removed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ---------------------------------------------------------------
    # Choose problem size based on the allowed budget
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    elif budget_s >= 300:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σ a_i is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ---------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_val = np.inf
    best_params = (0.5, 0.0, 0.0)   # placeholders

    p_grid = np.linspace(0.2, 1.2, 15)
    q_grid = np.linspace(0.0, 8.0, 15)
    r_grid = np.array([0.0, 0.5, 1.0, 2.0])

    for p in p_grid:
        for q in q_grid:
            for r in r_grid:
                seq = _shape_sequence(n, float(p), float(q), float(r), target_sum)
                val = _score(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    best_params = (p, q, r)

    # ---------------------------------------------------------------
    # If a previous solution exists, reuse / resize it
    # ---------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_val:
            best_val = val
            best_seq = prev.copy()
            # we do not know the (p,q,r) for this vector – keep current params

    print(f"[{time.time():.2f}] Initial best after parametric grid = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 0b – local hill‑climbing in the (p,q,r) space
    # ---------------------------------------------------------------
    hill_end = time.time() + (deadline - time.time()) * 0.15
    sigma_p, sigma_q, sigma_r = 0.03, 0.4, 0.15

    p_cur, q_cur, r_cur = best_params
    while time.time() < hill_end:
        p_new = max(1e-6, p_cur + rng.normal(scale=sigma_p))
        q_new = max(0.0, q_cur + rng.normal(scale=sigma_q))
        r_new = max(0.0, r_cur + rng.normal(scale=sigma_r))

        seq = _shape_sequence(n, p_new, q_new, r_new, target_sum)
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            p_cur, q_cur, r_cur = p_new, q_new, r_new
            print(f"[{time.time():.2f}] Hill‑climb improved best = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ---------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_val - 1e-12:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD restart improved best = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact(best_seq,
                                                            deadline,
                                                            target_sum)
        if val_ref < best_val - 1e-12:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT improved best = {best_val:.6f}")

        # occasional global random moves to escape tiny plateaux
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_val - 1e-12:
            best_val = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_val:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ---------------------------------------------------------------
    # Phase C – Final polishing (higher temperature)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_val - 1e-12:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing zeros and return as Python list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
