def _simulated_annealing(a_init, target_sum, deadline):
    a = a_init.copy()
    n = a.shape[0]
    conv = _conv_fft(a)  # shape 2n-1
    max_val = conv.max()
    # track second largest for temperature
    sorted_vals = np.partition(conv, -2)
    second_max = sorted_vals[-2] if conv.size > 1 else max_val
    best_a = a.copy()
    best_max = max_val
    # temperature init
    T0 = (max_val - second_max) * 5.0 if max_val > second_max else max_val * 0.01
    start = time.time()
    rng = np.random.default_rng()
    iter = 0
    while time.time() < deadline:
        # compute fractions
        top_frac = 0.02
        low_frac = 0.02
        k_top = max(1, int(top_frac * n))
        k_low = max(1, int(low_frac * n))
        idx_desc = np.argpartition(-a, k_top-1)[:k_top]
        donor = idx_desc[rng.integers(k_top)]
        idx_asc = np.argpartition(a, k_low-1)[:k_low]
        receiver = idx_asc[rng.integers(k_low)]
        if donor == receiver:
            continue
        # delta bound
        max_delta = a[donor] * 0.5
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # compute conv_delta via slices
        conv_delta = np.zeros_like(conv)
        conv_delta[receiver:receiver+n] += a
        conv_delta[donor:donor+n] -= a
        # new max
        new_conv = conv + 2.0 * delta * conv_delta
        new_max = new_conv.max()
        # accept?
        elapsed = time.time() - start
        progress = elapsed / (deadline - start)
        T = T0 * (1.0 - progress)
        accept = False
        if new_max < max_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_max - max_val) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            # apply
            a[donor] -= delta
            a[receiver] += delta
            conv = new_conv
            max_val = new_max
            # track best
            if max_val < best_max:
                best_max = max_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA iter={iter} best_max={best_max:.6f}")
        iter += 1
    # compute final score
    final_score = float(2.0 * n * best_max / (target_sum**2))
    return best_a, final_score

"""
Multi‑phase optimizer for the step‑function benchmark.

The algorithm works in five stages:

1. **Parametric pre‑search** – evaluates a dense set of power‑law,
   exponential‑tail, and two‑piece power‑law families.  The best
   candidate is kept.

2. **Projected Adam gradient descent** – several short Adam runs with
   a soft‑max approximation of the max‑convolution, each starting from a
   jittered version of the current best point.

3. **Exact deterministic mass‑transfer** – a greedy scheme that moves
   mass from the most harmful entry to the least harmful one so that the
   current convolution peak is reduced to the second‑largest value.

4. **Simulated annealing with O(n) updates** – fast moves that shift a
   small amount of mass from a high‑value index to a low‑value index.
   The full convolution is updated incrementally, allowing thousands of
   moves per second.

5. **Final polishing** – a longer Adam run with a very large soft‑max
   temperature.

All phases respect the overall time budget; the best sequence found
before the deadline is returned (trailing near‑zero entries are trimmed).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _two_piece_power_law_sequence(
    n: int, p1: float, p2: float, c_frac: float, target_sum: float
) -> np.ndarray:
    """(i+1)^(-p1) for i < c, (i+1)^(-p2) otherwise."""
    c = max(1, int(c_frac * n))
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(i)
    a[:c] = i[:c] ** (-p1)
    a[c:] = i[c:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max approximation
# ----------------------------------------------------------------------
def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer_exact(
    a0: np.ndarray, deadline: float, target_sum: float, tol: float = 1e-12
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # pick one maximal index (first)
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[k0 - idx]
        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contribution terms
        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]
        denom = 2.0 * (contrib_d - contrib_r)

        if denom <= 0.0:
            # cannot improve this peak – give up
            break

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.5, delta_needed * 0.99)

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur = _score(a)
        if cur < best_val - 1e-12:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing with fast O(n) convolution updates (Phase C)
# ----------------------------------------------------------------------
def _simulated_annealing(
    a_init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    a = a_init.copy()
    n = a.shape[0]
    conv = _conv_fft(a)                # full convolution, length 2n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    # temperature based on initial peak gap
    conv_copy = conv.copy()
    conv_copy[np.argmax(conv_copy)] = -np.inf
    second_max = conv_copy.max()
    T0 = max( (max_val - second_max) * 5.0 , max_val * 0.01 )
    start = time.time()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # pick donor from the top fraction, receiver from the bottom fraction
        top_frac = 0.02
        low_frac = 0.02
        k_top = max(1, int(top_frac * n))
        k_low = max(1, int(low_frac * n))

        idx_desc = np.argpartition(-a, k_top - 1)[:k_top]
        donor = idx_desc[rng.integers(k_top)]

        idx_asc = np.argpartition(a, k_low - 1)[:k_low]
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        if max_delta <= 0.0:
            continue

        delta = rng.random() * max_delta

        # build the convolution delta = shift(receiver) - shift(donor)
        conv_delta = np.zeros_like(conv)
        conv_delta[receiver : receiver + n] += a
        conv_delta[donor    : donor    + n] -= a

        new_conv = conv + 2.0 * delta * conv_delta
        new_max = new_conv.max()

        elapsed = time.time() - start
        progress = elapsed / (deadline - start)
        T = T0 * (1.0 - progress)   # linear cooling

        accept = False
        if new_max < max_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(new_max - max_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor]   -= delta
            a[receiver] += delta
            conv = new_conv
            max_val = new_max

            if max_val < best_max:
                best_max = max_val
                best_a = a.copy()
                obj = float(2.0 * n * best_max / (target_sum ** 2))
                print(f"[{time.time():.2f}] SA iter={it:6d} best_obj={obj:.6f}")

        it += 1

    final_obj = float(2.0 * n * best_max / (target_sum ** 2))
    return best_a, final_obj

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine obeys the time budget, returns the best sequence found,

    deadline = start_time + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------------

    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ---------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # (a) pure power‑law grid
    for p in np.linspace(0.30, 0.90, 12):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail grid
    for p in np.linspace(0.30, 0.90, 10):
        for q in np.linspace(0.0, 7.0, 10):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) two‑piece power‑law random samples
    for _ in range(200):
        p1 = rng.random() * 0.8 + 0.2      # 0.2 – 1.0
        p2 = rng.random() * 0.8 + 0.2
        c_frac = rng.random() * 0.8 + 0.1  # 0.1 – 0.9
        seq = _two_piece_power_law_sequence(n, p1, p2, c_frac, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (d) reuse previous global solution if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()

    # (e) a few jittered variants of the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ---------------------------------------------------------------
    gd_deadline = time.time() + min(0.25 * budget_s, (deadline - time.time()) * 0.30)
    if gd_deadline > time.time() + 5.0:
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq, target_sum, gd_deadline)
        if val_gd < best_score:
            best_score = val_gd
