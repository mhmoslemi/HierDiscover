**Annealing**:

```
def _anneal(a, target_sum, deadline):
    a = a.copy()
    n = len(a)
    b = _full_convolution(a)
    best_a = a.copy()
    best_score = b.max()
    T0 = best_score * 0.1
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.01, 1 - elapsed / (deadline - start))
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # choose delta as a fraction of a[i]
        delta = a[i] * rng.uniform(0.01, 0.2)
        if delta == 0:
            continue
        # Evaluate move:
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < -1e-12:
            continue
        a_new = _project_to_simplex(a_new, target_sum)  # ensures constraints
        b_new = _full_convolution(a_new)
        new_score = b_new.max()
        # acceptance
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        if new_score <= b.max() or rng.random() < math.exp(-(new_score - b.max()) / (T + 1e-12)):
            a = a_new
            b = b_new
        # optionally symmetrize
        if time.time() < deadline - 1.0:
            a = _symmetrize(a)
            a = _project_to_simplex(a, target_sum)
            b = _full_convolution(a)
    return best_a, best_score
```

"""Step‑function optimizer for the “evaluate_sequence” challenge.

The algorithm works in three stages:

1. **Small‑scale search** (n≈2000).  A cheap random‑family scan,
   projected Adam on a smooth‑max surrogate and a deterministic
   hill‑climb produce a good seed.
2. **Upsampling & large‑scale refinement** (n≈20000).  The seed is
   linearly interpolated, symmetrised and further optimised by a
   longer Adam run (+softmax temperature schedule) and a second
   hill‑climb.
3. **Final polishing** – a short simulated‑annealing phase and a last
   Adam pass finish the vector.  All iterates are projected onto the
   ℓ¹‑simplex Σa = √(2 n) and symmetrised, which empirically lowers the
   maximal autocorrelation.

The code respects the global 1000 s wall‑clock budget and returns the
best sequence found.  It makes no assumptions about the imported
`evaluate_sequence` function other than its existence.
"""

import time
import math
import numpy as np
from operator import itemgetter


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric (a[i] = a[n‑1‑i])."""
    b = (a + a[::-1]) * 0.5
    return np.maximum(b, 0.0)  # ensures non‑negativity after averaging


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    The iteration stops when the wall‑clock reaches ``deadline``.
    Symmetrisation after each projection tends to lower the max‑convolution.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.12 * target_sum

    # soft‑max temperature schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)

        # occasional jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            a = _symmetrize(a)

        # increase smoothing (i.e. raise β) progressively
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisation & up‑sampling
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)

    # pick a family uniformly
    family = rng.integers(0, 7)
    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # piecewise exponential
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 4:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 5:                   # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                               # linear ramp (triangular)
        a = np.linspace(1.0, 0.0, n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    a = _symmetrize(a)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    a_new = _symmetrize(a_new)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))                         # index of current max

        # indices that influence b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]                 # a[i]·a[M-i]

        # source with the largest contribution
        i = idx[np.argmax(contrib)]
        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations (a[M‑j])
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # destination with minimal counterpart (but not the source)
        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # try to move a fairly large fraction of the source mass
        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # compute the new convolution (exact update, O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # accept if we improve
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # otherwise try a smaller move (binary search on delta)
        improved = False
        delta2 = delta * 0.5
        while delta2 > 1e-14:
            b_test = b.copy()
            b_test[i:i + n] -= 2.0 * delta2 * a
            b_test[j:j + n] += 2.0 * delta2 * a
            if k2i < b_test.size:
                b_test[k2i] += delta2 * delta2
            if k2j < b_test.size:
                b_test[k2j] += delta2 * delta2
            if kij < b_test.size:
                b_test[kij] -= 2.0 * delta2 * delta2
            if b_test.max() < b.max() - 1e-12:
                a[i] -= delta2
                a[j] += delta2
                b = b_test
                if b.max() < best_score - 1e-12:
                    best_score = b.max()
                    best_a = a.copy()
                improved = True
                break
            delta2 *= 0.5

        if not improved:
            # small random jitter to escape a flat region
            jitter = rng.normal(scale=1e-6, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _symmetrize(a)
            a = _project_to_simplex(a, math.sqrt(2.0 * n))
            b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal(a0: np.ndarray, target_sum: float, deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing that moves a small fraction of mass
    from a random source to a random destination.  Moves are always
    accepted if they lower the maximal convolution, otherwise they are
    accepted with probability exp(-(Δ)/T).  Temperature T decays linearly.
    """
    a = a0.copy()
    n = a.size
    b = _full_convolution(a)
    best_a = a.copy()
    best_score = b.max()

    start = time.time()
    T0 = best_score * 0.08 + 1e-9

    while time.time() < deadline:
        elapsed = time.time() - start
        T = max(1e-9, T0 * (1.0 - elapsed / (deadline - start)))

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a modest fraction of the source
        delta = a[i] * rng.uniform(0.01, 0.15)
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < -1e-12:
            continue

        # exact update of the convolution (same pattern as hill‑climb)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        accept = False
        if new_score < b.max():
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(new_score - b.max()) / T):
                accept = True

        if accept:
            a = a_new
            b = b_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        # occasional symmetrisation (helps keep the autocorrelation flat)
        if (time.time() % 7) < 0.01:
            a = _symmetrize(a)
            a = _project_to_simplex(a, target_sum)
            b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    ``evaluate_sequence`` under a wall‑clock budget.

      1. Small‑scale optimisation (n≈2000) to obtain a good seed.
      2. Upsample to a larger size (n≈20000) and perform a longer Adam run,
         deterministic hill‑climb and simulated‑annealing fine‑tuning.
      3. Final projection onto the simplex and optional symmetry enforcement.
    """

    deadline = time.time() + budget_s - 0.6  # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Small‑scale optimisation (n = 2000)
    # ------------------------------------------------------------------

    target_small = math.sqrt(2.0 * n_small)

    best_small = None
    best_score_small = np.inf

    # allocate roughly 12 % of the total budget to this phase
    phase_deadline = time.time() + 0.12 * (budget_s - 2.0)
    while time.time() < phase_deadline:

        # quick Adam (≈15 s per run)
        adam_dead = min(phase_deadline, time.time() + 15.0)
        a_opt = _adam_optimize(a0, target_small, adam_dead, rng)
        score = _max_convolution_fft(a_opt)

            best_small = a_opt.copy()

            a_hc, sc_hc = _hill_climb_deterministic(
                a_opt, _full_convolution(a_opt), time.time() + 4.0)
            if sc_hc < best_score_small:
                best_score_small = sc_hc
                best_small = a_hc.copy()

    # ------------------------------------------------------------------
    # 2) Upsample to a larger dimension (n = 20000) and refine
    # ------------------------------------------------------------------
    n_large = 20000
    target_large = math.sqrt(2.0 * n_large)

    if best_small is None:
        best_small = _random_initial_sequence(n_small, rng)

    a_large = _upsample_sequence(best_small, n_large)

    # allocate the bulk of the remaining budget
    remaining = deadline - time.time()
    if remaining < 10.0:
        # not enough time – fall back to the small solution
        final_seq = best_small
        final_score = best_score_small
    else:
        # --- Adam optimisation (≈70 % of remaining time) -----------------
        adam_deadline = time.time() + 0.70 * remaining
        a_large_opt = _adam_optimize(a_large, target_large,
                                     adam_deadline, rng)
        score_adam = _max_convolution_fft(a_large_opt)

        # --- Deterministic hill‑climb ------------------------------------
        hc_deadline = min(deadline - 0.5,
                          time.time() + 0.25 * remaining)
        a_hc, score_hc = _hill_climb_deterministic(
            a_large_opt, _full_convolution(a_large_opt), hc_deadline)

        # keep the best of the two
        if score_hc < score_adam:
            best_seq = a_hc.copy()
            best_score = score_hc
        else:
            best_seq = a_large_opt.copy()
            best_score = score_adam

        # --- Simulated annealing (remaining budget) -----------------------
        if time.time() + 2.0 < deadline:
            anneal_deadline = deadline - 0.3
            best_seq, best_score = _anneal(best_seq, target_large,
                                           anneal_deadline, rng)

        final_seq = best_seq
        final_score = best_score

    # ------------------------------------------------------------------
    # 3) Final tidy‑up
    # ------------------------------------------------------------------
    final_seq = _symmetrize(final_seq)
    final_seq = _project_to_simplex(final_seq,
                                    math.sqrt(2.0 * final_seq.size))

    # drop trailing near‑zeros (helps a tiny bit with the evaluator)
    nz = np.where(final_seq > 1e-12)[0]
    if nz.size:
        final_seq = final_seq[: nz[-1] + 1]
    else:
        final_seq = final_seq[:1]

    # ------------------------------------------------------------------
    # 4) Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={final_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return final_seq.tolist()
