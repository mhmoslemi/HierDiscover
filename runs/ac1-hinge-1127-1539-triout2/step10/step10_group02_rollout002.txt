"""
Optimisation pipeline for the step‑function problem.

The goal is to minimise the maximum entry of the self‑convolution
a * a while keeping the ℓ¹‑norm Σ a = √(2 n) and a_i ≥ 0.
For a normalised sequence the objective equals the maximal convolution value.

Strategy
--------
1. Choose a problem size n (12000 for a long budget, smaller otherwise) and the
   target ℓ¹‑norm `target_sum = sqrt(2 n)`.
2. Build a diverse pool of candidate shapes:
   * power‑law (i+1)⁻ᵖ for a fine grid of p,
   * exponential decay e^{‑λ i},
   * sine‑raised shapes sin(π(i+½)/n)ᵖ.
   Every shape is projected onto the simplex {x≥0, Σx = target_sum}
   and a small jittered copy is also added.
3. Evaluate all candidates with an FFT‑based convolution and keep the best K.
4. Refine each of the K best candidates with a projected momentum GD on a
   soft‑max approximation of the max‑convolution (increasing β up to 3·10⁵).
5. Perform a short sub‑gradient descent on the true max‑convolution.
6. Apply a deterministic mass‑transfer / simulated‑annealing phase that moves
   mass from positions contributing most to the current peak to positions that
   contribute the least.
7. Final polishing with a high‑β soft‑max GD.
8. Trim trailing zeros, store the result in the global variable
   `height_sequence_1` for warm‑starts, and return it as a plain Python list.

All heavy work uses FFT (O(n log n)) and careful time checks to stay inside the
budget (default 1000 s).  The implementation is self‑contained and does not
require any external files or network access.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Unnormalised power‑law (i+1)⁻ᵖ."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Unnormalised exponential decay e^{‑lam·i}."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """Unnormalised (sin(π·(i+0.5)/n))ᵖ."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (used in GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (rev(a) ⋆ p) restricted to the central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected momentum GD on soft‑max
# ----------------------------------------------------------------------
def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 3e5,
                                   step_factor: float = 0.6,
                                   gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Momentum GD on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase X – Exact max sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2 * a[kmax - i] when valid
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic mass‑transfer (local annealing)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass transfer that reduces the current peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, kmax‑i)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        # choose a receiver among the low‑contributing indices
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the max self‑convolution.
    Returns a Python list of floats ready for `evaluate_sequence`.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Helper to evaluate a candidate (fast)
    # --------------------------------------------------------------
    def _eval(a: np.ndarray) -> float:
        return _max_convolution_fft(a)

    # --------------------------------------------------------------
    # Build a pool of parametric candidates
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # Warm‑start from previous solution (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        prev = _project_to_simplex(prev, target_sum)
        candidates.append(prev)

    # Power‑law grid
    for p in np.linspace(0.35, 0.85, 51):
        seq = _power_law_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

        # jittered copy
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Exponential decay grid
    for lam in np.linspace(5e-4, 0.012, 40):
        seq = _exponential_sequence(n, lam)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Sine‑power shapes
    for p in np.linspace(0.2, 2.0, 30):
        seq = _sin_power_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Random permutations of a few good candidates
    for base_seq in candidates[:6]:
        perm = base_seq.copy()
        rng.shuffle(perm)
        candidates.append(perm)

    # --------------------------------------------------------------
    # Quick evaluation of the pool
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:
        val = _eval(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    # Keep a handful of the best for intensive refinement
    top_k = 6
    best_seq, best_score = scored[0][1].copy(), scored[0][0]
    print(f"[{time.time():.2f}] Initial best score from pool = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Soft‑max momentum GD on the best few candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        gd_time = remaining * 0.35
        gd_deadline = time.time() + gd_time
        # allocate roughly equal budget per top candidate
        per_candidate = gd_time / top_k
        for i in range(top_k):
            if time.time() >= gd_deadline - 0.5:
                break
            seq0 = scored[i][1]
            sub_dead = time.time() + min(per_candidate, gd_deadline - time.time())
            a_opt, val_opt = _annealed_gradient_descent_ext(
                seq0, sub_dead, target_sum, max_beta=3e5,
                step_factor=0.6, gamma=0.995
            )
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] GD improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase X – Sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        sub_dead = time.time() + remaining * 0.25
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum, sub_dead,
                                            step_factor=0.55, gamma=0.95)
        if val_sg < best_score:
            best_score, best_seq = val_sg, a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        mt_dead = time.time() + remaining * 0.30
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                    target_sum,
                                                    start_temp=0.6,
                                                    end_temp=1e-5,
                                                    low_frac=0.04)
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Final polishing with a high‑β soft‑max GD
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_fin, val_fin = _annealed_gradient_descent_ext(
            best_seq,
            time.time() + remaining,
            target_sum,
            max_beta=5e5,
            step_factor=0.5,
            gamma=0.998,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑process, evaluate, and store globally
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)
    # trim trailing near‑zero entries to keep the representation compact
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]

    final_seq = final_arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
