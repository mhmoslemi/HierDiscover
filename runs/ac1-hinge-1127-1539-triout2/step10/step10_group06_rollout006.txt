"""
Improved non‑negative step sequence search.

The algorithm consists of five phases, all respecting a hard time budget.
1) Warm‑start (resize a previously found sequence or a symmetric power‑law).
2) Phase A – Adam (multiplicative) descent on a soft‑max surrogate of the
   maximal autocorrelation entry.  The current iterate is symmetrised after
   each update and projected onto the simplex Σa = √(2 n).
3) Phase B – Multiplicative sub‑gradient descent on the true max‑convolution.
4) Phase C – Projected gradient descent with momentum, random restarts and
   enforced symmetry.
5) Phase D – Exact local mass‑transfer hill‑climbing with simulated‑annealing
   acceptance.  After every accepted move the vector is symmetrised again.

All phases keep track of the best sequence (lowest max‑convolution).  The
final sequence is trimmed, evaluated with the provided `evaluate_sequence`,
stored globally for warm‑starts and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} (exact symmetry)."""
    return (a + a[::-1]) * 0.5


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)  – length n
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Adam descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative Adam‑style descent on the soft‑max surrogate.
    After each update the vector is symmetrised and projected onto the simplex.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    a = _symmetrize(a)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    beta = 1.0
    max_beta = 65536.0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # gradient of soft‑max approximation
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base learning rate decays with β
        lr = 0.07 / (beta + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # multiplicative update
        a = a * np.exp(-step)

        # enforce symmetry and feasibility
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _symmetrize(a)
            a = _project_to_simplex(a, target_sum)

        # increase β (sharpen the soft‑max)
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Subgradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent directly on max(a*a).
    Symmetrisation is applied after each update.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    a = _symmetrize(a)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # full convolution
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]   # sub‑gradient of the max entry

        # linear schedule for η
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.18
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)

        # symmetry & projection
        a = _symmetrize(a)
        a = a * (target_sum / a.sum())

        # jitter every so often
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _symmetrize(a)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Phase C – Projected GD with momentum, restarts & symmetry
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _symmetrize(a)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.43
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0013 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _symmetrize(a)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(28.0, remaining * 0.34)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.018, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _symmetrize(start)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – Exact local mass‑transfer refinement (symmetrised)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    a = _symmetrize(a)
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # locate donor and receiver (based on current max entry)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contribution of each index

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # propose tiny mass transfer
        max_delta = min(a[donor] * 0.04, target_sum * 0.0009)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution exactly (O(n) updates)
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # simulated annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            a = _symmetrize(a)                # force symmetry after move
            b = _conv_fft(a)                  # recompute full convolution (still cheap)
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if time.time() - last_recalc > 13.0:
            a = a * (target_sum / a.sum())
            a = _symmetrize(a)
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional tiny double‑jitter
        if int(elapsed * 101) % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.004, target_sum * 0.00035)
                a[i] -= eps
                a[j] += eps
                a = _symmetrize(a)
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.45      # safety margin for finalisation

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init = _symmetrize(init)
    else:
        # symmetric power‑law shape: a_i ∝ ((i+1)*(n-i))^{-p}
        p = 0.55
        x = np.arange(n, dtype=np.float64)
        init = ((x + 1) * (n - x)) ** (-p)
        init = _symmetrize(init)

    init = init * (target_sum / init.sum())
    init = _symmetrize(init)

    # --------------------------------------------------------------
    # Phase A – Adam descent on soft‑max surrogate (~15 % of budget)
    # --------------------------------------------------------------
    phase_a_deadline = time.time() + 0.15 * (deadline - time.time())
    a_a, val_a = _adam_descent(init, target_sum, phase_a_deadline)
    best_seq = a_a
    best_score = val_a
    print(f"[{time.time():.2f}] After Adam descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent on the true max‑convolution (~15 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_b_deadline = time.time() + 0.15 * (deadline - time.time())
        a_b, val_b = _subgradient_descent(best_seq, target_sum, phase_b_deadline)
        if val_b < best_score:
            best_seq, best_score = a_b, val_b
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts (~30 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_c_deadline = time.time() + 0.30 * (deadline - time.time())
        a_c, val_c = _gradient_descent_with_restarts(best_seq,
                                                     target_sum,
                                                     phase_c_deadline)
        if val_c < best_score:
            best_seq, best_score = a_c, val_c
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – exact local refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_d, val_d = _local_refine(best_seq, target_sum, deadline)
        if val_d < best_score:
            best_seq, best_score = a_d, val_d
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # trim trailing very small entries
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
