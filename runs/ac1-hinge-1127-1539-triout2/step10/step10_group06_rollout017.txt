            jitter = rng.normal(scale=target_sum*0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum/a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamExp it={t} beta={beta:.1f} score={best_val:.6f}")
        # increase beta schedule
        if t % 100 ==0 and beta < max_beta:
            beta = min(beta*2.0, max_beta)
    return best_a, best_val

"""
Improved search for a non‑negative step sequence minimizing the benchmark.
The algorithm uses:
1. An AdaGrad / Adam style projected gradient descent on a soft‑max
   surrogate of the convolution max (Phase A).
2. Multiplicative sub‑gradient descent directly on the true max entry
   (Phase B).
3. Projected gradient descent with momentum and random restarts (Phase C).
4. A longer simulated‑annealing based local refinement that moves tiny
   amounts of mass between indices while updating the convolution in O(n)
   (Phase D).
The routine adapts the problem size to the time budget, uses a larger
sequence (up to 20 000 elements) when time permits and always falls back to
a power‑law warm‑start (or the previously saved `height_sequence_1`).  The
final sequence is projected onto the feasible simplex and returned as a
plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (Phase A)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # safe exponentiation – values are ≤0
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p) – length n
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Adam‑style projected gradient descent on soft‑max
# ----------------------------------------------------------------------


def _adam_exponential_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam / AdaGrad style multiplicative descent on the soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    rng = np.random.default_rng()
    beta = 1.0
    max_beta = 1e6
    t = 0

    while time.time() < deadline:
        t += 1
        # soft‑max gradient
        grad, _ = _gradient_softmax(a, beta)

        # Adam update (additive, then projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step‑size adaption – larger for early β, smaller as β grows
        base_lr = target_sum * 0.25
        step = base_lr / (np.sqrt(v_hat) + eps) / (beta + 1.0)

        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # increase soft‑max temperature
        if t % 120 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamExp it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – multiplicative sub‑gradient descent on true max entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # full convolution
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient of the max‑entry
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear decay of step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 3.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # multiplicative update guarantees positivity
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with restarts (momentum version)
# ----------------------------------------------------------------------


def _momentum_gd(a0: np.ndarray,
                target_sum: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate using Nesterov‑style
    momentum and occasional random restarts.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1e6
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        # adaptive step size – normalise by the momentum norm
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * 0.5) / (norm * (beta + 1.0))
            a = a - lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred steps
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] MomGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β slowly – more faithful to the true max
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        # occasional restart from a slightly jittered version of the current best
        if it % 900 == 0 and it > 0:
            cand = best_a + rng.normal(scale=target_sum * 0.004, size=a.shape)
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_sum)
            cand_val = _score(cand)
            if cand_val < best_val:
                a = cand
                best_val = cand_val
                best_a = cand.copy()
                print(f"[{time.time():.2f}] Restart improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _gd_with_restarts(init: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Run several short momentum‑GD runs with random jitter; keep the best.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 4.0:
        remaining = deadline - time.time()
        # each sub‑run gets roughly 25 % of the remaining budget, but at least 5 s
        run_budget = max(5.0, min(30.0, remaining * 0.25))
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _momentum_gd(start, target_sum,
                               time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – simulated‑annealing mass‑transfer refinement (fast O(n) update)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)  # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # initial temperature proportional to the current excess over the average
    T0 = max(best_val - 1.0, 0.01) * 0.08
    last_recalc = start

    while time.time() < deadline:
        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the active peak
        contrib = a[idx] * a[k_max - idx]

        # pick a donor proportional to its contribution
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a receiver among indices with relatively small contribution
        # (the lower 30 % quantile of contributions)
        thresh = np.percentile(contrib, 30)
        low_candidates = idx[contrib <= thresh]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(0, n)

        if donor == receiver:
            continue

        # propose a tiny transfer
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution in O(n) using the linearity of the convolution
        b_new = b.copy()
        # linear part: 2·a·Δ
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic adjustments
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # simulated‑annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # full recomputation every ~10 s to avoid drift
        if time.time() - last_recalc > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")
            last_recalc = time.time()

        # tiny jitter every few hundred iterations (based on elapsed time)
        if int(elapsed * 100) % 2100 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.003, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – Adam‑style descent on soft‑max surrogate (≈15 % budget)
    # --------------------------------------------------------------
    exp_deadline = time.time() + 0.15 * (deadline - time.time())
    a_exp, val_exp = _adam_exponential_descent(init, target_sum, exp_deadline)
    best_seq = a_exp
    best_score = val_exp
    print(f"[{time.time():.2f}] After Adam‑exp descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent on the true max‑convolution (≈18 %)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.18 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts (≈30 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gd_with_restarts(best_seq,
                                          target_sum,
                                          gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – exact local refinement (remaining time)
    # --------------------------------------------------------------

"""
Improved search for a non‑negative step sequence minimizing the benchmark.
The algorithm uses:

1. Adam / AdaGrad style projected gradient descent on a soft‑max
   surrogate of the convolution max (Phase A).
2. Multiplicative sub‑gradient descent directly on the true max entry
   (Phase B).
3. Momentum‑based projected gradient descent with random restarts
   (Phase C).
4. A longer simulated‑annealing / hill‑climbing mass‑transfer refinement
   that updates the convolution in O(n) (Phase D).

The problem size is adapted to the time budget (up to 20 000 coefficients).
If a previous construction (`height_sequence_1`) is available it is used as
a warm‑start; otherwise a simple power‑law seed is generated.
The final sequence is projected onto the feasible simplex and returned as
a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) / (sum(a)²).  With Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phase A and Phase C)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))        # safe: b - max_b ≤ 0
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Adam‑style projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_exponential_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam/AdaGrad style descent on the soft‑max surrogate.
    The update is additive; positivity is enforced by a projection onto the
    simplex after each step.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]

    best_a = a.copy()
    best_val = _score(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    rng = np.random.default_rng()
    beta = 1.0
    max_beta = 1e6
    t = 0

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule that shrinks as β grows
        base_lr = target_sum * 0.25
        step = base_lr / (np.sqrt(v_hat) + eps) / (beta + 1.0)

        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # increase the soft‑max temperature
        if t % 120 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamExp it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – multiplicative sub‑gradient descent on the true max entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]          # sub‑gradient of the active entry

        # linearly decaying step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start, eta_end = 3.0, 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)               # multiplicative update
        a = a * (target_sum / a.sum())

        if it % 250 == 0:                       # jitter
            jitter = rng.normal(scale=target_sum * 0.0004, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with momentum and random restarts
# ----------------------------------------------------------------------


def _momentum_gd(a0: np.ndarray,
                target_sum: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate using momentum.
    Periodically restarts from a jittered copy of the best point seen so far.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a, best_val = a.copy(), _score(a)

    beta = 1.0
    max_beta = 1e6
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad

        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * 0.5) / (norm * (beta + 1.0))
            a = a - lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:                      # jitter
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] MomGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 150 == 0 and beta < max_beta:   # raise temperature slowly
            beta = min(beta * 2.0, max_beta)

        # occasional restart from a jittered version of the current best
        if it % 900 == 0 and it > 0:
            cand = best_a + rng.normal(scale=target_sum * 0.004, size=a.shape)
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_sum)
            cand_val = _score(cand)
            if cand_val < best_val:
                a, best_val, best_a = cand, cand_val, cand.copy()
                print(f"[{time.time():.2f}] Restart improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _gd_with_restarts(init: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Run several short momentum‑GD runs with random jitter and keep the best.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 4.0:
        remaining = deadline - time.time()
        run_budget = max(5.0, min(30.0, remaining * 0.25))
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _momentum_gd(start, target_sum,
                               time.time() + run_budget)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – simulated‑annealing mass‑transfer refinement (O(n) updates)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing / simulated‑annealing on the true max‑convolution.
    Tiny amounts of mass are moved from “bad’’ indices (large contribution
    to the current maximal entry) to “good’’ ones.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a, best_val = a.copy(), _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = max(best_val - 1.0, 0.01) * 0.08          # initial temperature
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the active peak
        contrib = a[idx] * a[k_max - idx]

        # donor – proportional to its contribution
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – among the low‑contribution tail
        thresh = np.percentile(contrib, 30)
        low = idx[contrib <= thresh]
        receiver = rng.choice(low) if low.size else rng.integers(0, n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) convolution update
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))

        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # full recomputation periodically to avoid drift
        if time.time() - last_recalc > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")
            last_recalc = time.time()

        # occasional tiny jitter
        if int(elapsed * 100) % 2100 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.003, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val, best_a = cur_val, a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5                # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))   # power‑law seed

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – Adam‑style descent on the soft‑max surrogate (≈15 % budget)
    # --------------------------------------------------------------
    exp_deadline = time.time() + 0.15 * (deadline - time.time())
    a_exp, val_exp = _adam_exponential_descent(init, target_sum, exp_deadline)
    best_seq, best_score = a_exp, val_exp
    print(f"[{time.time():.2f}] After Adam‑exp descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent on the true max‑convolution (≈18 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.18 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts (≈30 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gd_with_restarts(best_seq, target_sum, gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – exact local refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
