"""
Propose a step‑function with a very low max‑auto‑convolution value.

The pipeline builds multiple initial candidates (power‑law, Dirichlet,
jittered copies, random permutations) and runs several optimisation
phases:

1. Soft‑max projected gradient descent (fast reduction of the peak).
2. Exact max‑sub‑gradient descent.
3. Deterministic mass‑transfer refinement.
4. Permutation‑annealing (swap‑based simulated annealing).
5. Final high‑β soft‑max polishing.
6. **New** local hill‑climbing on the actual max‑convolution using
   random mass‑transfer / swap moves with a temperature schedule.

All stages respect the overall time budget (≈1000 s).  The best
sequence found is stored globally as ``height_sequence_1`` for warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (sum is normalised to √(2n) ⇒ score == max conv)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective value (equal to max (a*a) when sum=√(2n))."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps sum fixed and non‑negative.
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, target_sum: float) -> np.ndarray:
    rng = np.random.default_rng()
    # Gamma(1,1) == Exp(1) gives a Dirichlet(1,…,1) (uniform on simplex)
    a = rng.exponential(scale=1.0, size=n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Soft‑max projected gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start,
                                            time.time() + run_budget,
                                            target_sum,
                                            max_beta=1e5,
                                            step_factor=0.8,
                                            gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score

# Phase F – Local hill‑climbing on the true max‑convolution
# ----------------------------------------------------------------------


def _local_hill_climb(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Randomised mass‑transfer / swap hill‑climb with simulated‑annealing."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    start_time = time.time()
    start_temp = cur_max * 0.01 + 1e-6
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # choose move type
        if rng.random() < 0.2:
            # pure swap
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            # mass transfer from donor -> receiver
            # donor based on a (large values) 70% weighted, else random
            if rng.random() < 0.7:
                donor = np.argmax(a)
            else:
                donor = rng.choice(n, p=a / a.sum())
            receiver = rng.integers(0, n)
            while receiver == donor:
                receiver = rng.integers(0, n)

            max_delta = a[donor] * 0.5
            delta = rng.random() * max_delta
            a_new = a.copy()
            a_new[donor] -= delta
            a_new[receiver] += delta
            # projection not needed – sum unchanged and all entries stay non‑negative

        new_conv = _conv_fft(a_new)
        new_max = new_conv.max()

        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
            a = a_new
            conv = new_conv
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] HillClimb it={it:5d} new best={best_max:.6f}")

        it += 1

    return best_a, best_max

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the time budget (≈1000 s) and returns the best
    sequence found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Problem size (larger n → finer granularity)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        if budget_s >= 900:
            n = 15000
        elif budget_s >= 600:
            n = 12000
        else:
            n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a diverse set of initial candidates
    # --------------------------------------------------------------
    init_seqs = []

    # power‑law family
    for p in (0.35, 0.45, 0.55, 0.65):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # Dirichlet (uniform on simplex)
    for _ in range(2):
        init_seqs.append(_dirichlet_sequence(n, target_sum))

    # warm‑start from previous run, resized if needed
    if "height_sequence_1" in globals():
        init_seqs.append(_resize_sequence(prev_seq, n))

    rng = np.random.default_rng()
    # jittered copies of existing seeds
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # random permutations of a couple of power‑law candidates
    for p in (0.5, 0.6):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # --------------------------------------------------------------
    # Phase A – Soft‑max projected gradient descent (multiple restarts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.25

    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        cand, val = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Global best after GD = {best_score:.6f}")

    if best_seq is None:  # fall back
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase X – Exact max sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = time.time() + (deadline - time.time()) * 0.15
        cand, val = _subgradient_descent(best_seq, target_sum, sg_deadline,
                                         step_factor=0.6, gamma=0.95)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.25
        cand, val = _deterministic_mass_transfer(best_seq, mt_deadline,
                                                 target_sum,
                                                 start_temp=0.5,
                                                 end_temp=1e-4,
                                                 low_frac=0.05)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Re‑ordering heuristics
    # --------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        val = _max_convolution_fft(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        cand, val = _permutation_annealing(best_seq, target_sum,
                                           perm_deadline, init_score=best_score)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing (high‑β soft‑max GD)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        cand, val = _annealed_gradient_descent(best_seq,
                                               deadline,
                                               target_sum,
                                               max_beta=3e5,
                                               step_factor=0.6,
                                               gamma=0.995)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase F – Local hill‑climbing (simulated‑annealing moves)
    # --------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + (deadline - time.time()) * 0.10
        cand, val = _local_hill_climb(best_seq, target_sum, hill_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
