"""
Optimisation of a non‑negative step function for minimise
evaluate_sequence().

The algorithm is a multi‑stage optimiser:

1. **Initial seeding** – start from a previously found sequence (if
   available) or from a random point on the ℓ¹‑simplex.
2. **Broad random exploration** – generate thousands of candidates from
   a handful of simple families (exponential, power‑law, mixed and
   Dirichlet) and keep the best few.
3. **Projected Adam on a smooth‑max surrogate** – run many long Adam
   restarts (the smooth‑max is the log‑sum‑exp of the convolution) and
   project onto the simplex after every step.
4. **Deterministic hill‑climb** – exact greedy moves that shift mass from
   the indices that most contribute to the current worst convolution
   entry.
5. **Simulated‑annealing fine‑tuning** – small random mass transfers
   accepted with a temperature schedule, allowing the search to escape
   shallow local minima.
6. **Final polishing** – one more Adam run + hill‑climb.

The whole routine respects the wall‑clock budget (default 1000 s) and
returns the best sequence found.  The score printed at the end is the
value returned by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from operator import itemgetter


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a⊛a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⊛a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of  (1/β)·log Σ exp(β·b)   where   b = a⊛a.
    Returns (g, b)   with   g = 2·(a ⋆ p) ,   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # ----- b = a⊛a (FFT) -----
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # ----- gradient  g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until ``deadline`` (wall‑clock).
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning‑rate – proportional to the simplex radius
    base_lr = 0.12 * target_sum

    # smoothing schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake the system
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase the smoothing parameter
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Sample a non‑negative vector on the simplex Σ a = √(2 n).
    The vector is drawn from a small collection of hand‑crafted families.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves mass away from the current
    worst convolution entry.  Returns the best vector seen and its
    maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # index of the current maximal convolution entry
        M = int(np.argmax(b))

        # source indices that appear in b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # source with the biggest contribution
        i = idx[np.argmax(contrib)]
        if a[i] < 1e-15:
            break

        # pick a destination that minimises the counterpart term
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]

        # try destinations in increasing order of counterpart
        order = np.argsort(counterpart[i_low:i_high + 1])
        j = None
        for cand in np.arange(i_low, i_high + 1)[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # amount of mass to move (up to 80 % of the source)
        delta = 0.8 * a[i]
        if delta <= 0.0:
            break

        # ----- evaluate the move (exact linear update) -----
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ----- accept if we improve -----
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b_test.max() < best_score - 1e-12:
                        best_score = b_test.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # escape possible plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _simulated_annealing(a_start: np.ndarray,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a small amount of mass between two positions.
    Returns (best_vector, best_score) where the score is the *raw*
    maximal autocorrelation entry (i.e. the value fed to evaluate_sequence).
    """
    a = a_start.copy()
    n = a.size
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    # temperature schedule – start a few percent of the current score,
    # decay linearly to a tiny value.
    start_T = 0.02 * cur_score
    end_T = 1e-7

    total_time = deadline - time.time()
    while time.time() < deadline:
        remaining = deadline - time.time()
        # linear interpolation of temperature
        T = start_T * (remaining / total_time) + end_T * (1 - remaining / total_time)

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_fft(a_new)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                # occasional log
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # 1)  Choose a working length – 2000 works well and keeps the search cheap.
    # ------------------------------------------------------------------
    n = 2000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2)  Seed from previously discovered construction if it exists.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            prev = _project_to_simplex(prev, target_sum)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1, score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 3)  Broad random exploration (≈15 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        explore_deadline = time.time() + min(remaining * 0.15, 20.0)
        # generate a few thousand random candidates from the families
        while time.time() < explore_deadline:
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Random‑family improvement {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Projected Adam – many restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200.0:
        # allocate 60 % of the remaining time to Adam
        adam_deadline = time.time() + remaining * 0.60
        # decide how many restarts we can afford (≈5 s each)
        min_restart = 5.0
        n_restarts = max(1, int((adam_deadline - time.time()) // min_restart))
        per_restart = (adam_deadline - time.time()) / n_restarts

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            start_a = best_seq.copy() if r == 0 else _random_initial_sequence(n, rng)
            this_deadline = min(deadline - 0.2, time.time() + per_restart)
            a_opt = _adam_optimize(start_a, target_sum, this_deadline)
            score = _max_convolution_fft(a_opt)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] Adam restart {r+1}/{n_restarts} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5)  Deterministic hill‑climb on the exact convolution
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        hc_deadline = min(deadline - 0.2, time.time() + remaining * 0.30)
        a_cur = _project_to_simplex(best_seq, target_sum)
        b_cur = _full_convolution(a_cur)
        a_hc, hc_score = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
        if hc_score < best_score:
            best_score = hc_score
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6)  Simulated‑annealing fine‑tuning (use the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        sa_deadline = deadline - 0.5
        a_sa, sa_score = _simulated_annealing(best_seq, sa_deadline)
        if sa_score < best_score:
            best_score = sa_score
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] SA final improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7)  Final polishing – a short Adam run + a quick hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        final_deadline = deadline - 0.5
        a_ref = _adam_optimize(best_seq, target_sum, final_deadline)
        ref_score = _max_convolution_fft(a_ref)
        if ref_score < best_score:
            best_score = ref_score
            best_seq = a_ref.copy()
            print(f"[{time.time():.1f}] Final Adam polishing to {best_score:.6f}")

        # one quick hill‑climb
        b_final = _full_convolution(best_seq)
        a_hc2, score_hc2 = _hill_climb_deterministic(best_seq, b_final,
                                                      deadline - 0.3)
        if score_hc2 < best_score:
            best_score = score_hc2
            best_seq = a_hc2.copy()
            print(f"[{time.time():.1f}] Final hill‑climb polishing to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8)  Trim trailing near‑zeros for a tidy output
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback – uniform distribution (should never be needed)
        best_seq = np.full(n, target_sum / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 9)  Report the official evaluation (using the supplied function)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:      # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
