def propose_candidate(seed=42, budget_s=1000):
    rng = default_rng(seed)
    deadline = time.time()+budget_s-0.5

    best_seq = None
    best_score = np.inf

    # schedule lengths
    schedule = [2000, 4000, 8000, 20000]

    for n in schedule:
        if time.time()>deadline: break
        target = math.sqrt(2.0*n)
        # initialize: if have best_seq, upsample else random
        if best_seq is not None:
            init = _upsample_sequence(best_seq, n)
        else:
            init = _random_initial_sequence(n, rng)
        # build a set of candidate vectors: use many families
        candidates = []
        # param families
        for _ in range(30):
            # power law with offset
            p = rng.uniform(0.2, 0.9)
            offset = rng.uniform(0, 5.0)
            a = (np.arange(n, dtype=float) + offset)**(-p)
            a = _project_to_simplex(a, target)
            candidates.append(a)
        for _ in range(30):
            lam = rng.exponential(scale=0.02)
            a = np.exp(-lam * np.arange(n, dtype=float))
            a = _project_to_simplex(a, target)
            candidates.append(a)
        for _ in range(30):
            p = rng.uniform(0.2, 0.9)
            lam = rng.exponential(scale=0.02)
            a = (np.arange(n, dtype=float)+1.0) ** (-p) * np.exp(-lam*np.arange(n))
            a = _project_to_simplex(a, target)
            candidates.append(a)
        # add init
        candidates.append(init)
        # evaluate and keep top 5
        scores = [_max_convolution_fft(a) for a in candidates]
        top_idx = np.argsort(scores)[:5]
        population = [(scores[i], candidates[i]) for i in top_idx]
        # Now iterative refinement
        # allocate remaining time fraction for this n: 
        rem = deadline - time.time()
        # allocate portion for inner loops
        inner_deadline = time.time() + rem * 0.9
        while time.time() < inner_deadline:
            # selection: pick two best
            population.sort(key=lambda x: x[0])
            parent1 = population[0][1]
            parent2 = population[1][1] if len(population)>1 else parent1
            # crossover (average or weighted)
            child = (parent1 + parent2) / 2.0
            child = _project_to_simplex(child, target)
            # mutate
            if rng.random() < 0.5:
                # add Gaussian noise scaled by target
                child += rng.normal(scale=target*0.001, size=n)
                child = np.maximum(child, 0.)
                child = _project_to_simplex(child, target)
            # local Adam short run
            # compute per-run budget time as maybe 5 seconds or smaller
            sub_deadline = time.time() + 5.0
            child_opt = _adam_optimize(child, target, sub_deadline)
            child_score = _max_convolution_fft(child_opt)
            # add to population
            population.append((child_score, child_opt))
            # keep top 10
            population = sorted(population, key=itemgetter(0))[:10]
        # after inner loop, keep best in this n
        best_local_seq, best_local_score = min(population, key=itemgetter(0))
        if best_local_score < best_score - 1e-12:
            best_seq = best_local_seq.copy()
            best_score = best_local_score
            print(f"[time] n={n} improved best_score={best_score}")

        # final deterministic hillclimb on best_seq
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target)
            b_cur = _full_convolution(a_cur)
            h_deadline = deadline - 0.5
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, h_deadline)

"""
Simple evolutionary + projected‑Adam optimizer for the
max‑convolution problem.

The program builds a small population of candidates,
mutates them, refines them with Adam on a smooth‑max (log‑sum‑exp)
surrogate, and finally performs a deterministic hill‑climb that
exactly updates the convolution.  The length schedule grows up to
20 000 entries, which gives the optimizer many degrees of freedom
and typically drives the evaluation value below the previous best
≈ 1.508.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (fast)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.10 * target_sum

    # smoothing schedule (log‑sum‑exp temperature)
    beta = 1.5
    beta_max = 2e7
    beta_factor = 1.8

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-5,
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Random / parametric initialisations
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.2, 1.2)
        a = (idx + rng.uniform(0.0, 5.0) + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.012)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(scale=0.012)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.25 * n), int(0.75 * n))
        lam1 = rng.exponential(scale=0.018)
        lam2 = rng.exponential(scale=0.018)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length and projection onto the simplex."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climb (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves mass from the index that
    contributes most to the current maximal convolution entry to a
    location that hurts that entry the least.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    # small epsilon to avoid stagnation
    epsilon = 1e-12

    while time.time() < deadline:
        # index with maximal convolution
        M = int(np.argmax(b))

        # indices that affect b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # source index with largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < epsilon:
            break

        # counterpart values for possible destinations
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart = a[M - j_range]

        # pick destination with minimal counterpart (exclude i)
        order = np.argsort(counterpart)
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # move a moderate fraction of the source mass
        delta = a[i] * 0.6
        if delta <= epsilon:
            break

        # update convolution vector (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # accept move if it improves the max
        if new_max < b.max() - 1e-13:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-13:
                best_score = new_max
                best_a = a.copy()
            continue

        # otherwise try a smaller move (binary search)
        improved = False
        delta2 = delta * 0.5
        while delta2 > epsilon:
            b_test = b.copy()
            b_test[i:i + n] -= 2.0 * delta2 * a
            b_test[j:j + n] += 2.0 * delta2 * a
            if k2i < b_test.size:
                b_test[k2i] += delta2 * delta2
            if k2j < b_test.size:
                b_test[k2j] += delta2 * delta2
            if kij < b_test.size:
                b_test[kij] -= 2.0 * delta2 * delta2
            if b_test.max() < b.max() - 1e-13:
                a[i] -= delta2
                a[j] += delta2
                b = b_test
                if b.max() < best_score - 1e-13:
                    best_score = b.max()
                    best_a = a.copy()
                improved = True
                break
            delta2 *= 0.5

        if not improved:
            # occasional random jitter to break plateaus
            jitter = rng.normal(scale=1e-5, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            target = math.sqrt(2.0 * n)
            a = _project_to_simplex(a, target)
            b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimizes
    ``evaluate_sequence`` within the allotted wall‑clock budget.
    The routine uses a growing length schedule, a small evolutionary
    population and projected Adam on a smooth‑max surrogate, followed
    by a deterministic hill‑climb and a final polishing stage.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is available, seed from it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            prev = _project_to_simplex(prev, math.sqrt(2.0 * prev.size))
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – we grow the dimension gradually
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 20000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate (upsample previous best if possible)
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Build a small population of diverse candidates
        # --------------------------------------------------------------
        population = []

        # a few hand‑crafted families
        families = []

        # (i+shift)^-p
        for _ in range(15):
            p = rng.uniform(0.3, 0.9)
            shift = rng.uniform(0.0, 6.0)
            a = (np.arange(n, dtype=float) + shift + 1.0) ** (-p)
            families.append(a)

        # pure exponential
        for _ in range(15):
            lam = rng.exponential(scale=0.014)
            a = np.exp(-lam * np.arange(n, dtype=float))
            families.append(a)

        # mixed power‑law × exponential
        for _ in range(15):
            p = rng.uniform(0.3, 0.9)
            lam = rng.exponential(scale=0.014)
            a = (np.arange(n, dtype=float) + 1.0) ** (-p) * np.exp(-lam * np.arange(n))
            families.append(a)

        # piecewise exponential
        for _ in range(10):
            cut = rng.integers(int(0.2 * n), int(0.8 * n))
            lam1 = rng.exponential(scale=0.018)
            lam2 = rng.exponential(scale=0.018)
            a = np.concatenate([np.exp(-lam1 * np.arange(cut)),
                                np.exp(-lam2 * np.arange(cut, n))])
            families.append(a)

        # Dirichlet (uniform on simplex)
        for _ in range(10):
            a = rng.dirichlet(alpha=np.ones(n)) * target_sum
            families.append(a)

        # make sure the up‑sampled / random init also participates
        families.append(a_init)

        # project all to simplex and store their scores
        for vec in families:
            vec = np.maximum(vec, 0.0)
            vec = _project_to_simplex(vec, target_sum)
            score = _max_convolution_fft(vec)
            population.append((score, vec))

        # keep the best few
        population.sort(key=itemgetter(0))
        population = population[:6]

        # --------------------------------------------------------------
        # Evolutionary loop (mutate + short Adam refinements)
        # --------------------------------------------------------------
        while time.time() + 2.0 < deadline:
            # select two parents (the best two)
            parent1 = population[0][1]
            parent2 = population[1][1] if len(population) > 1 else parent1

            # crossover (random convex combination)
            alpha = rng.uniform(0.3, 0.7)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = _project_to_simplex(child, target_sum)

            # mutation – add tiny Gaussian jitter
            if rng.random() < 0.5:
                child += rng.normal(scale=target_sum * 0.0015, size=n)
                child = np.maximum(child, 0.0)
                child = _project_to_simplex(child, target_sum)

            # a very brief Adam refinement (≈3 s)
            sub_deadline = min(deadline - 0.1, time.time() + 3.0)
            child_opt = _adam_optimize(child, target_sum, sub_deadline)

            child_score = _max_convolution_fft(child_opt)

            # insert into population
            population.append((child_score, child_opt))
            population.sort(key=itemgetter(0))
            population = population[:8]   # keep the elite

        # --------------------------------------------------------------
        # Take the current best of this size
        # --------------------------------------------------------------
        local_best_seq, local_best_score = min(population, key=itemgetter(0))
        if local_best_score < best_score - 1e-12:
            best_seq = local_best_seq.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Size {n} new best score {best_score:.6f}")

        # --------------------------------------------------------------
        # Deterministic hill‑climb on the exact convolution
        # --------------------------------------------------------------
        if time.time() < deadline - 1.5:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + (deadline - time.time()) * 0.30)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb size {n} improved to {best_score:.6f}")

        # --------------------------------------------------------------
        # Short Adam polishing (use remaining slice of the budget)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            extra_deadline = deadline - 0.5
            best_seq = _adam_optimize(best_seq, target_sum, extra_deadline)
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Post‑HC Adam size {n} score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # final Adam run
        final_deadline = deadline - 0.6
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        # final exact hill‑climb
        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.2)
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: truncate trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: simple uniform sequence (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (for debugging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error in evaluate_sequence: {exc}")

    return best_seq.tolist()
