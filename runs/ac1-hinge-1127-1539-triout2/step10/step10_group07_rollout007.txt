```
m = np.zeros_like(a)
v = np.zeros_like(a)
beta1=0.9; beta2=0.999; eps=1e-8
t=0
while time...
    t+=1
    # compute loss and gradient
    c = _conv_fft(a)
    # compute softmax
    c_max = c.max()
    exp_alpha = np.exp(alpha * (c - c_max))
    w = exp_alpha / exp_alpha.sum()
    # gradient
    g = 2 * _linear_convolve_fft(a, w[::-1])[L - np.arange(n)]

    m = beta1*m + (1-beta1)*g
    v = beta2*v + (1-beta2)*(g**2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

"""
Propose a low‑score step‑function.

Algorithm
---------

* **Smooth max‑convolution** –  The true objective is
  `max_k (a*a)[k]`.  We replace the hard max with a
  soft‑max (log‑sum‑exp) with a temperature `α`.  The gradient of
  the smooth loss can be computed analytically and written as a
  convolution, so a single FFT pair yields the gradient.

* **Adam optimisation on the simplex** –  We run an Adam‑style
  optimiser on the smooth loss while projecting the iterate onto the
  simplex `{a ≥ 0, Σa = √(2n)}` after each step.  The temperature
  `α` is increased slowly so the optimisation first sees a smooth
  landscape and later focuses on the true max.

* **Random restarts** –  After a run we jitter the best sequence,
  re‑optimise for a few seconds and keep the improvement.

* **Exact mass‑transfer refinement** –  A fast O(n) simulated‑annealing
  move transfers a tiny amount of mass from a “bad’’ index (large
  contribution to the current maximal convolution entry) to a
  “good’’ one.

* **Multiple problem sizes** –  The budget is split over a few
  different lengths `n`.  The best sequence among all sizes is
  returned.

The code respects the required time budget and uses only NumPy
(FFT) and the supplied `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective 2·n·max(a*a) / (Σa)².
    After scaling Σa = √(2n) the function equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimises max(a*a) using a smoothed max (soft‑max) with Adam.
    The iterate is kept on the simplex Σa = target_sum.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    n = a.shape[0]
    L = 2 * n - 2               # index shift for gradient extraction
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # smoothing schedule
    alpha = 5.0                 # start temperature
    alpha_inc = 1.02            # multiplicative increase every 200 iters

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        if t % 200 == 0:
            alpha *= alpha_inc

        # --- convolution and soft‑max weights ---
        c = _conv_fft(a)                         # length 2n‑1
        c_max = c.max()
        # soft‑max (numerically stable)
        exp_term = np.exp(alpha * (c - c_max))
        w = exp_term / exp_term.sum()            # shape (2n‑1,)

        # --- gradient of the smoothed loss ---
        # g_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)   # length 3n‑2
        g = conv_full[L - np.arange(n)]
        grad = 2.0 * g

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate that slowly decays
        lr = target_sum * 0.001 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- simplex projection ---
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out of plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = c_max                     # because Σa = target_sum
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:6d} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact local mass‑transfer refinement (same as previous solution)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that moves a tiny
    amount of mass from a “bad’’ index to a “good’’ one.
    The full convolution is updated in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # ------------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – choose from the lower‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # ------------------------------------------------------------------
        # Propose a small transfer
        # ------------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta            # amount moved from donor → receiver

        # ------------------------------------------------------------------
        # Update convolution exactly (O(n))
        # ------------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # ------------------------------------------------------------------
        # Acceptance (simulated annealing)
        # ------------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # ------------------------------------------------------------------
        # Periodic re‑normalisation & full recomputation
        # ------------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional jitter to shake out of plateaus
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Choose a few problem sizes; larger n gives more freedom but costs time
    # ------------------------------------------------------------------
    if budget_s >= 800:
        sizes = [12000, 16000, 20000]
    elif budget_s >= 500:
        sizes = [8000, 12000, 16000]
    else:
        sizes = [4000, 6000, 8000]

    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    for n in sizes:
        # allocate a slice of the remaining time (≈1/len(sizes) of what is left)
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        sub_deadline = now + remaining / (len(sizes) - sizes.index(n))

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Warm‑start initial sequence
        # --------------------------------------------------------------
        if "height_sequence_1" in globals():
            init = _resize_sequence(
                np.array(globals()["height_sequence_1"], dtype=np.float64), n
            )
        else:
            # power‑law with a random exponent in a sensible range
            exp = rng.uniform(0.4, 0.7)
            init = (np.arange(1, n + 1, dtype=np.float64) ** (-exp))

        init = _project_to_simplex(init, target_sum)

        # --------------------------------------------------------------
        # Adam optimisation (≈60 % of the slice)
        # --------------------------------------------------------------
        opt_deadline = time.time() + 0.6 * (sub_deadline - time.time())
        a_opt, val_opt = _adam_optimize(init, target_sum, opt_deadline)

        # --------------------------------------------------------------
        # Short random restarts (≈25 % of the slice)
        # --------------------------------------------------------------
        while time.time() < sub_deadline - 2.0:
            # 5‑second jittered Adam run
            run_time = min(5.0, sub_deadline - time.time())
            if run_time < 1.0:
                break
            jitter = rng.normal(scale=target_sum * 0.02, size=n)
            start_seq = a_opt + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            a_tmp, val_tmp = _adam_optimize(start_seq, target_sum,
                                            time.time() + run_time)
            if val_tmp < val_opt:
                val_opt = val_tmp
                a_opt = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved score={val_opt:.6f}")

        # --------------------------------------------------------------
        # Exact local refinement (remaining time for this size)
        # --------------------------------------------------------------
        a_ref, val_ref = _local_refine(a_opt, target_sum, sub_deadline)

        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] New global best (n={n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up, evaluation and warm‑start storage
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – a simple decreasing power‑law
        n = 5000
        target_sum = np.sqrt(2.0 * n)
        best_seq = _project_to_simplex(
            (np.arange(1, n + 1, dtype=np.float64) ** (-0.55)), target_sum
        )

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq   # for warm‑starts in later calls

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
