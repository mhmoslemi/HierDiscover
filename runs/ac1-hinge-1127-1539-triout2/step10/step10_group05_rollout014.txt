import cvxpy as cp

def lp_step(a, target, M, max_iter):
    n = a.size
    A = build_conv_matrix(a)
    g = cp.Variable(n)
    constraints = [A @ g <= M,
                   g >= 0,
                   g <= target]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=max_iter)
    if g.value is None:
        return None
    return np.array(g.value, dtype=float)

"""
Propose a non‑negative step function that minimizes
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm consists of three phases.

1. **Broad exploration** – a cheap parametric sweep (power‑law,
   exponential, edge‑heavy) plus a few random draws on the ℓ¹‑simplex
   for several lengths n.  The best vector (according to the exact
   max‑convolution) is kept as a seed.

2. **LP‑based refinement** – given a seed *f* we solve the linear
   program

        maximize   Σ g_i
        subject to conv(f,g) ≤ M   (component‑wise),
                0 ≤ g_i ≤ √(2·n),

   where M = max(conv(f,f)).  The optimum *g* has the largest possible
   ℓ¹‑mass while still respecting the current worst convolution bound.
   We rescale *g* to have the target ℓ¹‑norm √(2·n) and replace *f*
   by the best convex combination of *f* and the rescaled *g*.
   Repeating this step strictly decreases the maximal convolution value.

3. **Final polishing** – a deterministic hill‑climb (exact updates) and
   a short simulated‑annealing phase are applied to the best vector
   found by the LP loop.

All work respects a wall‑clock budget (default 1000 s).  The produced
sequence is trimmed of trailing near‑zeros and returned as a plain
list of floats.
"""

# Linear‑algebra helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A such that (A @ g)[k] = (f * g)[k].
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        j = np.arange(i_min, i_max + 1)
        A[k, j] = f[k - j]
    return A

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan a few one‑parameter families (power‑law, exponential,
    edge‑heavy, mixed) and return the best vector found (according to the
    exact max‑convolution).
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law exponents
    for p in np.linspace(0.1, 2.0, 40):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.0005, 0.02, 40):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy (large at the ends, decays towards centre)
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 20):
        for lam in np.linspace(0.001, 0.01, 10):
            a = (idx ** (-p)) * np.exp(-lam * (lin - 0))
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    return best_a, best_score

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # pick a destination j that is not i
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

    # 1)  Look at several lengths (coarse grid) and keep the best
    # --------------------------------------------------------------
    n_grid = [400, 600, 800, 1000, 1200, 1500, 2000]

    for n in n_grid:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # --------- a) cheap parametric sweep -----------------------
        a_sweep, _ = _parameter_sweep(n, rng)
        a_sweep = _project_to_simplex(a_sweep, target)
        seqs = [a_sweep]

        # --------- b) a handful of random starts --------------------
        for _ in range(3):
            a0 = _random_initial_sequence(n, rng)
            a0 = np.maximum(a0, 1e-12)
            a0 = _project_to_simplex(a0, target)
            seqs.append(a0)

        # evaluate each seed and start the LP refinement
        for a0 in seqs:
            if time.time() >= deadline:
                break

            a = a0.copy()
            a = np.maximum(a, 1e-12)               # avoid exact zeros for LP
            a = _project_to_simplex(a, target)

            cur_M = _max_convolution_fft(a)
            local_best_a = a.copy()
            local_best_M = cur_M

            # ------------------------------------------------------
            # 2)  LP‑based refinement (skip if cvxpy unavailable)
            # ------------------------------------------------------
            if _HAS_CVX and time.time() < deadline - 5.0:
                while time.time() < deadline - 2.0:
                    # build the convolution matrix of the current a
                    A = _build_conv_matrix(a)

                    # linear program: maximize sum(g) subject to A@g <= cur_M, 0 <= g <= target
                    g = cp.Variable(n)
                    constraints = [A @ g <= cur_M,
                                   g >= 0,
                                   g <= target]
                    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
                    try:
                        prob.solve(solver=cp.ECOS, warm_start=True,
                                   max_iters=2000, abstol=1e-9, reltol=1e-9)
                    except Exception:
                        break
                    if g.value is None:
                        break
                    g_val = np.asarray(g.value).astype(float)
                    sum_g = g_val.sum()
                    if sum_g <= 0:
                        break

                    # rescale to the target ℓ¹‑norm
                    g_rescaled = g_val * (target / sum_g)

                    # try a few convex combinations (0, 0.5, 1)
                    candidates = [a,
                                  0.5 * a + 0.5 * g_rescaled,
                                  g_rescaled]
                    improved = False
                    best_local_M = cur_M
                    best_local_a = a
                    for cand in candidates:
                        cand = _project_to_simplex(cand, target)
                        M_c = _max_convolution_fft(cand)
                        if M_c < best_local_M - 1e-12:
                            best_local_M = M_c
                            best_local_a = cand
                            improved = True
                    if not improved:
                        break      # LP cannot improve further
                    a = best_local_a
                    cur_M = best_local_M
                    if cur_M < local_best_M - 1e-12:
                        local_best_M = cur_M
                        local_best_a = a.copy()
                        print(f"[{time.time():.1f}] n={n} LP step → M={cur_M:.6f}")

            # ------------------------------------------------------
            # 3)  Deterministic hill‑climb as final polish
            # ------------------------------------------------------
            if time.time() < deadline - 1.0:
                b_full = _full_convolution(a)
                a_hc, M_hc = _hill_climb_deterministic(a, b_full, deadline - 0.5)
                if M_hc < local_best_M - 1e-12:
                    local_best_M = M_hc
                    local_best_a = a_hc.copy()
                    print(f"[{time.time():.1f}] n={n} hill‑climb → M={M_hc:.6f}")

            # keep global best
            if local_best_M < best_score - 1e-12:
                best_score = local_best_M
                best_seq = local_best_a.copy()
                print(f"[{time.time():.1f}] New global best (n={n}) → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # ------------------------------------------------------------------
    # 5)  Trim trailing near‑zeros
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 6)  Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print("Evaluation error:", exc)

    # expose for possible next runs
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    return best_seq.tolist()
