"""
Propose a step‑function minimizing the peak of its self‑convolution.

The algorithm combines several complementary heuristics:

1.  **Diverse initial pool** – monotone power‑law, symmetric
    power‑law (high at both ends), the previous best (if any), jittered
    copies and random permutations.  All candidates are projected onto the
    ℓ¹‑simplex Σa = √(2 n).

2.  **Soft‑max projected gradient descent** – fast reduction of the
    convolution peak using a smoothed max (soft‑max) and momentum.  Multiple
    restarts explore different basins.

3.  **Exact sub‑gradient descent** – optimisation of the true max‑convolution
    using its sub‑gradient (2·a shifted) with momentum.

4.  **Deterministic mass‑transfer** – moves small amounts of mass from the
    index contributing most to the current peak towards indices with little
    contribution (original version).

5.  **Product‑based mass‑transfer** – a refined version that selects the
    donor according to the *product* a_i·a_{kmax−i} (the actual contribution
    to the peak) and transfers mass to a low‑contributing receiver.
    Simulated‑annealing acceptance allows occasional uphill moves.

6.  **Permutation annealing** – stochastic swapping of positions to
    discover better orderings.

7.  **Final polishing** – a high‑β soft‑max GD pass to fine‑tune the result.

All heavy convolutions use FFT (O(n log n)).  The time budget (≈ 1000 s)
is split among the phases, and any remaining time is spent in the final
polish.  The best sequence is stored in the global variable
`height_sequence_1` for warm‑starts in future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Monotone power‑law a_i ∝ (i+1)^{‑p}, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _sym_power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Symmetric high‑at‑ends shape a_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) + (n - i + 1) ** (-p)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 1e5,
                                   step_factor: float = 0.7,
                                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short GD runs with jitter; keep the best."""
    best_a = init.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    cur_val = best_val

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2 * a[kmax - i] where defined
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves targeting the current maximal convolution entry."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={int(time.time())} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] ProdMT new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the peak of its
    self‑convolution under the ℓ¹‑normalisation Σa = √(2 n).
    The routine respects the given time budget (≈ budget_s seconds).
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size selection (larger n gives finer granularity)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build initial candidate pool
    # --------------------------------------------------------------
    init_seqs = []

    # monotone power‑law families
    for p in (0.45, 0.5, 0.55, 0.6, 0.65):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # symmetric high‑ends power‑law families
    for p in (0.5, 0.6, 0.7):
        init_seqs.append(_sym_power_law_sequence(n, p, target_sum))

    # uniform baseline
    init_seqs.append(np.full(n, target_sum / n, dtype=np.float64))

    # previous best (if any)
    if "height_sequence_1" in globals():
        init_seqs.append(_resize_sequence(prev_seq, n))

    rng = np.random.default_rng()

    # jittered variants of everything above
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # a couple of random permutations of the power‑law candidates
    for p in (0.5, 0.55):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # --------------------------------------------------------------
    # Quick evaluation of the pool – keep the best as a starting point
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for cand in init_seqs:
        sc = _max_convolution_fft(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Soft‑max projected gradient descent (with restarts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.25
    a_gd, val_gd = _gradient_descent_with_restarts(best_seq, target_sum, gd_deadline)
    if val_gd < best_score:
        best_score = val_gd
        best_seq = a_gd.copy()
        print(f"[{time.time():.2f}] After GD best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase X – Exact max sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = time.time() + (deadline - time.time()) * 0.15
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum, sg_deadline,
                                            step_factor=0.6, gamma=0.95)
        if val_sg < best_score:
            best_score = val_sg
            best_seq = a_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (original)
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.12
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_deadline,
                                                    target_sum,
                                                    start_temp=0.5,
                                                    end_temp=1e-4,
                                                    low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After DMT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B2 – Product‑based mass‑transfer (new)
    # --------------------------------------------------------------
    if time.time() < deadline:
        pmt_deadline = time.time() + (deadline - time.time()) * 0.12
        a_pmt, val_pmt = _product_mass_transfer(best_seq, pmt_deadline,
                                                target_sum,
                                                start_temp=0.5,
                                                end_temp=1e-4,
                                                low_frac=0.07)
        if val_pmt < best_score:
            best_score = val_pmt
            best_seq = a_pmt.copy()
            print(f"[{time.time():.2f}] After product‑MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Re‑ordering heuristics
    # --------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        sc = _max_convolution_fft(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(best_seq,
                                                  target_sum,
                                                  perm_deadline,
                                                  init_score=best_score)
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing (high‑β soft‑max GD)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_ext(best_seq,
                                                        deadline,
                                                        target_sum,
                                                        max_beta=3e5,
                                                        step_factor=0.6,
                                                        gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up (trim trailing zeros) and report
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
