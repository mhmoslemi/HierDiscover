```
def _subgradient_multi(a0, target_sum, deadline, step_factor=0.5, gamma=0.9, top_k=3):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        # top_k indices
        idx = np.argpartition(conv, -top_k)[-top_k:]
        # compute average gradient
        grad = np.zeros_like(a)
        for k in idx:
            # compute subgradient for this k
            start = max(0, k-(len(a)-1))
            end = min(k, len(a)-1)
            # i from start..end: a_i * a_{k-i}
            # gradient for each i is 2*a_{k-i}
            # allocate to grad[i]
            # we can use vectorized slice
            i_range = np.arange(start, end+1)
            grad[i_range] += 2 * a[k - i_range]
        grad /= top_k
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] MultiSG it={it} best={best_val:.6f}")
        # optional jitter
        if it % 500 == 0 and it>0:
            jitter = rng.normal(scale=target_sum*1e-3, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        it+=1
    return best_a, best_val
```

"""
Optimisation pipeline for the step‑function benchmark.

The goal is to minimise   2·n·max(a∗a)/(Σa)²,
where Σa is forced to √(2 n).  With that normalisation the
objective equals the peak of the linear convolution of the
non‑negative sequence with itself.

The programme builds a large pool of diverse initial shapes
(power‑law, exponential, polynomial, random Dirichlet),
refines each using a multi‑peak sub‑gradient method, then
applies aggressive deterministic mass‑transfer, permutation
annealing and a final soft‑max (high‑β) projected gradient
polish.  Time is budgeted automatically (≈1000 s) and the
best sequence found is stored in the global variable
`height_sequence_1` for warm‑starts in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Σa = s, a_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (all return an ℓ¹‑simplex‑projected vector)
# ----------------------------------------------------------------------


def _power_law_seq(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (decreasing power law)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_seq(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (geometric decay)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _polynomial_seq(n: int, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (1 - i/(n‑1))^q  (smoothly decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - i / (n - 1)) ** q
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _random_dirichlet_seq(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex via Gamma(1,1) (exponential) sampling."""
    a = rng.exponential(scale=1.0, size=n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Multi‑peak sub‑gradient descent (true max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_multi(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       step_factor: float = 0.6,
                       gamma: float = 0.95,
                       top_k: int = 3,
                       jitter_interval: int = 500,
                       jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the true max‑convolution.
    The gradient is averaged over the top_k convolution entries.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)        # with Σa = √(2n) this equals the objective
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        # indices of the k largest convolution entries
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # i ranges [lo, hi]
            i_vals = np.arange(lo, hi + 1, dtype=np.int64)
            grad[i_vals] += 2.0 * a[k - i_vals]

        grad /= max(1, len(idxs))

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad‑multi it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic (but stochastic‑acceptance) mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 1.0,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer aimed at the current maximal convolution entry.
    Uses a temperature schedule to occasionally accept uphill moves.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # choose a low‑weight receiver
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Mass‑transfer it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            # occasional jitter to escape plateaus
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation (swap‑based) simulated annealing
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Swap two positions at a time, accepting with a temperature schedule.
    The objective is the peak of the convolution.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        ratio = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm‑anneal new best={best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full reshuffle when temperature is very low
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Soft‑max gradient descent (high‑β polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Soft‑max approximation of max(conv); returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (rev a) ⊛ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      max_beta: float = 5e5,
                                      step_factor: float = 0.5,
                                      gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Projected momentum GD on the soft‑max approximation of the max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Soft‑max GD it={it:5d} β={beta:.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step‑function with a low benchmark score.
    The routine builds diverse seeds, refines them with multi‑peak
    sub‑gradient descent, performs mass‑transfer, permutation annealing
    and a final high‑β soft‑max GD.  The best sequence found is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives more flexibility)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        if budget_s >= 800:
            n = 20000
        elif budget_s >= 500:
            n = 15000
        else:
            n = 10000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build an initial pool of candidate shapes
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    seed_gens = []

    # power‑law family
    for p in (0.30, 0.40, 0.50, 0.60, 0.70):
        seed_gens.append(lambda p=p: _power_law_seq(n, p, target_sum))

    # exponential family
    for lam in (5e-5, 1e-4, 3e-4, 6e-4):
        seed_gens.append(lambda lam=lam: _exponential_seq(n, lam, target_sum))

    # polynomial family (smoothly decreasing)
    for q in (0.5, 1.0, 2.0):
        seed_gens.append(lambda q=q: _polynomial_seq(n, q, target_sum))

    # random Dirichlet samples
    for _ in range(6):
        seed_gens.append(lambda rng=rng: _random_dirichlet_seq(n, target_sum, rng))

    # warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, prev.shape[0]), prev)
        seed_gens.append(lambda prev=prev: prev.copy())

    best_seq = None
    best_val = np.inf

    # evaluate all seeds (quick pass)
    for gen in seed_gens:
        if time.time() > deadline - 5.0:
            break
        seq = gen()

        best_seq = _power_law_seq(n, 0.55, target_sum)
        best_val = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Stage 1 – Multi‑peak sub‑gradient descent (most time‑intensive)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sg_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_sg, val_sg = _subgradient_multi(best_seq, target_sum,
                                            sg_deadline,
                                            step_factor=0.6,
                                            gamma=0.95,
                                            top_k=3)
        if val_sg < best_val:
            best_val = val_sg
            best_seq = seq_sg.copy()
            print(f"[{time.time():.2f}] After multi‑subgrad best={best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_mt, val_mt = _deterministic_mass_transfer(best_seq,
                                                      mt_deadline,
                                                      target_sum,
                                                      start_temp=1.0,
                                                      end_temp=1e-5,
                                                      low_frac=0.02)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = seq_mt.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best={best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Permutation annealing (swap‑based)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20

            print(f"[{time.time():.2f}] After permutation anneal best={best_val:.6f}")

                                                             deadline,

                                                             max_beta=5e5,
                                                             step_factor=0.5,
                                                             gamma=0.999)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = seq_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Log the final evaluation using the provided routine
    try:
        val = evaluate_sequence(final_seq)      # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
