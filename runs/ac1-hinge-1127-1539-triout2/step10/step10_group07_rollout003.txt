def _smooth_gradient_descent(a0, target_sum, deadline, alpha_start=0.5, alpha_max=200.0, alpha_factor=1.005, step0_factor=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_max = _conv_fft(a).max()
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    alpha = alpha_start
    while time.time() < deadline:
        # compute
        b = _conv_fft(a)
        max_b = b.max()
        # compute softmax weights
        exps = np.exp(alpha * (b - max_b))
        w = exps / exps.sum()
        # gradient
        grad = 2.0 * _cross_corr(w, a)
        # step size
        step = (target_sum * step0_factor) / np.sqrt(it + 1.0)
        # update
        a = np.maximum(a - step * grad, 0.0)
        sum_a = a.sum()
        if sum_a > 0:
            a *= target_sum / sum_a
        else:
            # reinitialize to uniform or random
            a = np.full(n, target_sum / n)
        # update alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)
        # bookkeeping
        if max_b < best_max:
            best_max = max_b
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth GD it={it} max_b={best_max:.6f} alpha={alpha:.2f}")
        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()
        it += 1
    return best_a, best_max

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. Choose a problem size `n` based on the time budget.
2. Warm‑start from a previously found construction (if any) or a simple
   decreasing power‑law, then project onto the simplex Σa = √(2 n).
3. **Phase 1 – Smoothed gradient descent**  
   Minimise the smoothed version of   max_k (a∗a)_k   using a soft‑max
   approximation with a gradually increasing temperature.  After each
   update the iterate is projected back onto the simplex (clip‑to‑zero
   + renormalise).  The schedule is:
        α ← α·α_factor   (α controls the smoothness, grows to α_max)
        step ← step0 / √iteration   (diminishing steps).
   The best iterate (lowest maximal convolution entry) is kept.
4. **Phase 2 – Short random restarts**  
   Re‑initialise the current best sequence with a modest Gaussian jitter
   and run a brief smoothed‑gradient run.  Keep any improvement.
5. **Phase 3 – Exact local refinement** (same routine as in the
   reference solution).  This moves a tiny amount of mass from a
   “bad’’ index to a “good’’ one, accepting moves with a simulated‑annealing
   schedule.
6. Trim trailing near‑zero entries, evaluate the final sequence with the
   provided `evaluate_sequence`, store it for warm‑starts and return it.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return np.real(conv)


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:m]
    return np.real(conv)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation (w ⋆ a) for w length 2n‑1, a length n.
    Returns a vector of length n: g_i = Σ_j w_{i+j} a_j.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_full = _linear_convolve_fft(w, a_rev)   # length 3n‑2
    start = n - 1
    return conv_full[start:start + n]


# ----------------------------------------------------------------------
# Smoothed gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             alpha_start: float = 0.5,
                             alpha_max: float = 200.0,
                             alpha_factor: float = 1.010,
                             step0_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise the true max‑convolution using a soft‑max approximation.
    Returns the best sequence encountered and the corresponding maximal
    convolution entry (i.e. the objective value after normalisation).
    """
    a = a0.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _conv_fft(a).max()

    rng = np.random.default_rng()
    it = 0
    alpha = alpha_start

    while time.time() < deadline:
        # convolution a*a
        b = _conv_fft(a)
        max_b = b.max()

        # soft‑max weights (stable)
        exps = np.exp(alpha * (b - max_b))
        w = exps / exps.sum()

        # gradient = 2 * (w ⋆ a)
        grad = 2.0 * _cross_corr(w, a)

        # diminishing step size
        step = (target_sum * step0_factor) / np.sqrt(it + 1.0)

        # gradient step + projection (clip‑to‑zero + renorm)
        a = np.maximum(a - step * grad, 0.0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:
            a = np.full(n, target_sum / n)

        # update smoothing parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # evaluate new max
        new_max = _conv_fft(a).max()
        if new_max < best_max:
            best_max = new_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] SmoothGD it={it:6d} max={best_max:.6f} α={alpha:.2f}")

        # occasional jitter to escape plateaus
        if it % 300 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Exact local mass‑transfer refinement (as in the previous solution)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the true max‑convolution.
    Moves a small amount of mass from a “bad’’ index to a “good’’ one,
    updating the full convolution in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = 2.0 * n * b.max() / (target_sum * target_sum)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # ----- identify donor (large contribution) and receiver (small) -----
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – choose from the lower‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # ----- propose a small transfer -----
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # ----- update convolution exactly (O(n)) -----
        b_new = b.copy()
        # linear part: 2*a*Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # ----- acceptance (simulated annealing) -----
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # ----- periodic full recomputation -----
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = 2.0 * n * b.max() / (target_sum * target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional tiny jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = 2.0 * n * b.max() / (target_sum * target_sum)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Problem size selection
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)            # Σa = √(2 n)

    # ------------------------------------------------------------------
    # Warm‑start (reuse previous construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Phase 1 – Smoothed gradient descent (≈55 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + 0.55 * (deadline - time.time())
    a_smooth, max_b_smooth = _smooth_gradient_descent(init, target_sum, phase1_deadline)
    best_seq = a_smooth
    best_score = max_b_smooth                # because Σa = √(2 n) ⇒ objective = max_b
    print(f"[{time.time():.2f}] Phase 1 completed, best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Short random restarts (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rest_deadline = time.time() + 0.20 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2.0:
            remaining = rest_deadline - time.time()
            run_time = min(7.0, remaining)    # each restart runs a few seconds
            if run_time < 2.0:
                break
            jitter = rng.normal(scale=target_sum * 0.02, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            a_tmp, max_b_tmp = _smooth_gradient_descent(start_seq, target_sum,
                                                        time.time() + run_time)
            if max_b_tmp < best_score:
                best_score = max_b_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up, final evaluation and storage
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # provided by the grader
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
