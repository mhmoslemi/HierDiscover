assistantfinal**Solution Explanation**

The goal is to supply a list `a = (a₀,…,a_{n‑1}) , a_i ≥ 0` that minimises  

```
score(a) = 2·n·max(a*a) / (∑a)² .
```  

Because the denominator is scale‑invariant we may *fix* the ℓ¹‑norm

```
∑ a_i = √(2·n)                (1)
```

and then the score is simply `max(a*a)`.  
The problem is a non‑convex minimisation of the maximum entry of the
autocorrelation of a non‑negative vector on the simplex (1).

The previous code used a plain sub‑gradient method.  
The new algorithm consists of three stages

1. **Fast coarse optimisation** – mirror‑descent on a *soft* version of the
   max‑function (log‑sum‑exp).  The soft‑max is differentiable,
   the gradient can be obtained with a single FFT and the update is performed
   on the simplex using an exponentiated‑gradient (mirror‑descent) step.
2. **Upscaling** – the coarse solution (few thousand coefficients) is linearly
   interpolated to the final size (`n = 12000`).  Afterwards the sum is brought
   back to (1).
3. **Exact local refinement** – a fast simulated‑annealing / mass‑transfer
   routine (the one already used in the reference solution) removes the last
   remaining peaks.

All stages respect the time limit (`budget_s` seconds).  
The implementation is completely self‑contained, uses only `numpy`
and the supplied `evaluate_sequence` routine.

```python
# --------------------------------------------------------------
# Step‑function optimiser
# --------------------------------------------------------------
# The function `propose_candidate(budget_s)` returns a list of non‑negative
# heights that (hopefully) beats the previous best score.
# --------------------------------------------------------------

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. FFT based convolutions (O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# 2. Simple utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# 3. Mirror‑descent on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _mirror_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the objective
          f(a) = log‑sum‑exp( β·(a*a) )
    The gradient of f is obtained with one FFT and the iterate is
    projected onto the simplex Σa = target_sum after every step.

    Returns the best vector seen and its (non‑smoothed) max‑convolution.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = np.inf

    rng = np.random.default_rng()
    it = 0
    eta0 = 1.0          # initial learning rate (will decay as 1/√t)
    beta = 0.5          # soft‑max sharpness – will be increased gradually
    n = a.shape[0]

    while time.time() < deadline:
        # ---- convolution and soft‑max weights ---------------------------------
        conv = _conv_fft(a)                     # length 2n‑1
        max_conv = conv.max()
        # soft‑max (shifted by max to keep numbers ≤ 0)
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:                                   # numerical safeguard
            w.fill(1.0 / w.size)

        # ---- gradient of the smoothed objective --------------------------------
        # ∂/∂a  ( Σ_i w_i (a*a)_i ) = 2 * a  * w   (cross‑correlation)
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)   # length 3n‑2
        L = len(w) - 1                                 # = 2n‑2
        g = 2.0 * conv_full[L - np.arange(n)]

        # ---- mirror (exponentiated) update ------------------------------------
        eta = eta0 / np.sqrt(it + 1.0)
        # exponent may overflow – clip safely
        exp_arg = -eta * g
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # ---- projection back onto the simplex ---------------------------------
        a = _project_to_simplex(a, target_sum)

        # ---- bookkeeping -------------------------------------------------------
        if max_conv < best_val:
            best_val = max_conv
            best_a = a.copy()
            print(f"[{time.time():.2f}] Mirror it={it:6d} best={best_val:.6f}")

        # ---- schedule: increase β (sharper soft‑max) -------------------------
        if it % 2000 == 0 and beta < 20.0:
            beta *= 1.3

        # ---- occasional jitter to escape plateaus -----------------------------
        if it % 3000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 4. Exact local refinement (same idea as the reference solution)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / mass‑transfer hill‑climbing.
    Moves a tiny amount of mass from a “bad’’ index (large contribution
    to the current maximal convolution entry) to a “good’’ one.
    The full convolution is updated in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact a*a
    best_a = a.copy()
    best_val = float(2.0 * n * b.max() / (target_sum * target_sum))

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature

    while time.time() < deadline:
        # ---- locate the current maximal convolution entry --------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # ---- choose a donor (index that participates heavily in the maximum) --
        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0.0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # ---- choose a receiver (index that contributes little) ---------------
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # ---- propose a small transfer ---------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # ---- update the convolution exactly (O(n)) ---------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # ---- acceptance (simulated annealing) -------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved={best_val:.6f}")

        # ---- periodic full recomputation (prevents drift) -------------------
        if time.time() - start > total_time * 0.5:
            # re‑normalise to keep the exact ℓ¹‑norm
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = float(2.0 * n * b.max() / (target_sum * target_sum))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved={best_val:.6f}")

    return best_a, best_val
```
