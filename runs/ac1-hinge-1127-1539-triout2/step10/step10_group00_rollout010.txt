def _adam_opt(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = _score(a)
    beta_temp = 1.0
    max_beta = 1e5  # maybe 2**16
    while time.time() < deadline:
        t += 1
        grad, conv = _gradient_softmax(a, beta_temp)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step_size = 0.05 / (np.sqrt(v_hat) + eps)
        a = a - step_size * m_hat
        a = np.maximum(a, 0.0)  # maintain nonneg
        a = a * (target_sum / a.sum())
        # occasionally increase beta to tighten approximation
        if t % 200 == 0 and beta_temp < max_beta:
            beta_temp = min(beta_temp * 2, max_beta)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iteration {t} beta={beta_temp} best={best_val:.6f}")
    return best_a, best_val

def _mirror_subgradient(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    eta0 = 0.5  # initial step
    t = 0
    rng = np.random.default_rng()
    n = a.size

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        # subgradient g_i = 2*a_{k_max - i} if index valid else 0
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = k_max - idx
        g[idx] = 2.0 * a[partner]

        eta = eta0 / np.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * g)
        # ensure sum and nonneg
        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] MirrorIter {t} best={best_val:.6f}")
        # occasional jitter
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
    return best_a, best_val

"""
Hybrid optimizer for the step‑function contest.

The objective, under the ℓ¹‑normalisation Σa = √(2·n), reduces to
    score = max_k (a * a)_k .
We therefore minimise the maximal entry of the self‑convolution.

The search consists of three phases:

1. **Parametric warm‑start** – a fine 2‑parameter grid on
   a_i = (i+1+δ)^(-p) (δ ≥ 0) on a reduced size, plus a few random
   refinements.  The best shape is interpolated to the full problem size.

2. **Adam optimisation** of a soft‑max surrogate of the max‑convolution.
   The surrogate gradient is obtained analytically via FFT, β (the
   inverse temperature) is doubled periodically to sharpen the approximation
   and the iterates are projected onto the ℓ¹‑simplex.

3. **Projected sub‑gradient descent** on the true max entry, followed by a
   cheap simulated‑annealing local search that transfers a small amount of
   mass from a high‑contributing index to a low‑contributing one.

All stages respect the global time budget (default 1000 s) and keep the
sequence non‑negative with the required ℓ¹‑norm.  The best sequence found
is trimmed of trailing zeros and stored in the global variable
`height_sequence_1` for warm‑starts in subsequent runs.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _fft_len(n: int) -> int:
    """Length of the next power‑of‑2 for a linear convolution of size 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real‑valued)."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(a*a) / (sum(a)²).  Assumes sum>0."""
    n = a.size
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Parametric warm‑start (power law with offset δ)
# ----------------------------------------------------------------------
def _grid_param_search_small(small_n: int,
                             target_sum: float,
                             deadline: float) -> np.ndarray | None:
    """Fine grid over p∈[0.35,0.80] and δ∈[0,30] (δ shift of the index)."""
    best_seq = None
    best_score = np.inf
    i = np.arange(1, small_n + 1, dtype=np.float64)  # 1‑based for the formula
    p_vals = np.linspace(0.35, 0.80, 45)          # 0.01 step ≈ 45 values
    delta_vals = np.linspace(0.0, 30.0, 31)       # step 1

    for p in p_vals:
        for delta in delta_vals:
            if time.time() > deadline:
                break
            raw = (i + delta) ** (-p)
            seq = raw * (target_sum / raw.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid p={p:.3f} δ={delta:.1f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    # random refinements while time permits
    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.35, 0.80)
        delta = rng.exponential(1.0) * 5.0   # bias to moderate δ
        raw = (i + delta) ** (-p)
        seq = raw * (target_sum / raw.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Rand p={p:.3f} δ={delta:.2f} score={best_score:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Soft‑max surrogate and Adam optimiser
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()                     # soft‑max weights

    # grad = 2 * (reverse(a) convolve p)  (central n entries)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_opt(a0: np.ndarray,
              target_sum: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max surrogate (β ramps up)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        grad, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.05 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            # small jitter to escape plateaus
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max entry
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Standard sub‑gradient descent with a 1/√t stepsize and simplex projection."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    t = 0
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of max‑entry w.r.t. a
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        step = (target_sum * 0.5) / (norm * np.sqrt(t))
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={t:5d} best={best_val:.6f}")

        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Cheap SA that moves a modest mass from a high‑contributing index to a low‑one."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)

    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.02 * best_val  # initial temperature proportional to the score

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        wsum = weight.sum()
        if wsum > 0:
            donor = rng.choice(idx, p=weight / wsum)
        else:
            donor = rng.integers(n)

        low_thr = np.percentile(weight, 10)
        low_candidates = idx[weight <= low_thr]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-14)

        if new_val < best_val or rng.random() < np.exp(-(new_val - best_val) / T):
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best = a_new.copy()
                print(f"[{time.time():.2f}] SA improved best={best_val:.6f}")

        # occasional random jitter
        if rng.random() < 0.003:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low score.
    The routine runs for at most `budget_s` seconds and returns the best
    sequence discovered as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size (larger n gives a tighter bound but costs a bit more time)
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start: either from previous run or a simple power‑law shape
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        # crude power‑law without shift
        i = np.arange(1, n + 1, dtype=np.float64)
        init = i ** (-0.55)
        init = init * (target_sum / init.sum())

    # ------------------------------------------------------------------
    # Stage 0 – fine parametric search on a reduced scale (p & δ)
    # ------------------------------------------------------------------
    small_n = max(2000, n // 5)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.10)
    best_small = _grid_param_search_small(small_n, small_target, param_deadline)
    if best_small is not None:
        init = _resize_sequence(best_small, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑started from refined parametric search")

    # ------------------------------------------------------------------
    # Stage 1 – Adam on the soft‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.35
        a_adam, val_adam = _adam_opt(init, target_sum, adam_deadline)
        best_seq = a_adam
        best_score = val_adam
        print(f"[{time.time():.2f}] After Adam best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent on the true max entry
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.35
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
            print(f"[{time.time():.2f}] After sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_deadline)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
            print(f"[{time.time():.2f}] After SA best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Verify with the supplied evaluator (optional, for logging)
    try:
        eval_val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {eval_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts next call
    globals()["height_sequence_1"] = final_seq
    return final_seq
