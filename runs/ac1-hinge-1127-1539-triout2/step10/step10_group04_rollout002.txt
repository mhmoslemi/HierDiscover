"""
Propose a non‑negative step function that (approximately) minimises
``evaluate_sequence``.
The algorithm combines:

1.  A fast scan of several parametric families (exponential,
    power‑law, mixed, beta‑shaped, Dirichlet).
2.  Projection of each candidate onto the ℓ¹‑simplex Σa = √(2 n).
3.  Refined optimisation with a projected Adam on a smooth
    log‑sum‑exp surrogate of the max‑convolution.
4.  A pure sub‑gradient descent on the exact max‑convolution
    (gradient = 2·a⋆1_{active indices}) with a diminishing step size.
5.  Deterministic hill‑climbing (as in the reference) for a final polish.
6.  Progressive up‑sampling: the best vector for a smaller length is
    interpolated to a larger length before optimisation, allowing the
    shape to be refined at higher resolutions.

The routine respects the global time budget (default 1000 s) and
returns the best vector found.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) surrogate and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    The algorithm runs until the wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.10 * target_sum

    # smoothed‑max schedule
    beta = 2.0
    beta_max = 1e8
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on   max_k (a*a)[k] .
    Uses a diminishing step size  step = α / sqrt(iter).
    """
    a = a0.copy()
    n = a.size
    rng = np.random.default_rng()
    it = 0
    alpha0 = target_sum * 0.5   # initial step magnitude

    while time.time() < deadline:
        it += 1
        b = _full_convolution(a)
        max_val = b.max()
        max_idx = np.where(b == max_val)[0]

        # compute sub‑gradient g_i = 2 * sum_{k∈max_idx} a_{k-i}
        g = np.zeros_like(a)
        for k in max_idx:
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # diminishing step size
        step = alpha0 / math.sqrt(it)
        a_new = a - step * g
        a = _project_to_simplex(a_new, target_sum)

        # tiny random perturbation to escape plateaus
        if it % 150 == 0:
            jitter = rng.normal(scale=target_sum * 1e-5, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution
    entry. Moves mass from the source that contributes most to the
    maximal entry towards a destination that contributes least.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # destination with minimal counterpart
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]
        cand_range = np.arange(i_low, i_high + 1, dtype=int)
        order = np.argsort(counterpart[cand_range])
        j = None
        for cand in cand_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        b_new = b.copy()
        # update convolution in O(n)
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move via binary search
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)

    # randomly pick a family
    family = rng.integers(0, 7)
    if family == 0:                     # power‑law
        p = rng.uniform(0.1, 1.5)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # beta‑shaped (U‑shape)
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        a = ( (idx + 1) ** (alpha - 1) ) * ( (n - idx) ** (beta - 1) )
        a = np.maximum(a, 0.0)

    elif family == 5:                   # piecewise exponential
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

            if i == j:
                continue
            delta = cand[i] * 0.07
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target_sum)
            score = _max_convolution_fft(cand)
            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score
                print(f"[{time.time():.1f}] Random tweak improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 7) Optional extra Adam run (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0 and extra_budget > 2.0:
            extra_deadline = deadline - 0.5
            a_ref = _adam_optimize(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC Adam improvement (n={n}) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 1.0
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)

        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up – drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    best_seq = best_seq[: nz[-1] + 1] if nz.size else best_seq[:1]

    # ------------------------------------------------------------------
    # Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
