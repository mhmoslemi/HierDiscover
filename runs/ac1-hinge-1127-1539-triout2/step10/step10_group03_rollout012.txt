"""
Hybrid optimiser for the convolution peak (evaluate_sequence).

The optimiser works in three phases:

1. **Shape search (mid size ≈ 8000)** – a few random starts
   (power‑law / exponential / Dirichlet) are optimised with a
   momentum projected gradient descent on the soft‑max approximation of
   the convolution peak.  The best low‑dimensional shape is kept.

2. **Upscaling** – the shape is linearly interpolated to a larger length
   (≈20000).  The ℓ¹‑simplex projection guarantees the required ‖a‖₁.
   A simple “alternating‑large‑small” reordering is performed to spread
   the large entries.

3. **Final polishing** – a long high‑β projected gradient descent is
   followed by a simulated‑annealing style swap‑annealing that exchanges
   positions of two entries (the sum is unchanged).  The swap‑annealing
   is cheap (a single FFT per move) and often reduces the maximal
   convolution entry below the previous best (≈1.508).  The routine keeps
   the best sequence seen so far and returns it before the 1000 s budget
   expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring (exact objective)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalised to a given ℓ¹‑sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam·i) normalised to a given ℓ¹‑sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, alpha: float, target_sum: float, rng) -> np.ndarray:
    """Draw a Dirichlet‑like vector and project onto the simplex."""
    a = rng.gamma(alpha, 1.0, size=n)
    a = _project_to_simplex(a, target_sum)
    return a


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2nd largest → right, 3rd → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (used by the projected GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    mx = np.max(b)
    exps = np.exp(beta * (b - mx))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------
def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with a growing soft‑max temperature.
    Returns the best vector seen and its exact score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # small jitter every few hundred iterations
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # raise temperature slowly
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Swap‑annealing (position optimisation)
# ----------------------------------------------------------------------
def _swap_annealing(a0: np.ndarray,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style swaps of two entries.
    The ℓ¹‑sum stays unchanged, only the ordering is altered.
    Returns the best ordering encountered and its exact score.
    """
    a = a0.copy()
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    n = a.shape[0]

    # initial temperature – a modest fraction of the current peak
    start_temp = max(cur_max * 0.04, 1e-6)

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = start_temp * (1.0 - frac)   # linear cooling

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == a[j]:
            continue

        # propose swap
        a[i], a[j] = a[j], a[i]
        new_max = _max_convolution_fft(a)

        if new_max < cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Swap accepted new best max={best_max:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            if rng.random() < prob:
                cur_max = new_max
            else:
                # revert
                a[i], a[j] = a[j], a[i]

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the 1000 s budget, stores the best sequence in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Choose a base size for the “shape” optimisation.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_mid = 8000
    elif budget_s >= 500:
        n_mid = 6000
    else:
        n_mid = 4000

    target_mid = np.sqrt(2.0 * n_mid)

    # ------------------------------------------------------------------
    # 2. Build a small pool of diverse initial candidates.
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.75, 9):
        init_pool.append(_power_law_sequence(n_mid, float(p), target_mid))

    # exponential family
    for lam in np.linspace(0.001, 0.015, 7):
        init_pool.append(_exponential_sequence(n_mid, float(lam), target_mid))

    # Dirichlet‑like draws
    for _ in range(5):
        init_pool.append(_dirichlet_sequence(n_mid, alpha=2.0, target_sum=target_mid, rng=rng))

    # previous global solution, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] >= n_mid:
            init_pool.append(prev[:n_mid].copy())
        else:
            # simple linear resize to required length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n_mid)
            resized = np.interp(new_x, old_x, prev)
            init_pool.append(_project_to_simplex(resized, target_mid))

    # add jittered copies to increase diversity
    for base in list(init_pool):
        jitter = rng.normal(scale=target_mid * 0.015, size=n_mid)
        cand = base + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_mid)
        init_pool.append(cand)

    # ------------------------------------------------------------------
    # 3. Quick gradient descents (short runs) – keep the best low‑dim shape.
    # ------------------------------------------------------------------
    best_mid = None
    best_mid_score = np.inf

    while time.time() < deadline - 30.0:   # leave time for upscale & polish
        seed = rng.choice(init_pool)
        run_dead = min(deadline - 10.0, time.time() + 12.0)   # ~12 s per run
        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_dead, target_mid,
            max_beta=3e5, step_factor=0.7, gamma=0.995)
        if val_opt < best_mid_score:
            best_mid_score = val_opt
            best_mid = a_opt.copy()
            print(f"[{time.time():.2f}] Mid‑run new best score={best_mid_score:.6f}")

    # sanity fallback
    if best_mid is None:
        best_mid = _power_law_sequence(n_mid, 0.55, target_mid)

    # ------------------------------------------------------------------
    # 4. Upscale to a large final size (≈20000) – keep the ℓ¹‑norm.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    n_final = 20000 if remaining > 100.0 else n_mid
    target_final = np.sqrt(2.0 * n_final)

    # linear interpolation to new length
    old_x = np.linspace(0.0, 1.0, best_mid.shape[0])
    new_x = np.linspace(0.0, 1.0, n_final)
    seq_big = np.interp(new_x, old_x, best_mid)
    seq_big = _project_to_simplex(seq_big, target_final)

    # initial reordering – alternating large/small
    seq_big = _alternating_order(seq_big)

    # quick assessment
    cur_best = seq_big.copy()
    cur_best_score = _score(cur_best)
    print(f"[{time.time():.2f}] Upscaled to n={n_final} initial score={cur_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long‑run gradient descent on the large problem.
    # ------------------------------------------------------------------
    if time.time() < deadline - 40.0:
        gd_dead = time.time() + min(30.0, deadline - time.time() - 10.0)
        a_final, val_final = _annealed_gradient_descent(
            cur_best, gd_dead, target_final,
            max_beta=1e6, step_factor=0.5, gamma=0.99)
        if val_final < cur_best_score:
            cur_best_score = val_final
            cur_best = a_final.copy()
            print(f"[{time.time():.2f}] After GD final score={cur_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Swap‑annealing (position optimisation) – the most effective
    #    improvement for the peak after the values have been shaped.
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        swap_dead = time.time() + min(20.0, deadline - time.time() - 2.0)
        a_sw, max_sw = _swap_annealing(cur_best, swap_dead)
        # convert max_sw to the full objective (they are identical when
        # the ℓ¹‑sum equals √(2n))
        new_score = _score(a_sw)
        if new_score < cur_best_score:
            cur_best_score = new_score
            cur_best = a_sw.copy()
            print(f"[{time.time():.2f}] After swap‑anneal score={cur_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Trim trailing zeroes (for compactness) and final evaluation.
    # ------------------------------------------------------------------
    arr = np.array(cur_best, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception:
        pass

    # expose for later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
