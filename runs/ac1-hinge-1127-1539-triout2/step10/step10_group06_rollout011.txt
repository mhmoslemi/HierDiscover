"""
Propose a non‑negative step function with a low benchmark score.

Algorithm
---------
1. Choose a problem size `n` (≈12000 for a 1000 s budget) and the target
   ℓ₁‑norm `S = sqrt(2 n)`.  All sequences are projected onto the simplex
   `{x ≥ 0, Σx = S}` after every update.

2. Initialise the search from several simple families
   (power‑law, exponential, sine‑window, linear decay and the previously
   discovered best sequence).  For each initialisation a short optimisation
   run is performed.

3. Optimisation pipeline (per run):
   * Phase A – exponential (mirror) descent on a soft‑max surrogate of the
     maximal convolution entry.
   * Phase B – multiplicative sub‑gradient descent on the *true* maximal
     convolution entry.
   * Phase C – deterministic local refinement: the current maximal
     convolution index `k*` is located, the contributions
     `c_i = a_i·a_{k*‑i}` are computed, and a small amount of mass is moved
     from the index with the largest contribution to the one with the
     smallest contribution.  The move is accepted only if the new maximal
     convolution does not increase (simple simulated‑annealing fallback is
     added).  This step aggressively flattens the convolution peak.

4. The best sequence over all runs is refined once more with the deterministic
   local refinement for whatever time is left.

5. The final sequence is truncated to remove trailing zeros, evaluated with
   the provided `evaluate_sequence`, stored for warm‑starts and returned.

All steps respect the global time budget; intermediate scores are printed.
"""

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    # The caller guarantees s > 0.
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def init_power_law(p: float, n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = a * (target_sum / a.sum())
    return a


def init_exponential(rate: float, n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i/n)."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64) / n)
    a = a * (target_sum / a.sum())
    return a


def init_sine(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine window."""
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = a * (target_sum / a.sum())
    return a


def init_linear_decay(n: int, target_sum: float) -> np.ndarray:
    """Linear decay from 1 to 0."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a = a * (target_sum / a.sum())
    return a


def init_warm(n: int, target_sum: float) -> np.ndarray:
    """Resize the globally stored best sequence if it exists."""
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        a = _resize_sequence(seq, n)
        a = a * (target_sum / a.sum())
        return a
    # fall back to a power‑law
    return init_power_law(0.55, n, target_sum)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used in Phase A)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – subgradient descent on the true max convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule for η
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – deterministic local refinement
# ----------------------------------------------------------------------


def deterministic_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing: repeatedly move a tiny amount of mass from the index
    contributing most to the current maximal convolution entry to the one
    contributing least.  The move is kept only if it does not increase the
    maximal convolution (with a small temperature‑based fallback).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    best_val = float(2.0 * n * best_max / (target_sum * target_sum))

    rng = np.random.default_rng()
    no_improve = 0
    start = time.time()

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        contrib = a[idx] * a[partner]

        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        if donor == receiver:
            # fallback to a random pair if the extreme indices coincide
            donor = rng.integers(lo, hi + 1)
            receiver = rng.integers(0, n)

        # maximum admissible transfer
        delta = min(a[donor] * 0.02, target_sum * 1e-4)
        if delta <= 0:
            break

        # apply the transfer
        a[donor] -= delta
        a[receiver] += delta

        # recompute max‑convolution
        new_max = _max_convolution_fft(a)

        # accept if we improve, otherwise revert or accept with temperature
        improved = new_max < best_max - 1e-12
        if improved:
            best_max = new_max
            best_a = a.copy()
            best_val = float(2.0 * n * best_max / (target_sum * target_sum))
            no_improve = 0
            print(f"[{time.time():.2f}] Deterministic refine improved max={best_max:.6f}")
        else:
            # simulated‑annealing fallback
            elapsed = time.time() - start
            total = deadline - start
            temperature = max(0.0, 0.04 * best_val * (1.0 - elapsed / total))
            if temperature > 0.0 and rng.random() < np.exp(-(new_max - best_max) / temperature):
                best_max = new_max
                best_a = a.copy()
                best_val = float(2.0 * n * best_max / (target_sum * target_sum))
                no_improve = 0
            else:
                # reject the move
                a[donor] += delta
                a[receiver] -= delta
                no_improve += 1

        # occasional full re‑normalisation to combat drift
        if (int(time.time() * 100) % 1000) == 0:
            a = a * (target_sum / a.sum())

        # give up if we have stalled for a while
        if no_improve > 500:
            # a small random jitter to escape a flat region
            jitter = rng.normal(scale=target_sum * 1e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Optimisation pipeline for a single run
# ----------------------------------------------------------------------


def _optimize_pipeline(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Run Phase A (mirror descent), Phase B (sub‑gradient) and
    Phase C (deterministic refinement) within the given time budget.
    Returns the best sequence found and its score.
    """
    now = time.time()
    remaining = deadline - now
    # Phase A – 15 % of remaining time
    phase_a_deadline = now + 0.15 * remaining
    a_a, val_a = _exponential_descent(a0, target_sum, phase_a_deadline)

    # Phase B – 30 % of what is left after A
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0:
        phase_b_deadline = now + 0.30 * remaining
        a_b, val_b = _subgradient_descent(a_a, target_sum, phase_b_deadline)
    else:
        a_b, val_b = a_a, val_a

    # Phase C – deterministic refinement for the rest of the budget
    now = time.time()
    if now < deadline:
        a_c, val_c = deterministic_refine(a_b, target_sum, deadline)
    else:
        a_c, val_c = a_b, _score(a_b)

    # pick the best of the three stages
    best_seq, best_val = a_a, val_a
    if val_b < best_val:
        best_seq, best_val = a_b, val_b
    if val_c < best_val:
        best_seq, best_val = a_c, val_c
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Define a pool of initialisers
    # --------------------------------------------------------------
    init_pool = [
        (init_power_law, {"p": 0.55}),
        (init_power_law, {"p": 0.70}),
        (init_power_law, {"p": 0.40}),
        (init_exponential, {"rate": 0.5}),
        (init_exponential, {"rate": 1.0}),
        (init_exponential, {"rate": 2.0}),
        (init_sine, {}),
        (init_linear_decay, {}),
        (init_warm, {}),
    ]

    rng = np.random.default_rng()
    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Multiple independent runs (each consumes a fraction of the budget)
    # --------------------------------------------------------------
    while time.time() < deadline - 20.0:  # keep ~20 s for final polishing
        remaining = deadline - time.time()
        # allocate at most 70 s per run, but never use more than half the rest
        run_budget = min(70.0, remaining * 0.5)
        run_deadline = time.time() + run_budget

        # pick a random initialiser
        func, params = rng.choice(init_pool)
        a0 = func(n=n, target_sum=target_sum, **params)

        # run the optimisation pipeline
        a_opt, val_opt = _optimize_pipeline(a0, target_sum, run_deadline)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best score {best_val:.6f}")

    # --------------------------------------------------------------
    # Final deterministic refinement using any remaining time
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ref, val_ref = deterministic_refine(best_seq.copy(),
                                               target_sum,
                                               deadline)
        if val_ref < best_val:
            best_seq = a_ref.copy()
            best_val = val_ref
            print(f"[{time.time():.2f}] Final refinement improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
