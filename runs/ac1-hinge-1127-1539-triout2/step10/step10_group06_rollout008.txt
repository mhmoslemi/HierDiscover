def propose_candidate(budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Choose n based on budget
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 600:
        n = 14000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # Warm start: try stored global or power-law sweep
    init = None
    if 'height_sequence_1' in globals():
        init = _resize_sequence(np.array(globals()['height_sequence_1']), n)
    p_candidates = [0.4, 0.5, 0.55, 0.6, 0.65, 0.7, 0.8]
    best_score = np.inf
    for p in p_candidates:
        a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
        a = a * (target_sum / a.sum())
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            init = a.copy()
    # Ensure init is normalized
    init = init * (target_sum / init.sum())
    best_a = init.copy()
    best_val = _score(best_a)
    print(f'Initial best score {best_val:.6f}')

    # Phase A: Mirror descent on softmax
    # allocate 0.15 of remaining time
    phaseA_deadline = time.time() + 0.15 * (deadline - time.time())
    aA, valA = _exponential_descent(best_a, target_sum, phaseA_deadline)
    if valA < best_val: best_a, best_val = aA, valA
    print(f'After Phase A best {best_val:.6f}')

    # Phase B: Simulated annealing mass transfer
    # allocate 0.20 of remaining time
    phaseB_deadline = time.time() + 0.20 * (deadline - time.time())
    aB, valB = _simulated_annealing_transfer(best_a, target_sum, phaseB_deadline)
    if valB < best_val: best_a, best_val = aB, valB
    print(f'After Phase B best {best_val:.6f}')

    # Phase C: Gradient descent with restarts
    # allocate 0.25 of remaining time
    phaseC_deadline = time.time() + 0.25 * (deadline - time.time())
    aC, valC = _gradient_descent_with_restarts(best_a, target_sum, phaseC_deadline)
    if valC < best_val: best_a, best_val = aC, valC
    print(f'After Phase C best {best_val:.6f}')

    # Phase D: Local refine for remaining time
    aD, valD = _local_refine(best_a, target_sum, deadline)
    if valD < best_val:
        best_a, best_val = aD, valD

        best_val = final_val  # just for logging
    except Exception as e:

def _simulated_annealing_transfer(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    n = a.shape[0]
    b = _conv_fft(a)

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # Temperature schedule: T0 = best_val * 0.02 maybe; decays linearly.
    T0 = best_val * 0.05

    while time.time() < deadline:
        # recompute k_max periodically to avoid drift? We'll recompute each iteration.
        # Actually compute from b each iteration update. Use b.
        k_max = int(np.argmax(b))

        # compute contributions for donor
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]  # size maybe hi-lo+1
        sum_contrib = contrib.sum()
        if sum_contrib > 0:
            donor = rng.choice(idx, p=contrib / sum_contrib)
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver: choose from indices with low contribution (maybe overall low a values)

        # but better choose low product also.
        # We'll compute weight_all for all positions to approximate low contributions.
        weight_all = np.zeros(n)
        # contributions to k_max only affect indices within [lo,hi]; for others weight 0.
        weight_all[lo:hi + 1] = contrib
        # choose receiver from low tail
        thresh = np.percentile(weight_all[weight_all > 0] if np.any(weight_all>0) else np.array([0]), 30)  # Actually we need low 30th percentile among all.
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(0, n)

        delta = rng.random() * max_delta

        b_new = b.copy()
        # linear updates
        b_new[donor:donor + n] += -2.0 * delta * a

        b_new[2*donor] += delta * delta
        b_new[2*receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        # compute new max and score
        new_max = b_new.max()
        new_score = float(2.0 * n * new_max / (target_sum * target_sum))

        cur_score = best_val  # keep best_val? Actually cur_score is current score for a, which we can compute with b.max()
        cur_score = float(2.0 * n * b.max() / (target_sum * target_sum))

        # Acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))
        accept = False
        if new_score < cur_score:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new

    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    b = _conv_fft(a)
    cur_max = b.max()
    cur_score = _score_from_max(cur_max, n, target_sum)  # compute formula
    best_a = a.copy()
    best_val = cur_score

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = best_val * 0.05

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]
        sum_contrib = contrib.sum()
        if sum_contrib > 0:
            donor = rng.choice(idx, p=contrib/sum_contrib)
        else:
            donor = rng.integers(lo, hi+1)

        # pick receiver uniformly not donor
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        # step size decreasing linearly with time
        elapsed = time.time() - start
        frac = max(0.001, (deadline - time.time()) / total)  # decreasing from 1 to small
        max_delta = min(a[donor] * frac * 0.5, target_sum * frac * 0.5)  # smaller fraction
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update b (copy)
        b_new = b.copy()
        b_new[donor:donor+n] -= 2.0 * delta * a
        b_new[receiver:receiver+n] += 2.0 * delta * a
        b_new[2*donor] += delta*delta
        b_new[2*receiver] += delta*delta
        if donor != receiver:
            b_new[donor+receiver] -= 2.0*delta*delta

        new_max = b_new.max()
        new_score = float(2.0 * n * new_max / (target_sum * target_sum))

        # temperature schedule
        temperature = max(0.0, T0 * (1.0 - elapsed / total))
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_max = new_max
            cur_score = new_score
            if new_score < best_val:
                best_val = new_score
                best_a = a.copy()
                print(f'[{time.time():.2f}] SA improvement {best_val:.6f}')

            b = _conv_fft(a)
            cur_max = b.max()
            cur_score = float(2.0 * n * cur_max / (target_sum * target_sum))
    return best_a, best_val

"""
Low‑score step function search.

The routine works in four stages under a fixed time budget (default 1000 s):

1. Warm‑start.  If a previous best sequence (global ``height_sequence_1``) is
   available it is resized; otherwise a short sweep over power‑law
   exponents provides a good initial shape.
2. Mirror‑descent on a soft‑max surrogate of the maximal convolution entry.
   This produces a rapid reduction of the objective.
3. Simulated‑annealing mass‑transfer on the *true* max‑convolution.
   Small amounts of mass are moved from indices that heavily
   contribute to the current maximal entry to random other indices,
   accepting worsening moves with a linear temperature schedule.
4. Heavy local refinement.  Exact O(n) convolution updates are used to
   move tiny amounts of mass from “bad’’ to “good’’ indices,
   with a simulated‑annealing acceptance rule.

All stages respect the overall budget and the best sequence found is
stored in ``height_sequence_1`` for warm‑starts in later calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective and simple helpers
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  Works for any sum."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _score_from_max(max_b: float, n: int, target_sum: float) -> float:
    """Objective expressed directly through the maximal convolution entry."""
    return float(2.0 * n * max_b / (target_sum * target_sum))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to length new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate and its gradient (used in Phase A & C)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – simulated‑annealing mass‑transfer on the true max‑convolution
# ----------------------------------------------------------------------


def _simulated_annealing_transfer(a0: np.ndarray,
                                  target_sum: float,
                                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Move tiny amounts of mass from indices that heavily contribute to the
    current maximal convolution entry to random other indices.
    Acceptance follows a linear temperature schedule.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_max = b.max()
    cur_score = _score_from_max(cur_max, n, target_sum)

    best_a = a.copy()
    best_val = cur_score

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = best_val * 0.05                 # initial temperature

    while time.time() < deadline:
        # locate the index of the current maximal entry
        k_max = int(np.argmax(b))

        # donor candidates – indices that actually affect entry k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]          # contribution to the max entry
        sum_contrib = contrib.sum()
        if sum_contrib > 0:
            donor = rng.choice(idx, p=contrib / sum_contrib)
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – uniform random index different from donor
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        # step size shrinks linearly with remaining time
        remaining = deadline - time.time()
        frac = max(0.001, remaining / total)       # in (0,1]
        max_delta = min(a[donor] * frac * 0.5, target_sum * frac * 0.5)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # ----- O(n) exact update of the convolution -----
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_score = _score_from_max(new_max, n, target_sum)

        # temperature decays linearly
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # acceptance rule
        accept = False
        if new_score < cur_score:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_max = new_max
            cur_score = new_score
            if new_score < best_val:
                best_val = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improve score={best_val:.6f}")

        # occasional renormalisation to keep Σa exact
        if (int(elapsed * 100) % 2500) == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_max = b.max()
            cur_score = _score_from_max(cur_max, n, target_sum)

    return best_a, best_val

# Phase C – projected gradient descent with random restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact local refinement (hill‑climbing with simulated‑annealing)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # locate current maximal entry
        k_max = int(np.argmax(b))

        # indices contributing to this entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contributions

        # pick a donor proportional to contribution
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a receiver among the light‑weight tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # propose a tiny transfer
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # exact O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # simulated‑annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        # occasional tiny jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the provided time budget (seconds) and stores the
    best sequence found in the global ``height_sequence_1`` for warm‑starts.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a finer search space)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 14000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start: either the stored best or a quick power‑law sweep
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        init = None

    # quick sweep over a few exponents if no good warm‑start exists
    exponents = [0.45, 0.50, 0.55, 0.60, 0.65, 0.70, 0.80]
    best_val = np.inf
    for p in exponents:
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            init = a.copy()

    # ensure the chosen init respects the simplex constraint
    init = init * (target_sum / init.sum())
    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent on soft‑max
    # --------------------------------------------------------------
    if time.time() < deadline:
        phaseA_deadline = time.time() + 0.12 * (deadline - time.time())
        aA, vA = _exponential_descent(best_seq, target_sum, phaseA_deadline)
        if vA < best_score:
            best_seq, best_score = aA, vA
        print(f"[{time.time():.2f}] After Phase A best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – simulated‑annealing mass transfer on true max‑conv
    # --------------------------------------------------------------
    if time.time() < deadline:
        phaseB_deadline = time.time() + 0.30 * (deadline - time.time())
        aB, vB = _simulated_annealing_transfer(best_seq, target_sum, phaseB_deadline)
        if vB < best_score:
            best_seq, best_score = aB, vB
        print(f"[{time.time():.2f}] After Phase B best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with random restarts
    # --------------------------------------------------------------
    if time.time() < deadline:
        phaseC_deadline = time.time() + 0.25 * (deadline - time.time())
        aC, vC = _gradient_descent_with_restarts(best_seq,
                                                  target_sum,
                                                  phaseC_deadline)
        if vC < best_score:
            best_seq, best_score = aC, vC
        print(f"[{time.time():.2f}] After Phase C best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – exact local refinement for the remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        aD, vD = _local_refine(best_seq, target_sum, deadline)
        if vD < best_score:
            best_seq, best_score = aD, vD
        print(f"[{time.time():.2f}] After Phase D best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up, evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (may raise if the environment does not expose it)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score={final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
