def _fft_len(n: int) -> int:
    """Return the next power‑of‑2 length for FFT of size 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    conv = _conv_fft(a)

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) convolve p) and then take the central n entries
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from high‑contributing indices to low‑contributing ones."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx]
        weight = a[idx] * partner

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise for numerical drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:

    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient: 2·a[k_max‑i] on feasible indices
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step size: do not push any coordinate negative
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Cheap SA that moves small mass from high‑weight to low‑weight indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.015 * best_val   # temperature proportional to current score

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # donor sampled proportionally to its contribution
        wsum = weight.sum()
        if wsum == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        # receiver from the low‑contribution tail
        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        # a modest transfer
        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved best score={best_val:.6f}")

        # occasional random jitter to diversify
        if rng.random() < 0.004:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val
