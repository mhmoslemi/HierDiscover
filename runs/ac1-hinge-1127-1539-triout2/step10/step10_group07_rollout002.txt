def _smooth_max_descent(a0, target_sum, deadline):
    a = _project_to_simplex(np.copy(a0), target_sum)
    best_a = a.copy()
    best_val = _score(a) # full score using max conv
    rng = np.random.default_rng()
    it = 0
    base_step = target_sum * 0.5
    beta = 0.5
    beta_max = 50.0
    while time.time() < deadline:
        # compute conv
        b = _conv_fft(a)  # length 2n-1
        # update beta gradually
        beta = min(beta_max, beta * 1.001 + 0.001) # simple increase
        # compute softmax weights
        maxb = b.max()
        exps = np.exp(beta * (b - maxb))
        w = exps / np.sum(exps)
        # gradient
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)
        L = len(b) - 1
        g = 2.0 * conv_full[L - np.arange(a.shape[0])]
        # step size schedule
        step = base_step / np.sqrt(it + 1.0)
        # update
        a -= step * g / (np.linalg.norm(g) + 1e-12)
        # symmetry enforcement (optional)
        # make a symmetric
        a = (a + a[::-1]) * 0.5
        # project back to simplex
        a = _project_to_simplex(a, target_sum)
        # optionally jitter occasionally
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum*0.0005, size=a.shape[0])
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        # evaluate current max_b (i.e., score)
        cur_val = b.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth it={it} score={best_val:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)
    # warm start
    init = ...
    # Stage1: smooth max descent for 0.5 of time
    stage1_deadline = time.time() + 0.5 * (deadline - time.time())
    best_seq, best_val = _smooth_max_descent(init, target_sum, stage1_deadline)
    # Stage2: short random restarts of smooth descent for 0.2 time
    if time.time() < deadline:
        rest_deadline = time.time() + 0.2 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2:
            jitter = rng.normal(scale=target_sum*0.02, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)
            a_tmp, val_tmp = _smooth_max_descent(start_seq, target_sum,
                                                time.time() + min(5.0, rest_deadline - time.time()))
            if val_tmp < best_val:
                best_val = val_tmp
                best_seq = a_tmp
                print(f"restart improved to {best_val}")
    # Stage3: local refine
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_val:
            best_seq = a_ref
            best_val = val_ref
    # trim zeros, evaluate
    arr = np.array(best_seq)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"Final score = {final_val}")
    except Exception as e:
        print(f"Evaluation error: {e}")
    # store
    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Propose a non‑negative step sequence that minimises the benchmark
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Smooth‑max gradient descent** – minimise the soft‑max of the
   autocorrelation `a*a`.  The temperature parameter `beta` is increased
   during the run so that the smooth approximation approaches the true
   `max`.  After each gradient step the vector is projected onto the
   simplex Σa = √(2 n) and forced to be symmetric (a[i] = a[n‑1‑i]).

2. **Randomised restarts** – the best sequence from stage 1 is jittered
   and the smooth‑max descent is run again for a few seconds.  The best
   of all restarts is kept.

3. **Exact local refinement** – an O(n) simulated‑annealing move that
   transfers a tiny amount of mass from a “bad’’ index (large contribution
   to the current maximal convolution entry) to a “good’’ one, updating
   the full convolution exactly.  This stage further reduces the
   maximum autocorrelation.

All intermediate vectors are kept non‑negative and lie on the simplex,
which makes the evaluation score equal to the maximal entry of the
autocorrelation.  The routine respects the given time budget
(`budget_s` seconds) and returns the best sequence found.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra / convolution utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σ x = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n‑1) using a power‑of‑two FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two arbitrary vectors using FFT.
    Result length = len(x) + len(y) - 1.
    """
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective (the same as evaluate_sequence for Σa = √(2n)):
        2·n·max(a*a) / (sum(a)²)
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Stage 1 – smooth‑max (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _smooth_max_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> Tuple[np.ndarray, float]:
    """
    Minimise the soft‑max of the autocorrelation a*a.
    The temperature parameter beta is increased during the run,
    and after each iteration the vector is projected onto the simplex
    and symmetrised.
    Returns the best vector found and its (max‑convolution) score.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    base_step = target_sum * 0.5
    beta = 0.5               # initial soft‑max temperature
    beta_max = 80.0          # final temperature (approaches true max)

    n = a.shape[0]
    L = 2 * n - 2            # index offset used for gradient extraction

    while time.time() < deadline:
        # ----- compute autocorrelation -----
        b = _conv_fft(a)                 # length 2n‑1

        # ----- increase temperature slowly -----
        # exponential schedule works well in practice
        beta = min(beta_max, beta * 1.001 + 0.001)

        # ----- soft‑max weights -----
        maxb = b.max()
        # stable soft‑max
        exps = np.exp(beta * (b - maxb))
        w = exps / exps.sum()            # shape (2n‑1,)

        # ----- gradient of Σ w_k·(a*a)_k -----
        # g_i = 2 * Σ_j w_{i+j} * a_j
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)   # length 3n‑1
        g = 2.0 * conv_full[L - np.arange(n)]

        # ----- step on the (sub)gradient -----
        step = base_step / math.sqrt(it + 1.0)
        gn = np.linalg.norm(g)
        if gn > 0:
            a -= step * g / gn

        # ----- enforce symmetry and simplex -----
        a = (a + a[::-1]) * 0.5           # a_i = a_{n-1-i}
        a = _project_to_simplex(a, target_sum)

        # ----- occasional jitter to escape plateaus -----
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # ----- bookkeeping -----
        cur_val = b.max()                 # because Σa = √(2n) ⇒ score = max(b)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth‑max it={it:6d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 2 – short random restarts of smooth‑max descent
# ----------------------------------------------------------------------


def _restart_smooth_max(
    best_seq: np.ndarray,
    target_sum: float,
    deadline: float,
) -> Tuple[np.ndarray, float]:
    """
    Perform a series of short smooth‑max descents starting from jittered
    versions of the current best sequence.  The overall best among the
    runs is returned.
    """
    rng = np.random.default_rng()
    best_a = best_seq.copy()
    best_val = _score(best_a)

    while time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        run_time = min(5.0, remaining)          # at most a few seconds per run
        if run_time < 1.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_seq.shape[0])
        start_seq = best_a + jitter
        start_seq = np.maximum(start_seq, 0.0)
        start_seq = _project_to_simplex(start_seq, target_sum)

        a_tmp, val_tmp = _smooth_max_descent(start_seq, target_sum,
                                             time.time() + run_time)
        if val_tmp < best_val:
            best_val = val_tmp
            best_a = a_tmp.copy()
            print(f"[{time.time():.2f}] Restart improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – exact local mass‑transfer refinement (simulated annealing)
# ----------------------------------------------------------------------


def _local_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that moves a tiny amount of mass
    from a “bad’’ index (large contribution to the maximal autocorrelation
    entry) to a “good’’ one.  The full convolution is updated in O(n)
    per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact autocorrelation
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature

    while time.time() < deadline:
        # ---- identify the peak position of the autocorrelation ----
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, k_max-i)
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # ---- choose a receiver from the low‑contribution tail ----
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo : hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # ---- propose a small transfer ----
        max_delta = min(a[donor] * 0.04, target_sum * 8e-4)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # ---- update the autocorrelation exactly (O(n)) ----
        b_new = b.copy()
        # linear terms: -2*delta*a at donor, +2*delta*a at receiver
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic self‑terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # ---- acceptance test (simulated annealing) ----
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # ---- periodic full recomputation to avoid drift ----
        if time.time() - start > total_time * 0.5:
            # renormalise once in the middle of the refinement
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

    return best_a, best_val
