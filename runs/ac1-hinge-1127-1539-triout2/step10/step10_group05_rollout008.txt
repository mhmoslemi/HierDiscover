"""
Non‑negative step‑function optimizer.

The evaluation metric is
    2·n·max(conv(a,a)) / (sum(a)**2)
with the L¹‑norm fixed to √(2 n).  The code performs

* a cheap parametric sweep to obtain a decent initial shape,
* several short Adam runs on a smooth log‑sum‑exp surrogate,
* a novel linear‑programming “exchange” step (max‑convolution ≤ M)
  that moves mass from regions where the current step‑function is large
  to regions where it is small,
* deterministic hill‑climbing and a brief simulated‑annealing finish,
* a final polishing Adam run.

All phases respect a wall‑clock budget (default 1000 s) and the best
sequence found is returned.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of non‑negative vector onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# Random initial families
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a point on the ℓ¹‑simplex Σa = √(2 n) from a few families."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential (two segments)
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_heavy_initial_sequence(n: int, rng) -> np.ndarray:
    """Large values near the two edges, decaying towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to the nearer edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """Explore a few one‑parameter families and keep the best (exact max‑conv)."""
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)   # 1‑based for power‑law
    lin = np.arange(n, dtype=float)          # 0‑based for exponentials

    # power‑law (coarse)
    for p in np.linspace(0.1, 2.0, 40):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # exponentials
    for lam in np.linspace(0.0005, 0.02, 40):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed
    for p in np.linspace(0.3, 1.5, 20):
        for lam in np.linspace(0.001, 0.01, 10):
            a = (idx ** (-p)) * np.exp(-lam * lin)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and projected Adam
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """Projected Adam on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0               # temperature for the soft‑max
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Linear‑programming exchange step
# ----------------------------------------------------------------------


def _lp_exchange_step(f: np.ndarray, target_sum: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_j     s.t.   f * g ≤ M   , g ≥ 0
    with M = max(f*f).  Scale g to the ℓ¹‑norm target and mix
    f ← (1‑t)·f + t·g.  Returns the updated f and its new max‑conv.
    The routine repeats the exchange while it yields improvement and
    time permits.
    """
    n = f.size
    M = _max_convolution_fft(f)
    best_f = f.copy()
    best_M = M

    while time.time() < deadline:
        # --- build the constraint matrix A (2n‑1) × n  -----------------
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            # for each j in [i_low, i_high] the coefficient is f[k‑j]
            for j in range(i_low, i_high + 1):
                rows.append(k)
                cols.append(j)
                data.append(f[k - j])
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M, dtype=float)

        # --- linear program -------------------------------------------------
        res = linprog(-np.ones(n), A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            # if the LP fails we abort the exchange phase
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # scale to the required ℓ¹‑norm
        g = g0 * (target_sum / sum_g0)

        # try a few mixing coefficients
        improved = False
        for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):
            new_f = (1.0 - t) * f + t * g
            new_f = _project_to_simplex(new_f, target_sum)
            new_M = _max_convolution_fft(new_f)
            if new_M < best_M - 1e-12:
                f = new_f
                best_f = new_f
                best_M = new_M
                improved = True
                print(f"[{time.time():.1f}] LP exchange → M={best_M:.6f}")
                break
        if not improved:
            # no progress – stop
            break
        # keep the current bound for the next iteration
        M = best_M
    return best_f, best_M


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that always attacks the current worst convolution entry."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # a good destination – pick the index with smallest counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try smaller steps
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # perturb a little to escape a plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing (O(n) per move)."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9999

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.5)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5

        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0) Seed from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded (n={n0}) → M={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # ------------------------------------------------------------------
    # 1) Fixed length (2000) – comfortable for all phases
    # ------------------------------------------------------------------
    n = 2000
    target = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1a) Parameter sweep (very cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_sweep, score_sweep = _parameter_sweep(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1b) A handful of Adam restarts (moderate budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # roughly 5‑second restarts
        n_restarts = max(1, int(remaining // 6.0))
        per_restart = remaining / n_restarts
        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            if r == 0 and best_seq is not None:
                a0 = best_seq.copy()
            else:
                # diversify initial points
                if rng.random() < 0.45:
                    a0 = _random_initial_sequence(n, rng)
                else:
                    a0 = _edge_heavy_initial_sequence(n, rng)
            a_opt = _adam_optimize(a0, target, time.time() + per_restart * 0.9)
            s = _max_convolution_fft(a_opt)
            if s < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = s
                print(f"[{time.time():.1f}] Adam restart {r+1}/{n_restarts} → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑programming exchange steps (core novelty)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        lp_deadline = deadline - 2.0
        best_seq, best_score = _lp_exchange_step(best_seq, target, lp_deadline)

    # ------------------------------------------------------------------
    # 3) Deterministic hill‑climb (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur, deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Hill‑climb → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final polish: a longer Adam run + a short hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_polish = _adam_optimize(best_seq, target, deadline - 1.5)
        b_polish = _full_convolution(a_polish)
        a_final, s_final = _hill_climb_deterministic(a_polish, b_polish,
                                                     deadline - 0.5)
        if s_final < best_score - 1e-12:
            best_seq = a_final.copy()
            best_score = s_final
            print(f"[{time.time():.1f}] Final polish → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 7) Official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
