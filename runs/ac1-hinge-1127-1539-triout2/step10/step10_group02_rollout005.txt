"""
propose_candidate
-----------------
A multi‑phase, time‑bounded optimizer for the min‑max auto‑convolution problem.

The pipeline:
1. Choose problem size `n` based on the time budget.
2. Generate a diverse pool of initial shapes:
   - Power‑law a_i ∝ (i+1)^(-p)  (p ∈ [0.30,0.80]).
   - Exponential a_i ∝ exp(-λ·i) (λ tuned for each `n`).
   - Mixtures of two power‑laws.
   Every candidate is projected onto the ℓ¹‑simplex (∑a = √(2n)).
3. Apply a fast “balanced‑order’’ permutation that spreads large masses
   to the edges (helps to keep the interior convolution low).
4. Refine each candidate with a projected Adam‑style sub‑gradient method
   that works on the true max‑convolution (gradient = 2·a shifted by the
   current peak index).
5. Deterministic mass‑transfer annealing (large‑donor / small‑receiver moves)
   with an exponential temperature schedule.
6. Swap‑based simulated annealing on the ordering (permutes the positions).
7. Random‑mass‑transfer annealing (purely stochastic donor/receiver selection)
   for the final escape from local minima.
8. A brief high‑β soft‑max GD polishing pass.

All phases respect the overall time budget (≈1000 s).  The best sequence
found is returned as a plain Python list and stored in the global variable
`height_sequence_1` for warm‑starts in future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (linear convolution a*a)
# ----------------------------------------------------------------------


def _fft_size_for_len(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (zero‑padding length for linear convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _fft_size_for_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (via FFT)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x_i ≥ 0, Σx_i = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex scaled by s."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Scoring helpers (objective = 2·n·max(conv) / (sum²); on the simplex sum = √(2n)
# ----------------------------------------------------------------------


def _score_simplex(a: np.ndarray, n: int) -> float:
    """Score when the ℓ¹‑norm is already √(2·n).  Equivalent to max(conv)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) normalised to ∑a = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) normalised to ∑a = target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _mixture_power_law(n: int, p1: float, p2: float, alpha: float,
                       target_sum: float) -> np.ndarray:
    """α·law(p1) + (1-α)·law(p2)  projected onto the simplex."""
    a1 = np.arange(1, n + 1, dtype=np.float64) ** (-p1)
    a2 = np.arange(1, n + 1, dtype=np.float64) ** (-p2)
    a = alpha * a1 + (1.0 - alpha) * a2
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Ordering heuristics
# ----------------------------------------------------------------------


def _balanced_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread large values to the two ends, then inward.
    Example (n=8): indices order = [0,7,1,6,2,5,3,4].
    """
    n = len(seq)
    order = []
    left, right = 0, n - 1
    toggle = True
    while left <= right:
        if toggle:
            order.append(left)
            left += 1
        else:
            order.append(right)
            right -= 1
        toggle = not toggle
    return seq[order]


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Legacy alternating order (largest to left/right)."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Projected Adam sub‑gradient descent (exact max convolution)
# ----------------------------------------------------------------------


def _adam_subgradient_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float,
                              step_lr: float = 0.4,
                              beta1: float = 0.9,
                              beta2: float = 0.999,
                              eps: float = 1e-8,
                              jitter_interval: int = 800,
                              jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the true max‑entry of the auto‑convolution.
    The gradient is the sub‑gradient at the current peak index.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    t = 0
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2·a[kmax-i] when the index is valid
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step_lr * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam subgrad it={t} best={best_val:.6f}")

        # occasional jitter to escape plateaus
        if t % jitter_interval == 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer annealing (large donor → small receiver)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                target_sum: float,
                                deadline: float,
                                start_temp: float = 0.8,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves guided by the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        # indices that actually contribute to the peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution of a pair (i, kmax-i)
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # propose δ
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)
        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 3000 == 0 and it > 0:
            # occasional jitter to avoid getting stuck
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

# Swap‑based permutation annealing
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Simulated annealing that swaps two positions."""
    cur_seq = a0.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        ratio = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] PermAnneal new best={best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional re‑shuffle near the end
        if ratio > 0.9 and rng.random() < 0.02:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Random mass‑transfer annealing (purely stochastic donor/receiver)
# ----------------------------------------------------------------------


def _random_mass_transfer_annealing(a0: np.ndarray,
                                   target_sum: float,
                                   deadline: float,
                                   start_temp: float = 0.8,
                                   end_temp: float = 1e-5,
                                   donor_bias: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Stochastic annealing that picks a donor proportional to its squared magnitude
    (biased towards large entries) and a uniform receiver.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # donor weighted by (a_i)^donor_bias
        probs = a ** donor_bias
        probs_sum = probs.sum()
        if probs_sum == 0:
            probs = np.ones_like(a) / n
        else:
            probs = probs / probs_sum
        donor = rng.choice(n, p=probs)

        # receiver uniform but not the same index
        receiver = rng.integers(0, n - 1)
        if receiver >= donor:
            receiver += 1

        # propose moving a fraction of donor's mass
        delta = rng.random() * a[donor] * 0.5
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] RandMT it={it} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max gradient descent (optional polishing)
# ----------------------------------------------------------------------


def _softmax_gradient_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float,
                              max_beta: float = 2e5,
                              step_factor: float = 0.6,
                              gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum‑GD on a soft‑max approximation of max(conv)."""
    def _grad_and_conv(x: np.ndarray, beta: float):
        n = x.shape[0]
        size = _fft_size_for_len(n)
        f = np.fft.rfft(x, n=size)
        conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
        m = conv.max()
        exps = np.exp(beta * (conv - m))
        p = exps / exps.sum()
        rev = x[::-1]
        rev_f = np.fft.rfft(rev, n=size)
        p_f = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(rev_f * p_f, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]
        return grad, m

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        g, _ = _grad_and_conv(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Softmax GD it={it} best={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max auto‑convolution.
    The routine respects the global time budget (≈budget_s seconds) and
    returns the best sequence it discovered.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1) Choose problem dimension n
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        # warm‑start from previous run – keep the same length
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev.shape[0]
    else:
        if budget_s >= 900:
            n = 15000
        elif budget_s >= 600:
            n = 12000
        else:
            n = 8000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 2) Build a pool of diverse initial shapes
    # --------------------------------------------------------------
    candidates = []

    # power‑law family
    for p in np.arange(0.30, 0.81, 0.05):
        cand = _power_law_sequence(n, float(p), target_sum)
        candidates.append(cand)

    # exponential family – λ chosen relative to n
    for lam_factor in (0.5, 1.0, 2.0, 4.0):
        lam = lam_factor / n
        cand = _exponential_sequence(n, lam, target_sum)
        candidates.append(cand)

    # mixtures of two power‑laws
    for (p1, p2, alpha) in [(0.3, 0.8, 0.6), (0.4, 0.7, 0.5), (0.45, 0.75, 0.7)]:
        cand = _mixture_power_law(n, p1, p2, alpha, target_sum)
        candidates.append(cand)

    # add warm‑start if available
    if "height_sequence_1" in globals():
        cand = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if cand.shape[0] != n:
            # linear resize preserving shape
            old = np.linspace(0, 1, cand.shape[0])
            new = np.linspace(0, 1, n)
            cand = np.interp(new, old, cand)
        cand = _project_to_simplex(cand, target_sum)
        candidates.append(cand)

    # jittered copies of each core candidate
    rng = np.random.default_rng()
    for base in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        jittered = np.maximum(base + jitter, 0.0)
        jittered = _project_to_simplex(jittered, target_sum)
        candidates.append(jittered)

    # apply a fast balanced ordering to all candidates (helps later)
    ordered_candidates = [_balanced_order(c) for c in candidates]

    # --------------------------------------------------------------
    # 3) Initial selection – keep the top few according to raw score
    # --------------------------------------------------------------
    scored = [(c, _max_convolution_fft(c)) for c in ordered_candidates]
    scored.sort(key=lambda x: x[1])
    # keep up to 6 best starters
    starters = [c for c, _ in scored[:6]]
    best_seq = starters[0].copy()
    best_val = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] Initial best raw score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Refine each starter with projected Adam sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.20
        for idx, init in enumerate(starters):
            if time.time() >= sub_dead:
                break
            a_opt, val = _adam_subgradient_descent(init, target_sum, sub_dead)
            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f"[{time.time():.2f}] Adam subgrad starter {idx} improved best={best_val:.6f}")

    # --------------------------------------------------------------
    # 5) Deterministic mass‑transfer annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_dead = time.time() + (deadline - time.time()) * 0.20
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, target_sum, mt_dead,
                                                    start_temp=0.9, end_temp=1e-5,
                                                    low_frac=0.015)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Deterministic MT improved best={best_val:.6f}")

    # --------------------------------------------------------------
    # 6) Permutation annealing (swap‑based)
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_dead = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(best_seq, target_sum, perm_dead,
                                                  init_score=best_val)
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Permutation anneal improved best={best_val:.6f}")

    # --------------------------------------------------------------
    # 7) Random mass‑transfer annealing (stochastic driver)
    # --------------------------------------------------------------
    if time.time() < deadline:
        rand_dead = time.time() + (deadline - time.time()) * 0.20
        a_rand, val_rand = _random_mass_transfer_annealing(best_seq, target_sum,
                                                          rand_dead,
                                                          start_temp=0.6,
                                                          end_temp=1e-5,
                                                          donor_bias=2.5)
        if val_rand < best_val:
            best_val = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Random MT anneal improved best={best_val:.6f}")

    # --------------------------------------------------------------
    # 8) Final soft‑max GD polishing (high β)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        fin_dead = deadline
        a_fin, val_fin = _softmax_gradient_descent(best_seq, target_sum,
                                                   fin_dead,
                                                   max_beta=5e5,
                                                   step_factor=0.5,
                                                   gamma=0.99)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final softmax polish best={best_val:.6f}")

    # --------------------------------------------------------------
    # 9) Post‑process: truncate trailing zeros, convert to list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # 10) Report final evaluation via provided function (optional)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
