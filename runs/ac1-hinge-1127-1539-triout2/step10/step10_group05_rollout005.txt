def solve_lp_max_sum(f, M, target, rows, cols, A_shape):

    data = np.repeat(f, n)   # shape n*n

    A = csr_matrix((data, (rows, cols)), shape=A_shape)

    g = cp.Variable(n)

    constraints = [A @ g <= M, g >= 0]

    objective = cp.Maximize(cp.sum(g))

    prob = cp.Problem(objective, constraints)

    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)

    if prob.status not in ["optimal", "optimal_inaccurate"]:

        return None

    g_val = g.value

    return g_val

def lp_improve(f, target, rows, cols, shape, time_deadline, max_iter=30):
    best_f = f.copy()
    best_M = _max_convolution_fft(best_f)
    t = 0.2
    n_iter = 0
    while time.time() < time_deadline and n_iter < max_iter:
        n_iter += 1
        M = best_M  # or compute each iteration from current f
        # Build matrix
        data = np.repeat(f, n) # using current f
        A = csr_matrix((data, (rows, cols)), shape=shape)
        # solve LP
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [A @ g_var <= M, g_var >= 0])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except cp.error.SolverError:
            break
        if g_var.value is None:
            break
        g = np.maximum(g_var.value, 0)
        sum_g = g.sum()
        if sum_g < 1e-12:
            break
        g = g * (target / sum_g)
        # Mix
        f_new = (1 - t) * f + t * g
        # enforce symmetry optionally
        f_new = (f_new + f_new[::-1]) / 2
        f_new = _project_to_simplex(f_new, target)
        M_new = _max_convolution_fft(f_new)
        if M_new < best_M - 1e-12:
            best_f = f_new
            best_M = M_new
            f = best_f  # continue from this improved f
            # maybe increase t for better step?
            t = min(0.5, t * 1.1)  # increase step if getting improvement
        else:
            # degrade t
            t = max(0.01, t * 0.9)
        # maybe break if improvement tiny
        if time.time() > time_deadline - 1.0:
            break
    return best_f, best_M

"""
Propose a non‑negative step function that (approximately) minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm is a three‑phase search for a fixed length `n` (default 2000):

1. **Parameter sweep** – cheap exploration of a few one‑parameter families
   (power‑law, exponential, edge‑heavy, mixed) using the exact max‑convolution.
   The best candidate is kept as a warm start.

2. **LP‑refinement** – repeatedly solve the linear program
      maximise   Σ g_j
      subject to f*g ≤ M ,  g ≥ 0
   where `M` is the current worst convolution value of `f`.  The solution `g`
   is scaled to the ℓ¹‑simplex and either replaces `f` (if it already improves
   the bound) or is mixed with `f` using a small step size.  The convolution
   matrix `f*g` is built once per iteration as a sparse Toeplitz matrix, so
   each LP solve is fast (≈0.2 s for `n=2000`).  The loop stops when the wall
   clock expires or after a fixed number of iterations.

3. **Local polishing** – deterministic hill‑climbing (exact max‑convolution
   updates) followed by a short simulated‑annealing phase.  Both respect the
   ℓ¹‑simplex constraint.

All steps respect the global 1000 s time budget.  The final sequence is trimmed
to remove trailing near‑zeros and returned as a Python list.  The code uses only
`numpy`, `scipy.sparse` and `cvxpy`; no file or network I/O is performed.
"""

import time
import math
import numpy as np
import cvxpy as cp
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Small parametric families (used for the cheap sweep)
# ----------------------------------------------------------------------


def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan a few one‑parameter families (power‑law, exponential,
    edge‑heavy, mixed) and return the best vector found according to the
    exact max‑convolution.
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law exponents
    for p in np.linspace(0.1, 2.0, 40):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.0005, 0.02, 40):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy (large at the ends, decays towards centre)
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 20):
        for lam in np.linspace(0.001, 0.01, 10):
            a = (idx ** (-p)) * np.exp(-lam * lin)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    return best_a, best_score


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max) and simulated annealing
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination j different from i
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # backtrack with smaller step
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # jitter and re‑project
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing (O(n) per move). Returns the best vector
    and its maximal convolution entry seen before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9999

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.5)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5
        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# LP refinement helper
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, target: float,
               rows: np.ndarray, cols: np.ndarray,
               shape_mat: tuple[int, int],
               deadline: float,
               max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve  max Σ g  s.t. f * g ≤ M and g ≥ 0.
    If the LP solution already improves the bound it replaces `f`,
    otherwise a convex combination is taken.
    Returns the best vector seen and its maximal convolution value.
    """
    n = f.size
    best_f = f.copy()
    best_M = _max_convolution_fft(best_f)

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1
        M = best_M

        # Build sparse convolution matrix A where (A g)_k = Σ_i f_i g_{k-i}
        data = np.repeat(best_f, n)               # length n²
        A = csr_matrix((data, (rows, cols)), shape=shape_mat)

        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                          [A @ g_var <= M,
                           g_var >= 0])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, verbose=False)
        except Exception as e:
            print(f"[{time.time():.1f}] LP solve exception: {e}")
            break

        if g_var.value is None:
            print(f"[{time.time():.1f}] LP failed (no solution).")
            break

        g = np.maximum(g_var.value, 0.0)
        s_g = g.sum()
        if s_g < 1e-12:
            break
        # Scale to the ℓ¹‑simplex
        g = g * (target / s_g)

        # Does the scaled vector already improve the bound?
        M_g = _max_convolution_fft(g)
        if M_g < best_M - 1e-12:
            best_f = g
            best_M = M_g
            print(f"[{time.time():.1f}] LP iter {iter_cnt}: direct replace → M={best_M:.6f}")
            continue

        # Otherwise mix with a modest step size and re‑project
        t = 0.2
        mixed = (1 - t) * best_f + t * g
        # Enforce symmetry (helps the convolution)
        mixed = (mixed + mixed[::-1]) / 2.0
        mixed = _project_to_simplex(mixed, target)
        mixed_M = _max_convolution_fft(mixed)

        if mixed_M < best_M - 1e-12:
            best_f = mixed
            best_M = mixed_M
            print(f"[{time.time():.1f}] LP iter {iter_cnt}: mixed improve → M={best_M:.6f}")
        else:
            # No improvement – stop early to save time
            print(f"[{time.time():.1f}] LP iter {iter_cnt}: no improvement (M={best_M:.6f})")
            break

    return best_f, best_M


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5       # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Fixed length – start with 2000 entries (good trade‑off)
    # ------------------------------------------------------------------
    n = 2000
    target = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 0)  Seed from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            if prev.size != n:
                prev = _upsample_sequence(prev, n)
            prev = _project_to_simplex(prev, target)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from previous run → M={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # ------------------------------------------------------------------
    # 1)  Quick parametric sweep (if time allows)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_sweep, score_sweep = _parameter_sweep(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep improvement → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a random start when nothing else is available
    # ------------------------------------------------------------------
    if best_seq is None:
        # a simple symmetric exponential around the edges
        lam = rng.uniform(0.001, 0.02)
        idx = np.arange(n)
        d = np.minimum(idx, n - 1 - idx)
        best_seq = np.exp(-lam * d)
        best_seq = _project_to_simplex(best_seq, target)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Random fallback → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  LP‑based refinement (core of the new method)
    # ------------------------------------------------------------------
    # Pre‑compute the sparse pattern of the convolution matrix (size n²)
    idx_arr = np.arange(n, dtype=int)
    rows = (idx_arr[:, None] + idx_arr[None, :]).ravel()
    cols = np.repeat(idx_arr, n)
    shape_mat = (2 * n - 1, n)

    # allocate a reasonable number of LP iterations depending on remaining time
    remaining = deadline - time.time()
    max_lp_iters = min(30, int(remaining // 5))   # ≈5 s per iteration budget
    best_seq, best_score = _lp_refine(best_seq, target,
                                      rows, cols, shape_mat,
                                      deadline,
                                      max_iters=max_lp_iters)

    # ------------------------------------------------------------------
    # 3)  Deterministic hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur,
                                                deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Hill‑climb → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur,
                                        deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal → M={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5)  Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 6)  Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
