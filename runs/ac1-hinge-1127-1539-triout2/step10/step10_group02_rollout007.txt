"""
Propose a step‑function (non‑negative sequence) that minimizes the
benchmark objective 2·n·max(conv)/(sum²).  The algorithm follows an
exploratory‑to‑refinement pipeline:

* Phase 0 – quick search on a modest size (n≈5000).  Many power‑law and
  exponential families are tried; each candidate is refined by a short
  soft‑max gradient descent and projected to the ℓ¹‑simplex (sum = √(2n)).
* Phase 1 – upscale the best small‑size sequence to a larger problem
  (n≈12000) and continue optimisation with more aggressive sub‑gradient
  descent, deterministic mass‑transfer and a high‑β soft‑max GD pass.
* Phase 2 – re‑ordering heuristics (alternating, golden‑ratio spacing,
  random shuffles) are examined and the best ordering is kept.
* Phase 3 – permutation annealing (swap‑based simulated annealing) to
  further lower the convolution peak.
* Phase 4 – final polishing with a very high soft‑max temperature.

All convolutions use FFT (O(n log n)).  The routine respects the given
budget (default 1000 s) and stores the final sequence in the global
variable ``height_sequence_1`` for warm‑starts in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(a*a)/(Σa)² (with Σa=√(2n) this equals max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)^(-p) normalized to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Return exp(-lam·i) normalized to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _golden_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread large entries uniformly using a golden‑ratio step.
    The largest value goes to position 0, the next largest to
    round(phi·n) mod n, etc.
    """
    n = seq.shape[0]
    phi = (np.sqrt(5) - 1) / 2.0          # ≈0.618
    positions = (np.arange(n) * phi) % 1.0
    order = np.argsort(positions)
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    out[order] = sorted_vals
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _softmax_gd(a0: np.ndarray,
                target_sum: float,
                deadline: float,
                max_beta: float = 1e7,
                step_factor: float = 0.6,
                gamma: float = 0.95) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    The temperature β is gradually increased up to ``max_beta``.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] softmax_gd it={it:5d} β={beta:.1e} score={best_val:.6f}")

        # increase β on a schedule
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Uses the sub‑gradient 2*a[k_max‑i] (zero outside support).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(a.shape[0])
        mask = (kmax - idx >= 0) & (kmax - idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[kmax - idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale,
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves targeting the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] perm_ann new best={best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score

    # power‑law family
    for p in np.arange(0.35, 0.66, 0.02):
        init_small.append(_power_law_sequence(n_small, float(p), target_small))

    # exponential family
    for lam in np.arange(0.001, 0.021, 0.004):
        init_small.append(_exponential_sequence(n_small, float(lam), target_small))

    # a few random jittered copies of the best known sequence (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        prev_small = _resize_sequence(prev, n_small)
        init_small.append(prev_small)

        for _ in range(3):
            jitter = rng.normal(scale=target_small * 0.01, size=n_small)
            cand = prev_small + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_small)
            init_small.append(cand)

    # random permutations of a couple of power‑law candidates
    for p in (0.45, 0.55):
        seq = _power_law_sequence(n_small, p, target_small)
        rng.shuffle(seq)
        init_small.append(seq)

    # ------------------------------------------------------------------
    # Quick optimisation of each small candidate (soft‑max GD, short budget)
    # ------------------------------------------------------------------
    best_small = None
    best_small_val = np.inf

    # allocate 20 % of total time to the small‑size search
    small_deadline = start + 0.20 * (deadline - start)

    for cand in init_small:
        if time.time() >= small_deadline:
            break
        # Give each candidate up to 12 s (or remaining time) for a quick GD
        cand_deadline = min(time.time() + 12.0, small_deadline)
        a_opt, val = _softmax_gd(cand, target_small, cand_deadline,
                                 max_beta=5e5, step_factor=0.7, gamma=0.97)
        if val < best_small_val:
            best_small_val = val
            best_small = a_opt.copy()
            print(f"[{time.time():.2f}] New best_small score={best_small_val:.6f}")

    # fallback in the unlikely case that no candidate succeeded
    if best_small is None:
        best_small = _power_law_sequence(n_small, 0.55, target_small)
        best_small_val = fast_score(best_small)

    # ------------------------------------------------------------------
    # Phase 1 – upscale to the big problem and perform heavier optimisation
    # ------------------------------------------------------------------
    best_big = _resize_sequence(best_small, n_big)
    best_big = _project_to_simplex(best_big, target_big)
    best_big_val = fast_score(best_big)

    # create a richer pool at the big size (different exponents, reorderings)
    init_big = [best_big]

    # power‑law variants and exponential variants
    for p in np.arange(0.40, 0.61, 0.03):
        seq = _power_law_sequence(n_big, float(p), target_big)
        init_big.append(seq)

    for lam in np.arange(0.002, 0.012, 0.003):
        seq = _exponential_sequence(n_big, float(lam), target_big)
        init_big.append(seq)

    # apply a few deterministic re‑orderings to each generated shape
    extra_big = []
    for seq in init_big:
        extra_big.append(_alternating_order(seq))
        extra_big.append(_golden_order(seq))
        tmp = seq.copy()
        rng.shuffle(tmp)
        extra_big.append(tmp)
    init_big.extend(extra_big)

    # ------------------------------------------------------------------
    # Heavier optimisation on big candidates (soft‑max GD + sub‑gradient)
    # ------------------------------------------------------------------
    while time.time() < deadline - 300.0:   # reserve ~5 min for later phases
        # pick a random init from the pool
        cand = rng.choice(init_big)
        # allocate a modest budget (≈15 s) per candidate
        cand_dead = min(time.time() + 15.0, deadline - 300.0)
        a1, v1 = _softmax_gd(cand, target_big, cand_dead,
                             max_beta=2e6, step_factor=0.5, gamma=0.97)
        # give a short sub‑gradient refinement if time permits
        if time.time() < cand_dead:
            a2, v2 = _subgradient_descent(a1, target_big,
                                          deadline=min(cand_dead, time.time() + 8.0),
                                          step_factor=0.55, gamma=0.92)
            a_candidate, v_candidate = (a2, v2) if v2 < v1 else (a1, v1)
        else:
            a_candidate, v_candidate = a1, v1

        if v_candidate < best_big_val:
            best_big_val = v_candidate
            best_big = a_candidate.copy()
            print(f"[{time.time():.2f}] Global best after big GD = {best_big_val:.6f}")

    # ------------------------------------------------------------------
    # Phase X – exact sub‑gradient descent (long run)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        sg_dead = time.time() + (deadline - time.time()) * 0.12
        a_sg, v_sg = _subgradient_descent(best_big, target_big, sg_dead,
                                          step_factor=0.5, gamma=0.94)
        if v_sg < best_big_val:
            best_big_val = v_sg
            best_big = a_sg.copy()
            print(f"[{time.time():.2f}] After sub‑gradient best={best_big_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.15
        a_mt, v_mt = _deterministic_mass_transfer(best_big, mt_dead,
                                                  target_big,
                                                  start_temp=0.6,
                                                  end_temp=5e-5,
                                                  low_frac=0.06)
        if v_mt < best_big_val:
            best_big_val = v_mt
            best_big = a_mt.copy()
            print(f"[{time.time():.2f}] After DMT best={best_big_val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – re‑ordering heuristics (try a few different permutations)
    # ------------------------------------------------------------------
    reorder_cands = [best_big,
                     _alternating_order(best_big),
                     _golden_order(best_big)]

    for _ in range(2):
        tmp = best_big.copy()
        rng.shuffle(tmp)
        reorder_cands.append(tmp)

    for cand in reorder_cands:
        cand_score = fast_score(cand)
        if cand_score < best_big_val:
            best_big_val = cand_score
            best_big = cand.copy()
            print(f"[{time.time():.2f}] Reorder improved score={best_big_val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – permutation annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 40.0:
        perm_dead = time.time() + (deadline - time.time()) * 0.20
        a_perm, v_perm = _permutation_annealing(best_big, target_big,
                                                perm_dead,
                                                init_score=best_big_val)
        if v_perm < best_big_val:
            best_big_val = v_perm
            best_big = a_perm.copy()
            print(f"[{time.time():.2f}] After perm anneal best={best_big_val:.6f}")

    # ------------------------------------------------------------------
    # Phase E – final high‑β soft‑max polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, v_fin = _softmax_gd(best_big, target_big,
                                   deadline,
                                   max_beta=5e6,
                                   step_factor=0.45,
                                   gamma=0.99)
        if v_fin < best_big_val:
            best_big_val = v_fin
            best_big = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_big_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_big, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # sanity check with the official evaluator (may raise – ignore if not present)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
