"""
Propose a low‑score step function.

The search consists of four stages:

1. **Coarse parametric warm‑start** – grid search over a two‑parameter
   family a_i = (i+1)^(-p)·exp(-λ·i/small_n) on a reduced problem size,
   followed by random refinements.

2. **Exponential mirror‑descent** – soft‑max surrogate of the max‑convolution
   with a slowly increasing temperature (β) and multiplicative updates.

3. **Deterministic mass‑transfer with exact line‑search** – the current
   maximal convolution entry is identified; a donor index (large contribution)
   and a receiver index (small contribution) are chosen and the maximal amount
   of mass that can be moved without raising any convolution entry above the
   second‑largest value is transferred (computed analytically).

4. **Very short simulated‑annealing polish** – cheap stochastic moves to escape
   tiny local plateaus.

All stages respect the time budget (default 1000 s).  The final sequence is
scaled to ℓ¹‑norm √(2 n) and trailing zeros are removed before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / scoring utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 size for FFT of a length‑n linear convolution."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (length 2·n‑1)."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    conv = _conv_fft(a)
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective value (equal to the maximal convolution entry after normalisation)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (for exponential descent)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                       # soft‑max weights, sum = 1

    # grad = 2 * (reverse(a) convolve p) and then take the central n entries
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """Search a 2‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n)."""
    p_vals = np.linspace(0.2, 1.5, 31)          # slightly finer grid on p
    lam_vals = np.linspace(0.0, 12.0, 31)      # finer λ grid
    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    rng = np.random.default_rng()
    # random refinements while time permits
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.5)
        lam = rng.exponential(1.0) * 4.0   # bias towards moderate λ
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score

# Stage 2 – exponential (multiplicative) descent on soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    base_eta = 0.06               # step‑size multiplier
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)            # multiplicative update
        a = a * (target_sum / a.sum())      # re‑normalise

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Stage 3 – deterministic mass‑transfer with exact line‑search
# ----------------------------------------------------------------------


def _precise_mass_transfer_refine(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move mass from a high‑contributing index of the current
    maximal convolution entry to a low‑contributing index.
    For each donor/receiver pair we compute the largest feasible transfer
    that guarantees *all* convolution entries stay ≤ the second‑largest value.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    pad_len = n - 1                # amount of zero padding on each side

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = float(conv.max())

        # second distinct maximum (helps us define a safe target)
        mask = conv < max_val - 1e-12
        if mask.any():
            second_max = float(conv[mask].max())
        else:
            second_max = max_val

        # convergence test – if gap extremely small we stop early
        if max_val - second_max < 1e-12:
            # add a tiny jitter and continue a bit, otherwise break
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            continue

        # pick one of the (possibly many) maximal positions at random
        max_positions = np.flatnonzero(conv >= max_val - 1e-12)
        k_max = int(rng.choice(max_positions))

        # contribution weights for this k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # candidate donors – top few weights
        donors = idx[np.argsort(weight)[::-1][:5]]
        # candidate receivers – smallest weights (including zeros)
        receivers = idx[np.argsort(weight)[:5]]

        # pre‑compute padded a for fast shift look‑ups
        a_pad = np.concatenate([np.zeros(pad_len), a, np.zeros(pad_len)])

        best_pair = None
        best_delta = 0.0

        # analyse each (donor,receiver) pair
        for donor in donors:
            shift_donor = a_pad[(np.arange(2 * n - 1) - donor) + pad_len]
            for receiver in receivers:
                if donor == receiver:
                    continue
                shift_receiver = a_pad[(np.arange(2 * n - 1) - receiver) + pad_len]
                diff = shift_receiver - shift_donor          # effect of moving +δ at receiver, –δ at donor
                pos = diff > 1e-15                           # only positions that would increase conv
                if pos.any():
                    # bound to keep every entry ≤ second_max
                    bound_vals = (second_max - conv[pos]) / (2.0 * diff[pos])
                    if (bound_vals < 0).any():
                        continue        # infeasible pair
                    feasible = min(a[donor], bound_vals.min())
                else:
                    feasible = a[donor]   # no upward pressure anywhere

                # safety cap – stay in the linear regime
                feasible = min(feasible, a[donor] * 0.5)

                if feasible > best_delta:
                    best_delta = feasible
                    best_pair = (donor, receiver)

        # if no feasible pair was found, fall back to a tiny random transfer
        if best_pair is None or best_delta < 1e-14:
            donor = int(rng.choice(idx))
            receiver = int(rng.choice(idx))
            if donor == receiver:
                continue
            max_delta = a[donor] * 0.02
            delta = rng.random() * max_delta
        else:
            donor, receiver = best_pair
            delta = best_delta

        # apply the transfer
        a[donor] -= delta
        a[receiver] += delta

        # renormalise (tiny drift may appear)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

        # evaluate improvement
        new_val = _max_convolution_fft(a)
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Precise transfer improve score={best_val:.6f}")

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Very short SA that moves small mass between random indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _max_convolution_fft(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.02 * best_val          # initial temperature

    while time.time() < deadline:
        # pick donor proportional to its contribution to the current max entry
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        wsum = weight.sum()
        if wsum == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        # receiver: any low‑weight index (different from donor)
        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = int(rng.choice(low_cands))
        else:
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _max_convolution_fft(a_new)
        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved best score={best_val:.6f}")

        # occasional random jitter to keep exploration alive
        if rng.random() < 0.003:
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _max_convolution_fft(a)

    return best, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the allotted time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size depending on the remaining budget
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start: either from a previous run or a fresh power‑law
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # --------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # --------------------------------------------------------------
    small_n = max(2500, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(25.0,
                                      (deadline - time.time()) * 0.12)
    best_small_seq, _ = _grid_param_search(small_n, small_target, param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑started from parametric search (n={n})")

    # --------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.30
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – deterministic precise mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.45
        a_mt, val_mt = _precise_mass_transfer_refine(best_seq,
                                                     target_sum,
                                                     mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After precise mass‑transfer best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – very short simulated‑annealing polish (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_deadline)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
        print(f"[{time.time():.2f}] After SA polish best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean up and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a future call
    globals()["height_sequence_1"] = final_seq
    return final_seq
