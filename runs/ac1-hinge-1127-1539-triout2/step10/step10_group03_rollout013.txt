"""
Hybrid optimizer with final permutation annealing.

The algorithm consists of three phases:

1. **Continuous shape optimisation** – a set of diverse initial
   candidates (power‑law, exponential and random Dirichlet) are
   projected onto the ℓ¹‑simplex Σa = √(2 n) and improved by a
   momentum projected‑gradient descent on a soft‑max approximation of
   the convolution peak.  The best shape found is kept.

2. **Discrete re‑ordering** – the values of the best shape are kept
   fixed while a simulated‑annealing walk swaps two positions at a time.
   The acceptance rule is Metropolis with an exponentially decaying
   temperature; the objective is the exact convolution peak (no soft‑max).

3. **Fine‑grained mass‑transfer** – a short deterministic refinement
   (the same routine used in the reference code) is applied to the
   reordered sequence.

All steps respect non‑negativity, the per‑element ceiling of 1000 and the
ℓ¹‑simplex constraint.  The function `propose_candidate` returns the best
sequence found before the time budget expires (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (equals the convolution peak when Σa=√(2n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence proportional to (i+1)^(-p) and renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_dirichlet(n: int, alpha: float, target_sum: float) -> np.ndarray:
    rng = np.random.default_rng()
    a = rng.gamma(alpha, 1.0, size=n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (used in continuous optimisation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (rev(a) ⋆ p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (same as reference)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves with a Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering via simulated annealing (swap moves)
# ----------------------------------------------------------------------


def _swap_annealing(a0: np.ndarray,
                   deadline: float,
                   start_temp: float = 0.1,
                   end_temp: float = 1e-6,
                   log_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Metropolis annealing that swaps two positions at a time.
    The objective is the exact convolution peak.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    t0 = time.time()
    total = deadline - t0
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - t0
        tau = elapsed / total
        T = start_temp * (end_temp / start_temp) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        a[i], a[j] = a[j], a[i]

        new_max = _max_convolution_fft(a)
        delta = new_max - cur_max

        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SwapAnneal it={it:5d} new best={best_max:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        if log_interval and it % log_interval == 0 and it > 0:
            # occasional recompute to guard against numerical drift
            cur_max = _max_convolution_fft(a)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Helper: alternating large/small ordering (kept for reference)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest value leftmost, second largest rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine uses continuous optimisation, a swap‑based annealing of the
    ordering and a short deterministic mass‑transfer refinement.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 4000

    target_sum = math.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build initial pool of candidates
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.3, 0.8, 7):
        init_pool.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.015, 5):
        init_pool.append(_exponential_sequence(n, float(lam), target_sum))

    # random Dirichlet‑like points
    for alpha in [0.5, 1.0, 2.0]:
        init_pool.append(_random_dirichlet(n, alpha, target_sum))

    # previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.array(prev, dtype=np.float64)
        init_pool.append(_resize_sequence(prev_arr, n))

    # jitter each element to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Phase 1 – continuous optimisation (gradient descent)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate about 50 % of the time to this phase
    phase1_deadline = min(deadline - 30.0, start_time + (deadline - start_time) * 0.5)

    while time.time() < phase1_deadline:
        # pick a seed: either jitter around current best or a random pool entry
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_sum * 0.005, size=n)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_sum)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 15.0, phase1_deadline)
        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed,
            run_dead,
            target_sum,
            max_beta=5e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Phase‑1 new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – discrete re‑ordering via swap annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        seq_swap, max_swap = _swap_annealing(
            best_seq,
            deadline - 5.0,
            start_temp=0.2,
            end_temp=1e-7,
            log_interval=500,
        )
        if max_swap < best_score:
            best_score = max_swap
            best_seq = seq_swap.copy()
            print(f"[{time.time():.2f}] Phase‑2 after swaps best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 1.0)
        a_mt, mt_score = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_sum,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05
        )
        if mt_score < best_score:
            best_score = mt_score
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Phase‑3 final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional logging with the provided evaluator
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
