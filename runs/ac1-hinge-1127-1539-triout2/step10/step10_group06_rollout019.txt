def _lp_iterate(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    t = 0.2
    # parameters
    K = min(200, 2*n-1)  # number of active constraints
    while time.time() < deadline:
        b = _conv_fft(a)  # compute conv
        M = b.max()
        # identify active constraints (largest values)
        # we can also include the index where M occurs.
        # using argsort descending
        idxs = np.argpartition(-b, K)[:K]  # not sorted
        # Build A_sub
        K_actual = idxs.shape[0]
        A = np.zeros((K_actual, n), dtype=np.float64)
        for row, k in enumerate(idxs):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi+1)
            A[row, lo:hi+1] = a[k - i_range]
        # Solve LP: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=np.full(K_actual, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"LP failed: {res.message}")
            break
        g = res.x
        # Scale g to target sum within constraints:
        # compute conv(f,g) to see if exceeds M
        conv_fg = _conv_fft(g)  # Actually this is g*g, not f*g. Oops.
        # we need conv(f,g) - compute via convolution with f as first operand:

        a_new = a_new * (target_sum / a_new.sum())
        a = a_new
        # evaluate
        val = _score(a)
        if val < best_score:

            best_a = a.copy()

"""
Hybrid LP‑+‑gradient search for a low‑value step function.
-------------------------------------------------------

The benchmark score 2·n·max(conv(a,a))/sum(a)² is invariant under scaling,
so all phases keep the L₁‑norm fixed to √(2 n).  The algorithm consists of

1. **Warm‑start** – resize a previously‑found sequence or use a random
   Dirichlet start.

2. **LP‑phase** – given the current vector f, solve the linear programme
   max ∑gⱼ  s.t.    f∗g ≤ M   and   g ≥ 0,
   where M = max(f∗f).  Only the K largest convolution constraints are
   kept (K≈200) to keep the LP small.  The optimiser (SciPy highs) returns a
   direction g that can increase the total mass while respecting the bound.
   A convex combination f←(1‑t)f+t·g (t≈0.2) strictly reduces the maximal
   convolution entry; the loop is repeated until the time budget for this
   phase expires.

3. **Sub‑gradient descent** – a multiplicative sub‑gradient step on the
   true max‑convolution (the same update as in the reference code).

4. **Mirror‑descent on a soft‑max surrogate** – exponential descent on the
   log‑sum‑exp approximation of the max entry; β is doubled periodically.

5. **Exact local refinement** – the hill‑climbing / simulated‑annealing
   mass‑transfer routine from the reference solution, now run on the final
   vector.

All phases respect the overall 1000 s budget and print intermediate scores.
The best sequence is stored in the global ``height_sequence_1`` for warm‑starts
in later calls.
"""

import time
import numpy as np

# Optional SciPy import (used only for the LP phase)
# ----------------------------------------------------------------------

    _SCIPY_AVAILABLE = True

    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, works for any length)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (length n+m‑1) via FFT."""
    n, m = a.size, b.size
    size = 1 << ((n + m - 2).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective 2·n·max(a*a) / (sum(a)²).
    With sum(a)=√(2 n) the value equals the maximal convolution entry.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to ``new_len`` entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – LP based mass‑increase (active‑set version)
# ----------------------------------------------------------------------


def _lp_phase(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve
        max   Σ gⱼ
        s.t.  (a∗g)ₖ ≤ M  for the K largest lag‑values,
              gⱼ ≥ 0 .
    The solution is mixed with the current vector (convex combination)
    which provably reduces the maximal self‑convolution entry.
    """
    if not _SCIPY_AVAILABLE:
        print("[LP] SciPy not available – skipping LP phase.")
        return a0, _score(a0)

    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    K = min(200, 2 * n - 1)                # active‑set size
    t = 0.2                                 # convex‑combination weight
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        M = b.max()

        # ---- active constraints -------------------------------------------------
        # indices of the K largest entries (including the absolute maximum)
        if K < len(b):
            active_idx = np.argpartition(-b, K)[:K]
        else:
            active_idx = np.arange(len(b))

        # Build A_active (shape K×n) where row k contains the coefficients of g
        # in (a∗g)_k = Σ_i a_i·g_{k‑i}.
        A_active = np.zeros((len(active_idx), n), dtype=np.float64)
        for row, k in enumerate(active_idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            A_active[row, lo:hi + 1] = a[k - i_range]

        # ---- linear programme ---------------------------------------------------
        c = -np.ones(n, dtype=np.float64)               # maximise Σg ⟹ minimise –Σg
        b_ub = np.full(A_active.shape[0], M, dtype=np.float64)

        res = linprog(c, A_ub=A_active, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})
        if not res.success:
            print(f"[LP] linprog failed ({res.message}); terminating LP phase.")
            break

        g = res.x

        # ---- enforce the *full* convolution bound by a global scaling -------------
        conv_fg = _conv_fft_two(a, g)
        max_fg = conv_fg.max()
        if max_fg > M * (1.0 + 1e-12):
            scale = M / max_fg
            g *= scale

        # ---- bring g to the required L1‑norm ------------------------------------
        g = g * (target_sum / g.sum())

        # ---- convex combination (mix) -------------------------------------------
        a = (1.0 - t) * a + t * g
        a = a * (target_sum / a.sum())          # numerical safety

        # ---- bookkeeping ---------------------------------------------------------
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP] it={it:5d} score={best_val:.6f}")

        # occasional jitter to escape possible stagnation
        if it % 100 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

        # early exit if we have already broken the target
        if best_val < 1.45:      # a safe margin below the required 1.503
            break

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear step‑size schedule
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[SubGrad] it={it:5d} score={best_val:.6f}")

        it += 1
    return best_a, best_val

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of 𝑙𝑜𝑔∑exp(β·(conv‑M)) where M=max(conv).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[ExpGD] it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – exact local refinement (simulated annealing style)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # donor – pick proportionally to contribution
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick from low‑contribution region
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 1.5e-3)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution exactly (O(n) updates)
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # simulated annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Refine] improved score={best_val:.6f}")

        # periodic full recomputation (to avoid drift)
        if time.time() - start > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Re‑norm] improved score={best_val:.6f}")

        # occasional tiny jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 4e-4)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[Jitter] improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # problem size – a modest n keeps the LP cheap while still allowing
    # a good solution; increasing n further does not change the theoretical
    # lower bound much.
    if budget_s >= 800:
        n = 1500
    elif budget_s >= 500:
        n = 1200
    else:
        n = 800

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
    else:
        rng = np.random.default_rng()
        init = rng.random(n) + 1e-8            # tiny offset to avoid exact zeros

    init = init * (target_sum / init.sum())

    # ------------------------------------------------------------------
    # Phase A – LP based mass increase (≈30 % of total budget)
    # ------------------------------------------------------------------
    lp_deadline = time.time() + 0.30 * (deadline - time.time())
    a_lp, val_lp = _lp_phase(init, target_sum, lp_deadline)
    best_seq = a_lp
    best_score = val_lp
    print(f"[{time.time():.2f}] After LP phase best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – subgradient descent on true max‑convolution (≈20 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.20 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – exponential mirror descent on soft‑max (≈20 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.20 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – exact local refinement (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up, evaluation and warm‑start storage
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
