"""
Propose a low‑score step‑function.

This implementation follows a multi‑stage pipeline similar to the previous state‑of‑the‑art
but adds a dedicated *greedy max‑reduction* phase that moves mass between indices while
guaranteeing that the maximal convolution entry never increases.  The algorithm:

1.   Warm‑start with a power‑law/exponential shape (or the previous best sequence).
2.   Small‑scale parametric grid search to obtain a better starting shape.
3.   Exponential (multiplicative) mirror‑descent on a soft‑max surrogate of the max‑conv.
4.   Projected gradient descent with random restarts (soft‑max surrogate).
5.   Deterministic mass‑transfer refinement (as in the reference code).
6.   **Greedy max‑reduction** – the new stage that computes, for the current
     maximal convolution index, a donor‑receiver pair and the largest admissible
     mass transfer that cannot raise any convolution entry above the current maximum.
7.   Final trimming and evaluation.

All stages respect the global time budget (default 1000 s) and keep the sequence
non‑negative with ℓ¹‑norm `sqrt(2·n)`.  The best sequence found is stored in the
global variable `height_sequence_1` for warm‑starts in future calls.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 size for a full linear convolution of length 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1)."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For normalized sum this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:  # safety (should never happen)
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to `new_len` entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence a_i = (i+1)^(-p) normalised to `target_sum`."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (used by exponential & GD stages)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) convolve p)  (central n entries)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """Search a 2‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n)."""
    p_vals = np.linspace(0.2, 1.5, 33)          # finer grid on p
    lam_vals = np.linspace(0.0, 12.0, 33)      # finer grid on λ
    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    rng = np.random.default_rng()
    # random refinements while time permits
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.5)
        lam = rng.exponential(1.0) * 4.0   # bias towards moderate λ
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    base_eta = 0.08               # slightly larger step
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)            # multiplicative update
        a = a * (target_sum / a.sum())      # re‑normalise

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – projected gradient descent with restarts (soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate (with momentum)."""
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(35.0, remaining * 0.35)   # a bit longer per restart
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 4 – deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from high‑contributing indices to low‑contributing ones."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx]
        weight = a[idx] * partner

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise for numerical drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 5 – greedy max‑reduction (new)
# ----------------------------------------------------------------------


def _greedy_max_reduce(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move mass from a high‑contributing donor to a low‑contributing
    receiver while guaranteeing that no convolution entry exceeds the current
    maximum.  The move size is the largest admissible delta computed analytically.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        k_max = int(conv.argmax())

        # donor candidates: indices that actually participate in the max entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        donors = np.arange(lo, hi + 1, dtype=np.int64)
        partners = k_max - donors
        weights = a[donors] * a[partners]

        if weights.size == 0:
            break

        # pick a few strongest donors
        top_idx = np.argsort(-weights)[:min(5, weights.size)]
        top_donors = donors[top_idx]

        # receiver candidates: indices that contribute little to the maximal entry
        contrib = np.zeros(n, dtype=np.float64)
        contrib[donors] = weights
        nonzero = contrib[contrib > 0]
        if nonzero.size > 0:
            thresh = np.percentile(nonzero, 15)
            low_cands = np.where((contrib <= thresh) & (contrib > 0))[0]
            if low_cands.size == 0:
                low_cands = np.where(contrib == 0)[0]
        else:
            low_cands = np.arange(n)

        # sample a handful of receivers
        if low_cands.size == 0:
            break
        rc_choices = rng.choice(low_cands,
                                size=min(5, low_cands.shape[0]),
                                replace=False)

        improved = False

        for i in top_donors:
            for j in rc_choices:
                if i == j:
                    continue
                # construct the shifted versions of a needed for the linear bound
                shift_i = np.concatenate([np.zeros(i), a, np.zeros(n - 1 - i)])
                shift_j = np.concatenate([np.zeros(j), a, np.zeros(n - 1 - j)])
                diff = shift_j - shift_i                     # contribution change per entry
                mask = diff > 1e-15
                if np.any(mask):
                    # admissible deltas so that no entry exceeds the current max
                    delta_cands = (M - conv[mask]) / (2.0 * diff[mask])
                    delta_cands = delta_cands[delta_cands > 0]
                    if delta_cands.size == 0:
                        continue
                    delta_max = min(delta_cands.min(), a[i])
                else:
                    delta_max = a[i]  # moving mass cannot increase any entry

                # limit the step size a bit for numerical stability
                delta_max = min(delta_max, a[i] * 0.5)

                if delta_max <= 1e-12:
                    continue

                a_new = a.copy()
                a_new[i] -= delta_max
                a_new[j] += delta_max
                a_new = a_new * (target_sum / a_new.sum())
                new_val = _score(a_new)

                if new_val < best_val - 1e-9:
                    best_val = new_val
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    print(f"[{time.time():.2f}] Greedy move i={i} j={j} δ={delta_max:.6e} new score={best_val:.6f}")
                    break
            if improved:
                break

        if not improved:
            # small random jitter to escape possible plateaus
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                delta = min(a[i] * 0.1, target_sum * 0.001)
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = a_new * (target_sum / a_new.sum())
                new_val = _score(a_new)
                if new_val < best_val:
                    best_val = new_val
                    best_a = a_new.copy()
                    a = a_new
                    print(f"[{time.time():.2f}] Random jitter i={i} j={j} δ={delta:.6e} new score={best_val:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on remaining budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (previous best or a simple power law)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # ------------------------------------------------------------------
    if time.time() < deadline:
        small_n = max(2000, n // 5)
        small_target = np.sqrt(2.0 * small_n)
        param_deadline = time.time() + min(25.0,
                                          (deadline - time.time()) * 0.10)
        best_small_seq, _ = _grid_param_search(small_n, small_target,
                                               param_deadline)
        if best_small_seq is not None:
            init = _resize_sequence(best_small_seq, n)
            init = init * (target_sum / init.sum())
            print(f"[{time.time():.2f}] Warm‑started from parametric search (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.13
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.33
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.18
        a_mt, val_mt = _mass_transfer_refine(best_seq,
                                             target_sum,
                                             mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – greedy max‑reduction (new stage)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gre_deadline = time.time() + (deadline - time.time()) * 0.24
        a_gre, val_gre = _greedy_max_reduce(best_seq,
                                            target_sum,
                                            gre_deadline)
        if val_gre < best_score:
            best_seq = a_gre
            best_score = val_gre
        print(f"[{time.time():.2f}] After greedy refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a future call
    globals()["height_sequence_1"] = final_seq
    return final_seq
