def _pairwise_hill_climb(a0, deadline, target_sum):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # choose epsilon as a small fraction of donor
        epsilon = min(a[i] * 0.1, a[i]*rng.random())
        if epsilon <= 0:

            # revert if no improvement?
            a[i] += epsilon
            a[j] -= epsilon
        # optionally also attempt random acceptance to avoid local minima?
    return best, best_val

"""
Step‑function minimizer for the benchmark objective.

The algorithm consists of four stages:

1. **Broad parametric search** – generate many candidate shapes
   (power‑law, power‑law + exponential tail, piecewise constant,
   truncated power‑law, mixtures) and keep the best according to a fast
   FFT based evaluation of 2·n·max(conv)/sum².

2. **Projected Adam‑gradient descent** – optimise the smooth soft‑max
   approximation of the max‑convolution.  The optimiser runs for a
   sizeable time budget, with an aggressive annealing of the soft‑max
   temperature (β) up to 10⁷ and Adam‑style adaptive steps, while each
   iterate is projected onto the simplex Σa = √(2 n), a ≥ 0.

3. **Exact deterministic mass‑transfer** – repeatedly move mass from
   the most harmful position to the most beneficial one so that the
   current maximal convolution entry is reduced to (approximately) the
   second largest one.  This phase strictly decreases the benchmark
   value.

4. **Final polishing** – a short high‑β Adam run and a brief random
   mass‑transfer sweep.

The code respects the overall time budget (default 1000 s), trims
trailing zeros and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark objective   2·n·max(a*a) / ( Σa )²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the initial search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) of length n, scaled to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p)·exp(-q·i/n) of length n, scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _piecewise_constant_sequence(n: int, m_frac: float, target_sum: float) -> np.ndarray:
    """First ⌈m_frac·n⌉ entries = 1, rest 0, then scaled."""
    m = max(1, int(m_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0
    a *= target_sum / a.sum()
    return a


def _truncated_power_law_sequence(n: int, p: float, L_frac: float, target_sum: float) -> np.ndarray:
    """Power‑law on the first ⌈L_frac·n⌉ entries, zeros afterwards."""
    L = max(1, int(L_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:L] = (np.arange(1, L + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _mixed_power_law_sequence(
    n: int, p1: float, p2: float, w: float, target_sum: float
) -> np.ndarray:
    """Convex combination w·i^{-p1} + (1‑w)·i^{-p2}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = w * (i ** (-p1)) + (1.0 - w) * (i ** (-p2))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam optimiser)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation using the soft‑max surrogate for the max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 200  # more frequent temperature increases

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # anneal β
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic exact mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves that directly reduce the currently maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]
        k0 = idx_max[0]  # work with the first maximal index

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[k0 - idx]
        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]
        denom = 2.0 * (contrib_d - contrib_r)

        if denom <= 0.0:
            # try a second maximal index if available
            if len(idx_max) > 1:
                k0 = idx_max[1]
                lo = max(0, k0 - (n - 1))
                hi = min(k0, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                w = a[idx] + a[k0 - idx]
                donor = idx[np.argmax(w)]
                receiver = idx[np.argmin(w)]
                contrib_d = a[k0 - donor]
                contrib_r = a[k0 - receiver]
                denom = 2.0 * (contrib_d - contrib_r)
                if denom <= 0.0:
                    break
            else:
                break

        # second largest convolution value (excluding all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.5, delta_needed * 0.99)

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing style moves: shift a small amount
    of mass from relatively large entries to relatively small ones.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the given time budget and returns the best
    sequence found (as a plain Python list).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # ---------------------------------------------------------------
    # Choose problem size based on the total budget
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 14000
    elif budget_s >= 500:
        n = 10000
    elif budget_s >= 300:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)          # Σa = √(2 n)

    # ---------------------------------------------------------------
    # Helper to keep the current best sequence
    # ---------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_score
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New best score = {best_score:.6f}")

    rng = np.random.default_rng()

    # ---------------------------------------------------------------
    # Phase 0 – deterministic grid search over several families
    # ---------------------------------------------------------------
    p_grid = np.linspace(0.20, 1.20, 25)
    q_grid = np.linspace(0.0, 10.0, 20)

    for p in p_grid:
        _maybe_update(_power_law_sequence(n, float(p), target_sum))

    for p in p_grid:
        for q in q_grid:
            _maybe_update(_exp_power_law_sequence(n, float(p), float(q), target_sum))

    # piecewise constant templates
    for m_frac in np.linspace(0.05, 0.50, 10):
        _maybe_update(_piecewise_constant_sequence(n, m_frac, target_sum))

    # truncated power‑law templates
    for p in p_grid[::4]:
        for L_frac in np.linspace(0.20, 1.00, 8):
            _maybe_update(_truncated_power_law_sequence(n, float(p), L_frac, target_sum))

    # ---------------------------------------------------------------
    # Phase 0b – random exploration of the parametric space
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    rnd_budget = min(200.0, remaining * 0.25)   # reserve at most 25 % of what's left
    rnd_deadline = time.time() + rnd_budget

    while time.time() < rnd_deadline:
        # mixed power‑law
        p1 = rng.uniform(0.20, 1.20)
        p2 = rng.uniform(0.20, 1.20)
        w = rng.uniform(0.0, 1.0)
        _maybe_update(_mixed_power_law_sequence(n, p1, p2, w, target_sum))

        # exponential‑power‑law with wider q range
        p = rng.uniform(0.20, 1.20)
        q = rng.uniform(0.0, 12.0)
        _maybe_update(_exp_power_law_sequence(n, p, q, target_sum))

    # ---------------------------------------------------------------
    # Phase 0c – reuse previous best if available
    # ---------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        _maybe_update(prev)

    # fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)

    # ---------------------------------------------------------------
    # Phase A – Adam‑style gradient descent with restarts (long budget)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.45  # ~45 % of the total time

    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        run_budget = min(40.0, (gd_deadline - time.time()) * 0.4)
        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            time.time() + run_budget,
            target_sum,
            max_beta=5e6,
            step_factor=0.6,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer + short random phase
    # ---------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ---------------------------------------------------------------
    # Phase C – Final high‑β polishing
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e7,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zeros and return a list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final verification (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible future rounds
    globals()["height_sequence_1"] = final_seq

    return final_seq
