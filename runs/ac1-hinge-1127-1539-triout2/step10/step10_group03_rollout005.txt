"""
Hybrid optimizer for the self‑convolution bound.

The algorithm consists of several stages:

1. **Exploration (small n ≈ 4‑5k).**  
   Many random starts (power‑law, exponential, Dirichlet) are refined
   with an annealed projected gradient descent on the soft‑max
   approximation of the convolution peak.

2. **Optional mass‑transfer refinement** on the small problem.

3. **Upscaling** to a larger length (up to 15 000).  The best low‑dimensional
   solution is interpolated, projected onto the ℓ¹‑simplex and a few simple
   reorderings are tried.

4. **Exact sub‑gradient descent** on the true max‑convolution (no
   soft‑max) to fine‑tune the large problem.

5. **Swap‑annealing** – a simulated‑annealing phase that swaps pairs of
   entries to flatten the autocorrelation further.

6. **Trimming** of trailing zeros and final reporting.

The routine respects all constraints of `evaluate_sequence` and aims to
beat the previous best bound (≈ 1.508) within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max based gradient descent (used in the exploratory phase)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 1e5,
                                   step_factor: float = 0.7,
                                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Deterministic mass‑transfer refinement (small scale)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, second largest, … alternating left–right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution (no soft‑max).
    Uses the fact that ∂ max_b / ∂ a_j = 2 * a_{k* - j} where k* is a
    maximizing index.  Simple step–size schedule with projection onto the
    simplex.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient
        n = a.shape[0]
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        # indices j = lo..hi correspond to a_idx = kmax - j
        grad[lo:hi + 1] = 2.0 * a[kmax - np.arange(lo, hi + 1)]

        norm = np.linalg.norm(grad)
        if norm > 0:
            # step size inversely proportional to norm; the factor 0.4 was
            # tuned empirically.
            lr = 0.4 * target_sum / norm
            a -= lr * grad

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Swap‑annealing (simulated annealing on permutations)
# ----------------------------------------------------------------------


def _swap_annealing(a0: np.ndarray,
                    deadline: float,
                    target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that swaps pairs of entries.
    The sum is preserved automatically; after each swap we project onto
    the simplex to keep numerical errors in check.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    # start temperature proportional to current score
    start_temp = max(best_score * 0.5, 1e-3)
    end_temp = 1e-5

    while time.time() < deadline:
        # linear cooling schedule
        progress = (time.time() - start) / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** progress

        i, j = rng.choice(n, size=2, replace=False)
        if a[i] == a[j]:
            continue
        a[i], a[j] = a[j], a[i]

        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SwapAnneal new best={best_score:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]
            # keep a projected (tiny drift may appear)
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to avoid stagnation
        if (time.time() - start) > 0.1 and int(progress * 1000) % 47 == 0:
            jitter = rng.normal(scale=target_sum * 0.0003, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_score = _score(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The method combines multi‑start soft‑max gradient descent,
    deterministic mass‑transfer, up‑scaling, exact sub‑gradient descent,
    and a final swap‑annealing pass.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Small‑scale exploration parameters
    # --------------------------------------------------------------
    n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    init_pool = []

    # power‑law
    for p in np.linspace(0.30, 0.80, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential
    for lam in np.linspace(0.001, 0.020, 7):
        a = np.exp(-lam * np.arange(n_expl, dtype=float))
        a *= target_expl / a.sum()
        init_pool.append(a)

    # random Dirichlet‑like
    for _ in range(6):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jitter copies for diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # --------------------------------------------------------------
    # Exploration with short gradient‑descent runs
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 15.0:
        # pick a seed
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.003, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 25.0, deadline - 5.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent_ext(
            seed,
            run_dead,
            target_expl,
            max_beta=5e5,
            step_factor=0.6,
            gamma=0.98,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Small‑scale deterministic mass‑transfer (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 12.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.18
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.5, end_temp=1e-4, low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Upscaling to a larger problem size
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 70:
        n_final = 15000
    elif remaining > 35:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few simple reorderings
    candidates = [seq_big, _alternating_order(seq_big)]
    for _ in range(3):
        perm = rng.permutation(seq_big)
        candidates.append(_project_to_simplex(perm, target_final))

    best_seq = candidates[0]
    best_score = _score(best_seq)
    for cand in candidates[1:]:
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand

    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # --------------------------------------------------------------
    # Exact sub‑gradient descent on the large problem
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            deadline - 4.0,
            target_final,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Swap‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_swap, val_swap = _swap_annealing(
            best_seq,
            deadline - 2.0,
            target_final,
        )
        if val_swap < best_score:
            best_score = val_swap
            best_seq = a_swap.copy()
            print(f"[{time.time():.2f}] Swap‑anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing zeros and final bookkeeping
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # expose for possible reuse in a later run
    globals()["height_sequence_1"] = final_seq

    # final log (optional evaluation)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
