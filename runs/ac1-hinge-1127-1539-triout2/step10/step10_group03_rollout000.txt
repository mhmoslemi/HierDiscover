"""
Hybrid sub‑gradient / mass‑transfer optimizer.

The evaluation objective for a non‑negative vector a of length n with
∑a = √(2n) reduces to minimizing the peak of its self‑convolution
(max_{k} (a*a)[k]).  The program performs three stages:

1. **Exploration** – many short runs of a projected sub‑gradient
   descent on the exact max‑convolution (no soft‑max).  Random
   power‑law, exponential and Dirichlet starts are used; the best
   candidate is kept.

2. **Upscaling + polishing** – the best low‑dimensional solution is
   linearly interpolated to a larger size (up to ~ 15000 entries)
   and refined with a longer sub‑gradient run followed by a brief
   deterministic mass‑transfer pass.  An alternating large/small
   re‑ordering is tried because it often lowers the peak.

3. **Final touch** – a short extra sub‑gradient run and a final
   mass‑transfer refinement.

All updates respect the ℓ¹‑simplex (non‑negativity + fixed sum) by
projecting after each step.  The routine stops before the time budget
expires and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Create an exponential decay e^{-lam*i} and renormalise."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder by placing the largest value leftmost, the second largest
    rightmost, the third largest next to the left, etc.
    """
    sorted_vals = np.sort(seq)[::-1]  # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact, no smoothing)
# ----------------------------------------------------------------------


def _subgradient(a: np.ndarray, eps: float = 1e-12):
    """
    Compute a sub‑gradient of the function max_k (a*a)[k] w.r.t. a.
    Returns (grad, current_max), where grad has shape a.shape.
    """
    n = a.shape[0]
    conv = _conv_fft(a)                     # length 2n‑1
    max_b = conv.max()
    mask = (conv >= max_b - eps).astype(np.float64)   # binary mask
    # Gradient = 2 * (mask * a_rev) convolution, sliced to length n
    a_rev = a[::-1]
    grad_full = np.convolve(mask, a_rev)    # length 3n‑2
    grad = 2.0 * grad_full[n - 1: n - 1 + n]  # central n entries
    return grad, max_b


# ----------------------------------------------------------------------
# Projected (momentum) sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.4,
                         momentum_coeff: float = 0.9,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Runs a projected sub‑gradient descent with momentum.
    Returns the best vector found and its corresponding max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    _, best_val = _subgradient(a, eps=eps)

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, cur_max = _subgradient(a, eps=eps)

        # momentum update
        momentum = momentum_coeff * momentum + grad

        # step size inversely proportional to gradient norm
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_factor * target_sum / (norm * (it + 1) ** 0.5)
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (it + 1), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # evaluate
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] sub‑grad it={it:5d} max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (same as previous implementation)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’ (large
    contribution to the current maximal convolution entry) to a “receiver’’
    (small contribution). Moves are accepted with a Metropolis rule whose
    temperature decays exponentially with elapsed time.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)
    cur_score = conv.max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        # feasible index range for donors (i, j) with i+j=kmax
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight contributed by each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # choose a low‑weight receiver among the smallest ``low_frac`` entries
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bound the amount of mass we move
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = conv.max()

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = conv.max()

        it += 1

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine proceeds through exploration, up‑scaling, and final polishing,
    all under the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Exploration phase (moderate size)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 400:
        n_expl = 5000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # Build a diverse pool of initial candidates
    init_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.80, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.002, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like points
    for _ in range(8):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to enlarge diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.012, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_val = np.inf

    # Run many short sub‑gradient runs
    while time.time() < deadline - 6.0:
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.006, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 18.0, deadline - 4.0)
        a_opt, val_opt = _subgradient_descent(
            seed,
            run_dead,
            target_expl,
            step_factor=0.45,
            momentum_coeff=0.92,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best max={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Short deterministic mass‑transfer refinement (small scale)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.15
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.6, end_temp=1e-4, low_frac=0.06
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] MT (small) new best max={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Upscaling to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # Decide final length based on remaining time
    remaining = deadline - time.time()
    if remaining > 150.0:
        n_final = 15000
    elif remaining > 80.0:
        n_final = 12000
    elif remaining > 40.0:
        n_final = 9000
    else:
        n_final = n_expl   # not enough time to upscale

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Try alternating large/small ordering, keep it if it improves the peak
    alt = _alternating_order(seq_big)
    if _max_convolution_fft(alt) < _max_convolution_fft(seq_big):
        seq_big = alt

    best_seq = seq_big
    best_val = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] Upscaled length={n_final} start max={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long sub‑gradient polishing on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_final, val_final = _subgradient_descent(
            best_seq,
            deadline,
            target_final,
            step_factor=0.33,
            momentum_coeff=0.93,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final sub‑grad best max={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final deterministic mass‑transfer pass (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(9.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_final,
            start_temp=0.3, end_temp=1e-5, low_frac=0.05
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best max={best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (just for logging)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
