"""
Propose a non‑negative step function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm works in several stages within the given wall‑clock budget.

1. **Rich parameter sweep** – Exhaustively scans several one‑parameter families
   (power‑law, exponential, edge‑heavy, truncated exponential, Gaussian,
   sparse Dirichlet) and keeps the best candidate according to the exact
   max‑convolution (FFT).

2. **Genetic optimisation** – Maintains a small population of vectors,
   repeatedly selects elite individuals, creates children by averaging and
   mutating them, projects back onto the ℓ¹‑simplex and keeps the best seen.
   This explores the non‑convex landscape much more diversely than the
   single‑restart Adam used previously.

3. **Projected Adam** – A longer Adam run on the smooth log‑sum‑exp surrogate
   of the max‑convolution, starting from the best vector obtained so far.

4. **Exact deterministic hill‑climb** – Greedy moves that always attack the
   current worst convolution entry using the exact update formula.

5. **Simulated‑annealing fine‑tuning** – Random mass‑transfer moves with a
   decreasing temperature schedule, again using the exact O(n) update of the
   convolution.

6. **Polish & clean‑up** – Final projection to the ℓ¹‑simplex, removal of
   trailing near‑zeros and a call to ``evaluate_sequence`` for reporting.

All steps respect the overall time limit (default 1000 s).  The code avoids
closures and uses only NumPy and the standard library, making it compatible
with the execution environment.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a (full convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # compute g = 2 * (a ⋆ p) (restricted to length n)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until ``deadline`` and returns the final vector.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.12 * target_sum          # a bit smaller than in the reference
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing parameter
        if t % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Random initial families (used by the sweep and the GA)

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay (origin at 0)
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # edge‑heavy (large near both ends)
        lam = rng.uniform(0.001, 0.02)
        d = np.minimum(idx, n - 1 - idx)
        a = np.exp(-lam * d)

    elif family == 5:                   # truncated exponential
        lam = rng.exponential(scale=0.015)
        cutoff = rng.integers(int(0.2 * n), int(0.8 * n))
        a = np.exp(-lam * np.arange(cutoff))
        a = np.concatenate([a, np.zeros(n - cutoff)])

    elif family == 6:                   # Gaussian bump
        mu = rng.uniform(0, n - 1)
        sigma = rng.exponential(scale=n / 10.0) + 1e-3
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    else:                               # sparse Dirichlet (α < 1)
        alpha = rng.uniform(0.01, 0.5)
        a = rng.dirichlet(alpha=np.full(n, alpha)) * target

    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Shape that is large near the two edges and decays towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves mass from the worst convolution
    entry to a better destination.  Returns the best vector seen and its
    maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # Choose a destination j that pairs with a small counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # back‑off line‑search
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (exact conv updates)
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing with a geometric temperature schedule.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9999

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.5)
        if delta <= 0.0:
            continue

        # O(n) convolution update
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5

        # occasional re‑heat to escape deep plateaus
        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Enriched parameter sweep (includes extra families)
# ----------------------------------------------------------------------


def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan several one‑parameter families (including the new truncated, Gaussian
    and sparse Dirichlet) and return the best vector seen (exact max‑conv).
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law
    for p in np.linspace(0.1, 2.0, 40):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponential
    for lam in np.linspace(0.0005, 0.02, 40):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 16):
        for lam in np.linspace(0.001, 0.01, 8):
            a = (idx ** (-p)) * np.exp(-lam * (lin - 0))
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # truncated exponential
    for lam in np.linspace(0.001, 0.03, 15):
        for cutoff_frac in np.linspace(0.2, 0.8, 9):
            cutoff = int(cutoff_frac * n)
            a = np.exp(-lam * np.arange(cutoff))
            a = np.concatenate([a, np.zeros(n - cutoff)])
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # Gaussian bump (off‑centre)
    for sigma in np.linspace(0.05, 0.4, 12):
        for mu_frac in np.linspace(0.0, 1.0, 8):
            mu = mu_frac * (n - 1)
            a = np.exp(-0.5 * ((lin - mu) / (sigma * n)) ** 2)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # Sparse Dirichlet (α<1)
    for alpha in np.linspace(0.02, 0.4, 12):
        a = rng.dirichlet(alpha=np.full(n, alpha)) * target
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple genetic algorithm (population‑based search)
# ----------------------------------------------------------------------


def _genetic_algorithm(initial_seq: np.ndarray, target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Small population genetic optimisation.  Returns the best individual and
    its exact max‑convolution value found before ``deadline``.
    """
    n = initial_seq.size
    rng = np.random.default_rng()

    # initialise a diverse population (size 16)
    pop = [initial_seq.copy()]
    for _ in range(7):
        pop.append(_random_initial_sequence(n, rng))
    for _ in range(7):
        pop.append(_edge_high_initial_sequence(n, rng))

    best_seq = None
    best_score = np.inf

    while time.time() < deadline:
        # evaluate current population
        scores = [_max_convolution_fft(ind) for ind in pop]

        # update global best
        for seq, sc in zip(pop, scores):
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

        # select elite (top 4)
        elite_idx = np.argsort(scores)[:4]
        elites = [pop[i] for i in elite_idx]

        # create next generation
        new_pop = elites.copy()
        while len(new_pop) < len(pop):
            p1, p2 = rng.choice(elites, size=2, replace=False)
            child = 0.5 * (p1 + p2)

            # mutation
            if rng.random() < 0.7:
                child += rng.normal(scale=target_sum * 1e-3, size=n)
                child = np.maximum(child, 0.0)

            child = _project_to_simplex(child, target_sum)
            new_pop.append(child)

        pop = new_pop

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optional seed from a previously computed global variable
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded (n={n0}) → score={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # --------------------------------------------------------------
    # Fixed problem size (2000) – enough resolution and still fast
    # --------------------------------------------------------------
    n = 2000
    target = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 0) Parameter sweep (cheap)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_sweep, score_sweep = _parameter_sweep(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep improvement → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Genetic optimisation (diverse search)
    # --------------------------------------------------------------
    if best_seq is None:
        # fall back to a random start if sweep returned nothing
        best_seq = _random_initial_sequence(n, rng)
        best_score = _max_convolution_fft(best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:
        ga_deadline = time.time() + remaining * 0.45
        a_ga, s_ga = _genetic_algorithm(best_seq, target, ga_deadline)
        if s_ga < best_score - 1e-12:
            best_seq, best_score = a_ga, s_ga
            print(f"[{time.time():.1f}] GA improvement → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Long Adam optimisation on the smooth surrogate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        adam_dead = time.time() + remaining * 0.35
        a_adam = _adam_optimize(best_seq, target, adam_dead)
        s_adam = _max_convolution_fft(a_adam)
        if s_adam < best_score - 1e-12:
            best_seq, best_score = a_adam, s_adam
            print(f"[{time.time():.1f}] Adam fine‑tune → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Deterministic hill‑climb (exact max)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur, deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq, best_score = a_hc, s_hc
            print(f"[{time.time():.1f}] Hill‑climb → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Simulated‑annealing refinement (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq, best_score = a_ann, s_ann
            print(f"[{time.time():.1f}] Anneal → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Clean‑up: drop trailing near‑zeros
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 6) Final reporting using the provided evaluate_sequence
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
