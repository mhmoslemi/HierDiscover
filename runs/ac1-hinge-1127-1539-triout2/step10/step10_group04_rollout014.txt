def _isotonic_decreasing(x):
    # return y that is nonincreasing close to x in L2 sense.
    n = len(x)
    y = x.astype(float).copy()
    # we'll use PAVA for decreasing: iterate merging blocks where y[i] < y[i+1], and replace them with average.
    # Represent block values and sizes.
    avg = []
    weight = []
    for val in y:
        avg.append(val)
        weight.append(1)
        # Merge while monotonic decreasing violated: last avg < second last avg.
        while len(avg) >= 2 and avg[-2] < avg[-1]:
            # Merge last two
            total_w = weight[-2] + weight[-1]
            merged_avg = (avg[-2] * weight[-2] + avg[-1] * weight[-1]) / total_w
            avg[-2] = merged_avg
            weight[-2] = total_w
            avg.pop()
            weight.pop()
    # Now expand average values to lengths
    y_out = np.empty(n, dtype=float)
    idx = 0
    for val, w in zip(avg, weight):
        y_out[idx: idx+w] = val
        idx += w
    return y_out
```

def _adam_optimize_monotone(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.15*target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t=0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = base_lr * m_hat/(np.sqrt(v_hat)+eps)
        a -= step
        # enforce non-negativity
        a = np.maximum(a, 0.0)
        # monotonic decreasing projection
        a = _isotonic_decreasing(a)
        # enforce sum constraint via scaling
        s = np.sum(a)
        if s > 0:
            a *= target_sum / s
        else:
            # fallback uniform
            a = np.full(n, target_sum/n)
        # jitter
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta+1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _isotonic_decreasing(a)
            s = np.sum(a)
            a *= target_sum / s
        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta*beta_factor, beta_max)
    return a
```

"""
propose_candidate
-----------------
A multi‑stage optimiser for the step‑function problem.

Stage 1 (n≈2000) : cheap family scan → monotone Adam → exact hill‑climb.
Stage 2 (larger n) : up‑sample the current best, run a longer monotone
Adam optimisation (smooth‑max surrogate) and keep any improvement.
Final polish : one more Adam run and a short hill‑climb.

All Adam steps enforce a non‑increasing shape by isotonic regression,
which is known to be optimal for the rearrangement inequality.
The implementation respects the 1000 s wall‑clock budget and returns
the best sequence found.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Isotonic regression (non‑increasing) – PAVA
# ----------------------------------------------------------------------


def _isotonic_decreasing(x: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the cone {y_i ≥ y_{i+1}}.
    Returns the monotone decreasing vector closest to x in L2 norm.
    """
    n = x.shape[0]
    # blocks: each block stores its average and its weight (length)
    avg = []
    weight = []
    for val in x:
        avg.append(val)
        weight.append(1)
        # while the monotonicity constraint is violated, merge the last two blocks
        while len(avg) >= 2 and avg[-2] < avg[-1]:
            tot_w = weight[-2] + weight[-1]
            merged = (avg[-2] * weight[-2] + avg[-1] * weight[-1]) / tot_w
            avg[-2] = merged
            weight[-2] = tot_w
            avg.pop()
            weight.pop()
    # expand back to a full vector
    y = np.empty(n, dtype=float)
    pos = 0
    for a, w in zip(avg, weight):
        y[pos: pos + w] = a
        pos += w
    return y


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize_monotone(a0: np.ndarray, target_sum: float,
                            deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    After each Adam step the vector is projected onto the cone of
    non‑increasing vectors and then renormalised to the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.15 * target_sum

    # smoothed‑max schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the smooth‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step

        # enforce non‑negativity
        a = np.maximum(a, 0.0)

        # monotone‑decreasing projection (PAVA)
        a = _isotonic_decreasing(a)

        # renormalise to the ℓ¹‑simplex (preserves monotonicity)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:
            a = np.full(n, target_sum / n)

        # occasional jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _isotonic_decreasing(a)
            s = a.sum()
            if s > 0:
                a *= target_sum / s

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisation (monotone families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative monotone decreasing vector on the ℓ¹‑simplex."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.5)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # piecewise exponential
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    # small random jitter to avoid pathological ties
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)

    # enforce monotonicity then renormalise to the simplex
    a = _isotonic_decreasing(a)
    a = a * target / a.sum()
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by monotone projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new = _isotonic_decreasing(a_new)
    target = math.sqrt(2.0 * new_n)
    a_new = a_new * target / a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution
    entry. Moves a (large) fraction of the responsible mass to a more
    favourable location. Returns the best vector seen and its maximal
    convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # choose destination with minimal counterpart (but not i)
        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # linear convolution update (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search style)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _isotonic_decreasing(a)
                a = a * a.sum() / a.sum()   # renormalise trivially
                b = _full_convolution(a)

    return best_a, best_score

def _family_candidates(n: int, rng, top_k: int = 8):
    """
    Scan several simple families (exponential, power‑law, mixed,
    piecewise, near‑uniform) and keep the best ``top_k`` candidates.
    Returns a list of (score, vector) pairs.
    """
    target = math.sqrt(2.0 * n)
    candidates = []

    # exponential decay
    for _ in range(30):
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = _isotonic_decreasing(a)
        a = a * target / a.sum()
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # power‑law
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
        a = _isotonic_decreasing(a)
        a = a * target / a.sum()
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # mixed power‑law × exponential
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p) * np.exp(-lam * np.arange(n))
        a = _isotonic_decreasing(a)
        a = a * target / a.sum()
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # piecewise exponential
    for _ in range(10):
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
        a = _isotonic_decreasing(a)
        a = a * target / a.sum()
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # near‑uniform with tiny jitter
    a = np.full(n, target / n)
    a += rng.normal(scale=target * 0.0005, size=n)
    a = np.maximum(a, 0.0)
    a = _isotonic_decreasing(a)
    a = a * target / a.sum()
    score = _max_convolution_fft(a)
    candidates.append((score, a))

    # keep best
    candidates.sort(key=itemgetter(0))
    return candidates[:top_k]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The algorithm proceeds in stages of increasing resolution and uses
    monotone Adam optimisation together with deterministic hill‑climbing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Stage 1 – modest dimension (n≈2000)
    # --------------------------------------------------------------
    n0 = 2000
    target0 = math.sqrt(2.0 * n0)

    # 1a) families
    cand = _family_candidates(n0, rng, top_k=10)

    # 1b) ensure the best candidate is used as seed
    cand.sort(key=itemgetter(0))
    a_cur = cand[0][1]
    best_seq = a_cur.copy()
    best_score = cand[0][0]
    print(f"[{time.time():.1f}] Stage 1 initial score n={n0} → {best_score:.6f}")

    # 1c) monotone Adam (allocate ~20 % of remaining budget)
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + min(remaining * 0.2, 30.0)
        a_opt = _adam_optimize_monotone(a_cur, target0, adam_deadline)
        score_opt = _max_convolution_fft(a_opt)
        if score_opt < best_score:
            best_seq = a_opt.copy()
            best_score = score_opt
            print(f"[{time.time():.1f}] Adam improvement (n={n0}) score={best_score:.6f}")

    # 1d) deterministic hill‑climb (if time allows)
    if time.time() < deadline - 5.0:
        b_cur = _full_convolution(best_seq)
        hc_deadline = deadline - 0.5
        a_hc, hc_score = _hill_climb_deterministic(best_seq, b_cur, hc_deadline)
        if hc_score < best_score:
            best_seq = a_hc.copy()
            best_score = hc_score
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n0}) score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – up‑sample to larger dimensions
    # --------------------------------------------------------------
    upsample_sizes = [8000, 20000, 50000]

    for n in upsample_sizes:
        if time.time() >= deadline - 5.0:
            break
        target = math.sqrt(2.0 * n)

        # up‑sample current best (preserves monotonicity)
        a_big = _upsample_sequence(best_seq, n)

        # quick evaluation
        score_big = _max_convolution_fft(a_big)
        print(f"[{time.time():.1f}] Upsampled to n={n}, init score={score_big:.6f}")

        # Adam optimisation on the large instance
        remaining = deadline - time.time()
        if remaining < 20.0:
            break
        # give each large instance roughly a third of the remaining time,
        # but not more than 120 s so we keep time for later polishing.
        adam_deadline = time.time() + min(remaining / 3.0, 120.0)
        a_big_opt = _adam_optimize_monotone(a_big, target, adam_deadline)
        score_big_opt = _max_convolution_fft(a_big_opt)
        print(f"[{time.time():.1f}] Adam after upsample n={n} → {score_big_opt:.6f}")

        if score_big_opt < best_score:
            best_seq = a_big_opt.copy()
            best_score = score_big_opt
            print(f"[{time.time():.1f}] New global best after upsample n={n} score={best_score:.6f}")

        # a very short hill‑climb on the large instance (optional, cheap)
        if time.time() < deadline - 3.0:
            b_big = _full_convolution(best_seq)
            hc_deadline = time.time() + 5.0
            a_hc, hc_score = _hill_climb_deterministic(best_seq, b_big, hc_deadline)
            if hc_score < best_score:
                best_seq = a_hc.copy()
                best_score = hc_score
                print(f"[{time.time():.1f}] Hill‑climb on large n={n} → {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final polishing on the current best size
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n_final = best_seq.size
        target_final = math.sqrt(2.0 * n_final)
        final_deadline = deadline - 0.5
        best_seq = _adam_optimize_monotone(best_seq, target_final, final_deadline)
        b_final = _full_convolution(best_seq)
        a_hc, final_score = _hill_climb_deterministic(best_seq, b_final,
                                                      deadline - 0.5)
        if final_score < best_score:
            best_seq = a_hc.copy()
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement → {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # Official evaluation
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
