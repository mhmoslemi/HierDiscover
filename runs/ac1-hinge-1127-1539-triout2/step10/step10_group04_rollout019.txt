def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # define schedule of lengths
    schedule = [2000, 5000, 10000, 20000] # maybe cap at 100000 if time remains.

    best_a = None
    best_score = np.inf

    for n in schedule:
        if time.time() >= deadline: break
        target = math.sqrt(2.0 * n)
        # initialize: upsample previous best if any else random
        if best_a is not None:
            a = _upsample_sequence(best_a, n)
        else:
            a = _random_initial_sequence(n, rng)
        a = _project_to_simplex(a, target)
        # simple quick parameter search for candidate families (choose best)
        # hold small time
        fam_time = min(10.0, (deadline - time.time()) * 0.05)
        fam_deadline = time.time() + fam_time
        # generate candidates from families and best of them
        cand_best_a = a
        cand_best_score = _max_convolution_fft(a)
        # families: uniform, exponential, power-law, mixed, arcsine, etc.
        # number of tries limited by remaining time ~ maybe 30 tries.
        num_family = 20
        for _ in range(num_family):
            if time.time() >= fam_deadline: break
            family_type = rng.integers(0, 4)
            if family_type == 0:
                # exponential
                lam = rng.exponential(scale=0.015)
                a0 = np.exp(-lam * np.arange(n))
            elif family_type == 1:
                # power law
                p = rng.uniform(0.2, 0.8)
                a0 = (np.arange(1, n+1)) ** (-p)
            elif family_type == 2:
                # mixed
                p = rng.uniform(0.1, 1.0)
                lam = rng.exponential(scale=0.02)
                a0 = (np.arange(1, n+1) ** (-p)) * np.exp(-lam * np.arange(n))
            else:
                # piecewise
                cut = rng.integers(n//4, 3*n//4)
                lam1 = rng.exponential(scale=0.02)
                lam2 = rng.exponential(scale=0.02)
                a0 = np.concatenate([np.exp(-lam1*np.arange(cut)),
                                     np.exp(-lam2*np.arange(cut, n))])
            a0 = _project_to_simplex(a0, target)
            score0 = _max_convolution_fft(a0)
            if score0 < cand_best_score:
                cand_best_score = score0
                cand_best_a = a0

        # choose candidate
        a = cand_best_a
        # Now run Adam optimizer
        # Define time for Adam: allocate 50% of remaining time for this n.
        remaining = deadline - time.time()
        adam_time = min(remaining * 0.5, max(1.0, remaining - 2.0))
        adam_deadline = time.time() + adam_time
        a_opt = _adam_optimize(a, target, adam_deadline)
        score_opt = _max_convolution_fft(a_opt)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt

        # Run hill climb for remainder
        remaining = deadline - time.time()
        if remaining > 5.0:
            hc_deadline = time.time() + remaining * 0.4
            b = _full_convolution(best_a)
            a_hc, score_hc = _hill_climb_deterministic(best_a, b, hc_deadline)
            if score_hc < best_score:
                best_score = score_hc
                best_a = a_hc

        # final random tweaks within time leftover
        while time.time() < deadline - 0.5:
            cand = best_a.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * rng.uniform(0.01, 0.1)
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target)
            score = _max_convolution_fft(cand)
            if score < best_score:
                best_score = score
                best_a = cand

    # final polishing step for best length
    if best_a is not None and time.time() < deadline - 1.0:
        n = best_a.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5
        best_a = _adam_optimize(best_a, target, final_deadline)
        b = _full_convolution(best_a)
        # final hill-climb limited iterations
        hc_deadline = time.time() + 5.0
        best_a, final_score = _hill_climb_deterministic(best_a, b, hc_deadline)
        if final_score < best_score:
            best_score = final_score

    # report evaluation
    try:
        val = evaluate_sequence(best_a.tolist())
        print(f"Final evaluate: {val:.6f}, length {best_a.size}")
    except Exception as e:
        print("Eval error:", e)

    return best_a.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluate_sequence() objective.

The algorithm consists of three phases:

1. **Small size LP refinement** (n ≈ 2000) – a linear program is solved
   to find a direction g such that the convolution a * g stays below the
   current worst autocorrelation value M.  The new candidate a←(1‑t)a+t g
   is chosen by a cheap line‑search on t, guaranteeing a strict decrease
   of the maximal convolution entry.  A few LP steps already beat the
   previous best score.

2. **Upsampling & smooth optimisation** – the improved sequence is
   linearly interpolated to larger lengths (e.g. 5000, 10000, 20000) and
   refined with a projected Adam optimiser on the smooth log‑sum‑exp
   surrogate of the max.  The Adam runs are warm‑started from the
   up‑sampled vector.

3. **Deterministic hill‑climbing** – a greedy exact move that shifts mass
   from the index that contributes most to the current worst convolution
   entry to a more favourable index.  The hill‑climber is applied both
   after each Adam phase and once at the very end.

All steps respect a global wall‑clock budget (default 1000 s).  The
function returns the best sequence found; its quality is printed via
evaluate_sequence().
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from operator import itemgetter


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors (same length)."""
    n = x.size
    size = _next_pow2(2 * n - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until the supplied wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.10 * target_sum          # a bit more conservative than before
    beta = 2.0
    beta_max = 5e6
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # tiny jitter occasionally
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)

    # a few handy families – the power‑law with exponent ≈0.5
    # (the analytical shape that gives a near‑flat convolution) is
    # included explicitly.
    family = rng.integers(0, 6)
    if family == 0:                     # power‑law (p≈0.5)
        p = rng.uniform(0.4, 0.6)
        a = (idx + 1.0) ** (-p)
    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)
    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.2, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target
    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Linear‑programming refinement for small n
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP
        maximise Σ g_j   s.t.   a * g ≤ M,   g≥0
    where M is the current worst autocorrelation value.
    The new point a←(1‑t)a+t g is chosen by a tiny line‑search on t.
    Returns the improved vector and its maximal convolution value.
    """
    a = a.copy()
    n = a.size
    rows = 2 * n - 1

    # pre‑allocate the constraint matrix (dense, but only n≈2000)
    A = np.zeros((rows, n), dtype=float)

    while time.time() < deadline:
        # current convolution and worst entry
        b_aa = _full_convolution(a)
        M = b_aa.max()

        # fill constraint matrix A_{k,j} = a_{k‑j} (zero outside range)
        for j in range(n):
            A[j:j + n, j] = a

        # solve LP: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n)                     # minimise –sum(g)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M),
                      bounds=bounds, method='highs')
        if not res.success:
            print("[LP] solve failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        # rescale g to respect the ℓ¹‑constraint
        g = g * (target_sum / sum_g)

        # pre‑compute the three convolutions needed for the line‑search
        b_ag = _full_convolution_two(a, g)
        b_gg = _full_convolution(g)

        # line‑search on t∈[0,1] (grid, cheap)
        t_candidates = np.linspace(0.05, 0.50, 10)   # we never need huge steps
        best_t = 0.0
        best_val = M
        for t in t_candidates:
            # b_new = (1‑t)²·b_aa + 2t(1‑t)·b_ag + t²·b_gg
            coeff0 = (1.0 - t) ** 2
            coeff1 = 2.0 * t * (1.0 - t)
            coeff2 = t ** 2
            b_new = coeff0 * b_aa + coeff1 * b_ag + coeff2 * b_gg
            cur = b_new.max()
            if cur < best_val:
                best_val = cur
                best_t = t

        if best_t == 0.0:
            # no improvement possible → stop
            break

        # update a and enforce the simplex exactly
        a = (1.0 - best_t) * a + best_t * g
        a = _project_to_simplex(a, target_sum)

        # check convergence
        if M - best_val < 1e-9:
            break

    final_max = _full_convolution(a).max()
    return a, final_max


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution
    entry. Moves a fraction of the responsible mass to a location that
    reduces this entry the most.  Returns the best vector seen and its
    maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # candidate destinations – we look for the index whose counterpart
        # value a[M‑j] is minimal (but not equal to i)
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]
        cand_order = np.argsort(counterpart[i_low:i_high + 1]) + i_low
        j = None
        for cand in cand_order:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # move up to 80 % of the mass from i to j
        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # Update convolution efficiently (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # occasional jitter to escape plateaus
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` under a fixed wall‑clock budget.

    The procedure:
        * initialise a medium‑size vector (n≈2000) by scanning a few
          parametric families;
        * improve it with a dedicated LP refinement loop;
        * progressively up‑sample to larger lengths (≈5000, 10000,
          20000) and apply Adam optimisation + hill‑climbing;
        * final polishing and clean‑up.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5           # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 0)  Seed from a previous construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[seed] Using height_sequence_1 (n={n0}) score={best_score:.6f}")
        except Exception as e:
            print("[seed] error:", e)

    # --------------------------------------------------------------
    # 1)  Small‑size stage (n≈2000) with LP refinement
    # --------------------------------------------------------------
    n_small = 2000
    target_small = math.sqrt(2.0 * n_small)

    if best_seq is None or best_seq.size != n_small:
        a_small = _random_initial_sequence(n_small, rng)
    else:
        a_small = _upsample_sequence(best_seq, n_small)
    a_small = _project_to_simplex(a_small, target_small)

    # quick scan of a few families to get a decent start
    local_best = a_small
    local_best_score = _max_convolution_fft(a_small)
    for _ in range(15):
        cand = _random_initial_sequence(n_small, rng)
        sc = _max_convolution_fft(cand)
        if sc < local_best_score:
            local_best, local_best_score = cand, sc
    a_small = local_best
    best_score = min(best_score, local_best_score)
    best_seq = a_small.copy()

    # LP refinement (spends up to ~15 % of total budget)
    lp_deadline = min(deadline, time.time() + 0.15 * budget_s)
    a_small, score_lp = _lp_refine(best_seq, target_small, lp_deadline)
    if score_lp < best_score - 1e-12:
        best_seq, best_score = a_small.copy(), score_lp
        print(f"[LP] improvement to {best_score:.6f} (n={n_small})")

    # --------------------------------------------------------------
    # 2)  Upsample and smooth optimisation for larger n
    # --------------------------------------------------------------
    larger_schedule = [5000, 10000, 20000]   # increase gradually
    for n in larger_schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # upsample current best
        a = _upsample_sequence(best_seq, n)

        # short Adam run (≈30 % of remaining time)
        remaining = deadline - time.time()
        adam_time = min(remaining * 0.30, max(5.0, remaining * 0.3))
        adam_deadline = time.time() + adam_time
        a_opt = _adam_optimize(a, target, adam_deadline)
        score_opt = _max_convolution_fft(a_opt)
        if score_opt < best_score - 1e-12:
            best_seq, best_score = a_opt.copy(), score_opt
            print(f"[Adam] n={n} improved to {best_score:.6f}")

        # deterministic hill‑climb (≈20 % of remaining time)
        remaining = deadline - time.time()
        if remaining > 5.0:
            hc_deadline = time.time() + remaining * 0.20
            a_hc, score_hc = _hill_climb_deterministic(best_seq,
                                                       _full_convolution(best_seq),
                                                       hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq, best_score = a_hc.copy(), score_hc
                print(f"[HC] n={n} improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3)  Final polishing on the remaining budget
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(best_seq,
                                                          b_final,
                                                          deadline - 0.5)
        if final_score < best_score - 1e-12:
            best_seq, best_score = best_seq, final_score
            print(f"[Final] polishing improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 4)  Clean‑up: trim trailing near‑zeros
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 5)  Official evaluation
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Error calling evaluate_sequence:", exc)

    return best_seq.tolist()
