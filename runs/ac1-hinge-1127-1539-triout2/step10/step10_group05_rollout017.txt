def _build_conv_ub_matrix(f):
    n = f.shape[0]
    conv_len = 2*n - 1
    # list of diagonal arrays
    diag_data = [np.full(n, val, dtype=float) for val in f]
    offsets = np.arange(n)
    A = sparse.diags(diag_data, offsets, shape=(conv_len, n), format='csr')
    # column for -M
    neg_one_col = sparse.csr_matrix((-np.ones(conv_len)).reshape(conv_len, 1))
    A_ub = sparse.hstack([A, neg_one_col], format='csr')
    return A_ub

def _lp_refine(f, target, deadline):
    n = f.size
    cur_f = f.copy()
    cur_score = _max_convolution_fft(cur_f)
    iter_idx = 0

    while time.time() < deadline - 0.5 and iter_idx < 10:
        iter_idx += 1
        A_ub = _build_conv_ub_matrix(cur_f)
        conv_len = A_ub.shape[0]
        c = np.zeros(n+1)
        c[-1] = 1.0  # minimise M

        b_ub = np.zeros(conv_len)

        # Equality sum(g) = target; M excluded
        row_eq = np.zeros(n+1)
        row_eq[:n] = 1.0
        A_eq = sparse.csr_matrix(row_eq.reshape(1, n+1))
        b_eq = np.array([target])

        bounds = [(0, None)] * (n+1)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {iter_idx} failed: {res.message}")
            break
        g = res.x[:n]
        # ensure exact sum (may be slight numeric error)
        g = _project_to_simplex(g, target)
        new_score = _max_convolution_fft(g)

        if new_score < cur_score - 1e-12:
            cur_f = g
            cur_score = new_score
            print(f"[{time.time():.1f}] LP iteration {iter_idx} improvement → score={cur_score:.6f}")
            continue

        # try a convex combination
        best_local_f = cur_f
        best_local_score = cur_score
        for t in (0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - t) * cur_f + t * g
            cand = _project_to_simplex(cand, target)
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_local_score:
                best_local_f = cand
                best_local_score = cand_score
        if best_local_score < cur_score - 1e-12:
            cur_f = best_local_f
            cur_score = best_local_score
            print(f"[{time.time():.1f}] LP line‑search improvement → score={cur_score:.6f}")
        else:
            break

    return cur_f, cur_score

"""
Step‑function optimizer for the autoconvolution max objective.

The algorithm proceeds in several stages, each respecting a global
wall‑clock budget (default 1000 s).

1. **Parameter sweep** – cheap one‑parameter families (power‑law,
   exponential, edge‑heavy, mixed) are evaluated with the exact
   max‑convolution (FFT) and the best candidate is retained.

2. **LP refinement** – for a fixed vector `f` we solve the linear program

        minimise   M
        subject to f * g ≤ M      (linear in g)
                   Σ g = √(2 n), g ≥ 0

   which finds a direction `g` that can potentially lower the true
   objective (the maximal self‑convolution).  After solving the LP we
   either move fully to `g` or perform a short line‑search between `f`
   and `g`.  Several restarts with random initial vectors are performed.

3. **Deterministic hill‑climb** – a greedy move of mass from the worst
   convolution entry to a better destination, exactly updating the
   convolution in O(n).

4. **Simulated‑annealing refinement** – a few random mass‑transfer moves
   with an exponential temperature schedule to escape small plateaus.

The best sequence found is truncated to remove trailing near‑zeros and
returned.  All numbers are forced to be non‑negative, and the sum is
projected onto the ℓ¹‑simplex Σa = √(2 n), which makes the evaluation
function equal to the maximal convolution entry.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Small parametric sweep (cheap, but often finds a good basin)
# ----------------------------------------------------------------------


def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """Scan a few one‑parameter families and return the best vector."""
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law
    for p in np.linspace(0.1, 2.0, 40):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.0005, 0.02, 40):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy (large at ends, decays towards centre)
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 20):
        for lam in np.linspace(0.001, 0.01, 10):
            a = (idx ** (-p)) * np.exp(-lam * lin)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Random initial families
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                         # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                       # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                       # mixed
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                       # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                       # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                                   # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Large near the two edges, decaying towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that always attacks the current worst convolution entry."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing (O(n) per move)."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9999

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.5)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5
        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _build_conv_ub_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the matrix A_ub for the constraints f * g ≤ M.
    Rows correspond to convolution indices (2 n‑1 rows),
    columns 0…n‑1 contain the Toeplitz matrix of f,
    a final column contains −1 (coefficient of M).
    """
    n = f.size
    conv_len = 2 * n - 1
    # each diagonal offset i (0 ≤ i < n) contains the constant f[i]
    diag_data = [np.full(n, val, dtype=float) for val in f]
    offsets = np.arange(n)
    A = sparse.diags(diag_data, offsets, shape=(conv_len, n), format='csr')
    neg_one = sparse.csr_matrix((-np.ones(conv_len)).reshape(conv_len, 1))
    A_ub = sparse.hstack([A, neg_one], format='csr')
    return A_ub


def _lp_refine(f0: np.ndarray, target: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Given a starting vector f0, repeatedly solve the LP
        minimise M   s.t.   f * g ≤ M , Σ g = target , g ≥ 0 .
    If the LP solution improves the true objective (max self‑convolution)
    we move to it, optionally using a short line‑search.
    """
    n = f0.size
    cur_f = f0.copy()
    cur_score = _max_convolution_fft(cur_f)
    it = 0

    while time.time() < deadline - 0.5 and it < 12:
        it += 1
        A_ub = _build_conv_ub_matrix(cur_f)
        conv_len = A_ub.shape[0]

        c = np.zeros(n + 1)
        c[-1] = 1.0                    # minimise M

        b_ub = np.zeros(conv_len)

        # Equality Σ g = target (M excluded)
        eq_row = np.zeros(n + 1)
        eq_row[:n] = 1.0
        A_eq = sparse.csr_matrix(eq_row.reshape(1, n + 1))
        b_eq = np.array([target])

        bounds = [(0, None)] * (n + 1)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {it} failed: {res.message}")
            break

        g = res.x[:n]
        # Project onto the simplex to remove tiny numerical drift
        g = _project_to_simplex(g, target)

        new_score = _max_convolution_fft(g)

        if new_score < cur_score - 1e-12:
            cur_f, cur_score = g, new_score
            print(f"[{time.time():.1f}] LP iter {it} direct improvement → {cur_score:.6f}")
            continue

        # try a few convex combinations
        best_local_f, best_local_score = cur_f, cur_score
        for t in (0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - t) * cur_f + t * g
            cand = _project_to_simplex(cand, target)
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_local_score:
                best_local_f, best_local_score = cand, cand_score

        if best_local_score < cur_score - 1e-12:
            cur_f, cur_score = best_local_f, best_local_score
            print(f"[{time.time():.1f}] LP iter {it} line‑search improvement → {cur_score:.6f}")
        else:
            break

    return cur_f, cur_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising
    ``evaluate_sequence``.  The routine respects a wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 0)  Attempt to seed from a previous run
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            # use the default length for the search
            n = 2000
            target = math.sqrt(2.0 * n)
            if prev.size != n:
                prev = _upsample_sequence(prev, n)
            prev = _project_to_simplex(prev, target)
            s = _max_convolution_fft(prev)
            best_seq, best_score = prev.copy(), s
            print(f"[{time.time():.1f}] Seeded (n={n}) → score={best_score:.6f}")
        except Exception as e:   # pragma: no cover
            print("Seeding error:", e)

    # --------------------------------------------------------------
    # 1)  Fixed length (2000) – enough granularity and fast enough
    # --------------------------------------------------------------
    n = 2000
    target = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1a)  Parameter sweep (cheap)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_sweep, score_sweep = _parameter_sweep(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq, best_score = a_sweep.copy(), score_sweep
            print(f"[{time.time():.1f}] Sweep improvement → {best_score:.6f}")

    # --------------------------------------------------------------
    # 2)  LP‑based refinements (few restarts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        n_lp_restarts = min(4, int(remaining // 8.0))
        for r in range(n_lp_restarts):
            if time.time() >= deadline - 0.5:
                break
            if r == 0 and best_seq is not None:
                init = best_seq.copy()
            else:
                init = _random_initial_sequence(n, rng) if rng.random() < 0.5 else _edge_high_initial_sequence(n, rng)
            init = _project_to_simplex(init, target)

            sub_dead = time.time() + min(6.0, (deadline - time.time()) / (n_lp_restarts - r))
            f_opt, sc_opt = _lp_refine(init, target, sub_dead)

            if sc_opt < best_score - 1e-12:
                best_seq, best_score = f_opt.copy(), sc_opt
                print(f"[{time.time():.1f}] LP restart {r+1}/{n_lp_restarts} → {best_score:.6f}")

    # --------------------------------------------------------------
    # 3)  Deterministic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur, deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq, best_score = a_hc.copy(), s_hc
            print(f"[{time.time():.1f}] Hill‑climb → {best_score:.6f}")

    # --------------------------------------------------------------
    # 4)  Simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq, best_score = a_ann.copy(), s_ann
            print(f"[{time.time():.1f}] Anneal → {best_score:.6f}")

    # --------------------------------------------------------------
    # 5)  Clean‑up: drop trailing near‑zeros
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 6)  Official evaluation
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
