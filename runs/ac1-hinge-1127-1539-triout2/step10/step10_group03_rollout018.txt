def _adam_softmax_descent(a0, deadline, target_sum, max_beta=1e6):
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    beta = 0.5
    while time.time() < deadline:
        g,_ = _gradient_softmax(a, beta)
        t += 1
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = target_sum * 0.01 / (np.sqrt(v_hat) + 1e-8)
        a = a - lr * m_hat
        a = _project_to_simplex(a, target_sum)
        # optional jitter
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t} β={beta:.1f} best={best_val:.6f}")
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta*2, max_beta)
    return best_a, best_val

def _permute_annealing(a_init, deadline):
    a = a_init.copy()
    n = len(a)
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    while time.time() < deadline:
        # temperature schedule: linear from start_temp to end_temp
        elapsed = time.time() - start_time
        total = deadline - start_time
        temp = max(0.01, (1 - elapsed/total) * 0.5)  # start 0.5, end 0.01
        # propose swap
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        a[i], a[j] = a[j], a[i]
        new_score = _score(a)
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / temp):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermAnneal new best={best_score:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]
    return best_a, best_score
```

"""
Hybrid optimizer for the self‑convolution bound.

Main ideas
----------
1. Initialise with the analytically motivated  a_i ∝ (i+1)^‑½  (p = 0.5) which
   gives a nearly flat convolution profile.

2. Refine the *values* using an Adam‑style projected‑gradient descent on a
   soft‑max approximation of the peak of the convolution (the same surrogate
   used in the reference code but with adaptive step‑sizes).

3. After the values have stabilised, improve the *ordering* of the entries.
   The convolution depends strongly on the permutation of the multiset.
   A simulated‑annealing walk over random swaps is performed, keeping the
   best sequence seen.

4. A short deterministic mass‑transfer phase (the “DMT” step from the
   reference) is executed as a final polishing pass.

The routine respects the non‑negativity and ℓ¹‑simplex constraint, uses
FFT‑based convolutions for speed, and automatically stops when the total
budget (≈1000 s) is exhausted.  The best sequence found is stored in the
global variable `height_sequence_1` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities for linear convolution
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using a single FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Objective and its soft‑max surrogate
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Exact objective  2·n·max(a*a) / (∑a)² ."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns (grad, conv) where conv is the full convolution a*a.
    The surrogate is  LogSumExp_{k}(β·conv_k)   which approaches max when β→∞.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    mx = np.max(conv)
    # stable soft‑max weights
    w = np.exp(beta * (conv - mx))
    w /= w.sum()

    # gradient = 2 * (a_rev ⋆ w)  (⋆ denotes convolution)
    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    w_fft = np.fft.rfft(w, n=size)
    grad_full = np.fft.irfft(rev_fft * w_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]       # keep only the central n entries
    return grad, conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for initial guesses
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^{-p}  then renormalised to have the prescribed ℓ¹‑sum."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a different length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray,
                          deadline: float,
                          target_sum: float,
                          max_beta: float = 1e6) -> tuple[np.ndarray, float]:
    """
    Adam optimizer on the soft‑max surrogate of the peak.
    Keeps the sequence on the ℓ¹‑simplex after each step.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5            # temperature for the soft‑max
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size adapts to the magnitude of the gradient
        step = (target_sum * 0.02) / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

        # raise the temperature gradually
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation annealing – improve ordering of a fixed multiset
# ----------------------------------------------------------------------


def _permute_annealing(a0: np.ndarray,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over random swaps.
    The ℓ¹‑sum is unchanged, so only the ordering is altered.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        # linear temperature schedule (0.5 → 0.001)
        elapsed = time.time() - start
        total = deadline - start
        temp = max(0.001, (1.0 - elapsed / total) * 0.5)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        new_score = _score(a)
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / temp):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSwap new best={best_score:.6f}")
        else:
            # revert the useless swap
            a[i], a[j] = a[j], a[i]

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (identical to the reference)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a simple Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        # exponential temperature schedule
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # true contribution of each index to the current peak (product)
        w = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(w)]
        # pick a receiver among the least‑contributing indices
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it score={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step function with a low self‑convolution bound.

    The method builds on three stages:
        1. initialise with a ½‑power law,
        2. refine the values using Adam‑style projected gradient descent,
        3. optimise the ordering by simulated‑annealing swaps,
        4. a short deterministic mass‑transfer polishing pass.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose a working length (making it as large as the budget allows)
    # --------------------------------------------------------------
    if budget_s > 800:
        n = 12000
    elif budget_s > 500:
        n = 10000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Assemble a small pool of seeds (power‑law + possible reuse)
    # --------------------------------------------------------------
    seeds = []
    # baseline power‑law (p = 0.5)
    seeds.append(_power_law_sequence(n, 0.5, target_sum))

    # a slightly perturbed version (helps when the optimiser gets stuck)
    rng = np.random.default_rng()
    jitter = rng.normal(scale=target_sum * 0.01, size=n)
    seed_j = seeds[0] + jitter
    seed_j = np.maximum(seed_j, 0.0)
    seed_j = _project_to_simplex(seed_j, target_sum)
    seeds.append(seed_j)

    # reuse previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # --------------------------------------------------------------
    # Value optimisation – run a few Adam restarts within the remaining budget
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # give each start roughly an equal share of the remaining time
    while time.time() < deadline - 30.0 and seeds:
        seed = seeds.pop(0)
        run_dead = min(deadline - 5.0, time.time() + 20.0)   # 20 s per run
        a_opt, val_opt = _adam_softmax_descent(seed, run_dead, target_sum, max_beta=5e5)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Seed optimisation new best={best_score:.6f}")

    # --------------------------------------------------------------
    # If nothing succeeded (unlikely) fall back to a raw power law
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.5, target_sum)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Ordering optimisation – simulated‑annealing swaps
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        perm_dead = deadline - 10.0
        a_perm, val_perm = _permute_annealing(best_seq, perm_dead)
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final deterministic mass‑transfer polishing (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 2.0)
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_dead, target_sum,
                                                    start_temp=0.3, end_temp=1e-5,
                                                    low_frac=0.04)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final DMT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the best sequence for possible later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
