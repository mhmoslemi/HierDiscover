"""
Propose a low‑score step‑function via a multi‑stage pipeline.

Stages
------
1. **Coarse parametric warm‑start** – search a two‑parameter
   (power‑law exponent `p`, exponential decay `lam`) family on a
   reduced problem size, then upscale.
2. **Mirror‑descent (exponential updates)** – fast early descent on a
   soft‑max surrogate of the maximal convolution entry.
3. **Projected sub‑gradient descent** – direct descent on the true
   max‑convolution using the exact sub‑gradient.
4. **Aggressive simulated annealing** – long stochastic local search
   with adaptive move size and a linear cooling schedule.  Each move
   transfers a random fraction of mass from a high‑contributing index
   to a low‑contributing one; moves are accepted with a Metropolis
   criterion.

All stages respect the global time budget (default 1000 s).  The best
sequence found is trimmed of trailing zeros before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (fast and numerically stable)
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 length sufficient for a linear convolution of size 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  Assumes sum(a) > 0."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence a_i = (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (for mirror descent)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 1 – coarse parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int, target_sum: float, deadline: float):
    """
    Scan a 2‑parameter family a_i = (i+1)^(-p)·exp(-lam·i/small_n)
    on a reduced size, keep the best.
    """
    p_vals = np.linspace(0.2, 1.5, 31)          # finer grid for p
    lam_vals = np.linspace(0.0, 12.0, 31)      # finer grid for λ
    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] grid p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    # some random refinements
    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.5)
        lam = rng.exponential(1.0) * 4.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] rand p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – exponential (mirror) descent on soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Multiplicative mirror descent using the soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    base_eta = 0.07                     # slightly larger base step
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)           # multiplicative update
        a = a * (target_sum / a.sum())        # re‑normalise

        if it % 200 == 0:
            # small jitter to avoid stagnation
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – projected sub‑gradient descent on the true max convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Standard sub‑gradient descent on the true objective.
    Uses a diminishing step size based on the safe maximal step
    that keeps all coordinates non‑negative.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient for the active shift k_max
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.size - 1))
        hi = min(k_max, a.size - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        # safe step size (largest η that keeps a ≥ 0)
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} score={best_val:.6f}")

        # occasional jitter to escape possible plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 4 – aggressive simulated annealing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Long SA with adaptive move size.  Each move transfers a random
    fraction of mass from a high‑contributing index (with respect to
    the current maximal convolution entry) to a low‑contributing one.
    Acceptance follows a Metropolis rule with a linearly decaying
    temperature.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start

    # Temperature proportional to current score; linearly cooling to 0.
    T0 = 0.02 * best_val

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # Choose donor proportionally to its contribution to the max entry
        wsum = weight.sum()
        if wsum > 0.0:
            donor = rng.choice(idx, p=weight / wsum)
        else:
            donor = rng.integers(n)

        # Choose receiver from the low‑contribution tail
        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        # Transfer amount: a random fraction up to a configurable maximum
        max_frac = 0.12 if (time.time() - start) < total * 0.5 else 0.05
        max_delta = a[donor] * max_frac
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        # Linear cooling schedule
        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / total), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_val:.6f}")

        # Rare large perturbation to diversify the search
        if rng.random() < 0.003:
            jitter = rng.normal(scale=target_sum * 0.0015, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val

# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low score.
    The routine respects the given time budget (seconds) and returns the best
    sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start: either reuse a previous run or generate a power‑law shape
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(22.0,
                                      (deadline - time.time()) * 0.10)
    best_small, _ = _grid_param_search(small_n, small_target, param_deadline)

    if best_small is not None:
        init = _resize_sequence(best_small, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑started from parametric search (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – exponential (mirror) descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.13
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_val = val_exp
        print(f"[{time.time():.2f}] After ExpGD best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.30
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_val:
            best_seq = a_sub
            best_val = val_sub
        print(f"[{time.time():.2f}] After SubGrad best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – aggressive simulated annealing (uses most of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_deadline)
        if val_sa < best_val:
            best_seq = a_sa
            best_val = val_sa
        print(f"[{time.time():.2f}] After SA best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros and final evaluation (for logging only)
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        final_score = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a future call
    globals()["height_sequence_1"] = final_seq
    return final_seq
