"""
Multi‑stage optimizer for the step‑function problem.

The goal is to minimise  max(a * a)  subject to  a_i ≥ 0  and  Σ a_i = √(2 n).
The evaluation function equals this maximum when the L¹‑norm is fixed,
so we work directly with the max‑convolution value.

Algorithm
---------
0. Warm‑start from the previously best construction (global variable
   ``height_sequence_1``); otherwise use a decreasing power‑law.

1. **Smoothed gradient descent**  
   We minimise the smooth surrogate  
        F_α(a) = (1/α)·log Σ_i exp(α·(a*a)_i) .  
   For α→∞ this tends to the true max, but for moderate α the gradient
   is well‑behaved.  The gradient is obtained by a single FFT‑based
   convolution with the soft‑max weights.  α is increased linearly
   during the run, and each trial step is projected onto the simplex
   Σa = √(2 n).  A short back‑tracking line search guarantees monotone
   progress.

2. **Short random restarts** – jitter the current best solution and
   rerun a brief smoothed descent; keep any improvement.

3. **Exact local mass‑transfer refinement** – the same O(n) simulated‑annealing
   move as in the previous solution, which transfers a tiny amount of mass
   from a “bad’’ index (large contribution to the maximal convolution entry)
   to a “good’’ one.

4. Clean‑up: truncate trailing zeros, evaluate with the supplied
   ``evaluate_sequence`` and store the result for future warm‑starts.

All stages respect the overall time budget (default 1000 s) and keep
the sequence non‑negative.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]

# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex target sum must be non‑negative")
    # The algorithm of Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _max_convolution(a: np.ndarray) -> float:
    """Return max(a*a).  Assumes Σa > 0."""
    b = _conv_fft(a)
    return float(b.max())


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             start_alpha: float = 2.0,
                             end_alpha: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed surrogate
        F_α(a) = (1/α)·log Σ_i exp(α·(a*a)_i)
    using projected gradient descent.  α is increased linearly from
    ``start_alpha`` to ``end_alpha`` as the run progresses.
    Returns the best sequence found and its true max‑convolution value.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _max_convolution(a)

    rng = np.random.default_rng()
    it = 0
    start = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        # ---- schedule α -------------------------------------------------
        frac = (time.time() - start) / (deadline - start)
        alpha = start_alpha + frac * (end_alpha - start_alpha)

        # ---- compute convolution and soft‑max weights --------------------
        b = _conv_fft(a)
        max_b = b.max()
        # avoid overflow / underflow
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0 or np.isinf(w_sum):
            w = np.ones_like(b) / b.size
        else:
            w = w / w_sum

        # ---- gradient of the surrogate ----------------------------------
        L = b.size - 1                     # = 2n‑2
        conv_full = _linear_convolve_fft(a, w[::-1])
        g = conv_full[L - np.arange(n)]    # length n

        # ---- step size (simple 1/√t schedule) ---------------------------
        step = (target_sum * 0.2) / np.sqrt(it + 1.0)

        # ---- back‑tracking line search (max 5 attempts) -----------------
        for _ in range(5):
            a_trial = _project_to_simplex(a - step * g, target_sum)
            new_val = _max_convolution(a_trial)
            # accept if we made progress, otherwise shrink step
            if new_val <= max_b or step < 1e-12:
                a = a_trial
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] SmoothGD it={it:6d} α={alpha:.1f} "
                          f"score={best_val:.6f}")
                break
            step *= 0.5

        # ---- occasional jitter to escape plateaus -----------------------
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact local mass‑transfer refinement (same as the previous solution)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the true max‑convolution.
    Moves a small amount of mass from a “bad’’ index to a “good’’ one,
    updating the full convolution in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _max_convolution(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – choose from the lower‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta            # amount moved from donor → receiver

        # --------------------------------------------------------------
        # Update convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic re‑normalisation & full recomputation
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _max_convolution(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional jitter (small random mass swap)
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _max_convolution(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)            # fixing Σa simplifies the objective

    # ------------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
    else:
        # fallback: decreasing power‑law
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – smoothed gradient descent (≈35 % of budget)
    # ------------------------------------------------------------------
    sg_deadline = time.time() + 0.35 * (deadline - time.time())
    a_sg, val_sg = _smooth_gradient_descent(init, target_sum, sg_deadline)
    best_seq = a_sg
    best_score = val_sg
    print(f"[{time.time():.2f}] After smoothed GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short random restarts of smoothed descent (~20 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rest_deadline = time.time() + 0.20 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2.0:
            remaining = rest_deadline - time.time()
            run_time = min(5.0, remaining)
            if run_time < 1.0:
                break
            jitter = rng.normal(scale=target_sum * 0.02, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            a_tmp, val_tmp = _smooth_gradient_descent(start_seq,
                                                      target_sum,
                                                      time.time() + run_time)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")
        final_val = None

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
