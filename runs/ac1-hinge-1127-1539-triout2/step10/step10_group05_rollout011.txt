"""
propose_candidate – a multi‑stage optimizer for the max‑convolution
objective  f(a)=2·n·max(conv(a,a))/(sum(a)^2).

The algorithm works inside the 1000 s wall‑clock budget:

1. **Depth‑first parameter sweep** – scans several one‑parameter families
   (power‑law, exponential, edge‑heavy, mixed) on a few different
   lengths (1500, 2000, 2500).  The best candidate so far is kept.

2. **Projected Adam on the smooth‑max surrogate** – many short restarts
   (≈6 s each) are performed.  The soft‑max temperature β is ramped from
   2 up to 1e7, giving a differentiable approximation of the max‑entry.
   After each restart the best sequence is retained.

3. **Deterministic hill‑climbing** – a greedy move transfers a fraction
   of the mass that contributes most to the current worst convolution
   entry to the location that hurts the least.  The delta is reduced
   gradually (0.5 → 0.25 → 0.125 …) to allow fine adjustments.

4. **Projected sub‑gradient descent** – uses the exact sub‑gradient of the
   max‑entry (∇_i max = 2·a_{M−i}) and a diminishing step‑size schedule.
   This quickly drives the max‑convolution down further.

5. **Short simulated‑annealing polish** – random mass swaps with an
   exponential cooling schedule help escape tiny local minima.

6. **Cleanup** – trailing near‑zero entries are stripped and the final
   sequence is reported via `evaluate_sequence`.

All phases respect the global deadline, and the best sequence found
throughout the run is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Exact linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed max (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and b the exact convolution.
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)

    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # shape (2n‑1,)

    # convolution a_rev * p  (only the middle n entries are needed)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate scale – proportional to the target ℓ¹‑mass
    base_lr = 0.13 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    fam = rng.integers(0, 8)

    idx = np.arange(n, dtype=float)

    if fam == 0:                     # pure power‑law
        p = rng.uniform(0.1, 1.5)
        a = (idx + 1.0) ** (-p)

    elif fam == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif fam == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(scale=0.012)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif fam == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif fam == 4:                   # edge‑heavy (distance to nearest edge)
        lam = rng.uniform(0.001, 0.025)
        d = np.minimum(idx, n - 1 - idx)
        a = np.exp(-lam * d)

    elif fam == 5:                   # piecewise linear decay
        cut = rng.integers(int(0.2 * n), int(0.8 * n))
        slope = rng.uniform(0.0, 2.0)
        left = np.linspace(1.0, max(0.0, 1.0 - slope * cut / n), cut)
        right = np.linspace(max(0.0, 1.0 - slope * (n - cut) / n), 1.0, n - cut)
        a = np.concatenate([left, right])

    elif fam == 6:                   # double‑exponential (symmetrically heavy)
        lam1 = rng.exponential(scale=0.01)
        lam2 = rng.exponential(scale=0.01)
        a = np.exp(-lam1 * idx) + np.exp(-lam2 * (n - 1 - idx))

    else:                             # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    # tiny Gaussian noise – helps the simplex projection be well‑behaved
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Large values near both edges, exponential decay towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.03)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()
    # start with a relatively aggressive delta factor and shrink it gradually
    delta_factor = 0.8
    while time.time() < deadline:
        M = int(np.argmax(b))                     # index of the current worst entry

        # indices that actually contribute to (a*a)[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        if contrib.size == 0:
            break

        i = idx[np.argmax(contrib)]               # source contributing the most

        # Find a destination j ≠ i that hurts the least this entry.
        # We rank candidates by the opposite contribution a[M-j].
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # sort destinations by increasing counterpart contribution
        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # transfer a fraction of source mass to j
        delta = a[i] * delta_factor
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-14:
                best_score = new_max
                best_a = a.copy()
        else:
            # try halving the move size a few times
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-15:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-14:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-14:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5

            if not improved:
                # small random jitter to escape flat regions
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

        # shrink the delta factor slowly – this lets us take big steps early
        # and fine‑tune later.
        delta_factor = max(0.05, delta_factor * 0.9995)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing (O(n) per move). Returns the best vector
    and the best maximal convolution entry observed.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9997

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.05, 0.3)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5

        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑entry
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on max(conv(a,a)).
    Uses the exact sub‑gradient (2·a_{M−i}) where M = argmax conv.
    """
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)

    start = time.time()
    iteration = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        iteration += 1
        b = _full_convolution(a)
        M = int(np.argmax(b))

        # compute sub‑gradient g_i = 2·a_{M−i} (zero if out of range)
        g = np.zeros_like(a)
        # vectorised fill
        i_vals = np.arange(n)
        idx = M - i_vals
        mask = (idx >= 0) & (idx < n)
        g[mask] = 2.0 * a[idx[mask]]

        # step size diminishing like α / sqrt(iter)
        alpha0 = 0.5 * target   # initial scale
        step = alpha0 / math.sqrt(iteration)

        a -= step * g
        # add very small jitter to keep us away from degenerate corners
        if iteration % 200 == 0:
            a += rng.normal(scale=target * 1e-5, size=n)

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Small parametric sweep (cheap, but often finds a good basin)
# ----------------------------------------------------------------------


def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan a few one‑parameter families (power‑law, exponential, edge‑heavy,
    mixed) and return the best vector found (according to the exact
    max‑convolution).
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law exponents
    for p in np.linspace(0.05, 2.5, 50):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.0003, 0.04, 60):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy (large at the ends, decays towards centre)
    for lam in np.linspace(0.0008, 0.04, 45):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.2, 1.8, 30):
        for lam in np.linspace(0.001, 0.02, 12):
            a = (idx ** (-p)) * np.exp(-lam * (lin - 0))
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 0)  Seed from a previous run if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded (n={n0}) → score={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # --------------------------------------------------------------
    # 1)  Explore a few lengths (1500‑2500).  Short sweeps give cheap basins.
    # --------------------------------------------------------------
    lengths = [1500, 1800, 2000, 2200, 2500]
    for n in lengths:
        if time.time() >= deadline - 3.0:
            break
        target = math.sqrt(2.0 * n)

        # ----- cheap param sweep -------------------------------------------------
        a_sweep, score_sweep = _parameter_sweep(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep (n={n}) → score={best_score:.6f}")

        # ----- Adam restarts (≈6 s each) ----------------------------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            continue
        # allocate about 35 % of remaining time to Adam for this length
        alloc = remaining * 0.35
        per_restart = max(4.0, alloc / max(1, int(alloc // 4.0)))
        n_restarts = int(alloc // per_restart)
        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            if r == 0 and best_seq is not None and best_seq.size == n:
                a0 = best_seq.copy()
            else:
                # mix families for diversity
                if rng.random() < 0.45:
                    a0 = _random_initial_sequence(n, rng)
                else:
                    a0 = _edge_high_initial_sequence(n, rng)

            a_opt = _adam_optimize(a0, target, time.time() + per_restart * 0.9)
            s = _max_convolution_fft(a_opt)
            if s < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = s
                print(f"[{time.time():.1f}] Adam (n={n}) restart {r+1}/{n_restarts} → score={best_score:.6f}")

        # ----- quick hill‑climb to polish this length ---------------------------
        if time.time() < deadline - 2.0:
            a_cur = _project_to_simplex(best_seq, target)
            b_cur = _full_convolution(a_cur)
            a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur,
                                                   deadline - 1.0)
            if s_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = s_hc
                print(f"[{time.time():.1f}] Hill‑climb (n={n}) → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2)  Intensive hill‑climbing on the global best (longer budget)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        a_cur = _project_to_simplex(best_seq, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur, deadline - 2.5)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Global hill‑climb → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3)  Projected sub‑gradient descent (exact max) – core fine‑tuning
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        a_sg = _subgradient_optimize(best_seq, deadline - 1.5)
        s_sg = _max_convolution_fft(a_sg)
        if s_sg < best_score - 1e-12:
            best_seq = a_sg.copy()
            best_score = s_sg
            print(f"[{time.time():.1f}] Sub‑gradient → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4)  Simulated‑annealing polish (short, but can escape tiny traps)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.size
        a_cur = _project_to_simplex(best_seq, math.sqrt(2.0 * n))
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal polish → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5)  Final clean‑up – drop trailing near‑zeros
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never trigger)
        n = 2000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 6)  Official evaluation report
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
