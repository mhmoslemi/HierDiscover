def _adam_descent(a0, target_sum, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best = a.copy()
    best_val = _score(best)
    beta = 1.0
    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step = 0.01 / (np.sqrt(v_hat) + epsilon)  # learning rate schedule
        a = a * np.exp(-step * m_hat)  # exponentiated gradient
        a = a * (target_sum / a.sum())
        if it % 100 == 0:
            val = _score(a)
            if val < best_val:
                best_val = val
                best = a.copy()
                print("[... ] Adam it={} score={:.6f}".format(it,best_val))
        if it % 200 == 0:
            beta = min(beta*2, 32768.0)
        it += 1
    return best, best_val
```

def _parametric_search(n, target_sum, deadline):
    best_seq = None
    best_val = np.inf
    # families
    rng = np.random.default_rng()
    # grid over p,q
    p_vals = np.arange(0.0, 5.1, 0.5)
    q_vals = np.arange(0.0, 5.1, 0.5)
    for p in p_vals:
        for q in q_vals:
            if time.time() > deadline:
                break
            i = np.arange(n, dtype=np.float64)
            seq = (i+1) ** (-p) * ((1 - i/(n-1)) ** q)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
    # random refinements
    while time.time() < deadline:
        p = rng.uniform(0,5)
        q = rng.uniform(0,5)
        lam = rng.exponential(2.0)  # maybe also lambda factor
        seq = (i+1) ** (-p) * ((1 - i/(n-1)) ** q) * np.exp(-lam * i/(n-1))
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    return best_seq, best_val

"""
Propose a low‑score step function.

The search consists of three phases:

1. **Rich parametric warm‑start** – a dense grid + random sampling over several
   families of analytic profiles (power‑law, exponential, Beta‑type, and mixed
   forms).  Each candidate is normalised to the required ℓ¹‑norm and scored
   with the exact objective.

2. **Adam‑style exponentiated gradient descent** on a smooth soft‑max
   surrogate of the max‑convolution.  The algorithm works directly on the
   non‑negative vector, uses the natural KL‑geometry (multiplicative update),
   and adapts first/second moment estimates of the gradient.  The temperature
   β of the soft‑max is increased steadily so that the surrogate converges to
   the true max.

3. **Deterministic mass‑transfer refinement** – repeatedly move a small
   fraction of mass from the most harmful index (the one that contributes the
   most to the current maximal convolution entry) to a low‑contribution
   index.  The move is accepted only if it improves the true objective.

All stages respect the global time budget (default 1000 s) and keep the
sequence non‑negative with the prescribed ℓ¹‑norm.  The final sequence is
trimmed of trailing zeros before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Length of the zero‑padded FFT suitable for a full convolution of size n."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact evaluation function: 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)

    # FFT of a
    a_fft = np.fft.rfft(a, n=size)

    # full convolution (real)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·reverse(a) ⊛ p  (restricted to central n entries)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Parametric families (used for the warm‑start)
# ----------------------------------------------------------------------


def _gen_power_law(n, p):
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def _gen_exponential(n, lam):
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i / (n - 1))


def _gen_polynomial_decay(n, p):
    i = np.arange(n, dtype=np.float64)
    return (1.0 - i / (n - 1)) ** p


def _gen_beta_shape(n, alpha, beta):
    """Beta‑type shape: (i+1)^α * (n‑i)^β  (zero at both ends)."""
    i = np.arange(n, dtype=np.float64)
    left = (i + 1) ** alpha
    right = (n - i) ** beta
    return left * right


def _gen_mixed(n, p, q, lam):
    """Mixed family: (i+1)^(-p) * (1-i/(n-1))^q * exp(-lam*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p)
    a *= (1.0 - i / (n - 1)) ** q
    a *= np.exp(-lam * i / (n - 1))
    return a

# Parametric broad search (grid + random)

def _parametric_search(n: int, target_sum: float, deadline: float):
    """Search many analytic families; keep the best sequence found."""
    best_seq = None
    best_val = np.inf
    rng = np.random.default_rng()
    i = np.arange(n, dtype=np.float64)

    # ----- deterministic grid -------------------------------------------------
    p_vals = np.linspace(0.0, 5.0, 11)          # for power‑law / polynomial
    lam_vals = np.linspace(0.0, 15.0, 11)       # exponential strength
    alpha_vals = np.linspace(0.0, 5.0, 11)      # beta‑shape exponents
    beta_vals = np.linspace(0.0, 5.0, 11)

    # power‑law
    for p in p_vals:
        if time.time() > deadline:
            break
        seq = _gen_power_law(n, p)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] grid power‑law p={p:.2f}  score={val:.6f}")

    # exponential
    for lam in lam_vals:
        if time.time() > deadline:
            break
        seq = _gen_exponential(n, lam)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] grid exponential λ={lam:.2f} score={val:.6f}")

    # polynomial decay (1‑t)^p
    for p in p_vals:
        if time.time() > deadline:
            break
        seq = _gen_polynomial_decay(n, p)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] grid poly‑decay p={p:.2f} score={val:.6f}")

    # beta‑type shapes
    for a in alpha_vals:
        for b in beta_vals:
            if time.time() > deadline:
                break
            seq = _gen_beta_shape(n, a, b)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[{time.time():.2f}] grid beta α={a:.2f} β={b:.2f} score={val:.6f}")

    # ----- random sampling ----------------------------------------------------
    while time.time() < deadline:
        p = rng.uniform(0.0, 5.0)
        q = rng.uniform(0.0, 5.0)
        lam = rng.exponential(1.5) * 5.0   # favour moderate λ
        choice = rng.integers(4)
        if choice == 0:
            seq = _gen_power_law(n, p)
        elif choice == 1:
            seq = _gen_exponential(n, lam)
        elif choice == 2:
            seq = _gen_polynomial_decay(n, p)
        elif choice == 3:
            seq = _gen_mixed(n, p, q, lam)
        else:
            seq = _gen_beta_shape(n, p, q)

        # avoid the all‑zeros case (possible for extreme exponents)
        if seq.max() == 0:
            continue

        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] random best val={val:.6f}")

def _adam_exponentiated_descent(a0, target_sum, deadline):
    """Exponentiated Adam on the soft‑max surrogate of max‑convolution."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0                         # soft‑max temperature
    max_beta = 32768.0
    m = np.zeros_like(a)               # first moment (gradient)
    v = np.zeros_like(a)               # second moment (squared gradient)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments (on the gradient of log‑a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step‑size schedule – larger early, then decay gently
        step = 0.05 / (np.sqrt(v_hat) + eps)

        # multiplicative (exponentiated) update
        a = a * np.exp(-step * m_hat)

        # re‑normalise to the required ℓ¹‑norm
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # evaluation using the true objective every few iterations
        if it % 200 == 0:
            val = _score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase the soft‑max sharpness gradually
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _mass_transfer_refine(a0, target_sum, deadline):
    """Move a small fraction of mass from high‑contribution to low‑contribution indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        # index of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # feasible index range for this k
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, k_max-i)
        partner = k_max - idx
        contrib = a[idx] * a[partner]

        # donor = index with the largest contribution (most responsible)
        donor = idx[np.argmax(contrib)]

        # pick a receiver among the lowest‑contributing indices
        low_thr = np.percentile(contrib, 15)
        low_candidates = idx[contrib <= low_thr]
        if low_candidates.size:
            receiver = low_candidates[rng.integers(low_candidates.size)]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_frac = 0.05
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] Transfer improved score={best_val:.6f}")

        # renormalise occasionally to suppress drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val
