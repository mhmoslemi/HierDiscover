"""
Search for a non‑negative step function that minimizes the benchmark score.

The algorithm combines three ideas:

1. **Broad random exploration** – many random families (power‑law,
   power‑law + exponential tail, plateau + tail, stretched exponential)
   are sampled and the best candidate is kept.  The families are cheap
   to generate and the score is evaluated with an FFT based convolution.

2. **Projected Adam gradient descent** – the current best candidate is
   refined by several Adam‑style GD runs that minimise a soft‑max
   approximation of the maximal convolution entry.  The variables are
   projected onto the simplex {x≥0, Σx = √(2 n)} after each update.

3. **Deterministic mass‑transfer & global random moves** – a greedy
   reduction of the maximal convolution entry is performed (exactly
   moving the amount of mass needed to equalise the current peak and
   the second‑largest peak).  Occasional random mass moves help to
   escape shallow plateaus.

The time budget (default 1000 s) is split into phases:
≈55 % for random search, ≈35 % for Adam‑GD, the rest for mass‑transfer
and a final polishing GD.  The best sequence found is trimmed of
trailing zeros and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark objective (equals the maximal convolution entry after scaling)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # with the fixed normalisation Σa = √(2 n) the factor cancels,
    # but we keep the original formula for safety.
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families of candidate sequences
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) scaled to the required sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) * exp(-q*i/n) scaled."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _plateau_tail_sequence(
    n: int,
    m: int,
    p: float,
    q: float,
    r: float,
    target_sum: float,
) -> np.ndarray:
    """
    First `m` entries form a plateau (height = r * first tail value),
    the remaining `n-m` entries follow a power‑law with exponential tail.
    """
    if m >= n:
        # pure plateau (constant)
        a = np.full(n, 1.0, dtype=np.float64)
        a *= target_sum / a.sum()
        return a

    tail_len = n - m
    idx_tail = np.arange(tail_len, dtype=np.float64)
    tail = (idx_tail + 1) ** (-p) * np.exp(-q * idx_tail / tail_len)

    plateau_height = tail[0] * r if tail_len > 0 else r
    plateau = np.full(m, plateau_height, dtype=np.float64)

    a = np.concatenate((plateau, tail))
    a *= target_sum / a.sum()
    return a


def _stretched_exponential_sequence(
    n: int, gamma: float, target_sum: float
) -> np.ndarray:
    """a_i = exp( - (i/(n-1))^γ ) scaled to the required sum."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-np.power(i / (n - 1), gamma))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max based gradient of max‑convolution
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where conv is the full a*a convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # soft‑max weights over convolution entries

    # gradient = 2 * (a_rev * p) (convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------
def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max of the convolution max."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Multiple short Adam‑GD runs with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic exact mass‑transfer (Phase B)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer_exact(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy moves that directly level the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # choose one maximal index (first)
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight for each i that contributes to conv[k0]
        w = a[idx] + a[k0 - idx]

        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contributions of donor/receiver to the peak
        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            # cannot improve this peak – try a second maximal index if present
            if len(idx_max) > 1:
                k0 = idx_max[1]
                lo = max(0, k0 - (n - 1))
                hi = min(k0, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                w = a[idx] + a[k0 - idx]
                donor = idx[np.argmax(w)]
                receiver = idx[np.argmin(w)]
                contrib_d = a[k0 - donor]
                contrib_r = a[k0 - receiver]
                denom = 2.0 * (contrib_d - contrib_r)
                if denom <= 0.0:
                    break
            else:
                break

        # second largest convolution value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.5, delta_needed * 0.99)

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------
def _global_random_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing moves that shift a small amount of mass."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the time budget, returns the best sequence found,
    and prints progress information.
    """
    start_time = time.time()
    total_budget = max(10.0, budget_s - 0.5)      # tiny safety margin
    deadline = start_time + total_budget

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scale‑invariant)

    # ------------------------------------------------------------------
    # Phase 0 – broad random exploration
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # If a previous construction exists, seed from it (after resizing if needed)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # random families
    phase0_deadline = start_time + total_budget * 0.55
    while time.time() < phase0_deadline:
        family = rng.integers(4)

        if family == 0:
            # pure power‑law
            p = rng.uniform(0.05, 2.0)
            seq = _power_law_sequence(n, p, target_sum)

        elif family == 1:
            # power‑law + exponential tail
            p = rng.uniform(0.05, 2.0)
            q = rng.uniform(0.0, 10.0)
            seq = _exp_power_law_sequence(n, p, q, target_sum)

        elif family == 2:
            # plateau + tail
            frac = rng.uniform(0.0, 0.5)
            m = int(frac * n)
            p = rng.uniform(0.05, 2.0)
            q = rng.uniform(0.0, 10.0)
            r = rng.uniform(1.0, 5.0)
            seq = _plateau_tail_sequence(n, m, p, q, r, target_sum)

        else:
            # stretched exponential
            gamma = rng.uniform(0.2, 2.0)
            seq = _stretched_exponential_sequence(n, gamma, target_sum)

        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random search new best={best_score:.6f}")

    # safety: ensure we have a sequence
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.6, target_sum)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 20.0:
        gd_deadline = time.time() + remaining_time * 0.35
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec, target_sum, gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer + global random moves
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # deterministic refinement
        a_ref, val_ref = _deterministic_mass_transfer_exact(best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best={best_score:.6f}")

        # a short burst of global random moves
        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best={best_score:.6f}")

        # break early if we are very close to the deadline
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Final polishing with a higher β
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store for possible reuse in a subsequent run
    globals()["height_sequence_1"] = final_seq

    # Optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
