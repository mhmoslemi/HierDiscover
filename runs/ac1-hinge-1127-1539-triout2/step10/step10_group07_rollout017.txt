"""
Optimisation of a step‑function to minimise
      2·n·max(conv(a,a)) / (sum(a)²)

The code builds a good initial guess from several parametric families,
refines it with a smoothed‑max gradient descent (soft‑max approximation
of the infinity norm) and finally performs a long simulated‑annealing
mass‑transfer pass.  All updates keep the sequence on the simplex
Σa = √(2 n) and respect non‑negativity, so the objective equals the
maximal convolution entry.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective: 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) the return equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Gradient of the smooth max (soft‑max) objective
# ----------------------------------------------------------------------


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a ( Σ_k w_k (a*a)[k] ) = 2·corr(a, w)
    where corr is the (non‑reversed) correlation.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_full = _linear_convolve_fft(a, w_rev)          # length 3n‑2
    L = w.shape[0]                                     # = 2n‑1
    # slice L‑n … L‑1 and reverse to obtain the correlation values
    slice_ = conv_full[L - n : L][::-1]
    return 2.0 * slice_


def _smooth_max_and_grad(a: np.ndarray, alpha: float):
    """
    Soft‑max approximation of max(b) where b = a*a.
    Returns (smooth_value, b, gradient) with respect to a.
    """
    b = _conv_fft(a)                # length 2n‑1
    max_b = b.max()
    # stable soft‑max
    exp_term = np.exp(alpha * (b - max_b))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp          # weights, sum to 1
    g = _grad_from_weights(a, w)    # sub‑gradient of the smooth max
    smooth_val = max_b + np.log(sum_exp) / alpha
    return smooth_val, b, g


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation of the max‑convolution.
    The iterate stays on the simplex Σa = target_sum.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    n = a.shape[0]

    # initial exact objective (max entry of the convolution)
    best_a = a.copy()
    best_val = _conv_fft(a).max()
    print(f"[{time.time():.2f}]  GD start best={best_val:.6f}")

    rng = np.random.default_rng()
    it = 0
    step0 = target_sum * 0.2               # base step magnitude
    alphas = np.geomspace(0.5, 64.0, num=12)   # schedule of smoothing parameters

    while time.time() < deadline:
        alpha = alphas[min(it // 500, len(alphas) - 1)]
        smooth_val, b, g = _smooth_max_and_grad(a, alpha)

        # Normalised update
        step = step0 / np.sqrt(it + 1.0)
        gn = np.linalg.norm(g)
        if gn > 0:
            a -= step * g / gn

        # Project back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # Exact objective for bookkeeping
        cur_val = _conv_fft(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}]  GD it={it:6d} α={alpha:5.2f}  best={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the true max‑convolution.
    Moves a tiny amount of mass from a “bad’’ index to a “good’’ one.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – choose from the lower‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.06, target_sum * 0.0012)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta            # amount moved donor → receiver

        # --------------------------------------------------------------
        # Update convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}]  Local refine improved={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic re‑normalisation & full recomputation
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            # re‑scale to exact target sum (prevents drift)
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}]  Re‑norm improved={best_val:.6f}")

        # occasional tiny jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.006, target_sum * 0.0006)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}]  Jitter improved={best_val:.6f}")

    return best_a, best_val

# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)            # fixing Σa simplifies the objective

    # ------------------------------------------------------------------
    # Generate a variety of hand‑crafted initial shapes
    # ------------------------------------------------------------------
    candidates = []

    # decreasing power‑law
    for p in [-0.30, -0.45, -0.55, -0.70, -0.90]:
        seq = (np.arange(1, n + 1, dtype=np.float64) ** p)
        candidates.append(seq)

    # exponential decay
    for rate in [3e-4, 5e-4, 1e-3, 2e-3]:
        seq = np.exp(-rate * np.arange(n, dtype=np.float64))
        candidates.append(seq)

    # sine‑based hump
    for power in [0.5, 1.0, 2.0, 3.0]:
        seq = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n) ** power
        candidates.append(seq)

    # symmetric edge‑heavy (distance to nearest end)
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx + 1, n - idx)
    for p in [-0.2, -0.5, -1.0]:
        seq = dist ** p
        candidates.append(seq)

    # half‑cosine bump
    for power in [1.0, 2.0]:
        seq = (0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))) ** power
        candidates.append(seq)

    # warm‑start with the globally best construction seen so far
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if hs.shape[0] != n:
            hs = _resize_sequence(hs, n)
        candidates.append(hs)

    # evaluate all candidates, keep the three best
    evaluated = []
    for seq in candidates:
        proj = _project_to_simplex(seq, target_sum)
        val = _score(proj)
        evaluated.append((val, proj))
    evaluated.sort(key=lambda x: x[0])
    top_seqs = [seq for _, seq in evaluated[:3]]
    best_seq = top_seqs[0].copy()
    best_score = evaluated[0][0]
    print(f"[{time.time():.2f}]  init best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (≈55 % of the budget)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + 0.55 * (deadline - time.time())
    for init in top_seqs:
        if time.time() >= gd_deadline:
            break
        remaining = gd_deadline - time.time()
        run_deadline = time.time() + min(200.0, remaining)   # cap per run
        a_opt, val_opt = _smooth_gradient_descent(init, target_sum, run_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}]  GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short stochastic restarts of gradient descent (~15 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        restart_deadline = time.time() + 0.15 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < restart_deadline - 2.0:
            remaining = restart_deadline - time.time()
            run_time = min(5.0, remaining)
            jitter = rng.normal(scale=target_sum * 0.025, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            a_tmp, val_tmp = _smooth_gradient_descent(start_seq,
                                                      target_sum,
                                                      time.time() + run_time)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}]  Restart GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
        print(f"[{time.time():.2f}]  After refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up, final evaluation and warm‑start storage
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}]  Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}]  Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
