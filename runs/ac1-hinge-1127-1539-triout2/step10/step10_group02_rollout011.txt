def _continuous_sa(a, conv, target_sum, deadline):
    n = len(a)
    best_a = a.copy()
    best_M = conv.max()
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_M = best_M
    # precompute length conv_len = 2*n-1
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    start = time.time()
    start_temp = 0.05
    end_temp = 1e-6
    # Determine max_delta = 0.1 * target_sum maybe
    max_delta_frac = 0.2 # fraction of donor value max move
    while time.time() < deadline:
        elapsed = time.time() - start
        ratio = elapsed / total_time if total_time > 0 else 1.0
        temp = start_temp * (end_temp/start_temp) ** ratio

        # Choose donor index with probability proportional to a_i (or w_i)
        # Use probabilities = cur_a / cur_a.sum()
        donor = rng.choice(n, p=cur_a/cur_a.sum())
        if cur_a[donor] <= 0:
            continue
        # Choose receiver distinct index randomly (uniform)
        receiver = rng.integers(0, n-1)
        while receiver == donor:
            receiver = rng.integers(0, n-1)

        # delta up to limited fraction of donor's value
        max_delta = max_delta_frac * cur_a[donor]
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # maybe also random sign? we only moving from donor to receiver (donor decreases).
        # Ensure after move donor remains nonnegative.
        # compute new conv via incremental update
        # copy conv for candidate
        new_conv = cur_conv.copy()
        # shift contributions
        # note: shift a using original cur_a
        new_conv[donor: donor+n] -= 2.0 * delta * cur_a
        new_conv[receiver: receiver+n] += 2.0 * delta * cur_a
        # delta^2 contributions
        idx2d = 2*donor
        new_conv[idx2d] += delta*delta
        idx2r = 2*receiver
        new_conv[idx2r] += delta*delta
        idxij = donor + receiver
        new_conv[idxij] -= 2.0 * delta*delta

        new_M = new_conv.max()
        # acceptance
        if new_M < cur_M or rng.random() < np.exp(-(new_M - cur_M) / max(temp, 1e-12)):
            # accept
            cur_a[donor] -= delta
            cur_a[receiver] += delta
            cur_conv = new_conv
            cur_M = new_M
            # update best
            if cur_M < best_M:
                best_M = cur_M
                best_a = cur_a.copy()
                # optionally print
                print(f"[{time.time():.2f}] SA new best M={best_M:.6f} it={int(elapsed*1000)}")
        # else reject: do nothing

    return best_a, best_M

"""
Propose a step‑function minimizing 2·n·max(f∗f)/(∑f)².

The search pipeline:
  1.  Initialise from a few power‑law shapes (exponents 0.45‑0.60) and,
      if available, the previous best sequence.  All candidates are projected
      onto the ℓ¹‑simplex {x≥0, Σx = √(2n)}.
  2.  Soft‑max projected gradient descent (with momentum) – fast descent on a
      smooth approximation of the peak of the convolution.
  3.  Exact max sub‑gradient descent – uses the true max‑convolution
      sub‑gradient.
  4.  Deterministic mass‑transfer refinement – greedy moves that shift mass
      from positions contributing most to the current peak towards low‑contrib
      positions, with an annealed acceptance probability.
  5.  Permutation annealing – simulated‑annealing swaps of two entries.
  6.  Continuous simulated‑annealing on the *values* (mass‑transfer with
      acceptance of worsening moves).  The convolution is updated
      incrementally, i.e. in O(n) per move, so many SA steps can be performed
      within the remaining time budget.
  7.  A short high‑β soft‑max GD “polish’’ phase.

All heavy convolutions use FFT (O(n log n)).  The routine respects the
overall 1000 s budget, keeps the best sequence found in the global
``height_sequence_1`` variable for warm‑starts, and returns the final list
of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective value 2·n·max(a*a)/(∑a)².
    When ∑a = √(2n) this equals the max‑convolution value.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short GD runs with jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves targeting the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase S – Continuous simulated annealing (mass transfer with acceptance)
# ----------------------------------------------------------------------


def _continuous_sa(a0: np.ndarray,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Continuous SA that moves a small amount of mass from a donor to a receiver.
    The convolution is updated incrementally in O(n) per move.
    Returns (best_sequence, best_max_conv).
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)                # full convolution (length 2n‑1)
    cur_M = float(conv.max())
    best_a = a.copy()
    best_M = cur_M

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0.0:
        return best_a, best_M

    start_temp = 0.05      # absolute temperature (tuned empirically)
    end_temp = 1e-6

    max_delta_frac = 0.15  # relative to donor's current mass

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        ratio = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** ratio

        # pick donor proportional to its mass (larger entries more likely)
        probs = a / a.sum()
        donor = rng.choice(n, p=probs)
        if a[donor] <= 0.0:
            continue

        # choose distinct receiver uniformly
        receiver = rng.integers(0, n - 1)
        while receiver == donor:
            receiver = rng.integers(0, n - 1)

        # limit the transferred mass
        max_delta = max_delta_frac * a[donor]
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # ---- incremental convolution update ----
        new_conv = conv.copy()
        # 2*a*d term
        new_conv[donor: donor + n] -= 2.0 * delta * a
        new_conv[receiver: receiver + n] += 2.0 * delta * a
        # d*d term (few isolated entries)
        idx2d = 2 * donor
        new_conv[idx2d] += delta * delta
        idx2r = 2 * receiver
        new_conv[idx2r] += delta * delta
        idxij = donor + receiver
        new_conv[idxij] -= 2.0 * delta * delta

        new_M = float(new_conv.max())

        # acceptance rule
        if new_M < cur_M or rng.random() < np.exp(-(new_M - cur_M) / max(temp, 1e-12)):
            # accept
            a[donor] -= delta
            a[receiver] += delta
            conv = new_conv
            cur_M = new_M
            if cur_M < best_M:
                best_M = cur_M
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best M={best_M:.6f}")
        # else reject – nothing changes

        it += 1

    return best_a, best_M


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The pipeline follows the phases described in the module doc‑string.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Problem size selection
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        # pick a size that balances quality and speed
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build initial pool of candidates
    # --------------------------------------------------------------
    init_seqs: list[np.ndarray] = []

    # power‑law seeds (a slightly finer grid than before)
    for p in (0.45, 0.48, 0.50, 0.53, 0.55, 0.58, 0.60):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    if "height_sequence_1" in globals():
        init_seqs.append(_resize_sequence(prev_seq, n))

    rng = np.random.default_rng()
    # jittered variants
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # a couple of random permutations of power‑law candidates
    for p in (0.5, 0.55):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # --------------------------------------------------------------
    # Phase A – soft‑max GD (restarts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.25

    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        cand, val = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New global best after GD = {best_score:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase X – exact max sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = time.time() + (deadline - time.time()) * 0.15
        cand, val = _subgradient_descent(best_seq, target_sum, sg_deadline,
                                         step_factor=0.6, gamma=0.95)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.25
        cand, val = _deterministic_mass_transfer(best_seq, mt_deadline,
                                                target_sum,
                                                start_temp=0.5,
                                                end_temp=1e-4,
                                                low_frac=0.05)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – re‑ordering heuristics
    # --------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        cand, val = _permutation_annealing(best_seq, target_sum, perm_deadline,
                                           init_score=best_score)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase S – continuous simulated annealing on the values
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # use roughly 40 % of the remaining time for SA
        sa_deadline = time.time() + remaining * 0.40
        cand, max_conv = _continuous_sa(best_seq, target_sum, sa_deadline)
        # note: max_conv equals the objective because Σa = √(2n)
        if max_conv < best_score:
            best_score = max_conv
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After continuous SA best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – final soft‑max polishing (short)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=3e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final post‑processing
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    # trim trailing near‑zero entries
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # evaluate once for logging purposes (evaluate_sequence is provided by the harness)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
