def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5

    n = choose based on deadline and budget maybe 20000.

    target_sum = sqrt(2*n)

    # Warm start
    if 'height_sequence_1' in globals():
        init = np.array(globals()['height_sequence_1'], dtype=np.float64)
        if init.shape[0] != n:
            # upsample or downsample accordingly using interpolation
            init = _resize_sequence(init, n)
    else:
        init = _power_law_sequence(n, 0.6, target_sum)  # initial guess

    best_seq = init.copy()
    best_score = _score(best_seq)

    # Stage A: param search. Use small n = n//4 for speed? Actually we could just do over full n maybe slower.
    param_end = min(deadline, time.time() + (deadline - time.time()) * 0.07)
    best_seq, best_score = parametric_search_small(n, target_sum, param_end, best_seq, best_score)

    # Stage B: mirror descent with softmax increasing beta.
    # Mirror descent using multiplicative updates.
    a = best_seq.copy()
    a = a * (target_sum / a.sum())
    beta = 1.0
    max_beta = 1e7
    eta_base = 0.01
    while time.time() < deadline:
        grad, conv = _gradient_softmax(a, beta)
        # step size: adapt to max gradient magnitude
        step = eta_base / (beta + 1.0)
        a = a * np.exp(-step * grad)
        a *= target_sum / a.sum()
        # update best
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"Mirror best {best_score:.6f} beta {beta}")
        # increase beta gradually
        if beta < max_beta:
            beta *= 1.2
        # optional jitter
        if np.random.rand() < 0.001:
            a += np.random.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a, 0)
            a *= target_sum / a.sum()
    # Stage C: mass transfer local search (simulated annealing style)
    a = best_seq.copy()
    a *= target_sum / a.sum()
    cur_score = best_score
    # compute conv for speed each iteration? We'll recompute as needed.
    temperature = cur_score * 0.02  # starting temperature
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]
        # donor selection weighted by weight
        if weight.sum() == 0:
            donor = np.random.randint(n)
        else:
            donor = np.random.choice(idx, p=weight/weight.sum())
        # choose receiver: sample from indices with low weight
        low_thr = np.percentile(weight, 5)
        low_candidates = idx[weight <= low_thr]
        if low_candidates.size:
            receiver = np.random.choice(low_candidates)
        else:
            receiver = np.random.randint(n)
        # transfer amount epsilon
        max_frac = 0.02
        epsilon = a[donor] * max_frac * np.random.rand()
        if epsilon <= 0:
            continue
        a_new = a.copy()
        a_new[donor] -= epsilon
        a_new[receiver] += epsilon
        a_new *= target_sum / a_new.sum()
        new_score = _score(a_new)
        it += 1
        # linear cooling
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = temperature * (1 - frac)
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] SA improve {best_score:.6f}")
        # occasional jitter
        if it % 1000 == 0:
            a += np.random.normal(scale=target_sum*0.0005, size=n)
            a = np.maximum(a, 0)
            a *= target_sum / a.sum()
            cur_score = _score(a)

    # Stage D: final subgradient descent.
    a = best_seq.copy()
    a *= target_sum / a.sum()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        # compute subgradient
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idx = np.arange(lo, hi+1)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]
        # step size schedule
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        eta = 0.5 / (it + 1)  # decreasing step
        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Subgradient improve {best_score:.6f}")
        it += 1
        if it % 200 == 0:
            # jitter
            a += np.random.normal(scale=target_sum*0.0005, size=n)
            a = np.maximum(a, 0)
            a *= target_sum / a.sum()
    # final evaluation and return
    arr = np.array(best_seq)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:

    seq = arr.tolist()
    try:
        val = evaluate_sequence(seq)
        print(f"Final score {val:.6f}")
    except Exception as e:
        print(f"Evaluation error: {e}")
    globals()['height_sequence_1'] = seq
    return seq

"""
Hybrid optimization for minimizing the max‑convolution score of a non‑negative
step sequence.

The routine works in several stages:

1.  Warm‑start – a power‑law / exponential family is explored on a reduced
    size, plus a closed‑form “sqrt‑derivative’’ (≈d√x) candidate.  The best
    candidate is resized to the final length.

2.  Mirror‑descent on a soft‑max surrogate of the max‑convolution.  The
    multiplicative (exponential) update keeps the variables positive and a
    projection onto the ℓ¹‑simplex enforces the required norm.

3.  Mass‑transfer simulated‑annealing – repeatedly moves a modest amount of
    mass from the index that contributes most to the current maximal
    convolution entry to a low‑contributing index, accepting worsening moves
    with a linearly cooling temperature.

4.  Exact sub‑gradient descent on the true max‑convolution entry, with a
    decreasing step‑size and occasional jitter.

All stages stop as soon as the global deadline (budget_s seconds) is reached,
and the best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 length adequate for a linear convolution of size 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  Assumes sum>0."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence a_i = (i+1)^(-p), normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _sqrt_derivative_sequence(n: int, target_sum: float) -> np.ndarray:
    """Discrete derivative of sqrt: a_i = √(i+1) - √i, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1) - np.sqrt(i)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution).  grad has the same shape as a.
    """
    n = a.shape[0]
    size = _fft_len(n)

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()                      # soft‑max weights

    # grad = 2 * (reverse(a) convolve p)  (central n entries)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Exhaustive grid search over a two‑parameter family
    a_i = (i+1)^(-p)·exp(-λ·i/small_n) and random refinements.
    Returns the best sequence and its score.
    """
    p_vals = np.linspace(0.2, 1.5, 21)          # grid on p
    lam_vals = np.linspace(0.0, 12.0, 25)      # grid on λ
    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    # exhaustive grid
    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    # random refinements
    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.5)
        lam = rng.exponential(1.0) * 4.0   # favour moderate λ
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score

# Mirror‑descent (exponential) on soft‑max surrogate

def _mirror_descent_exp(a0: np.ndarray,
                        target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative (mirror) descent on the soft‑max surrogate.
    Uses an ever‑increasing β to sharpen the approximation.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1e7
    base_eta = 0.04                     # a bit larger than the previous 0.06/step
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)                # multiplicative update
        a = a * (target_sum / a.sum())             # re‑normalise

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Mirror it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase softness
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.8, max_beta)

        it += 1

    return best_a, best_val

# Simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_sa(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap SA that repeatedly moves a small amount of mass from a high‑weight
    index (w.r.t. the current maximal convolution entry) to a low‑weight one.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    T0 = best_val * 0.015                # initial temperature proportional to score

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # indices that participate in the k_max entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # donor – weighted by its contribution to the peak
        wsum = weight.sum()
        if wsum == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        # receiver – prefer indices with low contribution
        low_thr = np.percentile(weight, 5)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # transfer a modest fraction of donor's mass
        max_frac = 0.03
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())
        new_val = _score(a_new)

        # linear cooling
        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        elif new_val < _score(a):  # accept if improves current
            a = a_new
        elif rng.random() < np.exp(-(new_val - _score(a)) / T):
            a = a_new

        # occasional random jitter to diversify search
        if rng.random() < 0.003:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true maximal convolution entry.
    Step size decreases like 1/√t and a small jitter is added periodically.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of (a*a)_{k_max}
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        eta = 0.5 / np.sqrt(it + 1.0)          # decreasing step
        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")

        # jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val
