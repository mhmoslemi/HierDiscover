def simulated_annealing(a, target_sum, deadline):
   n = len(a)
   b = _conv_fft(a)
   best_a = a.copy()
   best_val = _score(a)
   # initial temperature
   T0 = best_val * 0.1
   start_time = time.time()
   while time.time() < deadline:
       elapsed = time.time() - start_time
       total = deadline - start_time
       # linear temperature schedule
       T = T0 * (1 - elapsed / total)
       # recompute k_max perhaps every N steps
       if steps % recompute_interval == 0:
           b = _conv_fft(a)  # full recompute for accuracy
           k_max = int(np.argmax(b))
       # choose donor based on contributions to b[k_max]
       lo = max(0, k_max - (n - 1))
       hi = min(k_max, n - 1)
       idx = np.arange(lo, hi+1)
       contrib = a[idx] * a[k_max - idx]
       if contrib.sum() > 0:
           donor = rng.choice(idx, p=contrib / contrib.sum())
       else:
           donor = rng.integers(0, n)
       # choose receiver
       # choose from tail with low contribution to central region: we can compute same array of contributions for each index:
       # not needed, we just sample uniformly from all indices except donor
       while True:
            receiver = rng.integers(0, n)
            if receiver != donor:
                break
       # propose delta
       max_delta = min(a[donor], target_sum * 0.05 * (T/T0+0.1)) # scaled fraction
       if max_delta <= 0:
           continue
       delta = rng.random() * max_delta
       # compute b_new via O(n) update
       # update only after compute for acceptance
       b_new = b.copy()
       # -2*delta*a at donor shift
       b_new[donor:donor+n] -= 2 * delta * a
       # +2*delta*a at receiver shift
       b_new[receiver:receiver+n] += 2 * delta * a
       # adjust d*d
       b_new[2*donor] += delta*delta
       b_new[2*receiver] += delta*delta
       if donor != receiver:
           b_new[donor+receiver] -= 2 * delta * delta
       new_max = b_new.max()
       new_val = float(2.0 * n * new_max / (target_sum*target_sum))
       # acceptance
       if new_val < best_val or rng.random() < np.exp(-(new_val - best_val)/max(T, 1e-12)):
           # accept move for a
           a[donor] -= delta
           a[receiver] += delta
           b = b_new
           if new_val < best_val:
               best_val = new_val
               best_a = a.copy()
               # maybe print improvement
       steps += 1
```

def _simulated_annealing_refine(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)  # exact conv
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    total_time = deadline - start
    # Initial temperature maybe proportional to best_val
    T0 = best_val * 0.1  # tuning factor
    steps = 0
    recompute_every = max(1, int(n / 20))  # or time-based
    last_recompute = start
    while time.time() < deadline:
        # compute temperature linear decay
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / total_time)
        if T < 1e-12:
            T = 1e-12
        # locate current max index
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(0, n)
        # receiver uniformly from all indices not donor
        while True:
            receiver = rng.integers(0, n)
            if receiver != donor:
                break
        # Determine max delta: allow larger moves at high T
        # max fraction of donor's mass
        max_frac = 0.5 * (T / T0) + 0.01  # between 0.01 and 0.51
        max_delta = min(a[donor] * max_frac, target_sum * 0.02 * (T / T0 + 0.1))
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # compute new convolution via O(n) update
        # we can copy b
        b_new = b.copy()
        # update linear part
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # add quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))
        # acceptance rule
        accept = False
        if new_val < best_val:
            accept = True
        else:
            prob = np.exp(-(new_val - best_val) / T)
            if rng.random() < prob:
                accept = True
        if accept:

            a[receiver] += delta

                best_a = a.copy()

"""
Search for a non‑negative step sequence minimizing

    2·n·max(a*a) / (sum(a)^2)

The algorithm consists of five phases that respect a 1000 s budget.

1. Warm‑start from the best known construction (`height_sequence_1`) or a simple
   decreasing power‑law, resized to the chosen length `n` and normalised so that
   Σa = √(2 n).

2. Phase A – exponential (mirror) descent on a soft‑max surrogate of the
   maximal convolution entry.

3. Phase B – multiplicative sub‑gradient descent directly on the true
   maximal convolution entry.

4. Phase C – projected gradient descent with momentum and numerous random
   restarts.

5. Phase D – a dedicated simulated‑annealing refinement.  Each move transfers
   a small amount of mass `δ` from a “donor’’ index (chosen proportionally to
   its contribution to the current maximal convolution entry) to a random
   “receiver’’ index.  The convolution is updated exactly in O(n) time using
   the identity (a+d)*(a+d) = a*a + 2·a*d + d*d, so thousands of moves can be
   performed per second.  The temperature schedule is linear and the move
   size scales with the temperature, allowing large exploratory steps early
   and fine tuning later.

The best sequence found is returned as a plain list of floats and stored in
the global variable `height_sequence_1` for warm‑starts in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phase‑A and Phase‑C)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) *convolve* p) – length n
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b

# Phase A – exponential (mirror) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

            a += jitter

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – simulated‑annealing refinement (new)
# ----------------------------------------------------------------------


def _simulated_annealing_refine(a0: np.ndarray,
                               target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Exact O(n) SA on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                       # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # Temperature proportional to the current objective.
    T0 = best_val * 0.1 + 1e-12
    steps = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        if total_time <= 0:
            break
        T = T0 * (1.0 - elapsed / total_time)
        if T < 1e-12:
            T = 1e-12

        # current maximal convolution index
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – proportionally to its contribution to the maximal entry
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(0, n)

        # receiver – any other index (uniform)
        while True:
            receiver = rng.integers(0, n)
            if receiver != donor:
                break

        # size of the move – larger when temperature is high
        max_frac = 0.5 * (T / T0) + 0.02          # between ~0.02 and ~0.52
        max_delta = min(a[donor] * max_frac,
                        target_sum * 0.02 * (T / T0 + 0.1))
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) exact update of the convolution (see analysis)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        else:
            prob = np.exp(-(new_val - best_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # Periodic full recomputation to keep numerical errors in check
        if steps % 800 == 0:
            b = _conv_fft(a)

        steps += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – exponential descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.12 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")
    else:
        best_seq = init
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase B – subgradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.18 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing_refine(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
        print(f"[{time.time():.2f}] After simulated annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # provided by the runner
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
