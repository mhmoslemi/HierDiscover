    while len(S) < size_target and attempts < size_target*1000:

"""
Improved optimizer for the step‑function benchmark.

The strategy combines several complementary ideas:

* **Fast parametric pre‑search** – power‑law and exponential
  families give a good starting point.
* **Sub‑gradient descent on the true max‑convolution** – we compute the
  exact sub‑gradient of `max(a*a)` (no soft‑max approximation) and
  perform a projected sub‑gradient descent with a simple 1/√t step‑size.
* **Exact deterministic mass‑transfer** – a greedy “water‑filling”
  move analytically reduces the current maximal convolution entry to the
  second‑largest one.
* **Final Adam‑style polishing** – the same optimiser used in the
  reference code, but run only for a short remaining budget.
* **Early exit** – as soon as a sequence with score < 1.503 is found the
  routine stops and returns it.

All phases respect the global time budget (default 1000 s).  The
returned sequence is trimmed of trailing zeros.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σ x = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score (the benchmark objective)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """2·n·max(a*a) / (Σa)² – the benchmark score."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^{-p} scaled to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^{-p}·exp(-q·i/n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (exact)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Compute a sub‑gradient of f(a)=max_k (a*a)[k] and return
    (grad, max_val).  The sub‑gradient is
        g_i = 2·∑_{k∈K} a_{k-i},
    where K is the set of indices attaining the maximum.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    # active indices (allow a tiny tolerance)
    active = np.where(conv >= max_val - 1e-12)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]

    return grad, max_val


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         lr0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‐convolution.
    Step size follows η_t = lr0 / sqrt(t+1).
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _subgradient_max(a)

        # diminishing step size
        lr = lr0 / np.sqrt(it + 1.0)
        a -= lr * grad
        a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val - 1e-12:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy reduction of the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[k0 - idx]
        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            # try another maximal index if available
            if len(idx_max) > 1:
                k0 = idx_max[1]
                lo = max(0, k0 - (n - 1))
                hi = min(k0, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                w = a[idx] + a[k0 - idx]
                donor = idx[np.argmax(w)]
                receiver = idx[np.argmin(w)]
                contrib_d = a[k0 - donor]
                contrib_r = a[k0 - receiver]
                denom = 2.0 * (contrib_d - contrib_r)
                if denom <= 0.0:
                    break
            else:
                break

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.5, delta_needed * 0.99)

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine respects the time budget, prints progress and returns the
    best sequence found (trailing zeros are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (larger n gives a larger design space)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) – scale‑invariant

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # (a) pure power‑law grid
    for p in np.linspace(0.30, 0.85, 15):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail grid
    for p in np.linspace(0.30, 0.85, 12):
        for q in np.linspace(0.0, 6.0, 12):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) random dense start
    seq = rng.random(n)
    seq = _project_to_simplex(seq, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # (d) reuse previous best if available
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_score:
            best_score, best_seq = val, prev.copy()

    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Sub‑gradient descent with restarts (main optimisation)
    # ------------------------------------------------------------------
    subgrad_budget = (deadline - time.time()) * 0.45
    subgrad_deadline = time.time() + subgrad_budget

    while time.time() < subgrad_deadline - 2.0:
        # jittered start around current best
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start = best_seq + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        # allocate a few seconds for each run (max 25 s or remaining time)
        run_deadline = min(time.time() + 25.0, subgrad_deadline)
        a_opt, val_opt = _subgradient_descent(start,
                                              target_sum,
                                              run_deadline,
                                              lr0=target_sum * 0.01)

        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Sub‑grad restart improved best = {best_score:.6f}")
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact(best_seq,
                                                            deadline,
                                                            target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")
            if best_score < 1.503:
                break

        # short Adam‑GD polish after each sweep (helps escaping tiny stalls)
        if time.time() < deadline - 3.0:
            a_tmp, val_tmp = _adam_gradient_descent(best_seq,
                                                    time.time() + 5.0,
                                                    target_sum,
                                                    max_beta=3e5,
                                                    step_factor=0.7,
                                                    gamma=0.995)
            if val_tmp < best_score:
                best_score, best_seq = val_tmp, a_tmp.copy()
                print(f"[{time.time():.2f}] Adam polish improved best = {best_score:.6f}")
                if best_score < 1.503:
                    break

        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Final polishing (high β Adam)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final verification (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
