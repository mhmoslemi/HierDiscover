def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return float(conv.max())
```

def _swap_annealing(a, target_sum, deadline, init_score=None):
  cur_seq = a.copy()
  cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
  best_seq = cur_seq.copy()
  best_score = cur_score
  start_time = time.time()
  n = len(cur_seq)
  rng = np.random.default_rng()
  while time.time() < deadline:
      elapsed = time.time() - start_time
      ratio = elapsed / (deadline - start_time)
      temp = 0.2 * (1e-5 / 0.2) ** ratio  # using same schedule
      i,j = rng.integers(0,n, size=2)
      if i==j: continue
      cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
      new_score = _max_convolution_fft(cur_seq)
      delta = new_score - cur_score
      if delta < 0 or rng.random() < np.exp(-delta / max(temp,1e-12)):
          cur_score = new_score
          if new_score < best_score:
              best_score = new_score
              best_seq = cur_seq.copy()
              print(f"[{time.time():.2f}] Swap anneal new best: {best_score:.6f}")
      else:
          # revert
          cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
      # occasional random shuffle near end
      if ratio > 0.9 and rng.random() < 0.02:
          rng.shuffle(cur_seq)
          cur_score = _max_convolution_fft(cur_seq)
  return best_seq, best_score

"""
Improved search for a low‐peak self‑convolution step function.

The algorithm consists of the following stages
------------------------------------------------
1. **Problem size** – choose a length *n* (larger for larger budgets).
2. **Initial pool** – generate a diverse set of candidates:
   power‑law, exponential, uniform, jittered copies and a warm‑start
   from the previous run.
3. **Soft‑max projected gradient descent** – fast reduction of the peak
   using a differentiable soft‑max approximation of the max‑convolution.
4. **Exact sub‑gradient descent** – directly optimises the true max‑convolution.
5. **Deterministic mass‑transfer** – moves mass from indices heavily
   contributing to the current peak to low‑contributing indices.
6. **Swap‑annealing** – simulated‑annealing swaps of two positions.
7. **Final polishing** – high‑β soft‑max GD for fine tweaking.

All phases respect the overall time budget (default 1000 s).  The best
sequence found is stored in the global variable `height_sequence_1`
for warm‑starts in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (computed via FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Return exp(-lam·i) normalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where conv is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected soft‑max gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 1e5,
                                   step_factor: float = 0.7,
                                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD runs with jitter and keep the best."""
    best_a = init.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase X – Exact max sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves targeting the current maximal convolution entry."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # choose a low‑contributing receiver among the worst low_frac
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

# Phase D – Swap‑based simulated annealing
# ----------------------------------------------------------------------


def _swap_annealing(a: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Swap anneal new best={best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full reshuffle near the end
        if ratio > 0.9 and rng.random() < 0.02:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low max self‑convolution.
    The function respects the given time budget (seconds) and returns the
    best sequence found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 400:
        n = 13000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a diverse initial pool
    # --------------------------------------------------------------
    init_seqs = []

    # Power‑law family
    for p in np.linspace(0.35, 0.80, 10):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # Exponential family
    for lam in np.linspace(0.001, 0.02, 10):
        init_seqs.append(_exponential_sequence(n, lam, target_sum))

    # Uniform sequence (baseline)
    init_seqs.append(np.full(n, target_sum / n))

    rng = np.random.default_rng()

    # Jittered copies of the above
    for base in init_seqs[:]:
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq = base + jitter
        seq = np.maximum(seq, 0.0)
        seq = _project_to_simplex(seq, target_sum)
        init_seqs.append(seq)

    # Warm‑start from previous run, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        resized = _resize_sequence(prev, n)
        resized = _project_to_simplex(resized, target_sum)
        init_seqs.append(resized)

    # Random permutations of a few representatives
    for base in init_seqs[:5]:
        perm = base.copy()
        rng.shuffle(perm)
        init_seqs.append(perm)

    # --------------------------------------------------------------
    # Phase A – Soft‑max projected gradient descent (exploration)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    phase_a_deadline = time.time() + (deadline - time.time()) * 0.20
    for init in init_seqs:
        if time.time() >= phase_a_deadline:
            break
        cand, val = _gradient_descent_with_restarts(init, target_sum, phase_a_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New global best after GD = {best_score:.6f}")

    # Fallback in the unlikely event that no candidate was produced
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Phase X – Exact sub‑gradient descent (refinement)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sg_deadline = time.time() + (deadline - time.time()) * 0.15
        cand, val = _subgradient_descent(best_seq, target_sum, sg_deadline,
                                         step_factor=0.8, gamma=0.95)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        cand, val = _deterministic_mass_transfer(best_seq, mt_deadline,
                                                target_sum,
                                                start_temp=0.4,
                                                end_temp=1e-5,
                                                low_frac=0.04)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Swap‑based simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sa_deadline = time.time() + (deadline - time.time()) * 0.20
        cand, val = _swap_annealing(best_seq, target_sum, sa_deadline,
                                    init_score=_max_convolution_fft(best_seq))
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After swap annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing (high‑β soft‑max GD)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        cand, val = _annealed_gradient_descent_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional logging of the exact evaluation result
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # Store for warm‑starts in subsequent runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
