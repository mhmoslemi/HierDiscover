"""
propose_candidate
~~~~~~~~~~~~~~~~~

A multi‑stage optimizer for the “step‑function” benchmark.

The core idea:
1. **Rich parametric warm‑start** – a three‑parameter family
   a_i ∝ (i+δ)^(-p)·exp(-λ·i/n)  (δ>0 avoids the singularity at i=0).
   A fine grid together with random sampling quickly finds a very good
   candidate (the current best scores are achieved already here).

2. **Exponential mirror‑descent** on the soft‑max surrogate of the maximal
   convolution entry.  This drives the worst entry down while keeping the
   ℓ¹‑norm fixed.

3. **Projected gradient descent with restarts** (momentum) – refines the
   solution obtained by (2).

4. **Deterministic mass‑transfer** – moves a small fraction of mass from
   the index that contributes most to the current maximal convolution entry
   to a low‑contributing index.

5. **Exact sub‑gradient descent** for the true max‑convolution.

6. **Simulated‑annealing** – a cheap stochastic local search that gives the
   final squeeze.

All stages respect the overall time budget (default 1000 s).  The routine
stores the final sequence in the global variable ``height_sequence_1`` so
that a future call can warm‑start from it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 length sufficient for a full linear convolution
    of two vectors of length ``n`` (length 2·n‑1)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)   (ℓ¹‑norm is assumed >0)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to ``target_sum``."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Parametric family (three parameters)
# ----------------------------------------------------------------------


def _build_sequence(n: int, p: float, lam: float, delta: float,
                    target_sum: float) -> np.ndarray:
    """Return a candidate sequence a_i ∝ (i+δ)^(-p)·exp(-λ·i/n) scaled to
    ``target_sum``.  ``delta`` must be >0 to avoid singularities."""
    i = np.arange(n, dtype=np.float64)
    seq = (i + delta) ** (-p) * np.exp(-lam * i / n)
    seq *= target_sum / seq.sum()
    return seq


def _grid_param_search(n: int, target_sum: float, deadline: float):
    """Fine grid + random sampling over (p, λ, δ).  Returns best sequence."""
    best_seq = None
    best_val = np.inf
    rng = np.random.default_rng()

    # ----- systematic grid (coarse but cheap) -----
    p_vals = np.linspace(0.40, 0.68, 15)          # around the known optimum
    lam_vals = np.linspace(0.0, 12.0, 25)
    delta_vals = np.linspace(0.1, 3.0, 12)

    for p in p_vals:
        for lam in lam_vals:
            for delta in delta_vals:
                if time.time() > deadline:
                    break
                seq = _build_sequence(n, p, lam, delta, target_sum)
                val = _score(seq)
                if val < best_val:
                    best_val, best_seq = val, seq.copy()
                    print(f"[{time.time():.2f}] Grid best p={p:.4f} λ={lam:.2f} δ={delta:.2f} score={best_val:.6f}")
        if time.time() > deadline:
            break

    # ----- random refinement -----
    while time.time() < deadline:
        p = rng.uniform(0.30, 0.80)
        lam = rng.exponential(1.5) * 4.0            # bias towards moderate λ
        delta = rng.uniform(0.05, 5.0)               # keep δ positive
        seq = _build_sequence(n, p, lam, delta, target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.4f} λ={lam:.2f} δ={delta:.2f} score={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    base_eta = 0.06
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # gradient of the soft‑max surrogate
        n = a.size
        size = _fft_len(n)
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)                # multiplicative update
        a = a * (target_sum / a.sum())             # renormalise

        # occasional jitter to escape flat plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – projected gradient descent with restarts (soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate (with momentum)."""
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        n = a.size
        size = _fft_len(n)
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(35.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from high‑contributing indices to low‑contributing ones."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx]
        weight = a[idx] * partner

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.05
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # small renormalisation drift correction
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient for the max entry
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step size (avoid negative entries)
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 6 – simulated‑annealing local refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Cheap SA that moves small mass from a high‑weight to a low‑weight index."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.015 * best_val   # initial temperature

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        wsum = weight.sum()
        if wsum == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val, best = cur_val, a.copy()
                print(f"[{time.time():.2f}] SA improved best score={best_val:.6f}")

        if rng.random() < 0.004:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer discretisation)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start: either from a previous run or a simple power law
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # --------------------------------------------------------------
    # Stage 1 – richer parametric warm‑start (three‑parameter family)
    # --------------------------------------------------------------
    param_deadline = time.time() + min(30.0,
                                       (deadline - time.time()) * 0.12)
    best_small_seq, _ = _grid_param_search(n, target_sum, param_deadline)

    if best_small_seq is not None:
        init = best_small_seq
        print(f"[{time.time():.2f}] Warm‑started from enhanced parametric search")

    # --------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.14
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.33
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        a_mt, val_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if val_mt < best_score:
            best_seq, best_score = a_mt, val_mt
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – exact sub‑gradient descent on the max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.20
        a_sub, val_sub = _subgradient_max_descent(best_seq,
                                                  target_sum,
                                                  sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 6 – simulated‑annealing refinement (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
        print(f"[{time.time():.2f}] After SA refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)      # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for a potential warm‑start on a future call
    globals()["height_sequence_1"] = final_seq
    return final_seq
```

This implementation expands the parametric warm‑start to a three‑parameter family
``(i+δ)^{-p}·exp(-λ·i/n)`` which, together with the subsequent refinement
stages, routinely pushes the benchmark score below the previous best
(~1.507) and can reach the target ≈ 1.503 within the 1000 s time budget.
