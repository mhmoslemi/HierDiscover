"""
Propose a step‑function sequence that minimizes the benchmark score.

The algorithm builds on the previous multi‑phase scheme (soft‑max mirror
descent → sub‑gradient descent on the true max‑convolution → projected GD
with restarts → local mass‑transfer refinement) and adds a final
high‑β soft‑max fine‑tuning phase.  The whole search respects a 1000 s
budget, repeatedly re‑initialising from the current best (or a random
Dirichlet) and keeping the globally best sequence found.

Key ideas
---------

* Use a precise FFT‑based gradient of the soft‑max surrogate, with a
  convolution size large enough to avoid wrap‑around artefacts.
* Allocate the time budget dynamically among the phases, keeping a few
  seconds for a final low‑step high‑β soft‑max refinement that can
  push the score below the previous best (≈1.508).
* Perform several outer restarts (while time permits) to escape possible
  local minima; each restart runs the full pipeline on a jittered copy
  of the current global best or on a fresh random start.
* All updates respect the simplex constraint Σa = √(2 n) via Euclidean
  projection.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  For Σa = √(2 n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_conv_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Soft‑max surrogate utilities

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]

    # --- convolution a*a -------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # --- soft‑max weights -------------------------------------------------
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))          # stable: max exponent = 1
    p = exps / exps.sum()                     # distribution over 2·n‑1 positions

    # --- gradient = 2·reverse(a) convolve p -----------------------------
    # linear convolution length = n + (2n‑1) – 1 = 3n‑2
    size2 = 1 << ((3 * n - 2).bit_length())
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size2)[: 3 * n - 2]

    # contribution for index i lies at position (n‑1 + i)
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b

def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1 << 20          # ≈ 1 048 576
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to keep diversity
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β geometrically
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule η∈[η_start, η_end]
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _gradient_softmax_with_momentum(a: np.ndarray,
                                    beta: float,
                                    momentum: np.ndarray,
                                    gamma: float,
                                    step_base: float) -> tuple[np.ndarray, np.ndarray]:
    """One GD step on the soft‑max surrogate with momentum."""
    g, _ = _gradient_softmax(a, beta)
    momentum = gamma * momentum + g
    norm = np.linalg.norm(momentum)
    if norm > 0:
        lr = step_base / (norm * (beta + 1.0))
        a = a - lr * momentum
    return a, momentum


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1 << 20
    step_base = target_sum * 0.45
    gamma = 0.93
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        a, momentum = _gradient_softmax_with_momentum(a, beta, momentum, gamma, step_base)
        a = _project_to_simplex(a, target_sum)

        # jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               target_sum,
                                               time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature

    while time.time() < deadline:
        # locate donor and receiver based on current max entry
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contribution of each pair

        # choose a donor proportionally to its contribution
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # choose a receiver among indices that currently contribute little
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # propose a tiny transfer
        max_delta = min(a[donor] * 0.08, target_sum * 1.2e-3)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution exactly (O(n) slices)
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²) – only affect a few entries
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # simulated annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - start - elapsed > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        # occasional tiny jitter between random positions
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 4e-4)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

    return best_a, best_val

def _optimize_one_run(a_init: np.ndarray,
                      target_sum: float,

    Returns the best sequence and its score.
    """
    # Phase A – exponential descent
    exp_deadline = time.time() + 0.15 * (deadline - time.time())
    a, val = _exponential_descent(a_init, target_sum, exp_deadline)
    best_seq, best_val = a, val
    print(f"[{time.time():.2f}] After ExpDescent best={best_val:.6f}")

    # Phase B – subgradient descent
    if time.time() < deadline:
        sub_deadline = time.time() + 0.20 * (deadline - time.time())
        a, val = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val < best_val:
            best_seq, best_val = a, val

    # Phase C – GD with restarts
    if time.time() < deadline:

        if val < best_val:

    # Phase D – local refinement
    if time.time() < deadline:
        loc_deadline = time.time() + 0.20 * (deadline - time.time())
        a, val = _local_refine(best_seq, target_sum, loc_deadline)

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget and returns the best sequence
    found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5       # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 16000
    elif budget_s >= 800:
        n = 14000
    elif budget_s >= 600:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start from the global best, otherwise use a power‑law fallback
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = _resize_sequence(np.array(globals()["height_sequence_1"],
                                            dtype=np.float64), n)
    else:
        init_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init_seq = init_seq * (target_sum / init_seq.sum())
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Outer restart loop: as long as we have a reasonable chunk of time,
    # run a complete optimisation starting from a jittered copy of the
    # current global best (or from scratch on the first iteration).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    run_idx = 0

    while time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        # each run gets at most 250 s or a fraction of the remaining budget
        run_budget = min(250.0, remaining * 0.8)
        run_deadline = time.time() + run_budget

        if best_seq is not None and rng.random() < 0.7:
            # jittered copy of the current best
            a0 = _resize_sequence(np.array(best_seq, dtype=np.float64), n)
            jitter = rng.normal(scale=target_sum * 0.015, size=n)
            a0 = np.maximum(a0 + jitter, 0.0)
        else:
            # fresh random start (Dirichlet‑like)
            a0 = rng.random(n)
        a0 = _project_to_simplex(a0, target_sum)

        seq, score = _optimize_one_run(a0, target_sum, run_deadline)

        if score < best_score:
            best_score, best_seq = score, seq
            print(f"[{time.time():.2f}] New global best after run {run_idx}: {best_score:.6f}")

        run_idx += 1

    # ------------------------------------------------------------------
    # Final clean‑up: drop trailing zeros, evaluate with the provided function,
    # store for warm‑starts and return.
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
