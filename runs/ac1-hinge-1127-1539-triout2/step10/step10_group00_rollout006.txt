"""
Propose a low‑score step‑function.

The method consists of several stages:

1. **Parametric warm‑start** – exhaustive grid + random search over a
   two‑parameter family a_i ∝ (i+1)^(-p)·exp(-λ·i/n) on a reduced size
   problem, then resize to the target length.

2. **Mirror‑descent (exponential update)** on a soft‑max surrogate of the
   maximum convolution entry.

3. **Projected gradient descent with random restarts** on the same surrogate.

4. **Deterministic mass‑transfer** – repeatedly moves a modest fraction
   of mass from the “worst” contributing index of the current maximal
   convolution entry to a low‑contribution index.

5. **Exact sub‑gradient descent** on the true maximal convolution entry.

6. **Smoothing + simulated‑annealing** – a few smoothing passes followed
   by a cheap SA that transfers small mass blocks while accepting
   uphill moves with a linearly cooling temperature.

All stages respect the global time budget (default 1000 s) and keep the
sequence non‑negative with ℓ¹‑norm √(2·n).  The best sequence found is
trimmed of trailing zeros and returned as a plain list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 length for a full linear convolution of size 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  Assumes ℓ¹‑norm >0."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) convolve p) and then take the central n entries
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 0 – richer parametric warm‑start on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search_extended(small_n: int,
                                target_sum: float,
                                deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Exhaustive grid + random refinement over
    a_i = (i+1)^(-p)·exp(-λ·i / small_n).
    """
    p_vals = np.linspace(0.2, 2.0, 37)       # finer grid for p
    lam_vals = np.linspace(0.0, 20.0, 41)    # allow stronger exponential decay
    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 2.0)
        lam = rng.uniform(0.0, 20.0)
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 1 – exponential (mirror) descent on soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror‑descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.06
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)               # multiplicative update
        a = a * (target_sum / a.sum())         # renormalise

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """PGD on the soft‑max surrogate (with momentum)."""
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short PGD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(35.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               target_sum,
                                               time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – deterministic mass‑transfer refinement (stronger)
# ----------------------------------------------------------------------


def _mass_transfer_refine_stronger(a0: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Move a modest fraction of mass from a high‑contributing index of the
    current maximal convolution entry to a low‑contributing one.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx]
        weight = a[idx] * partner

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 15)          # a slightly larger tail
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        # allow up to 12 % of the donor's mass to move in one step
        max_frac = 0.12
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise for numerical drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 4 – exact sub‑gradient descent on maximal convolution entry
# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent directly on the true max‑convolution."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient 2·a[k_max‑i] on feasible indices
        n = a.shape[0]
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step: do not drive any coordinate negative
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 5 – simulated annealing (cheap local search)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Cheap SA that moves small mass blocks, cooling linearly."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start_time = time.time()
    T0 = 0.015 * best_val

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # donor sampled proportionally to its contribution to the peak
        wsum = weight.sum()
        donor = rng.choice(idx, p=weight / wsum) if wsum > 0 else rng.integers(n)

        # receiver from low‑contribution tail
        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        receiver = rng.choice(low_cands) if low_cands.size else rng.integers(n)

        # modest transfer (up to 25 % of donor)
        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved best score={best_val:.6f}")

        # occasional random jitter to diversify
        if rng.random() < 0.004:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val


# ----------------------------------------------------------------------
# Stage 6 – smoothing followed by SA
# ----------------------------------------------------------------------


def _smooth_then_sa(a0: np.ndarray,
                    target_sum: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """A few moving‑average passes to smooth the shape, then SA."""
    a = a0.copy()
    a = a * (target_sum / a.sum())

    # smoothing phase (≈10 % of remaining budget)
    smoothing_deadline = time.time() + 0.10 * (deadline - time.time())
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    while time.time() < smoothing_deadline:
        a = np.convolve(a, kernel, mode='same')
        a = a * (target_sum / a.sum())

    # SA phase
    a_sa, val_sa = _simulated_annealing(a, target_sum, deadline)
    return a_sa, val_sa


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size according to available time
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (global variable if available, otherwise simple power law)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.6, target_sum)

    # --------------------------------------------------------------
    # Stage 0 – parametric warm‑start on a reduced scale
    # --------------------------------------------------------------
    small_n = max(3000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(0.30 * (deadline - time.time()), 120.0)
    best_small_seq, _ = _grid_param_search_extended(small_n,
                                                    small_target,
                                                    param_deadline)
    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑started from extended parametric search (n={n})")

    # --------------------------------------------------------------
    # Stage 1 – exponential (mirror) descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.15 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected gradient descent with restarts
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.25 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – deterministic mass‑transfer refinement (stronger)
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + 0.15 * (deadline - time.time())
        a_mt, val_mt = _mass_transfer_refine_stronger(best_seq,
                                                      target_sum,
                                                      mt_deadline)
        if val_mt < best_score:
            best_seq, best_score = a_mt, val_mt
        print(f"[{time.time():.2f}] After strong mass‑transfer best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – exact sub‑gradient descent on max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.10 * (deadline - time.time())
        a_sub, val_sub = _subgradient_max_descent(best_seq,
                                                  target_sum,
                                                  sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – smoothing + simulated annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_fin, val_fin = _smooth_then_sa(best_seq, target_sum, deadline)
        if val_fin < best_score:
            best_seq, best_score = a_fin, val_fin
        print(f"[{time.time():.2f}] After final SA/smoothing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a future call
    globals()["height_sequence_1"] = final_seq
    return final_seq
