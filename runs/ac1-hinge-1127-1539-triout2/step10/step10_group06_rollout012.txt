"""
Multi‑restart search for a non‑negative step function with minimal max‑convolution.

The algorithm repeatedly runs a lightweight optimisation pipeline (soft‑max
mirror descent → subgradient descent on the true max → projected GD with
random restarts → exact local hill‑climbing) on randomly chosen problem
sizes and initialisations.  All candidates are kept on the simplex
∑a = √(2 n).  The best sequence seen before the global deadline is
returned.  The implementation re‑uses the helper functions from the
reference solution with a few hyper‑parameter tweaks (larger jitter,
finer step‑size schedule) to increase the chance of beating the previous
best bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using zero‑padding FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). When sum(a)=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has length len(a).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.04          # slightly larger than reference
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – subgradient descent on true max entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on the true max convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # full convolution and index of maximum
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        # sub‑gradient of the max entry
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule η ∈ [η_start, η_end]
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.5
        eta_end = 0.05                     # finer final steps
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with random restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0012 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(25.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact local refinement (hill‑climbing / simulated annealing)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # locate donor and receiver
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contribution of each index

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile tail
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # propose a tiny transfer
        max_delta = min(a[donor] * 0.08, target_sum * 0.0012)  # larger tentative steps
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution efficiently (O(n))
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # simulated annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_recalc > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        # occasional tiny jitter to escape plateaus
        if (int(elapsed * 100) % 2300) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.004, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Random jitter improved score={best_val:.6f}")

    return best_a, best_val

def _run_full_pipeline(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Execute Phases A → B → C → D within the given time budget.
    Returns the best sequence and its score.
    """
    # Phase A – exponential descent (≈15 % of the trial budget)
    t_now = time.time()
    exp_deadline = t_now + 0.15 * (deadline - t_now)
    a_exp, val_exp = _exponential_descent(a0, target_sum, exp_deadline)
    best_a = a_exp
    best_val = val_exp
    print(f"[{time.time():.2f}] After Exp descent best={best_val:.6f}")

    # Phase B – subgradient descent (≈20 % of remaining)
    if time.time() < deadline:
        t_now = time.time()
        sub_deadline = t_now + 0.20 * (deadline - t_now)
        a_sub, val_sub = _subgradient_descent(best_a, target_sum, sub_deadline)
        if val_sub < best_val:
            best_a, best_val = a_sub, val_sub
        print(f"[{time.time():.2f}] After SubGrad best={best_val:.6f}")

    # Phase C – projected GD with restarts (≈35 % of remaining)
    if time.time() < deadline:
        t_now = time.time()
        gd_deadline = t_now + 0.35 * (deadline - t_now)
        a_gd, val_gd = _gradient_descent_with_restarts(best_a,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_val:
            best_a, best_val = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best={best_val:.6f}")

    # Phase D – exact local refinement (use whatever time remains)
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_a, target_sum, deadline)
        if val_ref < best_val:
            best_a, best_val = a_ref, val_ref
        print(f"[{time.time():.2f}] After Local refine best={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    global_best_seq = None
    global_best_score = float('inf')
    trial = 0

    # Main multi‑restart loop
    while time.time() < deadline - 5.0:
        trial += 1
        now = time.time()
        remaining = deadline - now

        # Choose problem size based on remaining time
        if remaining > 800:
            n = 12000
        elif remaining > 400:
            n = 8000
        elif remaining > 200:
            n = 5000
        else:
            n = 3000

        target_sum = np.sqrt(2.0 * n)

        # Initialise a – warm start if possible, otherwise randomised power‑law
        if "height_sequence_1" in globals():
            a0 = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        else:
            a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

        # Random perturbation to diversify starts
        rng = np.random.default_rng()
        a0 += rng.normal(scale=target_sum * 0.0015, size=n)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 * (target_sum / a0.sum())

        # Allocate a modest slice of the remaining budget for this trial
        trial_budget = min(remaining * 0.4, 35.0)   # never exceed ~35 s per trial
        trial_deadline = time.time() + trial_budget

        # Run the optimisation pipeline
        a_opt, score_opt = _run_full_pipeline(a0, target_sum, trial_deadline)

        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after trial {trial}: score={global_best_score:.6f}")

        # Stop early if we are already very close to the target
        if global_best_score < 1.503 and (time.time() > deadline - 10):
            break

    # Final polishing on the best candidate with the remaining time
    if global_best_seq is not None and time.time() < deadline - 2.0:
        final_seq = global_best_seq
        final_n = final_seq.shape[0]
        final_target = np.sqrt(2.0 * final_n)
        final_seq, final_score = _local_refine(final_seq, final_target, deadline)
        if final_score < global_best_score:

"""
Multi‑restart optimisation for a non‑negative step function that minimises
the maximal self‑convolution entry.

The algorithm repeatedly runs a lightweight optimisation pipeline on
different random initialisations and problem sizes:

    1. Exponential (mirror) descent on a soft‑max surrogate of max (a*a)
    2. Sub‑gradient descent on the true max‑convolution entry
    3. Projected gradient descent with momentum and random restarts
    4. Exact local hill‑climbing / simulated‑annealing refinement

All candidates are projected onto the simplex
Σa = √(2 n) (the normalisation used by the benchmark).  The best
sequence seen before the global deadline is returned, and the result is
stored in ``height_sequence_1`` for warm‑starts in later calls.

Hyper‑parameter tweaks (larger jitter, finer sub‑gradient step‑size,
slightly larger transfer steps in the local refinement) are introduced
to increase the chance of beating the previous bound of 1.508382.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using zero‑padding FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  With Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has length len(a).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.04          # a bit larger than the reference implementation
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – subgradient descent on true max entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on the true max‑convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.5
        eta_end = 0.05                     # finer final steps for better polishing
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with random restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0012 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(25.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact local refinement (hill‑climbing / simulated annealing)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # slightly larger tentative steps (helps escape shallow plateaus)
        max_delta = min(a[donor] * 0.08, target_sum * 0.0012)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_recalc > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        # occasional tiny jitter to escape tiny plateaus
        if (int(elapsed * 100) % 2300) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.004, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Random jitter improved score={best_val:.6f}")

    return best_a, best_val

def _run_full_pipeline(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Execute Phases A → B → C → D within the given time budget.
    Returns the best sequence and its score.
    """
    # Phase A – exponential descent (≈15 % of trial budget)
    t_now = time.time()
    exp_deadline = t_now + 0.15 * (deadline - t_now)
    a_exp, val_exp = _exponential_descent(a0, target_sum, exp_deadline)
    best_a, best_val = a_exp, val_exp
    print(f"[{time.time():.2f}] After Exp descent best={best_val:.6f}")

    # Phase B – subgradient descent (≈20 % of remaining)
    if time.time() < deadline:
        t_now = time.time()
        sub_deadline = t_now + 0.20 * (deadline - t_now)
        a_sub, val_sub = _subgradient_descent(best_a, target_sum, sub_deadline)
        if val_sub < best_val:
            best_a, best_val = a_sub, val_sub
        print(f"[{time.time():.2f}] After SubGrad best={best_val:.6f}")

    # Phase C – projected GD with restarts (≈35 % of remaining)
    if time.time() < deadline:
        t_now = time.time()
        gd_deadline = t_now + 0.35 * (deadline - t_now)
        a_gd, val_gd = _gradient_descent_with_restarts(best_a,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_val:
            best_a, best_val = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best={best_val:.6f}")

    # Phase D – exact local refinement (use whatever time remains)
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_a, target_sum, deadline)
        if val_ref < best_val:
            best_a, best_val = a_ref, val_ref
        print(f"[{time.time():.2f}] After Local refine best={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    global_best_seq = None
    global_best_score = float('inf')
    trial = 0

    # ------------------------------------------------------------------
    # Multi‑restart outer loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        trial += 1
        now = time.time()
        remaining = deadline - now

        # Choose problem size based on remaining time
        if remaining > 800:
            n = 12000
        elif remaining > 400:
            n = 8000
        elif remaining > 200:
            n = 5000
        else:
            n = 3000

        target_sum = np.sqrt(2.0 * n)

        # Initialise a – warm start if possible, otherwise a randomised power‑law
        if "height_sequence_1" in globals():
            a0 = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        else:

        a0 = a0 * (target_sum / a0.sum())

        # Allocate a modest slice of the remaining budget for this trial
        trial_budget = min(remaining * 0.4, 35.0)   # never exceed ~35 s per trial
        trial_deadline = time.time() + trial_budget

    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm‑starts in later calls
    globals()["height_sequence_1"] = final_seq
