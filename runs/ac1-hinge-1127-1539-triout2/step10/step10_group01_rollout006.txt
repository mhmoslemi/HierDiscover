"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Coarse parametric sweep** – power‑law sequences,
   with an optional exponential tail, are generated for a range of
   exponents *p* and tail strengths *q*.  The best candidate is kept.

2. **Low‑dimensional refinement** – we add a small offset to the
   power‑law argument and perform a fast random‑walk search in the
   (p, q, offset) space.  Each evaluation uses the fast FFT‑based
   convolution, so thousands of proposals fit easily in the time
   budget.

3. **Long Adam‑style projected gradient descent** – starting from the
   refined sequence we minimise the softened max‑convolution
   (soft‑max with an annealed temperature β) while projecting onto the
   simplex {x≥0, Σx = √(2 n)}.  The β schedule grows up to 2·10⁶,
   giving a very accurate approximation of the true max.

All phases respect the global 1000 s budget; the function returns the
best sequence found (trailing zeros trimmed).  The score printed at the
end is the exact value returned by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective 2·n·max(a*a) / (Σa)².
    For Σa = sqrt(2 n) this reduces to the maximal convolution entry.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and scale to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_offset_sequence(
    n: int, p: float, q: float, offset: float, target_sum: float
) -> np.ndarray:
    """((i+offset)+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e6,
    step_factor: float = 0.5,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the soft‑max approximation.
    The schedule for β starts at 0.5 and is doubled every `beta_inc_iter`
    iterations, up to `max_beta`.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)

        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # anneal β
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Low‑dimensional (p,q,offset) random refinement
# ----------------------------------------------------------------------


def _refine_parameters(
    init_params: tuple[float, float, float],
    n: int,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float, tuple[float, float, float]]:
    """
    Simple hill‑climbing in the (p,q,offset) space using random
    perturbations.  The routine stops when the time budget expires.
    """
    best_p, best_q, best_offset = init_params
    best_seq = _exp_power_law_offset_sequence(n, best_p, best_q, best_offset, target_sum)
    best_val = _score(best_seq)
    rng = np.random.default_rng()

    while time.time() < deadline:
        p = np.clip(best_p + rng.normal(scale=0.015), 0.05, 2.5)
        q = np.clip(best_q + rng.normal(scale=0.25), 0.0, 12.0)
        offset = np.clip(best_offset + rng.normal(scale=1.0), 0.0, 50.0)

        seq = _exp_power_law_offset_sequence(n, p, q, offset, target_sum)
        val = _score(seq)

        if val < best_val:
            best_val = val
            best_seq = seq
            best_p, best_q, best_offset = p, q, offset
            print(
                f"[{time.time():.2f}] param‑refine p={p:.4f} q={q:.3f} off={offset:.2f} "
                f"score={best_val:.6f}"
            )

    return best_seq, best_val, (best_p, best_q, best_offset)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine obeys the time budget, returns the best sequence found,
    and prints progress information.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ---------------------------------------------------------------
    # Choose problem size – larger n gives a finer shape and a lower
    # theoretical lower bound (average of the convolution).
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scale‑invariant)

    # ---------------------------------------------------------------
    # Phase 0 – coarse parametric pre‑search (power‑law ± exponential tail)
    # ---------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    best_params = (0.5, 0.0, 0.0)   # (p,q,offset)

    # (a) pure power‑law grid
    p_grid = np.linspace(0.35, 0.85, 15)
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            best_params = (float(p), 0.0, 0.0)

    # (b) power‑law + exponential tail grid
    q_grid = np.linspace(0.0, 6.0, 15)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_params = (float(p), float(q), 0.0)

    print(f"[{time.time():.2f}] After coarse sweep best={best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – low‑dimensional refinement (p, q, offset)
    # ---------------------------------------------------------------
    # give ~10 % of the total budget to this cheap search
    param_deadline = min(deadline, start_time + 0.10 * budget_s)
    if time.time() < param_deadline:
        best_seq, best_val, best_params = _refine_parameters(
            best_params, n, target_sum, param_deadline
        )
        print(f"[{time.time():.2f}] After param refinement best={best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – long Adam‑style projected gradient descent
    # ---------------------------------------------------------------
    if time.time() < deadline:
        # Run the optimizer for the rest of the time
        best_seq, best_val = _adam_gradient_descent(
            best_seq, deadline, target_sum, max_beta=2e6, step_factor=0.5, gamma=0.995
        )
        print(f"[{time.time():.2f}] After AdamGD final best={best_val:.6f}")

    # ---------------------------------------------------------------
    # Trim trailing zeros and return as a plain Python list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final verification – harmless if evaluate_sequence is not present
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
