"""
Optimisation pipeline for the step‑function benchmark.

The goal is to minimise 2·n·max(conv(a,a))/(∑a)².
If the ℓ¹–norm is forced to √(2n) the objective equals max(conv),
so the search works directly on the peak of the autocorrelation.

The pipeline consists of the following stages:

1. **Problem size selection** – choose a longer sequence for larger budgets.
2. **Initial pool** – power‑law, exponential, Dirichlet and warm‑start vectors,
   all projected onto the ℓ¹‑simplex {x≥0, Σx=√(2n)}.
3. **Soft‑max GD** – momentum projected gradient descent on a soft‑max
   approximation of the max‑convolution (fast reduction of the peak).
4. **Multi‑peak sub‑gradient descent** – computes the exact sub‑gradient using
   all (near) active constraints, i.e. all indices where the convolution is
   close to its current maximum.
5. **Deterministic mass‑transfer** – moves mass from the largest contributors
   to the smallest, with a simple simulated‑annealing acceptance rule.
6. **Permutation annealing** – swaps positions of the coefficients to break
   unfavourable alignments.
7. **Final polishing** – a high‑β soft‑max GD pass.

All phases respect the global deadline (≈1000 s).  The best sequence found is
stored in the global variable ``height_sequence_1`` for warm‑starts in later
runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex using the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^{-p} normalised to target_sum."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam·i) normalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet distribution (α=1) and scale to target_sum."""
    a = rng.random(n)  # Gamma(1,1) is exponential, which is fine for a Dirichlet
    a = a / a.sum() * target_sum
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution (gradient)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected momentum GD on the soft‑max of the convolution.
    Returns the best sequence and its (soft‑max) score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)  # for reporting we use the true max

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_multi_restart(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Run several short soft‑max GD runs with jitter, keep the best.
    """
    best_a = init.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, _ = _annealed_gradient_descent(start,
                                          time.time() + run_budget,
                                          target_sum,
                                          max_beta=1e5,
                                          step_factor=0.8,
                                          gamma=0.99)
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Multi‑peak exact sub‑gradient descent (uses all active constraints)
# ----------------------------------------------------------------------


def _multi_peak_subgradient_descent(a0: np.ndarray,
                                   target_sum: float,
                                   deadline: float,
                                   step_factor: float = 0.6,
                                   gamma: float = 0.95,
                                   tol: float = 1e-8,
                                   jitter_interval: int = 400,
                                   jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent where the sub‑gradient aggregates the contributions of
    *all* convolution entries that are within a relative tolerance of the current
    maximum.  This provides a more accurate descent direction than using a
    single arg‑max.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # active set: entries whose value is >= M - ε
        eps_abs = max(tol * M, 1e-12)
        active_idx = np.where(conv >= M - eps_abs)[0]

        # build weight vector w (size 2n-1) with uniform weight on the active set
        w = np.zeros_like(conv)
        if active_idx.size:
            w[active_idx] = 1.0 / active_idx.size

        # gradient = 2 * rev(a) * w   (convolution)
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        w_fft = np.fft.rfft(w, n=size)
        conv_rev = np.fft.irfft(rev_fft * w_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Multi‑subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (as in the previous pipeline)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves targeting the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with low benchmark score.
    The routine respects the time budget and returns the best sequence found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size – longer sequences give more freedom.
    # --------------------------------------------------------------
    if budget_s >= 950:
        n = 20000
    elif budget_s >= 800:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    init_seqs = []
    rng = np.random.default_rng()

    # power‑law exponents
    for p in (0.3, 0.4, 0.5, 0.6, 0.7, 0.8):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # exponential decays
    for lam in (0.02, 0.05, 0.1, 0.2):
        init_seqs.append(_exponential_sequence(n, lam, target_sum))

    # Dirichlet random vectors
    for _ in range(4):
        init_seqs.append(_dirichlet_sequence(n, target_sum, rng))

    # warm‑start if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies of everything above
    jitter_scale = target_sum * 0.01
    for seq in list(init_seqs):
        jitter = rng.normal(scale=jitter_scale, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # a few random permutations of power‑law seeds
    for p in (0.45, 0.55):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # --------------------------------------------------------------
    # Phase 1 – Soft‑max projected gradient descent (multiple restarts)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.25
    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= phase1_deadline:
            break
        cand, val = _gradient_descent_multi_restart(init, target_sum, phase1_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Global best after GD = {best_score:.6f}")

    if best_seq is None:   # fallback safety
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – Multi‑peak exact sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + (deadline - time.time()) * 0.20
        cand, val = _multi_peak_subgradient_descent(best_seq,
                                                    target_sum,
                                                    phase2_deadline,
                                                    step_factor=0.6,
                                                    gamma=0.95)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After multi‑subgrad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = time.time() + (deadline - time.time()) * 0.25
        cand, val = _deterministic_mass_transfer(best_seq,
                                                 phase3_deadline,
                                                 target_sum,
                                                 start_temp=0.5,
                                                 end_temp=1e-4,
                                                 low_frac=0.05)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After DMT best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Re‑ordering heuristics
    # --------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        val = _max_convolution_fft(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        cand, val = _permutation_annealing(best_seq,
                                           target_sum,
                                           perm_deadline,
                                           init_score=best_score)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After permutation anneal best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – Final polishing with a very high‑β soft‑max GD
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        cand, val = _annealed_gradient_descent(best_seq,
                                               deadline,
                                               target_sum,
                                               max_beta=5e5,
                                               step_factor=0.5,
                                               gamma=0.995)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Final polishing best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing zeros and return as a plain list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional sanity check (will be printed, not used for scoring)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
