def _smooth_descent(a0, target_sum, deadline):
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_score = _score(a) # compute max_b actual
    it = 0
    rng = np.random.default_rng()
    # parameters
    eps = 0.2   # initial smoothing
    eps_decay = 0.995  # per iteration
    eta0 = 0.5    # step size scale
    while time.time() < deadline:
        # compute convolution
        b = _conv_fft(a)
        # compute actual objective maybe occasional update
        max_b = b.max()
        # smoothing
        b_scaled = b / eps
        b_shifted = b_scaled - b_scaled.max()
        expb = np.exp(b_shifted)
        w = expb / expb.sum()
        # gradient
        w_rev = w[::-1]
        L = len(w) - 1
        conv_full = _linear_convolve_fft(a, w_rev)
        g = 2.0 * conv_full[L - np.arange(a.shape[0])]
        # step size
        eta = eta0 / np.sqrt(it + 1.0)
        # exponential update
        a = a * np.exp(-eta * g)
        # renormalize onto simplex
        a = _project_to_simplex(a, target_sum)
        # optionally reproject small values to avoid zeros
        # update best if new max_b lower
        if max_b < best_score:
            best_score = max_b
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth it={it} eps={eps:.5f} score={best_score:.6f}")
        # schedule epsilon
        eps = max(eps * eps_decay, 0.005) # ensure not zero
        it += 1
    return best_a, best_score

def _smooth_gradient_descent(a0, target_sum, deadline):
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    rng = np.random.default_rng()
    eps = 0.3
    eps_min = 0.005
    eps_decay = 0.995
    eta0 = 1.0  # learning rate factor
    while time.time() < deadline:
        b = _conv_fft(a)
        max_b = b.max()
        # compute smooth weights
        scaled = b / eps
        # subtract max to avoid overflow
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        # gradient of smoothed max
        w_rev = w[::-1]
        L = len(w) - 1
        conv_full = _linear_convolve_fft(a, w_rev)
        g = 2.0 * conv_full[L - np.arange(a.shape[0])]
        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)
        # multiplicative update
        log_a = np.log(np.maximum(a, 1e-16))
        log_a -= eta * g
        a = np.exp(log_a)
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)
        # occasional jitter to avoid being stuck
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape[0])
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        # evaluation
        cur_val = max_b  # since target sum fixed
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth GD it={it} eps={eps:.4f} score={best_val:.6f}")
        # decay epsilon
        eps = max(eps*eps_decay, eps_min)
        it += 1
    return best_a, best_val

def _adam_subgradient_descent(a0, target_sum, deadline):
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = target_sum * 0.5  # base learning rate
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        max_b = b.max()
        eps_small = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps_small).astype(np.float64)
        mask_rev = mask[::-1]

        # Adam update

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

"""
Optimisation of a non‑negative step function for the
``evaluate_sequence`` benchmark.

The algorithm consists of three stages:

1. **Adam‑based sub‑gradient descent** – uses the true
   sub‑gradient of the max‑convolution (exactly as in the reference
   code) but with an Adam optimiser.  This gives faster progress on
   the highly non‑smooth objective.

2. **Smoothed (soft‑max) gradient descent** – replaces the hard max
   by an ε‑soft‑max, yielding a smooth differentiable objective.
   Gradient steps are performed with a multiplicative (exponentiated)
   update and a simplex projection, which respects the non‑negativity
   and L¹‑normalisation constraints.

3. **Exact local mass‑transfer refinement** – a cheap O(n) simulated
   annealing move that transfers a tiny amount of mass from a large‑
   contribution index to a small‑contribution one.  This stage is the
   same as in the previous solution and provides a final polishing.

All stages respect the time budget (``budget_s`` seconds).  The best
sequence found is returned as a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective: 2·n·max(a*a) / (sum(a)²).
    When Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam based sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _adam_subgradient_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the true max‑convolution (sub‑gradient).
    Keeps the iterate on the simplex Σa = target_sum.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    it = 0
    # learning‑rate scale – 2 % of the L¹‑norm works well in practice
    lr = target_sum * 0.02

    while time.time() < deadline:
        # convolution and mask of maximal entries
        b = _conv_fft(a)
        max_b = b.max()
        eps = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps).astype(np.float64)

        # sub‑gradient of max(b) w.r.t. a
        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)          # length 3n‑2
        L = len(mask) - 1                                      # = 2n‑2
        g = conv_full[L - np.arange(n)]                        # size n

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * (m_hat / (np.sqrt(v_hat) + eps_adam))

        a = a - step
        a = np.maximum(a, 0.0)                                 # keep non‑negative
        a = _project_to_simplex(a, target_sum)

        # bookkeeping
        cur_val = max_b                                          # Σa = √(2n) ⇒ score = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} score={best_val:.6f}")

        # occasional jitter to escape very flat plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed (soft‑max) gradient descent on max(b)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the ε‑soft‑max approximation of the max‑convolution.
    The update is multiplicative (exponentiated gradient) and the iterate
    stays on the simplex Σa = target_sum.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]

    # smoothing schedule
    eps = 0.5                     # start fairly smooth
    eps_min = 0.005
    eps_decay = 0.995

    it = 0
    while time.time() < deadline:
        # full convolution
        b = _conv_fft(a)
        max_b = b.max()

        # soft‑max weights (stable)
        scaled = b / eps
        scaled -= scaled.max()                     # prevent overflow
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the smooth objective:
        #   ∇ = 2 * a * (w reversed)  (convolution)
        w_rev = w[::-1]
        L = len(w) - 1
        conv_full = _linear_convolve_fft(a, w_rev)          # length 3n‑2
        g = 2.0 * conv_full[L - np.arange(n)]

        # step size – 1/√(iteration) schedule (multiplicative update)
        eta = 0.4 / np.sqrt(it + 1.0)

        # exponentiated update (log‑domain for stability)
        log_a = np.log(np.maximum(a, 1e-16))
        log_a -= eta * g
        a = np.exp(log_a)

        # back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # keep the best (according to the true objective)
        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth it={it:6d} eps={eps:.4f} score={best_val:.6f}")

        # occasional jitter to avoid stagnation
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # decay the smoothing parameter
        eps = max(eps * eps_decay, eps_min)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact mass‑transfer refinement (same as previous solution)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the true max‑convolution.
    Moves a tiny amount of mass from a “bad’’ index to a “good’’ one,
    updating the full convolution in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – choose from the lower‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # Update convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic full re‑normalisation & recomputation
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional jitter (tiny shuffle)
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Choose problem size – a larger n gives a lower theoretical bound
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 16000
    elif budget_s >= 700:
        n = 14000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)            # fixing Σa simplifies the objective

    # ------------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
    else:
        # power‑law initialisation (slightly steeper than the previous code)
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.60))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – Adam based sub‑gradient descent (≈45 % of budget)
    # ------------------------------------------------------------------
    t1_deadline = time.time() + 0.45 * (deadline - time.time())
    a_adam, val_adam = _adam_subgradient_descent(init, target_sum, t1_deadline)
    best_seq = a_adam
    best_score = val_adam
    print(f"[{time.time():.2f}] After Adam stage best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Smoothed gradient descent (≈35 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        t2_deadline = time.time() + 0.35 * (deadline - time.time())
        a_smooth, val_smooth = _smooth_gradient_descent(best_seq,
                                                        target_sum,
                                                        t2_deadline)
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
        print(f"[{time.time():.2f}] After smooth stage best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – short random restarts of Adam (≈10 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        t3_deadline = time.time() + 0.10 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < t3_deadline - 2.0:
            jitter = rng.normal(scale=target_sum * 0.03, size=n)
            cand = best_seq + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_sum)
            a_tmp, val_tmp = _adam_subgradient_descent(cand,
                                                       target_sum,
                                                       time.time() + 5.0)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]

    final_seq = arr.tolist()

        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")

        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
