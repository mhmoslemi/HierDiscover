    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    Implements the O(n) algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective used during the search.
    For Σa = √(2 n) the return equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed max gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed max objective
        F_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    The smoothing parameter β grows linearly from 0.5 to 40.
    The iterate stays on the simplex Σa = target_sum.
    Returns the best sequence (lowest true max) found before the deadline.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    start = time.time()
    total_time = deadline - start

    # parameters for the schedule
    beta_start = 0.5
    beta_end = 40.0

    while time.time() < deadline:
        # progress fraction in [0,1]
        prog = (time.time() - start) / total_time
        beta = beta_start + prog * (beta_end - beta_start)

        # full convolution a*a
        b = _conv_fft(a)
        max_b = b.max()

        # soft‑max weights (stable)
        shift = b - max_b          # max becomes 0
        w = np.exp(beta * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum

        # gradient of F_β : 2·a ⊛ w_rev  (⊛ = linear convolution)
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)      # length 3n‑2
        L = len(w) - 1                                   # = 2n‑2
        g = conv_full[L - np.arange(a.shape[0])]

        # step size (decaying)
        step = target_sum * 0.6 / np.sqrt(it + 1.0)
        gn = np.linalg.norm(g)
        if gn > 0:
            a -= step * g / gn

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # keep the best (according to the *true* max, not the smoothed one)
        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth it={it:5d} β={beta:5.2f} score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact mass‑transfer refinement (same as the previous solution)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement that transfers a tiny amount of mass
    from a “bad’’ index (large convolution contribution) to a “good’’ one.
    Operates in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature

    last_recalc = start

    while time.time() < deadline:
        # ----- identify donor (large contribution) -----
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # ----- identify receiver (small contribution) -----
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # ----- propose a small transfer -----
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # ----- update convolution exactly (O(n)) -----
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # ----- acceptance (simulated annealing) -----
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # ----- periodic full recomputation -----
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional tiny jitter to shake out of plateaus
        if int(elapsed * 100) % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)            # Σa is fixed to simplify the objective

    # ------------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # fallback: decreasing power‑law (exponent tuned empirically)
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.60))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – smoothed gradient descent (≈50 % of the budget)
    # ------------------------------------------------------------------
    sub_deadline = time.time() + 0.5 * (deadline - time.time())
    a_smooth, val_smooth = _smooth_descent(init, target_sum, sub_deadline)
    best_seq = a_smooth
    best_score = val_smooth
    print(f"[{time.time():.2f}] After smooth descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short random restarts of smoothed descent (~20 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rest_deadline = time.time() + 0.2 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2.0:
            remaining = rest_deadline - time.time()
            run_time = min(6.0, remaining)
            if run_time < 1.0:
                break
            jitter = rng.normal(scale=target_sum * 0.03, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            a_tmp, val_tmp = _smooth_descent(start_seq, target_sum,
                                             time.time() + run_time)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
