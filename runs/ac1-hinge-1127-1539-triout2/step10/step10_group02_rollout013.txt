"""
Propose candidate sequence for the max-convolution minimisation problem.

Algorithm summary
-----------------
1. **Coarse optimisation (small size)**
   * Build a diverse pool of simple parametric shapes (power‑law,
     exponential, triangular) for a modest length n0 (≈4000).
   * Run a fast soft‑max projected GD (Adam‑style momentum) on each
     candidate for a short budget; keep the best.
2. **Upscaling**
   * Interpolate the best small‑size candidate to a larger size n (≈12000‑20000)
     and project onto the ℓ¹‑simplex (∑a = √(2n)).
3. **Fine optimisation (large size)**
   * Staged optimisation using:
        – Exact max‑sub‑gradient descent,
        – Deterministic mass‑transfer,
        – Permutation annealing,
        – Continuous simulated‑annealing (mass moves),
        – A final high‑β soft‑max GD polish.
   * Each stage works on the current best sequence and respects the remaining
     time budget.
4. **Post‑processing**
   * Trim trailing near‑zero entries, store the result for warm‑starts and
     return it as a plain Python list.

All heavy convolutions use FFT (O(n log n)).  The routine respects the
1000 s time limit; the final sequence is evaluated with the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Symmetric triangle: 1,2,…,⌈n/2⌉,…,2,1."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(x, n + 1 - x)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max (smooth) gradient utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _softmax_gd(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_beta: float = 3e5,
    step_factor: float = 0.6,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Momentum projected GD on the soft‑max approximation.
    Returns the best sequence seen and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best = a.copy()
    best_val = _max_convolution_fft(best)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Exact max sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    step_factor: float = 0.5,
    gamma: float = 0.9,
    jitter_interval: int = 500,
    jitter_scale: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution (max entry only).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for max entry
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
    low_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves focusing on the current peak.
    Accepted moves follow a simple temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_val = _max_convolution_fft(a)
    best = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_val = _max_convolution_fft(a)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} best={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_val = _max_convolution_fft(a)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_side = True
    for v in sorted_vals:
        if left_side:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_side = not left_side
    return out


# ----------------------------------------------------------------------
# Swap‑based permutation annealing
# ----------------------------------------------------------------------


def _permutation_annealing(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    init_score: float | None = None,
) -> tuple[np.ndarray, float]:
    cur = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    n = cur.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        ratio = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur[i], cur[j] = cur[j], cur[i]
        new_score = _max_convolution_fft(cur)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[{time.time():.2f}] Perm anneal best={best_score:.6f}")
        else:
            cur[i], cur[j] = cur[j], cur[i]

        # occasional large shuffle near the end
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur)
            cur_score = _max_convolution_fft(cur)

    return best, best_score


# ----------------------------------------------------------------------
# Continuous simulated annealing (mass moves)
# ----------------------------------------------------------------------


def _continuous_annealing(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    init_temp: float = 0.5,
    final_temp: float = 1e-5,
    max_move_frac: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Move a random amount of mass from a donor to a receiver.
    Acceptance follows a temperature schedule.
    """
    cur = a0.copy()
    cur_val = _max_convolution_fft(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    n = cur.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = init_temp * (final_temp / init_temp) ** tau

        donor = rng.integers(0, n)
        if cur[donor] <= 0:
            continue
        max_eps = cur[donor] * max_move_frac
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        receiver = rng.integers(0, n - 1)
        while receiver == donor:
            receiver = rng.integers(0, n)

        cur[donor] -= eps
        cur[receiver] += eps

        new_val = _max_convolution_fft(cur)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
                print(f"[{time.time():.2f}] Continuous SA new best={best_val:.6f}")
        else:
            # revert
            cur[donor] += eps
            cur[receiver] -= eps

    return best, best_val

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with low benchmark score.
    The method combines a small‑scale quick search, up‑scaling,
    and several refinement phases under the time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    # Small stage size (fast coarse search)
    n_small = 4000
    # Large stage size – more points give a better bound if time permits
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 700:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum_small = np.sqrt(2.0 * n_small)
    target_sum_large = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1. Coarse optimisation on small size
    # --------------------------------------------------------------
    init_small = []

    # power‑law candidates
    for p in (0.35, 0.4, 0.45, 0.5, 0.55, 0.6):
        init_small.append(_power_law_sequence(n_small, p, target_sum_small))
    # exponential candidates
    for lam in (0.0002, 0.0005, 0.001):
        init_small.append(_exp_sequence(n_small, lam, target_sum_small))
    # triangular
    init_small.append(_triangular_sequence(n_small, target_sum_small))

    rng = np.random.default_rng()
    # jittered copies
    for seq in list(init_small):
        jitter = rng.normal(scale=target_sum_small * 0.02, size=n_small)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum_small)
        init_small.append(seq_jit)

    # --------------------------------------------------------------
    # Quick soft‑max GD on each small candidate (budget ~10%)
    # --------------------------------------------------------------
    small_deadline = time.time() + (deadline - time.time()) * 0.10
    best_small = None
    best_small_val = np.inf

    for cand in init_small:
        if time.time() >= small_deadline:
            break
        a_opt, val = _softmax_gd(
            cand,
            target_sum_small,
            small_deadline,
            max_beta=5e4,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_small_val:
            best_small_val = val
            best_small = a_opt.copy()
            print(f"[{time.time():.2f}] Small GD improved best={best_small_val:.6f}")

    if best_small is None:
        best_small = _power_law_sequence(n_small, 0.55, target_sum_small)

    # --------------------------------------------------------------
    # 2. Upscaling to large size
    # --------------------------------------------------------------
    a_large = _resize_sequence(best_small, n)
    a_large = _project_to_simplex(a_large, target_sum_large)

    # --------------------------------------------------------------
    # 3. Large‑scale fine optimisation (remaining budget)
    # --------------------------------------------------------------
    # allocate portions of the remaining time
    remaining = deadline - time.time()
    # Phase A: sub‑gradient descent (15%)
    sg_deadline = time.time() + remaining * 0.15
    a_large, val = _subgradient_descent(
        a_large, target_sum_large, sg_deadline, step_factor=0.6, gamma=0.95
    )
    print(f"[{time.time():.2f}] After sub‑grad best={val:.6f}")

    # Phase B: deterministic mass‑transfer (20%)
    remaining = deadline - time.time()
    if remaining > 0:
        mt_deadline = time.time() + remaining * 0.20
        a_large, val = _deterministic_mass_transfer(
            a_large,
            target_sum_large,
            mt_deadline,
            start_temp=0.6,
            end_temp=5e-5,
            low_frac=0.07,
        )
        print(f"[{time.time():.2f}] After DMT best={val:.6f}")

    # Phase C: permutation annealing (15%)
    remaining = deadline - time.time()
    if remaining > 0:
        perm_deadline = time.time() + remaining * 0.15
        a_large, val = _permutation_annealing(
            a_large, target_sum_large, perm_deadline, init_score=val
        )
        print(f"[{time.time():.2f}] After perm anneal best={val:.6f}")

    # Phase D: continuous simulated annealing on values (15%)
    remaining = deadline - time.time()
    if remaining > 0:
        ca_deadline = time.time() + remaining * 0.15
        a_large, val = _continuous_annealing(
            a_large, target_sum_large, ca_deadline, init_temp=0.4, final_temp=1e-5
        )
        print(f"[{time.time():.2f}] After continuous SA best={val:.6f}")

    # Phase E: final high‑β soft‑max GD (remaining time)
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_large, val = _softmax_gd(
            a_large,
            target_sum_large,
            deadline,
            max_beta=3e5,
            step_factor=0.5,
            gamma=0.998,
        )
        print(f"[{time.time():.2f}] Final GD polishing best={val:.6f}")

    # --------------------------------------------------------------
    # Optional re‑ordering heuristic (very cheap)
    # --------------------------------------------------------------
    alt = _alternating_order(a_large)
    alt_val = _max_convolution_fft(alt)
    if alt_val < val:
        a_large = alt
        val = alt_val
        print(f"[{time.time():.2f}] Alternating order improved best={val:.6f}")

    # --------------------------------------------------------------
    # Final trimming and evaluation
    # --------------------------------------------------------------
    arr = np.array(a_large, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
