def max_convolution_and_gradient(a):
    b = np.convolve(a, a)  # compute using FFT for speed
    k = np.argmax(b)
    # compute gradient:
    # g[i] = 2*a[k-i] for i in [0,n-1] if 0 <= k-i < n
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - (n-1))
    high = min(k, n-1)
    # for i in [low, high]:
    #     j = k - i
    #     g[i] = 2*a[j]
    # Since we have vector a, can compute slice:
    i_vals = np.arange(low, high+1)
    j_vals = k - i_vals
    g[i_vals] = 2 * a[j_vals]
    # This gives gradient. For indices outside range, g=0.
    return b, k, g

    import scipy.linalg
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

assistantfinal**Solution Overview**

The evaluation function is scale‑invariant:

```
evaluate(a) = 2·n·max(a∗a) / ( Σa )²
```

If we keep Σa = √(2 n) (the “simplex” used in the reference code) the
objective is simply the maximal entry of the linear self‑convolution
`a∗a`.  The problem therefore is:

*find a non‑negative vector `a` (Σa = √(2 n)) whose convolution with itself is
as flat as possible.*

The code below combines three complementary ideas

1. **Fast sub‑gradient descent on the exact max** –  
   the gradient of the active convolution entry `k*` is
   `g_i = 2·a_{k*‑i}` (zero outside the support).  After a step we project
   back onto the simplex.

2. **Linear‑programming refinement** –  
   for a fixed `f` let `M = max(f∗f)`.  The LP  

   ```
   maximise   Σ g_j
   subject to  f ∗ g ≤ M      (2n‑1 linear constraints)
               g_j ≥ 0
   ```

   finds a vector `g` that can be mixed with `f` while keeping the worst
   convolution entry ≤ M.  Repeating the LP a few times (with a small
   mixing factor `t`) drives the maximal entry down.

3. **Deterministic hill‑climb on the exact convolution** – an exact,
   O(n) greedy move that shifts mass from the index that contributes most
   to the current worst entry to the index that contributes least.

The three phases are executed many times on a modest size (`n≈800`),
the best result is up‑sampled to the target length (`n=2000`), and a final
Adam‑style smoothing + hill‑climb polish is performed with the remaining
time budget.

All helper functions are top‑level, the algorithm respects the 1000 s
budget, prints progress, and finally returns the best sequence found.

```python
"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1.   Initialise a moderate‑size vector (n0≈800) with several families
     (random Dirichlet, power‑law, symmetric arcsine‑like).
2.   Sub‑gradient descent on the exact max‑convolution entry, projected onto
     the ℓ¹‑simplex Σa = √(2 n0).
3.   Linear‑programming refinement:
     maximise Σg  subject to  f∗g ≤ M  (M = current max) and g≥0.
     Mix the solution with the current vector (mixing factor t).
4.   Upsample the best vector to the final length (n=2000) and run a
     deterministic hill‑climb on the exact convolution.
5.   Use a short Adam optimisation on the smooth‑max surrogate as a final
     polish.
All phases respect the global time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# Random initialisation

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _arcsine_initial_sequence(n: int) -> np.ndarray:
    """Symmetric arcsine‑like shape a_i = 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = np.maximum(a, 0.0)
    return _project_to_simplex(a, math.sqrt(2.0 * n))


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Sub‑gradient descent on the exact max entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float,
                         deadline: float, max_iter: int = 5000):
    """Projected sub‑gradient descent on max(a∗a)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        b = _full_convolution(a)
        k = int(np.argmax(b))

        # gradient: g_i = 2·a_{k‑i} for indices where this makes sense
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        # simple decreasing step size
        step = 0.1 * target_sum / (np.sqrt(it + 1) + 1.0)
        a -= step * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        sc = _max_convolution_fft(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
        it += 1

    return best_a, best_score

# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray, target_sum: float,
               deadline: float, max_iters: int = 5, t: float = 0.2):
    """Iteratively improve a via LP: maximise Σg under f∗g ≤ M."""
    import scipy.optimize
    import scipy.linalg

    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        b = _full_convolution(a)
        M = b.max()

        # Build Toeplitz matrix A where A[k, j] = a_{k‑j} (0 if out of range)
        col = np.concatenate([a, np.zeros(n - 1)])
        row = np.concatenate([a[:1], np.zeros(n - 1)])
        A = scipy.linalg.toeplitz(col, row)      # shape (2n‑1, n)

        c = -np.ones(n)                           # maximise Σg  → minimise -Σg
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs',
                                    options={'presolve': True})
        if not res.success:
            # LP failed – stop refining
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # Scale g to the simplex and mix with the current vector
        g = g * (target_sum / sum_g)
        a = (1.0 - t) * a + t * g
        a = _project_to_simplex(a, target_sum)

        sc = _max_convolution_fft(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that attacks the current worst convolution entry."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

# Adam optimisation on a smooth‑max surrogate (kept from the reference)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """Projected Adam on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``
    under a wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5       # a tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seed from a previously found construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq = np.asarray(globals()["height_sequence_1"], dtype=float)
            seq = np.maximum(seq, 0.0)
            target0 = math.sqrt(2.0 * seq.size)
            seq = _project_to_simplex(seq, target0)
            sc = _max_convolution_fft(seq)
            best_seq = seq.copy()
            best_score = sc
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={seq.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Search on a moderate size (n0≈800) – many restarts
    # ------------------------------------------------------------------
    n0 = 800
    target0 = math.sqrt(2.0 * n0)

    while time.time() < deadline:
        # remaining time check
        if time.time() + 60 > deadline:
            break

        # ---- random initialisation ------------------------------------
        typ = rng.integers(0, 3)
        if typ == 0:
            a = _random_initial_sequence(n0, rng)
        elif typ == 1:
            a = _arcsine_initial_sequence(n0)
        else:
            # symmetric Dirichlet (mirror half)
            half_len = (n0 + 1) // 2
            half = rng.dirichlet(alpha=np.ones(half_len))
            if n0 % 2 == 0:
                a = np.concatenate([half, half[::-1]])
            else:
                a = np.concatenate([half[:-1], half[::-1]])
            a = _project_to_simplex(a, target0)

        # ---- sub‑gradient descent ------------------------------------
        sub_dead = min(deadline, time.time() + 30.0)
        a_sd, sc_sd = _subgradient_descent(a, target0, sub_dead, max_iter=5000)
        if sc_sd < best_score - 1e-12:
            best_seq = a_sd.copy()
            best_score = sc_sd
            print(f"[{time.time():.1f}] Sub‑gradient improvement "
                  f"score={best_score:.6f}")

        # ---- LP refinement -------------------------------------------
        lp_dead = min(deadline, time.time() + 20.0)
        a_lp, sc_lp = _lp_refine(a_sd, target0, lp_dead, max_iters=5, t=0.2)
        if sc_lp < best_score - 1e-12:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"[{time.time():.1f}] LP refinement improvement "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upsample to the final length (n=2000) and finish with hill‑climb
    # ------------------------------------------------------------------
    final_n = 2000
    if best_seq is None:
        best_seq = _random_initial_sequence(final_n, rng)
    elif best_seq.size != final_n:
        best_seq = _upsample_sequence(best_seq, final_n)

    best_seq = _project_to_simplex(best_seq, math.sqrt(2.0 * final_n))

    # Hill‑climb polishing
    hc_dead = min(deadline, time.time() + 30.0)
    b_cur = _full_convolution(best_seq)
    seq_hc, sc_hc = _hill_climb_deterministic(best_seq, b_cur, hc_dead)
    if sc_hc < best_score - 1e-12:
        best_seq = seq_hc.copy()
        best_score = sc_hc
        print(f"[{time.time():.1f}] Hill‑climb improvement "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final Adam + hill‑climb polish (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        fin_dead = deadline - 2.0
        best_seq = _adam_optimize(best_seq,
                                  math.sqrt(2.0 * final_n),
                                  fin_dead)
        b_final = _full_convolution(best_seq)
        seq_fin, sc_fin = _hill_climb_deterministic(best_seq, b_final,
                                                    deadline - 1.0)
        if sc_fin < best_score - 1e-12:
            best_seq = seq_fin.copy()
            best_score = sc_fin
            print(f"[{time.time():.1f}] Final Adam+Hill‑climb improvement "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```
