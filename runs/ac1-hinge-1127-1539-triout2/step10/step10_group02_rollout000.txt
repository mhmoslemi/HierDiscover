"""
Improved search for a non‑negative step function minimizing
    2·n·max(a∗a)/(∑a)²  = max(a∗a)   (when ∑a = √(2n))

The pipeline follows several orthogonal phases:
   1.  A rich pool of initial candidates (power‑law, exponential, warm‑start,
       jittered copies, random permutations).
   2.  Momentum projected GD on a soft‑max approximation of the max‑convolution,
       with an annealed β (β→∞) and many short restarts.
   3.  Exact sub‑gradient descent on the true max‑convolution.
   4.  Deterministic mass‑transfer (greedy donor→receiver moves with simulated‑annealing
       acceptance).
   5.  Permutation annealing (swap‑based simulated annealing on the ordering).
   6.  Final high‑β soft‑max GD polishing.

All phases respect the overall time budget (default 1000 s).  The best
sequence is stored in the global variable `height_sequence_1` for warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – onto {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.where(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)^(-p) normalised to have ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Exponential decay a_i = exp(-lam·i) normalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max approximation utilities
# ----------------------------------------------------------------------


def _softmax_grad(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv_array) where grad has same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # numerical stability
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (rev(a) * p)_{central slice}
    a_rev = a[::-1]
    A_rev_fft = np.fft.rfft(a_rev, n=size)
    P_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_softmax_gd(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_beta: float = 1e7,
                         step_factor: float = 0.6,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    The schedule slowly increases β up to max_beta.
    Returns the best sequence seen and its (soft‑max) objective value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    # true objective (max convolution) – used for bookkeeping
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _softmax_grad(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        # increase β geometrically
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gd_with_restarts(init: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Runs several short GD restarts with jitter; keeps the best."""
    best_a = init.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 4.0:
        remaining = deadline - time.time()
        # each restart gets at most 25 s, but never more than 40 % of remaining time
        run_budget = min(25.0, remaining * 0.4)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_softmax_gd(start,
                                      time.time() + run_budget,
                                      target_sum,
                                      max_beta=1e6,
                                      step_factor=0.8,
                                      gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.95,
                         jitter_interval: int = 400,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent with momentum on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2 * a[kmax-i] for valid indices
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale,
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (greedy + simulated annealing)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.8,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """Greedy mass transfer guided by the current peak of the convolution."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight contributed to the peak by each index in the supporting range
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid being stuck
        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best={best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full shuffle for exploration
        if ratio > 0.9 and rng.random() < 0.02:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low max‑convolution value.
    The function respects the supplied time budget (default 1000 s) and
    returns a list of non‑negative heights.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        # warm‑start length – keep it unless we have plenty of time to enlarge
        base_n = len(globals()["height_sequence_1"])
    else:
        base_n = 0

    if budget_s >= 800:
        n = max(20000, base_n)
    elif budget_s >= 500:
        n = max(15000, base_n)
    else:
        n = max(10000, base_n)

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a rich pool of initial candidates
    # --------------------------------------------------------------
    init_seqs = []

    # power‑law family
    for p in np.linspace(0.45, 0.6, 6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # exponential decays
    for lam in (0.08, 0.12, 0.18):
        init_seqs.append(_exp_sequence(n, lam, target_sum))

    # warm start
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        init_seqs.append(prev)

    # uniform baseline
    init_seqs.append(np.full(n, target_sum / n))

    # jittered copies of everything above (two per seed)
    rng = np.random.default_rng()
    jitter_scale = target_sum * 0.015
    seeds = list(init_seqs)   # copy
    for seq in seeds:
        for _ in range(2):
            jitter = rng.normal(scale=jitter_scale, size=n)
            seq_jit = seq + jitter
            seq_jit = np.maximum(seq_jit, 0.0)
            seq_jit = _project_to_simplex(seq_jit, target_sum)
            init_seqs.append(seq_jit)

    # random permutations of a couple of strong candidates
    for p in (0.5, 0.55):
        tmp = _power_law_sequence(n, p, target_sum)
        rng.shuffle(tmp)
        init_seqs.append(tmp)

    # --------------------------------------------------------------
    # Phase 1 – Soft‑max GD with multiple restarts
    # --------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.30
    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= phase1_deadline:
            break
        cand_seq, cand_score = _gd_with_restarts(init,
                                                 target_sum,
                                                 phase1_deadline)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] Global best after GD = {best_score:.6f}")

    if best_seq is None:  # fallback safety
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = time.time() + (deadline - time.time()) * 0.20
        a_sg, sc_sg = _subgradient_descent(best_seq,
                                           target_sum,
                                           sg_deadline,
                                           step_factor=0.45,
                                           gamma=0.96)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        a_mt, sc_mt = _deterministic_mass_transfer(best_seq,
                                                   mt_deadline,
                                                   target_sum,
                                                   start_temp=0.6,
                                                   end_temp=5e-5,
                                                   low_frac=0.02)
        if sc_mt < best_score:
            best_score = sc_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Re‑ordering heuristics (alternating & random shuffles)
    # --------------------------------------------------------------
    reorder_cands = [best_seq, _alternating_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_cands.append(tmp)

    for cand in reorder_cands:
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.15
        a_perm, sc_perm = _permutation_annealing(best_seq,
                                                 target_sum,
                                                 perm_deadline,
                                                 init_score=best_score)
        if sc_perm < best_score:
            best_score = sc_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – Final high‑β soft‑max GD polishing
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, sc_fin = _annealed_softmax_gd(best_seq,
                                             deadline,
                                             target_sum,
                                             max_beta=3e6,
                                             step_factor=0.5,
                                             gamma=0.998)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in possible future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
