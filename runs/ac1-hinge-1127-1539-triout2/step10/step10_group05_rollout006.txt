"""
Propose a non‑negative step function that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm consists of

1.  A richer parametric sweep (exponential edges, double‑exponential,
    cosine‑power windows, logistic edges) to obtain a good initial basin.
2.  Several projected‑Adam runs on the smooth log‑sum‑exp surrogate of the
    max‑convolution, each restart lasting a few seconds.
3.  A deterministic hill‑climb that moves mass from the worst convolution
    entry to the most helpful destination (exact convolution updates).
4.  An up‑sampling stage (to a longer sequence) followed by another round of
    Adam + hill‑climb – longer sequences give the optimizer more freedom.
5.  A final polishing Adam + hill‑climb pass.

All steps respect a wall‑clock budget (default 1000 s).  The best sequence
found is trimmed of trailing near‑zeros and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate of the max‑convolution.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Learning‑rate scaled by the target L1‑norm.
    base_lr = 0.12 * target_sum
    beta = 2.0
    beta_max = 5e6
    beta_factor = 1.8

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Shape that is large near the two edges and decays towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

    # 2) Double‑exponential with independent left/right rates
    for lam_left in np.linspace(0.0005, 0.02, 20):
        for lam_right in np.linspace(0.0005, 0.02, 20):
            d_left = idx
            d_right = n - 1 - idx
            a = np.exp(-lam_left * d_left) + np.exp(-lam_right * d_right)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # 3) Cosine‑power windows (high at both ends, low in the middle)
    base = (1.0 + np.cos(2.0 * np.pi * idx / (n - 1))) / 2.0   # range [0,1]
    for p in np.linspace(0.5, 5.0, 30):
        a = base ** p
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # 4) Logistic edge shapes (mirrored sigmoids)
    mid = (n - 1) / 2.0
    for alpha in np.linspace(0.02, 10.0, 30):
        x = idx - mid
        sig = 1.0 / (1.0 + np.exp(alpha * x))
        sig2 = 1.0 / (1.0 + np.exp(-alpha * x))
        a = sig + sig2 - 1.0
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # 5) Random family seeds (to diversify the search)
    for _ in range(30):
        a = _random_initial_sequence(n, rng)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # 6) Edge‑heavy (identical to the original edge‑high family)
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(idx, n - 1 - idx)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb (exact max) – unchanged but a tiny tweak
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))
        # index range that contributes to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        # pick a destination that contributes the least to M
        counterpart = a[M - idx]
        order = np.argsort(counterpart)
        j = None
        for cand in idx[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # try to move a sizeable fraction; we will fall back to a line‑search
        delta = a[i] * 0.75
        if delta <= 0.0:
            break

        # --- exact O(n) update of the convolution ---
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # fall back to a more cautious, halving line‑search
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b_test.max() < best_score - 1e-12:
                        best_score = b_test.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small random perturbation to escape possible plateaus
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (kept as a fallback)
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing (O(n) per move).  Returns the best vector
    and its maximal convolution entry seen before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9999

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.5)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5
        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a wall‑clock budget (default
    1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 0)  Seed from a previous run if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded (n={n0}) → score={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # --------------------------------------------------------------
    # 1)  Fixed‑length optimisation (initial length = 2000)
    # --------------------------------------------------------------
    n = 2000
    target = math.sqrt(2.0 * n)

    # ---- 1a) Parameter sweep (cheap) ----------------------------
    if time.time() < deadline - 2.0:
        a_sweep, score_sweep = _parameter_sweep_extended(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep improvement → score={best_score:.6f}")

    # ---- 1b) Projected Adam restarts ----------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # allocate roughly 6 s per restart (including the time spent in Adam)
        n_restarts = max(1, int(remaining // 6.0))
        per_restart = remaining / n_restarts
        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            # start from the current best on the first restart, otherwise random
            if r == 0 and best_seq is not None:
                a0 = best_seq.copy()
            else:
                if rng.random() < 0.5:
                    a0 = _random_initial_sequence(n, rng)
                else:
                    a0 = _edge_high_initial_sequence(n, rng)
            a_opt = _adam_optimize(a0, target, time.time() + per_restart * 0.9)
            s = _max_convolution_fft(a_opt)
            if s < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = s
                print(f"[{time.time():.1f}] Adam restart {r+1}/{n_restarts} → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2)  Deterministic hill‑climb (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur, deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Hill‑climb → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3)  Simulated‑annealing refinement (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4)  Upsample to a longer sequence and refine again
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        n_big = 3000
        a_big = _upsample_sequence(best_seq, n_big)
        target_big = math.sqrt(2.0 * n_big)
        # short Adam run on the larger vector
        a_big_opt = _adam_optimize(a_big, target_big, deadline - 2.5)
        b_big = _full_convolution(a_big_opt)
        a_big_opt, s_big = _hill_climb_deterministic(a_big_opt, b_big, deadline - 1.5)
        if s_big < best_score - 1e-12:
            best_seq = a_big_opt.copy()
            best_score = s_big
            print(f"[{time.time():.1f}] Upsample + hill‑climb → score={best_score:.6f}")

        # final short anneal on the upsampled result
        if time.time() < deadline - 1.0:
            b_big = _full_convolution(best_seq)
            a_big_opt, s_big = _anneal_optimize(best_seq, b_big, deadline - 0.5)
            if s_big < best_score - 1e-12:
                best_seq = a_big_opt.copy()
                best_score = s_big
                print(f"[{time.time():.1f}] Upsample + anneal → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5)  Final polishing: a short Adam run + hill‑climb on the current length
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        n_cur = best_seq.size
        target_cur = math.sqrt(2.0 * n_cur)
        a_pol = _adam_optimize(best_seq, target_cur, deadline - 1.5)
        b_pol = _full_convolution(a_pol)
        a_pol, s_pol = _hill_climb_deterministic(a_pol, b_pol, deadline - 0.5)
        if s_pol < best_score - 1e-12:
            best_seq = a_pol.copy()
            best_score = s_pol
            print(f"[{time.time():.1f}] Final polish → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6)  Clean‑up: drop trailing near‑zeros
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 7)  Report official evaluation
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
