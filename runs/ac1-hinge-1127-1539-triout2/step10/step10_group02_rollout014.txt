def propose_candidate(budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # decide n based on budget
    if budget_s >= 800:
        n = 10000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)

    # generate initial sequences
    init_seqs = []
    # power law exponents 0.3-0.9
    for p in np.linspace(0.3,0.9,7):
        seq = _power_law_sequence(n, p, target_sum)
        init_seqs.append(seq)

    # exponential decays
    for lam in np.linspace(0.01, 0.2, 5):
        x = np.arange(n)
        seq = np.exp(-lam * x)
        seq = _project_to_simplex(seq.astype(float), target_sum)
        init_seqs.append(seq)

    # double-peak: sum of two gaussians centered at 0.25 and 0.75
    for width in [0.05, 0.08, 0.12]:
        x = np.linspace(0,1,n)
        g1 = np.exp(- ((x-0.25)/width)**2 )
        g2 = np.exp(- ((x-0.75)/width)**2 )
        seq = g1 + g2
        seq = _project_to_simplex(seq, target_sum)
        init_seqs.append(seq)

    # maybe previous best
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            # resize using interpolation
            prev = np.interp(np.linspace(0,1,n), np.linspace(0,1,prev.shape[0]), prev)
        init_seqs.append(prev)

    # keep random jittered versions
    rng = np.random.default_rng()
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_jit = np.maximum(seq + jitter, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    best_seq = None
    best_val = np.inf

    # Phase 1: subgradient descent (exact max)
    phase1_deadline = start_time + (deadline - start_time) * 0.35
    for init in init_seqs:
        if time.time() >= phase1_deadline:
            break
        a, val = _subgradient_descent(init, target_sum, phase1_deadline)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Phase1 new best (maxconv)={best_val:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_val = _max_convolution_fft(best_seq)

    # Phase 2: permutation annealing heavy
    phase2_deadline = deadline
    a_perm, val_perm = _permutation_annealing(best_seq, target_sum, phase2_deadline, init_score=best_val)
    if val_perm < best_val:
        best_val = val_perm
        best_seq = a_perm.copy()
        print(f"[{time.time():.2f}] Phase2 improved best={best_val:.6f}")

    # Phase 3: optional softmax gradient refinement for polishing
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e5,
            step_factor=0.5,
            gamma=0.99,
        )
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final smoothing best={best_val:.6f}")

    # Trim trailing near-zero
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq

    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _double_peak_sequence(n: int, width: float, target_sum: float) -> np.ndarray:
    """Two Gaussian‑like peaks centred at 0.25 and 0.75 of the interval."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    g1 = np.exp(-((x - 0.25) / width) ** 2)
    g2 = np.exp(-((x - 0.75) / width) ** 2)
    a = g1 + g2
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 400,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum of the linear convolution.
    Returns the best sequence found and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of max‑convolution at lag kmax:
        # g[i] = 2 * a[kmax - i]  if 0 ≤ kmax - i < n, else 0
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Swap‑based simulated annealing that only changes the ordering of the
    entries.  The L1 sum is preserved by the projection step.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.3
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # propose swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] perm anneal new best={best_score:.6f}")
        else:
            # reject – revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional large perturbation near the end of the schedule
        if ratio > 0.9 and rng.random() < 0.02:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Soft‑max projected gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (a_rev ⋆ p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected gradient descent on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low maximal self‑convolution.
    The routine respects the given time budget and returns a list of heights.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer shape, but slower per iteration)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 10000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    init_seqs = []

    # power‑law family
    for p in np.linspace(0.3, 0.9, 7):
        init_seqs.append(_power_law_sequence(n, float(p), target_sum))

    # exponential decays
    for lam in np.linspace(0.01, 0.20, 5):
        init_seqs.append(_exponential_sequence(n, float(lam), target_sum))

    # two‑peak shapes
    for w in (0.05, 0.08, 0.12):
        init_seqs.append(_double_peak_sequence(n, w, target_sum))

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            # resize by simple linear interpolation
            prev = np.interp(np.linspace(0.0, 1.0, n),
                             np.linspace(0.0, 1.0, prev.shape[0]),
                             prev)
        init_seqs.append(prev)

    # jittered copies of every starter (adds stochasticity)
    rng = np.random.default_rng()
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_jit = np.maximum(seq + jitter, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # --------------------------------------------------------------
    # Phase 1 – sub‑gradient descent (exact max‑convolution)
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    phase1_deadline = start + (deadline - start) * 0.35
    for init in init_seqs:
        if time.time() >= phase1_deadline:
            break
        a, val = _subgradient_descent(init, target_sum, phase1_deadline)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Phase 1 new best (maxconv) = {best_val:.6f}")

    # fallback – should never happen
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_val = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – heavy permutation annealing
    # --------------------------------------------------------------
    phase2_deadline = deadline - 5.0   # leave a few seconds for polishing
    a_perm, val_perm = _permutation_annealing(best_seq, target_sum,
                                              phase2_deadline,
                                              init_score=best_val)
    if val_perm < best_val:
        best_val = val_perm
        best_seq = a_perm.copy()
        print(f"[{time.time():.2f}] Phase 2 improved best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – short sub‑gradient fine‑tuning after ordering improvement
    # --------------------------------------------------------------
    if time.time() + 4.0 < deadline:
        sub_deadline = time.time() + min(8.0, deadline - time.time() - 2.0)
        a_fine, val_fine = _subgradient_descent(best_seq, target_sum, sub_deadline,
                                                step_factor=0.4, gamma=0.92)
        if val_fine < best_val:
            best_val = val_fine
            best_seq = a_fine.copy()
            print(f"[{time.time():.2f}] Phase 3 fine‑tuned best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final polishing with soft‑max GD (optional, time‑dependent)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(best_seq,
                                                    deadline,
                                                    target_sum,
                                                    max_beta=3e5,
                                                    step_factor=0.6,
                                                    gamma=0.995)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing zeros and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # sanity check with the provided evaluator (optional)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose sequence for warm‑starts in subsequent runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
