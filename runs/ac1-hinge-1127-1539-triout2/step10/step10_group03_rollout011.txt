def permute_anneal(a, time_deadline):
    n = len(a)
    best = a.copy()
    bestmax = _max_convolution_fft(best)
    cur = best.copy()
    curmax = bestmax
    rng = np.random.default_rng()
    temp0 = 1.0
    while time.time() < time_deadline:
        i,j = rng.integers(0, n, size=2)
        # swap cur[i], cur[j]
        cur[i], cur[j] = cur[j], cur[i]
        # compute new max
        newmax = _max_convolution_fft(cur)
        if newmax < curmax or rng.random() < math.exp((curmax - newmax) / temp):
            curmax = newmax
            if newmax < bestmax:
                bestmax = newmax
                best = cur.copy()
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
        # update temperature schedule
        # linear decay based on elapsed fraction
        elapsed = time.time() - start
        fraction = elapsed / total
        temp = temp0 * max(0.01, 1 - fraction)
    return best, bestmax

def _spacing_order(seq, spacing):

"""
Hybrid search for a low‑self‑convolution step function.

The algorithm combines several ideas:

* Generate many candidate value‑distributions (power‑law, exponential,
  random Dirichlet, previous best) at a modest size (≈5‑6 k).
* For each candidate try several orderings (original, alternating,
  and a family of “spacing’’ orderings that spread the biggest values
  uniformly).  Keep the arrangement with the smallest convolution peak.
* Short projected‑gradient runs on the simplex (soft‑max approximation of the max)
  improve the values while preserving the sum √(2 n).
* A brief deterministic mass‑transfer refinement is applied.
* Upscale the best small solution to a larger length (up to 12 k) using
  both linear interpolation and simple repetition; the better of the
  two is kept.
* Re‑order the upscaled vector again (alternating + spacing) and run a
  high‑temperature gradient descent.
* Finally a lightweight simulated‑annealing on permutations (swap moves)
  reduces the peak further.

All steps respect non‑negativity and the ℓ¹‑simplex constraint, use
FFT‑based convolutions for speed and stop automatically before the
time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT implementation)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (FFT implementation)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (with sum normalisation) and simplex projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2n) this reduces to max(a*a), i.e. the convolution peak.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _repeat_upscale(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Upscale by repetition (piecewise‑constant) and truncate to new_len."""
    n = seq.shape[0]
    if new_len <= n:
        return _resize_sequence(seq, new_len)
    repeats = new_len // n
    remainder = new_len - repeats * n
    out = np.repeat(seq, repeats)
    if remainder > 0:
        out = np.concatenate([out, seq[:remainder]])
    return out[:new_len]


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, second largest, third largest … alternating left/right."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _spacing_order(seq: np.ndarray, spacing: int) -> np.ndarray:
    """
    Distribute the sorted values over the array with a fixed spacing.
    The largest values are placed at residues 0,1,…,spacing‑1 cyclically.
    """
    n = seq.shape[0]
    vals = np.sort(seq)[::-1]               # descending
    out = np.empty_like(seq)
    idx = 0
    for offset in range(spacing):
        positions = np.arange(offset, n, spacing, dtype=int)
        out[positions] = vals[idx: idx + len(positions)]
        idx += len(positions)
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution (used by the projected GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv) where grad has shape a.shape.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) * p) (convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max approximation
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    Returns the best vector seen and its (soft‑max) score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (optional)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves (donor → receiver) with a Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple permutation annealing (swap moves)
# ----------------------------------------------------------------------


def _permute_anneal(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated annealing on the permutation of the vector.
    Swaps two positions, recomputes the max‑convolution and accepts
    with a temperature schedule.
    """
    a = a0.copy()
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    # linear temperature decay from 1.0 to 0.01
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        new_max = _max_convolution_fft(a)

        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = max(0.01, 1.0 - frac)

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] permute anneal new best max={best_max:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low self‑convolution peak.
    The routine respects the 1000 s budget, performs a multi‑stage
    exploration / up‑scaling / final polishing and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Small‑scale exploration (n ≈ 5‑6 k)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_small = 6000
    elif budget_s >= 400:
        n_small = 5000
    else:
        n_small = 4000
    target_small = math.sqrt(2.0 * n_small)

    rng = np.random.default_rng()

    # build a pool of raw value‑vectors (no ordering yet)
    pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.75, 7):
        pool.append(_power_law_sequence(n_small, float(p), target_small))

    # exponential family
    for lam in np.linspace(0.002, 0.012, 5):
        a = np.exp(-lam * np.arange(n_small, dtype=float))
        a *= target_small / a.sum()
        pool.append(a)

    # random Dirichlet‑like vectors
    for _ in range(5):
        a = rng.random(n_small)
        a *= target_small / a.sum()
        pool.append(a)

    # previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev = _resize_sequence(prev, n_small)
        pool.append(prev)

    # jittered copies of everything
    expanded = []
    for seq in pool:
        expanded.append(seq)
        jitter = rng.normal(scale=target_small * 0.02, size=n_small)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        expanded.append(seq_j)
    pool = expanded

    # --------------------------------------------------------------
    # 2) Search for a good ordering at the small scale
    # --------------------------------------------------------------
    best_seq = None
    best_max = np.inf

    for seq in pool:
        # original order
        m0 = _max_convolution_fft(seq)
        if m0 < best_max:
            best_max = m0
            best_seq = seq.copy()

        # alternating order
        alt = _alternating_order(seq)
        m_alt = _max_convolution_fft(alt)
        if m_alt < best_max:
            best_max = m_alt
            best_seq = alt.copy()

        # spacing orders
        for s in range(2, 7):
            sp = _spacing_order(seq, s)
            m_sp = _max_convolution_fft(sp)
            if m_sp < best_max:
                best_max = m_sp
                best_seq = sp.copy()

    print(f"[{time.time():.2f}] Best small max={best_max:.6f}")

    # --------------------------------------------------------------
    # 3) Short projected‑gradient refinement (many restarts)
    # --------------------------------------------------------------
    while time.time() < deadline - 30 and best_seq is not None:
        if rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_small * 0.005, size=n_small)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(pool)

        run_dead = min(time.time() + 12.0, deadline - 20.0)
        a_opt, _ = _annealed_gradient_descent_v2_ext(
            seed, run_dead, target_small,
            max_beta=5e5, step_factor=0.7, gamma=0.99
        )
        cur_max = _max_convolution_fft(a_opt)
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved small max={best_max:.6f}")

    # --------------------------------------------------------------
    # 4) Optional deterministic mass‑transfer (quick)
    # --------------------------------------------------------------
    if time.time() < deadline - 20 and best_seq is not None:
        mt_dead = time.time() + (deadline - time.time()) * 0.1
        a_mt, _ = _deterministic_mass_transfer(
            best_seq, mt_dead, target_small,
            start_temp=0.5, end_temp=1e-4, low_frac=0.05
        )
        mt_max = _max_convolution_fft(a_mt)
        if mt_max < best_max:
            best_max = mt_max
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] MT improved small max={best_max:.6f}")

    # --------------------------------------------------------------
    # 5) Upscaling to a larger problem size
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    n_final = n_small
    if time_left > 60:
        n_final = min(12000, n_small * 2)

    target_final = math.sqrt(2.0 * n_final)

    # two up‑scaling strategies
    seq_lin = _resize_sequence(best_seq, n_final)
    seq_rep = _repeat_upscale(best_seq, n_final)

    seq_lin = _project_to_simplex(seq_lin, target_final)
    seq_rep = _project_to_simplex(seq_rep, target_final)

    # re‑order each candidate further
    def improve_order(seq):
        cur_seq, cur_max = seq, _max_convolution_fft(seq)
        alt = _alternating_order(cur_seq)
        m_alt = _max_convolution_fft(alt)
        if m_alt < cur_max:
            cur_seq, cur_max = alt, m_alt
        for s in range(2, 7):
            sp = _spacing_order(cur_seq, s)
            m_sp = _max_convolution_fft(sp)
            if m_sp < cur_max:
                cur_seq, cur_max = sp, m_sp
        return cur_seq, cur_max

    seq_lin, max_lin = improve_order(seq_lin)
    seq_rep, max_rep = improve_order(seq_rep)

    if max_rep < max_lin:
        best_seq = seq_rep
        best_max = max_rep
    else:
        best_seq = seq_lin
        best_max = max_lin

    print(f"[{time.time():.2f}] Upscaled max={best_max:.6f}")

    # --------------------------------------------------------------
    # 6) High‑temperature gradient descent on the large instance
    # --------------------------------------------------------------
    if time.time() < deadline - 10:
        a_final, _ = _annealed_gradient_descent_v2_ext(
            best_seq, deadline, target_final,
            max_beta=1e6, step_factor=0.5, gamma=0.995
        )
        max_final = _max_convolution_fft(a_final)
        if max_final < best_max:
            best_max = max_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final GD improved max={best_max:.6f}")

    # --------------------------------------------------------------
    # 7) Light permutation annealing for the last few seconds
    # --------------------------------------------------------------
    if time.time() < deadline - 2:
        perm_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        best_seq, best_max = _permute_anneal(best_seq, perm_dead)
        print(f"[{time.time():.2f}] Permutation anneal final max={best_max:.6f}")

    # --------------------------------------------------------------
    # 8) Trim trailing zeros and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for possible later runs
    globals()["height_sequence_1"] = final_seq

    # optional final log using the provided evaluator
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
