"""
Hybrid optimizer with extra up‑scaling strategies and permutation search.

The pipeline
--------------
1. Build a diverse pool of small‑size (≈4‑5k) candidates (power‑law,
   exponential, Dirichlet, previous best) and improve each with a
   projected‑gradient descent on a soft‑max approximation of the
   self‑convolution peak.

2. Keep the best small solution and generate several large‑size
   candidates (linear interpolation, integer repetition, a “spread”
   placement that keeps large values far apart).  Choose the one with
   the smallest peak.

3. Polish the chosen large candidate with a high‑β gradient descent.

4. Finally, while time permits, repeatedly random‑permute the multiset
   of values (the L¹‑norm is unchanged) and keep any permutation that
   lowers the peak.  This cheap combinatorial step often beats the
   deterministic alternating order used previously.

All steps respect the non‑negativity and simplex constraints,
use FFT‑based convolutions, and stop before the 1000 s budget expires.
"""

import time
import numpy as np
import heapq

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(Σa)² (with Σa=√(2n) this equals max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _repeat_upscale(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Integer repetition (floor) and a short tail to reach exactly new_len."""
    rep = new_len // seq.shape[0]
    tail = new_len % seq.shape[0]
    out = np.repeat(seq, rep)
    if tail:
        out = np.concatenate([out, seq[:tail]])
    return out.astype(np.float64)


def _spread_upscale(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Place the values of `seq` (sorted descending) on a line of length
    `new_len` using a far‑thest‑point (binary‑split) heuristic, which
    keeps large values as far apart as possible.
    """
    vals = np.sort(seq)[::-1]               # descending
    n = new_len
    out = np.empty(n, dtype=np.float64)

    # max‑heap of intervals: (negative length, left, right)
    heap = [(-(n), 0, n - 1)]
    i = 0
    while heap and i < n:
        neg_len, lo, hi = heapq.heappop(heap)
        mid = (lo + hi) // 2
        out[mid] = vals[i]
        i += 1
        if lo <= mid - 1:
            heapq.heappush(heap, (-(mid - lo), lo, mid - 1))
        if mid + 1 <= hi:
            heapq.heappush(heap, (-(hi - mid), mid + 1, hi))
    return out


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest‑left, second‑largest‑right, third‑largest‑next‑left, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_side = True
    for v in sorted_vals:
        if left_side:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_side = not left_side
    return out


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max bound.
    The schedule slowly raises the soft‑max temperature (beta) while
    decreasing the effective step size.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _permutation_search(base_vals: np.ndarray,
                        deadline: float,
                        max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Repeatedly shuffle `base_vals` (which already satisfies the simplex)

    """
    best = base_vals.copy()
    best_val = _score(best)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iters:
        perm = rng.permutation(len(base_vals))
        cand = base_vals[perm]
        val = _score(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            print(f"[{time.time():.2f}] perm it={it:5d} improvement={best_val:.6f}")
        it += 1
    return best, best_val

            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 15.0, deadline - 2.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(seed,
                                                    run_dead,
                                                    target_expl,
                                                    max_beta=5e5,
                                                    step_factor=0.7,
                                                    gamma=0.99)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional deterministic mass‑transfer at the small scale
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.2
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                       mt_dead,
                                                       target_expl,
                                                       start_temp=0.5,
                                                       end_temp=1e-4,
                                                       low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Upscaling to a larger problem size
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    n_final = 12000 if remaining > 30.0 else n_expl
    target_final = np.sqrt(2.0 * n_final)

    candidates = []

    # linear interpolation
    seq_interp = _resize_sequence(best_seq, n_final)
    seq_interp = _project_to_simplex(seq_interp, target_final)
    candidates.append(seq_interp)

    # integer repetition
    seq_repeat = _repeat_upscale(best_seq, n_final)
    seq_repeat = _project_to_simplex(seq_repeat, target_final)
    candidates.append(seq_repeat)

    # spread placement
    seq_spread = _spread_upscale(best_seq, n_final)
    seq_spread = _project_to_simplex(seq_spread, target_final)
    candidates.append(seq_spread)

    # pick the best among the three
    best_seq = None
    best_score = np.inf
    for seq in candidates:
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # try deterministic alternating order
    alt_seq = _alternating_order(best_seq)
    alt_score = _score(alt_seq)
    if alt_score < best_score:
        best_seq = alt_seq
        best_score = alt_score

    print(f"[{time.time():.2f}] Upscaled n={n_final} start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Final polishing on the large problem
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_fin, val_fin = _annealed_gradient_descent(best_seq,
                                                    deadline,
                                                    target_final,
                                                    max_beta=1e6,
                                                    step_factor=0.5,
                                                    gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Permutation search (cheap combinatorial refinement)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = min(deadline, time.time() + 12.0)   # at most ~12 s for permutations
        best_seq, best_score = _permutation_search(best_seq, perm_dead, max_iters=4000)

    # --------------------------------------------------------------
    # 5. Trim trailing zeros and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final logging using the provided evaluator
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
