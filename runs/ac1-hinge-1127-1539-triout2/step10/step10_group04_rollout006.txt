"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – use the provided ``height_sequence_1`` (if any)
   and up‑sample it to a series of lengths.  For each length we also
   generate a short deterministic grid over a few parametric families
   (exponential, power‑law, mixed, piecewise) and keep the best few.
2. **Smooth‑max Adam** – run several short Adam optimisations on the
   log‑sum‑exp surrogate of the maximal convolution.  The smoothing
   parameter β is increased aggressively so the optimiser quickly
   focuses on the true max.
3. **Projected sub‑gradient descent** – optimise the exact max directly.
   At each step we move opposite the sub‑gradient of the maximal entry,
   project onto the ℓ¹‑simplex and use a diminishing step size.
4. **Deterministic hill‑climb** – exact greedy moves that transfer mass
   from the worst contributing index to the best receiving index while
   respecting the simplex constraint.
5. **Random jitter** – small simulated‑annealing style swaps to escape
   plateaus.
6. **Final polishing** – a longer Adam run followed by a last
   sub‑gradient / hill‑climb pass on the remaining budget.

All steps respect the global time budget (default 1000 s) and keep the
best sequence found.  The final result is returned as a plain list of
floats.
"""

import time
import math
import numpy as np
from operator import itemgetter


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Surrogate (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning rate (scaled with target_sum)
    base_lr = 0.15 * target_sum

    # smoothing schedule
    beta = 2.0
    beta_max = 1e9
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _subgradient_optimize(a0: np.ndarray, target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact maximal convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    # step‑size schedule parameters
    eta0 = 0.5 * target_sum          # initial step‑size factor
    t = 0
    while time.time() < deadline:
        t += 1
        # full convolution and index of the current worst entry
        b = _full_convolution(a)
        k = int(np.argmax(b))

        # sub‑gradient of the max entry:
        #   g_i = 2 * a[k - i]  if 0 ≤ k-i < n, else 0
        g = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_vals = np.arange(i_low, i_high + 1)
        j_vals = k - i_vals
        g[i_vals] = 2.0 * a[j_vals]

        # diminishing step size
        eta = eta0 / math.sqrt(t)
        a -= eta * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Sub‑grad iter {t:5d}  max={best_val:.6f}")

        # a cheap convergence test
        if eta * np.linalg.norm(g) < 1e-12:
            break

    return best_a, best_val

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Systematic family grid search (cheap, deterministic)
# ----------------------------------------------------------------------


def _family_grid_search(n: int, rng, top_k: int = 8):
    """Simple grid over a few parametric families, keep the best ``top_k``."""
    target = math.sqrt(2.0 * n)
    candidates = []
    idx = np.arange(n, dtype=float)

    # exponential decay (lam in a modest range)
    for lam in np.linspace(0.001, 0.12, 12):
        a = np.exp(-lam * idx)
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # power‑law (p in [0.1, 2.0])
    for p in np.linspace(0.1, 2.0, 12):
        a = (idx + 1.0) ** (-p)
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # mixed power‑law × exponential
    for p in np.linspace(0.1, 1.5, 6):
        for lam in np.linspace(0.001, 0.06, 6):
            a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
            a = _project_to_simplex(a, target)
            candidates.append((_max_convolution_fft(a), a))

    # piecewise exponential (break‑point grid)
    for br in np.linspace(0.2, 0.8, 6):
        b = int(br * n)
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # Dirichlet draws (few random draws)
    for _ in range(5):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # keep the best ``top_k``
    candidates.sort(key=itemgetter(0))
    return candidates[:top_k]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` under a wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from the globally available construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – increase dimension gradually
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 50000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise candidate (upsample previous best if possible)
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        remaining = deadline - time.time()
        if remaining < 12.0:
            break

        # --------------------------------------------------------------
        # 1) Small systematic family scan (≈10 % of remaining time)
        # --------------------------------------------------------------
        fam_deadline = time.time() + min(remaining * 0.10, 25.0)
        top_candidates = _family_grid_search(n, rng, top_k=8)
        # also keep the up‑sampled start vector
        init_score = _max_convolution_fft(a_init)
        top_candidates.append((init_score, a_init))
        top_candidates.sort(key=itemgetter(0))
        top_candidates = top_candidates[:8]

        # --------------------------------------------------------------
        # 2) Adam optimisation on the best few candidates
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate ≈55 % of the remaining budget to Adam
        adam_budget = remaining * 0.55
        per_restart = max(5.0, adam_budget / max(1, len(top_candidates)))
        local_best_a = None
        local_best_score = np.inf

        for score, vec in top_candidates:
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)
            a_opt = _adam_optimize(vec, target_sum, this_deadline)
            sc = _max_convolution_fft(a_opt)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = a_opt.copy()
            # keep a global best if the run succeeded spectacularly
            if sc < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = sc
                print(f"[{time.time():.1f}] Adam early improvement (n={n}) "
                      f"score={best_score:.6f}")

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam improvement after restarts (n={n}) "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 3) Projected sub‑gradient descent on the exact max
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            sub_deadline = min(deadline - 0.2, time.time() + remaining * 0.30)
            a_sub, score_sub = _subgradient_optimize(a_cur, target_sum, sub_deadline)
            if score_sub < best_score - 1e-12:
                best_seq = a_sub.copy()
                best_score = score_sub
                print(f"[{time.time():.1f}] Sub‑gradient improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 4) Deterministic hill‑climb (exact max) for a short window
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + remaining * 0.20)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 5) Random jitter / simulated‑annealing style tweaks
        # --------------------------------------------------------------
        while time.time() + 1.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target_sum)
            sc = _max_convolution_fft(cand)
            if sc < best_score - 1e-12:
                best_seq = cand
                best_score = sc
                print(f"[{time.time():.1f}] Random‑tweak improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 6) Optional extra short Adam run if any time left
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            extra_deadline = deadline - 0.5
            a_extra = _adam_optimize(best_seq, target_sum, extra_deadline)
            sc = _max_convolution_fft(a_extra)
            if sc < best_score - 1e-12:
                best_seq = a_extra.copy()
                best_score = sc
                print(f"[{time.time():.1f}] Extra Adam improvement (n={n}) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        final_deadline = deadline - 1.0
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)

        if final_score < best_score - 1e-12:
            best_seq = best_seq.copy()
            best_score = final_score
            print(f"[{time.time():.1f}] Final hill‑climb improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation (the true objective)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
