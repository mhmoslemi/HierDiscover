"""
A fast evolutionary optimiser for the step‑function problem.

The algorithm works in three increasing sizes (4000 → 16 000 → 64 000).
For each size it

1. builds a modest population (random families + the up‑sampled best so far),
2. repeatedly creates children by linear crossover + Gaussian mutation,
   projecting every offspring onto the ℓ¹‑simplex Σa = √(2 n),
3. keeps the best individuals and stops when the time budget for that size runs out,
4. finishes the current best vector with a deterministic hill‑climb that moves mass
   from the index responsible for the maximal convolution entry to a more benign
   location (exact O(n) update of the full convolution).

The final vector is projected once more, evaluated with ``evaluate_sequence``,
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Linear‑algebra utilities
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm from Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution a*a.
    Computed via an FFT of length the next power of two.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole 2n‑1 array)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a larger length, then simplex projection."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex(a_new, target)


# ------------------------------------------------------------
# Random initialisation families
# ------------------------------------------------------------


def _random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n) from a
    mixture of simple parametric families.
    """
    target = math.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)
    family = rng.integers(0, 6)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.2)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform jittered
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    return _project_to_simplex(a, target)


# ------------------------------------------------------------
# Evolutionary optimisation (population + crossover + mutation)
# ------------------------------------------------------------


def _evolve_population(init_vec: np.ndarray, n: int, target_sum: float,
                       deadline: float, rng: np.random.Generator):
    """
    Simple steady‑state EA.
    Returns the best vector observed and its maximal‑convolution value.
    """
    POP_SIZE = 30
    TOP_K = 5

    # initial population: the supplied vector + random draws
    pop = [init_vec.copy()]
    while len(pop) < POP_SIZE:
        pop.append(_random_candidate(n, rng))

    scores = [_max_convolution_fft(v) for v in pop]
    best_vec = pop[int(np.argmin(scores))].copy()
    best_score = min(scores)

    while time.time() < deadline:
        # select elites
        elite_idx = np.argsort(scores)[:TOP_K]
        elites = [pop[i] for i in elite_idx]

        # keep elites
        new_pop = elites.copy()

        # generate children until we hit POP_SIZE
        while len(new_pop) < POP_SIZE:
            p1, p2 = rng.choice(TOP_K, size=2, replace=False)
            parent1, parent2 = elites[p1], elites[p2]
            alpha = rng.random()
            child = alpha * parent1 + (1.0 - alpha) * parent2

            # Gaussian mutation (scale relative to target_sum)
            noise = rng.normal(scale=target_sum * 0.02, size=n)
            child += noise
            child = np.maximum(child, 0.0)
            child = _project_to_simplex(child, target_sum)

            new_pop.append(child)

        # evaluate
        pop = new_pop
        scores = [_max_convolution_fft(v) for v in pop]

        # update global best
        cur_best_idx = int(np.argmin(scores))
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score - 1e-12:
            best_score = cur_best_score
            best_vec = pop[cur_best_idx].copy()

    return best_vec, best_score

def _hill_climb(a0: np.ndarray, deadline: float):

    """

    target = math.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    b = _full_convolution(a)

    rng = np.random.default_rng()

        i = idx[int(np.argmax(contrib))]

        j_high = i_high

            if cand != i:

        if delta <= 0.0:
            break

        # ----- propose a full move -------------------------------------------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ----- accept full move ----------------------------------------------------
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # ----- otherwise try a smaller move (binary search) ------------------------
        improved = False
        delta2 = delta * 0.5
        while delta2 > 1e-14:
            b_test = b.copy()
            b_test[i:i + n] -= 2.0 * delta2 * a
            b_test[j:j + n] += 2.0 * delta2 * a
            if k2i < b_test.size:
                b_test[k2i] += delta2 * delta2
            if k2j < b_test.size:
                b_test[k2j] += delta2 * delta2
            if kij < b_test.size:
                b_test[kij] -= 2.0 * delta2 * delta2
            if b_test.max() < b.max() - 1e-12:
                a[i] -= delta2
                a[j] += delta2
                b = b_test
                if b.max() < best_score - 1e-12:
                    best_score = b.max()
                    best_a = a.copy()
                improved = True
                break
            delta2 *= 0.5

        if not improved:
            # minor jitter to get out of a flat region
            jitter = rng.normal(scale=1e-6, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method uses an evolutionary search on three increasing dimensions
    (4000 → 16 000 → 64 000) followed by a deterministic hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seed from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            candidate = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = candidate.size
            target0 = math.sqrt(2.0 * n0)
            candidate = _project_to_simplex(candidate, target0)
            best_seq = candidate.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Size schedule – we gradually increase dimensionality
    # ------------------------------------------------------------------
    schedule = [4000, 16000, 64000]

    for stage, n in enumerate(schedule):
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # initialise with an upsampled version of the current best, if any
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_candidate(n, rng)

        # allocate time for the evolutionary part (≈ 45 % of remaining)
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        evo_time = remaining * 0.45
        evo_deadline = time.time() + evo_time

        # ------------------------------------------------------------------
        # Evolutionary optimisation
        # ------------------------------------------------------------------
        vec, score = _evolve_population(a_init, n, target_sum,
                                         evo_deadline, rng)
        if score < best_score - 1e-12:
            best_seq = vec.copy()
            best_score = score
            print(f"[{time.time():.1f}] Evolution improvement (n={n}) "
                  f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Deterministic hill‑climb on the exact max (≈ 35 % of remaining)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 3.0:
            break
        hc_time = remaining * 0.35
        hc_deadline = time.time() + hc_time
        best_seq, hc_score = _hill_climb(best_seq, hc_deadline)
        if hc_score < best_score - 1e-12:
            best_score = hc_score
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                  f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Tiny random perturbations (the rest of the budget for this stage)
        # ------------------------------------------------------------------
        while time.time() + 1.0 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target_sum)
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_score - 1e-12:
                best_seq = cand
                best_score = cand_score
                print(f"[{time.time():.1f}] Random tweak improvement (n={n}) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the remaining budget (Adam‑like smoothing + HC)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5
        # a light Adam‑type run using the smooth‑max surrogate
        # (re‑implemented locally to avoid pulling in the previous heavy code)
        # --------------------------------------------------
        a = best_seq.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        base_lr = 0.12 * target
        beta = 2.0
        beta_max = 1e7
        t = 0
        while time.time() < final_deadline:
            t += 1
            # compute smoothed max gradient
            size1 = _next_pow2(2 * n - 1)
            af = np.fft.rfft(a, n=size1)
            b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]
            max_b = np.max(b)
            exps = np.exp(beta * (b - max_b))
            p = exps / np.sum(exps)

            # gradient = 2 * (a ⋆ p)
            a_rev = a[::-1]
            conv_len = n + (2 * n - 1) - 1          # 3n‑2
            size2 = _next_pow2(conv_len)
            Ar_fft = np.fft.rfft(a_rev, n=size2)
            Pr_fft = np.fft.rfft(p, n=size2)
            conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
            g = 2.0 * conv_rev[n - 1: n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a -= step
            a = _project_to_simplex(a, target)

            # increase smoothing periodically
            if t % 250 == 0:
                beta = min(beta * 2.0, beta_max)

        best_seq = a.copy()
        # final exact hill‑climb (short)
        final_hc_deadline = deadline - 0.2
        best_seq, final_hc_score = _hill_climb(best_seq, final_hc_deadline)
        if final_hc_score < best_score - 1e-12:
            best_score = final_hc_score
            print(f"[{time.time():.1f}] Final hill‑climb improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # a uniform fallback (should never happen)
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
