"""
Propose a step function with a smaller max‑self‑convolution.

The algorithm works in several stages within the 1000 s wall‑clock budget:

1.  Load a previous solution (if any) and use it as a seed.
2.  A cheap parameter sweep on a modest length (1024) to obtain a good
    initial shape.
3.  Several short Adam runs on the same length to improve the shape.
4.  Upsample the best vector to a larger length (4000) – longer sequences
    can spread the mass more evenly and usually lower the max‑convolution.
5.  Adam optimisation on the large size.
6.  A dedicated sub‑gradient descent that works directly on the exact
    max‑convolution (no smoothing).  The sub‑gradient of `max(a*a)` is
    obtained via a correlation of the active mask with `a`.
7.  A final deterministic hill‑climb (exact max) as a polish step.
8.  Trailing near‑zero entries are trimmed and the official
    `evaluate_sequence` is called for reporting.

All optimisation steps respect the simplex constraint Σa = √(2 n) and
non‑negativity by Euclidean projection onto the ℓ¹‑simplex.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (fast)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Surrogate (soft‑max) gradient used by Adam
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth max (log‑sum‑exp) of the self‑convolution.
    Returns (g, b) where b = a*a and g = 2·(a ⋆ p),  p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initial families
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Shape that is large near the two edges and decays towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Exact max‑convolution gradient (sub‑gradient)
# ----------------------------------------------------------------------


def _max_and_subgradient(a: np.ndarray, tol: float = 1e-12):
    """
    Return (M, g) where M = max(a*a) and g is a sub‑gradient of M.
    The active set is the mask of entries within `tol` of M.
    """
    n = a.size
    conv_full = _full_convolution(a)
    M = conv_full.max()
    mask = (conv_full >= M - tol).astype(float)

    size = _next_pow2(len(mask) + n - 1)          # >= 3n‑2
    mask_fft = np.fft.rfft(mask, n=size)
    a_fft = np.fft.rfft(a, n=size)
    # correlation = ifft(mask_fft * conj(a_fft))
    corr_full = np.fft.irfft(mask_fft * np.conj(a_fft), n=size)
    corr_full = corr_full[: len(mask) + n - 1]

    # gradient = 2 * correlation shifted by (n‑1)
    g = 2.0 * corr_full[n - 1 : n - 1 + n]
    return M, g


def _subgradient_optimize(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Simple sub‑gradient descent on the exact max‑convolution.
    Projects onto the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    # initial step size heuristic
    step_factor = 0.5 * target_sum
    while time.time() < deadline:
        M, g = _max_and_subgradient(a)
        # compute a step (norm‑scaled)
        norm_g = np.linalg.norm(g) + 1e-15
        eta = step_factor / norm_g
        # trial update
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        M_new = _max_convolution_fft(a_new)
        if M_new < M - 1e-8:
            a = a_new
            # increase step size modestly after improvement
            step_factor *= 1.05
        else:
            # decrease step size if no improvement
            step_factor *= 0.7
            if step_factor * target_sum < 1e-12:
                break
    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller transfer
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan a few one‑parameter families (power‑law, exponential,
    edge‑heavy, mixed) and return the best vector found (according to the
    exact max‑convolution).
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law exponents
    for p in np.linspace(0.1, 2.0, 40):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.0005, 0.02, 40):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy (large at the ends, decays towards centre)
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 20):
        for lam in np.linspace(0.001, 0.01, 10):
            a = (idx ** (-p)) * np.exp(-lam * lin)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    return best_a, best_score

    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        # allocate ~5 s per restart
        per_restart = 5.0
        n_restarts = max(1, int(remaining // per_restart))
        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            if r == 0 and best_seq is not None and best_seq.size == n_small:
                a0 = best_seq.copy()
            else:
                # mix the two families for diversity
                a0 = _random_initial_sequence(n_small, rng) if rng.random() < 0.6 else _edge_high_initial_sequence(n_small, rng)

            a_opt = _adam_optimize(a0, target_small,
                                   min(deadline, time.time() + per_restart * 0.9))
            s = _max_convolution_fft(a_opt)
            if s < best_score - 1e-12:
                best_seq, best_score = a_opt.copy(), s
                print(f"[{time.time():.1f}] Adam restart {r+1}/{n_restarts} (n={n_small}) → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Upsample to a larger size (n=4000) and refine with Adam
    # --------------------------------------------------------------
    n_large = 4000
    target_large = math.sqrt(2.0 * n_large)

    if time.time() < deadline - 10.0:
        a_large = _upsample_sequence(best_seq, n_large)
        # a short Adam run
        a_opt_large = _adam_optimize(a_large, target_large,
                                     deadline - 5.0)
        s_large = _max_convolution_fft(a_opt_large)
        if s_large < best_score - 1e-12:
            best_seq, best_score = a_opt_large.copy(), s_large
            print(f"[{time.time():.1f}] Adam on large (n={n_large}) → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Sub‑gradient descent on the exact max‑convolution (large size)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_sub = _subgradient_optimize(best_seq, target_large,
                                      deadline - 2.0)
        s_sub = _max_convolution_fft(a_sub)
        if s_sub < best_score - 1e-12:
            best_seq, best_score = a_sub.copy(), s_sub
            print(f"[{time.time():.1f}] Sub‑gradient descent → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final deterministic hill‑climb (exact max) for polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        b_cur = _full_convolution(best_seq)
        a_hc, s_hc = _hill_climb_deterministic(best_seq, b_cur,
                                               deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq, best_score = a_hc.copy(), s_hc
            print(f"[{time.time():.1f}] Final hill‑climb → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Trim trailing near‑zeros
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n_fallback = 1024
        best_seq = np.full(n_fallback, math.sqrt(2.0 * n_fallback) / n_fallback)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 7) Official evaluation report
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
```

This implementation follows the outlined multi‑stage strategy, adds an exact sub‑gradient optimisation stage, and works with an up‑sampled larger sequence that often yields a lower maximal self‑convolution, thus improving the reported bound.
