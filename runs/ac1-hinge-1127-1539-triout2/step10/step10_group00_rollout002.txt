def _simulated_annealing(a0, target_sum, deadline):

    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    # initial temperature based on best_val
    T0 = best_val * 0.04   # maybe fewer? earlier used 0.015 * best_val; adjust.
    # we will schedule temperature linearly down to near 0.
    while time.time() < deadline:
        # compute conv
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # choose donor: sample from idx weighted by weight (more contribution)
        wsum = weight.sum()
        if wsum > 0:
            donor = rng.choice(idx, p=weight / wsum)
        else:
            donor = rng.integers(n)

        # choose receiver: choose from low contribution tail or random
        low_thr = np.percentile(weight, 7)  # 7% lower tail
        low_candidates = idx[weight <= low_thr]
        if low_candidates.size > 0:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        # transition: delta fraction depends on temperature
        # high T: allow large delta (up to 0.6 of donor mass), low T: small
        temp_ratio = max(0.0, 1.0 - (time.time() - start) / (deadline - start))
        max_frac = 0.6 * temp_ratio + 0.04  # between 0.04 and 0.64
        max_delta = a[donor] * max_frac
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # propose new a
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # renorm
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        # temperature for acceptance
        elapsed = time.time() - start
        T = max(T0 * (1 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()

            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

def _subgradient_max_descent(a0, target_sum, deadline):

    a = a * (target_sum / a.sum())

    it = 0

        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (a.size - 1))
        hi = min(k_max, a.size - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # compute safe step size
        pos = grad > 1e-15
        if not np.any(pos):
            break
        safe_eta = np.min(a[pos] / grad[pos])
        # Use decreasing step size: start with safe_eta * 0.9 etc
        eta = min(0.9 * safe_eta, 0.5 / np.sqrt(it + 1))
        # update
        a = a - eta * grad
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad iteration {it} improved score={best_val:.6f}")

            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        it += 1
    return best, best_val

"""
A fast multi‑stage optimiser for the non‑negative step‑function problem.

The algorithm works as follows

1.  A light parametric sweep (power‑law + exponential decay) on a reduced
    length finds a promising shape.
2.  The best shape is resized to the full problem size and projected onto the
    ℓ¹‑simplex (Σa = √(2·n)).
3.  A short projected sub‑gradient descent on the *max* convolution entry
    improves the shape quickly.
4.  A long simulated‑annealing phase performs mass‑transfer moves that
    specifically target the dominant convolution entry, using a temperature
    schedule and a dynamic step‑size.  This phase occupies the majority of the
    time budget and is able to squeeze a few extra points out of the objective.
5.  The final sequence is trimmed of trailing zeros and returned.

All stages respect a user‑supplied time budget (default 1000 s) and print
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 length suitable for a full convolution of size 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Evaluate the objective 2·n·max(a*a)/(Σa)².
    The L¹ norm is assumed >0.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to change the length of a sequence."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence (i+1)^{‑p} normalised to have ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Stage 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """Search a 2‑parameter family (i+1)^‑p·exp(‑λ·i/small_n)."""
    p_vals = np.linspace(0.2, 1.5, 21)
    lam_vals = np.linspace(0.0, 12.0, 25)
    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.5)
        lam = rng.exponential(1.0) * 4.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – projected sub‑gradient descent on the maximal convolution entry
# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max_k (a*a)_k."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # safe step (prevent negativity)
        pos = grad > 1e-15
        if not np.any(pos):
            break
        safe_eta = np.min(a[pos] / grad[pos])
        eta = min(0.9 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} improved score={best_val:.6f}")

        # occasional jitter to escape shallow plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer SA that focuses on the currently maximal convolution entry.
    The temperature cools linearly; the allowed transfer size shrinks with T.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    # initial temperature proportional to the current score
    T0 = best_val * 0.02

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # donor – proportional to contribution to the maximal entry
        wsum = weight.sum()
        if wsum > 0:
            donor = rng.choice(idx, p=weight / wsum)
        else:
            donor = rng.integers(n)

        # receiver – from the low‑contribution tail
        low_thr = np.percentile(weight, 7)  # ~7 % tail
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        # dynamic maximum fraction of donor mass that may be moved
        elapsed = time.time() - start
        temp_frac = max(0.0, 1.0 - elapsed / (deadline - start))
        max_frac = 0.04 + 0.56 * temp_frac          # 0.60 → 0.04
        max_delta = a[donor] * max_frac
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        # linear cooling schedule
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved best score={best_val:.6f}")

        # rare random jitter to diversify
        if rng.random() < 0.002:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val

        best_seq = best_seq * (target_sum / best_seq.sum())
        best_score = _score(best_seq)
        print(f"[{time.time():.2f}] After parametric warm‑start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (short)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + min(0.20 * (deadline - time.time()), 120.0)
        a_sub, val_sub = _subgradient_max_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
            print(f"[{time.time():.2f}] After sub‑gradient score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – long simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
            print(f"[{time.time():.2f}] After SA refinement score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Remember the result for possible warm‑starts in the next call
    globals()["height_sequence_1"] = final_seq
    return final_seq
