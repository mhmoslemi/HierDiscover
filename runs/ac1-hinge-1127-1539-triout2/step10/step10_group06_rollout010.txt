```python
"""
Simulated annealing + exponential mirror descent search for low max
convolution step function.
Algorithm:
 1. Choose length n based on remaining time budget.
 2. Warm start from previously found best sequence or random Dirichlet.
 3. Short exponential mirror descent on a soft‑max approximation of the
    maximal autocorrelation (helps quickly reach a good basin).
 4. Long simulated‑annealing stage that repeatedly moves a tiny amount of
    mass from a “donor’’ index that participates strongly in the current
    maximal convolution entry to a “receiver’’ index that participates
    weakly.  Acceptance follows a linear cooling schedule.
 5. Return the best sequence discovered.
"""

import time, math, numpy as np

def _conv_fft(a):
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _max_convolution_fft(a):
    return float(_conv_fft(a).max())

def _score(a):
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s*s))

def _project_to_simplex(v, s):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1]+1)
    w = np.maximum(v - theta, 0.0)
    return w

def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, b

def _exponential_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1
    return best_a, best_val

def _simulated_annealing_refine(a_start, target_sum, deadline):
    n = a_start.shape[0]
    a = a_start.copy()
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature as a fraction of current max
    T0 = cur_max * 0.05 if cur_max > 0 else 1.0
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling
        T = max(0.0, T0 * (1.0 - elapsed / total))
        # locate current maximal entry
        b = _conv_fft(a)  # full convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)
        # pick a receiver that currently has small amplitude
        # simple uniform sample, but bias towards low values
        if rng.random() < 0.7:
            # pick from lower quantile
            thresh = np.percentile(a, 30)
            low_idx = np.where(a <= thresh)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)
        else:
            receiver = rng.integers(n)
        if donor == receiver:
            continue
        # magnitude of transfer
        max_delta = min(a[donor] * 0.3, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta + 1e-12
        # create candidate vector
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # evaluate
        new_max = _max_convolution_fft(a_new)
        # acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        elif T > 0:
            prob = math.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} best_max={best_max:.6f}")
        it += 1
    return best_a, best_max

def propose_candidate(budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low benchmark score.
    Uses a short exponential mirror‑descent warm‑up followed by a long
    simulated‑annealing refinement that moves tiny amounts of mass
    from indices that heavily contribute to the current maximal
    convolution entry towards weaker indices.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # problem size (larger n gives more degrees of freedom)
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000

    target_sum = math.sqrt(2.0 * n)

    # warm start
    if 'height_sequence_1' in globals():
        init = _resize_sequence(np.array(globals()['height_sequence_1'], dtype=np.float64), n)
    else:
        rng = np.random.default_rng()
        init = rng.random(n) + 1e-6   # ensure positivity
    init = init * (target_sum / init.sum())

    # Phase A: exponential mirror descent (≈15 % of time)
    phaseA_deadline = time.time() + 0.15 * (deadline - time.time())
    a_phase, score_phase = _exponential_descent(init, target_sum, phaseA_deadline)
    print(f"[{time.time():.2f}] After exponential phase best score={score_phase:.6f}")

    # Phase B: simulated annealing refinement (remaining budget)
    a_final, max_final = _simulated_annealing_refine(a_phase, target_sum, deadline)
    final_score = max_final   # because sum is already normalised
    print(f"[{time.time():.2f}] After SA best max={final_score:.6f}")

    # Trim trailing zeros (if any) and final evaluation
    arr = np.array(a_final, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1] + 1]
    final_seq = arr.tolist()
    try:

"""
Simulated‑annealing + exponential mirror descent search for a low
max‑convolution step function.

The algorithm proceeds in two stages:

1. **Exponential mirror descent** on a soft‑max surrogate of the
   maximal autocorrelation.  This quickly brings a random / warm‑start
   vector into a good basin while honouring the simplex constraint
   Σa = √(2 n).

2. **Simulated annealing refinement** that repeatedly moves a tiny
   amount of mass from a “donor’’ index (large contribution to the
   current maximal convolution entry) to a “receiver’’ index (small
   contribution).  Acceptance follows a linear cooling schedule.
   The convolution is evaluated with FFTs, which makes many
   iterations cheap even for n≈15000.

The routine respects the supplied time budget (default 1000 s) and
stores the best sequence in the global variable `height_sequence_1`
for warm‑starts in later calls.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  With Σa = √(2 n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phase‑A)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p) – length n
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # double β every few hundred iterations (up to a large cap)
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _simulated_annealing_refine(a_start: np.ndarray,
                                target_sum: float,
                                deadline: float) -> tuple[np.ndarray, float]:
    """
    Long SA loop that moves tiny amounts of mass from a donor index that
    strongly contributes to the current maximal convolution entry towards a
    receiver index with weaker contribution.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()

    start = time.time()
    total = deadline - start
    # initial temperature as a modest fraction of the current maximum
    T0 = max(cur_max * 0.2, 1e-3)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling
        T = max(0.0, T0 * (1.0 - elapsed / total))

        # full convolution to locate the current worst index
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        # donor – pick an index that participates heavily in entry k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – bias towards small entries (30‑th percentile)
        if rng.random() < 0.7:
            thresh = np.percentile(a, 30)
            low = np.where(a <= thresh)[0]
            if low.size:
                receiver = rng.choice(low)
            else:
                receiver = rng.integers(n)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # amount of mass to transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.4, target_sum * 5e-4) * (T / T0 + 0.1)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta + 1e-12

        # propose new candidate
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        new_max = _max_convolution_fft(a_new)

        # acceptance test
        accept = False
        if new_max < cur_max:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} best_max={best_max:.6f}")

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the caller

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more freedom)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        rng = np.random.default_rng()
        init = rng.random(n) + 1e-6          # guarantee strictly positive entries

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – short exponential mirror descent (≈15% of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = time.time() + 0.15 * (deadline - time.time())
    a_phase, score_phase = _exponential_descent(init, target_sum, phase_a_deadline)
    print(f"[{time.time():.2f}] After exponential phase best score={score_phase:.6f}")

    # --------------------------------------------------------------
    # Phase B – long simulated annealing refinement (remaining budget)
    # --------------------------------------------------------------
    a_final, max_final = _simulated_annealing_refine(a_phase, target_sum, deadline)
    print(f"[{time.time():.2f}] After SA best max={max_final:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up, evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(a_final, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
