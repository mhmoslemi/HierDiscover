"""
Improved step‑function optimizer.

* Phase 0 – dense parametric sweep (pure/exp power law, symmetric and
  bi‑power laws).  Fast evaluation via an FFT based max‑convolution.
* Phase A – projected sub‑gradient descent on the exact max‑convolution.
  The sub‑gradient is computed analytically (no soft‑max) and the iterate
  is projected onto the simplex {x≥0, Σx = √(2 n)}.
* Phase B – deterministic mass‑transfer (greedy reduction of the current
  peak) followed by a light‑weight random move phase.
* Phase C – a final short Adam‑style soft‑max descent as a polish step.

The routine respects the supplied time budget and returns the best
non‑negative sequence found (trailing zeros are trimmed).  The score is
the benchmark objective `evaluate_sequence`, which under the fixed‑sum
normalisation equals the maximal entry of the auto‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_and_mask(a: np.ndarray):
    """Return (max value, full convolution, binary mask of maximal entries)."""
    conv = _conv_fft(a)
    max_val = conv.max()
    mask = (conv >= max_val - 1e-12).astype(np.float64)
    return max_val, conv, mask


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i≥0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (Phase 0)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sym_power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    half = (n + 1) // 2
    a_half = (np.arange(1, half + 1, dtype=np.float64) ** (-p))
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[-2::-1]])
    a *= target_sum / a.sum()
    return a


def _sym_exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    half = (n + 1) // 2
    i = np.arange(half, dtype=np.float64)
    a_half = (i + 1) ** (-p) * np.exp(-q * i / n)
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[-2::-1]])
    a *= target_sum / a.sum()
    return a


def _bi_power_law_sequence(n: int, p: float, r: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * (n - i) ** (-r)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring shortcut (max entry of auto‑convolution)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Benchmark objective under the fixed‑sum normalisation."""
    return _max_convolution_and_mask(a)[0]


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _subgradient(a: np.ndarray):
    """
    Compute a sub‑gradient of max_i (a*a)_i  and the current maximum value.
    """
    n = a.shape[0]
    max_val, conv, _ = _max_convolution_and_mask(a)
    idx_max = np.where(conv >= max_val - 1e-12)[0]

    grad = np.zeros_like(a)
    for k in idx_max:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # derivative ∂b_k/∂a_i = 2 * a_{k-i}
        grad[idx] += 2.0 * a[k - idx]
    return grad, max_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (Phase A)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Simple diminishing step size is used.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    while time.time() < deadline:
        grad, cur_max = _subgradient(a)
        # decreasing step – the denominator grows linearly
        step = step0 / (1.0 + it * 0.05)
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} max={best_val:.6f}")

        it += 1
    return best_a, best_val


def _subgradient_descent_with_restarts(init: np.ndarray,
                                      target_sum: float,
                                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform several short sub‑gradient runs starting from jittered points.
    Keeps the best result.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.3)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a_opt, val_opt = _subgradient_descent(start,
                                              target_sum,
                                              time.time() + run_budget,
                                              step0=target_sum * 0.2)
        if val_opt < best_val - 1e-12:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Sub‑grad restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the current maximal convolution entry.
    Moves mass from the most harmful to the least harmful position so that
    the peak is lowered towards the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]
        # pick the first maximal index
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weighted contributions w_i = a_i + a_{k0-i}
        w = a[idx] + a[k0 - idx]

        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contributions of donor / receiver to the peak
        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            # cannot improve this peak – give up
            break

        # second‑largest value (excluding all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # required reduction in peak to meet the second maximum
        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.5, delta_needed * 0.99)

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Light random mass‑transfer (exploratory phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simple random moves: shift a small fraction of mass from a large entry
    to a small entry.  The candidate is kept if it improves the score.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score

# Adam‑style soft‑max GD (Phase C polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, full convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style projected GD on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
