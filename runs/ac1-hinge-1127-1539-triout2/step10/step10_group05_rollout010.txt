"""
Propose a non‑negative step function (vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The implementation follows a three‑stage recipe:

1. **Broad parameter sweep** – cheap families (exponential, power‑law,
   edge‑heavy, piece‑wise constant, mixtures) are evaluated with an FFT
   based max‑convolution.  The best candidate supplies a good warm‑start.

2. **Projected Adam on the smooth‑max surrogate** – we minimise a
   log‑sum‑exp approximation of the max‑convolution while always
   projecting onto the ℓ¹‑simplex Σa = √(2 n).  The optimizer keeps the
   vector symmetric (a[i] = a[n‑1‑i]) which is known to be helpful for
   autocorrelation flattening.  The schedule for the smoothing parameter
   β grows exponentially, sharpening the objective.

3. **Deterministic hill‑climbing + short annealing** – using the exact
   convolution we repeatedly move a small amount of mass from the index
   that contributes most to the current worst convolution entry to a
   better destination.  Finally a few annealing moves are performed to
   escape shallow plateaus.

All phases respect a wall‑clock budget (default 1000 s).  The final
vector is trimmed of trailing near‑zeros before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of  (1/β)·log Σ exp(β·b)  where  b = a∗a.
    Returns (g, b) with
        g = 2·(a ⋆ p)            (⋆ = linear convolution)
    where p_k ∝ exp(β·b_k) .
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)

    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a ⋆ p  (full length 3n‑2, we only need the middle n entries)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)

    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]   # central n entries
    return g, b


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce v[i] = v[n‑1‑i] (palindromic vector)."""
    return 0.5 * (v + v[::-1])


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Enforces symmetry after each projection.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate – roughly a few percent of the target sum
    base_lr = 0.12 * target_sum
    # smoothing parameter schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)

        # occasional jitter to avoid stagnation
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            a = _symmetrize(a)

        # increase smoothing strength
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initial families (all symmetric)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a symmetric non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, mixed) are sampled.
    """
    target = math.sqrt(2.0 * n)
    half = (n + 1) // 2
    idx = np.arange(half, dtype=float)

    family = rng.integers(0, 7)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.2, 1.2)
        a_half = (idx + 1) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.02)
        a_half = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.2, 1.0)
        lam = rng.exponential(scale=0.02)
        a_half = (idx + 1) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # piecewise constant (high edges, low centre)
        low = rng.uniform(0.001, 0.02)
        high = rng.uniform(0.02, 0.1)
        split = rng.integers(int(0.3 * half), int(0.7 * half))
        a_half = np.concatenate([np.full(split, high),
                                 np.full(half - split, low)])

    elif family == 4:                   # two‑exponential mixture
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        mix = rng.random()
        a_half = mix * np.exp(-lam1 * idx) + (1 - mix) * np.exp(-lam2 * idx)

    elif family == 5:                   # Dirichlet (uniform on simplex)
        a_half = rng.dirichlet(alpha=np.ones(half)) * target

    else:                               # near‑uniform with tiny jitter
        a_half = np.full(half, target / n)

    # mirror to full length
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[-2::-1]])

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    a = _symmetrize(a)
    return a


def _edge_heavy_sequence(n: int, rng) -> np.ndarray:
    """Large values near the two edges, decaying towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.015)
    idx = np.arange(n, dtype=float)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    a = _symmetrize(a)
    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # index of the current peak
        k = int(np.argmax(b))

        # indices that actually contribute to this peak
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[k - idx]

        # source index – the one with largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination j (different from i) that reduces the peak
        # we look for the index whose counterpart values are smallest
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[k - np.arange(i_low, i_high + 1)]

        # sort candidate destinations by counterpart size
        cand_order = np.argsort(counterpart)
        j = None
        for cand in cand_order:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # move a fraction of the source mass
        delta = a[i] * 0.7          # 70 % of the source mass
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # fallback: try a smaller step
            delta2 = delta * 0.5
            improved = False
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # jitter to escape possible plateaus
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                a = _symmetrize(a)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing (O(n) per move).  Returns the best vector
    and its maximal convolution entry seen before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9995

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.4)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5
        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Small parametric sweep (cheap but often finds a good basin)
# ----------------------------------------------------------------------


def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan a few one‑parameter families (power‑law, exponential,
    edge‑heavy, mixed, piecewise) and return the best vector found
    (according to exact max‑convolution).
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law
    for p in np.linspace(0.1, 2.0, 30):
        a_half = idx ** (-p)
        a = np.concatenate([a_half[: (n + 1) // 2],
                            a_half[: n // 2][::-1]])[:n]
        a = _project_to_simplex(a, target)
        a = _symmetrize(a)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # exponential decay
    for lam in np.linspace(0.0004, 0.02, 30):
        a_half = np.exp(-lam * np.arange((n + 1) // 2))
        a = np.concatenate([a_half,
                            a_half[: n // 2][::-1]])[:n]
        a = _project_to_simplex(a, target)
        a = _symmetrize(a)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy (large near ends)
    for lam in np.linspace(0.001, 0.02, 20):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        a = _symmetrize(a)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 12):
        for lam in np.linspace(0.001, 0.01, 8):
            a_half = (idx[: (n + 1) // 2] ** (-p)) * np.exp(-lam * np.arange((n + 1) // 2))
            a = np.concatenate([a_half,
                                a_half[: n // 2][::-1]])[:n]
            a = _project_to_simplex(a, target)
            a = _symmetrize(a)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # piecewise constant (high edges, low centre)
    for high in np.linspace(0.02, 0.15, 10):
        low = high * rng.uniform(0.05, 0.3)
        split = rng.integers(int(0.25 * n), int(0.75 * n))
        a = np.full(n, low)
        a[:split] = high
        a[-split:] = high
        a = _project_to_simplex(a, target)
        a = _symmetrize(a)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Seed from a previously found construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded (n={n0}) → score={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # ------------------------------------------------------------------
    # 1)  Choose a working length – larger than 2000 gives more degrees
    # ------------------------------------------------------------------
    n = 4000                     # feel free to increase if time permits
    target = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1a)  Fast parametric sweep
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_sweep, score_sweep = _parameter_sweep(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep improvement → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1b)  Projected Adam restarts (moderate budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        # allocate ~7 s per restart; use as many restarts as time allows
        n_restarts = max(1, int(remaining // 7.0))
        per_restart = remaining / n_restarts
        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            if r == 0 and best_seq is not None:
                a0 = best_seq.copy()
            else:
                # pick a diverse random start
                if rng.random() < 0.5:
                    a0 = _random_initial_sequence(n, rng)
                else:
                    a0 = _edge_heavy_sequence(n, rng)

            a_opt = _adam_optimize(a0, target, time.time() + per_restart * 0.9)
            s = _max_convolution_fft(a_opt)
            if s < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = s
                print(f"[{time.time():.1f}] Adam restart {r+1}/{n_restarts} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Deterministic hill‑climb (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target)
        a_cur = _symmetrize(a_cur)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur,
                                                deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Hill‑climb → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Final polishing – short Adam run + hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_polish = _adam_optimize(best_seq, target, deadline - 1.5)
        b_polish = _full_convolution(a_polish)
        a_final, s_final = _hill_climb_deterministic(a_polish, b_polish,
                                                     deadline - 0.5)
        if s_final < best_score - 1e-12:
            best_seq = a_final.copy()
            best_score = s_final
            print(f"[{time.time():.1f}] Final polish → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5)  Trim trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple uniform vector (should never happen)
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 6)  Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
