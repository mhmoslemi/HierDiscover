def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose n
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 11000
    else:
        n = 6000

    target_sum = math.sqrt(2.0 * n)

    # warm start
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        init = (np.arange(1, n+1, dtype=np.float64) ** (-0.6))

    init = _project_to_simplex(init, target_sum)

    # stage 1: smooth gradient descent
    sub_deadline = time.time() + 0.42 * (deadline - time.time())
    best_a, best_val = _smooth_gradient_descent(init, target_sum, sub_deadline)
    print(f"[{time.time():.2f}] After smooth gradient best max={best_val:.6f}")

    # stage 2: short random restarts of smooth gradient
    while time.time() < deadline - 0.1 * (deadline - start):
        # allocate small chunk for each restart
        remaining_time = deadline - time.time()
        if remaining_time < 5.0:
            break
        run_time = min(6.0, remaining_time * 0.2)
        jitter = np.random.default_rng().normal(scale=target_sum * 0.02, size=n)
        start_seq = best_a + jitter
        start_seq = np.maximum(start_seq, 0.0)
        start_seq = _project_to_simplex(start_seq, target_sum)
        a_tmp, val_tmp = _smooth_gradient_descent(start_seq, target_sum,
                                                  time.time() + run_time)
        if val_tmp < best_val:
            best_val = val_tmp
            best_a = a_tmp.copy()
            print(f"[{time.time():.2f}] Restart improved best max={best_val:.6f}")

    # stage 3: local refinement
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_a, target_sum, deadline)
        if val_ref < best_val:
            best_val = val_ref
            best_a = a_ref

    # final clean
    arr = np.array(best_a)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error: {e}")

    globals()["height_sequence_1"] = final_seq

    return final_seq

"""
Hybrid smoothed‑gradient + simulated‑annealing refinement.

The search consists of three stages:

1.  **Smoothed gradient descent** –  
    Minimise a smooth approximation of the maximal convolution entry.
    The smoothness is controlled by a temperature‑like parameter α which
    is slowly increased, turning the objective from a soft‑max to the true
    max.  After each step the iterate is projected onto the simplex
    Σa = √(2 n) to keep the scale invariant.

2.  **Short random restarts** –  
    Jitter the current best sequence and run a brief smoothed‑gradient
    optimisation.  The best restart is kept.

3.  **Exact mass‑transfer refinement** –  
    A fast O(n) simulated‑annealing hill‑climber that transfers a small
    amount of mass from “bad’’ (large contribution to the maximal
    convolution entry) to “good’’ indices.  This stage uses the same
    update logic as the previous best solution but starts from the
    improved sequence.

The routine respects the 1000 s time budget, prints intermediate scores
and stores the best sequence in the global variable ``height_sequence_1``
for warm‑starts in subsequent calls.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Smoothed gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _smooth_gradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of max(conv(a,a)).
    The schedule for the smoothness parameter α starts small and is
    multiplied by 1.02 each iteration (capped at 80).  The step size is
    ``base_step / sqrt(it+1)`` and the gradient direction is normalised.
    Returns the best sequence seen and its (unsclaed) maximal convolution
    value.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    b = _conv_fft(a)
    best_val = b.max()

    rng = np.random.default_rng()
    it = 0
    base_step = target_sum * 0.5
    alpha = 0.5          # initial smoothness
    max_alpha = 80.0

    while time.time() < deadline:
        # convolution and smooth weights
        b = _conv_fft(a)
        max_b = b.max()
        # soft‑max weights (avoid overflow by subtracting max)
        weights = np.exp(alpha * (b - max_b))
        w_sum = weights.sum()
        if w_sum == 0.0:
            # extreme underflow – fall back to uniform weights
            weights = np.ones_like(b) / b.shape[0]
        else:
            weights /= w_sum

        # gradient = 2 * sum_j a_j * weights_{i+j}
        rev = weights[::-1]
        conv_full = _linear_convolve_fft(a, rev)          # length 3n‑2
        L = len(weights) - 1                               # = 2n‑2
        grad = 2.0 * conv_full[L - np.arange(a.shape[0])]

        # step (normalised direction)
        step = base_step / math.sqrt(it + 1.0)
        gn = np.linalg.norm(grad)
        if gn > 0.0:
            a -= step * grad / gn

        # keep feasibility
        a = _project_to_simplex(a, target_sum)

        # bookkeeping
        if max_b < best_val:
            best_val = max_b
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] SmoothGrad it={it:6d} max={best_val:.6f} α={alpha:.2f}"
            )

        # increase smoothing (approach true max)
        alpha = min(alpha * 1.02, max_alpha)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape[0])
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact mass‑transfer refinement (same as previous solution)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[
    np.ndarray, float
]:
    """
    Simulated‑annealing hill‑climbing on the true max‑convolution.
    Moves a tiny amount of mass from a “bad’’ index (large contribution
    to the current maximal convolution entry) to a “good’’ one.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)  # exact convolution a*a
    best_a = a.copy()
    best_val = float(2.0 * n * b.max() / (target_sum * target_sum))

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04  # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]  # contribution of each pair
        if prod.sum() > 0.0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – choose from the lower‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo : hi + 1] = prod
        thresh = np.percentile(weight_all, 30)  # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer (size adapts with temperature)
        # --------------------------------------------------------------
        # larger moves early, smaller later
        elapsed = time.time() - start
        temp_factor = max(0.0, 1.0 - elapsed / total_time)
        max_delta = min(
            a[donor] * (0.04 + 0.06 * temp_factor),
            target_sum * (0.0008 + 0.0012 * temp_factor),
        )
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # Update convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor : donor + n] += -2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic full recomputation (keeps numerical drift low)
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = float(2.0 * n * b.max() / (target_sum * target_sum))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional jitter to shake out of plateaus
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = float(2.0 * n * b.max() / (target_sum * target_sum))
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget (larger n → richer search)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = math.sqrt(2.0 * n)  # fixing Σa simplifies the objective

    # ------------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
    else:
        # simple decreasing power‑law as fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – smoothed gradient descent (≈45 % of budget)
    # ------------------------------------------------------------------
    grad_deadline = time.time() + 0.45 * (deadline - time.time())
    best_a, best_max = _smooth_gradient_descent(init, target_sum, grad_deadline)
    best_score = float(2.0 * n * best_max / (target_sum * target_sum))
    print(f"[{time.time():.2f}] After smooth grad best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short random restarts of the smoothed gradient descent (~15 % budget)
    # ------------------------------------------------------------------
    while time.time() < deadline - 0.15 * (deadline - start):
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        run_time = min(6.0, remaining * 0.2)
        rng = np.random.default_rng()
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        restart_seq = best_a + jitter
        restart_seq = np.maximum(restart_seq, 0.0)
        restart_seq = _project_to_simplex(restart_seq, target_sum)

        a_tmp, max_tmp = _smooth_gradient_descent(
            restart_seq, target_sum, time.time() + run_time
        )
        tmp_score = float(2.0 * n * max_tmp / (target_sum * target_sum))
        if tmp_score < best_score:
            best_score = tmp_score
            best_a = a_tmp.copy()
            best_max = max_tmp
            print(f"[{time.time():.2f}] Restart improved best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_a, target_sum, deadline)
        if val_ref < best_score:
            best_score = val_ref
            best_a = a_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
