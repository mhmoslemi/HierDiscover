def subgradient_descent_max(a, target_sum, deadline):
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))
        # gradient for each a_i: 2 * a_{k_max-i} if i contributes
        grad = np.zeros_like(a)
        lo = max(0, k_max - (len(a)-1))
        hi = min(k_max, len(a)-1)
        idx = np.arange(lo, hi+1)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]   # because derivative w.r.t a_i of conv[k_max] = a_{partner}
        # Step size: easier: eta = 1/(it+1) * safe_eta? Let's compute safe_eta.
        # safe_eta is max step ensuring a_i - eta*grad_i >= 0 for all i => eta <= a_i/grad_i for grad_i>0.
        positive = grad > 0
        if np.any(positive):
            safe_eta = np.min(a[positive] / grad[positive])
        else:
            safe_eta = np.inf
        # Choose step size as factor * safe_eta, with factor decreasing over iteration.
        factor = min(0.9, 1.0 / np.sqrt(it+1))   # start high then decreasing.
        eta = factor * safe_eta
        # update
        a = a - eta * grad
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # print progress
            print(f"[{time.time():.2f}] Subgrad iteration {it} best {best_val:.6f}")
        it += 1
    return best_a, best_val

def mass_transfer_refine(a, target_sum, deadline):

        partner = k_max - idx

"""
Improved step‑function search.

The algorithm consists of the following stages:

1. **Parametric warm‑start** – a 5‑parameter family  
   a_i = (i+shift)^{-p} · exp( -λ·(i/(n‑1))^{γ} )  
   is explored on a reduced size problem (≈4000 points).  A coarse grid
   and a time‑limited random refinement are performed and the best
   shape is up‑sampled to the final length.

2. **Projected sub‑gradient descent on the true max‑convolution** –  
   using the exact maximal entry (no soft‑max surrogate).  The step size
   is a large fraction of the safe step (≤ a_i / grad_i) and decays
   like 1/√t, with occasional jitter to escape plateaus.

3. **Deterministic mass‑transfer refinement** – moves a modest fraction
   of mass from the index that contributes most to the current maximal
   convolution entry toward a low‑mass index.

4. **Extended simulated‑annealing** – a cheap stochastic local search
   that repeatedly transfers a small amount of mass from a high‑weight
   index (with respect to the current max entry) to a low‑weight index.
   The temperature is linearly cooled and the acceptance rule is the
   standard Metropolis criterion.

All stages respect the global time budget (`budget_s`).  The final
sequence is trimmed of trailing zeros, normalised to ‖a‖₁ = √(2·n) (the
objective is invariant to scaling) and returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power of two length sufficient for linear convolution of size n."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective = max convolution entry (sum constraint is assumed satisfied)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Parametric shape
# ----------------------------------------------------------------------


def _shape_sequence(n: int, p: float, lam: float, gamma: float, shift: float,
                    target_sum: float) -> np.ndarray:
    """Generate a_i = (i+shift)^{-p} * exp(-λ·(i/(n‑1))^{γ}) and normalise."""
    i = np.arange(n, dtype=np.float64) + shift          # shift avoids a singularity at 0
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)   # normalised coordinate in [0,1]
    a = (i + 1) ** (-p) * np.exp(-lam * (x ** gamma))
    # avoid the degenerate case where all entries become zero
    if a.sum() == 0:
        a = np.ones_like(a)
    a *= target_sum / a.sum()
    return a


def _grid_param_search(small_n: int, target_sum: float, deadline: float):
    """
    Coarse grid + random refinement over the 5‑parameter family.
    Returns the best sequence (as ndarray) and its score.
    """
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # coarse grid
    p_vals = np.linspace(0.2, 1.5, 9)               # 0.2 … 1.5
    lam_vals = np.linspace(0.0, 15.0, 9)            # 0 … 15
    gamma_vals = np.array([0.5, 0.9, 1.0, 1.3, 1.6, 2.0])
    shift_vals = np.array([0.0, 2.0, 5.0])

    for p in p_vals:
        for lam in lam_vals:
            for gamma in gamma_vals:
                for shift in shift_vals:
                    if time.time() > deadline:
                        return best_seq, best_score
                    seq = _shape_sequence(small_n, p, lam, gamma, shift, target_sum)
                    val = _score(seq)
                    if val < best_score:
                        best_score = val
                        best_seq = seq.copy()
                        print(f"[{time.time():.2f}] Grid best p={p:.2f} λ={lam:.2f} γ={gamma:.2f} sh={shift:.1f} score={best_score:.6f}")

    # random refinement while time permits
    while time.time() < deadline:
        p = rng.uniform(0.1, 1.8)
        lam = rng.exponential(1.0) * 5.0          # favour moderate λ
        gamma = rng.uniform(0.4, 2.2)
        shift = rng.uniform(0.0, 8.0)
        seq = _shape_sequence(small_n, p, lam, gamma, shift, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} γ={gamma:.3f} sh={shift:.3f} score={best_score:.6f}")

    return best_seq, best_score

# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Sub‑gradient descent on max convolution.
    Uses a large fraction of the safe step (≤ a_i / grad_i) and
    a 1/√t decay.  Jitter is injected periodically.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    last_improve = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        # gradient for the max entry
        n = a.size
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # safe step (largest step that does not make any coordinate negative)
        pos = grad > 0
        if np.any(pos):
            safe_eta = np.min(a[pos] / grad[pos])
        else:
            safe_eta = 0.0

        # schedule: fraction of safe_eta decays like 1/√t, capped at 0.9
        factor = min(0.9, 1.0 / np.sqrt(it + 1))
        eta = factor * safe_eta
        if eta > 0:
            a = a - eta * grad

        # projection onto the ℓ¹‑simplex (keeps sum fixed)
        a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time():.2f}] SubGD it={it:6d} best={best_val:.6f}")

        # occasional jitter to avoid stagnation
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # if we have not improved for a long time, add a bigger jitter
        if it - last_improve > 4000:
            jitter = rng.normal(scale=target_sum * 0.003, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            last_improve = it

        it += 1

    return best_a, best_val

def _mass_transfer_refine(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Move a modest fraction of mass from the most contributing index of the
    current maximal convolution entry to a low‑contribution index.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]

        # pick a receiver among indices with comparatively small a values
        low_thr = np.percentile(a, 12)
        candidates = np.where(a <= low_thr)[0]
        if candidates.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = rng.choice(candidates)

        if donor == receiver:
            # fall back to a random different pair
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.07
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] MT improve {best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renorm for numerical drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simple SA that moves a small amount of mass from a high‑weight index
    (with respect to the current maximal convolution entry) to a low‑weight
    index.  Temperature is linearly cooled.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.04 * best_val  # initial temperature (few percent of the score)

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # donor proportional to its contribution to the maximal entry
        wsum = weight.sum()
        if wsum > 0:
            donor = rng.choice(idx, p=weight / wsum)
        else:
            donor = rng.integers(n)

        # receiver chosen from low‑contribution tail
        low_thr = np.percentile(weight, 6)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())
        new_val = _score(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA new best {best_val:.6f}")

        # occasional random jitter to increase diversity
        if rng.random() < 0.008:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val

    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.30
        a_sub, val_sub = _subgradient_descent(init, target_sum, sub_deadline)
        best_seq = a_sub
        best_score = val_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        a_mt, val_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – simulated‑annealing refinement (last chunk of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_deadline)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
        print(f"[{time.time():.2f}] After SA best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # trim trailing zeros (evaluation treats them as absent)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation for logging (will be shown in the runner output)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a future call
    globals()["height_sequence_1"] = final_seq
    return final_seq
