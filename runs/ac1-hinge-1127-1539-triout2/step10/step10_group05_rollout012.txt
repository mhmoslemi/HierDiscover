"""
Search for a non‑negative step function minimizing

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm works in several stages, each respecting the wall‑clock
budget (default 1000 s):

1. **Extended parametric sweep** – a cheap exhaustive scan of several
   simple families (edge‑exponential, edge‑exponential with baseline,
   cosine‑U‑shape, power‑law, mixed power‑law × exponential, two
   Gaussian peaks).  The exact max‑convolution is obtained via FFT,
   keeping the best candidate.

2. **Projected Adam restarts** – several restarts of Adam on the
   smooth log‑sum‑exp surrogate of the max‑convolution.  The iterates
   are projected onto the ℓ¹‑simplex Σa = √(2 n) after each step,
   guaranteeing a valid sequence.

3. **Deterministic hill‑climb** – a greedy O(n) update that moves mass
   from the index contributing most to the current worst convolution
   entry to a more favourable destination, with occasional jitter.

4. **Simulated‑annealing refinement** – random mass‑transfer moves with
   a temperature schedule, also using O(n) convolution updates.

5. **Upsampling polish** – if time permits, the best vector is linearly
   interpolated to twice its length, refined again by Adam + hill‑climb,
   and finally returned (longer sequences often achieve a lower bound).

All phases print progress information and the final `evaluate_sequence`
value.  The returned list is truncated to drop trailing near‑zeros.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p  (size 3n‑2, we need the middle n entries)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random families for initialisation
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Shape that is large near the two edges and decays towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # find a destination j that is not i and has relatively small counterpart
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]

        # sort candidates by counterpart value
        order = np.argsort(counterpart[i_low:i_high + 1])
        j = None
        for cand in np.arange(i_low, i_high + 1)[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # tiny random jitter to escape possible plateaus
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing (O(n) per move). Returns the best vector
    and its maximal convolution entry seen before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9999

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.5)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5

        if rng.random() < 0.001:            # occasional re‑heat
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Additional shape families for the sweep
# ----------------------------------------------------------------------


def _shape_edge_exp(n: int, lam: float, p: float = 1.0, baseline: float = 0.0) -> np.ndarray:
    """Edge‑exponential: a_i = baseline + (1‑baseline)*exp(-lam·d^p)."""
    idx = np.arange(n, dtype=float)
    d = np.minimum(idx, n - 1 - idx)
    a = np.exp(-lam * (d ** p))
    if baseline > 0.0:
        a = baseline + (1.0 - baseline) * a
    return a


def _shape_cos_abs_pow(n: int, p: float) -> np.ndarray:
    """U‑shape via |cos|: a_i = |cos(π·i/(n‑1))|^p."""
    idx = np.arange(n, dtype=float)
    a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
    return a


def _shape_power_law(n: int, p: float) -> np.ndarray:
    """Pure power‑law a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    return a


def _shape_pow_exp(n: int, pow_exp: float, lam: float) -> np.ndarray:
    """Mixed power‑law × exponential."""
    idx = np.arange(n, dtype=float)
    d = np.minimum(idx, n - 1 - idx)
    a = (idx + 1.0) ** (-pow_exp) * np.exp(-lam * d)
    return a


def _shape_two_gaussians(n: int, sigma: float) -> np.ndarray:
    """Sum of two edge‑centered Gaussians."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-0.5 * (idx / sigma) ** 2) + np.exp(-0.5 * ((n - 1 - idx) / sigma) ** 2)
    return a


# ----------------------------------------------------------------------
# Extended parametric sweep
# ----------------------------------------------------------------------


def _parameter_sweep_extended(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Exhaustive search over a broader family of 1‑parameter (or 2‑parameter)
    shapes.  Returns the best vector and its exact max‑convolution value.
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    # 1) Edge‑exponential (no baseline)
    for lam in np.linspace(0.001, 0.02, 30):
        for p in np.linspace(0.5, 2.0, 9):
            a = _shape_edge_exp(n, lam, p)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # 2) Edge‑exponential with baseline
    for baseline in np.linspace(0.05, 0.3, 5):
        for lam in np.linspace(0.001, 0.02, 20):
            for p in (0.5, 1.0, 1.5, 2.0):
                a = _shape_edge_exp(n, lam, p, baseline)
                a = _project_to_simplex(a, target)
                s = _max_convolution_fft(a)
                if s < best_score:
                    best_score, best_a = s, a.copy()

    # 3) Cosine U‑shape
    for p in np.linspace(0.5, 5.0, 25):
        a = _shape_cos_abs_pow(n, p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # 4) Pure power‑law
    for p in np.linspace(0.1, 2.0, 40):
        a = _shape_power_law(n, p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # 5) Mixed power‑law × exponential
    for pow_exp in np.linspace(0.2, 1.5, 14):
        for lam in np.linspace(0.001, 0.02, 15):
            a = _shape_pow_exp(n, pow_exp, lam)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # 6) Two edge Gaussians
    for sigma_factor in np.linspace(0.02, 0.2, 10):
        sigma = sigma_factor * n
        a = _shape_two_gaussians(n, sigma)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a wall‑clock budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 0) Seed from a previous run if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            # make it the right length (2000) for the first stage
            base_n = 2000
            if prev.size != base_n:
                prev = _upsample_sequence(prev, base_n)
            target0 = math.sqrt(2.0 * base_n)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from global → score={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # --------------------------------------------------------------
    # 1) Fixed‑length search (n = 2000)
    # --------------------------------------------------------------
    n = 2000
    target = math.sqrt(2.0 * n)

    # 1a) Extended parameter sweep
    if time.time() < deadline - 2.0:
        a_sweep, score_sweep = _parameter_sweep_extended(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep improvement → score={best_score:.6f}")

    # 1b) Projected Adam restarts
    remaining = deadline - time.time()
    if remaining > 6.0:
        total_adam = remaining * 0.8
        n_restarts = max(1, int(total_adam // 5.0))      # ~5 s per restart
        per_restart = total_adam / n_restarts
        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            if r == 0 and best_seq is not None:
                a0 = best_seq.copy()
            else:
                # diversify initialisation
                choice = rng.integers(0, 5)
                if choice == 0:
                    a0 = _random_initial_sequence(n, rng)
                elif choice == 1:
                    a0 = _edge_high_initial_sequence(n, rng)
                elif choice == 2:
                    lam = rng.uniform(0.001, 0.02)
                    p = rng.uniform(0.5, 2.0)
                    a0 = _shape_edge_exp(n, lam, p)
                elif choice == 3:
                    p = rng.uniform(0.5, 5.0)
                    a0 = _shape_cos_abs_pow(n, p)
                else:
                    pow_exp = rng.uniform(0.2, 1.5)
                    lam = rng.uniform(0.001, 0.02)
                    a0 = _shape_pow_exp(n, pow_exp, lam)
                a0 = _project_to_simplex(a0, target)

            a_opt = _adam_optimize(a0, target, time.time() + per_restart * 0.9)
            s = _max_convolution_fft(a_opt)
            if s < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = s
                print(f"[{time.time():.1f}] Adam restart {r+1}/{n_restarts} → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Deterministic hill‑climb (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur,
                                               deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Hill‑climb → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Upsampling polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        larger_n = n * 2
        if larger_n <= 5000:               # keep the problem size manageable
            a_ups = _upsample_sequence(best_seq, larger_n)
            target_big = math.sqrt(2.0 * larger_n)
            # short Adam refinement on the larger problem
            a_ref = _adam_optimize(a_ups, target_big,
                                   deadline - 0.8)
            # final hill‑climb on the larger grid
            b_ref = _full_convolution(a_ref)
            a_final, s_final = _hill_climb_deterministic(a_ref, b_ref,
                                                         deadline - 0.3)
            if s_final < best_score - 1e-12:
                best_seq = a_final
                best_score = s_final
                print(f"[{time.time():.1f}] Upsampled polish → score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Clean‑up: drop trailing near‑zeros
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 6) Report the official evaluation
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={len(best_seq)})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
