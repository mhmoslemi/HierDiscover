# ----------------------------------------------------------------------
# FFT‑based convolution utilities

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(a*a)/(Σa)² (with Σa=√(2n) this equals max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# Simplex projection (Euclidean)

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _annealed_gradient_descent_v2_ext(a0: np.ndarray,

def _deterministic_mass_transfer(a0: np.ndarray,

                                end_temp: float = 1e-4,

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))

            continue

        a[receiver] += delta

        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")

            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Uses the sub‑gradient 2*a[k_max-i] (zero outside the support) and
    Euclidean projection onto the simplex after each update.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2 * a[kmax - i] if index in range, else 0
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val
