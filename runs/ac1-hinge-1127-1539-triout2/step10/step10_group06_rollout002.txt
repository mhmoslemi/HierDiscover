def _simulated_annealing_refine(a0, target_sum, deadline):

    n = a.size
    b = _conv_fft(a)  # convolution
    best_a = a.copy()
    best_val = _score(a)
    best_b = b.copy()
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # temperature schedule parameters
    T0 = best_val * 0.05   # maybe 0.05 times best
    # fraction_of_time for exploring
    while time.time() < deadline:
        elapsed = time.time() - start
        T = max(0.0, T0 * (1 - elapsed / total))
        # choose move type: 80% targeted, 20% random
        if rng.random() < 0.8:
            # targeted move
            k_max = int(np.argmax(b))
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            prod = a[idx] * a[k_max - idx]
            # donor weighted by prod
            if prod.sum() > 0:
                donor = rng.choice(idx, p=prod / prod.sum())
            else:
                donor = rng.choice(idx)
            # receiver chosen from low contribution indices
            weight_all = np.zeros(n, dtype=a.dtype)
            weight_all[lo:hi+1] = prod
            thresh = np.percentile(weight_all[weight_all>0], 30) if np.any(weight_all>0) else 0
            low_idx = np.where(weight_all <= thresh)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)
        else:
            # random move
            donor = rng.integers(n)
            while a[donor] < 1e-12:
                donor = rng.integers(n)
            receiver = rng.integers(n)
        # delta factor:
        max_delta = min(a[donor] * (0.3 if elapsed < total/2 else 0.1), target_sum * (0.001 if elapsed < total/2 else 0.0003))
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # compute new convolution via incremental update
        # (same as in local_refine)
        b_new = b.copy()
        # linear part
        b_new[donor:donor+n] += -2.0 * delta * a
        b_new[receiver:receiver+n] += 2.0 * delta * a
        # quadratic
        b_new[2*donor] += delta * delta
        b_new[2*receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))
        # acceptance
        if new_val < best_val or (T > 0 and rng.random() < np.exp(-(new_val - best_val) / T)):
            # accept move: update a, b, best if improved
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.2f}] SA improve score={best_val:.6f}")
        # periodic re-normalisation to avoid drift
        if (int(elapsed * 100) % 2000) == 0:
            # ensure sum stays exact
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.2f}] Re‑norm improve score={best_val:.6f}")
        # occasional jitter to escape local minima
        if rng.random() < 0.001:

                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.2f}] random jitter improve score={best_val:.6f}")
    return best_a, best_val

"""
Propose a non‑negative step sequence with a low benchmark score.

Algorithm Overview
------------------
1. **Warm‑start** – Use the best known construction (`height_sequence_1`) if
   available; otherwise initialise a decreasing power‑law and renormalise
   Σa = √(2 n).

2. **Coarse optimisation** – A short exponential (mirror) descent on a
   soft‑max surrogate of the maximal convolution entry (Phase A).

3. **Subgradient descent** – Direct multiplicative sub‑gradient descent on the
   true max‑convolution (Phase B).

4. **Projected gradient descent with restarts** – Momentum‑based GD on the
   soft‑max, many random jittered restarts (Phase C).

5. **Aggressive simulated‑annealing refinement** – Starting from the best
   candidate, repeatedly move a random amount of mass from an index that
   contributes heavily to the current maximal convolution entry towards an
   index with low contribution.  The move size is large early on and shrinks
   with a linear cooling schedule.  The convolution is updated *exactly* in
   O(n) time using the same delta‑update as the original local‑refine code,
   allowing many thousands of iterations even for n≈12000.

   This phase occupies the remaining time budget and is the main source of
   improvement over the previous implementation.

6. Truncate trailing near‑zero entries, evaluate with the provided
   `evaluate_sequence`, store the result for warm‑starts and return the list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based for full recomputation, O(n) delta‑update)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phases A & C)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution).  Grad has same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _gradient_softmax_cached(a: np.ndarray, beta: float):
    """Thin wrapper that forwards to _gradient_softmax (kept for naming)."""
    return _gradient_softmax(a, beta)


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax_cached(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _annealing_refine(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressive simulated‑annealing on the true max‑convolution.
    Moves a small amount of mass from indices that contribute heavily to the
    current maximal convolution entry to low‑contribution indices.
    The convolution is updated exactly in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)
    best_b = b.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start

    # Temperature proportional to the current score; will linearly decay.
    T0 = best_val * 0.12

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = max(0.0, T0 * (1.0 - frac))

        # ------------------------------------------------------------------
        # pick donor & receiver
        # ------------------------------------------------------------------
        # 80 % of moves are “targeted’’ (focus on the current maximal entry)
        if rng.random() < 0.8:
            k_max = int(np.argmax(b))
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)

            prod = a[idx] * a[k_max - idx]          # contribution of each pair
            if prod.sum() > 0:
                donor = rng.choice(idx, p=prod / prod.sum())
            else:
                donor = rng.choice(idx)

            # choose a receiver among indices that contribute little to the max entry
            weight_all = np.zeros(n, dtype=a.dtype)
            weight_all[lo:hi + 1] = prod
            # use a low percentile as “low contribution’’ set
            if np.any(weight_all):
                thresh = np.percentile(weight_all[weight_all > 0], 30) if np.any(weight_all > 0) else 0.0
                low_idx = np.where(weight_all <= thresh)[0]
                if low_idx.size:
                    receiver = rng.choice(low_idx)
                else:
                    receiver = rng.integers(n)
            else:
                receiver = rng.integers(n)
        else:
            # completely random move (helps escape flat regions)
            donor = rng.integers(n)
            while a[donor] < 1e-14:
                donor = rng.integers(n)
            receiver = rng.integers(n)

        # ------------------------------------------------------------------
        # size of the transfer
        # ------------------------------------------------------------------
        # early in the run we allow relatively large moves, later we shrink them
        max_factor = 0.30 * (1.0 - frac) + 0.05 * frac          # up to 30 % → 5 %
        abs_factor = 0.0010 * (1.0 - frac) + 0.0002 * frac      # up to 0.001 S → 0.0002 S
        max_delta = min(a[donor] * max_factor, target_sum * abs_factor)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # ------------------------------------------------------------------
        # exact O(n) convolution update
        # ------------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (the slice updates)
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # ------------------------------------------------------------------
        # acceptance rule
        # ------------------------------------------------------------------
        accept = False
        if new_val < best_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(new_val - best_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.2f}] SA improve score={best_val:.6f}")

        # ------------------------------------------------------------------
        # periodic full recomputation to avoid drift (every ~8 s)
        # ------------------------------------------------------------------
        if time.time() - (start + elapsed) > 8.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.2f}] Re‑norm improve score={best_val:.6f}")

        # ------------------------------------------------------------------
        # tiny random jitter (helps escape plateaus)
        # ------------------------------------------------------------------
        if rng.random() < 0.001:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.02, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    best_b = b.copy()
                    print(f"[{time.time():.2f}] Jitter improve score={best_val:.6f}")

        it += 1

    return best_a, best_val

    """

    # Choose problem size (larger n gives finer granularity but costs time)

    elif budget_s >= 500:

    else:

    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------

    if time.time() < deadline:
        sub_deadline = time.time() + 0.12 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts (≈30 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – aggressive simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _annealing_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After SA refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
