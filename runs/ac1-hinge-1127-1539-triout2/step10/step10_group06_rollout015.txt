def _greedy_local_refine(a0, target_sum, deadline):
    a = a0.copy()
    n = len(a)
    b = _conv_fft(a)
    best_a, best_val = a.copy(), _score(a)

    while time.time() < deadline:
        # compute max entry and its index
        k_max = int(np.argmax(b))
        # compute contributions to that entry: contributions = a[i] * a[k_max-i] for i in range.
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        contrib = a[idx] * a[k_max - idx]
        # donors: top few indices with largest contribution
        top_k = 5

"""
Propose a non‑negative step function with a low benchmark score.

Algorithm Overview
------------------
1. **Problem size** – pick `n` based on the time budget (12000 for a long run).
2. **Warm‑start** – use the global `height_sequence_1` if present,
   otherwise fall back to a power‑law.
3. **Parameter sweep** – test a few simple families (power‑law,
   exponential) and keep the best candidate.
4. **Exponential (mirror) descent** on a soft‑max surrogate of the max‑convolution.
5. **Sub‑gradient descent** on the true max‑convolution (additive updates
   with simplex projection).
6. **Projected GD with random restarts** (same as the previous runner).
7. **Greedy local refinement** – repeatedly move a tiny amount of mass
   from the index that contributes most to the current maximal entry
   to a low‑mass index; every move is evaluated exactly in O(n).
8. **Optional smoothing** – a few low‑pass filters are tried; any
   improvement is kept.
9. Trim trailing zeros, evaluate with the provided `evaluate_sequence`,
   store the result for warm‑starts and return the sequence.

The routine respects a 1000 s wall‑clock limit and prints intermediate
scores for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2 n) this equals max(a*a) and is the quantity printed
    throughout the search.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (gradient needed for Phase‑A and Phase‑C)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – subgradient descent on the true max‑convolution (additive)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Projected additive subgradient descent on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # compute full convolution once
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient: 2·a_{k_max-i}
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # Polyak‑like decreasing schedule (sqrt‑law works well for convex)
        elapsed = time.time() - start
        total = deadline - start
        eta0 = 2.0
        eta = eta0 / np.sqrt(1.0 + elapsed / total * 100.0)

        # additive update + projection
        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Phase C – projected GD with random restarts (same as before)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# Phase D – greedy exact local refinement
# ----------------------------------------------------------------------


def _greedy_local_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing: move a tiny amount of mass from the index that
    contributes most to the current maximal convolution entry to a low‑mass
    index. Each move is evaluated exactly in O(n) using the incremental
    update formulas from the original local‑refine routine.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution of each index to the current peak
        contrib = a[idx] * a[k_max - idx]

        # donors = top few contributors
        top_k = min(8, idx.size)
        donor_candidates = idx[np.argsort(-contrib)[:top_k]]

        # receivers = indices with smallest a (outside the peak window)
        low_k = min(8, n)
        receiver_candidates = np.argsort(a)[:low_k]

        improved = False
        best_local_val = best_val
        best_move = None

        # try all donor/receiver pairs (small combinatorial search)
        for donor in donor_candidates:
            for recv in receiver_candidates:
                if donor == recv:
                    continue
                # step size – a fraction of donor mass, but not too large
                eps = min(a[donor] * 0.35, target_sum * 0.0010)
                if eps <= 1e-12:
                    continue

                # exact O(n) update of the convolution
                b_new = b.copy()
                b_new[donor:donor + n] -= 2.0 * eps * a
                b_new[recv:recv + n] += 2.0 * eps * a
                b_new[2 * donor] += eps * eps
                b_new[2 * recv] += eps * eps
                if donor != recv:
                    b_new[donor + recv] -= 2.0 * eps * eps

                new_val = float(2.0 * n * b_new.max() /
                                (target_sum * target_sum))

                if new_val < best_local_val - 1e-9:
                    best_local_val = new_val
                    best_move = (donor, recv, eps, b_new.copy())

        if best_move:
            donor, recv, eps, b = best_move
            a[donor] -= eps
            a[recv] += eps
            best_val = best_local_val
            best_a = a.copy()
            improved = True
            print(f"[{time.time():.2f}] Greedy move {donor}->{recv} eps={eps:.6f} score={best_val:.6f}")
            continue  # restart loop with the updated state

        # No improving donor/receiver found – fall back to a tiny random jitter
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] <= 0.0:
            continue
        eps = min(a[i] * 0.15, target_sum * 0.0007)
        if eps <= 0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * eps * a
        b_new[j:j + n] += 2.0 * eps * a
        b_new[2 * i] += eps * eps
        b_new[2 * j] += eps * eps
        if i != j:
            b_new[i + j] -= 2.0 * eps * eps

        new_val = float(2.0 * n * b_new.max() /
                        (target_sum * target_sum))

        if new_val < best_val:
            a[i] -= eps
            a[j] += eps
            b = b_new
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random jitter improved score={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if time.time() - start_time > 15.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            start_time = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recomputation improved score={best_val:.6f}")

    return best_a, best_val

def _search_parametric_families(n: int, target_sum: float):
    """Return the best sequence found among a few simple parametric families."""
    best_seq = None
    best_score = np.inf

    # ---------- power‑law ----------
    p_grid = np.linspace(0.35, 1.0, 66)   # 0.010 steps
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in p_grid:
        a = idx ** (-p)
        a *= target_sum / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    print(f"[{time.time():.2f}] Power‑law sweep best score={best_score:.6f}")

    # ---------- exponential ----------
    r_grid = np.linspace(0.900, 0.9999, 70)
    for r in r_grid:
        a = r ** np.arange(n, dtype=np.float64)
        a *= target_sum / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    print(f"[{time.time():.2f}] Exponential sweep best score={best_score:.6f}")

    return best_seq, best_score

        n = 15000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse previous best if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
    else:
        # fallback power‑law
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase 0 – quick sweep over simple parametric families
    # --------------------------------------------------------------
    sweep_seq, sweep_score = _search_parametric_families(n, target_sum)

    # pick the best among warm‑start and sweep
    best_seq = init
    best_score = _score(best_seq)
    if sweep_score < best_score:
        best_seq, best_score = sweep_seq, sweep_score
    print(f"[{time.time():.2f}] After parametric sweep best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Allocate time for the remaining phases
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # Reasonable fractions (they sum to < 1)
    phase_a_frac = 0.10
    phase_b_frac = 0.15
    phase_c_frac = 0.20
    # whatever is left goes to the greedy refinement
    phase_a_dead = time.time() + phase_a_frac * remaining
    phase_b_dead = time.time() + phase_b_frac * remaining
    phase_c_dead = time.time() + phase_c_frac * remaining
    phase_d_dead = deadline

    # --------------------------------------------------------------
    # Phase A – exponential mirror descent (soft‑max)
    # --------------------------------------------------------------
    a_exp, val_exp = _exponential_descent(best_seq, target_sum, phase_a_dead)
    if val_exp < best_score:
        best_seq, best_score = a_exp, val_exp
    print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent (additive)
    # --------------------------------------------------------------
    if time.time() < phase_b_dead:
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, phase_b_dead)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts
    # --------------------------------------------------------------
    if time.time() < phase_c_dead:
        a_gd, val_gd = _gradient_descent_with_restarts(
            best_seq, target_sum, phase_c_dead
        )
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – greedy exact local refinement (major part of remaining time)
    # --------------------------------------------------------------
    if time.time() < phase_d_dead:
        a_ref, val_ref = _greedy_local_refine(best_seq, target_sum, phase_d_dead)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After greedy refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional light smoothing passes (try a few low‑pass filters)
    # --------------------------------------------------------------
    kernels = [np.ones(k) / k for k in (3, 5, 7)]
    for ker in kernels:
        smooth = np.convolve(best_seq, ker, mode="same")
        smooth = np.maximum(smooth, 0.0)
        smooth = smooth * (target_sum / smooth.sum())
        sc = _score(smooth)
        if sc < best_score:
            best_seq = smooth
            best_score = sc
            print(f"[{time.time():.2f}] Smoothing (k={len(ker)}) improved score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up: drop trailing zeros, final evaluation and storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
