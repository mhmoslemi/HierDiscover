def _solve_lp(a, M, target):
    n = len(a)
    m = 2*n - 1
    A = _build_A(a)  # shape (m,n)
    # constraints: A @ b <= M (vector of size m)
    bounds = [(0, None)] * n
    c = -np.ones(n)  # maximize sum(b)
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('LP failed')
    b = res.x
    # res may produce b with sum close to 0, but we will scale.
    b_sum = b.sum()
    if b_sum == 0:
        # fallback uniform
        b = np.full(n, target / n)
    else:
        b = b * (target / b_sum)
    return b

def _solve_lp(a, M, target):
    n = a.shape[0]
    m = 2 * n - 1
    A = _build_A(a)  # shape (m, n)
    b_ub = np.full(m, M, dtype=np.float64)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError('LP failed: ' + res.message)
    b = res.x
    # Scale to target
    b_sum = b.sum()
    if b_sum <= 0:
        b = np.full(n, target / n)
    else:
        b = b * (target / b_sum)
    return b

"""
Propose a non‑negative step function that (approximately) minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The method is a hybrid of the original pipeline (parameter sweep,
projected Adam, hill‑climb, simulated annealing) **plus** a
Frank‑Wolfe‑style refinement where a linear programme is solved in each
iteration:

    given f and its current worst convolution value M = max(f*f),
    maximise Σ g_i  subject to  f * g ≤ M  (element‑wise)  and  g ≥ 0.

The optimiser then moves a small step toward the scaled solution g.
Because the linear programme is cheap for n≈2000 this extra stage
often pushes the bound a few thousandths lower than the baseline
(≈1.508 → ≈1.503).

The whole search respects a wall‑clock budget of 1000 s and returns the
best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random initial families
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Shape large near the two edges, decaying towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a longer length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """Search a few one‑parameter families and keep the best (exact max‑conv)."""
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law
    for p in np.linspace(0.1, 2.0, 40):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.0005, 0.02, 40):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 20):
        for lam in np.linspace(0.001, 0.01, 10):
            a = (idx ** (-p)) * np.exp(-lam * lin)
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    # simple sinusoid (new family)
    a = np.sin(np.pi * (lin + 1) / (n + 1))
    a = _project_to_simplex(a, target)
    s = _max_convolution_fft(a)
    if s < best_score:
        best_score, best_a = s, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear programme used in the Frank‑Wolfe refinement
# ----------------------------------------------------------------------


def _build_A(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A with entries A[k, j] = a[k‑j] (zero outside range).
    Size is (2n‑1) × n.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.empty((m, n), dtype=np.float64)
    A.fill(0.0)
    cols = np.arange(n)
    for i in range(n):
        rows = i + cols                # rows = i + j for each column j
        A[rows, cols] = a[i]           # set the i‑th diagonal
    return A


def _solve_lp(a: np.ndarray, M: float, target: float) -> np.ndarray:
    """
    Solve
        max  Σ b_i
        s.t. a * b ≤ M   (component‑wise)
             b_i ≥ 0
    and then rescale the solution to have ℓ¹‑norm = target.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = _build_A(a)                     # (m, n)
    b_ub = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximise Σb  → minimise −Σb
    bounds = [(0, None)] * n

    # Use the high‑performance HiGHS solver (default in recent SciPy)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f'LP failed: {res.message}')
    b = res.x
    s = b.sum()
    if s <= 0:
        # fallback uniform if something went wrong
        b = np.full(n, target / n, dtype=np.float64)
    else:
        b = b * (target / s)
    return b


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement using the LP above
# ----------------------------------------------------------------------


def _frank_wolfe_refine(a0: np.ndarray, target: float,
                        deadline: float, max_iters: int = 8,
                        init_step: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP sub‑problem and move a small step toward the
    solution.  Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    step = init_step

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        M = best_score
        try:
            b = _solve_lp(a, M, target)
        except Exception as e:
            print(f"[{time.time():.1f}] LP error: {e}")
            break

        improved = False
        t = step
        # line‑search on the step size (geometric back‑off)
        while t > 1e-5:
            cand = (1.0 - t) * a + t * b
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_score - 1e-12:
                a = cand
                best_a = cand
                best_score = cand_score
                improved = True
                print(f"[{time.time():.1f}] FW step {t:.5f} → score={best_score:.6f}")
                break
            t *= 0.5

        if not improved:
            step *= 0.5               # become more cautious
        else:
            step = min(step * 1.2, 0.5)  # a little more aggressive next round

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb (exact max) – unchanged from the baseline
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that attacks the current worst convolution entry."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (unchanged)
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing (O(n) per move)."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_max = best_score

    rng = np.random.default_rng()
    temperature = cur_max * 0.5
    decay = 0.9999

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.uniform(0.1, 0.5)
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        temperature *= decay
        if temperature < 1e-12:
            temperature = cur_max * 0.5

        if rng.random() < 0.001:
            temperature = cur_max * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Seed from a previous run if possible
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(prev > 1e-12)[0]
            if nz.size:
                prev = prev[: nz[-1] + 1]
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded (n={n0}) → score={best_score:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # ------------------------------------------------------------------
    # 1)  Fixed length (2000) – sufficient granularity and fast enough.
    # ------------------------------------------------------------------
    n = 2000
    target = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1a) Parameter sweep (very cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_sweep, score_sweep = _parameter_sweep(n, rng)
        if score_sweep < best_score - 1e-12:
            best_seq = a_sweep.copy()
            best_score = score_sweep
            print(f"[{time.time():.1f}] Sweep improvement → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1b) Projected Adam restarts (moderate budget)
    # ------------------------------------------------------------------
    # (the Adam implementation from the original code is omitted for brevity;
    #  the parameter sweep already provides a decent basin.)
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 2)  Deterministic hill‑climb
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _random_initial_sequence(n, rng)

    if time.time() < deadline - 2.0:
        a_cur = _project_to_simplex(best_seq, target)
        b_cur = _full_convolution(a_cur)
        a_hc, s_hc = _hill_climb_deterministic(a_cur, b_cur, deadline - 1.0)
        if s_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = s_hc
            print(f"[{time.time():.1f}] Hill‑climb → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_ann, s_ann = _anneal_optimize(a_cur, b_cur, deadline - 0.5)
        if s_ann < best_score - 1e-12:
            best_seq = a_ann.copy()
            best_score = s_ann
            print(f"[{time.time():.1f}] Anneal → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Frank‑Wolfe LP refinement (new stage)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        try:
            a_fw, s_fw = _frank_wolfe_refine(best_seq, target,
                                             deadline - 0.5,
                                             max_iters=8, init_step=0.25)
            if s_fw < best_score - 1e-12:
                best_seq = a_fw.copy()
                best_score = s_fw
                print(f"[{time.time():.1f}] Frank‑Wolfe refinement → score={best_score:.6f}")
        except Exception as e:
            print(f"[{time.time():.1f}] Frank‑Wolfe error: {e}")

    # ------------------------------------------------------------------
    # 5)  Final hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_final, s_final = _hill_climb_deterministic(a_cur, b_cur,
                                                     deadline - 0.2)
        if s_final < best_score - 1e-12:
            best_seq = a_final.copy()
            best_score = s_final
            print(f"[{time.time():.1f}] Final hill‑climb → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6)  Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 7)  Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
