"""
Propose a non‑negative step function that (approximately) minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm proceeds in three stages:

1. **Cheap parametric sweep** – try a few families (power‑law,
   exponential, edge‑heavy, mixed) and keep the best candidate (exact
   max‑convolution via FFT).

2. **Exact‑max sub‑gradient descent** – repeatedly move the current vector
   in a sub‑gradient direction of the max‑convolution (the gradient of a
   single active shift is just a shifted copy of the vector).  After each
   step we project onto the ℓ¹‑simplex Σa = √(2 n).  The step size is
   adapted with a simple back‑tracking line search.  This stage works
   directly with the true objective (no soft‑max surrogate) and is able
   to push the maximal convolution a little lower than the previous Adam
   approach.

3. **Deterministic hill‑climb** – a fast exact‑max greedy refinement
   (the same routine used in the reference code) to polish the answer.

All stages respect the wall‑clock budget (default 1000 s).  The final
sequence is trimmed of trailing near‑zeros before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x : x_i ≥ 0, Σ_i x_i = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Small parametric sweep (cheap, but often finds a good basin)
# ----------------------------------------------------------------------


def _parameter_sweep(n: int, rng) -> tuple[np.ndarray, float]:
    """
    Scan a few one‑parameter families (power‑law, exponential,
    edge‑heavy, mixed) and return the best vector found (according to the
    exact max‑convolution).
    """
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    idx = np.arange(1, n + 1, dtype=float)
    lin = np.arange(n, dtype=float)

    # power‑law exponents
    for p in np.linspace(0.1, 2.0, 40):
        a = idx ** (-p)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # pure exponentials
    for lam in np.linspace(0.0005, 0.02, 40):
        a = np.exp(-lam * lin)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # edge‑heavy (large at the ends, decays towards centre)
    for lam in np.linspace(0.001, 0.02, 30):
        d = np.minimum(lin, n - 1 - lin)
        a = np.exp(-lam * d)
        a = _project_to_simplex(a, target)
        s = _max_convolution_fft(a)
        if s < best_score:
            best_score, best_a = s, a.copy()

    # mixed power‑law × exponential
    for p in np.linspace(0.3, 1.5, 20):
        for lam in np.linspace(0.001, 0.01, 10):
            a = (idx ** (-p)) * np.exp(-lam * (lin - 0))
            a = _project_to_simplex(a, target)
            s = _max_convolution_fft(a)
            if s < best_score:
                best_score, best_a = s, a.copy()

    return best_a, best_score

# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination j (different from i) with small counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # tiny random jitter to escape a plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient descent
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray, target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true objective
        M(a) = max_k (a*a)_k,
    with projection onto the ℓ¹‑simplex Σa = target_sum.
    Returns the best vector encountered and its maximal convolution value.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_M = _max_convolution_fft(a)

    # initial step size – roughly a few percent of the total mass
    step = max(target_sum * 0.02, 1e-3)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # current convolution and maximal entry
        b = _full_convolution(a)
        M = b.max()

        if M < best_M:
            best_M = M
            best_a = a.copy()

        # --- compute a sub‑gradient of the max ---
        # all indices k with (almost) maximal value are active
        eps_active = max(1e-12, M * 1e-14)
        active = np.where(b >= M - eps_active)[0]

        if active.size == 0:
            break  # should not happen

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # contribution a[lo:hi+1] reflected around k
            rev = a[lo:hi + 1][::-1]
            g[lo:hi + 1] += rev

        # averaging over the active set (the sub‑gradient is any convex combo)
        g *= 2.0 / active.size

        # --- back‑tracking line search ---
        accepted = False
        trial_step = step
        while not accepted and trial_step > target_sum * 1e-12:
            a_new = a - trial_step * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_to_simplex(a_new, target_sum)

            M_new = _max_convolution_fft(a_new)

            if M_new <= M:          # improvement (or at least no worsening)
                accepted = True
                a = a_new
                step = trial_step * 1.02   # modestly increase step
                break
            else:
                trial_step *= 0.5

        if not accepted:
            # step too small – inject a tiny random jitter to escape stagnation
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            step = max(target_sum * 0.02, step * 0.5)

    return best_a, best_M

# Upsampling helper (optional – to try a longer sequence)
# ----------------------------------------------------------------------


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new
