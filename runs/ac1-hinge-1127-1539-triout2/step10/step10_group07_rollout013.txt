def _enhanced_local_refine(a0, target_sum, deadline):
    a = a0.copy()
    n = len(a)
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)   # equals max b
    # (score = max_b)
    # temperature schedule
    start = time.time()
    total_time = deadline - start
    # initial temperature maybe 0.1 * best_val (function see earlier)
    T0 = best_val * 0.2
    iter_count = 0
    rng = np.random.default_rng()
    last_full = time.time()
    while time.time() < deadline:
        iter_count += 1
        # find max index
        k_max = b.argmax()
        # Determine donor:
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idx_range = np.arange(lo, hi+1)
        # contributions of each idx: a[idx] * a[k_max - idx]
        contrib = a[idx_range] * a[k_max - idx_range]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            donor = rng.choice(idx_range, p=contrib/total_contrib)
        else:
            donor = rng.integers(lo, hi+1)
        # Receiver choose index with low a (global low)
        # We'll choose from entire array weighted inversely to a
        # Ensure not donor
        # Compute probability inversely proportional to a+eps
        # Resist picking high a location.
        probs = 1.0/(a + 1e-12)
        probs[donor] = 0
        probs = probs / probs.sum()
        receiver = rng.choice(n, p=probs)
        # Compute delta bound
        max_delta = min(a[donor] * 0.3, target_sum * 0.001)  # allow larger
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        # Update b in O(n)
        b_new = b.copy()
        # linear part
        # subtract from donor
        start_idx = donor
        end_idx = donor + n
        b_new[start_idx:end_idx] -= 2.0 * delta * a
        # add to receiver
        start_idx = receiver
        end_idx = receiver + n
        b_new[start_idx:end_idx] += 2.0 * delta * a
        # quadratic terms
        b_new[2*donor] += delta*delta
        b_new[2*receiver] += delta*delta
        b_new[donor+receiver] -= 2.0 * delta*delta
        # evaluate new max
        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))
        # acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0:
            # use Metropolis
            prob = np.exp(-(new_val - best_val)/temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                # print progress occasionally
                if iter_count % 1000 == 0:
                    print(f"[{time.time():.2f}] Refine iter {iter_count} new best {best_val:.6f}")
        else:
            # optionally apply jitter occasionally
            pass
        # occasionally full recompute for numerical drift
        if time.time() - last_full > 10.0:
            # renorm a to exact target_sum (to correct rounding)
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved {best_val:.6f}")
```

def _adam_subgradient(a0, target_sum, deadline):
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = target_sum * 0.2
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    while time.time() < deadline:
        # compute mask of max b
        b = _conv_fft(a)
        max_b = b.max()
        mask = (b >= max_b - (max_b * 1e-12 + 1e-15)).astype(np.float64)
        # compute subgradient g = conv(a, mask_rev)
        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)  # length 3n-2
        L = len(mask) - 1
        g = conv_full[L - np.arange(len(a))]
        # adam update
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr / (np.sqrt(v_hat) + epsilon)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)
        # optional jitter occasionally
        if it % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam subgrad it={it} best={best_val:.6f}")
    return best_a, best_val
```

    if budget_s >= 800:
        n = 16000
    elif budget_s >= 500:
        n = 12000
    else:

    # warm start
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        # if different length, resize
        if init_seq.shape[0] != n:
            init_seq = _resize_sequence(init_seq, n)

        # power law
        init_seq = (np.arange(1, n+1, dtype=np.float64) ** (-0.5))

    init_seq = _project_to_simplex(init_seq, target_sum)

    # Phase 1: quick Adam subgradient
    remaining = deadline - time.time()
    phase1_deadline = time.time() + 0.15 * remaining
    a1, val1 = _adam_subgradient(init_seq, target_sum, phase1_deadline)

    a2, val2 = _enhanced_local_refine(a1, target_sum, deadline)

    try:

        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")

        print(f"[{time.time():.2f}] Evaluation error: {e}")

"""
Propose a step‑function with a low evaluation score.

Algorithm (≈1000 s budget)
--------------------------------
1. Warm‑start – reuse the best known sequence (`height_sequence_1`) if it exists,
   otherwise initialise with a decreasing power‑law.
2. Phase 1 – a short Adam‑style sub‑gradient descent on the true max‑convolution.
   This quickly balances the mass while staying on the simplex Σa = √(2 n).
3. Phase 2 – a long simulated‑annealing / mass‑transfer refinement.
   In each move a tiny amount of mass is shifted from an index that contributes
   heavily to the current maximal convolution entry to an index that contributes
   little.  The step size is adaptive, the temperature schedule is aggressive
   (initial temperature ≈ 0.2·current max) and the move acceptance uses the
   true objective (no surrogate).  Full recomputation of the convolution is
   performed periodically to keep numerical errors in check.
4. Final clean‑up – trim trailing zeros, evaluate the sequence with the
   provided ``evaluate_sequence`` and store it for warm‑starts.

All updates respect non‑negativity and the simplex constraint.  FFT‑based
convolutions give O(n log n) cost for the sub‑gradient phase; the refinement
phase works in O(n) per move.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT – based convolution utilities (O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2 n) the return equals max(a*a) directly.
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – Adam sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent for minimizing max(a*a)
    while keeping a on the simplex Σa = target_sum.
    Returns the best sequence seen and its max‑convolution value.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.astype(np.float64), target_sum)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = target_sum * 0.2  # initial learning‑rate scale

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        # ----- convolution & mask of maximal entries -----
        b = _conv_fft(a)
        max_b = b.max()
        eps_mask = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps_mask).astype(np.float64)         # 0/1 mask

        # ----- sub‑gradient of max(b) w.r.t. a -----
        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)            # length 3n‑2
        L = len(mask) - 1                                         # = 2n‑2
        g = conv_full[L - np.arange(a.shape[0])]

        # ----- Adam update (projected) -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat plateaux
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam sub‑grad it={it:6d} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Enhanced simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _enhanced_local_refine(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style refinement.
    Moves mass from a “bad’’ index (large contribution to the current maximal
    convolution entry) to a “good’’ index.  The move size adapts to the
    imbalance between donor and receiver and the temperature schedule is
    intentionally aggressive (initial temperature ≈ 0.2·current max).
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]

    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)                              # = max(b) when Σa fixed
    start = time.time()
    total_time = deadline - start
    # Aggressive temperature: 20 % of the initial max
    T0 = best_val * 0.20

    last_full = start

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- current maximal convolution index (choose randomly among ties) -----
        max_b = b.max()
        eps_idx = max_b * 1e-12 + 1e-15
        max_positions = np.where(b >= max_b - eps_idx)[0]
        k_max = rng.choice(max_positions)

        # ----- donor: index with largest contribution to this peak -----
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contributions for each pair
        total_prod = prod.sum()
        if total_prod > 0:
            donor = rng.choice(idx, p=prod / total_prod)
        else:
            donor = rng.integers(lo, hi + 1)

        # ----- receiver: index with *small* contribution to the peak -----
        # simply favour indices with small a‑values (inverse weighting)
        inv_weights = 1.0 / (a + 1e-12)
        inv_weights[donor] = 0.0                     # avoid trivial move
        inv_weights = inv_weights / inv_weights.sum()
        receiver = rng.choice(n, p=inv_weights)

        if donor == receiver:
            continue

        # ----- adaptive step size (move a fraction of the donor’s excess) -----
        excess = a[donor] - a[receiver]
        if excess <= 1e-15:
            continue
        # allow up to 20 % of donor’s mass or 0.1 % of total sum, whichever is smaller
        max_delta = min(a[donor] * 0.20, target_sum * 0.001)
        delta = rng.random() * max_delta
        # Do not overshoot the imbalance
        delta = min(delta, excess * 0.5)
        if delta <= 0:
            continue

        # ----- update convolution exactly (O(n)) -----
        b_new = b.copy()
        # linear part: -2·δ·a shifted by donor, +2·δ·a shifted by receiver
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # ----- simulated‑annealing acceptance -----
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                if iteration % 500 == 0:
                    print(f"[{time.time():.2f}] Refine it={iteration:6d} best={best_val:.6f}")

        # ----- periodic full recomputation to avoid drift -----
        if time.time() - last_full > 12.0:
            a = a * (target_sum / a.sum())          # renormalise exactly
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # ----- occasional global jitter (helps escape plateaus) -----
        if (iteration % 2000) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.01, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size – larger n gives more degrees of freedom
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 16000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # Σa is fixed to √(2 n)

    # --------------------------------------------------------------
    # Warm‑start (reuse previous construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init.shape[0] != n:
            init = _resize_sequence(init, n)
    else:
        # simple decreasing power‑law as a fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = _project_to_simplex(init, target_sum)

    # --------------------------------------------------------------
    # Phase 1 – short Adam sub‑gradient descent (≈15 % of remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + 0.15 * remaining
    a_mid, val_mid = _adam_subgradient(init, target_sum, phase1_deadline)
    print(f"[{time.time():.2f}] End of Adam sub‑grad, best score={val_mid:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long mass‑transfer refinement (≈80 % of remaining time)
    # --------------------------------------------------------------
    a_refined, val_refined = _enhanced_local_refine(a_mid, target_sum, deadline)
    print(f"[{time.time():.2f}] End of local refinement, best score={val_refined:.6f}")

    # --------------------------------------------------------------
    # Optional final Adam pass if any tiny time slice remains
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_final, val_final = _adam_subgradient(a_refined, target_sum, deadline)
        print(f"[{time.time():.2f}] Final Adam pass, best score={val_final:.6f}")
    else:
        a_final, val_final = a_refined, val_refined

    # --------------------------------------------------------------
    # Clean‑up: trim trailing zeros and evaluate with the provided function
    # --------------------------------------------------------------
    arr = a_final
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    try:
        final_score = evaluate_sequence(final_seq)            # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
