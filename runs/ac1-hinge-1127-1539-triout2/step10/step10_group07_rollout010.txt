"""
Improved search for a low‑evaluation step function.

The algorithm mixes three ideas:
1. **Rich warm‑start** – tries several simple parametric families
   (power‑law, exponential, random) and keeps the best one (or the
   previously saved `height_sequence_1`).
2. **Smoothed sub‑gradient descent** – minimises the true
   `max(a*a)` using a soft‑max surrogate.
   The gradient of the surrogate is a convolution with a soft‑max
   weight vector, which is cheap to compute via FFT.
3. **Guided mass‑transfer refinement** – repeatedly moves a tiny
   amount of mass from the index that contributes most to the current
   maximal convolution entry to the index that contributes least.
   An annealing schedule accepts occasional uphill moves.

All steps honour the simplex constraint Σa = √(2 n) and non‑negativity.
The routine adapts the sequence length to the time budget (≈ 12 k for a
full 1000 s run) and returns the best sequence found in plain list
form.
"""

import time
import numpy as np
from typing import List, Tuple


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Scoring (the objective after fixing the L¹‑norm)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  With Σa = √(2 n) this equals max(a*a)."""
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed sub‑gradient descent (soft‑max surrogate of max)
# ----------------------------------------------------------------------


def _smooth_subgradient_descent(a0: np.ndarray,
                                target_sum: float,
                                deadline: float,
                                lambda_scale: float = 8.0) -> Tuple[np.ndarray, float]:
    """
    Minimise max(a*a) using a smooth surrogate:
        f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)

    The gradient is a convolution with the soft‑max weights.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    base_step = target_sum * 0.3          # a modest initial step size
    n = a.shape[0]

    while time.time() < deadline:
        # convolution and soft‑max weights
        b = _conv_fft(a)
        max_b = b.max()
        # stabilised exponentials
        diff = b - max_b
        diff = np.maximum(diff, -50.0)               # avoid underflow
        w = np.exp(lambda_scale * diff)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = w / w_sum                               # soft‑max probabilities

        # gradient = conv(a, w_rev) restricted to the first n entries
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)     # length 3n‑2
        L = len(w) - 1                                   # = 2n‑2
        g = conv_full[L - np.arange(n)]

        # step (diminishing)
        step = base_step / np.sqrt(it + 1.0)
        g_norm = np.linalg.norm(g)
        if g_norm > 0.0:
            a -= step * g / g_norm

        a = _project_to_simplex(a, target_sum)

        # tracking best (true max, not the surrogate)
        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth sub‑grad it={it:6d} score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided mass‑transfer local refinement
# ----------------------------------------------------------------------


def _guided_local_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing style move that transfers a tiny amount of mass
    from the index contributing most to the current maximal convolution
    entry to the index contributing least (as measured by the sub‑gradient
    of the max).  Uses the O(n) exact update of the convolution after a
    transfer.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                     # initial temperature
    last_full = start                        # last full recomputation time

    while time.time() < deadline:
        max_b = b.max()
        # sub‑gradient of the max (mask of tight entries)
        eps = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps).astype(np.float64)

        # compute sub‑gradient vector g_i = Σ_j a_j·mask_{i+j}
        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)
        L = len(mask) - 1
        g = conv_full[L - np.arange(n)]

        donor = int(np.argmax(g))
        receiver = int(np.argmin(g))
        if donor == receiver:
            # fall back to a random pair
            donor = rng.integers(n)
            receiver = rng.integers(n)
            while donor == receiver:
                receiver = rng.integers(n)

        # propose a modest transfer
        max_delta = min(a[donor] * 0.025, target_sum * 0.001)
        if max_delta <= 0.0:
            # nothing to move – break early
            break
        delta = rng.random() * max_delta

        # exact O(n) update of the full convolution after the transfer
        b_new = b.copy()
        # linear part – 2·Δ·a shifted by donor / receiver
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # acceptance with a linear cooling schedule
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Refine improved score={best_val:.6f}")

        # periodic full recomputation to cure accumulated numeric drift
        if time.time() - last_full > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")
            last_full = time.time()

        # tiny random jitter every few seconds (helps escape stagnation)
        if int(elapsed * 150) % 3000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.004, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Warm‑start: generate a few simple families and keep the best
# ----------------------------------------------------------------------


def _initial_sequence(n: int, target_sum: float) -> np.ndarray:
    """Return a promising starting vector (projected onto the simplex)."""
    candidates = []

    # 1. saved sequence, if any
    if "height_sequence_1" in globals():
        cand = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        candidates.append(cand)

    # 2. power‑law families
    for exp in (-0.45, -0.5, -0.55, -0.6, -0.65):
        seq = np.arange(1, n + 1, dtype=np.float64) ** exp
        candidates.append(seq)

    # 3. exponential decay families
    for beta in (0.0005, 0.001, 0.0015, 0.002):
        seq = np.exp(-beta * np.arange(n, dtype=np.float64))
        candidates.append(seq)

    # 4. a few random draws (projected)
    rng = np.random.default_rng()
    for _ in range(2):
        seq = rng.random(n)
        candidates.append(seq)

    # project all candidates to the simplex and keep the best by the true score
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        proj = _project_to_simplex(seq, target_sum)
        val = _score(proj)
        if val < best_val:
            best_val = val
            best_seq = proj.copy()
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the supplied time budget (seconds) and returns
    the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.25      # tiny safety margin

    # --------------------------------------------------------------
    # Choose a problem size that balances quality and speed
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)                # we keep Σa = √(2n)

    # --------------------------------------------------------------
    # Warm‑start
    # --------------------------------------------------------------
    init_seq = _initial_sequence(n, target_sum)

    # --------------------------------------------------------------
    # Stage 1 – smoothed sub‑gradient descent (majority of the budget)
    # --------------------------------------------------------------
    sub_deadline = time.time() + 0.60 * (deadline - time.time())
    a_sub, val_sub = _smooth_subgradient_descent(init_seq,
                                                 target_sum,
                                                 sub_deadline,
                                                 lambda_scale=8.0)
    best_seq = a_sub
    best_score = val_sub
    print(f"[{time.time():.2f}] After smooth sub‑grad best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – short random restarts (≈15 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        rest_deadline = time.time() + 0.15 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2.0:
            remaining = rest_deadline - time.time()
            run_for = min(5.0, remaining)
            # create a jittered version of the current best
            jitter = rng.normal(scale=target_sum * 0.025, size=n)
            cand = best_seq + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_sum)

            a_tmp, val_tmp = _smooth_subgradient_descent(cand,
                                                         target_sum,
                                                         time.time() + run_for,
                                                         lambda_scale=6.0)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – guided local refinement (use all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _guided_local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final cleanup – strip trailing zeros and evaluate once more
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a later call
    globals()["height_sequence_1"] = final_seq

    return final_seq
