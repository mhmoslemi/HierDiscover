"""
Search for a non‑negative step sequence with a low evaluation score.

The algorithm runs in three stages:

1. **Fast sub‑gradient descent** –  minimise the true
   ``max(convolution)`` on the full fine‑grained vector.
   The iterate stays on the simplex Σa = √(2 n) (scale‑invariant).

2. **Block‑wise simulated annealing** –  the vector is approximated by a
   piecewise‑constant function (≈30–60 blocks).  In each iteration a
   small amount of mass is transferred from the block that contributes
   most to the current maximal convolution entry to the block that
   contributes the least.  The step size decays with the iteration
   count and occasional uphill moves are accepted according to a linear
   temperature schedule.

3. **Cleanup** –  trailing zeros are removed, the final sequence is
   evaluated with the provided ``evaluate_sequence`` and stored for a
   possible warm‑start in a later run.

The whole routine respects the 1000 s time budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective: 2·n·max(a*a) / (sum(a)²).
    With Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution (dense vector)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    # initial step magnitude – a fraction of the total mass
    base_step = target_sum * 0.5
    n = a.shape[0]

    while time.time() < deadline:
        b = _conv_fft(a)
        max_b = b.max()
        eps = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps).astype(np.float64)

        # sub‑gradient of max(b) w.r.t. a (the factor 2 is omitted – it
        # only rescales the direction)
        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)
        L = len(mask) - 1          # = 2n‑2
        g = conv_full[L - np.arange(n)]

        step = base_step / np.sqrt(it + 1.0)
        gn = np.linalg.norm(g)
        if gn > 0:
            a -= step * g / gn

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:6d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Block‑wise simulated‑annealing refinement
# ----------------------------------------------------------------------


def _block_refine(a_dense: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Piecewise‑constant (block) refinement.
    The vector a_dense is represented by B blocks (≈30–60).  In each
    iteration a small mass δ is moved from the block that most
    contributes to the current maximal convolution entry to the block
    that contributes the least.  A simple linear temperature schedule
    allows occasional uphill moves.
    """
    n = a_dense.shape[0]

    # choose a reasonable number of blocks (more for larger n)
    B = max(20, min(70, n // 200))          # ≈30‑60 blocks for n≈8000‑12000
    base_block = n // B
    block_sizes = np.full(B, base_block, dtype=int)
    # distribute the remainder into the first blocks
    rem = n - base_block * B
    block_sizes[:rem] += 1

    # start indices of each block
    block_starts = np.concatenate(([0], np.cumsum(block_sizes)[:-1]))
    block_slices = [slice(s, s + sz) for s, sz in zip(block_starts, block_sizes)]

    # initialise block heights as the average of the dense vector
    h = np.empty(B, dtype=np.float64)
    for i, sl in enumerate(block_slices):
        h[i] = a_dense[sl].mean()

    # enforce the exact sum constraint (the total mass is Σ h_i·size_i)
    cur_mass = np.dot(h, block_sizes)
    h *= target_sum / cur_mass
    for i, sl in enumerate(block_slices):
        a_dense[sl] = h[i]

    best_a = a_dense.copy()
    best_val = _score(best_a)

    rng = np.random.default_rng()
    it = 0

    # temperature schedule – start relatively hot and cool linearly
    total_time = deadline - time.time()
    T0 = best_val * 0.05                     # a few percent of the current score

    while time.time() < deadline:
        # ---- convolution and sub‑gradient of the current max ----
        b = _conv_fft(a_dense)
        max_b = b.max()
        eps = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps).astype(np.float64)

        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a_dense, mask_rev)
        L = len(mask) - 1
        g = conv_full[L - np.arange(n)]       # sub‑gradient of max(b)

        # block‑wise gradient (sum of g over each block)
        block_grad = np.add.reduceat(g, block_starts)

        donor = int(np.argmax(block_grad))
        receiver = int(np.argmin(block_grad))
        if donor == receiver:
            # nothing to do – perturb a little and continue
            jitter = rng.normal(scale=target_sum * 4e-4, size=n)
            a_dense = np.maximum(a_dense + jitter, 0.0)
            a_dense = _project_to_simplex(a_dense, target_sum)
            # recompute block heights
            for i, sl in enumerate(block_slices):
                h[i] = a_dense[sl].mean()
            it += 1
            continue

        donor_mass = h[donor] * block_sizes[donor]
        # step size decays with the iteration count
        step_factor = 0.15 / np.sqrt(it + 1.0)
        delta = min(donor_mass * step_factor, target_sum * 6e-4)
        if delta <= 0:
            it += 1
            continue

        # propose move
        h_new = h.copy()
        h_new[donor] -= delta / block_sizes[donor]
        h_new[receiver] += delta / block_sizes[receiver]

        # enforce non‑negativity (if donor would become negative, clip)
        if h_new[donor] < 0:
            # take only the amount that can be removed
            delta = h[donor] * block_sizes[donor]
            h_new[donor] = 0.0
            h_new[receiver] = h[receiver] + delta / block_sizes[receiver]

        # build the new dense vector
        a_new = a_dense.copy()
        a_new[block_slices[donor]] = h_new[donor]
        a_new[block_slices[receiver]] = h_new[receiver]

        # ---- evaluate new max ----
        b_new = _conv_fft(a_new)
        new_max = b_new.max()
        new_val = new_max                     # because Σa = √(2n)

        # ---- acceptance ----
        elapsed = time.time() - (deadline - total_time)
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            h = h_new
            a_dense = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_dense.copy()
                print(f"[{time.time():.2f}] Block refine it={it:6d} score={best_val:.6f}")

        # occasional full re‑normalisation to combat numerical drift
        if it % 1000 == 0:
            cur_mass = np.dot(h, block_sizes)
            h *= target_sum / cur_mass
            for i, sl in enumerate(block_slices):
                a_dense[sl] = h[i]

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)            # fixing Σa simplifies the objective

    # ------------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_dense = _resize_sequence(np.array(globals()["height_sequence_1"],
                                               dtype=np.float64), n)
    else:
        # simple decreasing power‑law as a fallback
        init_dense = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init_dense = _project_to_simplex(init_dense, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – projected sub‑gradient descent (≈20 % of budget)
    # ------------------------------------------------------------------
    sub_deadline = time.time() + 0.20 * (deadline - time.time())
    a_sub, val_sub = _subgradient_descent(init_dense, target_sum, sub_deadline)
    best_seq = a_sub
    best_score = val_sub
    print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – block‑wise simulated annealing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _block_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After block refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
