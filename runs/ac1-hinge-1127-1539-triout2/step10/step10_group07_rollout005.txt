def _smooth_gradient_descent(a0, target_sum, deadline):
    a = projection of a0 onto simplex target_sum.
    n = len(a)
    best_seq = a.copy()
    best_val = _score(a) # equals max conv after sum = sqrt(2n)
    step_factor = target_sum * 0.8 maybe; param.
    beta = 0.9
    m = np.zeros_like(a)
    iteration = 0
    while time.time() < deadline:
        # Compute convolution
        b = _conv_fft(a)  # length 2n-1
        max_b = b.max()
        # compute smoothing epsilon, can't be too small else numeric issues. Use epsilon = max_b * 0.2 + 1e-6
        eps = max_b * 0.2 + 1e-9
        # compute softmax weights
        w = np.exp((b - max_b) / eps)
        s = w.sum()
        if s == 0:
            # fallback to mask
            mask = (b >= max_b - 1e-12).astype(np.float64)
            w = mask
            s = w.sum()
        w = w / s
        # compute gradient via convolution of a & w_rev
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)
        L = len(w) - 1
        g = conv_full[L - np.arange(n)]
        grad = 2 * g
        # momentum
        m = beta * m + (1 - beta) * grad
        # step size schedule
        step = step_factor / np.sqrt(iteration + 1.0)
        a_new = a - step * m
        # project onto simplex
        a_new = _project_to_simplex(a_new, target_sum)
        # evaluate
        cur_val = _score(a_new) # includes sum factor but sum fixed so cur_val = max conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] Smooth GD it={iteration:6d} best={best_val:.6f}")
        a = a_new
        iteration += 1
    return best_seq, best_val

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. Warm‑start – use the previously best known sequence if available,
   otherwise a decreasing power‑law, resized to the chosen dimension
   and projected onto the L¹‑simplex Σa = √(2 n).

2. Smooth gradient descent (SGD) – minimise a smoothed version of the
   “max‑convolution’’ objective.
   * The smoothing uses a soft‑max (log‑sum‑exp) with a temperature
     proportional to the current maximal convolution entry.
   * The gradient of the smoothed objective is obtained by one FFT‑based
     convolution with the soft‑max weights.
   * A simple momentum term and a diminishing step size are employed.
   * The iterate is projected onto the simplex after each update.
   * The best iterate (lowest max‑convolution) is kept.

3. Random‑restart SGD – briefly jitter the current best sequence and
   run a short SGD run; keep any improvement.  Several such restarts
   are performed until the allocated time slice expires.

4. Local mass‑transfer refinement – a simulated‑annealing hill‑climbing
   that transfers a tiny amount of mass from a “bad’’ index (large
   contribution to the current maximal convolution entry) to a “good’’
   one, updating the full convolution in O(n).  The temperature schedule
   is linear and the move size is adaptive.

5. Clean‑up – trim trailing zeros, evaluate with ``evaluate_sequence``,
   store the result for possible warm‑starts and return the list of
   heights.

All steps respect the non‑negativity constraint; the L¹‑norm is fixed
to √(2 n) so the returned objective value equals the maximal
convolution entry.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective: 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) the return equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth gradient descent on a soft‑max approximation of max‑convolution
# ----------------------------------------------------------------------


def _smooth_gradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max‑convolution objective.
    The smoothing uses a temperature proportional to the current maximum.
    Momentum and a diminishing step size are employed.
    Returns the best sequence seen and its (unsmoothed) max‑convolution value.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    beta = 0.85  # momentum coefficient
    m = np.zeros_like(a)  # momentum buffer
    it = 0
    # Base step magnitude – a few percent of the total mass
    base_step = target_sum * 0.05

    while time.time() < deadline:
        # --- convolution of a with itself (the true objective) -----------------
        b = _conv_fft(a)               # length 2n‑1
        max_b = b.max()
        if max_b < best_val:
            best_val = max_b
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth‑GD it={it:6d} best={best_val:.6f}")

        # --- smoothing parameter -------------------------------------------------
        eps = max_b * 0.20 + 1e-12       # keeps the softmax numerically stable

        # soft‑max weights (size 2n‑1)
        w = np.exp((b - max_b) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:                # degenerate case – fall back to a mask
            w = (b >= max_b - 1e-12).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --- gradient of the smoothed objective ---------------------------------
        # g_i = 2 * sum_j a_j * w_{i+j}
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)      # length 3n‑2
        L = len(w) - 1                                 # = 2n‑2
        g = conv_full[L - np.arange(n)]                # shape (n,)
        grad = 2.0 * g

        # --- momentum update ----------------------------------------------------
        m = beta * m + (1.0 - beta) * grad

        # --- step size (diminishing) -------------------------------------------
        step = base_step / np.sqrt(it + 1.0)
        # cap step to keep updates reasonable
        step = min(step, target_sum * 0.5)

        # --- gradient step + projection -----------------------------------------
        a_new = a - step * m
        a = _project_to_simplex(a_new, target_sum)

        # occasional small jitter to escape flat plateaus
        if it % 1200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact mass‑transfer refinement (simulated annealing)
# ----------------------------------------------------------------------


def _local_refine(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that moves a small amount of mass
    from a “bad’’ index (large contribution to the current maximal
    convolution entry) to a “good’’ one.  The full convolution is updated
    in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.05                 # initial temperature (a bit higher)
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0.0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick from the lower tail of contributions
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.06, target_sum * 0.001)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # Update convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ (subtract from donor, add to receiver)
        b_new[donor : donor + n] += -2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic corrections
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local‑refine improved={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic full recomputation (keeps numerical drift in check)
        # --------------------------------------------------------------
        if time.time() - last_recalc > 15.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved={best_val:.6f}")

        # occasional jitter to shake out of plateaus
        if (int(elapsed * 120) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-15:
                eps = min(a[i] * 0.004, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved={best_val:.6f}")

    return best_a, best_val

        # Simple decreasing power‑law as a generic starter
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.60))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – smooth gradient descent (≈50 % of budget)
    # ------------------------------------------------------------------
    split_time = time.time()
    sg_deadline = split_time + 0.48 * (deadline - split_time)
    best_seq, best_val = _smooth_gradient_descent(init, target_sum, sg_deadline)
    print(f"[{time.time():.2f}] After SGD best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short random restarts of SGD (~20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rest_deadline = time.time() + 0.20 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2.0:
            rem = rest_deadline - time.time()
            run_time = min(6.0, rem)
            if run_time < 1.0:
                break
            jitter = rng.normal(scale=target_sum * 0.02, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            seq_tmp, val_tmp = _smooth_gradient_descent(
                start_seq, target_sum, time.time() + run_time
            )
            if val_tmp < best_val:
                best_val = val_tmp
                best_seq = seq_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_val:
            best_seq = seq_ref
            best_val = val_ref
        print(f"[{time.time():.2f}] After local refinement best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
