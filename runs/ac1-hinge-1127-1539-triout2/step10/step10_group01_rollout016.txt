```python
"""
Hybrid search for low max–convolution step function.

The routine combines:
* dense parametric random sampling (power‑law, exponential, polynomial);
* projected Adam‑gradient descent on a soft‑max surrogate of the max‑convolution;
* a simple simulated‑annealing phase that moves mass between entries;
* an exact greedy “mass‑transfer” sweep that flattens the convolution peak;
* a final polishing Adam run.

All phases respect the global time budget and keep the sequence on the
simplex Σa = √(2 n).  The best sequence found is trimmed of trailing zeros
and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective (score) – the benchmark function without the security wrapper
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the fast pre‑search)
# ----------------------------------------------------------------------


def _pow_law_sequence(n: int, p: float, shift: float, target_sum: float) -> np.ndarray:
    """(i+shift)^{-p} of length n, scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_pow_law_sequence(n: int, p: float, q: float, shift: float, target_sum: float) -> np.ndarray:
    """(i+shift)^{-p} * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam * i) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a *= target_sum / a.sum()
    return a


def _poly_decay_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(1 - i/(n-1))^{p}  (zero for i=n-1) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - i / (n - 1))
    a = a ** p
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected Adam‑style gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 5e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Adam‑style projected GD on the soft‑max of the convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)

        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # anneal beta
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(start_vec: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short Adam‑GD runs with jitter; keep the best result."""
    best_a = start_vec.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    # keep launching short runs while we have time
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=5e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy moves that directly lower the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()

def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:

    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    # initial temperature proportional to current score
    T0 = max(1e-4, cur_score * 0.05)

    while time.time() < deadline:
        elapsed = time.time() - start
        t = elapsed / total
        T = T0 * (1.0 - t)  # linear cooling

        # donor weighted by current mass (prefer large entries)
        probs = a / a.sum()
        donor = rng.choice(n, p=probs)
        # receiver uniform among all indices
        receiver = rng.integers(n)
        if donor == receiver:
            continue

        # step size – a fraction of donor mass (max 5 %)
        max_delta = a[donor] * 0.05
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # re‑project to keep exact sum (helps with numerical drift)
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new best={best_score:.6f}")

    return best_a, best_score

"""
Hybrid search for a low max–convolution step function.

The routine mixes several complementary strategies:

* dense random parametric sampling (power‑law, exponential, polynomial);
* Adam‑style projected gradient descent on a soft‑max surrogate of max‑conv;
* a simple simulated‑annealing phase that shifts small masses between entries;
* an exact greedy “mass‑transfer” sweep that flattens the highest convolution peak;
* a final polishing Adam run.

All phases respect the global time budget and keep the sequence on the
simplex Σa = √(2 n).  The best sequence found is trimmed of trailing zeros and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective (score) – the benchmark function without the security wrapper
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the fast pre‑search)
# ----------------------------------------------------------------------


def _pow_law_sequence(n: int, p: float, shift: float, target_sum: float) -> np.ndarray:
    """(i+shift)^(-p) of length n, scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_pow_law_sequence(n: int, p: float, q: float, shift: float,
                          target_sum: float) -> np.ndarray:
    """(i+shift)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam * i) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a *= target_sum / a.sum()
    return a


def _poly_decay_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(1 - i/(n‑1))^p (zero at the end) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - i / (n - 1))
    a = a ** p
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected Adam‑style gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 5e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Adam‑style projected GD on the soft‑max of the convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)

        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake off plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # anneal beta
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(start_vec: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Perform several short Adam‑GD runs with random jitter and keep the best."""
    best_a = start_vec.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=5e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

        elif family == 1:  # power‑law + exponential tail
            p = rng.uniform(0.3, 1.5)
            q = rng.uniform(0.0, 9.0)
            shift = rng.uniform(0.0, 4.0)
            seq = _exp_pow_law_sequence(n, p, q, shift, target_sum)
        elif family == 2:  # pure exponential decay
            lam = rng.uniform(0.0005, 0.02)
            seq = _exp_sequence(n, lam, target_sum)
        elif family == 3:  # polynomial decay
            p = rng.uniform(0.2, 5.0)
            seq = _poly_decay_sequence(n, p, target_sum)
        else:  # blended exponential with a tiny uniform baseline
            lam = rng.uniform(0.0005, 0.02)
            base = rng.uniform(0.0, 0.2) * target_sum / n
            seq = _exp_sequence(n, lam, target_sum)
            seq += base
            seq = _project_to_simplex(seq, target_sum)

        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Phase0 improved best={best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts (≈30 % of remaining time)
    # ---------------------------------------------------------------
    gd_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best={best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – Simple simulated annealing (≈20 % of remaining time)
    # ---------------------------------------------------------------
    sa_deadline = time.time() + 0.20 * (deadline - time.time())
    a_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_deadline)
    if val_sa < best_score:
        best_score = val_sa
        best_seq = a_sa.copy()
        print(f"[{time.time():.2f}] SA improved best={best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase C – Exact deterministic mass‑transfer sweep (≈30 % of remaining time)
    # ---------------------------------------------------------------
    dmt_deadline = time.time() + 0.30 * (deadline - time.time())
    a_dmt, val_dmt = _deterministic_mass_transfer_exact(best_seq,
                                                         dmt_deadline,
                                                         target_sum)
    if val_dmt < best_score:
        best_score = val_dmt
        best_seq = a_dmt.copy()
        print(f"[{time.time():.2f}] Exact DMT improved best={best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase D – Final polishing with a longer Adam run
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation with the official wrapper
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
