"""
Propose a low‑score step function using the classic “LP + mix” iteration
described in the state‑of‑the‑art paper.

The idea:
1.  Choose a modest problem size (n≈2000) – this keeps the linear
    program (n variables, 2 n‑1 constraints) well within memory/time.
2.  Start from a simple power‑law shape (a_i ∝ (i+1)^‑p) that already
    gives a score ≈1.51.
3.  Repeatedly solve the linear program

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   ∀k               (M = max(f*f))
                    b_j ≥ 0

    The solution b (called g_0) is the “most massive’’ sequence that can be
    convolved with the current f without exceeding the present worst‑case
    convolution value M.
4.  Rescale g_0 to the required ℓ¹–norm √(2 n) and convex‑combine it
    with the current f:

        f ← (1‑t)·f + t·g ,     0 < t ≤ 1

    Small t guarantees that the maximal convolution does not increase;
    we try t=0.2 and halve it if the new score does not improve.
5.  Keep the best sequence seen so far and stop when the time budget is
    exhausted.

The whole procedure respects the 1000 s budget, uses only NumPy and SciPy,
and returns a list of non‑negative heights ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Scoring utilities (replicate the evaluate_sequence formula, but faster)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2·n·max(a*a) / (sum(a)²); assumes sum(a)>0."""
    n = a.size
    max_b = np.convolve(a, a).max()
    s = a.sum()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Build the linear‑program constraint matrix A such that (f * b)_k ≤ M
# ----------------------------------------------------------------------
def _build_A(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A (shape (2n‑1, n)) with
       A[k, j] = f[k‑j] if 0≤k‑j<n else 0.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contains f shifted down by j rows
    for j in range(n):
        A[j : j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + max(budget_s - 0.5, 0.0)   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a sane problem size (large enough to be competitive,
    # small enough for a fast LP).
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 2000
    elif budget_s >= 400:
        n = 1500
    else:
        n = 1000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm start – a simple power‑law shape (the previous best used p≈0.55)
    # ------------------------------------------------------------------
    p = 0.55
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = a * (target_sum / a.sum())
    best_seq = a.copy()
    best_val = _score(best_seq)
    print(f"[{time.time():.2f}] initial   score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main LP‑based refinement loop
    # ------------------------------------------------------------------
    iteration = 0
    max_iters = 30                     # safety cap
    while time.time() < deadline - 5.0 and iteration < max_iters:
        # current maximal convolution M
        conv = np.convolve(best_seq, best_seq)
        M = conv.max()

        # build constraint matrix A (dense, but modest size)
        A = _build_A(best_seq)
