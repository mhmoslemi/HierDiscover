"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Key ideas
---------
*   Increase the search space by trying much larger lengths (up to 50 000).
*   Add a dedicated “sparse” family: a small number of equal spikes.
    Pairwise sums of spike positions are the only contributors to the
    convolution, so by spreading the spikes we can keep the maximal entry
    very low.
*   Use a short parametric scan (dense families + many random sparse
    candidates), keep the best few and improve them with
    – deterministic hill‑climb (exact updates of the convolution)
    – a projected Adam optimiser on the smooth‑max surrogate.
*   The whole routine respects the wall‑clock budget (default 1000 s) and
    returns the best sequence found.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b

                    deadline: float) -> np.ndarray:

    Projected Adam on the smooth‑max surrogate.
    Runs until the supplied deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning‑rate (scaled with the simplex radius)
    base_lr = 0.15 * target_sum

    # smoothing schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisation (dense families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Sparse random family
# ----------------------------------------------------------------------


def _random_sparse_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a sparse vector with a few equal spikes.
    The number of spikes k is drawn around √(2 n).
    """
    target = math.sqrt(2.0 * n)
    k0 = max(2, int(math.sqrt(2.0 * n)))          # baseline size
    factor = rng.uniform(0.4, 2.0)                 # allow a wide range
    k = int(min(n, max(2, round(k0 * factor))))

    # choose distinct positions
    pos = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    w = target / k
    a[pos] = w

    # tiny jitter to break exact ties (optional)
    a[pos] += rng.normal(scale=target * 1e-5, size=k)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution
    entry.  Moves a (large) fraction of the responsible mass to a more
    favourable location.  Returns the best vector seen and its maximal
    convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # index of the current maximal convolution entry
        M = int(np.argmax(b))

        # source indices influencing b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # source with the largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # destination with minimal counterpart (but not i)
        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # amount of mass to move (up to 80 % of source)
        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # --- evaluate the move ------------------------------------------------
        b_new = b.copy()
        # linear convolution update (O(n))
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # --- accept if we improve ------------------------------------------------
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search style)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small random jitter to escape a possible plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                # re‑project onto the simplex and recompute convolution
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimizes ``evaluate_sequence``.
    The strategy combines:
        * a schedule of growing lengths (up to 50 000)
        * dense parametric families + a dedicated sparse family
        * deterministic hill‑climb on the exact convolution
        * short projected‑Adam refinements
    The routine respects the global time budget and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Length schedule – start modest and grow
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000, 50000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # 1)  Generate a modest pool of candidates (dense + sparse)
        # ------------------------------------------------------------------
        candidates = []

        # dense uniform (baseline)
        a_uni = np.full(n, target_sum / n)
        candidates.append((_max_convolution_fft(a_uni), a_uni))

        # a few exponential decays (fast, cheap)
        for _ in range(8):
            lam = rng.exponential(scale=0.02)
            a = np.exp(-lam * np.arange(n, dtype=float))
            a = _project_to_simplex(a, target_sum)
            candidates.append((_max_convolution_fft(a), a))

        # a few power‑law decays
        for _ in range(8):
            p = rng.uniform(0.1, 1.5)
            a = (np.arange(1, n + 1, dtype=float)) ** (-p)
            a = _project_to_simplex(a, target_sum)
            candidates.append((_max_convolution_fft(a), a))

        # sparse random family – this is the new ingredient
        for _ in range(20):
            a = _random_sparse_sequence(n, rng)
            candidates.append((_max_convolution_fft(a), a))

        # keep the top few for further work
        candidates.sort(key=itemgetter(0))
        top_candidates = candidates[:6]

        # ------------------------------------------------------------------
        # 2)  Update global best if any of the top candidates improve
        # ------------------------------------------------------------------
        for score, vec in top_candidates:
            if score < best_score - 1e-12:
                best_seq = vec.copy()
                best_score = score
                print(f"[{time.time():.1f}] New global best (n={n}) score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 3)  Deterministic hill‑climb on the best vector for this length
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            a_cur = best_seq.copy()
        else:
            a_cur = _random_initial_sequence(n, rng)

        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        # allocate about 30 % of the remaining budget to hill‑climbing
        remaining = deadline - time.time()
        hc_deadline = min(deadline - 0.2, time.time() + 0.30 * remaining)
        a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)

        if score_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = score_hc
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 4)  Short Adam refinement (smooth‑max surrogate)
        # ------------------------------------------------------------------
        if time.time() < deadline - 1.0:
            # give the Adam run the rest of the time for this n
            adam_deadline = min(deadline - 0.5, time.time() + 0.15 * (deadline - time.time()))
            a_opt = _adam_optimize(best_seq, target_sum, adam_deadline)
            score_opt = _max_convolution_fft(a_opt)
            if score_opt < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = score_opt
                print(f"[{time.time():.1f}] Adam refinement (n={n}) score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) Final polishing on the remaining budget (long Adam + hill‑climb)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        final_deadline = deadline - 0.5
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)

        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 6) Clean‑up: drop trailing near‑zeros
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ----------------------------------------------------------------------
    # 7) Official evaluation report
    # ----------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
