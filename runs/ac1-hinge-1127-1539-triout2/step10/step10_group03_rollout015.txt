"""
Propose a low‑peak step function.

The optimizer works in three scales:

1. **Small‑scale exploration** (n≈3‑4 k).  A handful of short
   projected‑gradient runs on a soft‑max approximation of the
   convolution peak are performed.  The best small vector is kept.

2. **Upscaling** – the small vector is linearly interpolated to a
   large length (default 12 000).  After a simple alternating
   re‑ordering a high‑temperature gradient descent finishes the
   large‑scale optimisation.

3. **Fine polishing** – a short deterministic mass‑transfer pass and a
   random‑swap hill‑climber (swap two entries if the peak decreases)
   are applied while time permits.  The final sequence is trimmed of
   trailing zeros.

All steps respect non‑negativity and the ℓ¹‑simplex projection
( Σa = √(2 n) ).  The objective value after the projection is exactly
the maximum entry of the self‑convolution, i.e. the quantity that
`evaluate_sequence` returns (the factor `2·n / Σa²` cancels out).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    For vectors projected onto Σa = √(2·n) the value equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) on [0,n) renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max approximation of max‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth max{conv(a,a)}.
    Returns (grad, conv) where conv is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # safe shift
    p = exps / np.sum(exps)                    # soft‑max weights

    # grad = 2 * (a_rev) ⋆ p
    a_rev = a[::-1]
    Ar_fft = rfft(a_rev, n=size)
    Pr_fft = rfft(p, n=size)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected (momentum) gradient descent on the smooth objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max objective.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter – helps escape flat plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase temperature (beta) gradually
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves accepted with a decaying Metropolis rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))                       # index of current peak
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]                        # contribution of each donor index
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple alternating re‑ordering (large‑small‑large‑small …)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Random‑swap hill climber on the ordering
# ----------------------------------------------------------------------


def _permute_hill_climb(a0: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb that swaps two entries if the
    convolution peak (objective) improves.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    # very low temperature – we only accept strict improvements
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert – we do not accept uphill moves
            a[i], a[j] = a[j], a[i]

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine performs a fast small‑scale exploration, up‑scales to
    12 000 entries and finishes with high‑temperature gradient descent,
    a deterministic mass‑transfer pass and a random‑swap hill‑climber.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # modest safety margin

    # --------------------------------------------------------------
    # 1. small‑scale exploration
    # --------------------------------------------------------------
    n_small = 3000
    target_small = np.sqrt(2.0 * n_small)

    rng = np.random.default_rng()
    init_pool = []

    # power‑law seeds
    for p in np.linspace(0.35, 0.75, 7):
        init_pool.append(_power_law_sequence(n_small, float(p), target_small))

    # exponential seeds
    for lam in np.linspace(0.002, 0.012, 5):
        init_pool.append(_exponential_sequence(n_small, float(lam), target_small))

    # Dirichlet‑like random seeds
    for _ in range(5):
        a = rng.random(n_small)
        a *= target_small / a.sum()
        init_pool.append(a)

    # previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_small))

    # jittered copies to increase diversity
    extra = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_small * 0.01, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        extra.append(seq_j)
    init_pool.extend(extra)

    best_small = None
    best_small_val = np.inf

    # allocate roughly 20 % of the total budget for this stage
    stage_dead = start_time + budget_s * 0.20
    while time.time() < stage_dead - 5.0:
        # pick seed: either jitter around current best or random pool element
        if best_small is not None and rng.random() < 0.6:
            seed = best_small.copy()
            jitter = rng.normal(scale=target_small * 0.005, size=n_small)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(init_pool)

        # each run gets at most 25 s (or whatever fits before stage_dead)
        run_dead = min(time.time() + 25.0, stage_dead - 2.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(seed,
                                                     run_dead,
                                                     target_small,
                                                     max_beta=5e5,
                                                     step_factor=0.7,
                                                     gamma=0.99)
        if val_opt < best_small_val:
            best_small_val = val_opt
            best_small = a_opt.copy()
            print(f"[{time.time():.2f}] Small‑scale new best={best_small_val:.6f}")

    # one short deterministic refinement at the small scale (if time permits)
    if best_small is not None and time.time() < stage_dead - 2.0:
        mt_dead = time.time() + min(5.0, stage_dead - time.time() - 1.0)
        a_ref, val_ref = _deterministic_mass_transfer(best_small,
                                                      mt_dead,
                                                      target_small,
                                                      start_temp=0.5,
                                                      end_temp=1e-4,
                                                      low_frac=0.07)
        if val_ref < best_small_val:
            best_small_val = val_ref
            best_small = a_ref.copy()
            print(f"[{time.time():.2f}] Small‑scale MT refined best={best_small_val:.6f}")

    # --------------------------------------------------------------
    # 2. Upscaling to the final problem size (≈12 k)
    # --------------------------------------------------------------
    n_big = 12000
    target_big = np.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_small, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # a quick alternating re‑ordering often reduces the peak
    alt = _alternating_order(seq_big)
    if _max_convolution_fft(alt) < _max_convolution_fft(seq_big):
        seq_big = alt

    # --------------------------------------------------------------
    # 3. Large‑scale polishing (gradient descent + mass‑transfer)
    # --------------------------------------------------------------
    # remaining time for the big stage
    remaining = deadline - time.time()
    # keep a few seconds for the final permutation pass
    perm_deadline = deadline - 3.0
    final_grad_dead = perm_deadline

    if time.time() < final_grad_dead:
        seq_big, big_val = _annealed_gradient_descent(seq_big,
                                                      final_grad_dead,
                                                      target_big,
                                                      max_beta=1e6,
                                                      step_factor=0.5,
                                                      gamma=0.995)
        print(f"[{time.time():.2f}] Large‑scale GD finished, score={big_val:.6f}")

    # optional deterministic mass‑transfer after GD
    if time.time() < perm_deadline - 2.0:
        mt_dead = time.time() + min(4.0, perm_deadline - time.time() - 1.0)
        seq_big, mt_val = _deterministic_mass_transfer(seq_big,
                                                       mt_dead,
                                                       target_big,
                                                       start_temp=0.3,
                                                       end_temp=1e-5,
                                                       low_frac=0.07)
        if mt_val < big_val:
            big_val = mt_val
            seq_big = seq_big.copy()
            print(f"[{time.time():.2f}] Post‑GD MT improved score={big_val:.6f}")

    # --------------------------------------------------------------
    # 4. Random‑swap hill‑climber on the ordering
    # --------------------------------------------------------------
    if time.time() < perm_deadline:
        seq_big, perm_val = _permute_hill_climb(seq_big, perm_deadline)
        if perm_val < big_val:
            big_val = perm_val
            seq_big = seq_big.copy()
            print(f"[{time.time():.2f}] Permutation climb best={big_val:.6f}")

    # --------------------------------------------------------------
    # 5. Trim trailing zeros for compactness
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # 6. Final log (optional)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception:
        pass

    # expose for potential later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
