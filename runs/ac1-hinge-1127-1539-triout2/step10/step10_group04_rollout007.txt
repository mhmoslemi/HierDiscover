"""
Improved search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm combines:

1.  A richer parametric family scan (exponential, power‑law,
    mixed, piecewise‑exponential, Dirichlet, Gaussian, cosine).
2.  Projection onto the ℓ¹‑simplex Σa = √(2 n) after every move.
3.  Two gradient‑based refinements:
    * Adam on a smooth‑max (log‑sum‑exp) surrogate (as in the
      reference implementation) – long runs with aggressive
      learning‑rate and β‑schedule.
    * A custom “flatten‑convolution” optimiser that directly reduces
      the ℓ²‑distance of the autocorrelation to its average value.
4.  A deterministic hill‑climb that moves mass from the most
    offending index to a better one (executed only on the smallest
    lengths for speed).
5.  Small random perturbations for escaping plateaux.
6.  Gradual up‑sampling of the current best vector to larger problem
    sizes, re‑optimising at each scale.

All steps respect a global wall‑clock budget (default 1000 s).  The
best sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    The algorithm runs until the wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.25 * target_sum          # a bit larger than the reference

    # start with modest smoothing, double every ~200 steps
    beta = 3.0
    beta_max = 2e7
    beta_factor = 2.3

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # jitter every ~400 iterations to avoid getting stuck
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-5 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _flatten_optimize(a0: np.ndarray, target_sum: float,
                     deadline: float) -> np.ndarray:
    """
    Gradient descent that reduces the ℓ²‑distance of the autocorrelation
    a*a to its average value (which is the theoretical lower bound).
    """
    a = a0.copy()
    n = a.size
    avg = target_sum * target_sum / (2 * n - 1)  # desired average of a*a

    lr = 0.04 * target_sum   # modest learning rate (adapted later)

    while time.time() < deadline:
        # full convolution
        size1 = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(a, n=size1)
        b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

        # residual (b - avg)
        r = b - avg

        # gradient = 4 * (r ⋆ a_rev)
        a_rev = a[::-1]
        conv_len = (2 * n - 1) + n - 1            # = 3n‑2
        size2 = _next_pow2(conv_len)
        R_fft = np.fft.rfft(r, n=size2)
        A_rev_fft = np.fft.rfft(a_rev, n=size2)
        grad_full = np.fft.irfft(R_fft * A_rev_fft, n=size2)[:conv_len]
        g = 4.0 * grad_full[n - 1: n - 1 + n]

        # update + projection
        a -= lr * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # slowly decay learning rate
        lr *= 0.9997

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n) from a
    randomly chosen parametric family.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 8)   # now 8 families
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.1, 1.2)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # piecewise exponential
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 4:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 5:                   # Gaussian‑like (half‑Gaussian)
        sigma = rng.uniform(0.15, 0.45) * n
        a = np.exp(- (idx ** 2) / (2.0 * sigma ** 2))

    elif family == 6:                   # cosine‑power (smooth start)
        p = rng.uniform(0.5, 2.5)
        a = np.cos(np.pi * idx / (2.0 * n)) ** p

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    # add a little noise, enforce non‑negativity and project
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves mass away from the current
    worst convolution entry.  Returns the best vector seen and its
    maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))                     # worst index
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]               # source that contributes most

        if a[i] < 1e-14:
            break

        # find a destination that hurts the least
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]

        j_candidates = np.argsort(counterpart[i_low:i_high + 1])
        j = None
        for cand in j_candidates:
            cand_idx = i_low + cand
            if cand_idx != i:
                j = cand_idx
                break
        if j is None:
            break

        delta = 0.7 * a[i]      # a slightly smaller fraction than before
        if delta <= 0.0:
            break

        # cheap O(n) convolution update
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move – binary search style
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small jitter to escape potential plateaux
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small parametric family scan (extended)
# ----------------------------------------------------------------------


def _family_candidates(n: int, rng, top_k: int = 8):
    """
    Scan several simple families (exponential, power‑law, mixed,
    piecewise, Dirichlet, Gaussian, cosine) and keep the best
    ``top_k`` candidates.
    """
    target = math.sqrt(2.0 * n)
    candidates = []

    # exponential decay
    for _ in range(25):
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # power‑law
    for _ in range(25):
        p = rng.uniform(0.1, 1.4)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # mixed power‑law × exponential
    for _ in range(20):
        p = rng.uniform(0.1, 1.4)
        lam = rng.exponential(scale=0.015)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p) * np.exp(-lam * np.arange(n))
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # piecewise exponential
    for _ in range(10):
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # Dirichlet (uniform on simplex)
    for _ in range(15):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # Gaussian (half‑Gaussian)
    for _ in range(15):
        sigma = rng.uniform(0.15, 0.45) * n
        a = np.exp(- (np.arange(n, dtype=float) ** 2) / (2.0 * sigma ** 2))
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # Cosine‑power
    for _ in range(15):
        p = rng.uniform(0.5, 2.5)
        a = np.cos(np.pi * np.arange(n, dtype=float) / (2.0 * n)) ** p
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # sort and keep the best ``top_k``
    candidates.sort(key=itemgetter(0))
    return candidates[:top_k]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` under a wall‑clock budget.

    The procedure:
        1.  Start from a known construction (if available) or a
            random vector from an enlarged family pool.
        2.  Grow the problem size gradually (2000 → 5000 → 12000 → 30000).
        3.  At each size:
                • a quick parametric scan,
                • ensemble averaging,
                • several Adam runs (soft‑max surrogate),
                • a short “flatten‑convolution” refinement,
                • optional deterministic hill‑climb for the smallest size,
                • a few random perturbations.
        4.  Final long Adam + flatten run on the largest size.
        5.  Trim trailing near‑zeros and report the official score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.6   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Seed from a previously discovered construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 1)  Length schedule – start small, finish large
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 12000, 30000]

    for n in schedule:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise (upsample previous best if possible)
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        remaining = deadline - time.time()
        if remaining < 8.0:
            break

        # --------------------------------------------------------------
        # 1a)  Quick parametric family scan (≈12 % of remaining time)
        # --------------------------------------------------------------
        fam_deadline = time.time() + min(remaining * 0.12, 25.0)
        top_candidates = _family_candidates(n, rng, top_k=8)
        # ensure the up‑sampled or random start participates
        init_score = _max_convolution_fft(a_init)
        top_candidates.append((init_score, a_init))
        top_candidates.sort(key=itemgetter(0))
        top_candidates = top_candidates[:8]

        # --------------------------------------------------------------
        # 1b)  Ensemble averaging of the best candidates
        # --------------------------------------------------------------
        ensemble = np.mean([vec for _, vec in top_candidates], axis=0)
        ensemble = _project_to_simplex(ensemble, target_sum)
        ens_score = _max_convolution_fft(ensemble)
        if ens_score < best_score - 1e-12:
            best_seq = ensemble.copy()
            best_score = ens_score
            print(f"[{time.time():.1f}] Ensemble improvement (n={n}) "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1c)  Adam optimisation (soft‑max surrogate)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 4.0:
            break
        # allocate most of the time to Adam + flatten
        adam_time = remaining * 0.55
        flatten_time = remaining * 0.30
        extra_time = remaining * 0.15

        # determine number of restarts (≈6 s each)
        min_restart = 5.5
        n_restarts = max(1, int(adam_time // min_restart))
        per_restart = adam_time / n_restarts

        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_a, target_sum, this_deadline)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam improvement (n={n}) "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1d)  Flatten‑convolution refinement
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            flatten_deadline = min(deadline - 0.2, time.time() + flatten_time)
            a_flat = _flatten_optimize(best_seq, target_sum, flatten_deadline)
            flat_score = _max_convolution_fft(a_flat)
            if flat_score < best_score - 1e-12:
                best_seq = a_flat.copy()
                best_score = flat_score
                print(f"[{time.time():.1f}] Flatten improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1e)  Deterministic hill‑climb (only for the smallest size)
        # --------------------------------------------------------------
        if n <= 2000 and time.time() < deadline - 1.0:
            hc_deadline = min(deadline - 0.2, time.time() + extra_time * 0.5)
            a_cur = best_seq.copy()
            b_cur = _full_convolution(a_cur)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb (n={n}) score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1f)  Tiny random tweaks (simulated‑annealing style)
        # --------------------------------------------------------------
        while time.time() + 1.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target_sum)
            score = _max_convolution_fft(cand)
            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score
                print(f"[{time.time():.1f}] Random tweak (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1g)  Optional short extra Adam run (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 0.8 and extra_time > 2.0:
            extra_deadline = deadline - 0.4
            a_ref = _adam_optimize(best_seq, target_sum, extra_deadline)
            ref_score = _max_convolution_fft(a_ref)
            if ref_score < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = ref_score
                print(f"[{time.time():.1f}] Post‑HC Adam (n={n}) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing on the remaining budget (largest size)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        final_deadline = deadline - 0.9
        print(f"[{time.time():.1f}] Final Adam start (n={n})")
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        # a short flatten pass
        flat_deadline = deadline - 0.5
        best_seq = _flatten_optimize(best_seq, target, flat_deadline)

        # final deterministic hill‑climb (lightweight)
        if n <= 5000:
            b_final = _full_convolution(best_seq)
            hc_deadline = deadline - 0.2
            best_seq, final_hc_score = _hill_climb_deterministic(
                best_seq, b_final, hc_deadline)
            if final_hc_score < best_score - 1e-12:
                best_score = final_hc_score
                print(f"[{time.time():.1f}] Final hill‑climb improvement "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a uniform vector (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Official evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
