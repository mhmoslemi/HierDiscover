            k_max = int(np.argmax(b))
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            # choose donor from contributing indices proportional to product
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            prod = a[idx] * a[k_max - idx]
            if prod.sum() > 0:
                donor = rng.choice(idx, p=prod / prod.sum())
            else:
                donor = rng.integers(lo, hi + 1)
            # pick a receiver among low‑contribution indices
            weight = np.zeros(n, dtype=np.float64)
            weight[lo:hi + 1] = prod
            thresh = np.percentile(weight, 30)
            low = np.where(weight <= thresh)[0]
            if low.size:
                receiver = rng.choice(low)
            else:
                receiver = rng.integers(n)

            if donor == receiver:
                continue
            max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta

            # Update a and b efficiently (O(n))
            b_new = b.copy()
            # linear part: -2*delta*a added at donor window, +2*delta*a at receiver window
            b_new[donor:donor + n] -= 2.0 * delta * a
            b_new[receiver:receiver + n] += 2.0 * delta * a

            b_new[2 * receiver] += delta * delta
            if donor != receiver:

            new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))
            # simple acceptance (always accept if better)
            if new_val < best_val or rng.random() < np.exp(-(new_val - best_val) * 0.1):
                a[donor] -= delta
                a[receiver] += delta
                b = b_new
                if new_val < best_val:
                    best_val = new_val
                    best_seq = a.copy()
                    best_score = best_val
                    print(f"[{time.time():.2f}] refined score={best_score:.6f}")

    # Trim trailing zeros (unlikely) and return list:
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()
    # final check with provided evaluate_sequence

        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error: {e}")

    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Fast parametric search for a step‐function with a very low benchmark score.

The optimiser uses the one‑parameter family  

    a_i = (1 - (i/(n‑1))**p) ,   i = 0 … n‑1

scaled so that Σ a_i = √(2 n).  The objective
    2 n·max(a*a) / (Σa)²  =  max(a*a)     (because Σa = √(2 n))

is therefore just the maximal entry of the autocorrelation.
For a fixed n the value of p determines the score analytically,
and a short grid search (coarse + fine) finds the best p.
Optionally a tiny simulated‑annealing hill‑climb refines the result.

The whole routine respects the 1000 s budget and returns the best
sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, score)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score_from_max(n: int, max_b: float, target_sum: float) -> float:
    """Exact benchmark score from the maximal convolution entry."""
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Small local refinement (optional)
# ----------------------------------------------------------------------


def _local_refine(a: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """A few simulated‑annealing moves that improve the maximal convolution."""
    n = a.size
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score_from_max(n, b.max(), target_sum)

    rng = np.random.default_rng()
    start = time.time()
    # number of attempts scales with remaining time
    while time.time() < deadline:
        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)

        # indices that actually contribute to the peak
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # donor – pick proportionally to its contribution
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick a low‑contribution index
        weight = np.zeros(n, dtype=np.float64)
        weight[lo:hi + 1] = prod
        thresh = np.percentile(weight, 30)
        low = np.where(weight <= thresh)[0]
        if low.size:
            receiver = rng.choice(low)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # propose a tiny transfer of mass
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution exactly (O(n))
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = _score_from_max(n, b_new.max(), target_sum)

        # simple SA acceptance
        if new_val < best_val or rng.random() < np.exp(-(new_val - best_val) * 0.1):
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] refined score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The algorithm scans a simple one‑parameter family (power‑law decay) and
    keeps the best sequence; a brief hill‑climbing phase may improve it further.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # keep a small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a finer approximation but costs a bit more)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # the required Σa
    idx = np.arange(n, dtype=np.float64)
    x = idx / (n - 1)                     # normalized positions in [0,1]

    best_score = np.inf
    best_seq = None
    best_p = None

    # --------------------------------------------------------------
    # Phase 1 – coarse grid search for the exponent p
    # --------------------------------------------------------------
    coarse_ps = np.linspace(0.10, 1.00, 91)        # step 0.01
    nfft = 1 << ((2 * n - 1).bit_length())

    for p in coarse_ps:
        # shape (1 - x**p)
        a_raw = 1.0 - np.power(x, p)
        a_raw = np.maximum(a_raw, 0.0)          # numerical safety
        a = a_raw * (target_sum / a_raw.sum())

        # convolution via FFT
        fa = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * n - 1]
        max_b = conv.max()
        score = _score_from_max(n, max_b, target_sum)

        if score < best_score:
            best_score = float(score)
            best_seq = a.copy()
            best_p = p
            print(f"[{time.time():.2f}] coarse p={p:.3f} score={best_score:.6f}")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – fine search around the best exponent
    # --------------------------------------------------------------
    if time.time() < deadline and best_p is not None:
        p_low = max(0.05, best_p - 0.05)
        p_high = best_p + 0.05
        fine_ps = np.linspace(p_low, p_high, 61)   # finer steps

        for p in fine_ps:
            a_raw = 1.0 - np.power(x, p)
            a_raw = np.maximum(a_raw, 0.0)
            a = a_raw * (target_sum / a_raw.sum())

            fa = np.fft.rfft(a, n=nfft)
            conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * n - 1]
            max_b = conv.max()
            score = _score_from_max(n, max_b, target_sum)

            if score < best_score:
                best_score = float(score)
                best_seq = a.copy()
                best_p = p
                print(f"[{time.time():.2f}] fine p={p:.5f} score={best_score:.6f}")

            if time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # Phase 3 – optional tiny local refinement (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        seq_refined, val_refined = _local_refine(best_seq.copy(),
                                                target_sum,
                                                deadline)
        if val_refined < best_score:
            best_score = val_refined
            best_seq = seq_refined
            print(f"[{time.time():.2f}] after refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and evaluation
    # --------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("No sequence was generated – something went wrong.")

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]               # drop trailing zeros

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in later calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
