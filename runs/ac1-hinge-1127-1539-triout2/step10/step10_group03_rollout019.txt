"""
Hybrid optimizer for the self‑convolution bound.

The pipeline works in three stages:

1. **Initialisation** – start from the best known construction (if any) or
   from a simple exponential decay whose rate is chosen heuristically.
2. **Long gradient polish** – a momentum projected‑gradient descent on a
   soft‑max smoothed version of the max‑convolution, gradually raising the
   temperature and finally switching to the exact sub‑gradient of the
   max‑convolution.  The whole remaining time budget is spent here.
3. **Deterministic mass‑transfer refinement** – a short simulated‑annealing
   style local search that moves tiny amounts of mass from “donor’’ indices
   (large contribution to the peak) to “receiver’’ indices (small
   contribution).  Finally trailing zeros are stripped.

All steps respect non‑negativity and the ℓ¹‑simplex constraint
(Σa = √(2 n)).  The routine returns the best sequence found before the
deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score (same formula as ``evaluate_sequence`` without the security checks)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa fixed this equals the max‑convolution)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple families of sequences
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, eps: float, target_sum: float) -> np.ndarray:
    """
    a_i = c·(1‑eps)^i  on i=0…n‑1,
    renormalised so that Σa_i = target_sum.
    """
    r = 1.0 - eps
    # avoid division by zero when eps is extremely tiny
    if eps < 1e-12:
        return np.full(n, target_sum / n, dtype=np.float64)

    r_pow_n = r ** n
    denom = 1.0 - r_pow_n
    c = target_sum * eps / denom
    a = c * (r ** np.arange(n, dtype=np.float64))
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum projected gradient descent (soft‑max -> exact max switch)
# ----------------------------------------------------------------------


def _momentum_optimize(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       max_beta: float = 1e7,
                       step_factor: float = 0.5,
                       gamma: float = 0.99,
                       switch_beta: float = 1e3) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent.
    Starts with a soft‑max approximation of the max‑convolution and, once
    the temperature β exceeds *switch_beta*, switches to the exact
    sub‑gradient of the (non‑smooth) max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Gradient (soft‑max or exact)
        # --------------------------------------------------------------
        if beta < switch_beta:
            g, _ = _gradient_softmax(a, beta)
        else:
            # exact sub‑gradient of the current maximal convolution entry
            conv = _conv_fft(a)
            k = int(np.argmax(conv))
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[k - idx]

        # --------------------------------------------------------------
        # Momentum step
        # --------------------------------------------------------------
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # book‑keeping
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # schedule for β (temperature)
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer with a simple Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # choose a low‑weight receiver among the smallest ``low_frac`` entries
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bound the amount of mass we move
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple re‑ordering that often reduces the peak
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Place the largest value on the left, the second largest on the right,
    the third largest next to the left, etc.
    """
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine spends the whole time budget on a long momentum based
    optimisation that evolves from a soft‑max gradient to the exact
    max‑convolution sub‑gradient, followed by a short deterministic
    mass‑transfer refinement.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Choose a reasonably large problem size (still fast with FFT)
    # --------------------------------------------------------------
    n = 12000 if (deadline - time.time()) > 30.0 else 6000
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build an initial sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.shape[0] != n:
            init_seq = _resize_sequence(init_seq, n)
    else:
        # exponential decay, rate chosen heuristically
        init_seq = _exponential_sequence(n, eps=0.00025, target_sum=target_sum)

    init_seq = _project_to_simplex(init_seq, target_sum)

    # --------------------------------------------------------------
    # Long momentum based optimisation (soft‑max → exact)
    # --------------------------------------------------------------
    a_opt, val_opt = _momentum_optimize(
        init_seq,
        target_sum,
        deadline,
        max_beta=1e7,
        step_factor=0.5,
        gamma=0.99,
        switch_beta=1e2,          # switch to exact gradient relatively early
    )
    best_seq = a_opt
    best_score = val_opt

    # --------------------------------------------------------------
    # Try a simple alternating re‑ordering – often beneficial
    # --------------------------------------------------------------
    alt = _alternating_order(best_seq)
    alt_score = _score(alt)
    if alt_score < best_score:
        best_seq = alt
        best_score = alt_score
        print(f"[{time.time():.2f}] Alternating order improved score to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final deterministic mass‑transfer pass (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.3, end_temp=1e-5, low_frac=0.05
        )
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
            print(f"[{time.time():.2f}] Post‑polish MT improved score to {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Final external evaluation (only for logging)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
