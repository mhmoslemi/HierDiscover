    Tmax = T0

    rng = np.random.default_rng()

"""
Search for a non‑negative step‑function that minimizes the evaluation
function (essentially the maximal entry of the self‑convolution after
normalising the ℓ¹‑norm).  

The pipeline consists of:
1. Initialise a decreasing power‑law sequence and break the symmetry by
   placing the largest coefficients at the left, the smallest at the
   right (alternating order).
2. Run a long simulated‑annealing on tiny mass‑transfer moves
   (donor → receiver) while keeping the ℓ¹‑norm fixed.  The temperature
   is cooled linearly; occasional re‑ordering is performed to escape
   stagnation.
3. Finish with a short sub‑gradient descent (exact max‑convolution
   sub‑gradient) to polish the best candidate.
The whole routine respects the 1000 s budget and stores the final
sequence in the global ``height_sequence_1`` for warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)**(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values (breaks symmetry)."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    turn = True
    for v in sorted_vals:
        if turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        turn = not turn
    return out


# ----------------------------------------------------------------------
# Exact max‑sub‑gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    step_factor: float = 0.5,
    gamma: float = 0.9,
    jitter_interval: int = 500,
    jitter_scale: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Returns the best sequence found and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2 * a[kmax-i] (if inside the support)
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low benchmark score.
    The method uses a power‑law start, symmetry breaking, long SA
    on mass‑transfer moves, and a short sub‑gradient polishing phase.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)   # Σ a_i = √(2 n)

    # --------------------------------------------------------------
    # Initialise a decreasing power‑law and break symmetry
    # --------------------------------------------------------------
    a = _power_law_sequence(n, 0.55, target_sum)
    a = _alternating_order(a)

    # Incorporate a warm‑start if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        # Linear interpolation to desired length
        if prev.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        # try both the old sequence and a blend of old+new
        candidates = [prev, (a + prev) / 2.0]
        for cand in candidates:
            cand = _project_to_simplex(cand, target_sum)
            val = _max_convolution_fft(cand)
            if val < _max_convolution_fft(a):
                a = cand.copy()

    cur_score = _max_convolution_fft(a)
    best_seq = a.copy()
    best_score = cur_score
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing on tiny mass‑transfer moves
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    it = 0
    # linear cooling from Tmax to Tmin
    Tmax = best_score * 0.5 + 1e-2
    Tmin = 1e-6

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = Tmax * (1.0 - frac) + Tmin * frac

        # choose donor proportionally to its mass (larger mass → more likely)
        if a.sum() == 0:
            break
        donor = rng.choice(n, p=a / a.sum())
        # choose a receiver uniformly (different index)
        recv = rng.integers(0, n)
        while recv == donor:
            recv = rng.integers(0, n)

        # move a modest fraction of the donor's mass
        max_move = a[donor] * 0.2
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[recv] += delta

        new_score = _max_convolution_fft(a_new)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] SA iter={it:6d} new best={best_score:.6f}")

        # occasional re‑ordering to break possible periodic traps
        if it % 5000 == 0 and it > 0:
            a = _alternating_order(a)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Re‑order improvement iter={it} score={best_score:.6f}")

        it += 1

    # --------------------------------------------------------------
    # Final polishing with a short sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        polish_deadline = time.time() + (deadline - time.time()) * 0.20
        a_polished, val_polished = _subgradient_descent(
            best_seq.copy(),
            target_sum,
            polish_deadline,
            step_factor=0.8,
            gamma=0.95,
            jitter_interval=800,
            jitter_scale=5e-4,
        )
        if val_polished < best_score:
            best_score = val_polished
            best_seq = a_polished.copy()
            print(f"[{time.time():.2f}] Polishing improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Log the final evaluated score (if the harness supplies evaluate_sequence)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
