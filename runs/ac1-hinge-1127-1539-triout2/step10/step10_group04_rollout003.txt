"""
propose_candidate
-----------------

Search for a non‑negative step function (a list of heights) that minimises
``evaluate_sequence`` :

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three phases:

1. **Exploratory scan** – generate candidates from several cheap
   parametric families (exponential, power‑law, mixed, piecewise,
   Dirichlet and a two‑exponential mixture).  The best few are kept.

2. **Continuous optimisation** – apply a projected Adam optimiser on a
   smooth‑max surrogate (log‑sum‑exp of the convolution).  The optimiser
   respects the ℓ¹‑simplex Σa = √(2 n) and non‑negativity.  Several
   restarts are performed; the best result of this phase is kept.

3. **Local refinement** – a deterministic hill‑climber moves mass from
   the index that contributes most to the current worst convolution entry
   to a more favourable location.  Finally a short simulated‑annealing
   walk is performed to escape possible plateaus.

The search proceeds on a schedule of increasing lengths
(2000 → 4000 → 8000 → 20000).  Time is split dynamically so that the
larger instances receive most of the remaining budget.  All helper
functions are top‑level and contain no closures, satisfying the
execution constraints.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` (without the
    extra clipping performed inside that function)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.size
    m = _max_convolution_fft(a)
    return 2.0 * n * m / (s * s)


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # Full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # Soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # Gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Stops when the wall‑clock reaches ``deadline``.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Learning‑rate base (scaled with the simplex radius)
    base_lr = 0.12 * target_sum

    # Smoothed‑max schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # Tiny jitter every few hundred steps
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # Increase smoothing slowly
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)

    # pick a family at random
    family = rng.integers(0, 7)

    if family == 0:                     # power‑law
        p = rng.uniform(0.1, 1.5)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # piecewise exponential
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 4:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 5:                   # near‑uniform with jitter
        a = np.full(n, target / n)

    else:                               # two‑exponential mixture
        lam1 = rng.exponential(scale=0.015)
        lam2 = rng.exponential(scale=0.015)
        w = rng.uniform(0.2, 0.8)
        a = w * np.exp(-lam1 * idx) + (1.0 - w) * np.exp(-lam2 * idx)

    # small jitter + projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length and projection onto the simplex."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


def _family_candidates(n: int, rng, top_k: int = 8):
    """
    Scan several simple families (exponential, power‑law, mixed,
    piecewise, Dirichlet, two‑exponential mixture) and keep the best
    ``top_k`` candidates.
    Returns a list of (score, vector) pairs.
    """
    target = math.sqrt(2.0 * n)
    candidates = []

    # -------------------------- exponential ---------------------------
    for _ in range(30):
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # -------------------------- power‑law -----------------------------
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # ---------------------- mixed power‑law × exponential -------------
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p) * np.exp(-lam * np.arange(n))
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # -------------------------- piecewise -----------------------------
    for _ in range(15):
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # --------------------------- Dirichlet ----------------------------
    for _ in range(20):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # -------------- two‑exponential mixture (random weights) ----------
    for _ in range(20):
        lam1 = rng.exponential(scale=0.015)
        lam2 = rng.exponential(scale=0.015)
        w = rng.uniform(0.2, 0.8)
        a = w * np.exp(-lam1 * np.arange(n)) + (1.0 - w) * np.exp(-lam2 * np.arange(n))
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # keep the best ``top_k``
    candidates.sort(key=itemgetter(0))
    return candidates[:top_k]

# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution
    entry.  Moves a (large) fraction of the responsible mass to the
    index that contributes the least to that entry.  Returns the best
    vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        # source indices influencing b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart = a[M - j_range]

        # choose destination with minimal counterpart (different from i)
        order = np.argsort(counterpart)
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # amount of mass to move (up to 80 % of source)
        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # evaluate the move (linear update of convolution)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # accept if improvement
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # try a smaller move (binary search)
        improved = False
        delta2 = delta * 0.5
        while delta2 > 1e-14:
            b_test = b.copy()
            b_test[i:i + n] -= 2.0 * delta2 * a
            b_test[j:j + n] += 2.0 * delta2 * a
            if k2i < b_test.size:
                b_test[k2i] += delta2 * delta2
            if k2j < b_test.size:
                b_test[k2j] += delta2 * delta2
            if kij < b_test.size:
                b_test[kij] -= 2.0 * delta2 * delta2
            if b_test.max() < b.max() - 1e-12:
                a[i] -= delta2
                a[j] += delta2
                b = b_test
                if b.max() < best_score - 1e-12:
                    best_score = b.max()
                    best_a = a.copy()
                improved = True
                break
            delta2 *= 0.5

        if not improved:
            # add a tiny jitter and recompute convolution
            jitter = rng.normal(scale=1e-6, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            target = math.sqrt(2.0 * n)
            a = _project_to_simplex(a, target)
            b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (mass transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, target_sum: float,
                         deadline: float, init_temp: float = 0.1,
                         cooling: float = 0.999) -> np.ndarray:
    """Simple SA that moves a fraction of mass between two random indices."""
    a = a0.copy()
    cur_score = _objective(a)
    temp = init_temp
    rng = np.random.default_rng()

    while time.time() < deadline:
        # propose a transfer
        i = rng.integers(0, a.size)
        j = rng.integers(0, a.size)
        if i == j or a[i] < 1e-12:
            continue
        frac = rng.uniform(0.02, 0.15)            # transfer fraction
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        cand_score = _objective(cand)

        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / temp):
            a = cand
            cur_score = cand_score

        # geometric cooling
        temp *= cooling
        # keep temperature from collapsing to zero too fast
        if temp < 1e-8:
            temp = 1e-8

    return a

        # 1b)  Ensemble averaging of the best few candidates
        # --------------------------------------------------------------
        ensemble = np.mean([vec for _s, vec in top_candidates], axis=0)
        ensemble = _project_to_simplex(ensemble, target_sum)
        ens_score = _objective(ensemble)
        if ens_score < best_score - 1e-12:
            best_seq = ensemble.copy()
            best_score = ens_score
            print(f"[{time.time():.1f}] Ensemble improvement (n={n}) obj={best_score:.6f}")

        # --------------------------------------------------------------
        # 1c)  Adam optimisation – several restarts
        # --------------------------------------------------------------
        remain = deadline - time.time()
        if remain < 5.0:
            break
        # allocate ~55 % of remaining budget to Adam
        adam_budget = remain * 0.55
        per_restart = max(5.0, adam_budget / 8)   # at most 8 restarts
        n_restarts = int(adam_budget // per_restart)
        n_restarts = max(1, n_restarts)

        local_best = None
        local_score = float("inf")

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            restart_deadline = min(deadline - 0.2, time.time() + per_restart)
            a_opt = _adam_optimize(start_a, target_sum, restart_deadline)
            obj = _objective(a_opt)
            if obj < local_score:
                local_score = obj
                local_best = a_opt.copy()

        if local_best is not None and local_score < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_score
            print(f"[{time.time():.1f}] Adam improvement (n={n}) obj={best_score:.6f}")

        # --------------------------------------------------------------
        # 1d)  Deterministic hill‑climb (exact max)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + (remain * 0.35))
            a_hc, hc_val = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            hc_obj = 2.0 * n * hc_val / (target_sum * target_sum)
            if hc_obj < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = hc_obj
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) obj={best_score:.6f}")

        # --------------------------------------------------------------
        # 1e)  Simulated annealing (mass‑transfer moves)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            sa_deadline = deadline - 0.5
            a_sa = _simulated_annealing(best_seq, target_sum, sa_deadline,
                                        init_temp=0.05, cooling=0.9995)
            sa_obj = _objective(a_sa)
            if sa_obj < best_score - 1e-12:
                best_seq = a_sa.copy()
                best_score = sa_obj
                print(f"[{time.time():.1f}] Sim‑anneal improvement (n={n}) obj={best_score:.6f}")

        # --------------------------------------------------------------
        # 1f)  Tiny random perturbations (local tweaks)
        # --------------------------------------------------------------
        while time.time() + 1.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.07
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target_sum)
            cand_obj = _objective(cand)
            if cand_obj < best_score - 1e-12:
                best_seq = cand
                best_score = cand_obj
                print(f"[{time.time():.1f}] Random tweak improvement (n={n}) obj={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing on the remaining budget (if any)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # final Adam run
        final_deadline = deadline - 1.0
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        # final hill‑climb
        b_final = _full_convolution(best_seq)
        hc_deadline = deadline - 0.5
        best_seq, final_val = _hill_climb_deterministic(best_seq, b_final, hc_deadline)
        final_obj = 2.0 * n * final_val / (target * target)
        if final_obj < best_score - 1e-12:
            best_score = final_obj
            print(f"[{time.time():.1f}] Final polishing improvement obj={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform on a modest size
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Report the official evaluation (for debugging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
