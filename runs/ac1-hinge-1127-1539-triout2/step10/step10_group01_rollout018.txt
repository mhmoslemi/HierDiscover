    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    step = target_sum * 0.5  # initial step relative to sum amount
    decay = 0.9995
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline - 1.0:
        # compute gradient at current max index(s)
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros_like(a)
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idxs = np.arange(lo, hi + 1)
            grad[idxs] += a[k - idxs]  # contribution as a_{k-i}
        grad *= 2.0 / len(idx_max)

"""
A fast multi‑phase optimizer for the step‑function benchmark.

The algorithm proceeds as follows

1.  Choose a problem size n (bigger n gives a finer discretisation).
2.  Initialise the sequence from the previous best if it exists,
    otherwise use a simple power‑law.
3.  Explore a dense family of parametric shapes (pure power‑law,
    power‑law with exponential tail and pure exponential) and keep the
    best candidate.
4.  Perform a large number of sub‑gradient steps on the exact max‑convolution
    (the objective is convex).  The sub‑gradient is the (averaged) gradient
    of the active convolution entries.  After each step the vector is
    projected back onto the simplex {a_i≥0, Σa_i = √(2n)}.
5.  Apply a deterministic “mass‑transfer” sweep that reduces the current peak
    to the second‑largest value in a single exact move.
6.  Run a short random‑move phase and an Adam‑style soft‑max gradient‑descent
    polish if time permits.
7.  Trim trailing zeros and return the best sequence found.

All phases respect the global 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) scaled to have Σ = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to Σ = target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """pure exponential exp(-lam*i) scaled to Σ = target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by the Adam‑style optimizer)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style projected gradient descent on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short Adam‑GD runs with random jitter; keep the best result."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy mass transfers that directly reduce the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[k0 - idx]

        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            break

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.5, delta_needed * 0.99)

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing moves that shift a small amount of mass."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (new phase)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    step = 0.5          # initial step size (absolute)
    decay = 0.9999      # geometric decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        grad = np.zeros_like(a)
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idxs = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idxs] += a[k - idxs]
        grad *= 2.0 / len(idx_max)

        a_new = a - step * grad
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.6f} score={best_val:.6f}")

        a = a_new
        step *= decay
        it += 1

        # occasional small jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / n, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper: initialise from previous solution or a simple power‑law
# ----------------------------------------------------------------------


def _initial_sequence(n: int, target_sum: float) -> np.ndarray:
    """Return a starting sequence of length n."""
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size != n:
            old_len = seq.shape[0]
            old_x = np.linspace(0.0, 1.0, old_len)
            new_x = np.linspace(0.0, 1.0, n)
            seq = np.interp(new_x, old_x, seq)
        seq = _project_to_simplex(seq, target_sum)
    else:
        seq = _power_law_sequence(n, 0.6, target_sum)
    return seq


# ----------------------------------------------------------------------
# Parametric pre‑search (slightly richer than the baseline)
# ----------------------------------------------------------------------


def _parametric_search(n: int,
                       target_sum: float,
                       best_seq: np.ndarray,
                       best_score: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Explore several parametric families and keep the best candidate."""
    # power‑law grid
    p_grid = np.linspace(0.2, 1.2, 30)
    for p in p_grid:
        if time.time() >= deadline:
            break
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # power‑law + exponential tail grid
    q_grid = np.linspace(0.0, 8.0, 30)
    for p in p_grid:
        for q in q_grid:
            if time.time() >= deadline:
                break
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # pure exponential grid
    lam_grid = np.linspace(0.0, 1.0, 20)
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = _exp_sequence(n, float(lam), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # jittered variants of the current best (small population)
    rng = np.random.default_rng()
    for _ in range(6):
        if time.time() >= deadline:
            break
        jitter = rng.normal(scale=target_sum * 0.005 / n, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score, best_seq = val, seq_j.copy()

    print(f"[{time.time():.2f}] Parametric search best score = {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns the best sequence found before the time budget expires.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------------
    # Choose problem size (larger n gives a finer discretisation)
    # ---------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scale‑invariant)

    # ---------------------------------------------------------------
    # Phase 0 – initialise and parametric pre‑search
    # ---------------------------------------------------------------
    best_seq = _initial_sequence(n, target_sum)
    best_score = _score(best_seq)

    # Allocate ~10 % of the time budget for the cheap parametric sweep
    phase_end = min(deadline, time.time() + max(1.0, budget_s * 0.10))
    best_seq, best_score = _parametric_search(n, target_sum,
                                              best_seq, best_score,
                                              phase_end)

    # ---------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (many cheap iterations)
    # ---------------------------------------------------------------
    subgrad_end = deadline - max(30.0, budget_s * 0.30)   # keep some time for later phases
    if time.time() < subgrad_end:
        best_seq, best_score = _subgradient_descent(best_seq,
                                                    target_sum,
                                                    subgrad_end)
        print(f"[{time.time():.2f}] After sub‑gradient phase best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – deterministic exact mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact(best_seq,
                                                            deadline,
                                                            target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT improved best = {best_score:.6f}")

        # short random‑move phase after each sweep
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 8.0:
            break

    # ---------------------------------------------------------------
    # Phase 3 – final polishing with Adam‑style soft‑max GD (if time)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e7,
                                                step_factor=0.5,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing zeros, return as Python list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
