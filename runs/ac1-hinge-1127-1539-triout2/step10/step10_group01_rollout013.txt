def _subgradient_descent(a0, target_sum, deadline, max_iter=2000, step_scale=1.0):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        n = a.size
        # compute convolution via FFT
        size = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2*n - 1]

        max_val = conv.max()
        # find second max
        mask = conv == max_val
        # due to floating errors use tolerance
        tol = 1e-12
        mask = conv >= max_val - tol

        # compute second max
        conv2 = conv.copy()
        conv2[mask] = -np.inf
        second_max = conv2.max()

        # compute gradient only for max indices
        m = mask.astype(np.float64)

        # compute gradient via FFT (convolution of m with rev_a)
        rev_a = a[::-1]
        size2 = 1 << ((3*n - 2).bit_length())
        M_fft = np.fft.rfft(m, n=size2)
        rev_fft = np.fft.rfft(rev_a, n=size2)
        grad_full = np.fft.irfft(M_fft * rev_fft, n=size2)[: 3*n - 2]
        grad = 2.0 * grad_full[n-1:n-1+n]

        # step size based on difference
        diff = max_val - second_max
        if diff < 1e-12:
            # no more improvement
            break
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq == 0:
            break

        # compute step as diff / grad_norm_sq scaled
        step = step_scale * diff / grad_norm_sq
        # optionally clamp step to not exceed a_max
        step = min(step, a.max() * 0.5)  # don't move too much

        a = a - step * grad
        # ensure non-negative and project onto simplex
        a = _project_to_simplex(a, target_sum)

        # evaluate

        if val < best_val:
            best_val = val
            best_a = a.copy()
            #print(f"[{time.time():.2f}] Subgrad it={it} diff={diff:.6f} score={best_val:.6f}")

        it += 1
    return best_a, best_val

"""
Sub‑gradient / deterministic mass‑transfer optimizer for the step‑function
benchmark.

The algorithm builds a large candidate (n≈12000) and then iteratively
reduces the quantity

    score = 2·n·max(conv(a,a)) / (sum(a)²)          (sum(a) is kept = √(2n))

by three complementary phases:

1. **Parametric pre‑search** – a dense grid of power‑law / exponential
   families together with cheap random mixtures.  The best candidate from
   this stage becomes the starting point.

2. **Projected sub‑gradient descent** – the true sub‑gradient of the
   maximum auto‑convolution (not the soft‑max approximation) is computed
   via FFT, a step size derived from the current “peak‑gap” is taken,
   and the iterate is projected onto the simplex {x≥0, Σx = √(2n)}.
   Multiple restarts with jitter explore the landscape.

3. **Exact deterministic mass‑transfer (DMT)** – for the current maximal
   convolution index the amount of mass to move from the most harmful
   coordinate to the least harmful one is computed analytically so that
   the peak is driven down to the second‑largest value.  Repeating DMT
   after each sub‑gradient phase tightens the bound.

All phases respect the global time budget; after trimming trailing zeros
the best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the fast pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and scale to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000,
                         step_scale: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(a*a) with projection onto the simplex.
    The step size is diff / ‖grad‖² (diff = current‑peak – second‑peak)
    multiplied by *step_scale*.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    n = a.size
    # pre‑compute FFT size for the gradient convolution (length 3n‑2)
    grad_fft_size = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline and it < max_iter:
        # ----- convolution (FFT) -----
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_val = conv.max()
        # mask of (almost) maximal entries
        tol = 1e-12
        mask = conv >= max_val - tol

        # second largest value (ignoring the current maxima)
        conv2 = conv.copy()
        conv2[mask] = -np.inf
        second_max = conv2.max()

        # ----- sub‑gradient of the max (via FFT) -----
        m = mask.astype(np.float64)
        rev_a = a[::-1]

        M_fft = np.fft.rfft(m, n=grad_fft_size)
        rev_fft = np.fft.rfft(rev_a, n=grad_fft_size)
        grad_full = np.fft.irfft(M_fft * rev_fft, n=grad_fft_size)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ----- step size -----
        diff = max_val - second_max
        if diff <= 1e-15:
            break
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq == 0.0:
            break
        step = step_scale * diff / grad_norm_sq
        # protect against too large steps
        step = min(step, a.max() * 0.5)

        # ----- update & projection -----
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        # ----- bookkeeping -----
        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            # uncomment the next line for verbose progress
            # print(f"[{time.time():.2f}] Subgrad it={it:5d} diff={diff:.6e} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly reduce the current maximal convolution entry.
    For the chosen maximal index k, the algorithm computes the exact amount
    of mass to transfer from the most harmful position to the least harmful
    one so that the peak becomes (approximately) the second‑largest convolution
    value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)                     # full convolution
        max_val = conv.max()
        # indices achieving the maximal value (within tolerance)
        idx_max = np.where(conv >= max_val - tol)[0]

        # choose one of the maximal indices (rotate if needed)
        k0 = idx_max[it % len(idx_max)]

        # range of i that contributes to conv[k0]
        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # heuristic weight w_i = a_i + a_{k0-i}
        w = a[idx] + a[k0 - idx]

        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contributions of donor / receiver to the maximal entry
        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            # cannot improve this peak – try next maximal index
            if len(idx_max) > 1:
                k0 = idx_max[(it + 1) % len(idx_max)]
                lo = max(0, k0 - (n - 1))
                hi = min(k0, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                w = a[idx] + a[k0 - idx]
                donor = idx[np.argmax(w)]
                receiver = idx[np.argmin(w)]
                contrib_d = a[k0 - donor]
                contrib_r = a[k0 - receiver]
                denom = 2.0 * (contrib_d - contrib_r)
                if denom <= 0.0:
                    break
            else:
                break

        # second largest convolution value (excluding current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # exact delta needed to bring the peak down to second_max
        delta_needed = (max_val - second_max) / denom
        # safety: never move more than half of the donor mass
        max_delta = min(a[donor] * 0.5, delta_needed * 0.99)

        if max_delta <= 0.0:
            break

        # perform the transfer
        a[donor] -= max_delta
        a[receiver] += max_delta
        # sum is preserved, non‑negativity guaranteed by the bound

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # uncomment for verbose progress
            # print(f"[{time.time():.2f}] Exact DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing moves that shift a small amount of mass
    from the largest entries to the smallest ones.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # uncomment for verbose progress
            # print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the global time budget, returns the best sequence
    found, and prints occasional progress information.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ---------------------------------------------------------------
    # Choose problem size based on the allowed budget
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa is kept fixed (scale‑invariant)

    rng = np.random.default_rng()

    best_seq = None
    best_score = np.inf

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search + random mixtures
    # ---------------------------------------------------------------
    # (a) pure power‑law grid
    p_grid = np.linspace(0.30, 1.00, 25)
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail grid
    q_grid = np.linspace(0.0, 8.0, 25)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) pure exponential families
    for _ in range(30):
        alpha = rng.uniform(0.0, 8.0)
        seq = np.exp(-alpha * np.arange(n, dtype=np.float64))
        seq *= target_sum / seq.sum()
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (d) random mixtures of two exponentials
    for _ in range(30):
        alpha = rng.uniform(0.0, 8.0)
        beta = rng.uniform(0.0, 8.0)
        w = rng.random()
        seq = w * np.exp(-alpha * np.arange(n, dtype=np.float64))
        seq += (1 - w) * np.exp(-beta * np.arange(n, dtype=np.float64))
        seq *= target_sum / seq.sum()
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (e) previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_len = prev.shape[0]
            old_x = np.linspace(0.0, 1.0, old_len)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()

    # (f) jittered variants of the current best (small population)
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Projected sub‑gradient descent with restarts
    # ---------------------------------------------------------------
    while time.time() < deadline - 12.0:
        # start from a jittered version of the current best
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        sub_deadline = time.time() + min(12.0, deadline - time.time() - 4.0)

                                              target_sum,

        if val_opt < best_score:
            best_score = val_opt

            print(f"[{time.time():.2f}] Sub‑grad improved best = {best_score:.6f}")

                                                                deadline,

            if val_ref < best_score:
