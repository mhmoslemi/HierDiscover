def isotonic_decreasing(v):
    # returns monotone nonincreasing sequence that is L2 projection onto monotone cone.
    n=v.shape[0]
    v = v.astype(np.float64)
    # Use PAVA for decreasing: we can run on reversed array with increasing, then reverse back.
    # Standard PAVA for nondecreasing: maintain blocks.
    # For decreasing, we can apply PAVA to -v with nondecreasing constraint, then take -result.
    # so:
    rev = -v
    # pava increasing:
    solution = np.empty_like(rev)
    sum_block = []
    count_block = []
    for i, x in enumerate(rev):
        sum_block.append(x)
        count_block.append(1)
        # merge if needed
        while len(sum_block) >= 2 and sum_block[-2]/count_block[-2] > sum_block[-1]/count_block[-1]:
            # merge last two blocks
            s = sum_block[-2] + sum_block[-1]
            c = count_block[-2] + count_block[-1]
            sum_block[-2] = s
            count_block[-2] = c
            sum_block.pop()
            count_block.pop()
    # assign averages to solution
    idx = 0
    for s,c in zip(sum_block, count_block):
        avg = s/c
        solution[idx:idx+c] = avg
        idx += c
    # revert sign
    return -solution

"""
Propose a low‑score step function.

The algorithm builds on the previous multi‑stage pipeline but adds:

* a multiscale schedule for the problem size,
* a dedicated isotonic (monotone‑decreasing) projection,
* a custom sub‑gradient descent with simple line‑search,
* more aggressive time splitting and several quick restarts,
* final polishing with a short simulated‑annealing run.

All heavy work uses FFT‑based convolutions, and every update keeps the
ℓ¹‑norm fixed to √(2·n) by projection onto the simplex.  The routine
returns the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / scoring utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power of two length sufficient for a linear convolution of size n."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(sum(a)²); assumes sum(a) > 0."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)⁻ᵖ, then scaled to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (used by exponential / GD stages)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) convolve p) restricted to central n entries
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Isotonic regression (monotone decreasing) using PAVA
# ----------------------------------------------------------------------


def _isotonic_decreasing(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the cone of non‑increasing sequences.
    Implements the pool‑adjacent‑violators algorithm (PAVA) on -v.
    """
    # work on the reversed sign to obtain decreasing constraint via increasing PAVA
    x = -v.astype(np.float64)
    n = x.size
    # blocks
    sums = []
    counts = []
    for val in x:
        sums.append(val)
        counts.append(1)
        # merge while the block averages are decreasing (violates increasing constraint)
        while len(sums) >= 2 and sums[-2] / counts[-2] > sums[-1] / counts[-1]:
            s = sums[-2] + sums[-1]
            c = counts[-2] + counts[-1]
            sums[-2] = s
            counts[-2] = c
            sums.pop()
            counts.pop()
    # expand the averaged blocks
    result = np.empty_like(x)
    idx = 0
    for s, c in zip(sums, counts):
        avg = s / c
        result[idx: idx + c] = avg
        idx += c
    # revert sign
    return -result

def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,

    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

def _gradient_descent_with_restarts(init: np.ndarray,

def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from high‑contributing indices to low‑contributing ones."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renorm for numerical drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Cheap SA that moves small mass from high‑weight to low‑weight indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.015 * best_val  # initial temperature

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        wsum = weight.sum()
        if wsum == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved best score={best_val:.6f}")

        if rng.random() < 0.004:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size according to the time budget
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    elif budget_s >= 200:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse previous run if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # --------------------------------------------------------------
    small_n = max(2000, n // 5)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.08)
    seq_small, _ = _grid_param_search(small_n, small_target, param_deadline)

    if seq_small is not None:
        cand = _resize_sequence(seq_small, n)
        cand = cand * (target_sum / cand.sum())
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After param search score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + min(150.0, (deadline - time.time()) * 0.18)
        a_exp, sc_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if sc_exp < best_score:
            best_score = sc_exp
            best_seq = a_exp.copy()
            print(f"[{time.time():.2f}] After ExpGD score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts (soft‑max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + min(250.0, (deadline - time.time()) * 0.30)
        a_gd, sc_gd = _gradient_descent_with_restarts(best_seq,
                                                      target_sum,
                                                      gd_deadline)
        if sc_gd < best_score:
            best_score = sc_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After GD score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + min(150.0, (deadline - time.time()) * 0.10)
        a_mt, sc_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if sc_mt < best_score:
            best_score = sc_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After mass‑transfer score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – custom sub‑gradient descent (max convolution) with line‑search
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + min(200.0, (deadline - time.time()) * 0.20)
        a_sub, sc_sub = _subgradient_max_descent_custom(best_seq,
                                                        target_sum,
                                                        sub_deadline)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] After sub‑gradient score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 6 – isotonic (monotone decreasing) projection
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_iso = _isotonic_decreasing(best_seq)
        a_iso = a_iso * (target_sum / a_iso.sum())
        sc_iso = _score(a_iso)
        if sc_iso < best_score:
            best_score = sc_iso
            best_seq = a_iso.copy()
            print(f"[{time.time():.2f}] After isotonic projection score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 7 – simulated‑annealing fine‑tuning (uses remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sa, sc_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if sc_sa < best_score:
            best_score = sc_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] After final SA score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing (short sub‑gradient pass) if a few seconds remain
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_fin, sc_fin = _subgradient_max_descent_custom(best_seq,
                                                        target_sum,
                                                        deadline - 5.0)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] After final polish score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean up, trim trailing zeros and store for possible warm‑starts
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store globally for next invocation
    globals()["height_sequence_1"] = final_seq

    # Final verification (may be slower, but executed only once)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
