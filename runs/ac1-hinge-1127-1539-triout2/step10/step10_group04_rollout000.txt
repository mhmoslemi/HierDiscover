"""
Propose a non‑negative step function that minimises
``evaluate_sequence``.

Algorithm
---------

* **Length** – we concentrate on a single large dimension (≈20 000)
  where the optimisation landscape is richer.
* **Random families** – a cheap scan of several parametric families
  (exponential, power‑law, mixed, Dirichlet) is performed, each vector
  is projected onto the ℓ¹‑simplex Σa = √(2 n) and the best one (by the
  exact max‑convolution) is kept as a seed.
* **Projected Adam** – we minimise a smooth approximation of the
  ``max`` using the log‑sum‑exp surrogate.  The gradient is obtained
  analytically via FFTs, the update is Adam‑style and the iterate is
  projected back onto the simplex after every step.  The smoothing
  parameter β is increased geometrically so that the optimisation first
  moves quickly and then focuses on the true max.
* **Deterministic hill‑climb** – after Adam finishes we perform a
  greedy move that transfers mass from the index that contributes most
  to the current worst convolution entry to a more favourable location.
  The move is exact (the convolution is updated in O(n) using the
  linear‑algebraic formulas from the previous state‑of‑the‑art code).
* **Polishing** – a short extra Adam run followed by another hill‑climb
  is done if time permits.

All steps respect a global wall‑clock budget (default 1000 s).  The
final sequence is trimmed of trailing near‑zeros before being returned.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth‑max (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until ``deadline``.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.12 * target_sum

    # smoothing schedule
    beta = 2.0          # start very smooth
    beta_max = 5e7
    beta_factor = 1.9   # increase factor per schedule block

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat plateaus
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 2e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# ----------------------------------------------------------------------


def _random_family(n: int, rng) -> np.ndarray:
    """Sample a vector from a randomly chosen parametric family."""
    target = math.sqrt(2.0 * n)
    fam = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if fam == 0:          # exponential decay
        lam = rng.exponential(scale=0.012)
        a = np.exp(-lam * idx)

    elif fam == 1:        # power‑law
        p = rng.uniform(0.12, 1.3)
        a = (idx + 1.0) ** (-p)

    elif fam == 2:        # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.3)
        lam = rng.exponential(scale=0.012)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif fam == 3:        # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    else:                 # near‑uniform with tiny jitter
        a = np.full(n, 1.0)

    a = a.astype(float)
    a = _project_to_simplex(a, target)
    return a


def _family_scan(n: int, rng, trials: int = 60):
    """Return the best vector among a few random families."""
    target = math.sqrt(2.0 * n)
    best_a = None
    best_score = np.inf

    # include the pure uniform vector (often a decent baseline)
    uniform = np.full(n, target / n)
    uniform_score = _max_convolution_fft(uniform)
    best_a = uniform
    best_score = uniform_score

    for _ in range(trials):
        a = _random_family(n, rng)
        score = _max_convolution_fft(a)
        if score < best_score:
            best_score = score
            best_a = a

    return best_a, best_score

# Deterministic hill‑climb (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))                      # worst entry
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]               # contribution of each source

        i = idx[np.argmax(contrib)]                 # source with largest contribution
        if a[i] < 1e-14:
            break

        # destination that hurts the least (small counterpart)
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]
        cand_range = np.arange(i_low, i_high + 1, dtype=int)
        order = np.argsort(counterpart[cand_range])
        j = None
        for c in cand_range[order]:
            if c != i:
                j = c
                break
        if j is None:
            break

        delta = a[i] * 0.7  # move 70 % of the source mass
        if delta <= 0.0:
            break

        # linear‑time update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-13:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-13:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-15:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-13:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-13:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small random jitter to escape possible plateaus
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

            print(f"[{time.time():.1f}] Extra Adam polished score = {best_score:.6f}")

        if time.time() + 1.5 < deadline:
            b_cur = _full_convolution(best_seq)
            a_hc, hc_score = _hill_climb_deterministic(best_seq, b_cur, deadline - 0.2)
            if hc_score < best_score - 1e-13:
                best_seq = a_hc
                best_score = hc_score
                print(f"[{time.time():.1f}] Final hill‑climb polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6)  Trim trailing near‑zeros and report official evaluation
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
