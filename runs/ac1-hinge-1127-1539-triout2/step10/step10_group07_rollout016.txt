The optimiser works in three stages.

1. **Exploratory search on a small problem (≈ ¼ of the time budget)**  
   – Random Dirichlet vectors (different concentration parameters) are generated, projected onto the required simplex and improved by a *fixed‑iteration* sub‑gradient descent.  
   – The best small‑size sequence is kept and later interpolated to the final dimension.

2. **Long optimisation on the full size (≈ ½ of the remaining time)**  
   – A projected **Adam** optimiser minimises the true max‑convolution (the sub‑gradient of the ℓ∞‑norm).  
   – The step size follows a 1/√t schedule, the iterates are always projected back onto the simplex.

3. **Aggressive local refinement (the rest of the budget)**  
   – Mass‑transfer simulated‑annealing moves a (larger) amount of mass from a “bad’’ index (high contribution to the current maximal convolution entry) to a “good’’ index (low contribution).  
   – The temperature schedule is tuned for a more exploratory search.

All helper routines are implemented with FFT‑based convolutions ( O(n log n) ) and a Euclidean projection onto the simplex
\(\{x\ge 0,\;\sum x = \sqrt{2n}\}\), which makes the evaluation function equal to `max(a*a)`.

The code respects the 1000 s time limit, prints progress, stores the best sequence globally for warm‑starts and finally returns the best list of heights.

```python
"""
Optimise a non‑negative step function for the objective

    2·n·max(a*a) / (∑a)²

(the evaluation function returns exactly this value).  The algorithm
consists of

1. a fast exploratory search on a small dimension (random Dirichlet
   starts + a few sub‑gradient steps);
2. a long Adam‑based projected sub‑gradient descent on the final
   dimension;
3. an aggressive simulated‑annealing mass‑transfer refinement.

All steps obey the simplex constraint  ∑a = √(2 n)  (the scale is
fixed because the objective is scale‑invariant).  The routine runs for
the supplied time budget (≤ 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective: 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) the return equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Fixed‑iteration sub‑gradient descent (used in the exploratory phase)
# ----------------------------------------------------------------------


def _subgradient_descent_fixed(a0: np.ndarray,
                               target_sum: float,
                               max_iter: int = 300) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent for a fixed number of iterations.
    Returns the best sequence encountered and its (max‑convolution) score.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    base_step = target_sum * 0.5

    for it in range(max_iter):
        b = _conv_fft(a)
        max_b = b.max()
        eps = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps).astype(np.float64)

        # sub‑gradient of the max entry (same as in the original code)
        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)   # length 3n‑2
        L = len(mask) - 1                      # = 2n‑2
        g = conv_full[L - np.arange(a.shape[0])]

        step = base_step / np.sqrt(it + 1.0)
        gn = np.linalg.norm(g)
        if gn > 0:
            a -= step * g / gn

        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if it % 100 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape[0])
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        if max_b < best_val:
            best_val = max_b
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam‑based projected sub‑gradient descent (main long phase)
# ----------------------------------------------------------------------


def _adam_subgradient_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the true max‑convolution.
    The iterates stay on the simplex Σa = target_sum.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.astype(np.float64), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_step = target_sum * 0.5

    while time.time() < deadline:
        b = _conv_fft(a)
        max_b = b.max()
        eps_mask = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps_mask).astype(np.float64)

        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)
        L = len(mask) - 1
        g = conv_full[L - np.arange(a.shape[0])]

        # Adam moments
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_step / np.sqrt(it)          # decreasing magnitude
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to keep the dynamics alive
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape[0])
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        if max_b < best_val:
            best_val = max_b
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer simulated‑annealing refinement
# ----------------------------------------------------------------------


def _local_refine_aggressive(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that moves a relatively larger
    amount of mass from a “bad’’ index (large contribution to the current
    maximal convolution entry) to a “good’’ one.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.06               # a slightly larger initial temperature
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))               # index of the current max entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to the maximal entry
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with a *relatively* small contribution
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)          # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a (larger) transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.10, target_sum * 0.002)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # Update the full convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part  -2*a*Δ on donor side, +2*a*Δ on receiver side
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic full recomputation & re‑normalisation
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional extra jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Problem sizes
    # --------------------------------------------------------------
    if budget_s >= 800:
        large_n = 12000
    elif budget_s >= 500:
        large_n = 8000
    else:
        large_n = 5000
    # a much smaller size for the exploratory stage
    small_n = max(2000, large_n // 4)

    large_sum = np.sqrt(2.0 * large_n)
    small_sum = np.sqrt(2.0 * small_n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 0 – exploratory search on a small problem
    # --------------------------------------------------------------
    small_deadline = start + 0.25 * (deadline - start)
    best_small = None
    best_small_val = np.inf
    while time.time() < small_deadline:
        # random Dirichlet concentration between 0.3 and 2.0
        alpha = rng.uniform(0.3, 2.0)
        a0 = rng.dirichlet(np.full(small_n, alpha)) * small_sum
        a_opt, val = _subgradient_descent_fixed(a0, small_sum, max_iter=300)
        if val < best_small_val:
            best_small_val = val
            best_small = a_opt.copy()
            print(f"[{time.time():.2f}] Small stage new best val={best_small_val:.6f}")

    if best_small is None:
        # fallback – simple decreasing power‑law
        best_small = (np.arange(1, small_n + 1, dtype=np.float64) ** (-0.55))
        best_small = _project_to_simplex(best_small, small_sum)

    # --------------------------------------------------------------
    # Upscale to the final dimension
    # --------------------------------------------------------------
    a_large = _resize_sequence(best_small, large_n)
    a_large = _project_to_simplex(a_large, large_sum)

    # --------------------------------------------------------------
    # Phase 1 – long Adam based projected sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    sub_deadline = time.time() + 0.55 * remaining
    a_sub, val_sub = _adam_subgradient_descent(a_large, large_sum, sub_deadline)
    best_seq = a_sub
    best_score = val_sub
    print(f"[{time.time():.2f}] After Adam subgradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – short random restarts of sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        rest_deadline = time.time() + 0.20 * remaining
        while time.time() < rest_deadline - 2.0:
            jitter = rng.normal(scale=large_sum * 0.02, size=large_n)
            cand = best_seq + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, large_sum)
            a_tmp, val_tmp = _subgradient_descent_fixed(cand, large_sum, max_iter=200)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – aggressive local refinement (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine_aggressive(best_seq, large_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
```
