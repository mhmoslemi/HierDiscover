def _adam_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0
    max_beta = 65536.0
    # schedule for beta similar to earlier: double every certain it.

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # Adam step size, start with alpha = 0.01 maybe scaling to target_sum
        alpha = 0.02 * target_sum  # adjust
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        # Update a multiplicatively
        a = a * np.exp(-step)
        a = a * (target_sum / a.sum())

        # occasional jitter + sigma scaling similar to earlier
        if t % 300 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0)
            a = a * (target_sum / a.sum())

        # compute score
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t} β={beta:.1f} score={best_val:.6f}")
        # schedule beta increase
        if t % 400 == 0 and beta < max_beta:
            beta = min(beta*2.0, max_beta)

    return best_a, best_val

def _simulated_annealing(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature as fraction of best_val, maybe 0.3 * best_val
    T0 = best_val * 0.15
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # linear temperature cooling
        T = T0 * max(0.0, 1.0 - elapsed / total)

        # donor and receiver selection as before
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi+1)

        # choose receiver from low product indices or low value to limit max increase
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi+1] = prod
        # compute threshold based on low percentiles of weight_all
        thresh = np.percentile(weight_all[weight_all > 0], 30) if np.any(weight_all > 0) else 0.0
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # delta
        max_delta = min(a[donor] * 0.07, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # compute b_new incrementally
        b_new = b.copy()
        b_new[donor:donor+n] += -2.0 * delta * a
        b_new[receiver:receiver+n] += 2.0 * delta * a
        b_new[2*donor] += delta * delta
        b_new[2*receiver] += delta * delta
        if donor != receiver:
            b_new[donor+receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # accept criterion
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} T={T:.6f} best={best_val:.6f}")

        # periodically renormalize to avoid drift due to rounding:
        if it % 500 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)

        it += 1

    return best_a, best_val

"""
Propose a non‑negative step sequence with a low benchmark score.

Algorithm Overview
------------------
1. **Problem size** – choose `n` based on the time budget (larger budgets get a
   finer discretisation).

2. **Warm‑start** – reuse the best global construction if it exists,
   otherwise initialise several plausible candidates (power‑law, exponential,
   etc.) and keep the best according to the surrogate score.

3. **Optimization phases**
   * Phase A – exponential (mirror) descent on a soft‑max surrogate of the
     maximal convolution entry (β grows exponentially up to 2**20).
   * Phase B – multiplicative sub‑gradient descent on the true max‑convolution.
   * Phase C – Adam (adaptive‑step) mirror descent on the same soft‑max surrogate.
   * Phase D – exact hill‑climbing by moving tiny amounts of mass from
     “bad’’ indices to “good’’ ones (as in the reference solution).
   * Phase E – simulated‑annealing style random mass transfers to escape
     possible local minima.

   Time is split proportionally among the phases; each phase respects the
   remaining deadline.

4. **Post‑processing** – truncate trailing zeros, evaluate the final score with
   `evaluate_sequence`, store the result for future warm‑starts and return it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For a fixed sum Σa = √(2 n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used by mirror‑descent & Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns
    -------
    grad : np.ndarray
        Same length as a.
    full_conv : np.ndarray
        Full convolution a*a (length 2·n‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b

def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1 << 20          # ≈ 1 048 576
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule for η
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _adam_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Adam‑style adaptive mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    beta = 1.0
    max_beta = 1 << 20
    rng = np.random.default_rng()
    base_alpha = 0.018 * target_sum   # base step size for the log‑space update

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)                     # multiplicative Adam update
        a = a * (target_sum / a.sum())

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-3 / (beta + 1.0), size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

        if t % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val

def _local_refine(a0: np.ndarray, target_sum: float, deadline: float):
    """Hill‑climbing with simulated‑annealing acceptance on the true max‑conv."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04  # initial temperature

    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # donor – weighted by contribution to the current max
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with low contribution
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all[weight_all > 0], 30) if np.any(weight_all > 0) else 0.0
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            it += 1
            continue

        # propose a tiny transfer
        max_delta = min(a[donor] * 0.04, target_sum * 8e-4)
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - start - elapsed > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur = _score(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional tiny jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 4e-4)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur = _score(a)
                if cur < best_val:
                    best_val = cur
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase E – simulated‑annealing random mass transfers (global search)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, target_sum: float, deadline: float):
    """Random mass‑transfer SA using the exact convolution update."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = best_val * 0.15  # a hotter start than in the local refine

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # locate current maximal convolution index
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # donor (prefer high contributors)
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver (prefer low contributors / low values)
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all[weight_all > 0], 30) if np.any(weight_all > 0) else 0.0
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            it += 1
            continue

        # propose transfer amount (larger early, smaller later)
        max_delta = min(a[donor] * 0.07, target_sum * 0.0012)
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta

        # incremental convolution update
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # occasional full re‑normalisation (prevents drift)
        if it % 500 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 850:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a few candidate initialisations and keep the best one
    # --------------------------------------------------------------
    init_candidates = []

    # warm‑start (if any)
    if "height_sequence_1" in globals():
        ws = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        ws = ws * (target_sum / ws.sum())
        init_candidates.append(ws)

    # power‑law (exponent ≈ 0.5)
    pl = (np.arange(1, n + 1, dtype=np.float64) ** (-0.5))
    pl = pl * (target_sum / pl.sum())
    init_candidates.append(pl)

    # exponential decay (a rough alternative)
    lam = 1.5 / n
    expd = np.exp(-lam * np.arange(n, dtype=np.float64))
    expd = expd * (target_sum / expd.sum())
    init_candidates.append(expd)

    # pick the best according to the surrogate score
    best_seq = init_candidates[0]
    best_val = _score(best_seq)
    for cand in init_candidates[1:]:
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
    print(f"[{time.time():.2f}] Initial best score≈{best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.15 * (deadline - time.time())
        best_seq, best_val = _exponential_descent(best_seq, target_sum, phase_deadline)
        print(f"[{time.time():.2f}] After ExpGD best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent on true max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.15 * (deadline - time.time())
        best_seq, best_val = _subgradient_descent(best_seq, target_sum, phase_deadline)
        print(f"[{time.time():.2f}] After SubGrad best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase C – Adam mirror descent on soft‑max surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.20 * (deadline - time.time())
        best_seq, best_val = _adam_descent(best_seq, target_sum, phase_deadline)
        print(f"[{time.time():.2f}] After Adam best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase D – exact local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_val = _local_refine(best_seq, target_sum, deadline)
        print(f"[{time.time():.2f}] After Local refine best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase E – simulated‑annealing global search
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_val = _simulated_annealing(best_seq, target_sum, deadline)
        print(f"[{time.time():.2f}] After SA best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up, evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
