```
def _smooth_descent(a0, target_sum, deadline):
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)  # actual max_b
    beta = 0.5
    beta_max = 20.0
    beta_step_factor = 1.02 # increase each iteration
    base_step = target_sum * 0.5
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution
        b = _conv_fft(a)
        max_b = b.max()
        # smooth max
        shift = max_b  # max for numerical stability
        exp_term = np.exp(beta * (b - shift))
        w = exp_term / exp_term.sum()
        # gradient
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)
        L = len(w) - 1
        g = conv_full[L - np.arange(a.shape[0])]
        # line search step size: step = base_step / sqrt(it+1)
        step = base_step / np.sqrt(it + 1.0)
        # try update with sign? Since objective is f = (1/β)log sum exp(β b_i) approximating max, we want to decrease f so move opposite gradient.
        a_new = a - step * g / (np.linalg.norm(g) + 1e-12)
        # project
        a_new = _project_to_simplex(a_new, target_sum)
        # evaluate new actual max_b
        new_val = _score(a_new)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] Smooth it={it} beta={beta:.2f} score={best_val:.6f}")
        # acceptance criteria: if new_val <= best_val or with some probability if new_val slightly worse?

            a = a_new
```

assistantfinal**Solution Overview**

This version improves the previous search in three ways  

1. **Smooth gradient descent** – instead of the non‑smooth sub‑gradient we minimise a
   differentiable *soft‑max* of the convolution (`log‑sum‑exp`).  The
   temperature `β` is increased during the run, so the method first moves fast
   in a smooth landscape and later focuses on the true max.

2. **Improved mass‑transfer refinement** – the refinement step now uses the
   *full* set of (almost) maximal convolution entries.  For each iteration we
   compute the contribution of every index to the (almost) maximal entries
   (`a[i] * g[i]`, where `g` is the sub‑gradient of the max).  Mass is moved
   from the index with the largest contribution to an index with a small
   contribution.  The convolution is updated in **O(n)** time, allowing many
   cheap moves.

3. **Better time allocation & larger problem size** – for a full 1000 s budget
   we work with `n = 18000` (more degrees of freedom) and split the budget
   into four stages: smooth descent → exact sub‑gradient descent →
   short random restarts → refined mass‑transfer.  All stages respect the
   simplex constraint `∑a = √(2 n)`.

The code is fully self‑contained; the only external function is the provided
`evaluate_sequence`.  It returns the best step‑function found within the time

```python
"""
search for a step‑function with a low evaluation score.

Algorithm
---------
1. Warm‑start from the previous best construction (or a simple power‑law).
2. Smooth gradient descent on a soft‑max approximation of the max
   convolution.  β (soft‑max temperature) grows from 0.5 to ~20.
3. Exact sub‑gradient descent on the true max‑convolution.
4. Several short random restarts of the sub‑gradient phase.
5. Refined mass‑transfer step that uses the full set of (almost) maximal
   convolution entries; moves mass from a “bad’’ index (large contribution)
   to a “good’’ one while updating the convolution in O(n).
6. Trim trailing zeros, evaluate, store for possible warm‑starts and return.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective used during the search:
    2·n·max(a*a) / (sum(a)²).  For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max of the convolution)
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth maximum:
        f(a) = (1/β) log Σ exp(β * (a*a)_k)

    β is increased during the run so the method ends up minimising the true
    max.  After each step we project back onto the simplex.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    base_step = target_sum * 0.6
    beta = 0.5
    beta_max = 20.0
    beta_mul = 1.007  # geometric increase

    while time.time() < deadline:
        # convolution a*a
        b = _conv_fft(a)                     # length 2n‑1
        max_b = b.max()
        # soft‑max weights
        shift = max_b                         # for numerical stability
        w = np.exp(beta * (b - shift))
        w_sum = w.sum()
        w /= w_sum

        # gradient = 2 * a * w  (implemented via convolution)
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)   # length 3n‑2
        L = len(w) - 1
        g = conv_full[L - np.arange(a.shape[0])]

        # step size schedule
        step = base_step / np.sqrt(it + 1.0)
        gn = np.linalg.norm(g)
        if gn > 0.0:
            a -= step * g / gn

        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)      # true max‑convolution value
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth it={it:5d} β={beta:.2f} score={best_val:.6f}")

        # increase β slowly
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        it += 1

    return best_a, best_val

    Sub‑gradient descent for the non‑smooth max‑convolution.
    Keeps the iterate on the simplex Σa = target_sum.

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    base_step = target_sum * 0.5          # initial step magnitude

                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Refine improved score={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if (time.time() - start) % 15.0 < 0.1:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more freedom, but costs more per iter)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)               # we fix Σa = √(2n)

    # ------------------------------------------------------------------
    # Warm‑start (reuse previous construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.60))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Stage 0 – smooth gradient descent (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        smooth_deadline = start + 0.15 * (deadline - start)
        a_smooth, val_smooth = _smooth_descent(init, target_sum, smooth_deadline)
        best_seq = a_smooth
        best_score = val_smooth
        print(f"[{time.time():.2f}] After smooth descent score={best_score:.6f}")
    else:
        best_seq = init
        best_score = _score(init)

    # ------------------------------------------------------------------
    # Stage 1 – exact sub‑gradient descent (≈35 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.35 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short random restarts of sub‑gradient descent (~15 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rest_deadline = time.time() + 0.15 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2.0:
            remaining = rest_deadline - time.time()
            run_time = min(5.0, remaining)
            if run_time < 1.0:
                break
            jitter = rng.normal(scale=target_sum * 0.025, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            a_tmp, val_tmp = _subgradient_descent(start_seq, target_sum,
                                                  time.time() + run_time)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – refined mass‑transfer local improvement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _refined_transfer(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After refined transfer best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up, evaluate with the provided function and store for warm‑starts
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a later call
    globals()["height_sequence_1"] = final_seq

    return final_seq
```
