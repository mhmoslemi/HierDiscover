"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

The search proceeds in several stages:

1.  Warm‑start from a previously found construction if available.
2.  A schedule of growing lengths (2000 → 8000 → 32000).  At each length
    * an inexpensive parametric‐family scan produces a handful of candidates,
    * a projected Adam optimiser (smooth‑max surrogate) is run many times
      with different random starts,
    * a deterministic hill‑climb (exact max‑convolution update) refines the
      best candidate,
    * a simple sub‑gradient descent on the true max‑convolution is applied,
    * very cheap random pair‑wise mass transfers finish the stage.
3.  The best vector is up‑sampled to the next length and the process repeats.
4.  After the final length a longer Adam run and a final hill‑climb polish
    the solution.
5.  Trailing near‑zero entries are trimmed before returning.

All updates respect the simplex constraint Σa = √(2 n) (the evaluation
function is scaling‑invariant) and keep the sequence non‑negative.
The algorithm monitors the wall‑clock and stops before the allocated
budget (default 1000 s).
"""

import time
import math
import numpy as np
from operator import itemgetter


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    The algorithm runs until the wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Learning‑rate base (scaled with the simplex radius)
    base_lr = 0.15 * target_sum

    # smoothed‑max schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if t % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisation & up‑sampling
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Moves a (large) fraction of the responsible mass to the index that
    contributes the least to that entry.  Returns the best vector seen and its
    maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # destination candidates
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution efficiently
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small parametric family scan
# ----------------------------------------------------------------------


def _family_candidates(n: int, rng, top_k: int = 8):
    """
    Scan several simple families (exponential, power‑law, mixed,
    piecewise, Dirichlet) and keep the best ``top_k`` candidates.
    Returns a list of (score, vector) pairs.
    """
    target = math.sqrt(2.0 * n)
    candidates = []

    # exponential decay
    for _ in range(30):
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # power‑law
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # mixed power‑law × exponential
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p) * np.exp(-lam * np.arange(n))
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # piecewise exponential
    for _ in range(10):
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # Dirichlet (uniform on simplex)
    for _ in range(20):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    candidates.sort(key=itemgetter(0))
    return candidates[:top_k]


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iter: int = 500) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact max entry of a*a.
    At each step the index attaining the current maximum is identified and a
    step is taken opposite to its sub‑gradient (2·a shifted).  The procedure
    stops when the deadline is reached or a fixed number of iterations is
    exceeded.
    """
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)

    best_a = a.copy()
    b = _full_convolution(a)
    best_score = float(b.max())

    rng = np.random.default_rng()
    t = 0
    # initial step size – a few percent of the simplex radius
    eta0 = target * 0.02

    while time.time() < deadline and t < max_iter:
        t += 1
        b = _full_convolution(a)
        M = int(np.argmax(b))

        # counterpart a[M‑i] for each i that contributes to b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        counterpart = np.zeros(n, dtype=float)
        counterpart[idx] = a[M - idx]

        grad = 2.0 * counterpart
        eta = eta0 / math.sqrt(t)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target)

        b_new = _full_convolution(a_new)
        sc_new = float(b_new.max())

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 100 == 0:
            jitter = rng.normal(scale=target * 1e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the global time budget and returns the best sequence
    found (as a Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Warm start from a previously saved construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            if prev.size:
                n0 = prev.size
                target0 = math.sqrt(2.0 * n0)
                prev = _project_to_simplex(prev, target0)
                best_seq = prev.copy()
                best_score = _max_convolution_fft(best_seq)
                print(f"[init] Seed from height_sequence_1 (n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"[init] Failed to use height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # 1)  Length schedule – grow the dimension gradually
    # ------------------------------------------------------------------
    schedule = [2000, 8000, 32000]   # keep it modest to stay within the budget

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise candidate (upsample previous best if possible)
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)

        a = _project_to_simplex(a, target)

        # ------------------------------------------------------------------
        # 1a)  Small parametric family scan (cheap)
        # ------------------------------------------------------------------
        top_cand = _family_candidates(n, rng, top_k=6)
        init_score = _max_convolution_fft(a)
        top_cand.append((init_score, a))
        top_cand.sort(key=itemgetter(0))
        top_cand = top_cand[:6]

        # take the current best among them as seed
        a_start = top_cand[0][1]
        if top_cand[0][0] < best_score - 1e-12:
            best_seq = a_start.copy()
            best_score = float(top_cand[0][0])
            print(f"[{n}] Family‑scan seed improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 1b)  Adam optimisation (multiple restarts)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        adam_budget = remaining * 0.55
        n_restart = max(1, int(adam_budget // 5.0))
        per_restart = adam_budget / n_restart

        local_best = None
        local_best_sc = np.inf

        for r in range(n_restart):
            if time.time() >= deadline:
                break
            start_vec = a_start.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_vec, target,
                                   time.time() + per_restart - 0.1)
            sc = _max_convolution_fft(a_opt)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = a_opt.copy()

        if local_best is not None and local_best_sc < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_best_sc
            print(f"[{n}] Adam improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 1c)  Deterministic hill‑climb on the exact convolution
        # ------------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_hc, sc_hc = _hill_climb_deterministic(
                best_seq.copy(),
                _full_convolution(best_seq.copy()),
                deadline - 0.5)
            if sc_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = sc_hc
                print(f"[{n}] Hill‑climb improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 1d)  Sub‑gradient descent on the true max (exact)
        # ------------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_sub, sc_sub = _subgradient_descent(
                best_seq.copy(),
                deadline - 0.5)
            if sc_sub < best_score - 1e-12:
                best_seq = a_sub.copy()
                best_score = sc_sub
                print(f"[{n}] Sub‑gradient improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 1e)  Tiny random pairwise tweaks (simulated‑annealing style)
        # ------------------------------------------------------------------
        while time.time() + 1.0 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.07
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target)
            sc = _max_convolution_fft(cand)
            if sc < best_score - 1e-12:
                best_seq = cand
                best_score = sc
                print(f"[{n}] Random tweak improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Final polishing on the remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        # one last exact hill‑climb
        a_fin, sc_fin = _hill_climb_deterministic(
            best_seq,
            _full_convolution(best_seq),
            deadline - 0.2)
        if sc_fin < best_score - 1e-12:
            best_seq = a_fin
            best_score = sc_fin
            print(f"[final] Polishing hill‑climb gave {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Trim trailing near‑zeros (purely cosmetic)
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution of length 2000
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4)  Report the official evaluation (diagnostic)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"[final] Evaluation error: {exc}")

    return best_seq.tolist()
