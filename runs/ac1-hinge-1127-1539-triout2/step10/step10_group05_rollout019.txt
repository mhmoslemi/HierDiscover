"""
Propose a non‑negative step function that (approximately) minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The implementation uses three ideas:

1. **Log‑sum‑exp surrogate** –  (1/β)·log Σ exp(β·(a*a)).
   Its gradient w.r.t. the step heights can be computed exactly with two FFTs.
2. **Log‑it parametrisation** – we write a = S·softmax(y) (S = √(2n)).
   Optimising over the unconstrained vector y automatically enforces the
   ℓ¹‑simplex and non‑negativity.
3. **Continuation + L‑BFGS‑B** – start with a small β (smooth objective) and
   increase it successively (β = 0.5,1,2,…,40).  At each β we run a limited‑
   iteration L‑BFGS‑B optimisation (quasi‑Newton, far more efficient than
   first‑order Adam).  Multiple random restarts are performed within the
   allotted wall‑clock budget.

After the quasi‑Newton phase a cheap deterministic hill‑climb (exact
convolution updates) is applied to squeeze a little more out of the best
candidate.  The final sequence is trimmed of trailing zeros and returned.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. a
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   (⋆ = linear convolution),
        p_k ∝ exp(β·b_k) .
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # stable softmax on β·b
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p   (we need only the n‑length central part)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Objective and gradient w.r.t. logits y (softmax parametrisation)
# ----------------------------------------------------------------------


def _obj_and_grad_y(y: np.ndarray, target_sum: float, beta: float):
    """
    Returns (smooth_max, grad_y) where smooth_max is
        (1/β)·log Σ exp(β·(a*a))      with a = target·softmax(y)

    Gradient is obtained via the chain rule:
        ∂F/∂y = target·w·(∂F/∂a – Σ w·∂F/∂a)   ,  w = softmax(y)
    """
    # softmax to obtain a on the simplex
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # shift for stability
    w = exp_y / np.sum(exp_y)                # probability vector, Σw = 1
    a = target_sum * w

    # smooth max value and its gradient w.r.t. a
    g_a, b = _gradient_softmax(a, beta)     # g_a = ∂F/∂a , b = a*a
    smooth_val = b.max() + np.log(np.sum(np.exp(beta * (b - b.max())))) / beta

    # back‑propagation through the softmax
    dot = np.dot(w, g_a)                     # Σ w_i * g_i
    grad_y = target_sum * w * (g_a - dot)

    return smooth_val, grad_y


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max) – unchanged from previous code
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

    # ------------------------------------------------------------------

        except Exception as e:

    for r in range(max_restarts):

    except Exception as exc:                     # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
