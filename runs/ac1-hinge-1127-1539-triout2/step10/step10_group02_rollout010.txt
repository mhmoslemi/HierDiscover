"""
Optimise a non‑negative step function to minimise
    2·n·max(conv(a,a)) / (sum(a))²

The algorithm combines a diverse random initialisation with several
deterministic and stochastic refinement stages:

1. **Random pool** – power‑law, exponential, mixed, Dirichlet‑drawn
   candidates are generated and projected onto the ℓ¹‑simplex
   (Σa = √(2n)).

2. **Soft‑max GD** – momentum projected gradient descent on a
   smooth log‑sum‑exp approximation of the convolution maximum,
   with an annealed temperature β.

3. **Exact sub‑gradient descent** – uses the true max‑convolution
   sub‑gradient and Euclidean simplex projection.

4. **LP‑refinement** – solves a small linear programme that finds a
   partner sequence g obeying the current max‑convolution bound.
   Mixing a with g (small step‑size) provably reduces the bound.

5. **Deterministic mass‑transfer** – moves a little mass from the
   “donor’’ (largest contribution to the current peak) to a
   “receiver’’ (smallest contribution) with simulated‑annealing.

6. **Permutation annealing** – swap‑based simulated annealing on the
   ordering of the entries.

7. **Final polishing** – a high‑β soft‑max GD pass.

All stages respect the global time budget (default 1000 s) and the
best sequence found is stored in the global variable `height_sequence_1`
for warm‑starts in later runs.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(a*a)/(Σa)² (scale‑invariant)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate generators

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam*i) normalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _mixed_sequence(n: int, p: float, lam: float, target_sum: float) -> np.ndarray:
    """Product of power‑law and exponential, normalised."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, alpha: float, target_sum: float,
                        rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration alpha, then normalised."""
    if alpha <= 0:
        alpha = 0.1
    g = rng.gamma(alpha, 1.0, size=n)
    a = g / g.sum() * target_sum
    return a


def _generate_initial_candidates(n: int, target_sum: float,
                                 rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse pool of seeds."""
    cand = []

    # power‑law family
    for p in rng.uniform(0.35, 0.95, size=5):
        cand.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in rng.uniform(0.0, 4.0, size=5):
        cand.append(_exponential_sequence(n, float(lam), target_sum))

    # mixed family
    for _ in range(5):
        p = float(rng.uniform(0.35, 0.95))
        lam = float(rng.uniform(0.0, 4.0))
        cand.append(_mixed_sequence(n, p, lam, target_sum))

    # Dirichlet draws – sparse and dense variants
    for alpha in (0.2, 0.5, 1.0, 2.0):
        cand.append(_dirichlet_sequence(n, alpha, target_sum, rng))

    # jittered copies of the above (to increase diversity)
    jitter_scale = target_sum * 0.015
    for base in list(cand):
        jitter = rng.normal(scale=jitter_scale, size=n)
        v = np.maximum(base + jitter, 0.0)
        v = _project_to_simplex(v, target_sum)
        cand.append(v)

    # warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            # linear interpolation to required size
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        cand.append(prev.copy())

    return cand

# Soft‑max projected gradient descent
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """Return (grad, conv) for the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # gradient = 2 * (reverse(a) ⋆ p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _softmax_gd_phase(a0: np.ndarray, target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(best)

    beta = 0.5
    max_beta = 1e6
    step_factor = 0.7
    gamma = 0.98
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] GD β={beta:.1e} it={it:5d} score={best_val:.6f}")

        # anneal β upwards
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val

def _subgradient_descent_phase(a0: np.ndarray, target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    momentum = np.zeros_like(a)
    gamma = 0.93
    step_factor = 0.5
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        # sub‑gradient: g[i] = 2*a[kmax-i] if the index is valid
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best, best_val

def _lp_refine_phase(a0: np.ndarray, target_sum: float,
                     deadline: float,
                     max_iter: int = 200,
                     active_k: int = 12) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a tiny linear programme that finds a partner sequence g
    respecting the current max‑convolution bound. Mix a small amount of g
    into a.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        M = conv.max()
        # choose a few indices with the largest values (active set)
        if active_k >= conv.size:
            idx_set = np.arange(conv.size)
        else:
            idx_set = np.argpartition(conv, -active_k)[-active_k:]

        # Build A_ub (each row corresponds to a constraint for a selected k)
        rows = []
        for k in idx_set:
            row = np.zeros(n)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # coefficients for g_j are a_{k - j}
            seg = a[k - np.arange(lo, hi + 1)]
            row[lo:hi + 1] = seg
            rows.append(row)

        A_ub = np.vstack(rows)
        b_ub = np.full(A_ub.shape[0], M)

        # maximise sum(g)  → minimise -sum(g)
        c = -np.ones(n)

        res = _opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                           bounds=[(0, None)] * n, method='highs',
                           options={"presolve": True, "time_limit": 5.0})

        if not res.success:
            # LP failed – exit early
            break

        g = res.x
        if g.sum() < 1e-12:
            break

        # normalise g to the target ℓ¹ norm
        g = g * (target_sum / g.sum())

        # try a handful of mixing coefficients
        improved = False
        for t in (0.12, 0.08, 0.04, 0.02):
            cand = (1.0 - t) * a + t * g
            cand = _project_to_simplex(cand, target_sum)
            val = _score(cand)
            if val < best_val:
                best_val = val
                best = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time():.2f}] LP it={it:3d} t={t:.3f} score={best_val:.6f}")
                break
        if not improved:
            # if no improvement, inject a small random perturbation
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (temperature‑controlled)
# ----------------------------------------------------------------------


def _mass_transfer_phase(a0: np.ndarray, target_sum: float,
                         deadline: float,
                         start_temp: float = 0.6,
                         end_temp: float = 2e-4,
                         low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy movement of a small amount of mass from a donor to a receiver."""
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(conv.argmax())
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight contributed to the peak by each index pair (i, kmax-i)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # pick a receiver among the low‑contributing indices
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w.max() - w.min()
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.3)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        cur = _score(a)

        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] MassTx it={it:5d} score={best_val:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(cur - best_val) / max(temp, 1e-12))
            if rng.random() < prob:
                best_val = cur
                best = a.copy()
                print(f"[{time.time():.2f}] MassTx it={it:5d} (uphill) score={best_val:.6f}")
            else:
                # revert
                a[donor] += delta
                a[receiver] -= delta
                a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based SA)
# ----------------------------------------------------------------------


def _permutation_annealing_phase(a0: np.ndarray, target_sum: float,
                                 deadline: float,
                                 init_score: float | None = None) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur)
    best_seq = cur.copy()
    best_score = cur_score
    n = cur.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    start_temp = 0.25
    end_temp = 2e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        ratio = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        cur[i], cur[j] = cur[j], cur[i]

        new_score = _max_convolution_fft(cur)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur.copy()
                print(f"[{time.time():.2f}] PermSA new best={best_score:.6f}")
        else:
            cur[i], cur[j] = cur[j], cur[i]

        # occasional full shuffle towards the end
        if ratio > 0.85 and rng.random() < 0.01:
            rng.shuffle(cur)
            cur_score = _max_convolution_fft(cur)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final high‑β soft‑max polishing
# ----------------------------------------------------------------------


def _final_polish_phase(a0: np.ndarray, target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """High‑β GD for a few seconds to polish the solution."""
    return _softmax_gd_phase(a0, target_sum, deadline)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns the best sequence found before the time budget expires.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        # proportional to the available time budget
        if budget_s >= 800:
            n = 15000
        elif budget_s >= 500:
            n = 11000
        else:
            n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Random initial pool
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    init_pool = _generate_initial_candidates(n, target_sum, rng)

    best_seq = None
    best_score = np.inf
    for cand in init_pool:
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Init candidate score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Soft‑max GD (coarse)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + (deadline - time.time()) * 0.22
        cand, val = _softmax_gd_phase(best_seq, target_sum, phase_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After soft‑max GD score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + (deadline - time.time()) * 0.18
        cand, val = _subgradient_descent_phase(best_seq, target_sum, phase_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After sub‑grad score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – LP‑based refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + (deadline - time.time()) * 0.16
        cand, val = _lp_refine_phase(best_seq, target_sum, phase_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After LP refinement score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Deterministic mass‑transfer
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + (deadline - time.time()) * 0.16
        cand, val = _mass_transfer_phase(best_seq, target_sum, phase_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After mass‑transfer score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + (deadline - time.time()) * 0.14
        cand, val = _permutation_annealing_phase(best_seq, target_sum,
                                                 phase_deadline, init_score=best_score)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] After permutation annealing score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – Final polishing (high‑β GD)
    # --------------------------------------------------------------
    if time.time() + 1.0 < deadline:
        cand, val = _final_polish_phase(best_seq, target_sum, deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Final polish score={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: strip trailing zeros
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Evaluate with the provided black‑box (for logging only)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # Warm‑start for the next run
    globals()["height_sequence_1"] = final_seq

    return final_seq
