"""
Improved step‑function search for the convolution‑peak objective.

The pipeline combines many cheap stochastic optimisation phases while
respecting a 1000 s budget:

1. **Rich initial pool** – power‑law, exponential and jittered copies,
   all projected onto the ℓ¹–simplex {x ≥ 0, Σx = √(2 n)}.
2. **Projected soft‑max GD** – momentum GD on a smooth max‑approximation,
   with a higher β schedule and more restarts.
3. **Explicit sub‑gradient descent** – true max‑subgradient (no soft‑max)
   with a diminishing step‑size and Nesterov‑style momentum.
4. **Deterministic mass‑transfer** – greedy transfer from the “donor”
   (largest contribution to the current peak) to a low‑contributing
   “receiver”, with simulated‑annealing acceptance.
5. **Symmetrisation** – after every major phase we enforce aₖ = a_{n‑1‑k},
   which never harms the objective and often uncovers better minima.
6. **Permutation annealing** – swap‑based simulated annealing on the
   ordering of the coefficients.
7. **Local coordinate descent** – cheap random pair moves guided by the
   current max‑subgradient.
8. **Final polishing** – a high‑β soft‑max GD run.

All convolutions use an FFT of the next power‑of‑two size, giving
O(n log n) per evaluation.  The code respects the time budget and
stores the best found sequence in the global ``height_sequence_1`` for
warm‑starts in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (length of full linear convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _fft_size(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = s}.
    Implements the O(n log n) algorithm based on sorting.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^{-p} normalised to sum = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, r: float, target_sum: float) -> np.ndarray:
    """a_i = r^{i} normalised to sum = target_sum (0 < r < 1)."""
    a = r ** np.arange(n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest / smallest values (helps symmetry)."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] = a[n‑1‑i] (averaging with the reversed copy)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Soft‑max gradient utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Compute the gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where conv = a*a (full linear convolution).
    """
    n = a.shape[0]
    size = _fft_size(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # central n entries
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Momentum soft‑max projected GD
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    The schedule for β doubles every 300 iterations until `max_beta`.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD β={beta:.1f} it={it} best={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiple short GD runs with jitter.  The best result across all
    restarts is returned.
    """
    best_a = init.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e6,
            step_factor=0.7,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase X – Exact max sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.95,
                         jitter_interval: int = 400,
                         jitter_scale: float = 5e-4) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Uses a momentum term and occasional jitter.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient g[i] = 2 * a[kmax - i] for indices in range
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale,
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves based on the current maximal convolution entry.
    Accepts worsening moves according to a temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            # occasional jitter to escape plateaus
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based SA)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering of coefficients."""
    cur_seq = a.copy()
    cur_score = (
        init_score if init_score is not None else _max_convolution_fft(cur_seq)
    )
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best={best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full reshuffle near the end
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase F – Local coordinate descent (tiny random moves guided by sub‑grad)
# ----------------------------------------------------------------------


def _local_coordinate_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap random pair moves where mass is shifted from a high‑gradient
    index to a low‑gradient one.  This refines the solution after the
    heavier phases.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of max at current peak
        idx = np.arange(n)
        src = kmax - idx
        mask = (src >= 0) & (src < n)
        g = np.zeros(n)
        g[mask] = 2.0 * a[src[mask]]

        donor = np.argmax(g)
        receiver = np.argmin(g)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.4, target_sum * 1e-4)
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Local descent new best={best_val:.6f}")

    return best_a, best_val

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal convolution peak.
    Returns a Python list of floats suitable for evaluate_sequence().
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size selection
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build a diverse pool of starting candidates
    # ------------------------------------------------------------------
    init_seqs = []

    # power‑law family
    for p in (0.45, 0.48, 0.5, 0.53, 0.55, 0.58, 0.6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # exponential family
    for r in (0.98, 0.985, 0.99, 0.992, 0.995):
        init_seqs.append(_exponential_sequence(n, r, target_sum))

    # warm start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(
            globals()["height_sequence_1"], dtype=np.float64
        )
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        prev = _project_to_simplex(prev, target_sum)
        init_seqs.append(prev)

    # jittered copies of the above
    rng = np.random.default_rng()
    jittered = []
    for seq in init_seqs:
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        s = seq + jitter
        s = np.maximum(s, 0.0)
        s = _project_to_simplex(s, target_sum)
        jittered.append(s)
    init_seqs.extend(jittered)

    # random permutations of a few smooth candidates
    for p in (0.5, 0.55):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # ------------------------------------------------------------------
    # Phase A – Soft‑max projected GD (30 % of time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.30

    best_seq = None
    best_val = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        a, val = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Global best after GD = {best_val:.6f}")

    if best_seq is None:  # fallback safety
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_val = _max_convolution_fft(best_seq)

    # Symmetrise early – never harms the objective
    best_seq = _symmetrize(best_seq)
    best_seq = _project_to_simplex(best_seq, target_sum)

    # ------------------------------------------------------------------
    # Phase X – Exact max sub‑gradient descent (25 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = time.time() + (deadline - time.time()) * 0.25
        a_sg, val_sg = _subgradient_descent(
            best_seq,
            target_sum,
            sg_deadline,
            step_factor=0.6,
            gamma=0.95,
        )
        if val_sg < best_val:
            best_val = val_sg
            best_seq = a_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best={best_val:.6f}")

    best_seq = _symmetrize(best_seq)
    best_seq = _project_to_simplex(best_seq, target_sum)

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (25 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.25
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_deadline,
            target_sum,
            start_temp=0.5,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_val:.6f}")

    best_seq = _symmetrize(best_seq)
    best_seq = _project_to_simplex(best_seq, target_sum)

    # ------------------------------------------------------------------
    # Phase D – Permutation annealing (15 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.15
        a_perm, val_perm = _permutation_annealing(
            best_seq,
            target_sum,
            perm_deadline,
            init_score=best_val,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase F – Local coordinate descent (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_loc, val_loc = _local_coordinate_descent(
            best_seq, target_sum, deadline
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[{time.time():.2f}] After local descent best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
