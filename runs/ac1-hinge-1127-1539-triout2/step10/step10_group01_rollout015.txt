def _adam_subgradient_descent(a0, deadline, target_sum, step_factor=0.5, beta1=0.9, beta2=0.999):
    a = _project_to_simplex(a0.copy(), target_sum)
    n = len(a)
    best = a.copy()
    best_val = _max_convolution_fft(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    eps = 1e-8
    while time.time() < deadline:
        conv = _conv_fft(a)   # full conv
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            break
        w = w / w_sum
        # compute gradient
        a_rev = a[::-1]
        conv_rev = _conv_full_fft(a_rev, w)   # length 3n-2
        grad = 2.0 * conv_rev[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        it1 = it + 1
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # projection
        a = _project_to_simplex(a, target_sum)
        # evaluate
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] AdamSubgrad it={it:5d} score={best_val:.6f}")
        it += 1
    return best, best_val
```

"""
Propose a step function minimizing the maximum auto‑convolution (the benchmark
objective). The algorithm combines a dense parametric pre‑search over
power‑law / exponential / plateau families, a projected Adam optimizer that
uses the exact sub‑gradient of the max‑convolution, an aggressive deterministic
mass‑transfer phase that equalises the dominant convolution peak, and a brief
random‑move annealing stage. All phases respect the given time budget and
return the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using FFT."""
    size = 1 << ((a.size + b.size - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: a.size + b.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full, not circular)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0, Σ x_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _plateau_powerlaw_sequence(n: int, p: float, frac: float, target_sum: float) -> np.ndarray:
    """Plateau of constant height (value 1) for the first `frac` portion,
    then a pure power‑law tail."""
    L = int(frac * n)
    tail = (np.arange(L, n, dtype=np.float64) + 1) ** (-p)
    base = np.concatenate([np.ones(L, dtype=np.float64), tail])
    base *= target_sum / base.sum()
    return base


def _plateau_exp_powerlaw_sequence(
    n: int, p: float, q: float, frac: float, target_sum: float
) -> np.ndarray:
    """Plateau of 1’s followed by an exponentially tapered power law."""
    L = int(frac * n)
    i = np.arange(L, n, dtype=np.float64)
    tail = (i + 1) ** (-p) * np.exp(-q * (i - L) / (n - L))
    base = np.concatenate([np.ones(L, dtype=np.float64), tail])
    base *= target_sum / base.sum()
    return base


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Returns the sub‑gradient of max(conv) w.r.t. a.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    mask = (conv >= max_val - 1e-12)  # tolerance for numerical noise
    w = mask.astype(np.float64)
    if w.sum() == 0:
        return np.zeros_like(a)
    w /= w.sum()
    a_rev = a[::-1]
    conv_rev = _conv_full_fft(a_rev, w)  # length = 3n-2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimizer using the exact sub‑gradient
# ----------------------------------------------------------------------


def _adam_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected descent on the max‑convolution objective,
    using the exact sub‑gradient.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        grad = _subgradient_max_conv(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamSubgrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _adam_subgradient_restarts(
    init: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Multiple short Adam‑subgradient runs with random jitter, keeping the best.
    """
    best_a = init.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_subgradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            step_factor=0.5,
            beta1=0.9,
            beta2=0.999,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamSubgrad restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic exact mass‑transfer (aggressive)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer that directly reduces the current maximal convolution
    entry to (approximately) the second‑largest entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        if idx_max.size == 0:
            break

        # work on the first maximal index
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight w_i = a_i + a_{k0-i}
        w = a[idx] + a[k0 - idx]
        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contributions to the peak
        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            # cannot improve this peak; try another maximal index if any
            if len(idx_max) > 1:
                k0 = idx_max[1]
                lo = max(0, k0 - (n - 1))
                hi = min(k0, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                w = a[idx] + a[k0 - idx]
                donor = idx[np.argmax(w)]
                receiver = idx[np.argmin(w)]
                contrib_d = a[k0 - donor]
                contrib_r = a[k0 - receiver]
                denom = 2.0 * (contrib_d - contrib_r)
                if denom <= 0.0:
                    break
            else:
                break

        # second largest convolution value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        delta_needed = (max_val - second_max) / denom
        delta = min(delta_needed, a[donor])  # never make donor negative
        if delta <= 0.0:
            break

        a[donor] -= delta
        a[receiver] += delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass transfer (exploratory annealing)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style moves that shift a small amount of mass
    from large entries to small ones.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score

        best_seq, dmt_deadline, target_sum
    )
    print(f"[{time.time():.2f}] After deterministic mass transfer best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase C – Global random mass‑transfer (exploration)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    rand_deadline = time.time() + remaining * 0.15
    best_seq, best_score = _global_random_mass_transfer(best_seq, rand_deadline, target_sum)
    print(f"[{time.time():.2f}] After random moves best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase D – Final polishing with tighter Adam parameters
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        best_seq, best_score = _adam_subgradient_descent(
            best_seq,
            deadline,
            target_sum,
            step_factor=0.3,
            beta1=0.9,
            beta2=0.999,
        )
        print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries, return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is undefined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
