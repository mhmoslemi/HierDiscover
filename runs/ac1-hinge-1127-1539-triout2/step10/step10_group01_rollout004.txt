"""
Fast parametric + local‑search optimizer for the step‑function benchmark.

The algorithm proceeds in four stages:

1. **Wide parametric sweep** – 2‑parameter families (power‑law with optional
   exponential tail and a small offset) are sampled densely.  For each
   candidate we scale it to the invariant sum Σa = √(2 n) and keep the
   one with the smallest max‑convolution value.

2. **Refined gradient descent** –  Adam‑style projected gradient descent
   (soft‑max approximation of the max) is run from several jittered
   starts, keeping the best solution found before the deadline of this
   stage.

3. **Exact deterministic mass‑transfer** – a greedy “water‑filling’’ step
   that directly reduces the current peak of the convolution by moving
   mass from the most harmful index to the least harmful one.  The step
   is repeated until the time budget for this stage expires.

4. **Final polishing** – a short Adam run with a much larger soft‑max
   temperature (β) cleans up any remaining irregularities.

All operations respect the simplex constraint Σa = √(2 n) and non‑negativity.
The routine returns the best sequence discovered before the global
deadline (the competition gives 1000 s).  The code is self‑contained and
relies only on NumPy.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective (the benchmark evaluates the same formula)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the benchmark score for a (Σa is expected to be √(2 n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the non‑negative simplex with total mass s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the wide pre‑search)
# ----------------------------------------------------------------------
def _param_sequence(n: int,
                    p: float,
                    q: float,
                    offset: float,
                    target_sum: float) -> np.ndarray:
    """
    a_i = (i+offset+1)^(-p) * exp(-q * i / n)   (i = 0,…,n‑1)
    Afterwards the vector is scaled to have Σa = target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1.0) ** (-p) * np.exp(-q * i / n)
    a_sum = a.sum()
    if a_sum == 0.0:
        # fall back to uniform – should never happen for reasonable parameters
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a *= target_sum / a_sum
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (for Adam‑GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, full_convolution) where grad = ∂ softmax_β(conv) / ∂a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]  # full conv

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)          # soft‑max weights (sum to 1)

    # convolution of reversed a with p gives the gradient (see derivation)
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # keep only the n centred entries
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------
def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase soft‑max temperature
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short Adam‑GD runs with random jitter, keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy moves that lower the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # pick a single maximal index (first)
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # gradient‑like weights w_i = a_i + a_{k0-i}
        w = a[idx] + a[k0 - idx]

        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contribution to the peak from donor / receiver
        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            # try another maximal index if available
            if len(idx_max) > 1:
                k0 = idx_max[1]
                lo = max(0, k0 - (n - 1))
                hi = min(k0, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                w = a[idx] + a[k0 - idx]
                donor = idx[np.argmax(w)]
                receiver = idx[np.argmin(w)]
                contrib_d = a[k0 - donor]
                contrib_r = a[k0 - receiver]
                denom = 2.0 * (contrib_d - contrib_r)
                if denom <= 0.0:
                    break
            else:
                break

        # second largest convolution value (excluding all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)  # a bit more aggressive than 0.5

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploratory phase)
# ----------------------------------------------------------------------
def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing style moves that shift mass from big to small entries."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the global time budget and returns the best
    sequence discovered.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep it moderate but large enough for a good shape
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)          # invariant scaling Σa = √(2 n)

    # ------------------------------------------------------------------
    # Phase 0 – exhaustive parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Search ranges (chosen to be broader than the previous implementation)
    p_vals = np.linspace(0.10, 1.20, 112)        # step ≈0.01
    q_vals = np.linspace(0.0, 8.0, 81)          # step ≈0.1
    offset_vals = np.linspace(0.0, 5.0, 11)    # step =0.5

    phase0_deadline = start_time + 0.20 * budget_s
    print(f"[{time.time():.2f}] Starting Phase 0 (parametric sweep)")

    for offset in offset_vals:
        for p in p_vals:
            # Pre‑compute the power‑law part (without exponential tail)
            i = np.arange(n, dtype=np.float64)
            base = (i + offset + 1.0) ** (-p)

            # Normalisation constant for this (p, offset) – will be adjusted per q
            base_sum = base.sum()
            if base_sum == 0.0:
                continue

            for q in q_vals:
                if time.time() >= phase0_deadline:
                    break
                a = base * np.exp(-q * i / n)
                a *= target_sum / a.sum()
                val = _score(a)
                if val < best_score:
                    best_score = val
                    best_seq = a.copy()
        if time.time() >= phase0_deadline:
            break

    # If for some reason the sweep did not produce a sequence (should not happen)
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"[{time.time():.2f}] Phase 0 complete – best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Small random perturbations around the current best (still in Phase 0)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(12):
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Jitter improvement – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + 0.30 * (deadline - time.time())
    print(f"[{time.time():.2f}] Starting Phase A (Adam‑GD)")

    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer + short random walk
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting Phase B (deterministic mass‑transfer)")
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact(best_seq,
                                                            deadline,
                                                            target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT improved best = {best_score:.6f}")

        # A brief random phase to shake out possible stagnation
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        # give a chance to exit early if we are very close to the global deadline
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Final polishing (higher soft‑max temperature)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        print(f"[{time.time():.2f}] Final polishing with high‑β Adam")
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing zeros and return as plain list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional sanity check (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the best sequence for a potential future run
    globals()["height_sequence_1"] = final_seq

    return final_seq
