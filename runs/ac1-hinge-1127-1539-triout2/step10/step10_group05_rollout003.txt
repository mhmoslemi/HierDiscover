```
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # set n, target sum (maybe sqrt(2*n) but not needed)
    n = 2000
    target_sum = math.sqrt(2.0 * n)

    # initialize best from previous if exists else random

        # trim trailing zeros
        nz = np.where(a > 1e-12)[0]
        if nz.size:
            a = a[: nz[-1]+1]
        # if length different, upsample to n
        if a.size != n:
            a = _upsample_sequence(a, n)
    else:
        a = _random_initial_sequence(n, rng)

    a = _project_to_simplex(a, target_sum)

    best_seq = a.copy()
    best_score = _max_convolution_fft(best_seq)

    # iteration loop
    iteration = 0
    while time.time() < deadline - 5:
        iteration += 1
        # compute M
        M = _max_convolution_fft(a)
        # build convolution matrix A
        A = np.zeros((2*n-1, n), dtype=np.float64)
        # fill columns
        for j in range(n):
            A[j:j+n, j] = a
        # linear program: maximize sum(g) subject to A @ g <= M, g>=0
        c = -np.ones(n)  # minimize -sum(g)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP failed at iter {iteration}: {res.message}")
            break
        g = res.x
        # rescale g to same sum as a
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g * (target_sum / sum_g)
        # combine
        t = 0.5 if iteration < 5 else 0.2
        a_new = (1 - t) * a + t * g
        # ensure nonneg and project to simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)
```

assistantfinal**Solution Overview**

The score returned by `evaluate_sequence` is  
\[
\text{score}= \frac{2n\;\max(a*a)}{(\sum a)^2}.
\]

If we keep the ℓ¹‑norm of the step function fixed (the original code uses
`√(2n)`) the denominator cancels and the score is just the maximal
convolution entry `M = max(a*a)`.  Reducing `M` is therefore the whole
task.

For a fixed vector `f` (our current step function) we solve the **linear
program**

```
max   Σ g_j
s.t.  f * g  ≤  M    (entry‑wise)
      g_j ≥ 0
```

(`*` denotes linear convolution).  The constraints are linear in the
unknown vector `g`, while the objective pushes `g` to be as “spread out”
as possible under the same maximal autocorrelation bound `M`.  After the
LP we rescale `g` to the same ℓ¹‑norm as `f` and mix the two vectors

```
f ← (1‑t)·f + t·g ,   0 < t ≤ 0.5 .
```

Because `g` respects the bound `f * g ≤ M`, this convex combination never
increases the maximal self‑convolution and usually decreases it.
Repeating the LP‑mix step quickly drives `M` down.

To finish, a short deterministic hill‑climb (the exact‑max convolution
updates from the previous state‑of‑the‑art code) and a final Adam
optimisation are applied.  All work respects the 1000 s wall‑clock
budget.

The implementation below follows this plan.  It re‑uses the useful
helpers from the earlier code (`_project_to_simplex`, FFT‑based
convolution, random initial families, hill‑climb, Adam) and adds the
LP‑based improvement loop.

```python
"""
Iterative LP‑based improvement for the max‑convolution minimisation
problem.

1.  Initialise a non‑negative sequence (seeded from a previous run or a
    random family) and project it onto the ℓ¹‑simplex Σa = √(2 n).

2.  Repeatedly:
        • compute M = max(conv(a, a))
        • build the full convolution matrix A (size (2n‑1) × n) such that
          A @ g = a * g
        • solve   max Σ g   s.t.   A @ g ≤ M ,  g ≥ 0   (LP)
        • rescale g to the same ℓ¹‑norm as a
        • mix a ← (1‑t)·a + t·g   with a decreasing step size t
        • keep the best vector seen

    The LP drives the distribution to be more spread out, and the
    convex combination never increases the maximal self‑convolution.

3.  Occasionally run the exact deterministic hill‑climb from the
    previous state‑of‑the‑art code to clean remaining peaks.

4.  A short Adam optimisation on the smooth‑max surrogate serves as a
    final polish.

The whole procedure stops before the 1000 s budget expires and returns
the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (Adam refinement)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of (1/β)·log Σ exp(β·b) where b = a*a."""
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """Projected Adam on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random families (used for the initial guess)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _edge_high_initial_sequence(n: int, rng) -> np.ndarray:
    """Large near the two edges, decaying towards the centre."""
    target = math.sqrt(2.0 * n)
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-lam * d)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max‑convolution)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # pick a destination j (different from i) with the smallest contribution
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Length and ℓ¹‑norm target
    # ------------------------------------------------------------------
    n = 2000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Starting point (seed from a previous run if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a = np.asarray(globals()["height_sequence_1"], dtype=float)
            nz = np.where(a > 1e-12)[0]
            if nz.size:
                a = a[: nz[-1] + 1]
            if a.size != n:
                a = _upsample_sequence(a, n)
            a = _project_to_simplex(a, target_sum)
            print(f"[{time.time():.1f}] Seeded from previous run (len={a.size})")
        except Exception as e:
            print("Seeding error:", e)
            a = _random_initial_sequence(n, rng)
    else:
        a = _random_initial_sequence(n, rng)

    best_seq = a.copy()
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.1f}] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) LP‑based improvement loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 5.0:
        iteration += 1
        M = _max_convolution_fft(a)

        # Build the full convolution matrix A (dense, shape (2n‑1) × n)
        # A @ g == a * g
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        # Solve the LP: maximize Σ g_j  ⇔  minimize -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP failed (iter {iteration}): {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        # Rescale to the target ℓ¹‑norm
        g = g * (target_sum / sum_g)

        # Mix a and g
        if iteration < 5:
            t = 0.5
        elif iteration < 15:
            t = 0.2
        else:
            t = 0.1
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        # Evaluate the new candidate
        score_new = _max_convolution_fft(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.1f}] LP iter {iteration}: improved score = {best_score:.6f}")

        # Occasionally invoke a short deterministic hill‑climb
        if iteration % 3 == 0:
            a_hc, sc_hc = _hill_climb_deterministic(
                a_new, _full_convolution(a_new), deadline - 0.5)
            if sc_hc < best_score - 1e-12:
                best_score = sc_hc
                best_seq = a_hc.copy()
                a = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb after LP iter {iteration}: score = {best_score:.6f}")
                continue

        a = a_new

    # ------------------------------------------------------------------
    # 2) Final polishing with Adam (smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_opt = _adam_optimize(best_seq, target_sum, deadline - 1.5)
        sc_opt = _max_convolution_fft(a_opt)
        if sc_opt < best_score - 1e-12:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"[{time.time():.1f}] Final Adam polish: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Trim trailing near‑zeros and report
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Evaluation error:", exc)

    return best_seq.tolist()
```
