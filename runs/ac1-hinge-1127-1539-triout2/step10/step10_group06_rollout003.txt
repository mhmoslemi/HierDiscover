def random_search(n, target_sum, deadline):
    best_a = None; best_val = np.inf
    rng = np.random.default_rng()
    while time.time() < deadline:
        a = rng.exponential(scale=1.0, size=n)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random search new best {best_val:.6f}")
    return best_a, best_val

def propose_candidate(budget_s=1000, **kwargs):
    start=time.time()
    deadline = start + budget_s - 0.5

    # Choose n based on budget
    if budget_s >= 900:
        n = 24000
    elif budget_s >= 700:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # Warm start
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # power law
        p = 0.55
        init = (np.arange(1, n+1, dtype=np.float64) ** (-p))
    init = init * (target_sum / init.sum())

    best_seq = init
    best_score = _score(init)
    print(f"[{time.time():.2f}] Initial score {best_score:.6f}")

    # Random search phase (5% budget)
    if time.time() < deadline:
        random_deadline = time.time() + 0.05 * (deadline - time.time())
        a_rnd, val_rnd = random_search(n, target_sum, random_deadline)

        exp_deadline = time.time() + 0.15 * (deadline - time.time())
        a_exp, val_exp = exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best {best_score:.6f}")

        sub_deadline = time.time() + 0.20 * (deadline - time.time())
        a_sub, val_sub = subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best {best_score:.6f}")

    # Phase C gradient descent with restarts (30% budget)

        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = gradient_descent_with_restarts(best_seq, target_sum, gd_deadline)

"""
Propose a non‑negative step function with a low benchmark score.

The algorithm consists of several stages that together explore the
search space aggressively:

* 1️⃣  Warm‑start from a previously found construction (if any) and
    resize it to a problem size n chosen according to the time budget.
* 2️⃣  A short random‑search phase that draws many Dirichlet‑like vectors
    (exponential samples) and keeps the best.
* 3️⃣  Exponential (mirror) descent on a soft‑max surrogate of the
    maximal convolution entry (phase A).
* 4️⃣  Multiplicative sub‑gradient descent on the true max‑convolution
    (phase B).
* 5️⃣  Projected gradient descent with momentum and many random restarts
    (phase C).
* 6️⃣  Exact local refinement that repeatedly transfers a tiny amount of
    mass from “heavy’’ indices (large contribution to the current maximal
    convolution entry) to “light’’ ones, using an O(n) convolution update
    and a simulated‑annealing acceptance rule (phase D).

The time budget (default 1000 s) is split among the phases, and the
remaining time is devoted to the most powerful local refinement.
All operations are implemented with FFT‑based convolutions, so the
overall runtime scales as O(n log n) per iteration – fast enough for
n≈20 000 on a single CPU core.

The returned list is stored in the global variable `height_sequence_1`
to serve as a warm‑start for a possible subsequent call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    When Σa = √(2 n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phase A and Phase C)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # stable soft‑max
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)   – length n
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror descent on the soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        # learning rate slowly decays with beta
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – subgradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent directly on max(a*a).
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # full convolution and index of the current maximum
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]   # sub‑gradient for the active max entry

        # linear step‑size schedule
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with restarts (same idea as before)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact local mass‑transfer refinement (enhanced)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing / simulated‑annealing on the true max‑convolution.
    Transfers a small amount of mass from high‑contributing indices to
    low‑contributing ones, using an O(n) convolution update.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact full convolution
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # initial temperature proportional to the current objective
    T0 = best_val * 0.05

    while time.time() < deadline:
        # locate current maximal convolution entry
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to the maximal entry
        prod = a[idx] * a[k_max - idx]               # a_i * a_{k_max-i}

        # pick a donor proportional to its contribution
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a receiver among indices with relatively small contributions
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile tail
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --- step size (larger early, smaller later) -------------------
        elapsed = time.time() - start
        frac = 1.0 - elapsed / total_time           # 1 → early, 0 → late
        max_delta = min(a[donor] * (0.04 + 0.06 * frac),
                        target_sum * (0.0008 + 0.0008 * frac))
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --- update convolution exactly (O(n) slices) -------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --- simulated annealing acceptance ------------------------------
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - start > total_time * 0.25:  # every ~25 % of total time
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")
            start = time.time()   # reset the periodic timer

        # occasional tiny jitter (helps escaping very flat regions)
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Random search – cheap Dirichlet‑like sampling
# ----------------------------------------------------------------------


def _random_search(n: int,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Draw many exponential samples (which are Dirichlet with α=1) and keep the best.
    Returns (best_sequence or None, best_score).
    """
    best_a = None
    best_val = np.inf
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        a = rng.exponential(scale=1.0, size=n)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random search it={it:5d} best={best_val:.6f}")
        it += 1
        if it >= 2000:   # safeguard – we rarely need more than a few thousand samples
            break
    return best_a, best_val

# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size. Larger n gives a finer discretisation and
    # typically a lower achievable bound, at the price of heavier FFTs.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 24000
    elif budget_s >= 700:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (reuse previous construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback (p≈0.55 works well)
        p = 0.55
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-p))

    init = init * (target_sum / init.sum())
    best_seq = init
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap random search (≈5 % of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rs_deadline = time.time() + 0.05 * (deadline - time.time())
        a_rand, val_rand = _random_search(n, target_sum, rs_deadline)
        if a_rand is not None and val_rand < best_score:
            best_seq, best_score = a_rand, val_rand
            print(f"[{time.time():.2f}] After random search best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – exponential (mirror) descent on soft‑max (≈12 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.12 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – subgradient descent on true max‑convolution (≈18 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.18 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – projected GD with restarts (≈30 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – exact local refinement (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After local refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final cleanup, evaluation and warm‑start storage
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
