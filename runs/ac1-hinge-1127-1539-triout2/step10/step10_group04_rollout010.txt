assistantfinal**Solution Overview**

The evaluation metric  

\[
E(a)=\frac{2n\;\max (a*a)}{(\sum_i a_i)^2}
\]

is scale‑invariant – only the *shape* of the non‑negative step function matters.  
For a fixed length `n` we therefore minimise the maximum entry of the
autocorrelation `a*a` under the simplex constraint  

\[
a_i\ge 0,\qquad \sum_i a_i=\sqrt{2n}.
\]

The key idea is an *iterated linear‑program* step:

1.  For the current vector `a` compute `M = max(a*a)`.
2.  Solve the LP  

    \[
    \max\; \sum_i g_i\quad\text{s.t.}\quad (a*g)_k\le M\;\;(k=0,\dots,2n-2),\;
    g_i\ge0 .
    \]

    The constraints are linear because `a` is fixed, while the
    objective tries to pack as much `ℓ¹`‑mass as possible under the same
    worst‑case convolution bound.
3.  If the optimal sum `S_g` is larger than `S_a = \sum a_i`,
    scale the LP solution back to the original simplex  

    \[
    a' = \frac{S_a}{S_g}\;g .
    \]

    By construction  

    \[
    \max (a'*a') \le \Bigl(\frac{S_a}{S_g}\Bigr)^2 M < M ,
    \]

    so the objective strictly improves.
4.  Repeat the LP step until no improvement is possible or the
    time budget runs out.

The LP matrix is a *Toeplitz* matrix that represents the linear
convolution `a*g`.  It is built once per iteration with `scipy.sparse`
and solved with the fast HiGHS backend (`scipy.optimize.linprog`).

To obtain a high‑quality solution for the required length `2000` we
use a two‑stage schedule:

* **Stage 1 – modest size (≈ 1200).**  
  Randomly generated families (exponential, power‑law, mixed, piecewise,
  Dirichlet) provide several seeds.  
  Each seed is refined by the LP–iteration described above and a short
  Adam‑based smooth‑max optimisation.
* **Stage 2 – target size (2000).**  
  The best vector from stage 1 is linearly up‑sampled, projected onto the
  simplex, and further polished with a few Adam runs followed by the
  deterministic hill‑climber already used in the reference code.

All operations respect the security checks of `evaluate_sequence`.  
The algorithm stops just before the wall‑clock deadline and returns the
best sequence found.

# Linear‑algebra helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _family_candidates(n: int, rng, top_k: int = 8):
    """Generate a few simple‑family candidates and keep the best ``top_k``."""
    target = math.sqrt(2.0 * n)
    candidates = []

    # exponential decay
    for _ in range(30):
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # power‑law
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # mixed power‑law × exponential
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p) * np.exp(-lam * np.arange(n))
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # piecewise exponential
    for _ in range(10):
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # Dirichlet
    for _ in range(20):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # sort and keep the best ``top_k``
    candidates.sort(key=itemgetter(0))
    return candidates[:top_k]


# ----------------------------------------------------------------------
# Upsampling / interpolation
# ----------------------------------------------------------------------


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its projected Adam optimiser
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """Projected Adam on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]
        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search style)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program (LP) improvement step
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix H such that H @ g = a * g
    (linear convolution).  Returned as a CSR sparse matrix.
    """
    n = a.shape[0]
    size = 2 * n - 1
    rows = []
    cols = []
    data = []
    # column j corresponds to a shifted by j positions
    for j in range(n):
        length = size - j
        col_vals = a[:length]
        row_idx = np.arange(j, j + length)
        rows.append(row_idx)
        cols.append(np.full(length, j, dtype=int))
        data.append(col_vals)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    return csr_matrix((data, (rows, cols)), shape=(size, n))


def _lp_improve(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform the iterated LP improvement described in the paper.
    Stops when no ℓ¹‑mass increase is possible or the time budget expires.
    Returns the improved vector (still on the simplex Σa = √(2n)).
    """
    a = a0.copy()
    n = a.size
    target_sum = a.sum()  # should be √(2n) but we keep the exact sum
    while time.time() < deadline - 1.0:
        M = _max_convolution_fft(a)

        # Build the Toeplitz convolution matrix for the current a
        H = _build_conv_matrix(a)

        # Linear constraints: H @ g <= M
        b_ub = np.full(H.shape[0], M, dtype=float)

        # Linear objective: maximise Σ g_i  ==>  minimise -Σ g_i
        c = -np.ones(n, dtype=float)

        # Variable bounds (non‑negative, also respect the 1000‑clamp)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        # Time remaining for this LP
        time_left = deadline - time.time()
        lp_opts = {"presolve": True, "time_limit": max(0.0, time_left)}

        try:
            res = linprog(c, A_ub=H, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options=lp_opts)
        except Exception as exc:
            print(f"LP solve failed: {exc}")
            break

        if res.status != 0:
            # infeasible / time‑out / numerical trouble
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= target_sum * (1.0 + 1e-9):
            # no ℓ¹‑mass increase ⇒ cannot improve further
            break

        # Scale back to the original simplex
        scale = target_sum / sum_g
        a = g * scale
        a = np.clip(a, 0.0, 1000.0)
        a = _project_to_simplex(a, target_sum)

        # Check that we really decreased the worst convolution entry
        new_M = _max_convolution_fft(a)
        if new_M >= M - 1e-12:
            # stagnation – stop
            break
    return a

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` under the given wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Seeding from a previously stored construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            prev = _project_to_simplex(prev,
                                       math.sqrt(2.0 * n0))
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0})  max‑conv={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 1)  Coarse optimisation (size ≈ 1200)
    # ------------------------------------------------------------------
    n_coarse = 1200
    target_coarse = math.sqrt(2.0 * n_coarse)

    if best_seq is not None and best_seq.size != n_coarse:
        a = _upsample_sequence(best_seq, n_coarse)
    else:
        a = _random_initial_sequence(n_coarse, rng)

    a = _project_to_simplex(a, target_coarse)

    # 1a)  Scan simple families, keep a small set of good seeds
    fam_candidates = _family_candidates(n_coarse, rng, top_k=6)
    # ensure the current vector participates
    score_cur = _max_convolution_fft(a)
    fam_candidates.append((score_cur, a))
    fam_candidates.sort(key=itemgetter(0))
    best_local = fam_candidates[0][1].copy()
    best_local_score = fam_candidates[0][0]

    # 1b)  LP improvement on each candidate (time‑controlled)
    for idx, (sc, vec) in enumerate(fam_candidates):
        if time.time() >= deadline - 30.0:
            break
        rem = deadline - time.time()
        per_vec_deadline = time.time() + min(12.0, rem * 0.2)
        improved = _lp_improve(vec, per_vec_deadline)
        sc2 = _max_convolution_fft(improved)
        if sc2 < best_local_score - 1e-12:
            best_local = improved.copy()
            best_local_score = sc2
            print(f"[{time.time():.1f}] LP improvement (candidate {idx}) "
                  f"score={best_local_score:.6f}")

    # 1c)  A short Adam run to smooth the shape
    if time.time() < deadline - 10.0:
        adam_deadline = deadline - 5.0
        best_local = _adam_optimize(best_local,
                                    target_coarse,
                                    adam_deadline)
        best_local_score = _max_convolution_fft(best_local)
        print(f"[{time.time():.1f}] Adam after LP (coarse) score={best_local_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Upsample to the target length (2000) and final polishing
    # ------------------------------------------------------------------
    n_target = 2000
    target_target = math.sqrt(2.0 * n_target)
    a_big = _upsample_sequence(best_local, n_target)

    # optional quick LP step at the final size (only one iteration,
    # because the matrix becomes noticeably larger)
    if time.time() < deadline - 30.0:
        lp_deadline = time.time() + 10.0
        a_big = _lp_improve(a_big, lp_deadline)
        print(f"[{time.time():.1f}] LP at target size – max‑conv="
              f"{_max_convolution_fft(a_big):.6f}")

    # Adam refinements
    if time.time() < deadline - 15.0:
        remaining = deadline - time.time()
        n_restart = max(1, int(remaining // 8.0))
        per_restart = remaining / n_restart
        best_big = a_big.copy()
        best_big_score = _max_convolution_fft(best_big)
        for r in range(n_restart):
            if time.time() >= deadline - 2.0:
                break
            dl = min(deadline - 0.5, time.time() + per_restart)
            start = best_big.copy() if r == 0 else _random_initial_sequence(n_target, rng)
            opt = _adam_optimize(start, target_target, dl)
            sc = _max_convolution_fft(opt)
            if sc < best_big_score - 1e-12:
                best_big = opt.copy()
                best_big_score = sc
                print(f"[{time.time():.1f}] Adam restart {r} improvement "
                      f"score={best_big_score:.6f}")
        a_big = best_big
        best_score = best_big_score

    # Deterministic hill‑climb for the exact max convolution
    if time.time() < deadline - 5.0:
        b_big = _full_convolution(a_big)
        hc_deadline = deadline - 0.5
        a_big, sc_hc = _hill_climb_deterministic(a_big, b_big, hc_deadline)
        if sc_hc < best_score - 1e-12:
            best_score = sc_hc
            print(f"[{time.time():.1f}] Hill‑climb final improvement "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Final clean‑up and evaluation
    # ------------------------------------------------------------------
    if a_big is None:
        # Fallback – uniform (should never happen)
        n_target = 2000
        a_big = np.full(n_target,
                        math.sqrt(2.0 * n_target) / n_target,
                        dtype=float)

    # Drop trailing near‑zeros for aesthetic output
    nz = np.where(a_big > 1e-12)[0]
    if nz.size:
        a_big = a_big[: nz[-1] + 1]
    else:
        a_big = a_big[:1]

    try:
        val = evaluate_sequence(a_big.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={a_big.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return a_big.tolist()
