"""
Propose a step‑function that minimises ``evaluate_sequence``.

The optimiser works in several stages:

1. **Diverse initialisation** – random draws from several families,
   now including an arcsine‑like shape whose convolution is close to
   flat.  The best few are kept.

2. **Ensemble averaging** – the mean of the top candidates is projected
   onto the ℓ¹–simplex (∑a = √(2 n)).

3. **Projected Adam on a smooth‑max surrogate** – a few restarts with a
   log‑sum‑exp approximation of the max entry of the autocorrelation.

4. **Exact sub‑gradient descent on the true max** – after Adam the
   worst convolution entry is attacked directly.

5. **Linear‑programming mixing step** – given the current sequence `f`,
   solve a small LP that finds a non‑negative vector `g` with maximal
   ℓ¹‑mass while respecting `f*g ≤ max(f*f)`.  Mixing `f←(1‑t)f+tg`
   (t≈0.2) flattens the autocorrelation further.  The LP is used only
   for moderate lengths (≤2000) to keep memory bounded.

6. **Deterministic hill‑climbing** – greedy moves that shift mass from
   the source that contributes most to the current worst entry to the
   destination that contributes least.

7. **Final polishing** – a longer Adam run and a last hill‑climb.

The routine respects a wall‑clock budget (default 1000 s) and returns
the best sequence found.
"""

import time
import math
import numpy as np
from operator import itemgetter

# optional SciPy (used for an LP‑mixing step)
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until the supplied wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Learning‑rate base (scaled with the simplex radius)
    base_lr = 0.12 * target_sum

    # smoothed‑max schedule
    beta = 1.5
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max entry of the autocorrelation.
    One iteration moves a step opposite to the sub‑gradient at the current worst
    convolution index.
    """
    a = a0.copy()
    n = a.size
    rng = np.random.default_rng()

    # step size schedule – start relatively large and decay slowly
    init_lr = 0.3 * target_sum / n
    t = 0

    while time.time() < deadline:
        t += 1
        b = _full_convolution(a)
        k = int(np.argmax(b))

        # sub‑gradient of max entry
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # diminishing learning‑rate
        lr = init_lr / math.sqrt(t + 1)

        a -= lr * grad
        a = _project_to_simplex(a, target_sum)

        # occasional tiny random jitter (helps escape plateaus)
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-5, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Linear‑programming mixing step
# ----------------------------------------------------------------------


def _lp_mix_step(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    One LP step as described in the statement:
    maximise Σ g_j subject to g≥0 and (a*g)_k ≤ M,
    where M = max(a*a).  The solution g is then rescaled to the simplex
    and mixed with a (t=0.2).  For n>2000 the step is skipped (memory).
    """
    n = a.size
    if not _SCIPY_AVAILABLE or n > 2000:
        # fallback – do nothing
        return a

    M = _max_convolution_fft(a)

    # avoid exact zeros – they would give an unbounded LP
    a_eps = a.copy()
    a_eps[a_eps < 1e-12] = 1e-12

    rows = 2 * n - 1

    # Build sparse matrix A where A[k, j] = a_{k-j} if 0≤k-j<n else 0.
    col_idx = np.repeat(np.arange(n), n)                     # len = n*n
    row_blocks = [np.arange(i, i + n) for i in range(n)]
    row_idx = np.concatenate(row_blocks)
    data = np.repeat(a_eps, n)

    A = csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))
    b_vec = np.full(rows, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)      # maximise sum(g)  → minimise -sum(g)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    if not res.success:
        return a

    g0 = res.x
    s0 = np.sum(g0)
    if s0 <= 0:
        return a

    # rescale to the required ℓ¹‑radius
    g = g0 * (target_sum / s0)

    # mix (t = 0.2) and re‑project
    t = 0.2
    a_new = (1.0 - t) * a + t * g
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)                 # 0..6
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.2)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                   # arcsine‑like shape
        eps = 1e-12
        a = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5) + eps)

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    # add a tiny Gaussian perturbation and re‑project
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Moves a (large) fraction of the responsible mass to a location that
    contributes the least to that entry.  The full convolution is updated in
    O(n) per move.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for destinations
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]

        # destination with minimal counterpart (different from i)
        order = np.argsort(counterpart[i_low:i_high + 1])
        j = None
        for cand in order:
            cand_idx = i_low + cand
            if cand_idx != i:
                j = cand_idx
                break
        if j is None:
            break

        delta = a[i] * 0.8          # move up to 80 % of the source mass
        if delta <= 0.0:
            break

        # ---- evaluate the move (linear update of the convolution) ----
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- accept move if it improves -------------------------------
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move by binary searching on delta
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small random jitter to get out of a plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the global time budget (default 1000 s) and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from a previous construction, if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:                # pragma: no cover
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – gradually increase dimension
    # ------------------------------------------------------------------
    schedule = [500, 2000, 5000, 10000, 20000]

    for n in schedule:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise (upsample previous best if possible)
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # 2a)  Random family scan – keep a handful of promising seeds
        # --------------------------------------------------------------
        cand_scores = []
        for _ in range(40):
            a = _random_initial_sequence(n, rng)
            cand_scores.append((_max_convolution_fft(a), a))
        # include the up‑sampled initial vector
        cand_scores.append((_max_convolution_fft(a_init), a_init))
        cand_scores.sort(key=itemgetter(0))
        top_candidates = cand_scores[:5]

        # --------------------------------------------------------------
        # 2b)  Ensemble averaging of the best candidates
        # --------------------------------------------------------------
        ensemble = np.mean([vec for _s, vec in top_candidates], axis=0)
        ensemble = _project_to_simplex(ensemble, target_sum)
        ens_score = _max_convolution_fft(ensemble)
        if ens_score < best_score - 1e-12:
            best_seq = ensemble.copy()
            best_score = ens_score
            print(f"[{time.time():.1f}] Ensemble average (n={n}) "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2c)  Optimisation of each top candidate
        # --------------------------------------------------------------
        for score, a in top_candidates:
            if time.time() >= deadline - 5.0:
                break

            # --- Adam on smooth‑max ---
            a_opt = _adam_optimize(a, target_sum, deadline - 2.0)

            # --- sub‑gradient descent on true max ---
            a_opt = _subgradient_optimize(a_opt, target_sum, deadline - 2.0)

            # --- occasional LP mixing (only for moderate n) ---
            if n <= 2000:
                mix_deadline = deadline - 2.0
                while time.time() < mix_deadline:
                    a_opt = _lp_mix_step(a_opt, target_sum)
                    # stop after a few mixes to keep within the budget
                    if time.time() > mix_deadline - 1.0:
                        break

            # --- deterministic hill‑climb ---
            b_cur = _full_convolution(a_opt)
            a_opt, score_opt = _hill_climb_deterministic(
                a_opt, b_cur, deadline - 1.0)

            if score_opt < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = score_opt
                print(f"[{time.time():.1f}] New best after optimisation "
                      f"(n={n}) score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2d)  Light random tweaks until we move to the next size
        # --------------------------------------------------------------
        while time.time() + 1.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target_sum)
            s = _max_convolution_fft(cand)
            if s < best_score - 1e-12:
                best_seq = cand
                best_score = s
                print(f"[{time.time():.1f}] Random tweak improvement "
                      f"(n={n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing on the remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        final_deadline = deadline - 1.0
        best_seq = _adam_optimize(best_seq, target, final_deadline)
        best_seq = _subgradient_optimize(best_seq, target, final_deadline)
        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, final_deadline)

        if final_score < best_score - 1e-12:
            best_seq = best_seq.copy()
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:                     # pragma: no cover
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:                # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
