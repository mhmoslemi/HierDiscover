"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Wide parametric sweep** – generate candidate sequences from several
   families (power‑law, exponential, power·exponential, double‑power,
   two‑exponential mixture, triangular).  A modest grid is examined and then
   many random draws while time permits.  The best candidate is kept.

2. **Projected sub‑gradient descent with momentum** – using the exact
   sub‑gradient of the maximal auto‑convolution entry, a multiplicative
   (exponentiated‑gradient) update together with a momentum term moves mass
   away from the most critical indices.  Adaptive step‑sizes and occasional
   jitter avoid stagnation.

3. **Deterministic mass‑transfer** – moves a small fraction of mass from a
   high‑contributing index (w.r.t. the current maximal convolution entry) to
   a low‑contributing one, accepting only improving moves.

4. **Simulated‑annealing polishing** – a cheap SA that swaps a modest amount
   of mass between donor/receiver indices, accepting worsening moves with a
   linear cooling schedule.

The routine respects the supplied time budget (seconds) and returns the best
sequence found (trailing near‑zero entries are removed).  The global variable
`height_sequence_1` is stored for possible warm‑starts on subsequent calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / scoring utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 size sufficient for a linear convolution of length 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    conv = _conv_fft(a)
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  Assumes ℓ¹‑norm > 0."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i / n)
    a *= target_sum / a.sum()
    return a


def _pow_exp_sequence(n: int, p: float, lam: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-lam * i / n)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    a = (np.arange(n, dtype=np.float64)[::-1] + 1.0)  # n,...,1
    a *= target_sum / a.sum()
    return a


def _double_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:

    a = ((i + 1) * (n - i)) ** (-p)

# Sub‑gradient descent (multiplicative update with momentum)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)  # momentum term
    gamma = 0.85
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of max entry (2·a_{k_max‑i} on feasible indices)
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        # momentum
        m = gamma * m + (1.0 - gamma) * grad

        # safe multiplicative step size
        max_m = np.max(m)
        if max_m == 0:
            break
        safe_eta = np.min(np.where(m > 0, a / m, np.inf))
        eta = min(0.6 * safe_eta, 1.0 / np.sqrt(it + 1))

        # exponentiated‑gradient update
        a = a * np.exp(-eta * m)

        # re‑normalise to the ℓ¹‑simplex
        a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] subgrad it={it:5d} score={best_val:.6f}")

        # occasional jitter to shake out plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val

def _mass_transfer_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from a high‑contributing index to a low‑contributing one."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] mass‑transfer improve → {val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # re‑normalise for numerical drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Cheap SA that moves a small amount of mass from high‑weight to low‑weight indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.015 * best_val

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        wsum = weight.sum()
        if wsum == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved → {best_val:.6f}")

        if rng.random() < 0.004:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives more freedom, still fast enough)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 300:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (from previous run if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init.size != n:
            init = np.interp(
                np.linspace(0, 1, n), np.linspace(0, 1, init.size), init
            )
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑started from stored sequence (n={n})")
    else:
        # simple power‑law as a fallback
        init = _power_sequence(n, p=0.6, target_sum=target_sum)
        print(f"[{time.time():.2f}] Warm‑started from default power‑law (n={n})")

    # --------------------------------------------------------------
    # Stage 1 – broad parametric / random shape search
    # --------------------------------------------------------------
    shape_deadline = time.time() + min(180.0, (deadline - time.time()) * 0.30)
    best_seq, best_score = _search_shapes(n, target_sum, shape_deadline)
    if best_seq is not None and best_score < _score(init):
        init = best_seq
        best_score = best_score
        print(f"[{time.time():.2f}] Updated warm‑start after shape search, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + (deadline - time.time()) * 0.45
        a_sub, val_sub = _subgradient_descent(init, target_sum, subgrad_deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
            print(f"[{time.time():.2f}] After sub‑gradient descent, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.15
        a_mt, val_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
            print(f"[{time.time():.2f}] After mass‑transfer, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_deadline)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
            print(f"[{time.time():.2f}] After SA, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]      # trim trailing near‑zeros

    final_seq = arr.tolist()

    # final validation using the provided evaluation function
    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a future call
    globals()["height_sequence_1"] = final_seq
    return final_seq
