import numpy as np
import time
import math

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return float(conv.max())

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _project_to_simplex(v, s):
    # Euclidean projection.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _power_law_sequence(n, p, target_sum):
    a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _random_simplex(n, target_sum):
    # sample exponential (or gamma) then project
    x = np.random.exponential(scale=1.0, size=n)
    return _project_to_simplex(x, target_sum)

def _subgradient_descent(a0, target_sum, deadline,
                         step_factor=0.5, gamma=0.9,
                         jitter_interval=1000, jitter_scale=1e-3):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} best={best_val:.6f}")
        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        it += 1
    return best_a, best_val

def _permutation_annealing(a, target_sum, deadline, init_score=None):
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()
    start_temp = 0.2
    end_temp = 1e-5
    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best={best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)
    return best_seq, best_score

def _deterministic_mass_transfer(a0, target_sum, deadline,
                                 start_temp=0.5, end_temp=1e-4, low_frac=0.05):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)
        if donor == receiver:
            it += 1
            continue
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)
        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)
        it += 1
    return best_a, best_score

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, b

def _annealed_gradient_descent_ext(a0, deadline, target_sum,
                                   max_beta=1e5, step_factor=0.7, gamma=0.99):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Softmax GD it={it} β={beta:.1f} best={best_val:.6f}")
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1
    return best_a, best_val

"""
Hybrid random‑search + sub‑gradient + annealing for the max‑self‑convolution
problem.

The evaluation metric is   2·n·max(a∗a) / (∑a)².
Because the algorithm always keeps ∑a = √(2n), minimising the score is
equivalent to minimising the maximal entry of the linear convolution a∗a.

The search proceeds in five stages, each respecting the overall time budget:

1. **Exploratory random pool** – power‑law shapes, Dirichlet samples,
   warm‑starts from the previous run and small jitter.  Each candidate is
   given a brief sub‑gradient refinement; the best is retained.

2. **Long sub‑gradient descent** – a momentum based sub‑gradient method on the
   true max‑convolution, with occasional Gaussian jitter.

3. **Deterministic mass‑transfer** – greedy moves that shift a small amount
   of mass from indices that contribute most to the current peak towards
   those that contribute least, with a temperature schedule.

4. **Permutation annealing** – simulated‑annealing swaps of two positions,
   accepting deteriorations with a decaying temperature.

5. **Final soft‑max projected GD** – a high‑β soft‑max approximation of the
   max‑convolution, projected onto the ℓ¹‑simplex, for fine‑tuning.

The sequence length is chosen larger for larger budgets (up to ~30 k) to
exploit the extra degrees of freedom.  All operations use FFT‑based
convolutions (O(n log n)).  The final sequence is stored in the global
variable ``height_sequence_1`` for warm‑starts in subsequent runs.
"""

import math
import time

import numpy as np


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float) -> np.ndarray:
    """Sample a random point on the simplex using exponential variables."""
    x = np.random.exponential(scale=1.0, size=n)
    return _project_to_simplex(x, target_sum)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    step_factor: float = 0.5,
    gamma: float = 0.9,
    jitter_interval: int = 500,
    jitter_scale: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """Momentum sub‑gradient descent on max(a*a)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2 * a[kmax - i] for valid indices
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
    low_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Greedy mass transfers that target the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak
        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------
def _permutation_annealing(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    init_score: float | None = None,
) -> tuple[np.ndarray, float]:
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best={best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Soft‑max gradient descent (high‑β polishing)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


def _annealed_gradient_descent_ext(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 1e5,
    step_factor: float = 0.7,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Softmax GD it={it} β={beta:.1f} best={best_val:.6f}"
            )

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with low maximal self‑convolution.
    The routine adapts the problem size to the allotted time and returns a
    Python list ready for ``evaluate_sequence``.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more freedom)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 30000
    elif budget_s >= 400:
        n = 20000
    else:
        n = 12000

    target_sum = math.sqrt(2.0 * n)
    rng = np.random.default_rng()

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – exploratory random pool + short sub‑grad refinement
    # --------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.20
    while time.time() < phase1_deadline:
        # pick a generator
        gen = rng.integers(0, 3)
        if gen == 0:
            p = rng.uniform(0.40, 0.70)
            seq = _power_law_sequence(n, p, target_sum)
        elif gen == 1:
            seq = _random_simplex(n, target_sum)
        else:
            if "height_sequence_1" in globals():
                prev = np.array(
                    globals()["height_sequence_1"], dtype=np.float64
                )
                seq = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(prev)),
                    prev,
                )
                seq = _project_to_simplex(seq, target_sum)
            else:
                p = rng.uniform(0.45, 0.60)
                seq = _power_law_sequence(n, p, target_sum)

        # small jitter for diversity
        seq += rng.normal(scale=target_sum * 0.02, size=n)
        seq = np.maximum(seq, 0.0)
        seq = _project_to_simplex(seq, target_sum)

        # brief sub‑gradient descent (≈150 ms or ~200 iterations)
        sg_deadline = time.time() + min(0.15, phase1_deadline - time.time())
        seq, val = _subgradient_descent(
            seq,
            target_sum,
            sg_deadline,
            step_factor=0.6,
            gamma=0.9,
            jitter_interval=500,
            jitter_scale=5e-4,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Phase‑1 new best={best_val:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_val = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – long sub‑gradient descent on the current best
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        sg_deadline = time.time() + (deadline - time.time()) * 0.35
        seq, val = _subgradient_descent(
            best_seq,
            target_sum,
            sg_deadline,
            step_factor=0.5,
            gamma=0.95,
            jitter_interval=500,
            jitter_scale=1e-3,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Phase‑2 subgrad best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 20:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        seq, val = _deterministic_mass_transfer(
            best_seq,
            target_sum,
            mt_deadline,
            start_temp=0.6,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Phase‑3 mass‑transfer best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 10:
        perm_deadline = time.time() + (deadline - time.time()) * 0.15
        seq, val = _permutation_annealing(
            best_seq, target_sum, perm_deadline, init_score=best_val
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Phase‑4 permutation best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final high‑β soft‑max projected gradient descent
    # --------------------------------------------------------------
    if time.time() + 1.0 < deadline:
        seq, val = _annealed_gradient_descent_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.5,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Final polish best={best_val:.6f}")

    # --------------------------------------------------------------
    # Post‑process: trim trailing zeros and return as list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional verification (does not affect the return value)
    try:
        score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Evaluated final score = {score:.6f}")
    except Exception:
        pass

    # Store for warm‑starts in subsequent runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
