def _subgradient_descent(a0, target_sum, deadline, step_factor=0.9, min_step_factor=0.1):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_score = _score(a)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        # compute gradient on support
        n = a.size
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]
        # compute safe step
        mask = grad > 0
        if np.any(mask):
            safe_eta = np.min(a[mask] / grad[mask])
        else:
            safe_eta = 0.0
        # adaptive factor: decaying with it (e.g., 1/(1+it))
        factor = max(step_factor / np.sqrt(it+1), min_step_factor)
        eta = factor * safe_eta
        if eta <= 0:
            # not possible to move, add jitter
            if it % 10 == 0:
                jitter = rng.normal(scale=target_sum * 0.0005, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = a * (target_sum / a.sum())
            it += 1
            continue
        # gradient step
        a = a - eta * grad
        # enforce nonnegativity
        a = np.maximum(a, 0.0)
        # reproject to simplex
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_score:

            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        it += 1
    return best, best_score

"""
Propose a low‑score step‑function.

Algorithm outline
-----------------
1. **Warm‑start** – use the previously discovered sequence or a simple
   power‑law shape.
2. **Small‑scale sub‑gradient optimisation** – work on a reduced problem
   (≈ 4000 entries) where many cheap sub‑gradient steps can be performed.
   The sub‑gradient of the exact objective  max_k (a*a)_k  is used; a
   large step (≈ 90 % of the safe step) together with a diminishing
   factor 1/√t drives rapid progress.
3. **Upscale** – interpolate the best small solution to the target length
   (≈ 12000) and renormalise the ℓ¹‑norm.
4. **Large‑scale sub‑gradient descent** – the same aggressive scheme is
   repeated on the full size sequence.
5. **Deterministic mass‑transfer refinement** – move a modest fraction
   of mass from a donor (large contribution to the current maximal
   convolution entry) to a low‑contribution receiver; the fraction is
   increased to 8 % to accelerate convergence.
6. **Random perturbation + short sub‑gradient bursts** – occasional
   stochastic jitter followed by a few sub‑gradient iterations to escape
   shallow local minima.
7. **Final simulated‑annealing polish** – a cheap SA that swaps small
   amounts of mass, with a slightly higher temperature and larger moves.
8. **Trim trailing zeros**, store the result for warm‑starts and return
   the sequence.

All stages respect a global time budget (default 1000 s) and the
sequence is always projected onto the simplex {x ≥ 0, Σx = √(2 n)}.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 length for an FFT of size 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective  2·n·max(a*a) / (sum(a)²).  The ℓ¹‑norm is assumed >0."""
    n = a.size
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Aggressive projected sub‑gradient descent.
    step_factor ∈ (0,1] multiplies the maximal safe step (so 0.9 ≈ 90 % of the
    farthest we can move without violating non‑negativity).
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)
    n = a.size
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # gradient on the active entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # safe step (largest η that keeps a ≥ 0)
        mask = grad > 0
        if np.any(mask):
            safe_eta = np.min(a[mask] / grad[mask])
        else:
            safe_eta = 0.0

        # diminishing factor to guarantee convergence
        eta = min(step_factor * safe_eta, 1.0 / np.sqrt(it + 1))

        if eta > 0.0:
            a = a - eta * grad
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (larger moves)
# ----------------------------------------------------------------------


def _mass_transfer_refine2(a0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          max_frac: float = 0.08) -> tuple[np.ndarray, float]:
    """Move up to *max_frac* of a donor’s mass to a low‑contribution receiver."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]                     # contributions to the max entry

        donor = idx[np.argmax(weight)]                   # biggest contributor
        # pick a receiver among the 10 % smallest contributors
        low_thr = np.percentile(weight, 10)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = low_cands[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        delta = a[donor] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise occasionaly (numerical drift)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best, best_val


# ----------------------------------------------------------------------
# Simulated annealing (slightly more aggressive)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Cheap SA that swaps small mass fractions, with a higher temperature."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.025 * best_val   # a bit hotter than before

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        wsum = weight.sum()
        if wsum == 0.0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.35          # allow larger swaps
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())
        new_val = _score(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_val:.6f}")

        # Very occasional random jitter
        if rng.random() < 0.003:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size according to the time budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_target = 14000
    elif budget_s >= 500:
        n_target = 10000
    else:
        n_target = 6000

    target_sum = np.sqrt(2.0 * n_target)

    # ------------------------------------------------------------------
    # Warm‑start (previous run or simple power‑law)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n_target)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n_target, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 0 – aggressive sub‑gradient search on a reduced scale
    # ------------------------------------------------------------------
    small_n = min(4000, n_target // 3)
    small_target = np.sqrt(2.0 * small_n)
    small_deadline = time.time() + 0.25 * (deadline - time.time())
    small_init = _resize_sequence(init, small_n)
    small_init = _project_to_simplex(small_init, small_target)

    best_small, best_small_val = _subgradient_descent(
        small_init, small_target, small_deadline, step_factor=0.92)

    # ------------------------------------------------------------------
    # Upscale to the full size
    # ------------------------------------------------------------------
    best = _resize_sequence(best_small, n_target)
    best = best * (target_sum / best.sum())
    best_val = _score(best)
    print(f"[{time.time():.2f}] After up‑scaling score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – large scale sub‑gradient descent (aggressive)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        large_deadline = time.time() + 0.30 * (deadline - time.time())
        best, best_val = _subgradient_descent(
            best, target_sum, large_deadline, step_factor=0.90)
        print(f"[{time.time():.2f}] After large sub‑grad score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deterministic mass‑transfer refinement (larger moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + 0.15 * (deadline - time.time())
        best, best_val = _mass_transfer_refine2(best, target_sum,
                                                mt_deadline, max_frac=0.08)
        print(f"[{time.time():.2f}] After mass‑transfer score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – random perturbations + short sub‑gradient bursts
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 6.0:
        remaining = deadline - time.time()
        burst_deadline = time.time() + min(5.0, 0.12 * remaining)

        pert = best + rng.normal(scale=target_sum * 0.001, size=best.shape)
        pert = np.maximum(pert, 0.0)
        pert = _project_to_simplex(pert, target_sum)

        pert, val = _subgradient_descent(pert, target_sum,
                                         burst_deadline, step_factor=0.88)
        if val < best_val:
            best = pert
            best_val = val
            print(f"[{time.time():.2f}] Perturb+subgrad improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best, best_val = _simulated_annealing(best, target_sum, deadline)
        print(f"[{time.time():.2f}] After SA polish score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean up, final evaluation and return
    # ------------------------------------------------------------------
    arr = np.array(best, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # store for possible warm‑starts in a later invocation
    globals()["height_sequence_1"] = final_seq

    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
