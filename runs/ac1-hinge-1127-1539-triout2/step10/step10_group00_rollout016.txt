The goal is to minimise  

\[
\mathrm{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

where `a*a` is the linear self‑convolution of a non‑negative sequence `a`.  
After normalising the ℓ¹‑norm to \(\sqrt{2n}\) the score is simply the
maximum convolution entry.  

The programme below follows a *diversified* search strategy:

1. **Problem size** – `n` is chosen from the time budget (12000 for a full
   second).  
   `target_sum = sqrt(2·n)` is the required ℓ¹‑norm.

2. **Warm‑starts** –  
   * the previously found sequence (`height_sequence_1`) if it exists,  
   * a power‑law sequence (the old baseline).

3. **Explorative parametric families** – random members of the
   *power‑law × exponential*, *mixture of two exponentials* and *piecewise‑linear*
   families are generated, scored and the best few are kept.
   This enlarges the reachable region of the search space.

4. **Accelerated sub‑gradient descent (Nesterov)** –  
   For every promising start we run a projected Nesterov sub‑gradient method
   on the exact nonsmooth objective  
   `max(a*a)`.  The step size is chosen as the largest feasible one
   (so no coordinate becomes negative) and decays like `1/√t`.  
   The method is cheap (one FFT per iteration) and converges faster than the
   plain sub‑gradient loop used previously.

5. **Deterministic mass‑transfer** – a cheap greedy refinement that moves a
   small fraction of mass from the index contributing most to the current
   maximal convolution entry to a low‑contributing index.

6. **Simulated‑annealing fine‑tuning** – stochastic local search with a
   linear cooling schedule, identical to the previous pipeline but executed
   after the Nesterov phase.

7. **Final exact sub‑gradient pass** – a short classic sub‑gradient descent
   to polish the result.

All stages respect the global time budget and stop as soon as the budget is
exhausted.  The best sequence found is trimmed of trailing zeros, stored in
`height_sequence_1` for warm‑starting a future call, and returned as a plain
list of `float`s.

"""
Low‑score step‑function search.

The algorithm mixes diverse random parametric starts with a fast Nesterov
sub‑gradient optimiser for the exact max‑convolution objective, followed by
deterministic mass‑transfer, simulated‑annealing and a final classic
sub‑gradient polish.  The whole pipeline respects a 1000 s wall‑time budget
and returns the best non‑negative sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Next power‑of‑2 length ≥ 2·n‑1 (used for zero‑padded FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) via FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(sum(a)²).  Assumes sum>0."""
    n = a.size
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _make_powerlaw_exp(n: int, target_sum: float, rng: np.random.Generator):
    """Power‑law with exponential decay."""
    p = rng.uniform(0.2, 1.5)
    lam = rng.exponential(1.0) * 4.0   # typical λ in [0,12]
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1) ** (-p) * np.exp(-lam * i / n)
    seq *= target_sum / seq.sum()
    return seq


def _make_two_exp_mix(n: int, target_sum: float, rng: np.random.Generator):
    """Mixture of two exponentials."""
    lam1 = rng.uniform(0.0, 12.0)
    lam2 = rng.uniform(0.0, 12.0)
    w = rng.random()
    i = np.arange(n, dtype=np.float64)
    seq = w * np.exp(-lam1 * i / n) + (1.0 - w) * np.exp(-lam2 * i / n)
    seq *= target_sum / seq.sum()
    return seq


def _make_piecewise_linear(n: int, target_sum: float, rng: np.random.Generator):
    """K random control points linearly interpolated."""
    K = rng.integers(3, 9)            # 3 … 8 knots
    pos = np.sort(rng.random(K))
    heights = rng.random(K) + 1e-3    # keep strictly positive
    x = np.linspace(0.0, 1.0, n)
    seq = np.interp(x, pos, heights)
    seq *= target_sum / seq.sum()
    return seq


def _make_sine_shape(n: int, target_sum: float):
    """Half‑sine shape (strictly positive)."""
    i = np.arange(n, dtype=np.float64)
    seq = np.sin(np.pi * (i + 0.5) / n) + 0.001
    seq *= target_sum / seq.sum()
    return seq

# Nesterov accelerated sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _nesterov_subgradient(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Accelerated sub‑gradient method for max(a*a) with simplex projection."""
    a = _project_to_simplex(a0, target_sum)
    best = a.copy()
    best_val = _score(best)

    a_prev = a.copy()
    t = 0
    n = a.size
    rng = np.random.default_rng()

    while time.time() < deadline:
        # Nesterov extrapolation
        if t == 0:
            y = a
        else:
            beta = (t - 1) / (t + 2)
            y = a + beta * (a - a_prev)

        conv = _conv_fft(y)
        k_max = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(y)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * y[partner]

        # safe step size (keep non‑negative)
        safe_eta = np.min(
            np.where(grad > 0, y / grad, np.inf)
        )
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(t + 1))

        a_new = y - eta * grad
        a_new = _project_to_simplex(a_new, target_sum)

        # bookkeeping
        a_prev = a
        a = a_new
        val = _score(a)

        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Nesterov it={t:5d} score={best_val:.6f}")

        # occasional random jitter to jump out of plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        t += 1

    return best, best_val

# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Greedy transfer of a modest mass fraction from high‑ to low‑contributing indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Cheap SA that moves a small amount of mass, accepting worsening moves with a cooling schedule."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.015 * best_val  # initial temperature

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        wsum = weight.sum()
        if wsum == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        low_thr = np.percentile(weight, 8)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved best score={best_val:.6f}")

        # occasional random jitter
        if rng.random() < 0.004:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

    return best, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient pass (final polish)
# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Classical sub‑gradient descent with projection onto the simplex."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    n = a.size

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the supplied time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initial candidates (warm‑start + random parametric families)
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # 1. previous successful sequence if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        prev = _resize_sequence(prev, n)
        prev = prev * (target_sum / prev.sum())
        candidates.append((_score(prev), prev.copy()))

    # 2. simple power‑law baseline
    baseline = _power_law_sequence(n, 0.55, target_sum)
    candidates.append((_score(baseline), baseline.copy()))

    # 3. random parametric exploration (≈ 12 % of total budget)
    param_deadline = time.time() + (deadline - time.time()) * 0.12
    while time.time() < param_deadline:
        mode = rng.integers(4)
        if mode == 0:
            seq = _make_powerlaw_exp(n, target_sum, rng)
        elif mode == 1:
            seq = _make_two_exp_mix(n, target_sum, rng)
        elif mode == 2:
            seq = _make_piecewise_linear(n, target_sum, rng)
        else:
            seq = _make_sine_shape(n, target_sum)
        candidates.append((_score(seq), seq))

        # prune to keep only the best few (speed‑up later stages)
        if len(candidates) > 30:
            candidates.sort(key=lambda x: x[0])
            candidates = candidates[:20]

    # --------------------------------------------------------------
    # Select a handful of promising starts
    # --------------------------------------------------------------
    candidates.sort(key=lambda x: x[0])
    top_seqs = [seq for _, seq in candidates[:5]]
    best_seq = top_seqs[0].copy()
    best_val = _score(best_seq)

    # --------------------------------------------------------------
    # Nesterov accelerated sub‑gradient descent (≈ 50 % of remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        nesterov_budget = remaining * 0.55
        per_seq = nesterov_budget / max(1, len(top_seqs))
        for seq in top_seqs:
            sub_dead = time.time() + per_seq
            a_opt, val_opt = _nesterov_subgradient(seq, target_sum, sub_dead)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
            if time.time() >= deadline:
                break
        print(f"[{time.time():.2f}] After Nesterov best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer refinement (≈ 20 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_dead = time.time() + (deadline - time.time()) * 0.20
        best_seq, best_val = _mass_transfer_refine(best_seq, target_sum, mt_dead)
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing fine‑tuning (≈ 20 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sa_dead = time.time() + (deadline - time.time()) * 0.20
        best_seq, best_val = _simulated_annealing(best_seq, target_sum, sa_dead)
        print(f"[{time.time():.2f}] After SA best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Final exact sub‑gradient polish (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = deadline
        best_seq, best_val = _subgradient_max_descent(best_seq, target_sum, sub_dead)
        print(f"[{time.time():.2f}] After final sub‑gradient best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Clean‑up and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # store for next call
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluate_sequence()
    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
