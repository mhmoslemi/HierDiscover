"""
Propose a low‑score step‑function sequence.

The algorithm consists of three phases:

1. **Fast parametric seeding** – generate a few hundred candidates from
   simple families (power‑law, power‑law+exponential, mixtures, logistic)
   and keep the best one.

2. **Deterministic mass‑transfer** – a greedy exact move that reduces the
   current maximal convolution entry to (approximately) the second‑largest
   one.  This cheap phase often yields a visible improvement over the
   raw parametric seed.

3. **Long projected Adam‑GD** – minimise a soft‑max approximation of the
   convolution peak while staying on the simplex Σa = √(2 n).  The whole
   remaining time budget is spent here, using a high final β (≈5·10⁵)
   and an Adam‑style step‑size adaptation.

All steps respect the non‑negativity and sum constraints, and the final
sequence is trimmed of trailing zeros before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective for a (the value returned by evaluate_sequence
    when Σa = √(2 n)).  It is simply the maximal entry of the linear
    convolution a*a.
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    # the normalisation Σa = √(2 n) makes the denominator 2 n,
    # therefore the score equals max_b.
    return float(max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used for the seeding phase)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) and rescaled to have Σa = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-q·i/n)  (exponential tail) and rescaled."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv) where conv = a*a (full convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)               # soft‑max weights (length 2n‑1)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # the gradient w.r.t. a (only the central n entries matter)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max proxy
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 5e5,
                           step_factor: float = 0.5,
                           gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the soft‑max approximation.
    The whole remaining time up to *deadline* is used.
    Returns the best sequence seen and its (exact) score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5               # start with a very smooth soft‑max
    rng = np.random.default_rng()
    m = np.zeros_like(a)          # first moment estimate
    v = np.zeros_like(a)          # second moment estimate
    eps = 1e-8
    it = 0
    beta_inc_iter = 250           # double β every this many iterations

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)

        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the simplex (enforces Σa = target_sum, a ≥ 0)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to avoid exact plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # anneal β
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (greedy reduction of the current peak)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly reduce the current maximal convolution entry.
    For the currently maximal index k, the algorithm moves just enough mass
    from the most harmful position (largest contribution to that entry) to
    the least harmful one so that the peak becomes (approximately) the second‑largest
    convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # pick the first maximal index
        k0 = idx_max[0]

        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution weights w_i = a_i + a_{k0-i}
        w = a[idx] + a[k0 - idx]

        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        # contributions of donor / receiver to the peak
        contrib_d = a[k0 - donor]
        contrib_r = a[k0 - receiver]

        denom = 2.0 * (contrib_d - contrib_r)
        if denom <= 0.0:
            # no improving move possible for this k
            break

        # second largest convolution value (ignore the current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # amount that would bring the current peak down to the second one
        delta_needed = (max_val - second_max) / denom

        # never move more than half the donor's mass, and stay a bit below the exact target
        max_delta = min(a[donor] * 0.5, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta   # the total sum stays unchanged

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine obeys the time budget, returns the best sequence found,
    and prints progress information.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ---------------------------------------------------------------
    # Choose a problem size (larger n gives a richer search space)
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scale‑invariant)

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ---------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # (a) use the previous best if it exists (resized if necessary)
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_score:
            best_seq = prev.copy()
            best_score = val

    # (b) pure power‑law grid
    for p in np.linspace(0.35, 0.90, 25):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_seq, best_score = seq.copy(), val

    # (c) power‑law + exponential tail grid
    p_grid = np.linspace(0.35, 0.90, 15)
    q_grid = np.linspace(0.0, 8.0, 15)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_seq, best_score = seq.copy(), val

    # (d) mixture of two power‑laws (randomised)
    for _ in range(30):
        p1, p2 = rng.uniform(0.3, 1.2, size=2)
        w = rng.random()
        a1 = (np.arange(1, n + 1, dtype=np.float64) ** (-p1))
        a2 = (np.arange(1, n + 1, dtype=np.float64) ** (-p2))
        seq = w * a1 + (1 - w) * a2
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        if val < best_score:
            best_seq, best_score = seq.copy(), val

    # (e) logistic‑shaped profiles
    for mu_factor in [0.25, 0.5, 0.75]:
        for sigma_factor in [0.10, 0.15, 0.20]:
            x = np.arange(n, dtype=np.float64)
            mu = mu_factor * n
            sigma = sigma_factor * n
            seq = 1.0 / (1.0 + np.exp((x - mu) / sigma))
            seq = _project_to_simplex(seq, target_sum)
            val = _score(seq)
            if val < best_score:
                best_seq, best_score = seq.copy(), val

    print(f"[{time.time():.2f}] after parametric seeding best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – one quick deterministic mass‑transfer sweep
    # ---------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sweep_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.15)
        a_ref, val_ref = _deterministic_mass_transfer_exact(best_seq,
                                                            sweep_deadline,
                                                            target_sum)
        if val_ref < best_score:
            best_seq, best_score = a_ref.copy(), val_ref
            print(f"[{time.time():.2f}] after DMT sweep best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – long Adam‑style projected gradient descent (the core)
    # ---------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_opt, val_opt = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=5e5,
                                                step_factor=0.5,
                                                gamma=0.995)
        if val_opt < best_score:
            best_seq, best_score = a_opt.copy(), val_opt
            print(f"[{time.time():.2f}] after long ADAM best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 3 – final exact mass‑transfer polishing (short)
    # ---------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_ref2, val_ref2 = _deterministic_mass_transfer_exact(best_seq,
                                                              deadline,
                                                              target_sum)
        if val_ref2 < best_score:
            best_seq, best_score = a_ref2.copy(), val_ref2
            print(f"[{time.time():.2f}] final DMT best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing zeros and return the Python list
    # ---------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
