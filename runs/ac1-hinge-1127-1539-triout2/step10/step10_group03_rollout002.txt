def _swap_local_optimization(seq, deadline):
    a = seq.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    while time.time() < deadline:
        # compute remaining fraction
        elapsed = time.time() - start
        total = deadline - start
        tau = elapsed / total
        temp = max(1e-6, (1 - tau)) * cur_score * 0.1  # arbitrary schedule

        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # swap
        a[i], a[j] = a[j], a[i]  # in-place swap

        val = _score(a)
        if val < cur_score or rng.random() < np.exp(-(val - cur_score) / temp):
            cur_score = val
            if val < best_score:
                best_score = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Swap opt new best={best_score:.6f}")
        else:
            # revert
            a[i], a[j] = a[j], a[i]

    return best_a, best_score
```

"""
Propose a low‑score non‑negative step function.

The algorithm consists of three stages:

1️⃣  Exploration – generate many random shapes (power‑law,
    exponential, Dirichlet) of a fixed length (≈12 k) and run a
    projected gradient descent on a soft‑max approximation of the
    convolution peak.  Short runs (~20 s) are repeatedly performed
    until the global time budget is nearly exhausted; the best
    sequence found so far is retained.

2️⃣  Refinement – a deterministic “mass‑transfer’’ pass (greedy moves
    from large‑contributing indices to small ones) followed by a
    simple random‑swap local search that reshuffles the order of the
    values while preserving the ℓ¹‑norm.  Both passes respect the
    simplex constraint and aggressively lower the maximum self‑convolution.

3️⃣  Final polish – one more projected gradient descent with a very
    high soft‑max temperature (β up to 10⁶) and a final short
    deterministic transfer.

All steps use FFT‑based convolutions for speed, Euclidean projection
onto the simplex {x ≥ 0, Σx = √(2 n)} and keep the sum of the
coefficients fixed (the objective is scale‑invariant).  The best
sequence found before the deadline is returned and stored in the
global ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective & simple utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this reduces to max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max approximation of max‑convolution (and its gradient)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum PSD on the soft‑max approximation of the peak of the self‑convolution.
    The iterate is projected onto the ℓ¹‑simplex after each step.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase temperature gradually
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (greedy + Metropolis)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’ (large
    contribution to the current maximal convolution entry) to a “receiver’’
    (small contribution).  Moves are accepted with a Metropolis rule whose
    temperature decays exponentially with elapsed time.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight contributed by each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # choose a low‑weight receiver among the smallest ``low_frac`` entries
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bound the amount of mass we move
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to avoid stagnation
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random‑swap local ordering optimisation
# ----------------------------------------------------------------------


def _swap_local_optimisation(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Randomly swaps pairs of entries while keeping the ℓ¹‑norm.
    Accepts worsening swaps with a slowly decreasing temperature.
    """
    a = seq.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        # temperature proportional to remaining fraction
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = max(1e-8, (1.0 - frac)) * cur_score * 0.1

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        a[i], a[j] = a[j], a[i]          # swap
        val = _score(a)

        if val < cur_score or rng.random() < np.exp(-(val - cur_score) / temp):
            cur_score = val
            if val < best_score:
                best_score = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Swap opt new best={best_score:.6f}")
        else:
            a[i], a[j] = a[j], a[i]      # revert

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple alternating ordering (large‑small pattern)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder by placing the largest value at the leftmost position,
    the second largest at the rightmost, the third largest next to the left,
    etc.
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine runs a multi‑stage optimisation within the time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n generally gives a better bound,
    # but we keep it modest enough to stay within the time budget.
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 15000
    else:
        n = 12000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build an initial pool of diverse shapes (all already on the target simplex)
    # --------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.80, 9):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a *= target_sum / a.sum()
        init_pool.append(a)

    # exponential family
    for lam in np.linspace(0.001, 0.015, 7):
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a *= target_sum / a.sum()
        init_pool.append(a)

    # random Dirichlet‑like points
    for _ in range(8):
        a = rng.random(n)
        a *= target_sum / a.sum()
        init_pool.append(a)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            # simple nearest‑neighbour up‑sampling (preserves shape)
            old_len = prev.shape[0]
            idx = (np.linspace(0, old_len - 1, n)).astype(int)
            prev = prev[idx]
        prev = _project_to_simplex(prev, target_sum)
        init_pool.append(prev)

    # jittered copies to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        init_pool.append(seq_j)

    # --------------------------------------------------------------
    # Exploration phase – many short gradient‑descent runs
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 10.0:
        # pick a seed
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_sum * 0.006, size=n)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_sum)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 22.0, deadline - 4.0)
        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.6,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer refinement (short)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.18
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.6, end_temp=1e-4, low_frac=0.045
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Random‑swap reordering pass
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        swap_dead = time.time() + (deadline - time.time()) * 0.12
        a_swap, val_swap = _swap_local_optimisation(best_seq, swap_dead)
        if val_swap < best_score:
            best_score = val_swap
            best_seq = a_swap.copy()
            print(f"[{time.time():.2f}] After swap optimisation best={best_score:.6f}")

    # --------------------------------------------------------------
    # Apply a simple alternating large‑small ordering (often helpful)
    # --------------------------------------------------------------
    alt = _alternating_order(best_seq)
    if _max_convolution_fft(alt) < _max_convolution_fft(best_seq):
        best_seq = alt
        best_score = _score(best_seq)
        print(f"[{time.time():.2f}] Alternating order improved score to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing via a high‑β gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.45,
            gamma=0.995,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Tiny deterministic mass‑transfer pass (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(9.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.3, end_temp=1e-5, low_frac=0.04
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Logging of the final evaluated score
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
