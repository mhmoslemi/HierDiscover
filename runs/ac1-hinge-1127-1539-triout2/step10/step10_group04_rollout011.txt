"""
LP‑augmented search for a step function minimizing ``evaluate_sequence``.
The algorithm combines a cheap parametric family scan, a linear‑programming
improvement (the classic “linear program for g” step), a few Adam
iterations on the smooth‑max surrogate, and a final deterministic hill‑climb.
The LP stage is the main novelty compared with the reference code.
"""

import time, math, random
import numpy as np
from operator import itemgetter

# import optional scipy after checking availability

    import scipy.sparse as sp
    import scipy.optimize as opt

# Helper functions: projection, FFT conv, etc.
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# --- Random families ----------------------------------------------------
def _random_initial_sequence(n: int, rng) -> np.ndarray:
    target = math.sqrt(2.0 * n)
    # pick a random family
    family = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if family == 0:  # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)
    elif family == 1:  # exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)
    elif family == 2:  # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.2)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif family == 3:  # Dirichlet
        a = rng.dirichlet(alpha=np.ones(n)) * target
    else:  # near‑uniform with jitter
        a = np.full(n, target / n)
    # small jitter
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

# Smooth‑max surrogate and Adam
def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b

def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    a = a0.copy()
    n = a.size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning rate proportional to target sum
    base_lr = 0.12 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
    return a

# Deterministic hill‑climb (same as before but with small tweaks)
def _hill_climb(a0: np.ndarray, b0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M_idx = int(np.argmax(b))
        # find source index i that contributes most to b[M]
        i_low = max(0, M_idx - (n - 1))
        i_high = min(n - 1, M_idx)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M_idx - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-12:
            break

        # counterpart for each potential destination
        # counterpart[j] = a[M - j]
        j_range = np.arange(i_low, i_high + 1, dtype=int)
        counterpart = a[M_idx - j_range]

        # choose destination with minimal counterpart (excluding i)
        order = np.argsort(counterpart)
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # amount of mass to move
        delta = a[i] * 0.5
        if delta <= 0:
            break

        # update b efficiently
        # b_new = b - 2*delta*shifted a (from i) + 2*delta*shifted a (from j) …
        b_new = b.copy()
        b_new[i : i + n] -= 2.0 * delta * a
        b_new[j : j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try halving delta
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i : i + n] -= 2.0 * delta2 * a
                b_test[j : j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # occasional jitter
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)
    return best_a, best_score

# Linear‑programming improvement
def _lp_step(f: np.ndarray, target_sum: float, M: float, mix: float = 0.5) -> np.ndarray:
    """
    Solve   max Σ g_i   s.t. g_i ≥ 0, (f * g)_k ≤ M  (for all k).
    Scale the solution to the target sum and return the convex mixture
    f' = (1‑mix)·f + mix·g .
    """
    if not _SCIPY_AVAILABLE:
        return f  # fallback – no LP support

    n = f.shape[0]
    # Build sparse constraint matrix A (2n‑1 rows, n cols)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_low = max(0, k - (n - 1))
        j_high = min(k, n - 1)
        # for each j in this interval coefficient = f_{k-j}
        for j in range(j_low, j_high + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                     # maximise sum(g)  <=> minimise -sum(g)
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Using HiGHS interior‑point solver; fallback to default if it fails
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
    except Exception:
        # a very small fallback – return f unchanged
        return f

    if not res.success:
        return f

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 < 1e-12:
        return f

    scale = target_sum / sum_g0
    g = g0 * scale

    # convex combination
    f_new = (1 - mix) * f + mix * g
    f_new = np.maximum(f_new, 0.0)
    f_new = _project_to_simplex(f_new, target_sum)
    return f_new

"""
LP‑augmented search for a step function minimizing ``evaluate_sequence``.

The algorithm works in three stages:

1. **Family scan** – cheap random generators (exponential, power‑law,
   mixed, Dirichlet) produce an initial vector with a low autocorrelation peak.
2. **LP refinement** – the classic “optimize g while keeping f·g ≤ M”
   linear programme (see the description in the prompt) is solved a few
   times.  After each solve the new vector is mixed with the current one,
   which provably reduces the maximal convolution entry.
3. **Smooth‑max Adam + deterministic hill‑climb** – a few Adam
   iterations on a log‑sum‑exp surrogate followed by an exact greedy
   hill‑climb finish the optimisation.

All steps respect the global time budget (default 1000 s) and the
result is projected onto the ℓ¹‑simplex Σa = √(2 n), so the objective
``evaluate_sequence`` reduces to the maximal self‑convolution entry.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Optional SciPy import (used for the linear programme)
# ----------------------------------------------------------------------
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x ≥ 0, Σ x = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random families – produce a point on the ℓ¹‑simplex
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng) -> np.ndarray:
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if family == 0:               # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)
    elif family == 1:             # exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)
    elif family == 2:             # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.2)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif family == 3:             # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target
    else:                         # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient – for Adam optimisation
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smoothed max (1/β)·log Σ exp(β·b) with b = a*a."""
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the entries of b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until ``deadline``.
    """
    a = a0.copy()
    n = a.size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.12 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 1e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climbing on the exact convolution
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, b0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves mass from the index that
    contributes most to the current worst convolution entry towards a
    location that contributes the least.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M_idx = int(np.argmax(b))

        i_low = max(0, M_idx - (n - 1))
        i_high = min(n - 1, M_idx)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M_idx - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-12:
            break

        j_range = np.arange(i_low, i_high + 1, dtype=int)
        counterpart = a[M_idx - j_range]
        order = np.argsort(counterpart)

        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.5
        if delta <= 0.0:
            break

        # linear update of the convolution (O(n))
        b_new = b.copy()
        b_new[i : i + n] -= 2.0 * delta * a
        b_new[j : j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try shrinking the move geometrically
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i : i + n] -= 2.0 * delta2 * a
                b_test[j : j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “g‑optimization” step)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, target_sum: float, M: float, mix: float = 0.5) -> np.ndarray:
    """
    Solve

        max   Σ g_i
        s.t.  g_i ≥ 0,
              (f * g)_k ≤ M   for all k.

    The solution ``g`` is scaled to ``target_sum`` and convexly mixed
    with ``f`` using the factor ``mix``.
    """
    if not _SCIPY_AVAILABLE:        # pragma: no cover
        return f

    n = f.shape[0]
    rows, cols, data = [], [], []

    for k in range(2 * n - 1):
        j_low = max(0, k - (n - 1))
        j_high = min(k, n - 1)
        for j in range(j_low, j_high + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                     # maximise Σ g_i  → minimise −Σ g_i
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=float)

    try:
        res = opt.linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds,
            method='highs', options={'presolve': True}
        )
    except Exception:                 # pragma: no cover
        return f

    if not res.success:
        return f

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 < 1e-12:
        return f

    scale = target_sum / sum_g0
    g = g0 * scale

    f_new = (1 - mix) * f + mix * g
    f_new = np.maximum(f_new, 0.0)
    f_new = _project_to_simplex(f_new, target_sum)
    return f_new
