        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)   # indices i in [lo, hi]
        partner = k_max - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]
        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]

            receiver = rng.integers(n)
        if donor == receiver:
            receiver = rng.integers(n)
        max_frac = 0.04

        a[donor] -= delta

        a = a * (target_sum / a.sum())

def _fft_len(n):
    return 1 << ((2 * n - 1).bit_length())
def _conv_fft(a):
    n = a.shape[0]
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv
def _max_convolution_fft(a):
    return float(_conv_fft(a).max())
def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, b
def _project_to_simplex(v, s):
    # from earlier code
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)
def _power_law_sequence(n, p, target_sum):
    a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

    diags = [np.full(n, f_i) for f_i in f]

"""
Candidate generation for the step‑function minimization problem.

The algorithm has three stages:
1. **Compact LP descent** – a small‑scale linear program
   (maximising the ℓ¹‑norm of a new vector `b` while keeping the
   convolution `f*b` below the current worst autocorrelation `M`).
   The solution is mixed with the current sequence, guaranteeing a
   monotone reduction of `M`.  This stage quickly discovers a
   promising non‑uniform shape.
2. **Exponential (mirror) descent** – optimisation of a smooth
   log‑sum‑exp surrogate of the maximum convolution entry.
   The update is multiplicative (mirror‑descent) with an increasing
   temperature `β`, allowing fast convergence.
3. **Simulated‑annealing refinement** – a cheap stochastic local
   search that moves a small amount of mass from indices that
   contribute heavily to the current maximal convolution entry to
   indices that contribute little.  A linearly cooling temperature
   permits occasional uphill moves, helping to escape shallow local
   minima.

All stages respect the supplied time budget and keep the sequence
non‑negative with ℓ¹‑norm `√(2·n)`.  The final sequence is trimmed of
trailing zeros before being returned.
"""
import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Convolution / scoring utilities
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Return the next power‑of‑2 length for FFT of size 2·n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    conv = _conv_fft(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Linear‑programming (LP) phase on a reduced scale
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1)×n matrix A where (A·b)_k = ∑_i f_i·b_{k-i}.
    Implemented with a collection of constant diagonals.
    """
    n = f.shape[0]
    # each diagonal offset d contains the constant value f[d]
    diags = [np.full(n, f[d]) for d in range(n)]
    A = sp.diags(diags, offsets=range(n), shape=(2 * n - 1, n), format="csr")
    return A


def _lp_descent_small_n(
    n_small: int,
    target_sum: float,
    deadline: float,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    LP based descent on a reduced problem size.
    Returns the best sequence found (ℓ¹‑norm = target_sum) and its max‑convolution.
    """
    if init_seq is not None:
        f = init_seq.copy()
    else:
        f = _power_law_sequence(n_small, 0.55, target_sum)

    best_f = f.copy()
    best_score = _max_convolution_fft(f)
    t = 0.3  # mixing factor
    iteration = 0

    while time.time() < deadline:
        M = _max_convolution_fft(f)

        # Build the constraint matrix A (size (2n‑1)×n)
        A = _build_conv_matrix(f)

        # Upper bounds for each convolution entry
        b_ub = np.full(2 * n_small - 1, M)

        # Objective: maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n_small)

        bounds = [(0, None)] * n_small

        # Solve the LP
        res = opt.linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs"
        )
        if not res.success:
            print(f"[{time.time():.2f}] LP failed after {iteration} it: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        # No improvement → stop
        if sum_b <= target_sum * (1 + 1e-9):
            print(f"[{time.time():.2f}] LP iteration {iteration}: sum not increased (sum={sum_b:.6f})")
            break

        # Scale to the correct ℓ¹‑norm and mix
        g = b * (target_sum / sum_b)
        f = (1 - t) * f + t * g
        f = np.maximum(f, 0.0)
        f = f * (target_sum / f.sum())

        cur_score = _max_convolution_fft(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time():.2f}] LP it={iteration} score={best_score:.6f}")
        else:
            # if no progress, shrink the mixing factor
            t = max(0.05, t * 0.9)

        iteration += 1
        # safeguard against too many iterations
        if iteration >= 15:
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 1.0
    max_beta = 32768.0
    base_eta = 0.05
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)  # multiplicative update
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it} β={beta:.1f} val={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _simulated_annealing(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Cheap SA that moves small mass from high‑weight to low‑weight indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _max_convolution_fft(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.02 * best_val  # temperature proportional to current score

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)

    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to ``target_sum``."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as ``a``.
    """
    n = a.shape[0]
    size = _fft_len(n)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Linear‑programming (LP) phase on a reduced scale
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1)×n matrix A where (A·b)_k = Σ_i f_i·b_{k-i}.
    Implemented as a collection of constant diagonals.
    """
    n = f.shape[0]
    diags = [np.full(n, f[d]) for d in range(n)]
    A = sp.diags(diags, offsets=range(n), shape=(2 * n - 1, n), format="csr")
    return A


def _lp_descent_small_n(
    n_small: int,
    target_sum: float,
    deadline: float,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    LP‑based descent on a reduced problem size.
    Returns the best sequence found (ℓ¹‑norm = ``target_sum``) and its
    maximal convolution entry.
    """
    if init_seq is not None:
        f = init_seq.copy()
    else:
        f = _power_law_sequence(n_small, 0.55, target_sum)

    best_f = f.copy()
    best_score = _max_convolution_fft(f)
    t = 0.3                      # mixing factor
    iteration = 0

    while time.time() < deadline:
        M = _max_convolution_fft(f)
        A = _build_conv_matrix(f)                     # (2n‑1)×n matrix
        b_ub = np.full(2 * n_small - 1, M)            # each conv entry ≤ M
        c = -np.ones(n_small)                         # maximise Σb  → minimise -Σb
        bounds = [(0, None)] * n_small

        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            print(f"[{time.time():.2f}] LP failed after {iteration} it: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= target_sum * (1 + 1e-9):
            print(f"[{time.time():.2f}] LP iteration {iteration}: sum not increased (sum={sum_b:.6f})")
            break

        # Scale to the correct ℓ¹‑norm and mix
        g = b * (target_sum / sum_b)
        f = (1 - t) * f + t * g
        f = np.maximum(f, 0.0)
        f = f * (target_sum / f.sum())

        cur_score = _max_convolution_fft(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time():.2f}] LP it={iteration} score={best_score:.6f}")
        else:
            t = max(0.05, t * 0.9)   # shrink step if no progress

        iteration += 1
        if iteration >= 15:          # safety guard
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 1.0
    max_beta = 32768.0
    base_eta = 0.05
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)                     # multiplicative update
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it} β={beta:.1f} val={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement on the full size
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Cheap SA that moves small mass from high‑weight to low‑weight indices."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _max_convolution_fft(best)
    cur_val = best_val
    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.02 * best_val   # temperature scale

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        # donor proportional to its contribution
        wsum = weight.sum()
        if wsum == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=weight / wsum)

        # receiver from the low‑contribution tail
        low_thr = np.percentile(weight, 10)
        low_cands = idx[weight <= low_thr]
        if low_cands.size:
            receiver = rng.choice(low_cands)
        else:
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.25
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _max_convolution_fft(a_new)

        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA improve best={best_val:.6f}")

        # occasional jitter to keep diversity
        if rng.random() < 0.004:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _max_convolution_fft(a)

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 1 – compact LP descent on a reduced scale
    # ------------------------------------------------------------------
    small_n = max(800, n // 6)               # a few thousand at most
    small_target = np.sqrt(2.0 * small_n)

    # Warm‑start from a previously stored construction if available
    if "height_sequence_1" in globals():
        prior = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prior.shape[0] >= small_n:
            init_small = _resize_sequence(prior[:small_n], small_n)
        else:
            init_small = _resize_sequence(prior, small_n)
        init_small = init_small * (small_target / init_small.sum())
    else:
        init_small = None

    lp_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.20)
    f_small, score_small = _lp_descent_small_n(
        small_n, small_target, lp_deadline, init_seq=init_small
    )
    print(f"[{time.time():.2f}] LP phase done – small‑scale max‑conv={score_small:.6f}")

    # ------------------------------------------------------------------
    # Upscale to the full dimension
    # ------------------------------------------------------------------
    init_full = _resize_sequence(f_small, n)
    init_full = init_full * (target_sum / init_full.sum())

    # ------------------------------------------------------------------
    # Phase 2 – exponential (mirror) descent on the full problem
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + min(180.0, (deadline - time.time()) * 0.30)
        a_exp, val_exp = _exponential_descent(init_full, target_sum, exp_deadline)
        best_seq = a_exp
        best_val = val_exp
        print(f"[{time.time():.2f}] After ExpGD max‑conv={best_val:.6f}")
    else:
        best_seq = init_full
        best_val = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing refinement (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_deadline)
        if val_sa < best_val:
            best_seq = a_sa
            best_val = val_sa
        print(f"[{time.time():.2f}] After SA max‑conv={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in future calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
