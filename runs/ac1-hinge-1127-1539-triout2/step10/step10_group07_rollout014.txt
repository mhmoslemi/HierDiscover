The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

Because the score is scale‑invariant we fix  

\[
\sum_i a_i=\sqrt{2n},
\]

so the score is simply the maximal entry of the self‑convolution
`b = a*a`.  

The optimiser consists of four independent stages

1. **Very cheap parametric sweep** – evaluate a few hundred simple
   families (power‑law, exponential, product of both, triangular).
   The best family is used as a warm‑start.

2. **Exact sub‑gradient descent** – minimise the true max‑convolution.
   The sub‑gradient is obtained by a single FFT‑convolution with the
   indicator of the maximal entries.  After each step the iterate is
   projected onto the simplex `{a≥0, Σa = √(2n)}`.

3. **Short random restarts** – the current best vector is jittered,
   a brief sub‑gradient run is performed and the result is kept if it
   improves the objective.

4. **Mass‑transfer simulated annealing** – repeatedly move a small
   amount of mass from a “bad’’ coordinate (large sub‑gradient) to a
   “good’’ one (small sub‑gradient).  The move is accepted with a
   temperature schedule and the convolution is updated **exactly**
   in `O(n)` time.

All stages respect the time budget (the function receives `budget_s`);
the best sequence found before the deadline is returned.
The code works with any `evaluate_sequence` that follows the specification
and uses only `numpy` and the Python standard library.

```python
#!/usr/bin/env python3
"""
propose_candidate – search for a non‑negative step sequence with a low
evaluation score.

The algorithm combines a cheap parametric sweep, exact sub‑gradient
descent on the true max–convolution, a few random restarts and a
simulated‑annealing mass‑transfer refinement.  All steps stay on the
simplex Σ a_i = √(2 n) so the evaluation reduces to the maximal
convolution entry.  The routine respects the supplied time budget and
stores the best sequence globally for future warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (return length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _max_convolution(a: np.ndarray) -> float:
    """Return max(a*a) – the objective when Σa = √(2n)."""
    return _conv_fft(a).max()


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the true max‑convolution (sub‑gradient).
    The iterate stays on the simplex Σa = target_sum.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _max_convolution(a)

    rng = np.random.default_rng()
    it = 0
    base_step = target_sum * 0.6          # a bit larger than the previous 0.5
    n = a.shape[0]

    while time.time() < deadline:
        # convolution and mask of (almost) maximal entries
        b = _conv_fft(a)
        max_b = b.max()
        eps = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps).astype(np.float64)

        # sub‑gradient of max(b) w.r.t a
        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)   # length 3n‑2
        L = len(mask) - 1           # = 2n‑2
        g = conv_full[L - np.arange(n)]

        # step size (decays like 1/√t)
        step = base_step / np.sqrt(it + 1.0)
        gn = np.linalg.norm(g)
        if gn > 0:
            a -= step * g / gn

        # projection back onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0007, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] sub‑grad it={it:6d}  score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Mass‑transfer simulated annealing (improved version)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that moves a tiny amount of mass
    from a “bad’’ index (high sub‑gradient) to a “good’’ one (low
    sub‑gradient).  The convolution is updated exactly in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _max_convolution(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.05                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # ----- sub‑gradient (used as a proxy for “bad’’ / “good’’) -----
        max_b = b.max()
        eps = max_b * 1e-12 + 1e-15
        mask = (b >= max_b - eps).astype(np.float64)
        mask_rev = mask[::-1]
        conv_full = _linear_convolve_fft(a, mask_rev)
        L = len(mask) - 1
        g = conv_full[L - np.arange(n)]

        # donors – indices with large gradient
        prob_donor = g / (g.sum() + 1e-18)
        donor = rng.choice(n, p=prob_donor)

        # receivers – indices with small gradient
        inv = np.max(g) - g + 1e-12          # larger weight for small g
        prob_recv = inv / inv.sum()
        receiver = rng.choice(n, p=prob_recv)

        if donor == receiver:
            continue

        # ----- propose a transfer -----
        # allow larger transfers at the beginning, shrink with temperature
        frac = max(0.005, T0 * (1.0 - (time.time() - start) / total_time) / best_val)
        max_delta = min(a[donor] * 0.3, target_sum * frac)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # ----- exact update of the convolution (O(n)) -----
        b_new = b.copy()
        # linear part 2*a*Δ
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = new_max

        # ----- acceptance -----
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] local refine improved score={best_val:.6f}")

        # ----- periodic full recomputation (to avoid drift) -----
        if time.time() - last_recalc > 11.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _max_convolution(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] re‑norm improved score={best_val:.6f}")

        # ----- occasional tiny jitter -----
        if int(elapsed * 100) % 2400 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.004, target_sum * 0.0003)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _max_convolution(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families – cheap global search
# ----------------------------------------------------------------------


def _parametric_sweep(n: int, target_sum: float, time_limit: float):
    """
    Scan a few hundred simple families (power‑law, exponential,
    product) and return the best sequence found within *time_limit*.
    """
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # power‑law family a_i ∝ (i+1)^α
    alphas = np.linspace(-1.5, 0.0, 31)
    for α in alphas:
        if time.time() > time_limit:
            break
        base = (np.arange(1, n + 1, dtype=np.float64) ** α)
        seq = base / base.sum() * target_sum
        score = _max_convolution(seq)
        if score < best_score:
            best_score, best_seq = score, seq.copy()

    # exponential family a_i ∝ exp(β i)
    betas = np.linspace(-0.0015, 0.0, 16)
    for β in betas:
        if time.time() > time_limit:
            break
        base = np.exp(β * np.arange(n, dtype=np.float64))
        seq = base / base.sum() * target_sum
        score = _max_convolution(seq)
        if score < best_score:
            best_score, best_seq = score, seq.copy()

    # product of both
    for α in alphas:
        for β in betas:
            if time.time() > time_limit:
                break
            base = (np.arange(1, n + 1, dtype=np.float64) ** α) * np.exp(β * np.arange(n, dtype=np.float64))
            seq = base / base.sum() * target_sum
            score = _max_convolution(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()

    # simple triangular shape
    if time.time() < time_limit:
        base = np.linspace(1.0, 0.0, n, dtype=np.float64)
        seq = base / base.sum() * target_sum
        score = _max_convolution(seq)
        if score < best_score:
            best_score, best_seq = score, seq.copy()

    # warm‑start from a previously found construction (if any)
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm / warm.sum() * target_sum
        score = _max_convolution(warm)
        if score < best_score:
            best_score, best_seq = score, warm.copy()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the provided time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # small safety margin

    # --------------------------------------------------------------
    # Choose problem size according to the time budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)            # Σa = √(2 n)

    # --------------------------------------------------------------
    # Stage 0 – cheap parametric sweep (≈10 % of the budget)
    # --------------------------------------------------------------
    sweep_deadline = time.time() + 0.1 * (deadline - time.time())
    init_seq, init_score = _parametric_sweep(n, target_sum, sweep_deadline)
    print(f"[{time.time():.2f}] parametric sweep best score={init_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – exact sub‑gradient descent (≈50 % of the remaining budget)
    # --------------------------------------------------------------
    sub_deadline = time.time() + 0.5 * (deadline - time.time())
    a_sub, val_sub = _subgradient_descent(init_seq, target_sum, sub_deadline)
    best_seq = a_sub
    best_score = val_sub
    print(f"[{time.time():.2f}] after sub‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – short random restarts (≈20 % of the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        rest_deadline = time.time() + 0.2 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2.0:
            # jitter the current best and run a short descent
            jitter = rng.normal(scale=target_sum * 0.02, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = start_seq / start_seq.sum() * target_sum

            run_until = time.time() + min(5.0, rest_deadline - time.time())
            a_tmp, val_tmp = _subgradient_descent(start_seq, target_sum, run_until)
            if val_tmp < best_score:
                best_score, best_seq = val_tmp, a_tmp.copy()
                print(f"[{time.time():.2f}] restart improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – exact mass‑transfer refinement (use all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
        print(f"[{time.time():.2f}] after local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up, evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {final_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print(f"[{time.time():.2f}] evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
```
