"""
Low‑evaluation step‑function search.

The algorithm works in three stages (all respecting the time budget):

1. **Warm‑start** – reuse the best known construction if it exists,
   otherwise initialise a decreasing power‑law and project onto the
   simplex Σa = √(2 n).

2. **Adam‑softmax descent** – minimise a smooth approximation of the
   true objective max(a∗a).  A temperature schedule turns the soft‑max
   into the true max towards the end of the run, while an Adam optimiser
   gives fast, well‑scaled updates.  The iterate is always projected back
   onto the simplex (non‑negativity + fixed ℓ¹‑norm).

3. **Exact local refinement** – a cheap O(n) simulated‑annealing
   mass‑transfer move (the same routine used in the previous solution)
   that finishes the optimisation on the true objective.

The routine returns the best sequence found and stores it in the global
variable ``height_sequence_1`` for warm‑starts in subsequent calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective used during the search.
    For Σa = √(2 n) the return equals max(a*a).
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam descent on a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise the true max‑convolution via a soft‑max surrogate.
    The optimisation stays on the simplex Σa = target_sum.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0
    start = time.time()
    total = deadline - start
    # base learning‑rate (chosen empirically)
    base_lr = 0.35 * target_sum

    while time.time() < deadline:
        # 1️⃣ compute convolution and current max
        b = _conv_fft(a)
        max_b = b.max()

        # 2️⃣ temperature schedule – from 0.2·max_b down to ~0.005·max_b
        elapsed = time.time() - start
        frac = elapsed / total
        T = max_b * (0.2 * (1.0 - frac) + 0.005)

        # 3️⃣ soft‑max weights (probability distribution over all convolution entries)
        scaled = (b - max_b) / T          # stability: max element becomes 0
        exp_term = np.exp(scaled)
        w = exp_term / exp_term.sum()

        # 4️⃣ gradient of the soft‑max surrogate (convolution with reversed weight vector)
        w_rev = w[::-1]
        conv_full = _linear_convolve_fft(a, w_rev)   # length 3·n‑2
        L = len(w) - 1                                 # = 2·n‑2
        g = conv_full[L - np.arange(a.shape[0])]

        # 5️⃣ Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # 6️⃣ occasional jitter to escape plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape[0])
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # 7️⃣ bookkeeping
        cur_val = max_b                     # because Σa = √(2 n)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d}  score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact local mass‑transfer refinement (same as previous solution)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the true max‑convolution.
    Transfers a tiny amount of mass from a “bad’’ index to a “good’’ one.
    The routine respects non‑negativity and the ℓ¹‑norm constraint.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – choose from the lower‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta            # amount moved from donor → receiver

        # --------------------------------------------------------------
        # Update convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic re‑normalisation & full recomputation
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects a time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size based on the time budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)               # fixing Σa simplifies the objective

    # --------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = _project_to_simplex(init, target_sum)

    # --------------------------------------------------------------
    # Stage 1 – Adam soft‑max descent (≈40 % of budget)
    # --------------------------------------------------------------
    sub_deadline = time.time() + 0.4 * (deadline - time.time())
    a_best, best_score = _adam_softmax_descent(init, target_sum, sub_deadline)
    print(f"[{time.time():.2f}] After Adam stage best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – short random restarts of Adam descent (~15 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        rest_deadline = time.time() + 0.15 * (deadline - time.time())
        rng = np.random.default_rng()
        while time.time() < rest_deadline - 2.0:
            remaining = rest_deadline - time.time()
            run_time = min(6.0, remaining)
            if run_time < 1.0:
                break
            jitter = rng.normal(scale=target_sum * 0.02, size=n)
            start_seq = a_best + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            a_tmp, val_tmp = _adam_softmax_descent(start_seq, target_sum,
                                                   time.time() + run_time)
            if val_tmp < best_score:
                best_score = val_tmp
                a_best = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – exact local mass‑transfer refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(a_best, target_sum, deadline)
        if val_ref < best_score:
            best_score = val_ref
            a_best = a_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean up and final evaluation
    # --------------------------------------------------------------
    arr = np.array(a_best, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a later call
    globals()["height_sequence_1"] = final_seq

    return final_seq
