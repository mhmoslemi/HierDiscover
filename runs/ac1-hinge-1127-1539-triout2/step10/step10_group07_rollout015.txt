def _gradient_max_smoothing(a, w):
    # a: np.ndarray shape (n,)
    # w: np.ndarray shape (2n-1,)
    n = a.shape[0]
    # reverse w
    w_rev = w[::-1]
    # Full conv length = n + (2n-1) -1 = 3n-2
    conv_full = _linear_convolve_fft(a, w_rev)
    L = len(w) - 1  # 2n-2
    # Want g_i = 2 * conv_full[L - i] for i = 0..n-1
    indices = L - np.arange(n)
    g = 2.0 * conv_full[indices]
    return g

"""Improved search for low‑evaluation step sequences.

The algorithm consists of three stages:

1. **Smoothed gradient descent** – minimise the log‑sum‑exp of the
   convolution entries (a smooth approximation of the max).  The gradient
   is obtained via two FFT‑based convolutions; after each step the vector
   is projected onto the simplex {x≥0, Σx = √(2n)}.

2. **Short random restarts** – repeatedly jitter the current best
   sequence and run a brief smoothed‑gradient run; keep the best result.

3. **Exact mass‑transfer refinement** – a simulated‑annealing hill‑climb
   that moves a tiny amount of mass from indices that contribute heavily
   to the maximal convolution entry to indices that contribute little.

All stages respect the non‑negativity and sum constraints required by
``evaluate_sequence``.  The routine trades time for quality and stays
within the supplied budget (default 1000 s)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """Exact benchmark value (equals max convolution for Σa=√2 n)."""
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Gradient of smoothed max (log‑sum‑exp) objective
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a  L_α(b) where L_α(b) = log∑_k exp(α b_k).
    The gradient can be expressed as
        g_i = 2 * Σ_j a_j * w_{i+j}
    where w_k = exp(α b_k) / Σ_t exp(α b_t).
    """
    n = a.shape[0]
    # Numerical stability: shift by max(b)
    max_b = b.max()
    if alpha == 0.0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        # exponent is non‑positive, safe from overflow
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = w / w_sum

    # convolution a * w_rev and pick the middle n entries
    w_rev = w[::-1]
    conv_full = _linear_convolve_fft(a, w_rev)   # length 3n‑2
    L = len(w) - 1            # = 2n‑2
    g = 2.0 * conv_full[L - np.arange(n)]
    return g


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             alpha_max: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent on log‑sum‑exp of the convolution.
    Returns the best vector encountered and its (exact) max‑convolution value.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    b = _conv_fft(a)
    best_val = b.max()
    rng = np.random.default_rng()
    it = 0
    start = time.time()
    # ramp‑up length for α (number of iterations after which α reaches α_max)
    alpha_ramp = 2000

    while time.time() < deadline:
        # compute convolution
        b = _conv_fft(a)
        max_b = b.max()

        # schedule the smoothing parameter α
        if it < alpha_ramp:
            alpha = alpha_max * (it / alpha_ramp)
        else:
            alpha = alpha_max

        # gradient of smoothed objective
        g = _smooth_gradient(a, b, alpha)

        # step magnitude (decreasing)
        step = target_sum * 0.3 / np.sqrt(it + 1.0)

        # avoid division by zero
        gn = np.linalg.norm(g)
        if gn > 0:
            a -= step * g / gn

        # enforce simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 400 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # record best
        cur_val = max_b  # because Σa = √(2n) → objective = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SmoothGD it={it:6d} score={best_val:.6f}")

        # early exit if we already beat the target
        if best_val < 1.503:
            break

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact mass‑transfer refinement (same as previous solution, with
# a slightly higher initial temperature for more exploration)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that moves tiny amounts of mass
    from indices that heavily contribute to the current maximal convolution
    entry towards indices that contribute little.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # increase base temperature a bit (previous 0.04 → 0.06)
    T0 = best_val * 0.06                 # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]               # contribution of each pair
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – choose from the lower‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0009)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta            # amount moved from donor → receiver

        # --------------------------------------------------------------
        # Update convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic re‑normalisation & full recomputation
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional jitter to shake out of plateaus
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.006, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the supplied time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)            # fixing Σa simplifies the objective

    # ------------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # fallback: decreasing power‑law (exponent tuned heuristically)
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – smoothed gradient descent (≈45 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage1_deadline = time.time() + 0.45 * remaining
    a_sgd, val_sgd = _smooth_gradient_descent(init, target_sum, stage1_deadline)
    best_seq = a_sgd
    best_score = val_sgd
    print(f"[{time.time():.2f}] After smooth GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short random restarts of smoothed GD (~20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_deadline = time.time() + 0.20 * remaining
        rng = np.random.default_rng()
        while time.time() < stage2_deadline - 1.0:
            # jitter the current best and run a brief descent
            jitter = rng.normal(scale=target_sum * 0.025, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            small_deadline = time.time() + 4.0   # a few seconds per restart
            a_tmp, val_tmp = _smooth_gradient_descent(start_seq, target_sum, small_deadline)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq

    return final_seq

"""Improved search for a low‑evaluation step sequence.

The algorithm works in three stages:

1. **Smoothed gradient descent** – minimise a log‑sum‑exp surrogate of the
   maximal convolution entry.  The gradient is obtained with two FFT‑based
   convolutions and the iterate is projected onto the simplex
   {x≥0, Σx = √(2 n)}.

2. **Short random restarts** – repeatedly jitter the current best
   sequence and run a brief smoothed‑gradient run; the best result is kept.

3. **Exact mass‑transfer refinement** – a simulated‑annealing hill‑climb
   that moves a tiny amount of mass from indices contributing heavily to the
   current maximal convolution entry to indices that contribute little.

All steps respect the non‑negativity and sum constraints required by
``evaluate_sequence``.  The routine adapts its running time to the supplied
budget (default 1000 s)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)[:m]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """Exact benchmark value (equals max convolution when Σa=√2 n)."""
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Gradient of smoothed max (log‑sum‑exp) objective
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a  L_α(b) where
        L_α(b) = log Σ_k exp(α·b_k).

    The gradient can be written as
        g_i = 2· Σ_j a_j·w_{i+j},
    where w_k = exp(α·b_k) / Σ_t exp(α·b_t).
    """
    n = a.shape[0]

    # Build the probability vector w from the current convolution b
    max_b = b.max()
    if alpha == 0.0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w = np.exp(alpha * (b - max_b))          # non‑positive exponent
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = w / w_sum

    # Convolution a * w_rev and pick the central n entries
    w_rev = w[::-1]
    conv_full = _linear_convolve_fft(a, w_rev)   # length 3·n‑2
    L = len(w) - 1                               # = 2·n‑2
    g = 2.0 * conv_full[L - np.arange(n)]
    return g


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             alpha_max: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent on the log‑sum‑exp surrogate.
    Returns the best vector encountered and the exact maximal convolution
    value for that vector.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    best_a = a.copy()
    b = _conv_fft(a)
    best_val = b.max()

    rng = np.random.default_rng()
    it = 0
    start = time.time()
    # α grows linearly to alpha_max over this many iterations
    alpha_ramp = 2000
    base_step = target_sum * 0.2  # less aggressive than the original sub‑grad code

    while time.time() < deadline:
        # current convolution and its maximum
        b = _conv_fft(a)
        max_b = b.max()

        # schedule α
        alpha = alpha_max * (it / alpha_ramp) if it < alpha_ramp else alpha_max

        # gradient of the smoothed objective
        g = _smooth_gradient(a, b, alpha)

        # step size (decreasing)
        step = base_step / np.sqrt(it + 1.0)

        # normalized update
        gn = np.linalg.norm(g)
        if gn > 0.0:
            a -= step * g / gn

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 400 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # bookkeeping
        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SmoothGD it={it:6d} score={best_val:.6f}")

        # early exit if target already reached
        if best_val < 1.503:
            break

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact mass‑transfer refinement (simulated annealing)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Move tiny amounts of mass from “bad’’ indices (large contribution to the
    current maximal convolution entry) to “good’’ ones, accepting moves with
    a temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.06                 # a slightly higher start temperature
    last_recalc = start

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Choose donor (contributes strongly to the current max entry)
        # ------------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]                # pair contributions
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # ------------------------------------------------------------------
        # Choose receiver (indices with relatively low contribution)
        # ------------------------------------------------------------------
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # ------------------------------------------------------------------
        # Propose a small transfer
        # ------------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0009)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # ------------------------------------------------------------------
        # Update convolution exactly (O(n))
        # ------------------------------------------------------------------
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # ------------------------------------------------------------------
        # Acceptance test (simulated annealing)
        # ------------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # ------------------------------------------------------------------
        # Periodic full recomputation (stability)
        # ------------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional extra jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.006, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the supplied time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)            # fixing Σa simplifies the objective

    # ------------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
    else:
        # fallback: decreasing power‑law (exponent chosen heuristically)
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – smoothed gradient descent (≈45 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage1_deadline = time.time() + 0.45 * remaining
    a_sgd, val_sgd = _smooth_gradient_descent(init, target_sum, stage1_deadline)
    best_seq = a_sgd
    best_score = val_sgd
    print(f"[{time.time():.2f}] After smooth GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – short random restarts of smoothed GD (~20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_deadline = time.time() + 0.20 * remaining
        rng = np.random.default_rng()
        while time.time() < stage2_deadline - 1.0:
            jitter = rng.normal(scale=target_sum * 0.025, size=n)
            start_seq = best_seq + jitter
            start_seq = np.maximum(start_seq, 0.0)
            start_seq = _project_to_simplex(start_seq, target_sum)

            # a few seconds per restart
            small_deadline = time.time() + 4.0
            a_tmp, val_tmp = _smooth_gradient_descent(start_seq,
                                                      target_sum,
                                                      small_deadline)
            if val_tmp < best_score:
                best_score = val_tmp
                best_seq = a_tmp.copy()
                print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
