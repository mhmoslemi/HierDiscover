def _simulated_annealing(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    b = _full_convolution(a)
    best_a = a.copy()
    best_score = b.max()
    # We will use a linear cooling schedule: t in [0, total]; temperature = initial_temp * (1 - t/total)
    total = deadline - time.time()
    start_temp = best_score * 0.3  # maybe
    rng = np.random.default_rng()
    while time.time() < deadline:
        elapsed = time.time() - (deadline - total)  # not exactly, easier: compute fraction left
        fraction = max(0.0, (deadline - time.time()) / total)
        temperature = start_temp * fraction  # linearly decreasing

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # delta = small fraction (0.01 to 0.1)
        # ensure delta not too large relative to a[i]
        if a[i] < 1e-12:
            continue
        delta = a[i] * rng.uniform(0.01, 0.2)
        # compute new b after moving delta from i to j
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        new_score = b_new.max()
        # Accept if better or with probability exp(-(new_score - cur_score)/temp)
        cur_score = b.max()
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (temperature + 1e-12)):
            # accept
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # else maybe reject
    return best_a, best_score
```

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Key ideas
---------
1.  Use a very short quick search (n≈2000) only to obtain a decent shape.
2.  Upsample that shape to a much larger length (n≈20000) – the max
    autocorrelation typically drops with length.
3.  On the large problem run many independent Adam optimisations on the
    smooth‑max surrogate, followed by an exact deterministic hill‑climb
    and a short simulated‑annealing phase.
4.  All iterators respect the global time budget and the best sequence
    found so far is always projected onto the ℓ¹‑simplex
    Σa = √(2 n) (the evaluation is homogeneous, so this normalisation
    does not affect the final score).

The routine returns the best list of heights before the 1000 s deadline.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until ``deadline``.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.12 * target_sum

    # smoothing schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # tiny jitter every few hundred steps
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisation families
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # index of the current maximal convolution entry
        M = int(np.argmax(b))

        # indices that influence b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # source with the largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for potential destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # choose destination with minimal counterpart (but not i)
        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # amount of mass to move (up to 60 % of source)
        delta = a[i] * 0.6
        if delta <= 0.0:
            break

        # evaluate the move
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search style)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # tiny random jitter to escape a possible plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple family sweep (used to seed the optimisation)
# ----------------------------------------------------------------------


def _family_candidates(n: int, rng, top_k: int = 8):
    """
    Scan a few simple families and keep the best ``top_k`` vectors.
    Returns a list of (score, vector) pairs.
    """
    target = math.sqrt(2.0 * n)
    candidates = []

    # exponential decay
    for _ in range(20):
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # power‑law
    for _ in range(20):
        p = rng.uniform(0.1, 1.5)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # mixed power‑law × exponential
    for _ in range(20):
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p) * np.exp(-lam * np.arange(n))
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # piecewise exponential
    for _ in range(10):
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # Dirichlet (uniform on simplex)
    for _ in range(20):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        a = _project_to_simplex(a, target)
        candidates.append((_max_convolution_fft(a), a))

    # keep the best
    candidates.sort(key=itemgetter(0))
    return candidates[:top_k]

def _simulated_annealing(a0: np.ndarray, target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Short SA that moves a small fraction of mass between two indices.
    Returns the best sequence seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.size
    b = _full_convolution(a)
    best_a = a.copy()
    best_score = b.max()
    total = max(deadline - time.time(), 1e-12)
    start = time.time()
    rng = np.random.default_rng()
    # initial temperature proportional to current score
    temp0 = best_score * 0.3 + 1e-12

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = max(0.0, (total - elapsed) / total)
        temperature = temp0 * frac

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] < 1e-14:
            continue

        delta = a[i] * rng.uniform(0.01, 0.2)
        if delta < 1e-14:
            continue

        # update convolution (exact O(n) update)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()
        cur_score = b.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(
                -(new_score - cur_score) / (temperature + 1e-12)):
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

# Core optimisation at a fixed length

def _optimize_at_n(n: int, rng, deadline: float,
                   init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a vector of length ``n`` until ``deadline``.
    Returns the best vector and its maximal convolution entry.
    """
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # initialise ‑‑ possibly upsample from a previous result
    # ------------------------------------------------------------------
    if init_seq is not None:
        a_init = np.asarray(init_seq, dtype=float)
        if a_init.size != n:
            a_init = _upsample_sequence(a_init, n)
    else:
        a_init = _random_initial_sequence(n, rng)

    a_init = _project_to_simplex(a_init, target_sum)

    # ------------------------------------------------------------------
    # quick family scan and keep a few elite candidates
    # ------------------------------------------------------------------
    top_cands = _family_candidates(n, rng, top_k=6)
    init_score = _max_convolution_fft(a_init)
    top_cands.append((init_score, a_init))
    top_cands.sort(key=itemgetter(0))
    top_cands = top_cands[:6]

    best_seq = top_cands[0][1].copy()
    best_score = top_cands[0][0]

    # ------------------------------------------------------------------
    # optional ensemble averaging of the elite set
    # ------------------------------------------------------------------
    ensemble = np.mean([vec for _, vec in top_cands], axis=0)
    ensemble = _project_to_simplex(ensemble, target_sum)
    ens_score = _max_convolution_fft(ensemble)
    if ens_score < best_score:
        best_seq = ensemble
        best_score = ens_score

    # ------------------------------------------------------------------
    # main iterative improvement loop (Adam → hill‑climb → SA)
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # ----- Adam restarts -------------------------------------------------
        # Allocate at most half of the remaining time to Adam, but cap each
        # restart at ≈5 s to keep many independent attempts.
        adam_time = min(remaining * 0.45, 40.0)
        if adam_time > 5.0:
            n_restarts = max(1, int(adam_time // 5.0))
            per_restart = adam_time / n_restarts
            for r in range(n_restarts):
                if time.time() >= deadline - 2.0:
                    break
                sub_dead = time.time() + per_restart
                start_vec = best_seq if r == 0 else _random_initial_sequence(n, rng)
                a_opt = _adam_optimize(start_vec, target_sum, sub_dead)
                sc = _max_convolution_fft(a_opt)
                if sc < best_score - 1e-12:
                    best_seq = a_opt
                    best_score = sc
                    print(f"[{time.time():.1f}] Adam improvement n={n} score={best_score:.6f}")

        # ----- Deterministic hill‑climb ------------------------------------
        if time.time() < deadline - 2.0:
            hc_dead = time.time() + min(30.0, deadline - time.time() - 1.0)
            b_cur = _full_convolution(best_seq)
            a_hc, sc_hc = _hill_climb_deterministic(best_seq, b_cur, hc_dead)
            if sc_hc < best_score - 1e-12:
                best_seq = a_hc
                best_score = sc_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement n={n} score={best_score:.6f}")

        # ----- Simulated annealing (short burst) ---------------------------
        if time.time() < deadline - 2.0:
            sa_dead = time.time() + min(20.0, deadline - time.time() - 1.0)
            a_sa, sc_sa = _simulated_annealing(best_seq, target_sum, sa_dead)
            if sc_sa < best_score - 1e-12:
                best_seq = a_sa
                best_score = sc_sa
                print(f"[{time.time():.1f}] SA improvement n={n} score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine spends a few seconds on a modest problem (n≈2000) to obtain a
    good shape, upsamples it to a larger dimension (n≈20000) and then performs
    intensive optimisation (Adam restarts, hill‑climb, simulated annealing) within
    the remaining time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    end = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1) Quick optimisation at a small size (n=2000)
    # --------------------------------------------------------------
    small_n = 2000
    small_deadline = min(end, time.time() + 30.0)   # at most 30 s
    init_small = None
    if "height_sequence_1" in globals():
        try:
            init_small = np.asarray(globals()["height_sequence_1"], dtype=float)
        except Exception:
            init_small = None
    best_small, score_small = _optimize_at_n(small_n, rng, small_deadline, init_small)
    print(f"[{time.time():.1f}] Small n={small_n} best score={score_small:.6f}")

    # --------------------------------------------------------------
    # 2) Upsample to a larger size and perform the heavy optimisation
    # --------------------------------------------------------------
    large_n = 20000
    best_large, score_large = _optimize_at_n(
        large_n, rng,
        deadline=end,
        init_seq=best_small
    )
    print(f"[{time.time():.1f}] Large n={large_n} final best score={score_large:.6f}")

    # --------------------------------------------------------------
    # 3) Cleanup – drop trailing near‑zeros
    # --------------------------------------------------------------
    best_seq = np.asarray(best_large, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # 4) Official evaluation (just for logging)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
