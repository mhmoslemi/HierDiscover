"""
Improved step‑function optimizer.

The goal is to minimise the maximal entry of the self‑convolution
of a non‑negative sequence a with a fixed ℓ¹‑norm S = √(2·n).
Since 2·n·max(conv)/(S²) = max(conv), the problem is exactly a
‘flat‑autocorrelation’ problem.

The optimiser works in three phases:

1.  **Fast initialisation** – a family of power‑law sequences
    a[i] ∝ (i+1)^{-p} is generated for many exponents p and
    projected onto the simplex Σa = S.  The best of these is taken
    as a starting point.

2.  **Adam on a soft‑max surrogate** – the non‑smooth max is
    approximated by a soft‑max with temperature β.  Adam (adaptive
    moment estimation) is applied to the gradient of this surrogate,
    while the list is projected onto the simplex after every step.
    β is increased exponentially up to a large value (≥5·10⁵) so that
    the surrogate becomes tight.

3.  **Refined deterministic mass‑transfer** – only the index
    k* where the current convolution attains its maximum is
    examined.  For this k* we compute the sub‑gradient
    w_i = a_i + a_{k*‑i}.  Mass is moved from the index with the
    largest w_i (donor) to the one with the smallest w_i (receiver)
    by exactly the amount that would equalise the two w‑values.
    After each move the convolution is recomputed; moves that do
    not improve the objective are reverted.  Small random jitter
    is added occasionally to escape plateaux.

A final short Adam run with an even higher β finishes the optimisation.
The whole procedure respects the time budget (default 1000 s) and
outputs the trimmed sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a)/(∑a)² – reduces to max(conv)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)          # soft‑max weights

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Adam optimiser on soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_beta: float = 5e5,
                  step_factor: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of max(convolution).
    The iterate is projected onto the simplex after every step.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best = a.copy()
    best_val = _score(a)

    beta = 0.2                     # start temperature
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    step_base = target_sum * step_factor

    while time.time() < deadline:
        it += 1

        # gradient of soft‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # adaptive step (norm‑scaled, also depends on current β)
        norm = np.linalg.norm(m_hat)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * m_hat

        # stay feasible
        a = _project_to_simplex(a, target_sum)

        # increase temperature regularly
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:7.1f} score={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Phase B – Refined deterministic mass‑transfer
# ----------------------------------------------------------------------


def _refined_mass_transfer(a0: np.ndarray,
                          deadline: float,
                          target_sum: float,
                          jitter_interval: int = 1500) -> tuple[np.ndarray, float]:
    """
    Deterministic mass‑transfer moves that directly aim at the current
    maximal convolution entry.  In each move the amount transferred is
    the exact amount required to equalise the largest and smallest
    sub‑gradient w_i = a_i + a_{k*‑i}.  Moves that do not improve the
    objective are reverted.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    conv = _conv_fft(a)          # current full convolution
    rng = np.random.default_rng()
    it = 0
    start_time = time.time()

    while time.time() < deadline:
        # index where convolution attains its maximum
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient w_i = a_i + a_{kmax‑i}
        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]

        if donor != receiver:
            w_donor = w[np.argmax(w)]
            w_recv = w[np.argmin(w)]

            # amount that would bring the two w‑values together
            delta = min(a[donor], max(0.0, (w_donor - w_recv) / 2.0))

            if delta > 0:
                a[donor] -= delta
                a[receiver] += delta

                # projection only to clean tiny numerical negatives
                a = _project_to_simplex(a, target_sum)

                # recompute convolution and score
                conv = _conv_fft(a)
                new_score = _score(a)

                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] mt it={it:5d} new best={best_score:.6f}")

                # keep only improving moves
                if new_score < cur_score:
                    cur_score = new_score
                else:
                    # revert
                    a[donor] += delta
                    a[receiver] -= delta
                    a = _project_to_simplex(a, target_sum)
                    conv = _conv_fft(a)

        # occasional global jitter to avoid getting stuck
        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm combines:
      * diverse power‑law initialisations,
      * Adam optimisation on a soft‑max surrogate,
      * a refined deterministic mass‑transfer phase,
      * a short high‑β polishing Adam run.
    """
    start = time.time()
    deadline = start + budget_s - 0.2          # small safety margin

    # ------------------- problem size -------------------
    if budget_s >= 800:
        n = 16000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    init_seqs = []
    for p in np.linspace(0.35, 0.75, 9):      # 9 exponents

    # previous best, if any

    for seq in list(init_seqs):

            tmp = seq + jitter
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_to_simplex(tmp, target_sum)
            jittered.append(tmp)
    init_seqs.extend(jittered)

    # keep the best raw candidate
    best_seq = None
    best_score = np.inf
