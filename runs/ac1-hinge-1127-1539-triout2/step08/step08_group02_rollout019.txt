"""
Improved step‑function optimizer.

The goal is to minimise
    2·n·max(a*a)/(sum(a)**2)
for a non‑negative vector a (the evaluator rescales this to the
maximum of the self‑convolution when the ℓ₁‑norm of a is fixed to √(2n)).
The previous best score was ≈1.509.  This implementation
* uses a larger set of parametric families for the initial guess,
* runs a longer Adam optimiser that switches from a τ‑soft‑max
  approximation to the exact max‑gradient,
* allocates more time to the refinement phases and performs a
  simulated‑annealing hill‑climb at the end,
* works with sequence lengths up to 50 000 (still fast enough with FFT).

The optimiser keeps the ℓ₁‑norm of the vector exactly
√(2·n) by using a log‑softmax parametrisation (logits → probabilities
→ heights).  All operations respect the non‑negativity constraint.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scaling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sinusoidal shape – smooth hill, optionally flattened."""
    i = np.arange(n)
    theta = np.pi * (i + 0.5) / n
    base = np.sin(theta) ** rng.uniform(0.5, 2.5)
    mix = rng.uniform(0.0, 1.0)
    return mix * base + (1.0 - mix) * np.ones(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 reuse the best global sequence (scaled),
    otherwise draw from a random parametric family.
    The vector is finally rescaled to have ℓ₁‑norm √(2n).
    """
    S = np.sqrt(2.0 * n)

    # warm start
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        tot = a.sum()
        if tot > 0.0:
            a = a * (S / tot)
            return a

    style = rng.choice(
        ['powerlaw', 'exponential', 'linear', 'gamma',
         'uniform', 'dirichlet', 'ushape', 'sine']
    )
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    tot = raw.sum()
    if tot == 0.0:
        raw = np.full(n, 1.0)
        tot = n
    return raw * (S / tot)


# ----------------------------------------------------------------------
# Optimiser (Adam on the log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   switch_to_hard: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the max‑convolution objective.
    Starts with a τ‑soft‑max gradient (τ large → smooth) and,
    after a few thousand steps, switches to the exact sub‑gradient
    of the true max (if *switch_to_hard* is True).
    Returns the best sequence found and its corresponding max‑convolution.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # target ℓ₁‑norm

    # initialise logits (log‑softmax) from a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # start temperature for the τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_max = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & peak ----
        conv = _full_convolution_fft(a)
        peak = conv.max()
        k_max = int(conv.argmax())

        if peak < best_max:
            best_max = peak
            best_a = a.copy()

        # ---- gradient of the max term ----
        if switch_to_hard and it > 2000:
            # exact sub‑gradient of the hard max
            g = np.zeros_like(a)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idx = np.arange(i_start, i_end + 1)
            j = k_max - idx
            g[idx] = 2.0 * a[j]
        else:
            # τ‑soft‑max approximation (same implementation as before)
            b = conv
            max_b = b.max()
            if tau < 1e-12:
                mask = (b == max_b).astype(float)
            else:
                diff = b - max_b
                exps = np.exp(diff / tau)
                mask = exps / exps.sum()
            # gradient wrt a  (2 * (mask * a) convolved with reversed a)
            a_rev = a[::-1]
            conv_len = len(mask) + n - 1
            f_len = _next_pow2(conv_len)
            Fmask = np.fft.rfft(mask, n=f_len)
            Fa_rev = np.fft.rfft(a_rev, n=f_len)
            tmp = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
            g = 2.0 * tmp[n - 1:n - 1 + n]
            # keep max value for possible logging
            peak = max_b

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules ----
        tau *= 0.9998                # keep a little smoothing longer
        lr *= 0.9999                 # very gentle decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_max

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm consists of:
      1) Broad random exploration of many parametric families.
      2) Scaling the best candidate to several longer lengths and
         intensive Adam optimisation (soft‑max → hard‑max).
      3) A final, longer Adam run on the currently best length.
      4) A simulated‑annealing style hill‑climb to polish the result.
    The final sequence is stored as the global ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # short safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf   # we store the *maximum* convolution (lower is better)
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – random exploration (~55 % of the time budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0   # reserve ~5 min for refinement / hill‑climb
    while time.time() < explore_deadline:
        attempt += 1
        # Length – bias towards a few thousand entries but allow larger sizes
        if rng.random() < 0.7:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 20000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:4d} n={n:5d}  best_max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling and medium‑length Adam refinements
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback in the unlikely case that exploration never produced a sequence
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # Construct a list of target lengths (geometric growth)
    lengths = [best_n]
    for factor in [1.3, 1.6, 2.0, 2.8, 3.5]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 50000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 120.0:   # keep ~2 min for final phases
            break
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.2, 140.0)   # at most ~2 min per length

        # scale the current best sequence to the new length
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # medium‑length Adam optimisation (soft‑max → hard‑max)
        a_opt, peak = _adam_optimize(a_scaled,
                                      time.time() + slice_time,
                                      rng,
                                      switch_to_hard=True)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  new_best_max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final long Adam optimisation on the best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:   # leave a little room for hill‑climb
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng,
                                      switch_to_hard=True)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final] length={best_n:5d}  best_max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – stochastic hill‑climb (simulated annealing style)
    # ------------------------------------------------------------------
    # temperature schedule for accepting worse moves
    start_temp = 0.02
    final_temp = 0.0005
    total_hc_time = end_time - time.time() - 1.0
    hc_start = time.time()
    while time.time() < end_time - 1.0:
        elapsed = time.time() - hc_start
        # linear temperature decay
        T = max(final_temp,
                start_temp * (1.0 - elapsed / total_hc_time))

        a = best_seq.copy()
        n = a.shape[0]

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue
        # move a modest random fraction of mass
        frac = rng.uniform(0.01, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount

        # evaluate new max (cheap because we already have FFT utilities)
        new_peak = _max_convolution(a)

        # Metropolis acceptance
        if new_peak < best_score:
            best_score = new_peak
            best_seq = a.copy()
            print(f"[hill] improved max={best_score:.6f}")
        else:
            accept_prob = np.exp(-(new_peak - best_score) / T)
            if rng.random() < accept_prob:
                best_score = new_peak
                best_seq = a.copy()
                print(f"[hill] accepted worse max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
