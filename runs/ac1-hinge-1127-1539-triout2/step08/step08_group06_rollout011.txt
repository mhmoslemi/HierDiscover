"""
A two‑stage optimizer for the self‑convolution peak.
Stage 1 – a Frank‑Wolfe style method that uses a linear programme
          (maximise Σ g_j   s.t.   f * g ≤ M) to push mass to the
          positions where the current shape f is “weak”.  This is cheap
          for a few thousand coefficients and usually drives the peak
          well below the 1.51 barrier reached by pure Adam search.

Stage 2 – once a good medium‑size shape is found we linearly up‑sample
          it to larger lengths and polish it with the smooth‑max Adam
          optimiser from the reference code.  A final tiny random‑swap
          hill‑climb finishes the job.

All vectors are kept on the simplex Σ a_i = √(2 n) and are forced to be
non‑negative.  The internal objective is max_k (a * a)_k; the external
score returned by `evaluate_sequence` equals this value because of the
normalisation.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (fast for n > ~2 000)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _peak_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (the internal objective)."""
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Symmetry helper (most good solutions are symmetric)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear interpolation (used for up‑sampling)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Candidate families (used only for the initial random draws)
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    fam = rng.integers(0, 7)
    if fam == 0 and base is not None:
        a = _upsample_seq(base, n)
    elif fam == 1:
        a = rng.random(n)
    elif fam == 2:
        rate = rng.uniform(0.92, 0.9995)
        left = rate ** np.arange(n, dtype=float)
        a = (left + left[::-1]) * 0.5
    elif fam == 3:
        p = rng.uniform(0.2, 4.5)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    else:
        scale = rng.uniform(0.02, 0.15) * n
        mid = (n - 1) / 2.0
        xs = np.arange(n, dtype=float)
        left = 1.0 / (1.0 + np.exp((xs - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - xs) / scale))
        a = left + right
    return np.clip(a, 0.0, None)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution (same as the reference)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇_a Σ_i w_i (a*a)_i  (w must sum to 1)."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.05
    max_alpha = 800.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb used for the final polish
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _peak_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Build the dense convolution matrix needed for the LP:
#
#    (A g)_k = Σ_i f_i * g_{k-i}  (linear in g)
#    A has shape (2n‑1) × n.
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j receives f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Solve   max Σ g_j   s.t.   f * g ≤ M,  g ≥ 0,   g_j ≤ S
# (implemented as a linear programme with ‘highs’)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float, S: float) -> np.ndarray:
    n = f.shape[0]
    A = _build_conv_matrix(f)                     # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, S) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fall back to a simple symmetric shape (uniform)
        return np.full(n, S / n, dtype=np.float64)

    g_raw = res.x
    total = g_raw.sum()
    if total == 0.0:
        return np.full(n, S / n, dtype=np.float64)
    # re‑scale to the required simplex sum
    return g_raw * (S / total)


# ----------------------------------------------------------------------
# Frank‑Wolfe / LP based refinement for a medium sized sequence.
# ----------------------------------------------------------------------
def _refine_with_lp(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iters: int = 40) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_M = _peak_convolution(a)

    t = 0.3                     # initial convex‑combination weight
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        M_cur = best_M
        # linear programme – give us a direction g that can carry the most mass
        g = _solve_lp_max_sum(a, M_cur, S)

        # convex combination (Frank‑Wolfe step)
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new, S)
        a_new = _symmetrize(a_new)          # symmetry helps the bound

        M_new = _peak_convolution(a_new)

        if M_new < best_M - 1e-12:          # improvement ?
            best_M = M_new
            best_a = a_new.copy()
            a = a_new                       # accept the step
            t = min(0.5, t * 1.1)           # be a bit bolder next time
        else:
            # step too aggressive – back off
            t = max(0.05, t * 0.7)

    return best_a, best_M

# Main entry point required by the checker
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    self‑convolution peak.  The routine first runs a Frank‑Wolfe style
    LP optimiser on a modest size (≈2000) to obtain a good shape,
    then up‑samples and refines it with the Adam optimiser from the
    reference implementation, finally polishing with a tiny random‑swap
    hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety buffer for the outer environment
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Stage 1 – LP based Frank‑Wolfe on a medium sized vector
    # --------------------------------------------------------------
    n0 = 2000                     # moderate size, cheap for the LP
    S0 = np.sqrt(2.0 * n0)

    # try a few random restarts (each gets its own time slice)
    best_a = None
    best_score = np.inf
    attempts = 0

    while time.time() < deadline - 30.0:      # keep ~30 s for later stages
        attempts += 1

        # initialise a – either down‑sample the previous best if present
        base = None
        if 'height_sequence_1' in globals():
            try:
                prev = np.asarray(height_sequence_1, dtype=float)
                base = _upsample_seq(prev, n0)
            except Exception:
                base = None

        a0 = _gen_candidate(n0, rng, base)
        a0 = a0 / a0.sum() * S0
        a0 = _symmetrize(a0)

        # give each restart a short slice of the remaining budget
        sub_deadline = min(deadline, time.time() + 20.0)

        a_opt, M_opt = _refine_with_lp(a0, sub_deadline, rng)

        if M_opt < best_score:
            best_score = M_opt
            best_a = a_opt.copy()
            print(f"[LP‑restart {attempts}] new best M={best_score:.6f}")

    if best_a is None:                      # absolute fallback
        n0 = 1500
        S0 = np.sqrt(2.0 * n0)
        best_a = rng.random(n0)
        best_a = best_a / best_a.sum() * S0
        best_score = _peak_convolution(best_a)

    # --------------------------------------------------------------
    # Stage 2 – up‑sample and finish with Adam + tiny local search
    # --------------------------------------------------------------
    # a list of target lengths (bounded by 15000)
    target_lengths = [4000, 8000, 12000]
    for target_n in target_lengths:
        if time.time() >= deadline:
            break
        if target_n > 15000:
            continue

        a_up = _upsample_seq(best_a, target_n)
        S_up = np.sqrt(2.0 * target_n)
        a_up = a_up / a_up.sum() * S_up
        a_up = _symmetrize(a_up)

        # allocate a modest fraction of the remaining time to Adam
        now = time.time()
        time_left = deadline - now
        if time_left <= 2.0:
            break
        adam_deadline = now + min(0.25 * time_left, 20.0)   # ≤ 20 s per up‑sample

        a_up_opt, M_up = _optimize_adam(a_up, adam_deadline, rng, enforce_sym=False)

        if M_up < best_score:
            best_score = M_up
            best_a = a_up_opt.copy()
            print(f"[upsample+adam] n={target_n} improved M={best_score:.6f}")

    # --------------------------------------------------------------
    # Final tiny random‑swap hill‑climb (uses whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, M_final = _local_swap_search(best_a, deadline, rng, enforce_sym=False)
        if M_final < best_score:
            best_score = M_final
            best_a = a_final.copy()
            print(f"[final‑swap] improved M={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the sequence in the format required by the outer harness
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
