"""
Propose a non‑negative sequence that minimizes the self‑convolution peak.
The algorithm works in three stages:

1. **Random seeding** – a handful of simple parametric families
   (uniform, exponential, power‑law, piece‑wise linear, symmetric
   double‑exponential).  Each candidate is projected onto the ℓ¹‑simplex
   Σa_i = √(2 n).

2. **Adam optimisation of a smoothed max** – minimise the smooth
   approximation  (1/β)·log∑exp(β·(a⋆a))  while the vector stays on the
   simplex.  β is increased to 2·10⁶ so the surrogate approaches the true
   max.  The optimiser runs until the time budget for this stage is
   exhausted.

3. **Exact hill‑climbing** – a simulated‑annealing move that transfers a
   random amount of mass from a “source’’ index (chosen preferentially
   among the contributors to the current maximal convolution entry) to a
   “target’’ index (biased towards small values).  The full convolution is
   updated in O(n) per move; moves that lower the peak are always accepted,
   otherwise they are accepted with a Metropolis probability.

If time permits the final solution is up‑sampled to double length and a
short Adam + hill‑climb polishing pass is performed – larger support often
allows a lower peak.

The routine respects the wall‑clock limit (default 1000 s) and returns the
best sequence found as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (full length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a⋆a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Families for random initialisation
# ----------------------------------------------------------------------


def _family_uniform(n: int, rng) -> np.ndarray:
    return np.ones(n, dtype=float)


def _family_exponential(n: int, rng) -> np.ndarray:
    r = rng.uniform(0.95, 0.9999)
    return r ** np.arange(n, dtype=float)


def _family_power_law(n: int, rng) -> np.ndarray:
    p = rng.uniform(0.1, 2.0)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_piecewise_linear(n: int, rng) -> np.ndarray:
    # high at the ends, low in the middle
    left = np.linspace(1.0, 0.0, n // 2, endpoint=False)
    if n % 2 == 0:
        a = np.concatenate([left, left[::-1]])
    else:
        a = np.concatenate([left, [0.0], left[::-1]])
    return a.astype(float)


def _family_double_exponential(n: int, rng) -> np.ndarray:
    r = rng.uniform(0.95, 0.999)
    idx = np.arange(n, dtype=float)
    return r ** idx + r ** (n - 1 - idx)


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a random family, add a tiny jitter and project onto the simplex."""
    families = [_family_uniform,
                _family_exponential,
                _family_power_law,
                _family_piecewise_linear,
                _family_double_exponential]
    a = families[rng.integers(len(families))](n, rng).astype(float)
    a += rng.normal(scale=1e-3, size=n)          # jitter to break ties
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log∑exp(β·b_k)  where
    b = a⋆a (full convolution). Returns (grad, b) with

        grad_i = 2·∑_k p_k·a_{k-i}   (where p_k ∝ exp(β·b_k))
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a⋆p  (note the reversal of a)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # n‑dimensional gradient
    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 2e6,
                   beta_factor: float = 1.5) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max objective.
    β is increased geometrically until ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.15
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the smoothed max
        g, _ = _softmax_gradient(a, beta)

        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # small random jitter every few hundred steps – helps escape plateaux
        if step % 300 == 0:
            jitter = rng.normal(scale=target_sum * 3e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase β gradually
        if beta < beta_target and step % 200 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Exact hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing hill‑climbing.
    Moves a random amount of mass from a source index to a target index,
    updating the full self‑convolution in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-12   # initial temperature
    cooling = 0.999

    while time.time() < deadline:
        # pick a (random) index where the current maximum is attained
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # indices that contribute to b_k
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total <= 0.0:
            continue
        # source – weighted by contribution size (larger → more likely)
        probs = contrib / total
        i = rng.choice(i_range, p=probs)

        if a[i] < 1e-14:
            continue

        # target – bias towards small a values (inverse weighting)
        inv = 1.0 / (a + 1e-12)
        probs_t = inv / inv.sum()
        j = rng.choice(n, p=probs_t)

        if i == j:
            continue

        # amount of mass to transfer
        delta = rng.random() * a[i] * 0.5   # at most 50 % of source mass
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        # Remove contribution of source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # Add contribution of target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correction for the cross term at i+j (appears twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # recompute exact convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new
