def _adam_optimize(a0, deadline, rng, init_lr=0.6, lr_decay=0.99995,
                   tau_decay=0.9999, tau_factor=5.0, min_tau=1e-6):
    n = len(a0)
    S = np.sqrt(2.0 * n)

    # logits from a0 normalized
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    tau = max(_full_convolution_fft(a0).max(), 1.0) * tau_factor
    lr = init_lr

    best_a = a0.copy()
    best_peak = _max_convolution(a0)  # compute once

    it = 0
    while time.time() < deadline:
        it += 1

        # recover a
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient and current peak
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # compute gradient wrt logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clip if needed
        clip_norm = np.linalg.norm(grad_p, ord=np.inf)
        if clip_norm > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_norm)

        m = beta1 * m + (1 - beta1) * grad_p
        v = beta2 * v + (1 - beta2) * (grad_p * grad_p)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau = max(tau * tau_decay, min_tau)
        lr *= lr_decay

        # occasional noise
        if it % 1000 == 0:
            p += rng.normal(scale=0.005, size=n)

    return best_a, best_peak
```

"""
Propose a step‑function with a low evaluation score.

The search consists of four phases:

1. **Fast random exploration** – many short runs from a rich
   collection of parametric families (including a new sparse family
   and a sinusoid family).  The best candidate is kept.

2. **Multi‑scale refinement** – the current best vector is linearly
   resized to several larger lengths and each resized copy is
   refined with a medium‑length Adam run (two restarts per length).

3. **Long final Adam optimisation** – the best length from phase 2
   receives a long Adam optimisation with a very slow decay of the
   τ‑soft‑max temperature and a relatively high learning rate.

4. **Simulated‑annealing hill‑climb** – the final few seconds are
   spent moving small fractions of mass between random indices.
   Moves are accepted if they improve the peak of the self‑convolution
   or with a Metropolis probability that decreases with a linear
   cooling schedule.

All variables are kept on the probability simplex
(`log‑softmax` parametrisation) so the ℓ₁‑norm is always
√(2 n).  Consequently the external score equals the peak of the
full convolution `np.convolve(a, a)`.  The algorithm respects the
1000 s budget, stores the resulting sequence in the global variable
`height_sequence_1` and returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where
    b = a * a (full convolution).  Returns (grad, max_b) with
    grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------


def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    raw = rng.random(n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    “U‑shape’’ – large values at the edges, strongly suppressed in the centre.
    a_i ∝ ((i+1)*(n-i))^{-p} with optional uniform mixing.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse vector – a few random positions carry equal mass."""
    # pick a small number of non‑zero positions
    k = int(rng.integers(5, max(6, n // 20 + 1)))
    idx = rng.choice(n, size=k, replace=False)
    raw = np.zeros(n)
    raw[idx] = 1.0
    return raw


def _init_sinusoid(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mild sinusoidal modulation on a positive baseline."""
    freq = rng.uniform(0.5, 5.0)
    amp = rng.uniform(0.2, 1.0)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    raw = 1.0 + amp * np.sin(2 * np.pi * freq * t / n + phase)
    return np.clip(raw, 0.0, None)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence, otherwise
    draws from a random parametric family.
    The returned vector has ℓ₁‑norm √(2 n) and is non‑negative.
    """
    S = np.sqrt(2.0 * n)

    # warm‑start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'ushape', 'sparse', 'sinusoid'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sparse':
        raw = _init_sparse(n, rng)
    elif style == 'sinusoid':
        raw = _init_sinusoid(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   init_lr: float = 0.6,
                   lr_decay: float = 0.99995,
                   tau_decay: float = 0.9999,
                   tau_factor: float = 5.0,
                   min_tau: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the (hard) peak of the self‑convolution.
    The objective is approximated by a τ‑soft‑max; τ is slowly annealed.
    Returns the best vector seen and its **hard** peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)              # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # temperature initialisation
    tau = max(_full_convolution_fft(a0).max(), 1.0) * tau_factor

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft‑max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep the best *hard* peak encountered
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        clip_norm = np.linalg.norm(grad_p, ord=np.inf)
        if clip_norm > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_norm)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - init_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau = max(tau * tau_decay, min_tau)
        init_lr *= lr_decay

        # occasional jitter to shake out from plateaus
        if it % 1200 == 0:
            p += rng.normal(scale=0.004, size=n)

    return best_a, best_peak

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """

    The best sequence is stored globally as ``height_sequence_1``.
    """

    # leave a few seconds for the final sanity check
    end_time = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf

    attempts = 0

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:

            best_n = n

    # ------------------------------------------------------------------

    if best_seq is None:

    # ------------------------------------------------------------------

    if time.time() < end_time - 30.0:
