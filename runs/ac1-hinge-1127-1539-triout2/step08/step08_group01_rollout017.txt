def _subgradient_descent_max(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    base_lr = target_sum * 0.2   # step magnitude
    while time.time() < deadline:
        t += 1
        # compute full conv via FFT
        size = _next_pow2(2*n-1)
        fa = np.fft.rfft(a, n=size)
        b = np.fft.irfft(fa * fa, n=size)[:2*n-1]
        max_b = b.max()
        # find indices where b is close to max
        eps = max_b * 1e-12 + 1e-14
        mask = (b >= max_b - eps)
        # indicator vector p (0/1)
        p = mask.astype(float)
        # gradient = 2 * (a_rev * p) (convolution)
        a_rev = a[::-1]
        conv_len = n + p.size - 1
        size2 = _next_pow2(conv_len)
        conv = np.fft.irfft(np.fft.rfft(a_rev, n=size2) * np.fft.rfft(p, n=size2), n=size2)[:conv_len]
        g = 2.0 * conv[n-1:n-1+n]
        # step size = base_lr / sqrt(t)
        step = base_lr / np.sqrt(t)
        a -= step * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
    return a
```

evaluate_sequence.  The algorithm is a multi‑stage optimisation that works
entirely within the time budget (default 1000 s).

Main ideas
----------

* **Length schedule** – start with n≈2000 and slowly increase the length.
  At each length the current best is up‑sampled and serves as a warm start.

* **Projected Adam on a smooth max** – we minimise a log‑sum‑exp
  approximation of the max‑convolution.  The smoothing parameter β is
  increased exponentially so that the optimisation finally targets the
  true max.  Several random restarts are performed; the best result is kept.

* **Deterministic sub‑gradient descent** – after the Adam phase we run a
  few thousand steps of projected sub‑gradient descent on the exact max.
  This removes the remaining bias introduced by the smooth approximation.

* **Hill‑climbing (simulated‑annealing)** – mass is moved from a source
  index to a destination index while the full self‑convolution is updated
  in O(n) per move.  The move is biased towards indices that contribute
  heavily to the current worst convolution entry and towards destinations
  that are under‑represented.  Temperature is cooled very slowly.

* **Final polish** – a long Adam run followed by a final hill‑climb are
  executed on the longest sequence obtained.

All helper routines are top‑level; there are no closures or lambdas.
The code respects the wall‑clock limit and always returns a valid
(non‑negative) list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) with b = a*a.
    Returns (g, b) where
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities (log‑sum‑exp trick)
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser on the smoothed‑max objective.
    The schedule for β (smoothness) is exponential; after a few hundred
    iterations β is huge, turning the surrogate into the true max.
    """
    a = a0.copy()
    n = a.size

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # step‑size proportional to the simplex radius
    base_lr = 0.25 * target_sum

    beta = 2.0
    beta_max = 1e12
    beta_factor = 1.6

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the smooth max
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0),
                               size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if t % 320 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max (no smoothing)
# ----------------------------------------------------------------------


def _subgradient_descent_max(a0: np.ndarray, target_sum: float,
                             deadline: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    The sub‑gradient of max(b) is 2·(a ⋆ p) where p is the 0‑1 indicator of
    the current maximal entries (or their convex hull).  A diminishing
    step size (α/√t) guarantees convergence.
    """
    a = a0.copy()
    n = a.size
    t = 0
    rng = np.random.default_rng()
    base_lr = target_sum * 0.5   # empirical

    while time.time() < deadline:
        t += 1
        # full convolution
        b = _full_convolution(a)
        max_b = b.max()
        # mask of (almost) maximal entries
        eps = max_b * 1e-12 + 1e-14
        mask = (b >= max_b - eps).astype(float)
        # sub‑gradient (see paper for derivation)
        a_rev = a[::-1]
        conv_len = n + mask.size - 1
        size2 = _next_pow2(conv_len)
        grad_full = np.fft.irfft(
            np.fft.rfft(a_rev, n=size2) * np.fft.rfft(mask, n=size2),
            n=size2
        )[:conv_len]
        g = 2.0 * grad_full[n - 1:n - 1 + n]

        # diminishing step size
        step = base_lr / np.sqrt(t)
        a -= step * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σ a = √(2 n).  Several families are mixed to enhance diversity.
    """
    target = math.sqrt(2.0 * n)
    fam = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if fam == 0:                     # power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif fam == 1:                   # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif fam == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif fam == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif fam == 4:                   # piecewise exponential
        split = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(split)),
             np.exp(-lam2 * np.arange(split, n))]
        )

    elif fam == 5:                   # near‑uniform with jitter
        a = np.full(n, target / n)
        a += rng.normal(scale=target * 0.001, size=n)

    else:                            # sinusoidal (helps flatten convolution)
        a = np.sin(np.pi * (idx + 1) / (n + 1)) ** 2 + 1e-6

    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation onto a new length followed by projection onto the
    appropriate ℓ¹‑simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing (biased simulated annealing)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray, b0: np.ndarray,
                deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Move a small amount of mass from a source index to a destination index,
    keeping the full self‑convolution up‑to‑date in O(n) per move.
    The temperature is cooled extremely slowly (0.99995) to explore the
    landscape thoroughly.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur = b.max()
    best = cur
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur * 0.12 + 1e-9          # initial temperature
    cooling = 0.99995

    while time.time() < deadline:
        # pick a worst convolution entry
        max_val = cur
        idxs = np.flatnonzero(b == max_val)
        if idxs.size == 0:
            break
        M = rng.choice(idxs)

        # source index: probability proportional to its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        src_range = np.arange(i_low, i_high + 1, dtype=int)
        partner = M - src_range
        contrib = a[src_range] * a[partner]
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            i = rng.choice(src_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-15:
            continue

        # destination index: bias towards low counterpart a[M‑j]
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart = np.zeros(n)
        counterpart[j_range] = a[M - j_range]
        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution efficiently (vectorised O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur = new_score
            if cur < best - 1e-12:
                best = cur
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising evaluate_sequence.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            prev = _project_to_simplex(prev, math.sqrt(2.0 * n0))
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – allocate increasing effort to larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 8000, 20000, 50000]

    # total remaining time will be split equally among the remaining stages
    for stage, n in enumerate(schedule):
        if time.time() >= deadline:
            break

        # time budget for this stage (equally split among the not‑yet‑processed)
        remaining = deadline - time.time()
        stages_left = len(schedule) - stage
        stage_budget = remaining / stages_left

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – multiple Adam restarts on the smoothed max
        # --------------------------------------------------------------
        n_restarts = max(2, int(stage_budget * 0.25 // 5))   # at least a few
        per_restart = stage_budget * 0.25 / max(1, n_restarts)
        local_best = None
        local_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            restart_deadline = min(deadline - 0.1, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_a, target_sum, restart_deadline)
            score = _max_convolution_fft(a_opt)
            if score < local_score:
                local_score = score
                local_best = a_opt.copy()

        if local_best is not None and local_score < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_score
            print(f"[{time.time():.1f}] Adam improvement for n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – short sub‑gradient descent (exact max)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            sub_deadline = min(deadline - 0.1,
                               time.time() + stage_budget * 0.10)
            a_sub = _subgradient_descent_max(best_seq, target_sum, sub_deadline)
            score_sub = _max_convolution_fft(a_sub)
            if score_sub < best_score - 1e-12:
                best_seq = a_sub.copy()
                best_score = score_sub
                print(f"[{time.time():.1f}] Sub‑gradient improvement for n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase C – biased hill‑climbing (simulated annealing)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hill_deadline = min(deadline - 0.1,
                                time.time() + stage_budget * 0.40)
            a_hc, b_hc = _hill_climb(a_cur, b_cur, hill_deadline)
            score_hc = b_hc.max()
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Optional after‑stage polishing (extra Adam) if time permits
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            extra_deadline = deadline - 1.0
            a_refined = _adam_optimize(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution_fft(a_refined)
            if score_ref < best_score - 1e-12:
                best_seq = a_refined.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑stage Adam improvement for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the globally best sequence (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # long Adam run
        final_deadline = deadline - 2.0
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        # short hill‑climb with a very slow cooling schedule
        b_final = _full_convolution(best_seq)
        best_seq, b_final = _hill_climb(best_seq, b_final, deadline - 1.0)
        final_score = b_final.max()
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: strip trailing near‑zeros for a tidy output
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never be needed)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation metric
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
