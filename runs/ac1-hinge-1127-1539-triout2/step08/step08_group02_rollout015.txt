"""
Optimiser for the step‑function inequality.

The algorithm consists of several stages:

1. **Broad random exploration** – generate many candidate sequences from a
   diverse collection of parametric families, including many *asymmetric* ones
   (exponential, power‑law, linear decay).  Occasionally a random permutation
   is applied to break any remaining symmetry.

2. **Scaling & intensive Adam optimisation** – the best candidate is rescaled
   to several larger lengths and for each length a dedicated Adam run is
   performed on the simplex (log‑softmax parametrisation).  This directly
   minimises the peak of the self‑convolution (the objective after the normal
  isation of the ℓ₁‑norm).

3. **Long final Adam run** – all remaining time is spent refining the current
   best length.

4. **Stochastic hill‑climb / simulated‑annealing** – cheap random mass‑transfer
   moves (and occasional swaps) are attempted; moves that improve the peak are
   kept, occasional uphill moves are allowed using a slowly cooling temperature.

The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, scaling, gradient, Adam)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2*(mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the *exact* peak of the self‑convolution.
    Returns the best sequence found and its peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # enforced ℓ₁‑norm

    # initialise logits from a feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.55                           # a bit more aggressive than before

    # temperature schedule (start fairly smooth)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # Recover a from logits (log‑softmax)
        # ------------------------------------------------------------------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------------------------------------------------------------------
        # Convolution & gradient
        # ------------------------------------------------------------------
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------------------------------------------------------------------
        # Gradient w.r.t. logits (chain rule)
        # ------------------------------------------------------------------
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ------------------------------------------------------------------
        # Adam update
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------------------------------------------------------------------
        # Schedules (very gentle decay)
        # ------------------------------------------------------------------
        tau *= 0.99985          # keep a little smoothing for longer
        lr *= 0.99995           # gentle learning‑rate decay

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak

# Parametric families for random initialisation (both symmetric & asymmetric)
# ----------------------------------------------------------------------
def _init_powerlaw_sym(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric power‑law (edge‑heavy)."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_powerlaw_asym(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law (asymmetric)."""
    idx = np.arange(n)
    p = rng.uniform(0.2, 4.0)
    offset = rng.uniform(0.0, 30.0)
    mix = rng.uniform(0.0, 1.0)
    decay = (idx + offset) ** (-p)
    base = np.ones(n)
    return mix * decay + (1.0 - mix) * base


def _init_exponential_sym(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_exponential_asym(n: int, rng: np.random.Generator) -> np.ndarray:
    """Asymmetric exponential decay from the left edge."""
    idx = np.arange(n)
    lam = rng.uniform(1e-4, 0.06)
    mix = rng.uniform(0.0, 1.0)
    decay = np.exp(-lam * idx)
    base = np.ones(n)
    return mix * decay + (1.0 - mix) * base


def _init_linear_sym(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric triangular shape."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_linear_asym(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone linear decay from the left edge."""
    idx = np.arange(n)
    max_h = rng.uniform(0.5, 8.0)
    slope = rng.uniform(0.0, max_h / (n - 1 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * idx)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    return mix * lin + (1.0 - mix) * base


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    return rng.random(n)


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    “U‑shape’’ – large values at the edges, strongly suppressed in the centre.
    a_i ∝ ((i+1)*(n-i))^{-p} with optional uniform mixing.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base

# Produce an initial feasible vector (with correct ℓ₁‑norm)
# ----------------------------------------------------------------------
def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Create a feasible start vector of length n.
    With probability ~0.2 a warm‑start (scaled previous best) is used,
    otherwise a random parametric family (including asymmetric ones) is drawn.
    """
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # ------------------------------------------------------------------
    # Warm start
    # ------------------------------------------------------------------
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # ------------------------------------------------------------------
    # Random family
    # ------------------------------------------------------------------
    style = rng.choice(['powerlaw_asym', 'exponential_asym', 'linear_asym',
                        'powerlaw_sym', 'exponential_sym', 'linear_sym',
                        'gamma', 'uniform', 'dirichlet', 'ushape'])
    if style == 'powerlaw_asym':
        raw = _init_powerlaw_asym(n, rng)
    elif style == 'exponential_asym':
        raw = _init_exponential_asym(n, rng)
    elif style == 'linear_asym':
        raw = _init_linear_asym(n, rng)
    elif style == 'powerlaw_sym':
        raw = _init_powerlaw_sym(n, rng)
    elif style == 'exponential_sym':
        raw = _init_exponential_sym(n, rng)
    elif style == 'linear_sym':
        raw = _init_linear_sym(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:
        raw = _init_uniform(n, rng)

    # optional random permutation (helps break symmetry)
    if rng.random() < 0.12:
        rng.shuffle(raw)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list) and stores it in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin for the final check

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf          # this stores the *peak* of the convolution
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 60 % of the total budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 250.0   # reserve ~250 s for refinement
    while time.time() < explore_deadline:
        attempt += 1

        # Length choice – bias towards a few thousand entries but allow
        # occasional longer runs.
        if rng.random() < 0.73:
            n = int(rng.integers(1800, 7000))
        else:
            n = int(rng.integers(7000, 20000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:4d} n={n:5d}  new best peak={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & extended Adam refinement on several lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of moderate size
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # Choose a handful of lengths (geometric progression)
    lengths = [best_n]
    for factor in [1.3, 1.7, 2.2, 3.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 30.0:
            break
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.16, 180.0)   # at most 3 min per length

        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        a_opt, peak = _adam_optimize(a_scaled,
                                      time.time() + slice_time,
                                      rng)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  improved peak={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final long Adam optimisation on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 12.0:   # leave a couple of seconds for hill‑climb
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final refine] length={best_n:5d}  best peak={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb / simulated annealing
    # ------------------------------------------------------------------
    # Temperature schedule for occasional uphill moves
    temp = 0.02
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move a random fraction of mass from i to j
        frac = rng.uniform(0.03, 0.14)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount

        # evaluate the hard peak (fast FFT based)
        peak = _max_convolution(a)

        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[local] improved peak={best_score:.6f}")
            temp = 0.02                               # reset temperature
        else:
            # occasional uphill acceptance (simulated annealing)
            if rng.random() < np.exp(-(peak - best_score) / max(temp, 1e-9)):
                best_score = peak
                best_seq = a.copy()
                temp *= 0.995                         # cool down slowly

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal peak = {best_score:.6f}  length = {len(final_seq)}")

    # store globally for possible warm‑starts in a later run
    globals()['height_sequence_1'] = final_seq
    return final_seq
