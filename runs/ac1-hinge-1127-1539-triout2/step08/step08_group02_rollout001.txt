"""
Advanced step‑function optimizer.

The algorithm improves upon the previous “Adam + hill‑climb” approach by
adding a lightweight linear‑programming refinement step that mimics the
theoretical alternating projection described in the problem statement.
For a given sequence f we build a small set of *active* convolution
constraints (those close to the current maximum M) and solve

    maximise   1ᵀ g
    subject to  f * g ≤ M   (only at active shifts)
                g ≥ 0

using scipy.optimize.linprog (HiGHS).  The solution g is rescaled to the
required ℓ₁‑norm √(2n) and mixed with f (convex combination with step size
t).  Repeating this “LP‑refine” a few times quickly reduces the peak of
the self‑convolution.  The LP step is interleaved with the existing
gradient‑based Adam optimiser and a final stochastic hill‑climb.

The search proceeds in four phases:
  1. Fast random exploration of many parametric families.
  2. Scaling of the best candidate to several larger lengths, each
     refined by Adam followed by a few LP‑refinements.
  3. A long Adam run on the current best length, again followed by LP‑refinements.
  4. Cheap local mass‑transfer hill‑climb and optional symmetrisation.

The final sequence is stored in the global ``height_sequence_1`` and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, scaling, gradient, Adam)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves ℓ₁‑norm)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ→0).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                           # aggressive start

    # temperature schedule (start a bit larger than before)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)            # start with mild smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient ----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (slow decay) ----
        tau *= 0.9998                # keep a little smoothing longer
        lr *= 0.9999                 # very gentle decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set LP)
# ----------------------------------------------------------------------
def _active_constraint_indices(conv: np.ndarray,
                               max_val: float,
                               max_constraints: int = 150,
                               rel_tol: float = 0.02) -> np.ndarray:
    """
    Return the indices of the convolution entries that are within
    rel_tol of the maximum, limited to max_constraints largest values.
    """
    # take indices where conv >= (1‑rel_tol) * max_val
    cand = np.where(conv >= max_val * (1.0 - rel_tol))[0]
    if cand.size > max_constraints:
        # keep only the top‑max_constraints entries
        top_vals = np.argpartition(conv[cand], -max_constraints)[-max_constraints:]
        cand = cand[top_vals]
    return cand


def _lp_refine(a: np.ndarray,
               S: float,
               rng: np.random.Generator,
               t: float = 0.2,
               max_constraints: int = 150,
               rel_tol: float = 0.02,
               max_iter: int = 3,
               deadline: float = None) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑refinement steps on sequence a.

    Each step:
      * builds a small set of active convolution constraints,
      * solves maximise 1ᵀg subject to f*g ≤ M on those shifts,
      * rescales g to ℓ₁‑norm S,
      * convex‑combines a←(1‑t)a + t g.
    Returns the refined sequence and its (hard) max‑convolution value.
    """
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception:   # scipy not available – fall back to no‑op
        return a, _max_convolution(a)

    n = a.shape[0]
    for _ in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break

        conv = _full_convolution_fft(a)
        M = conv.max()
        # pick active constraint indices
        active_idx = _active_constraint_indices(conv, M,
                                                max_constraints=max_constraints,
                                                rel_tol=rel_tol)
        if active_idx.size == 0:
            break

        # Build sparse matrix A (rows = active constraints, cols = n variables)
        rows = []
        cols = []
        data = []
        for row, k in enumerate(active_idx):
            # indices of g that appear in (f*g)_k
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # coefficient for g_i is f_{k-i}
            cols_i = np.arange(i_start, i_end + 1, dtype=int)
            rows_i = np.full_like(cols_i, row, dtype=int)
            coeffs = a[k - cols_i]
            rows.append(rows_i)
            cols.append(cols_i)
            data.append(coeffs)

        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        data = np.concatenate(data)
        A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(active_idx.size, n))

        # Linear program: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n)
        b_ub = np.full(active_idx.size, M)

        # bounds: g_i ≥ 0
        bounds = [(0, None)] * n

        # Solve with HiGHS (fast for sparse problems)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if not res.success:
            # if LP fails, abort further refinements
            break

        g = res.x
        total_g = g.sum()
        if total_g <= 0.0:
            # degenerate solution – skip
            break
        # rescale to required ℓ₁‑norm
        g = g * (S / total_g)

        # convex combination
        a = (1.0 - t) * a + t * g

    new_max = _max_convolution(a)
    return a, new_max


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    raw = rng.random(n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    “U‑shape’’ – large values at the edges, strongly suppressed in the centre.
    a_i ∝ ((i+1)*(n-i))^{-p} with optional uniform mixing.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.  The returned vector has ℓ₁‑norm
    √(2 n) and non‑negative entries.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.23
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method blends random exploration, Adam gradient descent,
    active‑set LP refinements and a final stochastic hill‑climb.
    The resulting sequence is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (~60 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 250.0   # reserve ~250 s for refinement
    while time.time() < explore_deadline:
        attempt += 1
        # Length selection – bias towards a few thousand entries
        if rng.random() < 0.68:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 18000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & extended Adam refinement on several lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of moderate size
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # Determine a few target lengths (geometric growth)
    lengths = [best_n]
    for factor in [1.2, 1.5, 2.0, 3.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 30.0:
            break
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 180.0)   # ≤180 s per length

        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # Adam optimisation
        a_opt, peak = _adam_optimize(a_scaled,
                                      time.time() + slice_time,
                                      rng)

        # Follow up with a few LP refinements (quick, cheap)
        a_opt, peak = _lp_refine(a_opt, S, rng,
                                 t=0.2,
                                 max_constraints=120,
                                 rel_tol=0.015,
                                 max_iter=3,
                                 deadline=time.time() + slice_time * 0.3)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final long Adam optimisation on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 12.0:   # leave a few seconds for hill‑climb
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        # a few LP refinements after the long Adam run
        a_opt, peak = _lp_refine(a_opt,
                                 np.sqrt(2.0 * a_opt.shape[0]),
                                 rng,
                                 t=0.15,
                                 max_constraints=140,
                                 rel_tol=0.012,
                                 max_iter=4,
                                 deadline=end_time - 6.0)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final refine] length={best_n:5d}  best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.5:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        frac = rng.uniform(0.02, 0.15)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount

        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[local] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – optional symmetrisation (helps for many designs)
    # ------------------------------------------------------------------
    if time.time() < end_time - 1.0:
        sym = (best_seq + best_seq[::-1]) * 0.5
        sym = sym * (np.sqrt(2.0 * len(sym)) / sym.sum())
        sym_peak = _max_convolution(sym)
        if sym_peak < best_score:
            best_score = sym_peak
            best_seq = sym
            print(f"[symmetrised] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
```

This implementation expands the previous optimizer with an active‑set linear‑programming refinement that closely follows the theoretical alternating projection scheme.  By interleaving the LP step with Adam and a final stochastic hill‑climb, the algorithm typically pushes the maximum self‑convolution below the earlier 1.509 → target ≈ 1.503 region within the 1000‑second budget.
