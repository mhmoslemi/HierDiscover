"""
Minimal functional search for a non‑negative step sequence with low
benchmark score.

Algorithm overview
------------------
1. Choose a problem size `n` based on the time budget.
2. Warm‑start with a 2‑parameter family (power‑law × exponential) on a
   reduced problem and interpolate to the full size.
3. Exponential (mirror) descent on a soft‑max surrogate of the maximal
   convolution entry.
4. Projected gradient descent with random restarts (same surrogate).
5. Deterministic mass‑transfer refinement that moves a small fraction of
   mass from indices contributing most to the current maximal
   convolution entry to low‑contributing indices.
6. Simulated‑annealing‑style local search: repeatedly move a random
   amount of mass from a donor (chosen proportionally to its contribution
   to the active convolution peak) to a low‑mass receiver, accepting
   occasional uphill moves with a decaying temperature.
7. Return the best sequence found (truncated trailing zeros) and store it
   globally for warm‑starts in future calls.
"""

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with exponential updates using a soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – projected gradient descent with restarts (re‑used)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate of max(convolution).
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Search a two‑parameter family
        a_i = (i+1)^(-p) * exp(-λ·i / small_n)
    on a small problem.  Returns the best sequence found.
    """
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 5 – deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass from indices that contribute
    most to the current maximal convolution entry to indices that contribute
    least, accepting only improvements.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        # sensitivities (product contributions) for the current max entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        total_w = weight.sum()
        if total_w > 0:
            donor = rng.choice(n, p=weight / total_w)
        else:
            donor = rng.integers(n)

        low_thr = np.percentile(weight, 10)
        low_idx = np.where(weight <= low_thr)[0]
        receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

        if donor == receiver:
            continue

        max_frac = 0.03
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise occasionally
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 6 – simulated‑annealing‑style refinement
# ----------------------------------------------------------------------


def _simulated_annealing_refine(a0: np.ndarray,
                               target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """
    Local stochastic search that moves a small amount of mass from a
    donor (chosen proportionally to its contribution to the active
    convolution peak) to a low‑mass receiver, accepting occasional
    uphill moves with a decaying temperature.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    T0 = 1e-3
    Tmin = 1e-7
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # product contributions for donor selection
        length = n - k_max
        if length > 0:
            prods = a[:length] * a[k_max:k_max + length]
            tot = prods.sum()
            if tot > 0:
                donor = rng.choice(length, p=prods / tot)
            else:
                donor = rng.integers(length)
        else:
            donor = rng.integers(n)

        # receiver: pick from the low‑mass tail (bottom 20 %)
        thr = np.percentile(a, 20)
        low_idxs = np.where(a <= thr)[0]
        if low_idxs.size:
            receiver = rng.choice(low_idxs)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # move at most 5 % of the donor mass
        max_frac = 0.05
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        cur_sum = a_new.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            a_new *= (target_sum / cur_sum)

        new_val = _score(a_new)

        # Acceptance rule
        if new_val < cur_val:
            accept = True
        else:
            T = max(T0 * (0.995 ** it), Tmin)
            prob = np.exp(-(new_val - cur_val) / T)
            accept = rng.random() < prob

        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improve score={best_val:.6f}")

        # occasional global jitter to shake the system
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            cur_val = _score(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5           # safety margin

    # ---- problem size --------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ---- initial sequence (fallback) ------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ---- Stage 1: cheap parametric warm‑start ---------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.10)
    best_small_seq, _ = _grid_param_search(small_n, small_target, param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ---- Stage 2: exponential descent ----------------------------------
    exp_deadline = time.time() + (deadline - time.time()) * 0.15
    a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
    best_seq = a_exp
    best_score = val_exp
    print(f"[{time.time():.2f}] After exponential GD best score={best_score:.6f}")

    # ---- Stage 3: projected GD with restarts ---------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.35
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # ---- Stage 4: deterministic mass‑transfer refinement ---------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.25
        a_mt, val_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After transfer refinement best score={best_score:.6f}")

    # ---- Stage 5: simulated‑annealing style refinement -----------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, val_sa = _simulated_annealing_refine(best_seq,
                                                   target_sum,
                                                   sa_deadline)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
        print(f"[{time.time():.2f}] After simulated annealing best score={best_score:.6f}")

    # ---- Clean up and final evaluation ---------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # evaluate (the external function is assumed to be present)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
