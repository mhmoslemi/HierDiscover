"""
Propose a step‑function sequence that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Broad random exploration** – many simple parametric families
   (exponential, power‑law, linear, asymmetric…) are sampled.
   The best candidate (lowest hard max‑convolution) is remembered.

2. **Length scaling + long Adam optimisation** – the current best vector
   is linearly interpolated to several larger lengths (up to ~60 000)
   and each copy is refined with an unconstrained Adam optimiser that
   directly minimises a τ‑soft‑max approximation of the convolution
   peak.  The Adam routine works on the log‑softmax parametrisation,
   guaranteeing non‑negativity and the ℓ₁‑norm invariance of the
   objective.  The temperature τ decays very slowly, giving a smooth
   approach to the true max.

3. **Focused stochastic hill‑climbing** – after the gradient phase a
   cheap random “mass‑transfer” loop moves a small fraction of height
   from a high‑value entry to a low‑value one, keeping any improvement.
   This can slip through tiny plateaus that the smooth optimiser may
   miss.

The whole search is time‑boxed (default 1000 s).  All progress is
printed; the final sequence is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Utility functions (FFT based convolution etc.)
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves sum)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# -------------------------------------------------------------
# τ‑soft‑max gradient (differentiable approximation of max)
# -------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂/∂a τ‑soft‑max(b) and
    b = a * a (full convolution).  The gradient is the exact
    sub‑gradient of the weighted sum of convolutions.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # g = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# -------------------------------------------------------------
# Adam optimiser on the probability simplex (log‑softmax)
# -------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Adam optimiser that minimises the true max‑convolution.
    The parametrisation is p = log(softmax) so that the ℓ₁‑norm is
    fixed to S = sqrt(2 n) (any scaling would drop out of the ratio).
    Returns the best vector seen and its hard max.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm (does not affect the ratio)

    # initialise logits from the feasible a0
    eps = 1e-12
    a0 = np.maximum(a0, 0.0)
    a0_sum = a0.sum()
    if a0_sum == 0.0:
        a0 = np.full(n, 1.0 / n)
        a0_sum = 1.0
    a0 = a0 * (S / a0_sum)              # enforce ℓ₁‑norm = S
    soft = a0 / S
    p = np.log(np.maximum(soft, eps))   # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.30                           # a bit more conservative than the previous version

    # start with a generous temperature (smooth max)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 2.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # --- recover a from logits (log‑softmax) ---
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # --- convolution & gradient ---
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # --- gradient wrt logits (chain rule) ---
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for numerical stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 8.0 * S:
            grad_p = grad_p * (8.0 * S / clip_val)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- schedules (very slow decay) ---
        tau *= 0.99995                  # keep a little smoothing for a long time
        lr *= 0.99995

        # occasional jitter to escape plateaus
        if it % 1200 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak

def _init_exponential_left(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential from the left side."""
    lam = rng.uniform(0.0008, 0.02)          # decay rate
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * idx)
    return raw


def _init_powerlaw_left(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law from the left side."""
    p = rng.uniform(0.3, 2.2)
    idx = np.arange(1, n + 1, dtype=float)
    raw = idx ** (-p)
    return raw


def _init_linear_left(n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear ramp from a positive value down to zero at the right end."""
    max_h = rng.uniform(0.6, 4.5)
    raw = np.linspace(max_h, 0.0, n)
    return raw


def _init_ushape_asym(n: int, rng: np.random.Generator) -> np.ndarray:
    """Asymmetric U‑shape: heavy on left, light on right."""
    i = np.arange(n, dtype=float)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.5, 2.0)
    ush = prod ** (-p)
    # force asymmetry by scaling right half
    factor = rng.uniform(0.2, 0.7)
    ush[n // 2 :] *= factor
    base = np.ones(n)
    mix = rng.uniform(0.0, 1.0)
    return mix * ush + (1.0 - mix) * base


def _init_random_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse vector: a few random positions get random mass."""
    raw = np.zeros(n)
    k = rng.integers(3, 8)               # number of spikes
    positions = rng.choice(n, size=k, replace=False)
    masses = rng.random(k) + 0.1
    raw[positions] = masses
    # add a tiny background to keep sum positive
    raw += rng.random(n) * 1e-3
    return raw


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    raw = rng.random(n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    20 % of the time a warm‑start (scaled previous best) is used,
    otherwise a random family is drawn.
    """
    S = np.sqrt(2.0 * n)

    # warm start
    if warm_seq is not None and rng.random() < 0.21:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            return a * (S / total)

    style = rng.choice(['exp_left', 'pow_left', 'lin_left',
                        'ushape_asym', 'sparse', 'dirichlet'])
    if style == 'exp_left':
        raw = _init_exponential_left(n, rng)
    elif style == 'pow_left':
        raw = _init_powerlaw_left(n, rng)
    elif style == 'lin_left':
        raw = _init_linear_left(n, rng)
    elif style == 'ushape_asym':
        raw = _init_ushape_asym(n, rng)
    elif style == 'sparse':
        raw = _init_random_sparse(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    return raw * (S / total)
