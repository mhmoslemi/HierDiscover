        # Power law

            max_b = _max_convolution_fft(a)

                best_seq = a.copy()

"""
Improved search for the minimal maximal entry of a self‑convolution.

Strategy
--------
1. **Quick parametric sweep** – generate a few thousand candidates from
   simple families (exponential, power‑law, linear, piece‑wise exponential).
   For exponentials we sweep the decay rate `r = 1 - c/n` with `c` in a
   broad log‑range; this concentrates the search around the theoretically‑
   optimal scaling `c ≈ 1…2`.  Every candidate is projected onto the ℓ¹‑simplex
   (∑a_i = √(2n)) and its maximal convolution entry is evaluated by an FFT.

2. **Select the best shape** – keep the sequence with the smallest
   `max(conv(a,a))` found so far.

3. **Upsample** – if the best shape is short, linearly interpolate it to a
   larger support (default 8000 elements).  Larger supports give a finer
   discretisation of the continuous optimum, typically lowering the bound.

4. **Long hill‑climbing** – a simulated‑annealing move set that transfers a
   random amount of mass from a “source” index (chosen among contributors to
   the current maximal convolution entry) to a “target” index (biased towards
   low‑mass positions).  The full convolution is kept up‑to‑date in O(n) per
   move using NumPy slice operations.  The temperature is slowly cooled,
   allowing occasional uphill moves.  The loop runs until the wall‑clock
   budget expires, exploiting the whole 1000 s budget.

5. **Final evaluation** – the sequence is trimmed of trailing near‑zeros,
   reported to the provided `evaluate_sequence` function and returned as a
   plain list of floats.

The algorithm respects the time limit, prints progress messages and uses only
NumPy and the standard library (no external files or network access)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex
       {x | x_i ≥ 0, Σ x_i = s}.  Duchi et al. (2008) implementation."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Random families of step functions
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0 < r < 1).  r is drawn uniformly close to 1."""
    # r = 1 - c/n, with c sampled log‑uniformly
    c = rng.uniform(0.5, 8.0)          # broad range for the decay constant
    r = 1.0 - c / n
    if r <= 0.0:
        r = 0.5
    idx = np.arange(n, dtype=float)
    return r ** idx


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    p = rng.uniform(0.2, 2.0)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential decay with a random breakpoint."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = 1.0 - rng.uniform(0.5, 2.0) / n
    r2 = 1.0 - rng.uniform(2.0, 10.0) / n
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a random family, add tiny jitter and project onto the ℓ¹‑simplex."""
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)          # break exact ties
    a = np.maximum(a, 0.0)
    target = math.sqrt(2.0 * n)
    return _project_to_simplex(a, target)


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex(a_new, target)

# Hill‑climbing (simulated annealing) – exact convolution updates
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a source index to a target index while
    keeping the full convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9               # initial temperature
    cooling = 0.998

    while time.time() < deadline:
        # pick a position attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – weighted by its contribution to entry k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias toward light positions
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 75 % of the source)
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        # removal of source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # addition of target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correction for the double‑counted cross term at i+j
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # recompute a clean convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock limit and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Seed from a previously found construction, if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, math.sqrt(2.0 * n0))
            score0 = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = score0
            print(f"[{time.time():.1f}] seed from height_sequence_1 (n={n0}) score={score0:.6f}")
        except Exception as e:
            print(f"Could not use height_sequence_1: {e}")

    # --------------------------------------------------------------
    # 1️⃣   Small‑scale parametric exploration
    # --------------------------------------------------------------
    # Support sizes we shall try – a few are cheap enough for exhaustive sweeps
    small_lengths = [2000, 4000, 8000]

    for n in small_lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ---- exponential family (r = 1 - c/n) ----
        # c is sampled log‑uniformly; this concentrates the search around the
        # theoretically optimal scaling (c ≈ 1…2).
        for c in np.logspace(-0.3, 1.0, num=35):      # c ∈ [0.5, 10]
            if time.time() >= deadline:
                break
            eps = c / n
            r = 1.0 - eps
            if r <= 0.0:
                continue
            # finite geometric sum
            denom = 1.0 - r
            geo_sum = (1.0 - r**n) / denom
            scale = target / geo_sum
            a = scale * (r ** np.arange(n, dtype=float))
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} exp c={c:.3f} score={score:.6f}")

        # ---- power‑law family ----
        for p in np.linspace(0.1, 2.5, num=32):
            if time.time() >= deadline:
                break
            idx = np.arange(1, n + 1, dtype=float)
            a_raw = idx ** (-p)
            scale = target / a_raw.sum()
            a = a_raw * scale
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} power p={p:.3f} score={score:.6f}")

        # ---- a few random draws from the mixed families ----
        for _ in range(12):
            if time.time() >= deadline:
                break
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} random score={score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣   Upsample to a larger support for a finer discretisation
    # --------------------------------------------------------------
    final_n = max(best_seq.size, 8000) if best_seq is not None else 8000
    if best_seq is None or best_seq.size != final_n:
        best_seq = _upsample_sequence(best_seq if best_seq is not None else np.zeros(1),
                                      final_n)
    else:
        # ensure exact simplex projection after possible up‑sampling
        best_seq = _project_to_simplex(best_seq, math.sqrt(2.0 * final_n))

    # --------------------------------------------------------------
    # 3️⃣   Exhaustive hill‑climbing with the remaining budget
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        cur_b = _full_convolution(best_seq)
        best_seq, cur_b = _hill_climbing(best_seq, cur_b, deadline)
        best_score = float(cur_b.max())
        print(f"[{time.time():.1f}] post‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣   Trim trailing near‑zeros and final API evaluation
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error during final evaluation: {exc}")

    return best_seq.tolist()
