"""
Improved step‑function search.

The code follows a three‑stage strategy:

1. **Exploration** – generate many random analytic families (beta, power‑law,
   exponential, Gaussian, sinusoidal, logistic) possibly reordered by an
   alternating permutation that spreads large values.  Keep the K best
   candidates (lowest self‑convolution peak).

2. **Refinement** – for each elite run a long projected‑Adam optimisation on a
   soft‑max approximation of the peak (α grows to ≈800), then a projected
   sub‑gradient descent on the true max entry and finally an aggressive
   random‑swap/position‑swap hill‑climb.  All updates are projected onto the
   simplex {a≥0, Σa=√(2·n)} and the vector is symmetrised after each step
   (the optimum is expected to be symmetric).

3. **Upsampling & final polishing** – linearly upsample the best vector to
   larger lengths (×1.2, ×1.5, ×2.0, bounded by 20000) and give each a short
   Adam burst followed by a final hill‑climb.  Any remaining budget is spent
   on a final long Adam run.

The routine returns the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear self‑convolution via FFT (length = 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Return the peak of the self‑convolution
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Enforce exact mirror symmetry (v[i] = v[-1-i])
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear interpolation to a new length (preserves shape)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Alternate permutation – places large values at the two ends alternately
# ----------------------------------------------------------------------
def _alternate_permute(a: np.ndarray) -> np.ndarray:
    """Place sorted values in order 0, n‑1, 1, n‑2, 2, …."""
    vals = np.sort(a)[::-1]          # descending
    n = a.shape[0]
    res = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    idx = 0
    while left <= right:
        if toggle:
            res[left] = vals[idx]
            left += 1
        else:
            res[right] = vals[idx]
            right -= 1
        toggle = not toggle
        idx += 1
    return res


# ----------------------------------------------------------------------
# Candidate families (non‑negative vectors)
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    fam = rng.integers(0, 8)
    if fam == 0 and base is not None:
        a = _upsample_seq(base, n)
    elif fam == 1:
        a = rng.random(n)                                 # pure uniform random
    elif fam == 2:
        # symmetric power‑law (i·(n‑i))^p
        p = rng.uniform(0.1, 5.0)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 3:
        # symmetric beta‑like (x·(1‑x))^p
        p = rng.uniform(0.2, 8.0)
        x = np.linspace(0.0, 1.0, n)
        a = (x * (1.0 - x)) ** p
    elif fam == 4:
        # symmetric exponential decay from both ends
        rate = rng.uniform(0.90, 0.9999)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) * 0.5
    elif fam == 5:
        # Gaussian‑like bump
        mu = (n - 1) / 2.0
        sigma = rng.uniform(0.1, 0.5) * n
        xs = np.arange(n, dtype=float)
        a = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    elif fam == 6:
        # sinusoidal power
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    else:
        # logistic bump
        scale = rng.uniform(0.02, 0.15) * n
        mid = (n - 1) / 2.0
        xs = np.arange(n, dtype=float)
        left = 1.0 / (1.0 + np.exp((xs - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - xs) / scale))
        a = left + right

    # optional alternating permutation to spread large entries
    if rng.random() < 0.35:
        a = _alternate_permute(a)

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation of the max‑convolution
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the smooth max
    alpha = 0.2
    alpha_mul = 1.04
    max_alpha = 800.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max probabilities over convolution entries
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_shifted = np.exp(scaled - max_scaled)
        w = exp_shifted / exp_shifted.sum()

        # gradient of Σ w_i (a*a)_i  w.r.t. a
        w_rev = w[::-1]
        conv_len = n + (2 * n - 1) - 1          # = 3n‑2
        fft_len = 1 << (conv_len - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        Fw = np.fft.rfft(w_rev, n=fft_len)
        full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99996       # very gentle decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the genuine max entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the selected entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random mass transfer (simulated‑annealing flavour)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        # early break if stagnating
        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap / position‑swap hill‑climb (local refinement)
# ----------------------------------------------------------------------
def _local_swap_and_perm_search(a0: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # ---- move a tiny chunk from i to j ----
        if a[i] > 0.0:
            delta = 0.02 * a[i] * rng.random()
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand, S)
            if enforce_sym:
                cand = _symmetrize(cand)
            cand_score = _peak_convolution(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                continue   # go to next iteration

        # ---- full value swap (no projection needed) ----
        cand = a.copy()
        cand[i], cand[j] = cand[j], cand[i]          # keep sum unchanged
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _peak_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand

    return best_a, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    eval = 2·n·max(convolution) / (sum²).  Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Warm‑start: reuse previous best if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – cheap exploration, keep the K elite candidates
    # ------------------------------------------------------------------
    K = 8
    elite = []                    # list of (score, vector)
    attempts = 0
    phase0_deadline = start_time + 0.18 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1
        # pick a length (favor a modest perturbation of the previous best)
        if base_seq is not None and rng.random() < 0.55:
            scale = rng.uniform(0.85, 1.15)
            n = max(600, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(2000, 12001))

        a = _gen_candidate(n, rng, base_seq)

        # Normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # optional enforced symmetry (helps convergence)
        if rng.random() < 0.5:
            a = _symmetrize(a)

        cur_score = _peak_convolution(a)

        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback if exploration yielded nothing
    # ------------------------------------------------------------------
    if not elite:
        n = 5000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        elite.append((_peak_convolution(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – refinement of each elite candidate
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        # Edge case – out of time already
        best_overall_seq = elite[0][1]
        external_score = evaluate_sequence(best_overall_seq.tolist())
        globals()['height_sequence_1'] = best_overall_seq.copy()
        return best_overall_seq.tolist()

    # allocate equal budget per elite
    time_per_elite = remaining / len(elite)

    best_overall_seq = None
    best_overall_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        # sub‑budgets
        ada_dead = now + 0.55 * time_per_elite
        sub_dead = ada_dead + 0.30 * time_per_elite
        loc_dead = sub_dead + 0.15 * time_per_elite

        # Adam (smooth max)
        a_adam, sc_adam = _optimize_adam(vec, ada_dead, rng, enforce_sym=True)

        # Sub‑gradient on true max
        a_sub, sc_sub = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=True)

        # Tiny random‑swap / position‑swap hill‑climb
        a_loc, sc_loc = _local_swap_and_perm_search(a_sub, loc_dead, rng, enforce_sym=True)

        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[refine {idx}] improved to {best_overall_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – up‑sampling to larger lengths (if time permits)
    # ------------------------------------------------------------------
    max_len = 20000
    up_factors = [1.2, 1.5, 2.0, 2.5]
    for fac in up_factors:
        if best_overall_seq is None:
            break
        new_n = int(best_overall_seq.shape[0] * fac)
        if new_n > max_len or new_n < 2:
            continue
        a_up = _upsample_seq(best_overall_seq, new_n)

        # re‑normalise to new length
        S_new = np.sqrt(2.0 * new_n)
        a_up = a_up / a_up.sum() * S_new
        a_up = _symmetrize(a_up)

        now = time.time()
        if now >= deadline:
            break
        # short Adam burst (≈7 % of remaining time)
        short_dead = min(deadline, now + 0.07 * (deadline - now))
        a_up_opt, sc_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=True)

        if sc_up < best_overall_score:
            best_overall_score = sc_up
            best_overall_seq = a_up_opt.copy()
            print(f"[upsample] factor={fac:.2f} new_n={new_n} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing with any remaining budget
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline and best_overall_seq is not None:
        a_final, sc_final = _local_swap_and_perm_search(best_overall_seq,
                                                        deadline,
                                                        rng,
                                                        enforce_sym=True)
        if sc_final < best_overall_score:
            best_overall_score = sc_final
            best_overall_seq = a_final.copy()
            print(f"[final‑local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (store globally for warm‑restarts)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
