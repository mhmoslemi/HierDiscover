def solve_lp_max_sum_g(f, M):
    n = len(f)
    # A_ub shape (2n-1, n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    b = np.full(2*n-1, M, dtype=np.float64)
    # bounds for g variables: (0, None)
    bounds = [(0, None)] * n
    # objective: minimize -sum(g) to maximize sum(g)
    c = -np.ones(n, dtype=np.float64)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    else:
        raise RuntimeError("LP failed")

def monotone_decreasing_projection(v):
    n = len(v)
    # treat negative of v and apply increasing isotonic regression
    y = -v
    # PAVA
    w = y.copy()
    # block start indices and lengths
    start = [0]
    length = [1]
    avg = [y[0]]
    for i in range(1, n):
        start.append(i)
        length.append(1)
        avg.append(y[i])
        # merge while last avg <= previous avg
        while len(avg) >= 2 and avg[-2] > avg[-1] + 1e-12:
            # merge last two blocks
            total_len = length[-2] + length[-1]
            total_avg = (avg[-2] * length[-2] + avg[-1] * length[-1]) / total_len
            avg[-2] = total_avg
            length[-2] = total_len
            start.pop()
            length.pop()
            avg.pop()
    # now fill output
    out = np.empty(n, dtype=float)
    idx = 0
    for s, l, a in zip(start, length, avg):
        out[idx:idx+l] = a
        idx += l
    # revert sign
    return -out

   linear programme
        maximise   Σ g_j
        subject to g_j ≥ 0  and (f∗g)_k ≤ M   (M = max(f∗f))
   using ``scipy.optimize.linprog``.  The solution g is scaled to the
   same ℓ¹‐norm as f and mixed with f (convex combination with step‑size
   t≈0.2).  This step provably never raises the supremum and often
   yields a strict decrease.

3. **Adam descent on a smoothed max** – we minimise a smooth
   approximation of the maximum entry of the self‑convolution,
   (1/β)·log∑exp(β·(a∗a)).  The gradient is obtained analytically via
   FFTs and Adam’s adaptive step‑size.  β is increased geometrically
   from a small start to a large target (≈10⁶), concentrating the
   optimisation on the true maximum.

4. **Monotone projection + simplex projection** – after each major
   phase we enforce a non‑increasing shape (PAVA) and project onto the
   ℓ¹‑simplex {x≥0, Σx = √(2 n)}.  The projection is cheap (O(n log n))
   and respects the normalisation required by ``evaluate_sequence``,
   whose value then equals the current maximal convolution entry.

5. **Hill‑climbing (optional for small n)** – a simulated‑annealing
   move that shifts a random amount of mass from a “source” index to a
   “target” index while updating the full convolution in O(n).  It is
   only invoked when enough time remains and n is modest.

All phases respect the global deadline; the best sequence seen so far
is kept and finally returned (trailing near‑zeros are trimmed).  The
function prints progress messages and the final value obtained by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a (FFT, length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a∗a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Monotone (non‑increasing) projection – PAVA
# ----------------------------------------------------------------------


def _monotone_decreasing_projection(v: np.ndarray) -> np.ndarray:
    """Isotonic regression onto a non‑increasing sequence (L2)."""
    # Apply PAVA to -v to obtain an increasing sequence,
    # then restore the sign.
    y = -v.astype(float)
    n = len(y)
    # blocks: start indices, lengths and averages
    start = [0]
    length = [1]
    avg = [y[0]]
    for i in range(1, n):
        start.append(i)
        length.append(1)
        avg.append(y[i])
        # Merge while last average is larger than the new one
        while len(avg) >= 2 and avg[-2] > avg[-1] + 1e-12:
            tot_len = length[-2] + length[-1]
            tot_avg = (avg[-2] * length[-2] + avg[-1] * length[-1]) / tot_len
            avg[-2] = tot_avg
            length[-2] = tot_len
            start.pop()
            length.pop()
            avg.pop()
    out = np.empty(n, dtype=float)
    idx = 0
    for s, l, a in zip(start, length, avg):
        out[idx:idx + l] = a
        idx += l
    return -out  # revert sign


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum
        (1/β)·log Σ exp(β·b_k)   where b = a∗a.
    Returns (grad, b) where

        grad_i = 2·∑_k p_k·a_{k-i},
        p_k = exp(β·b_k) / Σ exp(β·b).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev ⋆ p   (the gradient factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 1e6,
                   beta_factor: float = 1.3) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max objective.
    The schedule for β grows geometrically until ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection onto the simplex (keeps the evaluation invariant)
        a = _project_to_simplex(a, target_sum)

        # occasional isotonic projection – keeps the shape sensible
        if step % 250 == 0:
            a = _monotone_decreasing_projection(a)
            a = _project_to_simplex(a, target_sum)

        # jitter occasionally to escape plateaux
        if step % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase β
        if beta < beta_target and step % 250 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Simple parametric families (used for random restarts)
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)."""
    r = rng.uniform(0.96, 0.9995)
    idx = np.arange(n, dtype=float)
    return r ** idx


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential decay with a breakpoint."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a random family, add tiny jitter and project onto the simplex."""
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=1e-3, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, np.sqrt(2.0 * n))
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = _project_to_simplex(a_new, np.sqrt(2.0 * new_n))
    return a_new


# ----------------------------------------------------------------------
# LP improvement (small lengths only)
# ----------------------------------------------------------------------


def _lp_improvement_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    One LP‑based improvement:
        maximise Σ g_j   s.t.   g≥0,   (f∗g)_k ≤ M   for all k.
    The solution g is scaled to have the same ℓ¹‑norm as f and mixed with f.
    Returns the updated f (or the original f if the LP failed or gave no
    improvement).
    """
    try:
        import scipy.optimize as opt
    except Exception:
        return f  # no LP capability

    n = f.size
    # Build A matrix: each column j contributes f shifted by j positions.
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(rows, M, dtype=float)
    bounds = [(0, None)] * n
    # maximise Σ g  <=> minimise -Σ g
    c = -np.ones(n, dtype=float)

    # Use HiGHS interior‑point solver; it is fast for our modest sizes.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success:
        return f

    g = res.x
    sum_f = f.sum()
    sum_g = g.sum()
    if sum_g <= sum_f + 1e-12:
        return f

    # Scale g to the same ℓ¹‑norm as f
    g = g * (sum_f / sum_g)

    # convex combination (t chosen modestly)
    t = 0.2
    f_new = (1.0 - t) * f + t * g
    f_new = _project_to_simplex(f_new, target_sum)
    f_new = _monotone_decreasing_projection(f_new)
    f_new = _project_to_simplex(f_new, target_sum)
    return f_new


# ----------------------------------------------------------------------
# Hill climbing (exact O(n) updates) – kept for modest n
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a source index to a target index
    while keeping the full convolution up‑to‑date in O(n) per move.
    Returns the best a found and its exact convolution.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9
    cooling = 0.998

    while time.time() < deadline:
        # pick a position attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – choose among contributors, weighted by product size
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias towards small‑mass positions
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- update convolution (exact O(n) slices) ----
        # Remove contribution of source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # Add contribution of target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correct the cross term at i+j (appears twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
        T *= cooling

    b_best = _full_convolution(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock limit and returns the best sequence
    found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Length schedule – start small, finish large.
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise at this length (upsample previous best if possible)
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)

        a = _monotone_decreasing_projection(a)
        a = _project_to_simplex(a, target_sum)

        # ------------------------------------------------------------------
        # 1) Optional LP improvement (only for modest n)
        # ------------------------------------------------------------------
        if n <= 3000:
            max_before = _max_convolution_fft(a)
            # perform a few LP‑based refinements
            for _ in range(3):
                if time.time() >= deadline - 5:
                    break
                a = _lp_improvement_step(a, max_before, target_sum)
                max_before = _max_convolution_fft(a)
            cur_val = max_before
        else:
            cur_val = _max_convolution_fft(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[{time.time():.1f}] init (n={n}) improvement: max={cur_val:.6f}")

        # ------------------------------------------------------------------
        # 2) Adam descent on smoothed max
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate roughly 50 % of the remaining budget for this length
        gd_time = min(remaining * 0.50, 250.0)
        gd_deadline = time.time() + gd_time
        a = _adam_optimize(a, gd_deadline,
                           beta_start=2.0,
                           beta_target=1e6,
                           beta_factor=1.3)
        a = _monotone_decreasing_projection(a)
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Adam (n={n}) improvement: max={cur_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Exact hill climbing (cheap for n ≤ 8000)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0 and n <= 8000:
            hc_deadline = time.time() + remaining * 0.30
            b_cur = _full_convolution(a)
            a_hc, b_hc = _hill_climbing(a, b_cur, hc_deadline)
            cur_val = float(b_hc.max())
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] hill‑climb (n={n}) improvement: max={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – if time permits, another Adam run on the best.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a_final = _adam_optimize(best_seq.copy(),
                                 deadline - 2.0,
                                 beta_start=4.0,
                                 beta_target=2e6,
                                 beta_factor=1.3)
        a_final = _monotone_decreasing_projection(a_final)
        a_final = _project_to_simplex(a_final, target_sum)
        final_val = _max_convolution_fft(a_final)
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final Adam improvement: max={final_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy return value
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a simple uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (max‑convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
