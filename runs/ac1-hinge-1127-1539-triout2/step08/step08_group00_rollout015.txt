def optimize_sequence(a_init, deadline):
    a = a_init.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _max_convolution_fft(a)
    while time.time() < deadline:
        # compute convolution
        b = _full_convolution(a)
        M = b.max()
        if M < best_val - 1e-12:
            best_val = M
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best (LP) val={best_val:.6f}")

        # tolerance for active set
        tol = 1e-12 * M + 1e-12  # relative tolerance
        active_idxs = np.where(b >= M - tol)[0]
        # Usually there might be many due to plateau; limit to e.g., first 20
        if active_idxs.size > 20:
            # Choose random subset or those far apart
            # pick every such index by stride, but for simplicity, randomly sample.
            rng = np.random.default_rng()
            active_idxs = rng.choice(active_idxs, size=20, replace=False)

        # Build LP
        #!
        # matrix rows indices.
        rows = []
        rhs = []
        for k in active_idxs:
            # Build coefficient row length n+1
            coeff = np.zeros(n+1, dtype=float)
            # i ranges for conv
            i_low = max(0, k - (n - 1))
            i_high = min(n-1, k)
            # i from i_low..i_high
            # j = k - i
            # contribution a_i * x_j
            # we can fill vector: for each i, add a_i to column j.
            i_vals = np.arange(i_low, i_high+1)
            j_vals = k - i_vals
            coeff[j_vals] += a[i_vals]
            coeff[-1] = 1.0  # epsilon coefficient
            rows.append(coeff)
            rhs.append(0.0)
        A_ub = np.vstack(rows) if rows else None
        b_ub = np.array(rhs) if rhs else None

        # equality constraint sum(x)=0
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # bounds
        bounds = [( -a[i], None) for i in range(n)] + [(0.0, None)]

        # objective
        c = np.zeros(n+1)
        c[-1] = -1.0

        # Solve
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                     bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # LP failed, break or try other heuristic
            print(f"[{time.time():.1f}] LP failed: {res.message}")
            break
        eps = res.x[-1]
        if eps < 1e-12:
            # No improvement possible
            # maybe we are stuck; break
            break
        delta = res.x[:n]  # direction
        # line search
        # compute candidate step size
        # Ensure a+alpha*delta >=0 (should hold for any alpha <=1 due to bound)
        # choose alpha = 1 initially
        alpha_low = 0.0
        alpha_high = 1.0
        # We'll do backtracking: start with high = 1.0
        best_alpha = None
        for _ in range(10):
            a_new = a + alpha_high * delta
            # enforce non-negative just in case
            if np.any(a_new < -1e-12):
                alpha_high *= 0.5
                continue
            new_max = _max_convolution_fft(a_new)
            if new_max < M - eps * 0.8: # guarantee theoretical improvement? 
                best_alpha = alpha_high
                break
            else:
                alpha_high *= 0.5
                if alpha_high < 1e-6:
                    break
        if best_alpha is None:
            # no acceptable step, fallback small step
            best_alpha = min(0.5, alpha_high)
        a = a + best_alpha * delta
        # reproject to simplex? Not needed; sum stays constant by equality. However due to numeric errors, we may renorm:
        a = np.maximum(a, 0.0)
        total = a.sum()
        if abs(total - np.sqrt(2 * n)) > 1e-6:
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
    return best_seq

    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    iteration = 0

        iteration += 1
        # compute conv
        b = _full_convolution(a)
        M = b.max()
        # choose active set
        tol = M*1e-10 + 1e-12
        active_idxs = np.where(b >= M - tol)[0]
        if active_idxs.size == 0:
            break
        max_active = 30
        if active_idxs.size > max_active:
            # pick random subset to keep size
            active_idxs = rng.choice(active_idxs, size=max_active, replace=False)
        # create A_ub
        rows = []
        for k in active_idxs:
            coeff = np.zeros(n+1)
            i_low = max(0, k - (n-1))
            i_high = min(n-1, k)
            i_vals = np.arange(i_low, i_high+1)
            j_vals = k - i_vals
            coeff[j_vals] += a[i_vals]
            coeff[-1] = 1.0  # epsilon term
            rows.append(coeff)
        A_ub = np.vstack(rows)
        b_ub = np.zeros(A_ub.shape[0])

        # Equality sum(delta)=0
        A_eq = np.zeros((1, n+1))
        A_eq[0,:n] = 1.0
        b_eq = np.array([0.0])

        # bounds
        bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]

        # objective: minimize -epsilon -> c = [0...0, -1]
        c = np.zeros(n+1)
        c[-1] = -1.0

        # solve

                                     A_eq=A_eq, b_eq=b_eq,
                                     bounds=bounds, method='highs',

            print(f"[{time.time():.1f}] LP solve failed: {res.message}")

"""
Propose a step‑function that minimizes the maximal entry of its self‑convolution.

The optimizer works in two stages:

* **Linear‑programming (LP) direction steps** –  For the current sequence `a`,
  we identify the indices where the convolution attains its maximum.
  An LP is solved to find a feasible direction `Δa` that reduces all those
  maximal entries by the same amount `ε` (maximising `ε`).  The LP contains
  only the active constraints, a sum‑zero equality, a non‑negativity bound,
  and a modest upper bound `Δa_i ≤ 0.75·a_i`.  After the LP we perform a
  back‑tracking line search and update `a ← a + α·Δa`.  The sequence always
  stays on the ℓ¹‑simplex (sum = √(2 n)).

* **Exact hill‑climbing (simulated annealing)** –  After a number of LP
  updates a short hill‑climbing pass (the same routine used in the previous
  state‑of‑the‑art) removes any remaining local roughness.

The algorithm is applied to several support sizes (2000 … 32000) with a
wall‑clock budget of 1000 s.  The best sequence found is returned as a plain
list of floats; `evaluate_sequence` is called once at the end to report the
official score.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families for random initialisers
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)."""
    r = rng.uniform(0.96, 0.9995)
    idx = np.arange(n, dtype=float)
    return r**idx


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx**(-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential decay with a breakpoint."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _family_semicircle(n: int, rng) -> np.ndarray:
    """Semi‑circular shape a_i ∝ √(1‑x²) on [-1,1]."""
    x = np.linspace(-1.0, 1.0, n)
    a = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    return a


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a random family, add jitter and project onto the ℓ¹‑simplex."""
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp,
                _family_semicircle]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)          # tiny jitter
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation followed by ℓ¹‑simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# LP‑based direction optimisation
# ----------------------------------------------------------------------


def _lp_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Repeatedly compute a direction that reduces the current maximal
    convolution entry by solving a small linear programme.
    """
    a = a0.copy()
    n = a.size
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    # maximal size of the active set (keeps the LP tiny)
    max_active = 30

    while time.time() < deadline:
        # ----- current convolution and active indices -----
        b = _full_convolution(a)
        M = b.max()
        # tolerance to collect “active” peaks
        tol = max(1e-12, M * 1e-10)
        active = np.flatnonzero(b >= M - tol)
        if active.size == 0:
            break
        if active.size > max_active:
            active = rng.choice(active, size=max_active, replace=False)

        # ----- build LP: maximize ε (i.e. minimise –ε) -----
        # variables: Δa (n) followed by ε (1)
        rows = []
        for k in active:
            row = np.zeros(n + 1, dtype=float)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_vals = np.arange(i_low, i_high + 1)
            j_vals = k - i_vals
            # contribution a_i * Δa_{j}
            row[j_vals] += a[i_vals]
            row[-1] = 1.0                     # coefficient of ε
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.zeros(A_ub.shape[0])

        # Equality: sum(Δa) = 0  (keeps the ℓ¹‑norm unchanged)
        A_eq = np.zeros((1, n + 1), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # Bounds:   –a_i ≤ Δa_i ≤ 0.75·a_i   (keeps entries non‑negative)
        bounds = [(-a[i], a[i] * 0.75) for i in range(n)] + [(0.0, None)]

        # Objective: minimise –ε  <=> maximise ε
        c = np.zeros(n + 1, dtype=float)
        c[-1] = -1.0

        res = scipy.optimize.linprog(c,
                                     A_ub=A_ub, b_ub=b_ub,
                                     A_eq=A_eq, b_eq=b_eq,
                                     bounds=bounds,
                                     method='highs',
                                     options={'presolve': True})

        if not res.success:
            # LP failure – give up on this length
            print(f"[{time.time():.1f}] LP failed: {res.message}")
            break

        eps = res.x[-1]
        if eps <= 1e-12:
            # No meaningful improvement can be found
            break

        delta = res.x[:n]

        # ----- line search for a safe step size -----
        step = 1.0
        for _ in range(12):
            a_try = a + step * delta
            if np.min(a_try) < -1e-12:
                step *= 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            new_max = _max_convolution_fft(a_try)
            # The linear model predicts at least a reduction of eps;
            # we accept a modest reduction to stay safe.
            if new_max < M - 0.4 * eps:
                a = a_try
                break
            step *= 0.5
            if step < 1e-8:
                break
        else:
            break  # could not find a decreasing step

        # re‑project onto the simplex to correct accumulated round‑off
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP improvement: max={best_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Exact hill‑climbing (simulated annealing) – unchanged from the reference
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """Simulated‑annealing hill‑climb that updates the convolution in O(n)."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9
    cooling = 0.998

    while time.time() < deadline:
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution O(n) -----
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    b_best = _full_convolution(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock limit and returns the best
    sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Progressive support sizes – we spend a fraction of the budget on each
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        slice_time = min(remaining * 0.30, 120.0)   # at most two minutes per size
        slice_deadline = time.time() + slice_time

        # a few random restarts
        for restart in range(3):
            if time.time() >= slice_deadline:
                break
            a0 = _random_initial_sequence(n, rng)

            # ----- LP‑based directional optimisation -----
            a_opt = _lp_optimize(a0, target_sum, slice_deadline)
            val = _max_convolution_fft(a_opt)
            if val < best_score:
                best_score = val
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] LP restart {restart} n={n} score={val:.6f}")

            # ----- short hill‑climb refinement (if time permits) -----
            rem = slice_deadline - time.time()
            if rem > 5.0:
                b_cur = _full_convolution(a_opt)
                a_hc, b_hc = _hill_climbing(a_opt, b_cur,
                                            time.time() + min(rem, 20.0))
                val2 = float(b_hc.max())
                if val2 < best_score:
                    best_score = val2
                    best_seq = a_hc.copy()
                    print(f"[{time.time():.1f}] LP+HC n={n} score={val2:.6f}")

        # up‑sample the current best sequence for the next, larger size
        if best_seq is not None and best_seq.size != n:
            best_seq = _upsample_sequence(best_seq, n)

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        a_final = _lp_optimize(best_seq.copy(), target_sum, deadline - 3.0)

        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 2.0)
        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polish improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy return value
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never be needed)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (max‑convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:                                   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
