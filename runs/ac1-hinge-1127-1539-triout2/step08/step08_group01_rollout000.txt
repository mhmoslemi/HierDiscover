def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi et al. (2008) algorithm.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns gradient g and the convolution b.
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)   (⋆ denotes linear convolution)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed‑max objective
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser that tries to minimise the smooth max of the
    self‑convolution.  The algorithm runs until *deadline* (wall‑clock).
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning‑rate (scaled with the ℓ¹ radius)
    base_lr = 0.25 * target_sum

    # smoothing schedule for the log‑sum‑exp surrogate
    beta = 2.0               # start with a gentle smoothing
    beta_max = 1e6
    beta_factor = 2.0        # grow faster than the previous version

    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1

        # gradient of the smooth‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter (helps escape plateaus)
        if t % 150 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (make the surrogate sharper)
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled to provide diverse starting points.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        split = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(split)),
             np.exp(-lam2 * np.arange(split, n))]
        )

    elif family == 5:                   # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                               # random spikes (few large entries)
        a = np.zeros(n)
        spikes = rng.integers(1, n // 10 + 1)
        positions = rng.choice(n, size=spikes, replace=False)
        a[positions] = rng.random(spikes) + 0.5

    # add a tiny Gaussian jitter and project back to simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (biased version)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray, b0: np.ndarray, deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.  Each move transfers a
    small amount of mass from a source index (chosen among the indices
    that contribute most to the current worst convolution entry) to a
    destination index that currently contributes little.  The full
    self‑convolution b is maintained in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    # initial temperature roughly 10 % of the worst value
    T = cur_score * 0.1 + 1e-12
    cooling = 0.9999  # slow cooling for fine search

    while time.time() < deadline:
        # pick a convolution index that is currently maximal
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # Determine the range of (i,j) such that i+j = M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx

        # Contribution of each (i,j) pair to entry M
        contrib = a[idx] * a[jdx]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # Choose a destination index that is *not* heavily paired
        # with the counterpart a[M‑j].
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # delta ≤ 80 % of the source mass
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # --- update convolution efficiently (vectorised slices) ---
        # removal of mass from i
        b[i:i + n] -= 2.0 * delta * a
        # addition of mass to j
        b[j:j + n] += 2.0 * delta * a

        # self‑interaction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b.size:
            b[k2i] += delta * delta
        if k2j < b.size:
            b[k2j] += delta * delta
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta
        # ---------------------------------------------------------

        new_score = b.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a[i] -= delta
            a[j] += delta
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
        # else reject – a and b stay as they were (we already reverted
        # the changes in the copy b_new by not committing)

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    (default 1000 s) and returns the best sequence found as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Seed handling
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            # project onto the correct ℓ¹‑simplex for its length
            prev = _project_to_simplex(prev, math.sqrt(2.0 * n0))
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
                  f"score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    if best_seq is None:
        # create a fresh random seed of length 2000
        n0 = 2000
        best_seq = _random_initial_sequence(n0, rng)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Random seed (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – larger n gives more freedom → potentially lower max
    # ------------------------------------------------------------------
    schedule = [2000, 8000, 20000, 50000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise candidate for this length
        # --------------------------------------------------------------
        if best_seq.size == n:
            a_init = best_seq.copy()               # reuse the current best
        else:
            a_init = _upsample_sequence(best_seq, n)  # up‑sample if needed

        # --------------------------------------------------------------
        # Time allocation for this dimension
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        # give the bulk to the current (largest) dimension, keep a safety net
        gd_budget = remaining * 0.45
        hc_budget = remaining * 0.40
        # the rest (~15 %) will be left for later phases

        # --------------------------------------------------------------
        # Phase A – Adam gradient descent (multiple restarts)
        # --------------------------------------------------------------
        n_restarts = 6
        per_restart = gd_budget / n_restarts
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            restart_deadline = min(deadline - 0.1, time.time() + per_restart)

            # first restart uses the deterministic initialisation,
            # the others use fresh random seeds (to diversify search)
            if r == 0:
                start_a = a_init.copy()
            else:
                start_a = _random_initial_sequence(n, rng)

            a_opt = _adam_gradient_descent(start_a, target_sum, restart_deadline)
            score = _max_convolution_fft(a_opt)

            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        # accept if improvement over the global best
        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] GD improvement for n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – biased hill climbing (multiple restarts)
        # --------------------------------------------------------------
        if time.time() >= deadline - 5.0:
            break

        # ensure we work with a vector of the current length
        if best_seq.size != n:
            a_cur = _upsample_sequence(best_seq, n)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        # we may run a few short hill‑climb attempts; the budget is split
        n_hc_runs = 3
        hc_per_run = hc_budget / n_hc_runs
        best_hc_a = a_cur.copy()
        best_hc_score = b_cur.max()

        for r in range(n_hc_runs):
            if time.time() >= deadline:
                break
            run_deadline = min(deadline - 0.1, time.time() + hc_per_run)
            a_hc, b_hc = _hill_climb(a_cur, b_cur, run_deadline)
            score_hc = b_hc.max()
            if score_hc < best_hc_score:
                best_hc_score = score_hc
                best_hc_a = a_hc.copy()

        if best_hc_score < best_score - 1e-12:
            best_seq = best_hc_a.copy()
            best_score = best_hc_score
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Optional short post‑hill‑climb Adam refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            extra_deadline = deadline - 1.0
            a_refined = _adam_gradient_descent(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution_fft(a_refined)
            if score_ref < best_score - 1e-12:
                best_seq = a_refined.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC GD improvement for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the (currently) best length
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # long Adam run (dominant portion of remaining time)
        final_deadline = deadline - 2.0
        best_seq = _adam_gradient_descent(best_seq, target, final_deadline)

        # short hill‑climb to clean up any remaining spikes
        b_final = _full_convolution(best_seq)
        best_seq, b_final = _hill_climb(best_seq, b_final, deadline - 1.0)

        final_score = b_final.max()
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy output
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
