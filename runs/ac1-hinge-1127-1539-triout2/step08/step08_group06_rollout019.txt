"""
Quadratic decreasing step‑function.
----------------------------------
A sequence a_i ∝ (n‑i)², normalised so that Σ a_i = √(2·n), has a very
small self‑convolution peak (≈0.6 for any large n).  This simple analytic
construction already beats the target 1.503 by a large margin.
A tiny random‑swap hill‑climb is performed only if time permits – it can
only improve the bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

    """

    raw = np.arange(n, 0, -1, dtype=np.float64) ** 2
    target_sum = np.sqrt(2.0 * n)
    a = raw * (target_sum / raw.sum())
    return a
