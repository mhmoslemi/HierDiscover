"""
Enhanced adversarial step‑function optimizer.

Main ideas beyond the reference implementation
------------------------------------------------
* **Symmetric search space** – the optimal step‑function is symmetric,
  therefore we project every candidate onto the symmetric sub‑space.
  This halves the effective dimensionality and improves convergence,
  especially for long sequences (up to 50 000 entries).

* **ε‑subgradient of the hard max** – once the temperature τ becomes very
  small we switch from a soft‑max weighting to an ε‑subgradient that
  averages gradients over all convolution entries within a tiny band
  of the current maximum.  This yields a true subgradient of the
  non‑smooth objective and allows τ to approach zero.

* **More aggressive τ‑schedule** – τ starts at 5·max b and decays by
  0.995 per iteration until a floor of 1e‑6 is hit, after which the
  ε‑subgradient is used.

* **Richer parametric families** – a new “piecewise‑constant” family
  is added and the existing families are used more often.

* **Multi‑start Adam on several scaled lengths** – after the quick
  random exploration the best candidate is up‑scaled to a few larger
  lengths and, for each length, several independent Adam runs are
  performed (subject to the remaining time budget).

* **Symmetric hill‑climb** – the final stochastic local search also
  respects symmetry, moving equal amounts of mass at mirrored positions.

The function ``propose_candidate`` follows the same four‑phase
structure as before (exploration → scaling‑refinement → final Adam →
hill‑climb) but with the improvements listed above.  It stores the
best sequence in the global variable ``height_sequence_1`` and returns
it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, scaling, gradient, Adam)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _make_symmetric(seq: np.ndarray) -> np.ndarray:
    """Force symmetry a[i] = a[n-1-i] (mirror about centre)."""
    n = seq.shape[0]
    if n % 2 == 0:
        half = n // 2
        return np.concatenate([seq[:half], seq[:half][::-1]])
    else:
        half = (n + 1) // 2
        return np.concatenate([seq[:half], seq[:half - 1][::-1]])


def _subgrad_mask(b: np.ndarray, tau: float, eps: float) -> np.ndarray:
    """Mask used for the (soft) max‑gradient."""
    max_b = b.max()
    if tau > 0.0:
        # classic τ‑soft‑max
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)
    else:
        # ε‑subgradient: average over all entries close to the max
        mask = (b >= max_b - eps).astype(float)
        if mask.sum() == 0.0:               # safety fallback
            mask = (b == max_b).astype(float)
        mask /= mask.sum()
    return mask, max_b


def _max_convolution_gradient(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k mask_k · (a*a)_k   w.r.t. a.
    Returns a vector of length n.
    """
    a_rev = a[::-1]
    conv_len = mask.shape[0] + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the symmetric simplex (log‑softmax).
    Uses an aggressive τ‑schedule and falls back to an ε‑subgradient
    once τ is very small.
    Returns the best sequence encountered and its hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits from a feasible a0 (already symmetric)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                           # relatively high start

    # τ‑schedule
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max() * 5.0, 1.0)      # start with generous smoothing
    tau_min = 1e-6
    tau_decay = 0.995

    best_a = a0.copy()
    best_peak = b0.max()
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----- recover a from logits, enforce symmetry, renormalise -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # project onto the symmetric sub‑space
        a = _make_symmetric(a)
        if a.sum() == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / a.sum())

        # ----- convolution & mask -----
        b = _full_convolution_fft(a)
        # ε for sub‑gradient, relative to current max
        eps = max(1e-9, b.max() * 1e-4)
        mask, peak = _subgrad_mask(b, tau, eps)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. a -----
        g = _max_convolution_gradient(a, mask)

        # ----- chain rule to logits -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau = max(tau * tau_decay, tau_min)
        lr *= 0.9999                     # gentle decay

        # occasional jitter to avoid plateaus
        if iteration % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_piecewise(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑constant shape with 2‑5 segments."""
    segs = rng.integers(2, 6)
    # Choose breakpoints (sorted, unique)
    cuts = np.sort(rng.choice(np.arange(1, n), size=segs - 1, replace=False))
    heights = rng.uniform(0.1, 5.0, size=segs)
    seq = np.empty(n)
    start = 0
    for cut, h in zip(np.append(cuts, n), heights):
        seq[start:cut] = h
        start = cut
    return seq


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Uses the global best as a warm‑start with probability ~0.2,
    otherwise draws from a random parametric family.
    The resulting vector is forced to be symmetric and ℓ₁‑normalised
    to √(2 n).
    """
    S = np.sqrt(2.0 * n)

    # ---- warm‑start chance ----
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.full(n, 1.0)
        a = a * (S / a.sum())
        a = _make_symmetric(a)
        a = a * (S / a.sum())
        return a

    # ---- pick a random family ----
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape',
                        'piecewise'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'piecewise':
        raw = _init_piecewise(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)

    # enforce symmetry
    a = _make_symmetric(a)
    a = a * (S / a.sum())
    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm proceeds through the following phases:

    1. **Fast random exploration** (≈60 % of the budget) – many candidates
       drawn from diverse parametric families; keep the best by the exact
       max‑convolution value.
    2. **Scaling & multi‑restart Adam refinement** – the current best
       candidate is up‑scaled to several larger lengths (up to ~45 k)
       and for each length several independent Adam runs are performed.
    3. **Final long Adam optimisation** on the best length found so far.
    4. **Symmetric stochastic hill‑climb** – cheap random mass transfers
       respecting symmetry, kept if they improve the hard max.
    5. **Final sanity check** via the provided ``evaluate_sequence``.
    The final construction is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0  # 5 s safety margin

    # ------------------------------------------------------------------
    # Warm‑start from any previously found sequence
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈60 % of total budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + budget_s * 0.60
    while time.time() < explore_deadline:
        attempt += 1
        # bias length towards a few thousand entries, but keep a tail up to 30 k
        if rng.random() < 0.75:
            n = int(rng.integers(2000, 8000))
        else:
            n = int(rng.integers(8000, 30000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & multi‑restart Adam refinement on several lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of moderate size
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # lengths to test – grow geometrically to a safe bound (~45 k)
    lengths = [best_n]
    for factor in [1.3, 1.6, 2.0, 3.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 45000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 35.0:   # reserve time for later phases
            break

        # allocate a slice of the remaining budget for this length
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.18, 150.0)   # at most 150 s per length

        # up‑scale the current best candidate
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = _make_symmetric(a_scaled)
        a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

        # number of independent Adam restarts we can afford for this length
        # (we keep at least one iteration of the loop to guarantee progress)
        restarts = max(1, int(slice_time // 30.0))

        for r in range(restarts):
            run_deadline = time.time() + (slice_time / restarts) - 0.5
            a_opt, peak = _adam_optimize(a_scaled, run_deadline, rng)

            if peak < best_score:
                best_score = peak
                best_seq = a_opt.copy()
                best_n = target_n
                print(f"[refine] length={target_n:5d} run={r+1}/{restarts}  improved={best_score:.6f}")

        # after the restarts for this length, shrink the candidate a_scaled back
        # to the (potentially) improved best_seq for the next scaling step
        a_scaled = best_seq.copy()

    # ------------------------------------------------------------------
    # Phase 3 – final long Adam optimisation on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 12.0:
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq, time.time() + remaining, rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final refine] length={best_n:5d}  best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – symmetric stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    S = np.sqrt(2.0 * best_seq.shape[0])
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]

        # pick two distinct positions (and move their mirrors as well)
        i, j = rng.integers(0, n // 2 + n % 2, size=2)
        if i == j:
            continue
        # ensure the source still has mass
        if a[i] <= 0.0:
            continue
        # move a fraction of mass from i (and its mirror) to j (and its mirror)
        frac = rng.uniform(0.02, 0.13)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount

        # also affect the mirrored indices
        mi = n - 1 - i
        mj = n - 1 - j
        if mi != i:
            a[mi] -= amount
            a[mj] += amount

        # re‑symmetrise and renormalise
        a = _make_symmetric(a)
        a = a * (S / a.sum())

        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[local] improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
