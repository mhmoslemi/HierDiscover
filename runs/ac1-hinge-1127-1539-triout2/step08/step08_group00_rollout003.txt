"""
Hybrid search for a non‑negative step function that minimises the maximal
self‑convolution entry (equivalently the value returned by
``evaluate_sequence`` after ℓ¹‑normalisation).

The algorithm consists of three stages that run repeatedly for several
support sizes increasing from a few thousand up to ~80 000:

1. **Randomised families** – generate many candidates from a small pool
   of parametric families (exponential, power‑law, linear, piece‑wise
   exponential, Gaussian and double‑Gaussian).  All candidates are
   projected onto the ℓ¹‑simplex of radius √(2 n) and the best one is
   kept.

2. **Smoothed‑max Adam descent** – minimise a smooth approximation of
   ``max(a⋆a)`` using the log‑sum‑exp (soft‑max) surrogate.  The gradient
   is obtained analytically via FFT, Adam adapts the step‑size and a
   geometric schedule ramps the smoothing parameter β up to 5·10⁵.
   After each step the iterate is projected back onto the simplex.

3. **Exact hill‑climbing (simulated annealing)** – a move transfers a
   random amount of mass from a “source” index to a “target” index while
   updating the full convolution in O(n).  Moves that lower the current
   maximum are always accepted; others are accepted with a Metropolis
   probability.  The temperature is slowly cooled (``cooling=0.999``).

After iterating over the schedule the best sequence is polished with a
final Adam run followed by a short hill‑climb.  The resulting list is
returned; its maximal self‑convolution entry equals the value that
``evaluate_sequence`` will report.

The routine respects a wall‑clock budget (default 1000 s) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log Σ exp(β·b_k)  where
    b = a⋆a (full convolution).  Returns (grad, b) with

        grad_i = 2·∑_k p_k·a_{k-i}
    where p_k ∝ exp(β·b_k) are soft‑max probabilities.
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    # subtract max for numerical stability
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a ⋆ p  (gradient factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 5e5,
                   beta_factor: float = 1.3) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max objective.
    The smoothing parameter β grows geometrically until ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.25
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if step % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # β schedule
        if beta < beta_target and step % 250 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)"""
    r = rng.uniform(0.96, 0.9995)
    idx = np.arange(n, dtype=float)
    return r ** idx


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}"""
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential decay with a breakpoint."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _family_gaussian(n: int, rng) -> np.ndarray:
    """Bell‑shaped Gaussian centred roughly in the middle."""
    mu = rng.uniform(0.3, 0.7) * (n - 1)
    sigma = rng.uniform(0.1, 0.4) * n
    idx = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    return a


def _family_double_gaussian(n: int, rng) -> np.ndarray:
    """Sum of two Gaussians with random weights."""
    w = rng.uniform(0.3, 0.7)
    a1 = _family_gaussian(n, rng) * w
    a2 = _family_gaussian(n, rng) * (1 - w)
    return a1 + a2


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Pick a random parametric family, add a tiny jitter and project onto the ℓ¹‑simplex.
    """
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp,
                _family_gaussian,
                _family_double_gaussian]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)     # avoid exact ties
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a source index to a target index while
    updating the convolution in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.25 + 1e-12      # initial temperature
    cooling = 0.999                # slower cooling for deeper search

    while time.time() < deadline:
        # pick one of the indices attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – pick among contributors to this entry weighted by product size
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias towards positions with small current mass
        if rng.random() < 0.4:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        # remove contribution from source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # add contribution from target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correct cross term at i + j (appears twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # final accurate convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock limit and returns the best sequence
    discovered as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seed from a previously found construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            score0 = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = score0
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) score={score0:.6f}")
        except Exception as e:
            print(f"Unable to seed from height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Length schedule – gradually increase the support size
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000, 50000, 80000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1) Random families (cheap exploration)
        # --------------------------------------------------------------
        n_samples = 30
        for _ in range(n_samples):
            if time.time() >= deadline:
                break
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] random family improvement (n={n}) score={score:.6f}")

        # --------------------------------------------------------------
        # 2) Adam‑based smoothed‑max descent
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        gd_time = min(remaining * 0.45, 250.0)
        gd_deadline = time.time() + gd_time

        # initialise Adam from the current best (upsampled if necessary)
        if best_seq is not None and best_seq.size != n:
            a_start = _upsample_sequence(best_seq, n)
        else:
            a_start = _random_initial_sequence(n, rng)

        a_opt = _adam_optimize(a_start, gd_deadline,
                               beta_start=5.0,
                               beta_target=5e5,
                               beta_factor=1.2)
        score_opt = _max_convolution_fft(a_opt)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Adam descent improvement (n={n}) score={score_opt:.6f}")

        # --------------------------------------------------------------
        # 3) Exact hill‑climbing (simulated annealing)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        hc_time = min(remaining * 0.35, 200.0)
        hc_deadline = time.time() + hc_time

        # ensure we work with a vector of the correct length
        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = float(b_hc.max())
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb improvement (n={n}) score={score_hc:.6f}")

        # --------------------------------------------------------------
        # 4) Optional short Adam follow‑up (quick polishing)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 8.0:
            extra_deadline = time.time() + min(remaining * 0.15, 80.0)
            a_extra = _adam_optimize(best_seq.copy(),
                                      extra_deadline,
                                      beta_start=5.0,
                                      beta_target=1e6,
                                      beta_factor=1.25)
            score_extra = _max_convolution_fft(a_extra)
            if score_extra < best_score:
                best_score = score_extra
                best_seq = a_extra.copy()
                print(f"[{time.time():.1f}] extra Adam polishing (n={n}) score={score_extra:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the best length found
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final Adam run
        final_deadline = time.time() + (deadline - time.time()) * 0.5
        a_final = _adam_optimize(best_seq.copy(),
                                  final_deadline,
                                  beta_start=8.0,
                                  beta_target=2e6,
                                  beta_factor=1.25)

        # short hill‑climb to smooth remaining rough edges
        b_final = _full_convolution(a_final)
        hc_final_deadline = time.time() + (deadline - time.time()) * 0.3
        a_final, b_final = _hill_climbing(a_final, b_final, hc_final_deadline)

        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy return value
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (max‑convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
