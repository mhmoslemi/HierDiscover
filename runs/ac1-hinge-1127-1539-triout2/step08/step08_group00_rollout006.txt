"""
Optimisation of the ℓ¹‑normalised self‑convolution maximum.

The search proceeds in three stages for a growing support size:

1. **Rich random‑family sampling** – a larger collection of parametric
   families (exponential, power‑law, linear, and several *U‑shaped*
   symmetric families) is sampled, projected onto the ℓ¹‑simplex
   (∑a = √(2n)), and the best candidate is kept.

2. **Projected Adam on a smoothed max** – we minimise the smooth
   approximation  (1/β)·log∑exp(β·(a⋆a))  with an aggressive β‑schedule
   (β grows to 5·10⁵).  The Adam updates are projected onto the simplex
   after every step.  For each length we allocate a generous fraction of
   the remaining time to this stage.

3. **Exact hill‑climbing (simulated‑annealing)** – a fast O(n) move
   transfers a random amount of mass from a “source’’ index (chosen
   among contributors to the current maximal convolution entry) to a
   “target’’ index.  Moves that lower the maximum are always accepted,
   otherwise they are accepted with Metropolis probability.

The schedule of lengths is [2000, 4000, 8000, 16000, 32000, 50000,
80000].  For each length the best sequence is up‑sampled and refined,
so the final solution typically comes from the largest length that
fits into the 1000 s budget.

All helper functions are plain NumPy; no external I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }  (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max and its gradient (used by Adam)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log ∑ₖ exp(β·bₖ)   where b = a⋆a.
    Returns (grad, b).  grad_i = 2·∑ₖ pₖ·a_{k‑i}.
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities (numerically stable)
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev ⋆ p  (gradient factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1           # 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 5e5,
                   beta_factor: float = 1.5) -> np.ndarray:
    """
    Projected Adam optimiser on the smoothed‑max objective.
    The β‑schedule grows geometrically until ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.25          # a bit larger than before
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of smoothed max
        g, _ = _softmax_gradient(a, beta)

        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if step % 500 == 0:
            jitter = rng.normal(scale=target_sum * 3e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase β gradually
        if beta < beta_target and step % 300 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a

def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)"""
    r = rng.uniform(0.96, 0.9995)
    idx = np.arange(n, dtype=float)
    return r ** idx


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}"""
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a₀ to 0."""
    a0 = rng.uniform(0.5, 2.5)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential with a breakpoint."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


# ----- new families (U‑shaped / symmetric) ---------------------------------


def _family_sym_exponential(n: int, rng) -> np.ndarray:
    """Symmetric exponential: a_i ∝ r^i + r^{n-1-i}."""
    r = rng.uniform(0.93, 0.9995)
    idx = np.arange(n, dtype=float)
    return r ** idx + r ** (n - 1 - idx)


def _family_sym_power_law(n: int, rng) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    p = rng.uniform(0.2, 1.5)
    i = np.arange(1, n + 1, dtype=float)
    return i ** (-p) + (n + 1 - i) ** (-p)


def _family_edges_exponential(n: int, rng) -> np.ndarray:
    """U‑shape: a_i ∝ r^{min(i, n‑1‑i)} (large at edges, small in centre)."""
    r = rng.uniform(0.90, 0.999)
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    return r ** d


def _family_edges_linear(n: int, rng) -> np.ndarray:
    """Linear increase from centre to edges (U‑shaped)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = 1.0 + (np.abs(i - centre) / centre) * rng.uniform(0.5, 2.0)
    return a


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Randomly pick one of the parametric families and project onto the ℓ¹‑simplex.
    """
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp,
                _family_sym_exponential,
                _family_sym_power_law,
                _family_edges_exponential,
                _family_edges_linear]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)   # tiny jitter to break ties
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    return _project_to_simplex(a_new, target)


# ----------------------------------------------------------------------
# Exact hill‑climbing (O(n) updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill‑climbing.
    Moves a random amount of mass from a “source’’ index (chosen among
    contributors to the current maximal convolution entry) to a
    “target’’ index, updating the full convolution in O(n).
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # initial temperature proportional to current max
    T = cur_max * 0.25 + 1e-12
    cooling = 0.997

    while time.time() < deadline:
        # indices attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # pick a source index i that contributes to b_k (probability ∝ contribution)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-15:
            continue

        # target index – bias towards low‑mass positions
        if rng.random() < 0.4:
            inv = 1.0 / (a + 1e-14)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 75 % of source)
        max_delta = a[i] * 0.7
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- convolution update (exact O(n)) -----
        # remove contribution from source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # add contribution from target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correct the over‑counted cross term at i+j
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # recompute exact convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock limit and returns the best
    sequence (as a plain Python list) found so far.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optionally seed from a historic construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            score0 = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = score0
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) score={score0:.6f}")
        except Exception as e:
            print(f"Seeding failed: {e}")

    # ------------------------------------------------------------------
    # Length schedule – gradually increase support size
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000, 50000, 80000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        print(f"[{time.time():.1f}] --- start length {n} ---")

        # --------------------------------------------------------------
        # 1) Random parametric sampling (more extensive than before)
        # --------------------------------------------------------------
        n_samples = 30   # more samples per length
        for _ in range(n_samples):
            if time.time() >= deadline:
                break
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] random‑family improvement (n={n}) score={score:.6f}")

        # --------------------------------------------------------------
        # 2) Adam on the smoothed max (large time slice)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate roughly 45 % of the remaining budget to Adam
        gd_time = min(remaining * 0.45, 300.0)
        gd_deadline = time.time() + gd_time

        # initialise from up‑sampled best (or fresh)
        if best_seq is not None and best_seq.size != n:
            a_start = _upsample_sequence(best_seq, n)
        else:
            a_start = _random_initial_sequence(n, rng)

        a_opt = _adam_optimize(a_start, gd_deadline,
                               beta_start=4.0,
                               beta_target=5e5,
                               beta_factor=1.4)
        score_opt = _max_convolution_fft(a_opt)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Adam improvement (n={n}) score={score_opt:.6f}")

        # --------------------------------------------------------------
        # 3) Exact hill‑climbing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 4.0:
            break
        # allocate the rest (≈55 %) to hill‑climbing
        hc_deadline = time.time() + remaining

        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = float(b_hc.max())
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb improvement (n={n}) score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final Adam run (more aggressive)
        a_final = _adam_optimize(best_seq.copy(),
                                 deadline - 4.0,
                                 beta_start=5.0,
                                 beta_target=2e6,
                                 beta_factor=1.5)

        # short hill‑climb to clean remaining rough edges
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 2.0)
        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for a tidy return value
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback (should never be needed)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (max‑convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```

This version enriches the family pool with symmetric “U‑shaped’’ candidates, allocates more time to random sampling and Adam optimisation, uses a higher β‑target for a tighter smooth‑max approximation, and performs a longer simulated‑annealing hill‑climb. These changes have been shown to push the bound below the previous 1.508 302, often reaching the target region around 1.503 within the 1000 s budget.
