    best = a.copy()
    best_peak = _max_convolution(a)
    n = len(a)
    while time.time() < deadline:

        max_b = b.max()

        # compute gradient

"""
Symmetry‑aware multi‑scale Adam optimiser with final sub‑gradient balancing.

The algorithm:
1. Randomly sample a set of candidate step‑functions from several
   parametric families (Dirichlet‑like, Gamma, power‑law,
   exponential, sparse bumps, Gaussian).  Each candidate is made
   symmetric (a = (a + a[::-1]) / 2) and normalised so that
   Σ a = √(2 n); for a symmetric vector the evaluation score equals
   max(a * a).

2. The best candidates are refined at increasing resolutions.
   Refinement uses Adam on the simplex (log‑softmax
   parametrisation) with a τ‑soft‑max approximation of the peak.
   After each Adam update the vector is symmetrised again.
   Several hyper‑parameter settings are tried and the best intermediate
   vector is kept.

3. When the time budget is almost exhausted a cheap
   sub‑gradient “mass‑transfer” walk is performed.  The exact
   gradient of the current peak (`τ = 0`) is used to move a small
   amount of mass from indices that increase the peak to those that
   decrease it, while keeping the vector symmetric and the ℓ₁‑norm
   fixed.

The final construction is stored in the global variable
`height_sequence_1` and returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of τ‑soft‑max (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500,
                   enforce_symmetry: bool = True) -> tuple[np.ndarray, float]:
    """
    Minimise the exact max‑convolution of ``a`` under the ℓ₁‑norm
    constraint using a soft‑max approximation.
    Returns the best vector seen and its *hard* peak (max convolution).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0  (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)            # for numerical stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- enforce optional symmetry -------
        if enforce_symmetry:
            a = (a + a[::-1]) * 0.5

        # ------- convolution & gradient (soft max) -------
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # how much background
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))    # low background
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Discrete Gaussian (possibly mixed with a uniform background)."""
    idx = np.arange(n)
    mu = rng.uniform(0, n - 1)
    sigma = rng.uniform(0.1 * n, 0.4 * n)
    raw = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    mix = rng.uniform(0.0, 0.6)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * raw
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ~0.22 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    The resulting vector is made symmetric and normalised to
    Σa = √(2 n).
    """
    S = np.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        a = (a + a[::-1]) * 0.5
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    else:  # gaussian
        raw = _init_gaussian(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)

    # enforce symmetry
    a = (a + a[::-1]) * 0.5
    return a


# ----------------------------------------------------------------------
# Sub‑gradient mass‑transfer walk (exact peak, τ = 0)
# ----------------------------------------------------------------------
def _balance_subgradient(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         init_step_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Cheap deterministic walk that transfers a small amount of mass
    from indices that increase the current peak to those that decrease it.
    The vector is kept symmetric and normalised after each move.
    Returns the best vector encountered and its peak value.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best = a.copy()
    best_peak = _max_convolution(a)

    step_frac = init_step_frac

    while time.time() < deadline:
        # exact peak and mask
        b = _full_convolution_fft(a)
        max_b = b.max()
        mask = (b == max_b).astype(float)

        # gradient of the exact peak (τ = 0)
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        grad = 2.0 * conv[n - 1:n - 1 + n]  # shape (n,)

        # choose source (largest gradient) and target (smallest gradient)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        if grad[src] <= grad[dst] + 1e-12:
            # cannot improve further
            break

        delta = step_frac * a[src]
        if delta < 1e-12:
            break

        # mass transfer
        a[src] -= delta
        a[dst] += delta

        # enforce symmetry and normalisation
        a = (a + a[::-1]) * 0.5
        a = a * (S / a.sum())

        cur_peak = _max_convolution(a)
        if cur_peak < best_peak:
            best_peak = cur_peak
            best = a.copy()

        # slowly decay step size
        step_frac *= 0.9999

    return best, best_peak

"""
Symmetry‑aware multi‑scale Adam optimiser with final sub‑gradient balancing.

The algorithm:
1. Randomly sample a set of candidate step‑functions from several
   parametric families (Dirichlet‑like, Gamma, power‑law,
   exponential, sparse bumps, Gaussian).  Each candidate is made
   symmetric (a = (a + a[::-1]) / 2) and normalised so that
   Σ a = √(2 n); for a symmetric vector the evaluation score equals
   max(a * a).

2. The best candidates are refined at increasing resolutions.
   Refinement uses Adam on the simplex (log‑softmax
   parametrisation) with a τ‑soft‑max approximation of the peak.
   After each Adam update the vector is symmetrised again.
   Several hyper‑parameter settings are tried and the best intermediate
   vector is kept.

3. When the time budget is almost exhausted a cheap
   sub‑gradient “mass‑transfer” walk is performed.  The exact
   gradient of the current peak (`τ = 0`) is used to move a small
   amount of mass from indices that increase the peak to those that
   decrease it, while keeping the vector symmetric and the ℓ₁‑norm
   fixed.

The final construction is stored in the global variable
`height_sequence_1` and also returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of τ‑soft‑max (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500,
                   enforce_symmetry: bool = True) -> tuple[np.ndarray, float]:
    """
    Minimise the exact max‑convolution of ``a`` under the ℓ₁‑norm
    constraint using a soft‑max approximation.
    Returns the best vector seen and its *hard* peak (max convolution).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0  (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)            # stabilise
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- enforce symmetry -------
        if enforce_symmetry:
            a = (a + a[::-1]) * 0.5

        # ------- convolution & gradient (soft max) -------
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits -------
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law mixed with background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)          # shift
    mix = rng.uniform(0.0, 1.0)              # background weight
    base = np.ones(n)
    return mix * base + (1.0 - mix) * (idx + offset) ** (-p)


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential mixed with background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * np.exp(-lam * idx)


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few rectangular bumps on a low background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Discrete Gaussian (optionally mixed with a uniform background)."""
    idx = np.arange(n)
    mu = rng.uniform(0, n - 1)
    sigma = rng.uniform(0.1 * n, 0.4 * n)
    raw = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    mix = rng.uniform(0.0, 0.6)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best, otherwise draws from
    a parametric family.  The result is symmetric and normalised to
    Σa = √(2 n).
    """
    S = np.sqrt(2.0 * n)

    # Warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        a = (a + a[::-1]) * 0.5
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    else:  # gaussian
        raw = _init_gaussian(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)

    # enforce symmetry
    a = (a + a[::-1]) * 0.5
    return a


# ----------------------------------------------------------------------
# Sub‑gradient mass‑transfer walk (exact peak, τ = 0)
# ----------------------------------------------------------------------
def _balance_subgradient(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         init_step_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Transfer a small amount of mass from indices that increase the current
    peak to those that decrease it, keeping the vector symmetric and
    ℓ₁‑normalised.  Returns the best vector encountered and its peak.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best = a.copy()
    best_peak = _max_convolution(a)
    step_frac = init_step_frac

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        max_b = b.max()
        mask = (b == max_b).astype(float)

        # exact gradient of the peak (τ = 0)
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        grad = 2.0 * conv[n - 1:n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        if grad[src] <= grad[dst] + 1e-12:
            break

        delta = step_frac * a[src]
        if delta < 1e-12:
            break

        a[src] -= delta
        a[dst] += delta

        # enforce symmetry and normalisation
        a = (a + a[::-1]) * 0.5
        a = a * (S / a.sum())

        cur_peak = _max_convolution(a)
        if cur_peak < best_peak:
            best_peak = cur_peak
            best = a.copy()

        step_frac *= 0.9999

    return best, best_peak


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm combines random exploration, multi‑scale Adam optimisation
    (with symmetry enforcement), and a final exact sub‑gradient balancing
    walk.  The best sequence is stored in ``height_sequence_1`` and also
    returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin for final check

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if present
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – random exploration (≈ 35 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + budget_s * 0.35
    top_k = 8
    top_candidates = []          # (peak, sequence)
    best_seq = None
    best_peak = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # choose length – favour a few‑thousand entries
        if rng.random() < 0.6:
            n = int(rng.integers(1800, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_peak:
            best_peak = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_peak:.6f}")

        # maintain a sorted list of the best candidates
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Fallback if nothing was found (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_peak = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_peak, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement (≈ 55 % of budget)
    # ------------------------------------------------------------------
    max_len = min(30000, best_n * 6)
    lengths = [best_n]
    if best_n * 1.3 < max_len:
        lengths.append(int(best_n * 1.3))
    if best_n * 1.8 < max_len:
        lengths.append(int(best_n * 1.8))
    if best_n * 2.5 < max_len:
        lengths.append(int(best_n * 2.5))
    lengths.append(int(max_len))
    lengths = sorted(set(lengths))

    for init_idx, (init_peak, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_n = len(cur_seq)

        for target_n in lengths:
            if time.time() >= deadline - 30.0:
                break

            # interpolate and normalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            S = np.sqrt(2.0 * target_n)
            a_scaled = a_scaled * (S / a_scaled.sum())
            a_scaled = (a_scaled + a_scaled[::-1]) * 0.5   # symmetry

            remaining = deadline - time.time()
            slice_budget = min(remaining * 0.45, 180.0)

            # coarse Adam run
            lr0 = rng.choice([0.42, 0.48, 0.55])
            tau_decay = rng.choice([0.9993, 0.9995, 0.9997])
            refined, peak = _adam_optimize(
                a_scaled,
                time.time() + slice_budget,
                rng,
                lr0=lr0,
                tau_decay=tau_decay,
                lr_decay=0.99985,
                jitter_every=400,
                enforce_symmetry=True,
            )

            # optional fine‑tune run if time permits
            if time.time() + 10.0 < deadline:
                refined, peak2 = _adam_optimize(
                    refined,
                    time.time() + min(30.0, (deadline - time.time()) * 0.15),
                    rng,
                    lr0=0.30,
                    tau_decay=0.9992,
                    lr_decay=0.9999,
                    jitter_every=600,
                    enforce_symmetry=True,
                )
                if peak2 < peak:
                    peak = peak2

            if peak < best_peak:
                best_peak = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_peak:.6f}")

            cur_seq = refined
            cur_n = target_n

        if time.time() >= deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient mass‑transfer balancing (≈ 5 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        final_seq, final_peak = _balance_subgradient(
            best_seq,
            time.time() + remaining,
            rng,
            init_step_frac=0.04,
        )
        if final_peak < best_peak:
            best_peak = final_peak
            best_seq = final_seq.copy()
            print(f"[balance] final balanced score={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Final check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_peak:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
```

This implementation expands on the previous approach by:

* **Enforcing symmetry** at every stage – a property that tends to lower the convolution peak.
* **Adding a Gaussian family** to the random search for richer initial shapes.
* **Running a secondary fine‑tuning Adam pass** on each scale.
* **Applying a deterministic sub‑gradient mass‑transfer walk** after Adam refinements to push the peak down further.

All helper functions are defined at the top level, no file I/O is performed, and the algorithm respects the 1000 s time budget while aiming to achieve a score below the target 1.503.
