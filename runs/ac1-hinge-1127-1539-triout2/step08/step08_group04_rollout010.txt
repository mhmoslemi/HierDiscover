    elif budget_s >= 400: n=8000
    else: n=5000
    target_sum = np.sqrt(2.0 * n)

        init = _resize_sequence(np.array(globals()['height_sequence_1'], dtype=np.float64), n)
        init = init * (target_sum / init.sum())

    it = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        # decide move
        if rng.random() < 0.7:
            # mass transfer as before but try larger fraction
            k_max = int(np.argmax(conv))
            # calculate contributions
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            sens = a[k_max - idx]  # contributions weight factor for each a[idx]
            weight = a[idx] * sens
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=weight / total_w)
            else:
                donor = rng.integers(n)
            # receiver: choose low contribution index
            low_thr = np.percentile(weight, 10)
            low_idxs = idx[weight <= low_thr]

                receiver = rng.choice(n)

                max_frac = 0.05
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
        else:
            # simulated annealing perturbation
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:

                a[i] -= delta
                a[j] += delta

A multi‑stage optimizer for the step‑function benchmark.

Stage 1 – cheap parametric warm‑start on a reduced problem.
Stage 2 – Adam‑style mirror‑descent on a soft‑max upper bound of the
          maximal convolution entry.
Stage 3 – sub‑gradient descent that directly targets the current maximal
          convolution point.
Stage 4 – deterministic mass‑transfer together with a simple simulated
          annealing scheme that moves a small amount of mass between
          indices, accepting occasional deteriorations while the temperature
          slowly cools.

All updates keep the sequence non‑negative and project onto the ℓ¹‑simplex
∑a = √(2 n).  The routine respects the supplied time budget and returns the
best sequence found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT implementation)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)² (used during optimisation)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _score_from_conv(conv: np.ndarray, target_sum: float, n: int) -> float:
    """Score computed from a pre‑computed convolution."""
    return float(2.0 * n * np.max(conv) / (target_sum ** 2))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to `new_len` entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence a_i = (i+1)^(-p) normalised to `target_sum`."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns (grad, full_convolution) where grad has the same length as `a`.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 1 – parametric warm‑start on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Grid + random refinement search of the 2‑parameter family
        a_i = (i+1)^(-p) * exp(-λ·i / small_n)
    """
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – Adam‑style mirror descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_mirror_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponential‑mirror descent with Adam‑like moment estimates.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eta = 0.06
    eps = 1e-12

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    beta = 1.0
    max_beta = 16384.0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam updates on the *log* of a (mirror‑descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        update = np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        a = a * update
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – sub‑gradient descent on the true maximum entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Direct sub‑gradient descent on max(a*a).  At each step the index of the
    current maximal convolution entry is identified and the corresponding
    sub‑gradient 2·a_{k‑i} is applied.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    avg = target_sum / n
    base_step = avg * 0.25  # 25 % of the average coefficient

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k_max - idx]

        step = base_step / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 2e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 4 – deterministic mass‑transfer + simulated annealing
# ----------------------------------------------------------------------


def _mass_transfer_and_anneal(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass from indices that
    contribute most to the current maximal convolution entry to indices
    that contribute least, with a Simulated‑Annealing acceptance rule.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    init_temp = best_val * 0.02
    temp = init_temp
    it = 0

    while time.time() < deadline:
        # keep a copy of the current state (for possible rejection)
        a_prev = a.copy()

        # current convolution and its maximal entry
        conv = _conv_fft(a)
        cur_val = _score_from_conv(conv, target_sum, n)
        k_max = int(np.argmax(conv))

        # ---- decide which elementary move to perform ----
        if rng.random() < 0.70:          # mass‑transfer focused move
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx_range = np.arange(lo, hi + 1, dtype=np.int64)

            # contribution weights for the maximal entry
            sens = a[k_max - idx_range]                 # a_{k‑i}
            weight = a[idx_range] * sens                # a_i * a_{k‑i}
            total_w = weight.sum()
            if total_w > 0.0:
                donor = rng.choice(idx_range, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # pick a low‑contributing receiver
            low_thr = np.percentile(weight, 10)
            low_idxs = idx_range[weight <= low_thr]
            if low_idxs.size:
                receiver = rng.choice(low_idxs)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                max_frac = 0.05
                delta = a[donor] * rng.random() * max_frac
                a[donor] -= delta
                a[receiver] += delta
        else:                           # pure simulated‑annealing perturbation
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_frac = 0.04
            delta = a[i] * rng.random() * max_frac
            a[i] -= delta
            a[j] += delta

        # enforce feasibility
        a = np.maximum(a, 0.0)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            a = a * (target_sum / cur_sum)

        # evaluate the new state (reuse full convolution)
        conv_new = _conv_fft(a)
        new_val = _score_from_conv(conv_new, target_sum, n)

        # ---- acceptance rule ----
        accepted = False
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
            accepted = True
            print(f"[{time.time():.2f}] Anneal improve score={best_val:.6f}")
        else:
            # Metropolis criterion
            prob = np.exp(-(new_val - best_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accepted = True  # move to a worse state
            else:
                a = a_prev      # revert
                # conv stays as conv (the previous one); no need to recompute

        # temperature schedule (slow exponential cooling)
        it += 1
        temp = init_temp * (0.995 ** it)

        # occasional jitter to keep the dynamics alive
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 2e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial sequence (fallback)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – parametric warm‑start on a reduced problem
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(80.0,
                                      (deadline - time.time()) * 0.10)
    best_small_seq, _ = _grid_param_search(small_n, small_target,
                                           param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑start from parametric shape (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – Adam‑style mirror descent (≈25 % of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        ad_deadline = time.time() + (deadline - time.time()) * 0.25
        a_ad, val_ad = _adam_mirror_descent(init, target_sum, ad_deadline)
        best_seq = a_ad
        best_score = val_ad
        print(f"[{time.time():.2f}] After Adam best score={best_score:.6f}")
    else:
        best_seq = init
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient descent on the true maximum (≈20 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.20
        a_sub, val_sub = _subgradient_descent(best_seq,
                                               target_sum,
                                               sub_deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – deterministic mass‑transfer + annealing (remainder)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _mass_transfer_and_anneal(best_seq,
                                                   target_sum,
                                                   deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq
    return final_seq
