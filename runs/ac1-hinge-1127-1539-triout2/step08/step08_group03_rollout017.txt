def project_simplex(v, S):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    # find rho
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def subgrad_optimize(a0, deadline, step0=0.5, decay=0.999, gamma=0.01):
    n = len(a0)
    S = np.sqrt(2*n)
    a = a0.copy()
    best_a = a.copy()
    best_val = max_convolution(a)
    step = step0
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        val, max_idx = max_convolution_and_max(a) # returns val, idx
        # gradient
        grad = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        # indices i = lo..hi correspond to inv = max_idx - i
        grad[lo:hi+1] = 2 * a[max_idx - np.arange(lo, hi+1)]
        # take step in negative gradient direction
        a_temp = a - step * grad
        # project onto simplex
        a = project_simplex(a_temp, S)
        # ensure nonnegative mean: done
        val_new = max_convolution(a)
        if val_new < best_val:
            best_val = val_new
            best_a = a.copy()
        # decay step
        step *= decay
        # occasional jitter
        if iteration % 1000 == 0:
            a = a + np.random.normal(scale=step*0.1, size=n)
            a = project_simplex(a, S)
    return best_a, best_val

"""
Advanced step‑function optimiser.

The algorithm consists of three stages:

1️⃣  Random exploration – many candidates are drawn from a set of
    parametric families (Dirichlet, Gamma, power‑law, exponential,
    sparse bumps).  The vectors are normalised to ℓ₁‑norm = √(2 n)
    (so that the objective equals the peak of the self‑convolution).
    The best *k* candidates are retained.

2️⃣  Multi‑scale refinement – every retained candidate is enlarged to a
    larger support (up to ~30 000 entries) and optimised with two
    gradient‑based procedures:
       • Adam optimiser on a τ‑soft‑max approximation of the max‑convolution
         (exactly as in the previous generation).
       • Adam optimiser on the *exact* sub‑gradient of the max‑convolution
         (gradient = 2·reversed a truncated to the active shift).  After
         each step the iterate is projected onto the simplex (non‑negative,
         prescribed ℓ₁‑norm).

3️⃣  Cheap simulated‑annealing fine‑tuning – a mass‑transfer walk on the
    simplex (Metropolis acceptance) for the remaining time.

All intermediate results are printed, the final construction is stored in
the global variable ``height_sequence_1`` and returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _max_convolution_and_gradient(a: np.ndarray):
    """Peak of the self‑convolution together with the exact sub‑gradient.

    The gradient of max(b) w.r.t. a at the active shift *k* is
        g_i = 2·a_{k‑i}   if 0 ≤ k‑i < n, else 0.
    """
    conv = _full_convolution_fft(a)
    k = int(conv.argmax())
    peak = float(conv[k])
    n = a.shape[0]

    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    rev = k - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[rev]
    return peak, grad


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = S}."""
    # Duchi et al. (2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        centre = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, centre - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _initial_vector(
    n: int,
    rng: np.random.Generator,
    warm_seq: np.ndarray | None,
) -> np.ndarray:
    """Generate a feasible start vector of length n."""
    S = np.sqrt(2.0 * n)

    # occasional warm‑start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a * (S / a.sum())
        return a

    style = rng.choice(
        ["dirichlet", "gamma", "uniform", "left_powerlaw", "left_exponential", "sparse"]
    )
    if style == "dirichlet":
        raw = _init_dirichlet(n, rng)
    elif style == "gamma":
        raw = _init_gamma(n, rng)
    elif style == "uniform":
        raw = _init_uniform(n, rng)
    elif style == "left_powerlaw":
        raw = _init_left_powerlaw(n, rng)
    elif style == "left_exponential":
        raw = _init_left_exponential(n, rng)
    else:
        raw = _init_sparse_bumps(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max version – same as previous code)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    lr0: float = 0.45,
    tau_decay: float = 0.9994,
    lr_decay: float = 0.9998,
    jitter_every: int = 500,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the τ‑soft‑max of the max‑convolution."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)  # required ℓ₁‑norm

    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)  # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient (soft max)
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient wrt logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # annealing schedules
        tau *= tau_decay
        lr *= lr_decay

        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Adam optimiser on the *exact* sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize_exact(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    lr0: float = 0.5,
    lr_decay: float = 0.9999,
    jitter_every: int = 500,
) -> tuple[np.ndarray, float]:
    """Adam optimiser using the exact sub‑gradient of max(conv)."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    a = a * (S / a.sum())  # ensure correct ℓ₁‑norm

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    best_a = a.copy()
    best_peak = _max_convolution(a)
    it = 0

    while time.time() < deadline:
        it += 1
        peak, grad = _max_convolution_and_gradient(a)

        # Adam accumulation
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # projection onto the simplex (non‑negative, fixed ℓ₁‑norm)
        a = _project_to_simplex(a, S)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        lr *= lr_decay

        if jitter_every and it % jitter_every == 0:
            a += rng.normal(scale=S * 0.01, size=n)
            a = _project_to_simplex(a, S)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (cheap version)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with Metropolis acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score  # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())  # renormalise exactly

        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm follows three stages (exploration, multi‑scale refinement,
    cheap annealing) and stores the best construction in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    margin = 5.0
    end_time = time.time() + budget_s - margin  # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previous construction, if present
    # ------------------------------------------------------------------
    warm_seq = None
    if "height_sequence_1" in globals():
        try:
            warm_seq = np.asarray(globals()["height_sequence_1"], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 350.0  # leave time for refinement & anneal
    best_seq = None
    best_score = np.inf
    top_k = 8
    top_candidates = []  # list of (score, seq)

    attempts = 0
    while time.time() < explore_deadline:
        attempts += 1

        # choose length – a biased log‑uniform spread
        if rng.random() < 0.55:
            n = int(rng.integers(1500, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain the top‑k list
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Ensure at least one candidate exists
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement
    # ------------------------------------------------------------------
    max_len = min(30000, best_n * 6)          # safe ceiling
    length_schedule = sorted(
        set(
            [
                best_n,
                int(best_n * 1.5),
                int(best_n * 2.5),
                max_len,
            ]
        )
    )

    for cand_idx, (cand_score, cand_seq) in enumerate(top_candidates):
        cur_seq = cand_seq.copy()
        cur_n = len(cur_seq)

        for target_n in length_schedule:
            if time.time() >= end_time - 40.0:
                break

            # enlarge / shrink to the target length
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining budget
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.35, 120.0)

            # 1️⃣  soft‑max Adam phase
            refined, peak = _adam_optimize(
                a_scaled,
                time.time() + slice_budget,
                rng,
                lr0=rng.choice([0.38, 0.44, 0.5]),
                tau_decay=rng.choice([0.9992, 0.9994, 0.9996]),
                lr_decay=0.99985,
                jitter_every=400,
            )

            # 2️⃣  exact‑gradient Adam phase (shorter budget)
            remaining2 = end_time - time.time()
            if remaining2 > 15.0:
                refined2, peak2 = _adam_optimize_exact(
                    refined,
                    time.time() + min(30.0, remaining2 * 0.15),
                    rng,
                    lr0=0.5,
                    lr_decay=0.9999,
                    jitter_every=500,
                )
                if peak2 < peak:
                    refined, peak = refined2, peak2

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] cand#{cand_idx+1} len={target_n:5d}  best={best_score:.6f}")

            # use refined vector as seed for the next scale
            cur_seq = refined
            cur_n = target_n

        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning if time permits
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        final_seq, final_peak = _anneal(best_seq, time.time() + remaining, rng)
        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_seq.copy()
            print(f"[anneal] final annealed score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(
        f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
