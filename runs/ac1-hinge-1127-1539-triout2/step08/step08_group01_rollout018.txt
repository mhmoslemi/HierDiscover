"""
Propose a candidate step function minimizing evaluate_sequence.

Method:
1. Initialise a non‑negative vector a of length N=2000 (seeded from previous best
   if available) and project onto the ℓ¹‑simplex Σa = √(2N).
2. Repeatedly solve a linear program
       max 1ᵀ g  subject to  a⋆g ≤ M , g ≥ 0,
   where M = max(a⋆a).  The LP increases the total mass while keeping the
   a‑convolution bounded.  The solution g is scaled back to the required sum
   and mixed with a (convex combination).  The process strictly reduces the
   evaluation value until no further improvement is possible.
3. A short polishing phase uses the projected Adam optimiser together with the
   deterministic biased hill‑climb from the previous state‑of‑the‑art code.
4. The trailing near‑zero entries are trimmed and the list is returned.

The routine respects a 1000 s wall‑clock limit and prints progress.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT version)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Adam optimiser (same as before, kept for polishing)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b

def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    a = a0.copy()
    n = a.size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.25 * target_sum
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
    return a

# ----------------------------------------------------------------------
# Hill‑climbing (same as previous implementation)
# ----------------------------------------------------------------------
def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9
    cooling = 0.9997
    while time.time() < deadline:
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
        T *= cooling
    return best_a, b

# ----------------------------------------------------------------------
# LP based improvement
# ----------------------------------------------------------------------
def _lp_improve_max(f: np.ndarray, target_sum: float,
                    deadline: float) -> tuple[np.ndarray, bool]:
    """
    Perform one LP step that tries to increase the total mass while keeping
    the convolution with the current vector bounded by its own maximal entry.
    Returns the new vector (projected onto the simplex) and a flag indicating
    whether a strict improvement was observed.
    """
    n = f.size
    M = _max_convolution_fft(f)

    # Build matrix A where A[k, j] = f[k - j] if 0 <= k-j < n else 0
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    # Linear program: maximise sum(g)  (i.e. minimise -sum(g))
    c = -np.ones(n, dtype=float)
    b_ub = np.full(m, M, dtype=float)

    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return f, False

    g = res.x
    sum_g = g.sum()
    # If the LP did not increase the ℓ¹‑norm we stop.
    if sum_g <= target_sum * (1.0 + 1e-8):
        return f, False

    # Scale g back to the required ℓ¹‑norm.
    g_scaled = g * (target_sum / sum_g)

    # Mix the old and the new vector (convex combination) and re‑project.
    t = 0.5
    new_f = (1.0 - t) * f + t * g_scaled
    new_f = _project_to_simplex(new_f, target_sum)

    # Evaluate improvement.
    old_val = _max_convolution_fft(f)
    new_val = _max_convolution_fft(new_f)

    improved = new_val < old_val - 1e-12
    return new_f, improved

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm first performs a few linear‑programming based updates,
    then finishes with the Adam + hill‑climb polishing of the previous
    state‑of‑the‑art code.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise a seed of length 2000
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = seq0.size
        target = math.sqrt(2.0 * n)
        seq0 = _project_to_simplex(seq0, target)
    else:
        n = 2000
        target = math.sqrt(2.0 * n)
        seq0 = np.full(n, target / n, dtype=float)

    best_seq = seq0.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] start (n={n}) → {best_val:.6f}")

    # ------------------------------------------------------------------
    # LP improvement phase – at most a few seconds per iteration.
    # ------------------------------------------------------------------
    while time.time() < end_time - 5.0:
        new_seq, improved = _lp_improve_max(best_seq, target, end_time)
        if not improved:
            # no progress; exit loop
            break
        best_seq = new_seq
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] LP step improvement → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional short polishing with Adam + hill‑climb (if time left)
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        # A couple of Adam restarts
        gd_deadline = time.time() + (end_time - time.time()) * 0.4
        a_opt = _adam_gradient_descent(best_seq, target, gd_deadline)
        val_adam = evaluate_sequence(a_opt.tolist())
        if val_adam < best_val:
            best_seq = a_opt
            best_val = val_adam
            print(f"[{time.time():.1f}] Adam polish → {best_val:.6f}")

        # final hill‑climb
        bic_deadline = end_time - 1.0
        b_cur = _full_convolution(best_seq)
        best_seq, b_final = _hill_climbing_improved(best_seq, b_cur, bic_deadline)
        final_score = b_final.max()
        if final_score < best_val:
            best_val = final_score
            print(f"[{time.time():.1f}] final hill‑climb → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros and return.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] final evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error during final evaluation: {exc}")

    return best_seq.tolist()

"""
Propose a candidate step‑function that (approximately) minimises
``evaluate_sequence``.

Method
------
1.  Initialise a non‑negative vector of length 2000 (use the previous
    construction if it exists) and project it onto the ℓ¹‑simplex
    Σa = √(2 n).

2.  Repeatedly solve a linear programme

        max   Σ g_j
        s.t.  (f * g)_k ≤ M   for all k,
              g_j ≥ 0,

    where ``M = max (f * f)``.  
    The LP increases the total mass while keeping the convolution with the
    current vector bounded.  After scaling the solution back to the required
    ℓ¹‑norm we convex‑combine it with the old vector; the maximal
    convolution value strictly decreases until the LP can no longer
    improve the objective.

3.  A short polishing phase (projected Adam optimisation followed by the
    deterministic biased hill‑climb from the previous state‑of‑the‑art
    implementation) is applied to the best vector found.

The routine respects a 1000 s wall‑clock limit and prints progress
information.  All helper functions are top‑level; no file‑IO or
network access is performed.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation on the smooth‑max objective.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.25 * target_sum
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (biased version)
# ----------------------------------------------------------------------
def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The destination is biased towards indices that contribute little to the
    current worst convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9997                    # slower cooling for finer search

    while time.time() < deadline:
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index (biased by its contribution to the worst entry)
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index (biased towards small counterpart a[M‑j])
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Linear‑programming based improvement
# ----------------------------------------------------------------------
def _lp_improve_max(f: np.ndarray, target_sum: float,
                    deadline: float) -> tuple[np.ndarray, bool]:
    """
    One LP step:
        maximise Σ g_j   subject to   (f * g)_k ≤ M,   g ≥ 0,
    where M = max (f * f).
    The solution is scaled back to the required ℓ¹‑norm and mixed
    with the current vector.  Returns the new vector and a flag
    indicating whether a strict improvement was observed.
    """
    n = f.size
    M = _max_convolution_fft(f)

    # Build convolution matrix A: A[k, j] = f[k‑j] if 0 ≤ k‑j < n else 0
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    # maximise Σ g_j  ⇔ minimise −Σ g_j
    c = -np.ones(n, dtype=float)
    b_ub = np.full(m, M, dtype=float)

    # solve the LP (method 'highs' is fast and reliable)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})

    if not res.success:
        return f, False

    g = res.x
    sum_g = g.sum()

    # If the LP failed to increase the ℓ¹‑norm we stop.
    if sum_g <= target_sum * (1.0 + 1e-8):
        return f, False

    # Scale back to the required ℓ¹‑norm and convex‑combine.
    g_scaled = g * (target_sum / sum_g)
    t = 0.5
    new_f = (1.0 - t) * f + t * g_scaled
    new_f = _project_to_simplex(new_f, target_sum)

    # Check whether the maximal self‑convolution decreased.
    old_val = _max_convolution_fft(f)
    new_val = _max_convolution_fft(new_f)
    improved = new_val < old_val - 1e-12
    return new_f, improved
