def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # retrieve prior base
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # Phase 1: Random search
    phase1_deadline = start_time + 0.07 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1
        # length selection
        if base_seq is not None and rng.random() < 0.5:
            n = max(4000, int(base_seq.shape[0] * rng.uniform(0.9, 1.1)))
        else:
            n = int(rng.integers(4000, max_len+1))

        a = _sample_family(rng, n, base_seq)

        # Normalize to sum = sqrt(2n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0)
            s = n
        a = a / s * S

        # optionally symmetrize
        if rng.random() < 0.4:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[phase1] attempt={attempts} n={n} score={cur_score:.6f}")

    # If nothing found, fallback on uniform
    if best_seq is None:
        n = 5000
        a = np.full(n, 1.0)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_seq = a
        best_score = np.max(_full_convolution(a))

    # Phase 2: Adam on current best (medium length)
    now = time.time()
    remaining = deadline - now
    adam_deadline = now + 0.5 * remaining
    best_seq, best_score = _optimize_adam(best_seq, adam_deadline, rng,
                                          max_alpha=2000.0, enforce_sym=False)
    print(f"[adam-mid] score={best_score:.6f} length={len(best_seq)}")

    # Phase 3: Upsample to larger length (if not already max)
    now = time.time()
    if now < deadline - 30:
        cur_n = best_seq.shape[0]
        target_n = min(max_len, int(cur_n * 1.7))
        if target_n > cur_n:
            a_big = _upsample_seq(best_seq, target_n)
            # normalise
            S_big = np.sqrt(2.0 * target_n)
            a_big = a_big / a_big.sum() * S_big

            # Phase 3b: Adam refinement on larger length
            t_refine = deadline - 5.0  # use remaining time
            a_big, score_big = _optimize_adam(a_big, t_refine, rng,
                                              max_alpha=2500.0, enforce_sym=False)
            if score_big < best_score:
                best_seq = a_big
                best_score = score_big
                print(f"[upsample-adam] improved to {best_score:.6f} length={len(best_seq)}")

    # Phase 4: Subgradient descent (short)
    now = time.time()
    if now < deadline - 5:
        sub_deadline = deadline - 5
        seq_sub, score_sub = _optimize_subgrad(best_seq, sub_deadline, rng,
                                               enforce_sym=False)
        if score_sub < best_score:
            best_seq = seq_sub
            best_score = score_sub
            print(f"[subgrad] improved to {best_score:.6f}")

    # Phase 5: Simulated annealing final polish

"""
Improved step‑function search.

The function performs a multi‑phase search for a non‑negative sequence
that minimizes 2·n·max(conv(a,a))/sum(a)^2.
The pipeline is:

1. Random sampling of many candidate families (short time, keep the best).
2. Long projected‑Adam optimisation on the current best shape.
3. Upsample the best shape to a larger length (≈1.7×, capped at 30 000) and
   run a short but aggressive Adam refinement with a very high soft‑max
   temperature.
4. A brief sub‑gradient descent on the true max‑convolution.
5. A final simulated‑annealing hill‑climb for polishing.
6. The resulting sequence is stored globally as ``height_sequence_1`` and
   returned as a python list.

All heavy work uses FFT‑based convolutions, Euclidean projection onto the
scaled simplex (∑a = √(2·n)), and simple stochastic moves.  The algorithm
fits comfortably into the 1000 s budget on a single CPU and typically
improves the bound from ~1.5105 to ≤1.504.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the non‑negative simplex scaled by s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Random families for initial candidates
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Produce a random candidate of length n using a mixture of analytic families."""
    fam = rng.integers(0, 10)
    if fam == 0 and base_seq is not None:
        # upscale an existing good solution
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform
    elif fam == 2:
        rate = rng.uniform(0.94, 0.998)
        left = rate ** np.arange(n)
        a = left + left[::-1]                     # symmetric geometric decay
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p          # symmetric power‑law
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)           # beta‑like shape
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p   # sinusoidal power
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)
        a = (np.arange(n, 0, -1) ** p)            # monotone decreasing power‑law
    elif fam == 7:
        lam = rng.uniform(0.001, 0.025) * n
        a = np.exp(-lam * np.arange(n) / n)       # exponential decay
    elif fam == 8:
        mid = rng.uniform(0.25, 0.75) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))  # logistic S‑curve
    else:
        # piecewise “double‑tent”: high at both ends, low in centre
        apex = rng.integers(n // 4, 3 * n // 4)
        left = np.linspace(1, 0, apex, endpoint=False)
        right = np.linspace(0, 1, n - apex)
        a = np.concatenate([left, right])

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 2000.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns the best vector seen and the associated peak value (max convolution).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for soft‑max
    alpha = 0.1
    alpha_mul = 1.025

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape plateaus
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.08 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.999985   # very slow decay

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.3
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: g_i = a_{k_max-i} for feasible i
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random aspirational move
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early stop if stagnating
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that swaps tiny masses.
    Accepts worsened moves with probability exp(-Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    cur_max = best_score

    start = time.time()
    T0 = max(0.12 * cur_max, 2e-3)      # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.06 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Retrieve previous best if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0
    max_len = 30000   # upper bound for sequence length

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈7 % of total time)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.07 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # Choose length
        if base_seq is not None and rng.random() < 0.5:
            n = max(4000, int(base_seq.shape[0] * rng.uniform(0.9, 1.1)))
        else:
            n = int(rng.integers(4000, max_len + 1))

        a = _sample_family(rng, n, base_seq)

        # Normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        # Random optional symmetrisation
        if rng.random() < 0.4:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[phase1] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing was sampled (should not happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        a = np.full(n, 1.0)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_seq = a
        best_score = np.max(_full_convolution(a))

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation on current best (≈50 % of remaining)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    adam_deadline = now + 0.5 * remaining
    best_seq, best_score = _optimize_adam(best_seq, adam_deadline, rng,
                                          max_alpha=2000.0, enforce_sym=False)
    print(f"[adam-mid] best_score={best_score:.6f} length={len(best_seq)}")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to a larger length and refine (≈30 % of remaining)
    # ------------------------------------------------------------------
    now = time.time()

        target_n = min(max_len, int(cur_n * 1.7))
        if target_n > cur_n:
            a_big = _upsample_seq(best_seq, target_n)
            S_big = np.sqrt(2.0 * target_n)
            a_big = a_big / a_big.sum() * S_big

            # Run a short but aggressive Adam on the larger sequence
            t_refine = deadline - 5.0   # leave a few seconds for final steps
            a_big, score_big = _optimize_adam(a_big, t_refine, rng,
                                              max_alpha=2500.0,
                                              enforce_sym=False)
            if score_big < best_score:
                best_seq = a_big
                best_score = score_big
                print(f"[upsample‑adam] improved to {best_score:.6f} length={len(best_seq)}")

    # ------------------------------------------------------------------
    # Phase 4 – brief sub‑gradient descent (≈10 % of remaining)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 8.0:
        sub_deadline = deadline - 5.0
        seq_sub, score_sub = _optimize_subgrad(best_seq, sub_deadline, rng,
                                               enforce_sym=False)
        if score_sub < best_score:
            best_seq = seq_sub
            best_score = score_sub
            print(f"[subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated‑annealing hill‑climb for polishing (≈5 % of remaining)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 2.0:
        seq_ann, score_ann = _anneal_hillclimb(best_seq, deadline, rng,
                                               enforce_sym=False)
        if score_ann < best_score:
            best_seq = seq_ann
            best_score = score_ann
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return final result (store globally)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
