"""
Optimisation of a non‑negative step function to minimise the peak of its
self‑convolution.

The evaluation metric (with the standard normalisation Σa = √(2·n))
coincides with the peak of a⋆a.  The algorithm proceeds in three stages:

1. **Exploration** – generate many candidate shapes from a diverse set of
   analytic families (uniform, geometric, power‑law, beta, sinusoidal,
   logistic, Gaussian, double‑Gaussian, piecewise‑linear).  The K best
   vectors (lowest internal peak) are kept as an elite pool.

2. **Refinement** – each elite vector is locally optimised by a
   three‑stage routine:
        * Projected Adam on a smooth log‑sum‑exp approximation of the
          max‑convolution.
        * Projected sub‑gradient descent on the true max entry.
        * Simulated‑annealing random‑swap hill‑climb.
   All updates respect the simplex constraint Σa = √(2·n) and enforce
   exact mirror symmetry, which is known to be beneficial for this
   problem.

3. **Multi‑scale up‑sampling** – the current best vector is repeatedly
   up‑sampled (factors 1.2, 1.33, 1.5, 1.75, 2.0, bounded by length 25000)
   and each up‑sampled version receives a short Adam refinement.
   Finally a long random‑swap annealing phase exhausts the remaining time.

The function `propose_candidate` follows the above plan and returns the
best sequence found before the time limit (default 1000 s).  The returned
list is also stored in the global variable `height_sequence_1` for warm‑starts
in subsequent runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Simplex projection (Euclidean) – onto {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of total mass s."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Linear convolution a⋆a (full length = 2·n‑1) via FFT
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
#  Gradient of Σ_i w_i·(a⋆a)_i  w.r.t. a
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇_a Σ_i w_i (a⋆a)_i .  w must be a probability vector (sum 1)."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
#  Soft‑max (log‑sum‑exp) probabilities over convolution entries
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return probs proportional to exp(alpha·conv)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
#  Force exact mirror symmetry (v[i] == v[-1-i])
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
#  Linear interpolation to a new length (preserves shape)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
#  Internal metric – peak of the self‑convolution
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
#  Candidate generator – a diverse collection of analytic families
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator,
                   base: np.ndarray | None) -> np.ndarray:
    """Return a non‑negative vector of length n drawn from a random family."""
    fam = rng.integers(0, 10)   # 0 … 9
    if fam == 0 and base is not None:
        # up‑sample previous best (no further randomisation)
        a = _upsample_seq(base, n)

    elif fam == 1:
        # pure uniform random
        a = rng.random(n)

    elif fam == 2:
        # symmetric geometric decay from both ends
        rate = rng.uniform(0.92, 0.9995)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) * 0.5

    elif fam == 3:
        # symmetric power‑law (i·(n‑i))^p
        p = rng.uniform(0.2, 4.5)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p

    elif fam == 4:
        # beta‑like (x^p1 * (1‑x)^p2)
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)

    elif fam == 5:
        # sinusoidal power
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

    elif fam == 6:
        # symmetric logistic bump
        scale = rng.uniform(0.02, 0.15) * n
        mid = (n - 1) / 2.0
        xs = np.arange(n, dtype=float)
        left = 1.0 / (1.0 + np.exp((xs - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - xs) / scale))
        a = left + right

    elif fam == 7:
        # Gaussian centred at the middle
        sigma = rng.uniform(0.12, 0.45) * n
        mid = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n) - mid) / sigma) ** 2)

    elif fam == 8:
        # Double Gaussian mixture (two symmetric bumps)
        sigma1 = rng.uniform(0.07, 0.25) * n
        sigma2 = rng.uniform(0.07, 0.25) * n
        mid = (n - 1) / 2.0
        g1 = np.exp(-0.5 * ((np.arange(n) - mid) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((np.arange(n) - mid) / sigma2) ** 2)
        w = rng.uniform(0.3, 0.7)
        a = w * g1 + (1.0 - w) * g2

    else:   # fam == 9
        # Piecewise‑linear with a handful of random break points
        m = rng.integers(4, 10)                     # number of knots
        xs = np.sort(rng.random(m))
        xs = np.concatenate(([0.0], xs, [1.0]))
        hs = rng.random(m + 2)                      # heights at knots
        # linear interpolation on the normalized grid
        x_grid = np.linspace(0.0, 1.0, n)
        a = np.interp(x_grid, xs, hs)

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
#  Projected Adam minimising the soft‑max approximation of the peak
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the smooth max
    alpha = 0.2
    alpha_mul = 1.08
    max_alpha = 2000.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # smooth weights (log‑sum‑exp)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional gradient clipping to avoid jumps
        step_norm = np.linalg.norm(step)
        if step_norm > 0.5:
            step = step * (0.5 / step_norm)

        a -= step

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to jump out of plateaux
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997   # extremely gentle decay

    return best_a, best_score


# ----------------------------------------------------------------------
#  Projected sub‑gradient descent on the true max entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random‑swap with SA style acceptance
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.15:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        # early stop if no progress for a while
        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
#  Simulated‑annealing random‑swap hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _local_anneal_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    start = time.time()
    # initial temperature based on current score range
    while time.time() < deadline:
        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.03 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _peak_convolution(cand)

        # deterministic improvement
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            continue

        # simulated‑annealing acceptance
        elapsed = time.time() - start
        # linear cooling from 0.02 to 0.0005
        T = max(5e-4, 0.02 * (1.0 - elapsed / (deadline - start)))
        if rng.random() < math.exp(-(cand_score - best_score) / T):
            a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    #  Warm‑start from previous run, if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    #  Phase 0 – cheap exploration, keep the K elite vectors
    # ------------------------------------------------------------------
    K = 12                           # richer elite pool
    elite = []                       # list of (score, vector)
    attempts = 0
    phase0_deadline = start_time + 0.14 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1

        # length choice – favour modestly sized vectors for speed,
        # but occasionally explore larger ones
        if base_seq is not None and rng.random() < 0.55:
            scale = rng.uniform(0.80, 1.30)
            n = max(500, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(3000, 15001))

        a = _gen_candidate(n, rng, base_seq)

        # normalise to Σa = √(2·n) (the standard simplex)
        S = np.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # enforce symmetry for half of the candidates
        if rng.random() < 0.5:
            a = _symmetrize(a)

        cur_score = _peak_convolution(a)

        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    #  Fallback – if for any reason elite is empty
    # ------------------------------------------------------------------
    if not elite:
        n = 8000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        elite.append((_peak_convolution(a), a))

    # ------------------------------------------------------------------
    #  Phase 1 – refinement of elite vectors
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # allocate roughly equal budget to each elite; top 3 get a bigger share
    per_elite = remaining / len(elite)

    best_overall_seq = None
    best_overall_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        # Fine‑tune the allocation: top 3 get 1.5× the share
        share = per_elite * (1.5 if idx < 3 else 1.0)
        ada_dead = now + 0.55 * share
        sub_dead = ada_dead + 0.30 * share
        loc_dead = sub_dead + 0.15 * share

        # ---- Adam (smooth max) ----
        a_adam, sc_adam = _optimize_adam(vec, ada_dead, rng, enforce_sym=True)
        # ---- Sub‑gradient (true max) ----
        a_sub, sc_sub = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=True)
        # ---- Local annealing ----
        a_loc, sc_loc = _local_anneal_search(a_sub, loc_dead, rng, enforce_sym=True)

        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[refine {idx}] improved to {best_overall_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    #  Phase 2 – multi‑scale up‑sampling with short Adam bursts
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 25000
        # progressive factors that fill the interval smoothly
        up_factors = [1.2, 1.33, 1.5, 1.75, 2.0]
        for fac in up_factors:
            new_n = int(best_overall_seq.shape[0] * fac)
            if new_n > max_len or new_n < 2:
                continue
            a_up = _upsample_seq(best_overall_seq, new_n)
            S_new = np.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * S_new
            a_up = _symmetrize(a_up)

            now = time.time()
            if now >= deadline:
                break
            # give each up‑sampled version about 8 % of what is left
            remain = deadline - now
            short_dead = now + min(0.08 * remain, 30.0)   # cap to keep it short
            a_up_opt, sc_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=True)

            if sc_up < best_overall_score:
                best_overall_score = sc_up
                best_overall_seq = a_up_opt.copy()
                print(f"[upsample] factor={fac:.2f} n={new_n} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    #  Phase 3 – final long annealing on the best vector
    # ------------------------------------------------------------------
    now = time.time()
    if best_overall_seq is not None and now < deadline:
        a_final, sc_final = _local_anneal_search(best_overall_seq, deadline, rng,
                                                enforce_sym=True)
        if sc_final < best_overall_score:
            best_overall_score = sc_final
            best_overall_seq = a_final.copy()
            print(f"[final‑local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    #  Return result (store globally for warm‑restarts)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
