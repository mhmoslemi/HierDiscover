def proj_simplex(v, S):
    # v is any real vector.
    # returns projection onto simplex of sum S.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid LP + gradient optimizer for the step‑function challenge.

The algorithm works in three phases:

1. **LP refinement on a modest size (≈1500).**  
   For a fixed length we iteratively solve a linear program that
   maximises the ℓ1‑norm of a new distribution g while keeping the
   self‑convolution f∗g below the current peak M of f∗f.  After each
   LP we scale g to the required ℓ1–norm (√(2n)) and mix it with the
   current iterate  f←(1‑t)f+t g .  This deterministic scheme quickly
   drives the maximal convolution down.

2. **Scale‑up & Adam refinement.**  
   The best small‑scale candidate is interpolated to a few larger
   lengths (up to ≈ 2·n_small) and optimised with the τ‑soft‑max Adam
   optimiser that was already proven effective.  Longer runs are given
   to the most promising length.

3. **Cheap stochastic hill‑climb** to escape tiny plateaus and a final
   sanity check with the provided ``evaluate_sequence`` function.

The implementation falls back to the pure Adam optimiser if SciPy is
unavailable (so it always runs).  All helper functions are top‑level,
no closures or lambdas are used, and the total runtime is kept safely
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, scaling, convolution, projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _proj_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, Σx = S}."""
    if S < 0:
        raise ValueError("S must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient based Adam optimiser (τ‑soft‑max)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2*(mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                           # aggressive start

    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)            # start with mild smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9998
        lr *= 0.9999

        if it % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random parametric families (used for warm‑starts)
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and
    otherwise draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with prob ≈0.2
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (convex‑combination step)
# ----------------------------------------------------------------------
def _lp_step(current: np.ndarray, M: float, t: float,
             rows: np.ndarray, cols: np.ndarray, shape: tuple,
             rng: np.random.Generator) -> np.ndarray:
    """
    One LP refinement step.
    Solves   maximise Σ g_j   s.t.   g ≥ 0  and  (current * g)_k ≤ M  ∀k .
    Returns the convex combination (1‑t)·current + t·g_scaled.
    """
    n = current.shape[0]
    # Build data for the constraint matrix A (size (2n‑1)×n)
    data = np.tile(current, n)                     # f_i repeated for every column
    A = sp.csr_matrix((data, (rows, cols)), shape=shape)

    # Linear programme: minimise –Σ g_j  (i.e. maximise Σ g_j)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # If the LP fails for any reason we simply return the unchanged vector.
        return current

    g = res.x
    S = np.sqrt(2.0 * n)
    sumg = g.sum()
    if sumg > 0:
        g = g * (S / sumg)

    # convex combination
    new_vec = (1.0 - t) * current + t * g
    return new_vec

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Hybrid LP / Adam optimiser for the step‑function problem.
    Returns a list of non‑negative heights.
    """
    import time
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – small‑scale LP refinement
    # ------------------------------------------------------------------
    n_small = 1500                     # modest size for fast LP solves
    if warm_seq is not None:
        a_cur = _scale_sequence(warm_seq, n_small)
        a_cur = np.clip(a_cur, 0.0, None)
        S_small = np.sqrt(2.0 * n_small)
        a_cur = a_cur * (S_small / a_cur.sum())
    else:
        a_cur = _initial_vector(n_small, rng, None)

    best_seq = a_cur.copy()
    # exact objective value (no external evaluator yet)
    best_score = 2.0 * n_small * _max_convolution(a_cur) / (np.sum(a_cur) ** 2)
    print(f"[init] n_small={n_small:,d}  score={best_score:.6f}")

    # Prepare static sparse‑matrix index pattern for the given n_small
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
        have_linprog = True
    except Exception:
        have_linprog = False

    if have_linprog:
        n = n_small
        i_idx = np.arange(n)[:, None]          # shape (n,1)
        j_idx = np.arange(n)[None, :]          # shape (1,n)
        rows = (i_idx + j_idx).ravel()         # (i+j) for each pair
        cols = np.tile(np.arange(n), n)        # column index j repeated for each i
        shape_mat = (2 * n - 1, n)

        lp_iter = 0
        while time.time() < end_time - 30.0 and lp_iter < 20:
            M = _max_convolution(a_cur)
            a_new = _lp_step(a_cur, M, t=0.2,
                             rows=rows, cols=cols, shape=shape_mat,
                             rng=rng)
            new_score = 2.0 * n_small * _max_convolution(a_new) / (np.sum(a_new) ** 2)

            lp_iter += 1
            print(f"[LP] iter={lp_iter:2d}  score={new_score:.6f}")

            if new_score + 1e-9 < best_score:
                best_score = new_score
                best_seq = a_new.copy()

            # simple stopping criterion
            if abs(new_score - best_score) < 1e-6:
                break
            a_cur = a_new
    else:
        print("[LP] SciPy not available – skipping LP phase")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & Adam refinement on several lengths
    # ------------------------------------------------------------------
    lengths = [best_seq.shape[0]]
    for factor in [1.3, 1.6, 2.0]:
        cand = int(best_seq.shape[0] * factor)
        if 2000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 30.0:
            break
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 180.0)

        a_scaled = _scale_sequence(best_seq, target_n)
        S_target = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S_target / a_scaled.sum())

        a_opt, peak = _adam_optimize(a_scaled,
                                      time.time() + slice_time,
                                      rng)

        score = 2.0 * target_n * peak / (np.sum(a_opt) ** 2)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[refine] target_n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final long Adam optimisation on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 12.0:
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        score = 2.0 * best_seq.shape[0] * peak / (np.sum(a_opt) ** 2)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[final refine] length={best_seq.shape[0]:5d}  best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.15)
        amt = a[i] * frac
        a[i] -= amt
        a[j] += amt
        peak = _max_convolution(a)
        score = 2.0 * n * peak / (np.sum(a) ** 2)
        if score + 1e-9 < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[local] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
