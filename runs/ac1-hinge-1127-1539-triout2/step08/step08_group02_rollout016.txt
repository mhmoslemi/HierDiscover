"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a list of heights) that minimises
``evaluate_sequence``.  The loss is proportional to the infinity‑norm of the
self‑convolution of the normalised sequence, i.e. we minimise

    2·n·max( a * a ) / (sum(a)**2)      (n = len(a))

which is invariant to a global scaling of the vector.  The algorithm
combines three ideas:

1. **Diverse random starts** – several parametric families (power‑law,
   exponential, triangular, U‑shape, sinusoid, quadratic, Dirichlet, …)
   are sampled and projected onto the ℓ₁‑simplex of radius √(2n).

2. **Adam optimisation on a smooth max** – the true max is approximated
   by a τ‑soft‑max (log‑sum‑exp) of the convolution.  Gradients are
   obtained efficiently with FFTs and the parameters are optimised on
   the log‑soft‑max (logits) using Adam.  τ starts large and is cooled
   slowly to 0, at which point the exact sub‑gradient of the max is used.

3. **Refinement and stochastic hill‑climbing** – after the Adam phase the
   best vector is repeatedly rescaled to a few larger lengths (up to
   ~20000 entries) and each scaled copy receives an intensive Adam run.
   Finally a cheap random perturbation loop moves small fractions of
   mass between coordinates (and occasionally swaps two entries) and
   accepts any improvement.

The function respects the 1000 s budget, printing progress and storing the
final construction in the global ``height_sequence_1`` variable.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = rfft(a, n=f_len)
    conv = irfft(A * A, n=f_len)[:total_len]
    return conv

def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()

# ----------------------------------------------------------------------
# Soft‑max (τ‑soft‑max) gradient of the max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂/∂a τ‑soft‑max( a*a )
    and max_b is the (hard) maximum of the convolution.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    # exact sub‑gradient when tau is tiny
    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # grad = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]

    conv_len = mask.shape[0] + a.shape[0] - 1
    f_len = _next_pow2(conv_len)

    Fmask = rfft(mask, n=f_len)
    Fa_rev = rfft(a_rev, n=f_len)
    conv = irfft(Fmask * Fa_rev, n=f_len)[:conv_len]

    # extract the part that aligns with a (central n entries)
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b

# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the exact max‑convolution (τ → 0).
    Returns the best sequence and its max‑convolution value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)               # ℓ₁‑norm we target

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.55                           # a bit more aggressive than before

    # initialise temperature – start fairly large, then cool
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 5.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S                     # respects ℓ₁‑norm automatically

        # ------- convolution & gradient -------
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip = np.max(np.abs(grad_p))
        if clip > 8.0 * S:
            grad_p = grad_p * (8.0 * S / clip)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- schedules -------
        tau *= 0.9994                     # slower decay – keeps smoothing a while
        lr *= 0.99995                    # very gentle decay

        # occasional jitter to escape plateaus
        if it % 700 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak

# ----------------------------------------------------------------------
# Interpolation utilities (for scaling to new lengths)
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw

def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw

def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw

def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:

    p = rng.uniform(0.4, 2.5)

    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base

def _init_sinusoid(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sinusoidal shape, optionally flipped."""
    i = np.arange(n)
    phase = rng.uniform(0, 2*np.pi)
    freq = rng.uniform(0.8, 2.5)   # a few periods over the interval
    raw = np.abs(np.sin(freq * i / (n-1) * np.pi + phase))
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    return mix * raw + (1.0 - mix) * base

def _init_quadratic(n: int, rng: np.random.Generator) -> np.ndarray:
    """Convex quadratic bowl, optionally flipped."""
    i = np.arange(n)
    a = rng.uniform(-2.0, 2.0)   # curvature (negative => U‑shape)
    b = rng.uniform(-0.5, 0.5) * n
    c = rng.uniform(0.0, 0.5) * n**2
    raw = a * (i - n/2)**2 + b * (i - n/2) + c
    raw = np.clip(raw, 0.0, None)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.1))
    return mix * raw + (1.0 - mix) * base

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:

    otherwise a parametric family is sampled.
    The result is projected onto the ℓ₁‑sphere of radius √(2n).
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # choose a family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'ushape', 'sinusoid', 'quadratic'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sinusoid':
        raw = _init_sinusoid(n, rng)
    elif style == 'quadratic':
        raw = _init_quadratic(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm follows the three‑phase scheme
    described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # reserve a few seconds for cleanup

    # ------------------------------------------------------------------
    # keep a warm start if a previous construction exists
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – quick random exploration (≈ 60 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0   # keep ~5 min for refinement & hill‑climb
    while time.time() < explore_deadline:
        attempts += 1
        # length selection – favour a few thousand entries but allow larger ones
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 6500))
        else:
            n = int(rng.integers(6500, 21000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)          # exact max (no smoothing)
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # If exploration failed (unlikely) fall back to a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_n = 3000
        best_seq = np.full(best_n, np.sqrt(2.0 * best_n) / best_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – scaling to several larger lengths + longer Adam runs
    # ------------------------------------------------------------------
    # generate a few candidate lengths (geometric growth, capped at 20000)
    lengths = [best_n]
    for factor in [1.2, 1.5, 2.0, 3.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 20000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 60.0:
            break
        # allocate a slice of the remaining budget (max 180 s per length)
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.18, 180.0)

        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        a_opt, peak = _adam_optimize(a_scaled,
                                      time.time() + slice_time,
                                      rng)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[scale‑refine] target_n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final long Adam run on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 12.0:
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final‑adam] length={best_n:5d}  best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        # pick two distinct indices and move a modest fraction of mass
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount

        # occasional three‑way swap (helps escape tiny plateaus)
        if rng.random() < 0.03:
            k = rng.integers(0, n)
            a[i], a[k] = a[k], a[i]

        peak = _max_convolution(a)
        if peak + 1e-10 < best_score:          # strict improvement
            best_score = peak
            best_seq = a.copy()
            print(f"[hill‑climb] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
