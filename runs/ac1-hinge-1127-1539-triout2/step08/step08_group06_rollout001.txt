"""
Improved step‑function optimizer
--------------------------------
The goal is to minimise the peak of the self‑convolution of a non‑negative
vector under the L1‑normalisation Σa = √(2·n).  The evaluation routine
therefore simply returns that peak.

Key ideas compared with the reference implementation
* Enforce exact mirror symmetry at every optimisation step – symmetric
  shapes are empirically much better for the peak‑reduction problem.
* Augment the family of analytic initialisations with a Gaussian bump
  (the only extra family needed to broaden the search space).
* Use a more aggressive Adam schedule (larger learning‑rate, larger final
  temperature) and always project onto the simplex after each update.
* After the usual Adam → sub‑gradient → tiny‑swap pipeline, perform a
  *guided* swap phase that explicitly tries to shave off the current
  maximal convolution entry.
* Upsample the best candidate to larger lengths (up to 15000) and give it a
  short Adam burst.
* All phases respect the simplex constraint via the Euclidean projection
  routine `_project_simplex`.  The code keeps the ten elite candidates
  found during the cheap exploration phase and refines each of them
  independently.

The function `propose_candidate` returns a list of floats that can be fed
directly to `evaluate_sequence`.  It respects the 1000‑second wall‑time
budget (a a few seconds margin is kept for the final return).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (zero‑padded)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights over convolution entries
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    m = np.max(scaled)
    exp_shifted = np.exp(scaled - m)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of Σ w_i (a*a)_i  w.r.t. a (weights sum to 1)
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = 3 * n - 2                     # length of a * w_rev * a
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # indices that correspond to the gradient components
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Symmetrisation (v[i] = v[-1-i])
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear interpolation to a new length
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Current peak of the self‑convolution (the objective we minimise)
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Analytic families for cheap exploration (now 8 families)
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    fam = rng.integers(0, 8)
    if fam == 0 and base is not None:
        # up‑sample previous best
        a = _upsample_seq(base, n)
    elif fam == 1:
        # pure uniform random
        a = rng.random(n)
    elif fam == 2:
        # symmetric geometric decay
        rate = rng.uniform(0.92, 0.9995)
        left = rate ** np.arange(n, dtype=float)
        a = (left + left[::-1]) * 0.5
    elif fam == 3:
        # symmetric power‑law (i·(n‑i))^p
        p = rng.uniform(0.2, 4.5)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        # beta‑like shape
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        # sinusoidal power
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        # symmetric logistic bump
        scale = rng.uniform(0.02, 0.15) * n
        mid = (n - 1) / 2.0
        xs = np.arange(n, dtype=float)
        left = 1.0 / (1.0 + np.exp((xs - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - xs) / scale))
        a = left + right
    else:
        # Gaussian bump centred at the middle
        sigma = rng.uniform(0.07, 0.20) * n
        xs = np.arange(n, dtype=float)
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((xs - centre) / sigma) ** 2)

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # hyper‑parameters (more aggressive than the reference)
    lr = 1.0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.2              # start temperature
    alpha_mul = 1.07
    max_alpha = 2000.0       # very tight approximation at the end

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter (helps escape plateaus)
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99996   # very gentle decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max entry (with symmetry)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.22
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # gradient of the selected entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random small swaps (simulated‑annealing flavour)
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.15:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        # early stop if stagnating
        if it % 12000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Guided swap phase – explicitly attacks the current maximal entry
# ----------------------------------------------------------------------
def _guided_swap_search(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # indices that contribute to the peak
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[k_max - idx_i]          # equals 0.5 * gradient component
        # Choose i with largest contribution (and positive mass)
        i_candidates = idx_i[contrib > 0.0]
        if i_candidates.size == 0:
            # fall back to random move
            i = rng.integers(0, n)
        else:
            i = i_candidates[np.argmax(contrib[i_candidates == i_candidates])]

        if a[i] <= 0.0:
            # random jitter if nothing to move
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 0.0:
                continue
            delta = 0.04 * a[i] * rng.random()
            a[i] -= delta
            a[j] += delta
            if enforce_sym:
                a = _symmetrize(a)
            a = _project_simplex(a, S)
            cur = _peak_convolution(a)
            if cur < best_score:
                best_score, best_a = cur, a.copy()
            continue

        # Try a few random destinations and keep the one that gives the biggest
        # estimated reduction of the current peak.
        best_imp = -np.inf
        best_j = None
        best_delta = None
        for _ in range(6):
            j = rng.integers(0, n)
            if j == i:
                continue
            # linear estimate of the change in conv[k_max]
            contrib_j = a[k_max - j] if (0 <= k_max - j < n) else 0.0
            if contrib <= contrib_j:
                continue
            delta = a[i] * rng.uniform(0.5, 1.0)   # move a sizable chunk
            delta_change = -delta * contrib + delta * contrib_j
            if i + j == k_max:
                delta_change -= 2.0 * delta * delta   # quadratic term (exact)
            est_new = conv[k_max] + delta_change
            imp = conv[k_max] - est_new
            if imp > best_imp:
                best_imp = imp
                best_j = j
                best_delta = delta

        if best_j is None:
            # fallback to a small random swap
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = 0.02 * a[i] * rng.random()
        else:
            j = best_j
            delta = best_delta

        # Apply the move
        a[i] -= delta
        a[j] += delta
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur = _peak_convolution(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()

    return best_a, best_score

            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # enforce symmetry for the exploration stage as well
        if rng.random() < 0.6:
            a = _symmetrize(a)

        cur_score = _peak_convolution(a)

        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case of no elite
    # ------------------------------------------------------------------
    if not elite:
        n = 8000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        elite.append((_peak_convolution(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – per‑elite refinement (Adam → sub‑grad → guided swaps)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    time_per_elite = remaining / len(elite)

    best_overall_seq = None
    best_overall_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        ada_dead = now + 0.55 * time_per_elite
        sub_dead = ada_dead + 0.28 * time_per_elite
        guid_dead = sub_dead + 0.12 * time_per_elite
        loc_dead = deadline if (guid_dead + 0.04 * time_per_elite) > deadline else guid_dead + 0.04 * time_per_elite

        # Adam (smooth max)
        a_adam, sc_adam = _optimize_adam(vec, ada_dead, rng, enforce_sym=True)

        # Sub‑gradient on true max
        a_sub, sc_sub = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=True)

        # Guided swaps tackling the current peak
        a_guid, sc_guid = _guided_swap_search(a_sub, guid_dead, rng, enforce_sym=True)

        # Tiny random swaps for final polishing
        a_loc, sc_loc = _local_swap_search(a_guid, loc_dead, rng, enforce_sym=True)

        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[refine {idx}] improved to {best_overall_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – upsample the current best to larger lengths
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 15000
        up_factors = [1.2, 1.5, 2.0]
        for fac in up_factors:
            new_n = int(best_overall_seq.shape[0] * fac)
            if new_n > max_len or new_n < 2:
                continue
            a_up = _upsample_seq(best_overall_seq, new_n)

            # normalise to the new length
            S_new = np.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * S_new
            a_up = _symmetrize(a_up)

            # short Adam burst (≈ 6 % of the time left)
            now = time.time()
            if now >= deadline:
                break
            short_dead = min(deadline, now + 0.06 * (deadline - now))
            a_up_opt, sc_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=True)

            if sc_up < best_overall_score:
                best_overall_score = sc_up
                best_overall_seq = a_up_opt.copy()
                print(f"[upsample] factor={fac:.2f} new_n={new_n} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final random‑swap polishing (any remaining budget)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_final, sc_final = _local_swap_search(best_overall_seq, deadline, rng, enforce_sym=True)
        if sc_final < best_overall_score:
            best_overall_score = sc_final
            best_overall_seq = a_final.copy()
            print(f"[final‑local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result, store globally for warm‑starts
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
