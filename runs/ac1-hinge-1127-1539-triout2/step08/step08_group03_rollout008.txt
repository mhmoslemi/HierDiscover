def _proj_simplex(v, S):

    # shift v so that sum(v) = S? Actually we need to solve.

    # From Duchi et al., algorithm:

    u = np.sort(v)[::-1]

    cssv = np.cumsum(u) - S

    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]

    theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0)

    return w

"""
Optimiser for the “max‑convolution” bound.

The algorithm proceeds in three stages within the 1000 s budget:

1️⃣  Random exploration – many vectors are drawn from a collection of
    parametric families (Dirichlet, Gamma, uniform, left‑hand power‑law,
    left‑hand exponential, sparse bumps, cosine bumps and random spikes).
    Each vector is normalised so that its ℓ₁‑norm equals √(2 n)
    (the objective then is simply max⁡(a∗a)).  The best few candidates are
    retained.

2️⃣  Multi‑scale refinement – every retained candidate is interpolated to a
    series of larger lengths.  For each length we run two complementary
    optimisers:
        • Adam on a τ‑soft‑max approximation of the max‑convolution.
        • Exact sub‑gradient descent on the true max‑convolution, with a
          projection onto the simplex (non‑negative, fixed ℓ₁‑norm).  
    The refined vector with the smallest peak is kept.

3️⃣  Final fine‑tuning – a cheap simulated‑annealing mass‑transfer walk
    runs for any remaining time.

The best sequence found is stored in the global variable
`height_sequence_1` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of ``a``."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Projection onto the simplex (ℓ₁ = S, non‑negative)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x ≥ 0, sum(x) = S}.
    """
    if S <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries would become zero – return uniform distribution
        theta = (np.sum(v) - S) / n
        w = np.maximum(v - theta, 0.0)
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max approximation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """Adam optimiser for the τ‑soft‑max of the max‑convolution."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from a feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient (soft max)
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # annealing schedules
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (max‑convolution, not softened)
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          alpha0: float = 0.5,
                          decay: float = 0.999,
                          momentum: float = 0.9,
                          jitter_every: int = 300) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector seen and the corresponding peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    b = _full_convolution_fft(a)
    best_peak = b.max()

    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and its maximum
        b = _full_convolution_fft(a)
        max_b = b.max()
        # mask of active (max) indices – average to obtain a sub‑gradient
        mask = (np.abs(b - max_b) <= 1e-12).astype(float)
        if mask.sum() > 0:
            mask /= mask.sum()

        # gradient of the weighted max (2 * (mask*a) * reversed a)
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

        # momentum update
        v = momentum * v + (1.0 - momentum) * g

        # learning‑rate schedule
        alpha = alpha0 * (decay ** it)

        # take a step and project back onto the simplex
        a = _proj_simplex(a - alpha * v, S)

        # occasional jitter to avoid stagnation
        if jitter_every and (it % jitter_every) == 0:
            # multiplicative log‑space noise
            eps = rng.normal(scale=0.004, size=n)
            a = a * np.exp(eps)
            a = _proj_simplex(a, S)

        # keep the best seen point
        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)

def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_cosine_bump(n: int, rng: np.random.Generator) -> np.ndarray:
    """A smooth bump given by a powered cosine profile."""
    idx = np.arange(n)
    p = rng.uniform(0.5, 4.0)
    raw = np.sin(np.pi * (idx + 0.5) / n) ** p
    # optional background mixture
    mix = rng.uniform(0.0, 0.3)
    raw = mix + (1.0 - mix) * raw
    # random reversal to increase diversity
    if rng.random() < 0.5:
        raw = raw[::-1]
    return raw


def _init_random_spikes(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few isolated high spikes on a tiny background."""
    raw = np.full(n, rng.uniform(0.0, 0.04))
    nspikes = rng.integers(1, max(2, n // 200))
    for _ in range(nspikes):
        pos = rng.integers(0, n)
        height = rng.uniform(0.8, 5.0)
        raw[pos] = max(raw[pos], height)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length ``n``.
    With probability ≈0.22 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'cosine', 'spikes'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'cosine':
        raw = _init_cosine_bump(n, rng)
    else:  # 'spikes'
        raw = _init_random_spikes(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (very cheap, used only if time left)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The final sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0      # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if present
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration
    # ------------------------------------------------------------------
    explore_deadline = end_time - 400.0          # keep time for refinement
    top_k = 15
    top_candidates = []   # list of (score, seq)
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1

        # pick a length – a mix of medium and large sizes
        if rng.random() < 0.65:
            n = int(rng.integers(1500, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        # maintain a sorted list of the best candidates
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

        if attempt % 500 == 0:
            print(f"[explore] attempt={attempt:5d} n={n:5d} best={top_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Ensure we have at least one candidate
    # ------------------------------------------------------------------
    if not top_candidates:
        n = 2000
        a0 = np.full(n, np.sqrt(2.0 * n) / n)
        top_candidates = [( _max_convolution(a0), a0.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale refinement (Adam + exact sub‑gradient)
    # ------------------------------------------------------------------
    best_seq, best_score = top_candidates[0][1].copy(), top_candidates[0][0]

    for init_idx, (init_score, init_seq) in enumerate(top_candidates[:5]):
        cur_seq = init_seq.copy()
        cur_n = len(cur_seq)

        # length schedule – grow geometrically up to a safe ceiling
        max_len = min(35000, cur_n * 8)
        lengths = [cur_n]
        factor = 1.5
        while lengths[-1] < max_len:
            nxt = int(lengths[-1] * factor)
            if nxt > max_len:
                nxt = max_len
            if nxt == lengths[-1]:
                break
            lengths.append(nxt)
            factor = 1 + (factor - 1) * 0.8
        lengths = sorted(set(lengths))

        for target_n in lengths:
            remaining = end_time - time.time()
            if remaining < 45:
                break

            # interpolate to target length and renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining budget for Adam
            slice_budget = min(remaining * 0.35, 180.0)
            lr0 = rng.choice([0.38, 0.44, 0.50])
            tau_decay = rng.choice([0.9992, 0.9994, 0.9996])
            refined, peak_adam = _adam_optimize(a_scaled,
                                                time.time() + slice_budget,
                                                rng,
                                                lr0=lr0,
                                                tau_decay=tau_decay,
                                                lr_decay=0.99985,
                                                jitter_every=400)

            # a short exact sub‑gradient pass after Adam
            sub_budget = min(remaining * 0.12, 90.0)
            refined2, peak_sub = _subgradient_optimize(refined,
                                                       time.time() + sub_budget,
                                                       rng,
                                                       alpha0=0.7,
                                                       decay=0.9995,
                                                       momentum=0.85,
                                                       jitter_every=250)

            # keep the better of the two refinements
            if peak_sub < peak_adam:
                refined, best_local = refined2, peak_sub
            else:
                refined, best_local = refined, peak_adam

            if best_local < best_score:
                best_score = best_local
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d} improved score={best_score:.6f}")

            cur_seq = refined
            cur_n = target_n

            if time.time() > end_time - 30:
                break

        if time.time() > end_time - 30:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final exact sub‑gradient on the (potentially larger) best vector
    # ------------------------------------------------------------------
    # enlarge a bit more if time permits
    final_target = min(35000, best_n * 2)
    if final_target > best_n:
        cur = _scale_sequence(best_seq, final_target)
        cur = np.clip(cur, 0.0, None)
        cur = cur * (np.sqrt(2.0 * final_target) / cur.sum())
        best_seq = cur.copy()
        best_n = final_target
        print(f"[scale_up] enlarged to {best_n}")

    remaining = end_time - time.time()
    if remaining > 20:
        final_seq, final_peak = _subgradient_optimize(best_seq,
                                                      time.time() + remaining,
                                                      rng,
                                                      alpha0=0.45,
                                                      decay=0.9985,
                                                      momentum=0.9,
                                                      jitter_every=300)
        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_seq.copy()
            print(f"[subgrad] final sub‑gradient improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap simulated‑annealing fine‑tuning if time remains
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 25:
        final_seq2, final_peak2 = _anneal(best_seq,
                                          time.time() + remaining,
                                          rng)
        if final_peak2 < best_score:
            best_score = final_peak2
            best_seq = final_seq2.copy()
            print(f"[anneal] final annealed score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    external = evaluate_sequence(final_list)
    print(f"[final] external score = {external:.6f} internal max = {best_score:.6f} length = {len(final_list)}")
    globals()['height_sequence_1'] = final_list
    return final_list
