def _optimize_adam_logit(a0, deadline, rng, max_alpha=400.0, enforce_sym=False):
    # a0 is initial sequence satisfying sum = sqrt(2n)
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    # initialize logits as log(a0+eps)
    eps = 1e-12
    v = np.log(a0 + eps)
    # Adam hyperparameters
    lr = 0.5 # adjust
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v) # rename
    it = 0
    best_a = a0.copy()
    best_score = np.max(_full_convolution(a0))
    alpha = 0.1
    alpha_mul = 1.04

    while time.time() < deadline:
        it += 1
        # compute a
        expv = np.exp(v - np.max(v)) # for numerical stability
        a = expv * (S / expv.sum())
        # optionally symmetrize a after conversion to maintain symmetry
        if enforce_sym:
            a = _symmetrize(a)
            # update v to reflect symmetrical? we can just symmetrize expv accordingly: but easier to symmetrize a and then recompute v = log(a+eps). This may break momentum; but it's okay.
            v = np.log(a + eps)
        # compute conv
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # compute weights
        w = _smooth_weights(conv, alpha)
        grad_a = _conv_with_weights(a, w)  # shape n
        # compute grad_v
        dot_ag = np.dot(a, grad_a)   # scalar
        grad_v = a * (grad_a - dot_ag / S)  # shape n
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v * grad_v)
        m_hat = m / (1 - beta1 ** it)
        v_hat_corr = v_hat / (1 - beta2 ** it)
        v -= lr * m_hat / (np.sqrt(v_hat_corr) + 1e-8)

        # schedule
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995

        # occasional jitter: small random perturbation of weights
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = 0.05 * a[i] * rng.random()
                # adjust logits to move some mass from i to j: easier to adjust a then recompute v
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)  # using proj for this small fix
                v = np.log(a + eps)

    return best_a, best_score
```

"""
Improved step‑function search for the convolution‑peak ratio.

The algorithm works in three stages:
1.  Generate a large number of random candidates from a diverse
    family of analytic shapes and keep the best (internal score = max
    self‑convolution, which equals the external evaluation when the
    L1‑norm is fixed to √(2·n)).
2.  Run a projected‑Adam optimiser many times with different initial
    vectors.  The optimiser works on *logits* (soft‑max parametrisation)
    so the non‑negativity and sum constraints are satisfied without an
    explicit Euclidean‑projection at every step.
3.  Optionally up‑sample the current best solution (≈30 % larger) and
    perform a final short Adam refinement.

All work is limited to the supplied time budget (default 1000 s);
the best sequence found before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: Euclidean projection onto the simplex (used only for jitter)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  log‑sum‑exp(α·(a*a))  w.r.t. a
    (w must be a probability vector over the convolution entries).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max helpers for the smooth‑max approximation
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over the entries of conv (larger α → sharper)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1‑i]."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families for the initial sampling stage
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int,
                   base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a variety of families."""
    fam = rng.integers(0, 9)
    if fam == 0 and base_seq is not None:
        # up‑sample an existing good solution
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                 # symmetric power‑law
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                # beta‑like skewed
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                 # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)                 # monotone decreasing power‑law
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n        # exponential decay
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    else:
        # piecewise linear “tent”
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser working on logits (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _optimize_adam_logit(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_alpha: float = 500.0,
                         enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex using a soft‑max parametrisation.
    Returns the best vector (list of floats) and its internal max‑conv score.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                 # required L1‑norm
    eps = 1e-12

    # initialise logits from a0
    v = np.log(a0 + eps)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(v)
    v_m = np.zeros_like(v)               # called 'v' in Adam theory
    it = 0

    # temperature schedule for the smooth‑max
    alpha = 0.1
    alpha_mul = 1.04

    best_a = np.exp(v - np.max(v))
    best_a = best_a * (S / best_a.sum())
    best_score = np.max(_full_convolution(best_a))

    while time.time() < deadline:
        it += 1

        # construct a from logits
        maxv = np.max(v)
        expv = np.exp(v - maxv)          # stability
        a = expv * (S / expv.sum())

        if enforce_sym:
            a = _symmetrize(a)
            # keep logits consistent with the symmetrised a
            v = np.log(a + eps)

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights over the convolution entries
        w = _smooth_weights(conv, alpha)

        # gradient w.r.t. a (using the analytic formula)
        grad_a = _conv_with_weights(a, w)        # shape (n,)

        # convert gradient to logits space
        dot_ag = np.dot(a, grad_a)                # scalar
        grad_v = a * (grad_a - dot_ag / S)

        # Adam update in the logit space
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_m = beta2 * v_m + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_m / (1.0 - beta2 ** it)

        v -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if rng.random() < 0.012:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.06 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
                v = np.log(a + eps)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99996
        # keep logits in a reasonable numeric range
        if np.abs(v).max() > 50:
            v = np.clip(v, -50, 50)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
