def _subgradient_max_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute conv and max index
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # gradient:
        g = np.zeros_like(a)
        lo = max(0, k - (len(a)-1))
        hi = min(k, len(a)-1)
        # i in [lo, hi], j = k - i
        # gradient for i:
        # if i == j -> a[i]; else 2*a[j]
        # So:
        for i in range(lo, hi+1):
            j = k - i
            if i == j:
                g[i] = a[i]  # derivative w.r.t a_i (since term appears once)
            else:
                g[i] = 2 * a[j]
        # g is non-negative

        # Determine step size:
        # max step to keep non-negative: min_i a[i]/g[i] for g[i]>0
        idx_pos = g > 0
        if not np.any(idx_pos):
            # gradient zero (maybe optimum). Break
            break
        max_alpha = np.min(a[idx_pos] / g[idx_pos])
        # Take a fraction of max_alpha, e.g., 0.8 initial
        # adapt step decay: use schedule decreasing with iteration
        step_frac = 0.7  # could vary
        alpha = step_frac * max_alpha

        # optionally add random jitter to avoid stuck
        # compute new a
        a_new = a - alpha * g
        # Ensure numeric positivity
        a_new = np.maximum(a_new, 0.0)

        # reproject to simplex
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} step={alpha:.3e} score={best_val:.6f}")

        # Accept a_new always (even if not improvement)? Could accept only if improvement; else revert maybe smaller step.

        a = a_new

        # Occasionally jitter:

            jitter = rng.normal(scale=target_sum*0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

"""
Propose a step‑function with a low benchmark score.

Algorithm
---------

1. **Problem size** – pick n based on the available time budget.
2. **Parametric warm‑start** – fast grid / random search on the
   2‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n) on a reduced
   problem (≈ n/4).  The best small sequence is interpolated to the
   target length and normalised to Σa = √(2n).
3. **Exact‑max subgradient descent** – repeatedly
   * compute the linear convolution a∗a,
   * locate the index k = argmax(a∗a),
   * build the sub‑gradient of the max‑convolution at k,
   * take a step in the negative sub‑gradient direction,
   * project back onto the ℓ¹‑simplex (Σa fixed, a≥0).
   The step size is a fraction of the largest feasible step that
   keeps all entries non‑negative; the fraction decays with time.
4. **Mass‑transfer refinement** – the deterministic “move a few percent
   of mass from the most harmful donor to a low‑impact receiver” routine
   from the previous solution is applied for the remaining time.
5. **Final clean‑up** – strip trailing zeros, evaluate the sequence with
   the supplied ``evaluate_sequence`` and store it for warm‑starts in
   subsequent calls.

All stages respect the non‑negativity and ℓ¹‑norm constraints, and the
search stops automatically when the allotted 1000 s budget expires.
"""

# Convolution utilities

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Search a two‑parameter family
        a_i = (i+1)^(-p)·exp(-λ·i/small_n)
    on a small problem.  Returns the best sequence found.
    """
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:

            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)

def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that directly minimises the maximal entry of a*a.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < deadline:
        # compute convolution and locate the active index
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # build sub‑gradient of the maximal convolution entry
        i = np.arange(n)
        j = k - i
        valid = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # generic case: contribution appears twice
        g[valid] = 2.0 * a[j[valid]]
        # diagonal contribution (i == j) should count only once
        diag = valid & (i == j)
        g[diag] = a[i[diag]]

        # if gradient is zero we are at a stationary point
        if not np.any(g > 0):
            break

        # maximal feasible step (keep a >= 0)
        pos = g > 1e-15
        max_alpha = np.min(a[pos] / g[pos])

        # progressive step‑size schedule
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total
        step_frac = max(0.1, 0.7 * (1.0 - frac))    # decays from 0.7 → 0.1
        alpha = step_frac * max_alpha

        # backtracking line‑search for a monotone improvement
        val_current = _score(a)
        improved = False
        while alpha > 1e-15:
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_to_simplex(a_new, target_sum)
            val_new = _score(a_new)
            if val_new < val_current:
                a = a_new
                improved = True
                if val_new < best_val:
                    best_val = val_new
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Subgrad it={it:5d} α={alpha:.3e} score={best_val:.6f}")
                break
            else:
                alpha *= 0.5

        if not improved:
            # No improvement – add a tiny jitter to escape flat region
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # occasional stronger jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass from indices that
    contribute most to the current maximal convolution entry to
    indices that contribute least, accepting only improvements.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        # sensitivities for the current maximal entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        total_w = weight.sum()
        if total_w > 0:
            donor = rng.choice(n, p=weight / total_w)
        else:
            donor = rng.integers(n)

        low_thr = np.percentile(weight, 10)
        low_idx = np.where(weight <= low_thr)[0]
        receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

        if donor == receiver:
            continue

        # transfer a modest fraction
        max_frac = 0.03
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # reject
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise if needed
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val
