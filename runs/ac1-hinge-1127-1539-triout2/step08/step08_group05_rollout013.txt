"""
Improved step‑function search for minimizing max self‑convolution.

Key ideas
---------
* Enforce exact mirror symmetry throughout optimisation – the optimum is
  believed to be symmetric, and symmetry reduces the search space.
* Use a richer set of analytic seed families (including a semicircle and a
  flat‑top “tent”) to start closer to the optimum.
* Run a single long Adam optimisation with a much higher soft‑max temperature
  (α up to 3000) while projecting onto the simplex scaled to √(2 n).
* Follow with a sub‑gradient descent and a short simulated‑annealing hill‑climb,
  all with symmetry enforced.
* Finally up‑sample the best sequence (≈+30 %) and perform a quick high‑α
  Adam refinement.
* All steps respect the non‑negativity and the sum‑normalisation required by
  `evaluate_sequence`.

The function `propose_candidate` returns the best sequence found before the
time budget (default 1000 s) expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    `w` must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 2000.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.04
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry enforcement
        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.08 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient (single‑term version)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random aspirational move
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit if stagnating
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that swaps tiny masses.
    Accepts worsened moves with probability exp(-Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    cur_max = best_score
    start = time.time()
    T0 = max(0.1 * cur_max, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.05 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score

# Analytic families for the initial sampling stage
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a variety of families."""
    fam = rng.integers(0, 11)   # 0‑10 inclusive
    if fam == 0 and base_seq is not None:
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                  # symmetric power‑law
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                 # beta‑like skewed
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                  # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)                  # monotone decreasing power‑law
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    elif fam == 9:
        # semicircle shape: sqrt(1 - ((2i/(n-1))-1)^2)
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.clip(1 - x * x, 0.0, None))
    else:  # fam == 10
        # flat‑top tent shape
        plateau_frac = rng.uniform(0.15, 0.45)
        plateau_len = max(2, int(plateau_frac * n))
        left_len = (n - plateau_len) // 2
        right_len = n - plateau_len - left_len
        left = np.linspace(0, 1, left_len, endpoint=False)
        right = np.linspace(1, 0, right_len, endpoint=False)
        a = np.concatenate([left, np.ones(plateau_len), right])
    a = np.clip(a, 0.0, None)
    return a

    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈15 % of total time)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1
        # choose a length; favour large n but stay ≤ 25000
        if base_seq is not None and rng.random() < 0.45:
            scale = rng.uniform(0.9, 1.1)
            n = max(2000, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(3000, 25001))

        a = _sample_family(rng, n, base_seq)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        a = _symmetrize(a)          # enforce symmetry immediately

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing was generated (should not happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        a = _symmetrize(a)
        best_seq = a
        best_score = np.max(_full_convolution(a))

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation (≈55 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # keep a safety buffer for later phases
    adam_time = now + max(0.0, min(0.55 * remaining, remaining - 120.0))
    a_adam, sc_adam = _optimize_adam(best_seq, adam_time, rng,
                                     max_alpha=3000.0, enforce_sym=True)
    if sc_adam < best_score:
        best_score = sc_adam
        best_seq = a_adam.copy()
        print(f"[adam] improved to {best_score:.6f} n={len(best_seq)}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent (≈25 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        sub_time = now + max(0.0, min(0.25 * (deadline - now), deadline - now - 60.0))
        a_sub, sc_sub = _optimize_subgrad(best_seq, sub_time, rng,
                                          enforce_sym=True)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (≈5 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        anneal_time = deadline   # use whatever is left for a short climb
        a_ann, sc_ann = _anneal_hillclimb(best_seq, anneal_time, rng,
                                          enforce_sym=True)
        if sc_ann < best_score:
            best_score = sc_ann
            best_seq = a_ann.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – up‑sampling + high‑alpha Adam refinement (≈remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 5.0:
        cur_n = best_seq.shape[0]
        # increase length by up to 30 % (capped at 30 k)
        new_n = min(30000, int(cur_n * 1.3))
        a_big = _upsample_seq(best_seq, new_n)

        S_big = np.sqrt(2.0 * new_n)
        a_big = a_big / a_big.sum() * S_big
        a_big = _symmetrize(a_big)

        # a brief high‑α Adam refinement
        refine_deadline = deadline
        a_ref, sc_ref = _optimize_adam(a_big, refine_deadline, rng,
                                       max_alpha=4000.0, enforce_sym=True)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[upsample‑adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (external score) and return
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
