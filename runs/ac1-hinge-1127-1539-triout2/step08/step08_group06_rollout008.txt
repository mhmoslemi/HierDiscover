Improved step‑function search for the self‑convolution peak problem.

Main ideas
----------
1. **Broader exploration** – generate candidates from many analytic families,
   including Dirichlet random vectors, mixtures of beta densities, Gaussian bumps
   and simple piecewise‑constant “edge‑mass” shapes.  The best `K` vectors are
   kept as elites.

2. **Refinement** – each elite undergoes three optimisation stages:
      * *soft‑max Adam*   – minimise a smooth log‑sum‑exp approximation of the
        convolution peak, with an increasing temperature (α) and aggressive
        learning‑rate schedule.
      * *exact sub‑gradient* – minimise the true max entry using projected
        sub‑gradient descent.
      * *local hill‑climb* – tiny random mass swaps.

   The optimiser works directly on the full convolution via FFT, and projects
   onto the simplex {a_i≥0, Σa_i = √(2·n)} after every update.

3. **Annealing phase** – starting from the best refined vector we run a fast
   simulated‑annealing loop that accepts occasional uphill moves (temperature
   slowly decays).  This helps to escape shallow local minima that the

   (≈1.2×, 1.5×, 2×, capped at 20 000).  Each up‑sample receives a short Adam
   burst and a final annealing pass.

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using FFT (length = 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _peak_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(_full_convolution(a)))


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) probabilities over the convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i (a*a)_i w.r.t. a.
    (w must sum to 1).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = 3 * n - 2            # length of full (a * w_rev) convolution
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)   # indices that correspond to the gradient
    grad = 2.0 * full[idx]
    return grad


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1‑i]."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a vector to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# 2. Candidate generation (many analytic families)
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    """Return a random non‑negative vector of length n."""
    fam = rng.integers(0, 9)
    # -- 0: up‑sample previous best (if available) --------------------
    if fam == 0 and base is not None:
        a = _upsample_seq(base, n)

    # -- 1: Dirichlet / Gamma random (sparse) -------------------------
    elif fam == 1:
        alpha = rng.uniform(0.2, 2.0)          # concentration
        a = rng.gamma(alpha, size=n)

    # -- 2: Uniform ----------------------------------------------------
    elif fam == 2:
        a = rng.random(n)

    # -- 3: Symmetric power‑law ((i+1)*(n‑i))^p ------------------------
    elif fam == 3:
        p = rng.uniform(0.1, 5.0)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p

    # -- 4: Beta‑like (x^p1 * (1‑x)^p2) ------------------------------
    elif fam == 4:
        p1 = rng.uniform(0.01, 5.0)
        p2 = rng.uniform(0.01, 5.0)
        x = np.linspace(0.0, 1.0, n)
        a = (x ** p1) * ((1 - x) ** p2)

    # -- 5: Mixture of two beta shapes --------------------------------
    elif fam == 5:
        w = rng.random()
        p1a = rng.uniform(0.01, 5.0)
        p2a = rng.uniform(0.01, 5.0)
        p1b = rng.uniform(0.01, 5.0)
        p2b = rng.uniform(0.01, 5.0)
        x = np.linspace(0.0, 1.0, n)
        a = w * (x ** p1a) * ((1 - x) ** p2a) + (1 - w) * (x ** p1b) * ((1 - x) ** p2b)

    # -- 6: Gaussian bump (possibly off‑centre) -----------------------
    elif fam == 6:
        mu = rng.uniform(0.2, 0.8) * (n - 1)
        sigma = rng.uniform(0.05, 0.2) * n
        xs = np.arange(n, dtype=float)
        a = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)

    # -- 7: Piecewise‑constant edge‑mass shape -------------------------
    elif fam == 7:
        # fraction of indices that belong to the “edge” region
        edge_frac = rng.uniform(0.01, 0.3)
        m = max(1, int(edge_frac * n))
        high = rng.uniform(0.5, 1.5)
        low = rng.uniform(0.0, 0.5)
        a = np.full(n, low)
        a[:m] = high
        a[-m:] = high

    # -- 8: Sinusoidal power -------------------------------------------
    else:
        p = rng.uniform(0.3, 5.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# 3. Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth log‑sum‑exp approximation of max."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.9
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    alpha = 0.2
    alpha_mul = 1.07
    max_alpha = 1200.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997   # extremely gentle decay

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the actual maximum convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the selected entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it + 1.0)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random exploratory swap (simulated‑annealing flavour)
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.15:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        # occasional early break if not improving
        if it % 12000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climb that moves tiny mass chunks between random indices."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _peak_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


def _anneal(a0: np.ndarray,
            deadline: float,
            rng: np.random.Generator,
            enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing loop that accepts occasional uphill moves.
    Operates directly on the max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_max = _peak_convolution(a)
    best_a = a.copy()
    best_val = cur_max

    # initialise a small temperature (relative to the excess over 1)
    T = max(0.005, (cur_max - 1.0) * 0.2)
    step_factor = 0.12

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * step_factor * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = _peak_convolution(cand)
        accept = False
        if cand_max < cur_max:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-(cand_max - cur_max) / T):
                accept = True
        if accept:
            a = cand
            cur_max = cand_max
            if cur_max < best_val:
                best_val = cur_max
                best_a = cand.copy()

        # annealing schedules
        T *= 0.99996
        step_factor *= 0.9999
        if T < 1e-7:
            T = 1e-7

    return best_a, best_val

# 4. Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    The result is also stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – exploration (keep K elite vectors)
    # ------------------------------------------------------------------
    K = 10                       # number of elite candidates retained
    elite = []                   # list of tuples (score, vector)
    attempts = 0
    phase0_deadline = start_time + 0.18 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1
        # length choice – favour moderate sizes but allow occasional large ones
        if base_seq is not None and rng.random() < 0.55:
            scale = rng.uniform(0.85, 1.15)
            n = max(500, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(1500, 13001))

        a = _gen_candidate(n, rng, base_seq)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        total = a.sum()
        if total <= 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # random symmetric enforcement (helps many instances)
        if rng.random() < 0.5:
            a = _symmetrize(a)

        cur_score = _peak_convolution(a)

        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    if not elite:
        # fallback – a simple uniform vector
        n = 5000
        a = np.full(n, 1.0)
        a = a / a.sum() * np.sqrt(2.0 * n)
        elite.append((_peak_convolution(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – elite refinement (Adam → sub‑grad → local → anneal)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    time_per_elite = remaining / len(elite)

    best_overall_seq = None
    best_overall_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        # allocate sub‑budgets
        ada_dead = now + 0.52 * time_per_elite
        sub_dead = ada_dead + 0.30 * time_per_elite
        loc_dead = sub_dead + 0.12 * time_per_elite
        anneal_dead = loc_dead + 0.06 * time_per_elite

        # 1) Adam (soft‑max)
        a_adam, sc_adam = _optimize_adam(vec, ada_dead, rng, enforce_sym=False)

        # 2) Sub‑gradient descent (exact max)
        a_sub, sc_sub = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=False)

        # 3) Tiny random‑swap hill‑climb
        a_loc, sc_loc = _local_swap_search(a_sub, loc_dead, rng, enforce_sym=False)

        # 4) Simulated annealing for final polish
        a_anneal, sc_anneal = _anneal(a_loc, anneal_dead, rng, enforce_sym=False)

        final_seq = a_anneal
        final_score = sc_anneal

        if final_score < best_overall_score:
            best_overall_score = final_score
            best_overall_seq = final_seq.copy()
            print(f"[refine {idx}] improved to {best_overall_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – up‑sampling to larger lengths (helps flatten the convolution)
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 20000                      # a safety cap
        up_factors = [1.2, 1.5, 2.0]
        for fac in up_factors:
            new_n = int(best_overall_seq.shape[0] * fac)
            if new_n > max_len or new_n < 4:
                continue
            a_up = _upsample_seq(best_overall_seq, new_n)

            # re‑normalise
            S_new = np.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * S_new

            now = time.time()
            if now >= deadline:
                break
            # short Adam burst (≈ 6 % of the remaining budget)
            short_dead = min(deadline, now + 0.06 * (deadline - now))
            a_up_opt, sc_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=False)

            # final anneal on the up‑sampled candidate
            now = time.time()
            if now < deadline:
                anneal_dead = min(deadline, now + 0.04 * (deadline - now))
                a_up_opt2, sc_up2 = _anneal(a_up_opt, anneal_dead, rng, enforce_sym=False)
                sc_up = sc_up2
                a_up_opt = a_up_opt2

            if sc_up < best_overall_score:
                best_overall_score = sc_up
                best_overall_seq = a_up_opt.copy()
                print(f"[upsample] factor={fac:.2f} n={new_n} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final tiny annealing pass on the globally best vector
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline and best_overall_seq is not None:
        a_final, sc_final = _anneal(best_overall_seq, deadline, rng, enforce_sym=False)
        if sc_final < best_overall_score:
            best_overall_score = sc_final
            best_overall_seq = a_final.copy()
            print(f"[final‑anneal] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return result (store globally for possible warm‑restarts)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
