def find_best_exponential(n, deadline):
    low=0.95; high=0.9999
    target = sqrt(2n)
    for i in range(30):
        r1 = low + (high - low)*0.382

"""Candidate generator for the self‑convolution max‑norm problem.

The algorithm mixes three cheap but complementary ideas:

1️⃣ **Parametric warm‑start** – a fast golden‑section search on pure
   exponential families (a_i ∝ r^i) gives a solid baseline.
2️⃣ **Smoothed‑max Adam descent** – minimise a smooth log‑sum‑exp
   approximation of max(a∗a) while staying on the ℓ¹‑simplex.
3️⃣ **Exact hill‑climbing (simulated annealing)** – moves a tiny amount
   of mass from one index to another, updating the convolution in O(n)
   per move, accepting only improving or slightly worsening steps.

The whole routine runs a tight loop until the 1000 s wall‑clock budget
expires, repeatedly re‑initialising from the current best (with a small
jitter) and applying Adam + hill‑climb phases.  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b_k) where b = a∗a.
    Returns (grad, b), where
        grad_i = 2·∑_k p_k·a_{k-i},
    and p_k ∝ exp(β·b_k) (softmax probabilities).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev ⋆ p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 2e5,
                   beta_factor: float = 1.2) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max objective.
    β grows geometrically from ``beta_start`` to ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.20
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaux
        if step % 350 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if beta < beta_target and step % 300 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Simple parametric families (used for random starts)
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i  with 0<r<1."""
    r = rng.uniform(0.96, 0.9995)
    return r ** np.arange(n, dtype=float)


def _family_two_exponential(n: int, rng) -> np.ndarray:
    """a_i = c·r1^i + (1‑c)·r2^i."""
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    c = rng.uniform(0.0, 1.0)
    i = np.arange(n, dtype=float)
    return c * (r1 ** i) + (1.0 - c) * (r2 ** i)


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential with a breakpoint."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a random family, jitter a little and project onto the simplex."""
    families = [_family_exponential,
                _family_two_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)        # tiny jitter
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Exact hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a source index to a target index
    while keeping the full convolution up‑to‑date in O(n) per move.
    Returns the best (a, b) encountered.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9               # initial temperature
    cooling = 0.998

    while time.time() < deadline:
        # pick a position attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – pick among contributors to this entry,
        # weighted by product size
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias towards positions with little mass
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        # Remove contribution from source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # Add contribution from target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # Correct the cross term that was added twice at i+j
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # recompute exact convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best

def _best_exponential_sequence(n: int,
                               deadline: float,
                               rng) -> tuple[np.ndarray, float]:
    """
    Finds (approximately) the best pure exponential a_i ∝ r^i on the simplex
    using a golden‑section search over r∈(0.95,0.9999).
    Returns (a_opt, value).
    """
    low, high = 0.95, 0.9999
    target = np.sqrt(2.0 * n)

    for _ in range(30):
        if time.time() >= deadline:
            break
        r1 = low + (high - low) * 0.382
        r2 = low + (high - low) * 0.618
        a1 = _project_to_simplex(r1 ** np.arange(n, dtype=float), target)
        a2 = _project_to_simplex(r2 ** np.arange(n, dtype=float), target)
        v1 = _max_convolution_fft(a1)
        v2 = _max_convolution_fft(a2)
        if v1 < v2:
            high = r2
        else:
            low = r1

    r_opt = (low + high) / 2.0
    a_opt = _project_to_simplex(r_opt ** np.arange(n, dtype=float), target)
    val_opt = _max_convolution_fft(a_opt)
    return a_opt, val_opt
