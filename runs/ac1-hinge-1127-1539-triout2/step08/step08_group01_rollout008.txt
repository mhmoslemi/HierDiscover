        b_cur = _full_convolution(a_cur)

            best_score = score_hc

    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size

"""
Sub‑gradient + hill‑climb optimizer for the max‑convolution objective.

Steps
-----
1. Initialise a non‑negative vector on the ℓ¹‑simplex   Σa = √(2 n).
2. Repeatedly perform a projected sub‑gradient descent that
   directly targets the worst convolution entry (the “hard‑max”).
   The direction is g_i = (#max)·a_i + Σ_{m∈argmax} a_{m‑i},
   i.e. the exact sub‑gradient of max_k (a*a)_k.
   A diminishing step‑size (∝1/√t) together with a simplex
   projection yields a fast descent phase.
3. Follow the descent with a biased simulated‑annealing hill‑climb
   (the same routine used in the reference code) to exploit
   discrete moves that are cheap to evaluate.
4. Run the two phases on a length schedule
   (2000 → 5000 → 10 000 → 20 000 → 40 000),
   up‑sampling the current best candidate before moving to a
   larger dimension.
5. A final polishing pass on the best sequence found.
The algorithm respects a wall‑clock budget (default 1000 s) and
prints intermediate scores.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a larger grid, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Sub‑gradient descent on the hard‑max (max‑convolution) objective
# ----------------------------------------------------------------------


def _hardmax_subgradient(a: np.ndarray):
    """
    Compute the exact sub‑gradient of max_k (a*a)_k and the current max value.
    Returns (g, max_b) where
        g_i = (#max)·a_i + Σ_{m∈argmax} a_{m‑i}   (valid indices only)
    """
    n = a.size
    b = _full_convolution(a)
    max_b = b.max()
    # indices attaining the maximum (exact equality is sufficient)
    idx_max = np.flatnonzero(b == max_b)
    num_max = idx_max.size

    # start with the term (#max)·a_i
    g = num_max * a.copy()

    # add Σ_{m∈argmax} a_{m‑i}
    for m in idx_max:
        i_start = max(0, m - (n - 1))
        i_end = min(n - 1, m)
        i_arr = np.arange(i_start, i_end + 1, dtype=int)
        j_arr = m - i_arr
        g[i_arr] += a[j_arr]

    return g, max_b


def _subgradient_descent(a0: np.ndarray, target_sum: float,
                         max_iter: int, deadline: float, rng) -> np.ndarray:
    """
    Projected sub‑gradient descent on max‑convolution.
    Step size = η₀ / sqrt(t) with a direction normalised to unit length.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    # base step magnitude (empirical)
    eta0 = target_sum * 0.6

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        g, cur_score = _hardmax_subgradient(a)

        # keep the best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # normalised descent direction
        norm = np.linalg.norm(g) + 1e-12
        direction = g / norm
        step = eta0 / math.sqrt(t)

        a -= step * direction
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.size)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (biased version)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The destination is biased towards indices that contribute little to the
    current worst convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9997                    # slower cooling for finer search

    while time.time() < deadline:
        # pick one of the currently worst convolution positions
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: bias towards small counterpart a[M‑j]
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    # ------------------------------------------------------------------
