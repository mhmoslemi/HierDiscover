"""
Propose a candidate step‐function with a very low benchmark score.

The algorithm consists of five stages:
1. **Warm‑start** – a cheap two‑parameter grid search on a reduced size,
   then linear interpolation to the target length.
2. **Multiplicative (exponential) descent** – mirror‑descent on a soft‑max
   surrogate of the maximal convolution value.
3. **Projected gradient descent with restarts** – additive updates with
   momentum and occasional jitter.
4. **Deterministic mass‑transfer refinement** – move a small fraction of the
   mass from indices that contribute most to the current maximal convolution
   entry to indices that contribute least.
5. **Sub‑gradient descent on the true max‑convolution** – a classic
   sub‑gradient method for the non‑smooth objective (max a*a) with
   projection onto the ℓ₁‑simplex.

All stages respect the non‑negativity and ℓ₁‑norm constraints (the sum is
fixed to √(2 n)), so the returned list can be evaluated directly by
``evaluate_sequence``.  The routine adapts the allocated time budget to the
available time (default 1000 s) and keeps the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    conv = _conv_fft(a)
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). For Σa = √(2n) this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalized to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

# 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Two‑parameter family: a_i = (i+1)^(-p) * exp(-λ·i / small_n).
    Returns the best sequence found (or None) and its score.
    """
    # deterministic grid
    p_vals = np.linspace(0.2, 1.2, 19)          # finer grid
    lam_vals = np.linspace(0.0, 6.0, 25)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    # random refinement
    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Rand best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2 – exponential (multiplicative) descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Soft‑max approximation of max(convolution). Returns gradient w.r.t. a
    and the full convolution (for diagnostics). Uses FFT for speed.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]   # full conv

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                               # soft‑max weights

    # gradient = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]               # length n

    return grad, b


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with exponential updates using the soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())            # enforce exact sum
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)              # multiplicative update
        a = a * (target_sum / a.sum())        # renormalise

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# 3 – projected gradient descent with restarts

def _softmax_grad_and_conv(a: np.ndarray, beta: float):
    """Convenient wrapper returning gradient and full conv."""
    g, conv = _softmax_gradient(a, beta)
    return g, conv


def _annealed_gradient_descent(a0: np.ndarray,
                               target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate, with momentum,
    temperature schedule and jitter.
    """
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_grad_and_conv(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short GD runs with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start, target_sum,
                                            time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from indices that contribute most to the
    current maximal convolution entry to indices that contribute least.
    Accept improvements only.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        # Sensitivities (product contributions) for this entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = a[k_max - idx]                      # partner values
        weight = a[idx] * sens                     # contribution a_i * a_{k-i}
        total_w = weight.sum()

        if total_w > 0:
            donor = rng.choice(n, p=weight / total_w)
        else:
            donor = rng.integers(n)

        low_thr = np.percentile(weight, 10)
        low_idx = np.where(weight <= low_thr)[0]
        receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

        if donor == receiver:
            continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # reject move
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # occasional renorm to avoid drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method for minimizing max(a*a) under the simplex.
    Uses step = f / ||g||² (with a safety factor) and projects back onto the simplex.
    """
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))
        f_val = float(conv[k_max])          # current max convolution value

        # gradient of the max‑entry (sub‑gradient)
        n = a.size
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partners = k_max - idx
        g[idx] = 2.0 * a[partners]          # factor 2 from both sides

        norm_sq = np.dot(g, g)
        if norm_sq == 0.0:
            break

        # Classic diminishing step: α = f / ||g||², scaled down a bit
        alpha = (f_val / norm_sq) * 0.9

        a_new = a - alpha * g
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    return best_a, best_val
