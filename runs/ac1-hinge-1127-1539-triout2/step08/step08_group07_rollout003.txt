"""
Multi‑phase optimizer for the step‑function benchmark.

The algorithm improves on the previous baseline by

1. **Very fast parametric pre‑search** – a grid over power‑law
   exponents and an exponential tail produces a handful of strong
   starting shapes.

2. **Momentum projected gradient descent** on a soft‑max approximation
   of the maximal convolution entry (β is annealed up to 1e6).  The
   optimiser uses Nesterov‑style momentum and a simple adaptive step
   size.

3. **Deterministic mass‑transfer** that directly attacks the current
   maximal convolution index (the same idea as before but with a
   longer temperature schedule).

4. **Global random mass‑transfer** – repeated moves of a small amount
   of mass from the biggest entries to the smallest ones, accepted
   only if they improve the score (or with a tiny Metropolis probability).

5. **Final polishing** – one more long gradient‑descent run.

All phases respect the global time budget (default 1000 s).  The
resulting sequence is trimmed of trailing zeros and returned as a plain
Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for the quick parametric search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and scale to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # avoid overflow – subtract max first
    exps = np.exp(beta * (b - max_b))
    # soft‑max weights
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent with annealed soft‑max temperature
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    - a0          : initial vector (will be projected to the simplex)
    - deadline    : absolute time when the run must stop
    - target_sum  : Σa = target_sum after each projection
    - max_beta    : final soft‑max temperature
    - step_factor : base step size is target_sum * step_factor
    - gamma       : momentum decay
    Returns (best_vector, best_score).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    step_base = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        # gradient of soft‑max approximation
        g, _ = _gradient_softmax(a, beta)

        # Nesterov‑style momentum update
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # slowly increase β
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiple short GD runs with random jitter; keep the best result.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start,
                                            time.time() + run_budget,
                                            target_sum,
                                            max_beta=2e5,
                                            step_factor=0.8,
                                            gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly reduce the current maximal convolution entry.
    Uses a Metropolis rule with an exponentially decaying temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bound the transfer amount
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution & score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to escape large plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing moves that shift a small amount of mass
    from the largest entries to the smallest ones.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    # parameters (percentages of the vector)
    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        # pick donor from the top fraction
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)  # descending
        donor = idx_desc[rng.integers(k_top)]

        # pick receiver from the bottom fraction
        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)   # ascending
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        # move at most 10 % of the donor’s mass
        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # optional tiny Metropolis acceptance – we keep only improving moves

        if cur_score > best_score:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scaling invariant)

    # ---------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search
    # ---------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # (a) pure power‑law
    for p in np.linspace(0.35, 0.70, 15):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # (b) power‑law + exponential tail
    for p in np.linspace(0.35, 0.70, 8):
        for q in np.linspace(0.0, 5.0, 8):
            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # (c) previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # (d) jittered variants of everything above
    for seq in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # keep only a manageable number for the cheap scoring pass
    max_initial_evals = min(180, len(candidates))
    sample_idx = rng.choice(len(candidates), size=max_initial_evals, replace=False)

    best_seq = None
    best_score = np.inf

    for i in sample_idx:
        seq = candidates[i]
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Projected gradient descent with restarts
    # ---------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        # small jittered start from the current best
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer + Global random moves
    # ---------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # deterministic refinement
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-6,
                                                      low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        # global random mass‑transfer
        a_ref2, val_ref2 = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_ref2 < best_score:
            best_score = val_ref2
            best_seq = a_ref2.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        # stop early if we are very close to the deadline
        if time.time() > deadline - 2.0:
            break

    # ---------------------------------------------------------------
    # Phase C – Final polishing gradient descent
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(best_seq,
                                                    deadline,
                                                    target_sum,
                                                    max_beta=1e6,
                                                    step_factor=0.6,
                                                    gamma=0.99)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing zeros, return as Python list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
