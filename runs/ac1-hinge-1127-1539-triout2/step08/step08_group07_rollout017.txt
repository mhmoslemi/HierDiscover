def _subgradient_descent(a0, target_sum, deadline):
    a = _project_to_simplex(a0, target_sum)
    best = a.copy()
    best_val = _score(a)
    r = np.random.default_rng()
    it = 0
    # step parameters
    init_lr = target_sum * 0.5
    # we'll schedule decreasing lr: lr = init_lr / (1 + it*0.001)
    while time.time() < deadline:
        # compute convolution
        conv = _conv_fft(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        # compute gradient
        n = a.shape[0]
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx = np.arange(lo, hi+1)
        j = max_idx - idx
        grad = np.zeros_like(a)
        grad[idx] += a[j]
        grad[j] += a[idx]
        # maybe include small random jitter
        # step size
        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = init_lr / (norm * (1 + it*0.001))
            a = a - lr * grad
        # keep feasible
        a = _project_to_simplex(a, target_sum)
        # evaluate score
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SGD it={it} score={best_val:.6f} maxIdx={max_idx}")
        # occasional jitter
        if it % 500 == 0 and it > 0:
            jitter = r.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)
        it += 1
    return best, best_val

"""
propose_candidate.py

Optimiser for the step‑function benchmark.

The objective (after fixing the ℓ¹‐norm to √(2·n)) reduces to
    score = max_k (a * a)[k] .
The code below searches for a low‐maximum convolution using three
stages:

* Stage A – fast sub‑gradient descent on the exact maximum entry.
  The gradient of the current maximal convolution term is computed
  analytically (no soft‑max).  A simplex projection keeps the ℓ¹‑norm
  fixed.  The cheap per‑iteration cost (one FFT + O(n) work) allows
  thousands of updates within the time budget.

* Stage B – deterministic mass‑transfer moves that directly attack the
  dominant convolution index (the technique from the reference
  implementation).

* Stage C – final polishing with the soft‑max projected gradient descent
  of the reference code (high β).  This sharpens the result already
  obtained by the earlier phases.

All helper functions are defined at top level, there are no closures,
and the routine respects the 1000 s runtime budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a)/(∑a)² (reduces to max convolution)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of initial sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) scaled to the desired ℓ¹ norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Geometric decay a_i = lam**i, scaled to target_sum."""
    raw = lam ** np.arange(n, dtype=np.float64)
    raw *= target_sum / raw.sum()
    return raw


def _gaussian_sequence(n: int, sigma_factor: float, target_sum: float) -> np.ndarray:
    """Half‑Gaussian decreasing from the left, scaled to target_sum."""
    x = np.arange(n, dtype=np.float64)
    sigma = sigma_factor * n / 6.0
    raw = np.exp(- (x ** 2) / (2.0 * sigma ** 2))
    raw *= target_sum / raw.sum()
    return raw


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact maximum convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(init_seq: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast descent using the exact gradient of the current maximal
    convolution entry (sub‑gradient).  The iterate stays on the ℓ¹‑simplex.
    """
    a = _project_to_simplex(init_seq, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]

    # hyper‑parameters
    gamma = 0.95                     # momentum decay
    step_factor = target_sum * 0.2   # controls absolute step size
    it = 0

    momentum = np.zeros_like(a)

    while time.time() < deadline:
        # ----- current maximal convolution entry -----
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # ----- sub‑gradient of the active quadratic term -----
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - idx

        grad = np.zeros_like(a)
        # each pair contributes to both coordinates (i and j)
        grad[idx] += a[j]
        grad[j] += a[idx]

        # ---- momentum update ----
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_factor / norm
            a -= lr * momentum

        # ---- keep feasible ----
        a = _project_to_simplex(a, target_sum)

        # ---- evaluate ----
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] subgrad it={it:6d} k={k:5d} score={best_val:.6f}")

        # occasional random jitter to escape shallow plateaus
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Soft‑max based projected gradient descent (reference implementation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full linear convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 1e5,
                                   step_factor: float = 0.7,
                                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (reference implementation)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly target the current maximal
    convolution entry.  Moves are accepted with a Metropolis rule where
    the temperature decays exponentially with time.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine adapts problem size to the time budget, builds a diverse
    pool of initial candidates, runs a fast sub‑gradient phase,
    refines the result with deterministic mass‑transfer moves and finally
    polishes it with the high‑β soft‑max gradient descent from the
    reference implementation.
    """
    start = time.time()
    deadline = start + budget_s - 0.5     # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more freedom but higher cost)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 30000
    elif budget_s >= 700:
        n = 24000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a diverse set of initial sequences
    # --------------------------------------------------------------
    init_seqs = []

    # power‑law families
    for p in (0.45, 0.5, 0.55, 0.6, 0.65):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # geometric (exponential) families
    for lam in (0.9998, 0.9995, 0.9992, 0.998):
        init_seqs.append(_exponential_sequence(n, lam, target_sum))

    # half‑Gaussian families
    for sf in (0.8, 1.0, 1.2):
        init_seqs.append(_gaussian_sequence(n, sf, target_sum))

    # resized previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies to increase diversity
    rng = np.random.default_rng()
    jittered = []
    for seq in init_seqs:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    init_seqs.extend(jittered)

    # --------------------------------------------------------------
    # Phase A – fast sub‑gradient descent (majority of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.55

    best_seq = None
    best_score = np.inf
    per_candidate_budget = 25.0  # seconds per candidate (fallback)

    for init in init_seqs:
        if time.time() >= phase_a_deadline:
            break
        cand_deadline = min(phase_a_deadline,
                            time.time() + per_candidate_budget)
        a_opt, val_opt = _subgradient_descent(init, target_sum, cand_deadline)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after sub‑grad = {best_score:.6f}")

    # safety fallback
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase B – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-4,
                                                      low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After mass transfer best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – final high‑β polishing (short window)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_ext(best_seq,
                                                        deadline,
                                                        target_sum,
                                                        max_beta=3e5,
                                                        step_factor=0.5,
                                                        gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: drop trailing zeros and return plain Python list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (the harness may ignore the print)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
