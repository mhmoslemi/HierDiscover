        # scale g to sum S

        # mixing factor t (may adapt)
        t = 0.3
        a_new = (1 - t) * a + t * g_scaled
        # ensure nonnegative (should be)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

"""
Advanced search for a non‑negative step‑function minimising the
self‑convolution peak.

Main ideas
----------
1. **Broad exploration** – generate diverse families (uniform,
   geometric, power‑law, beta, sinusoidal, logistic and a new sparse
   family) and keep the K best vectors.

2. **Refinement** – for every elite vector run the three‑stage optimiser
   from the previous solution (Adam → sub‑gradient → tiny‑swap).

3. **Upsampling** – linearly stretch the best vector to larger lengths
   and give each a short Adam burst.

4. **Linear‑programming boost** – after the local search solve a small
   LP that finds a direction `g` with a larger ℓ₁‑mass while keeping the
   convolution bounded by the current maximum `M`.  Mixing the current
   solution with the scaled `g` provably never increases `M`.

5. **Final tidy‑up** – a last swap‑climb and evaluation.

The code respects the simplex constraint (∑a = √(2·n)) by Euclidean
projection after every update.  All heavy work is done with NumPy/FFT;
the LP uses only the most critical convolution constraints (≈200 per
iteration) so that it stays fast even for n≈15000.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Peak of the self‑convolution (internal metric)
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    """Return max_i (a*a)_i."""
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Smooth soft‑max weights for the Adam phase
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return a probability vector proportional to exp(alpha·conv)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear interpolation (upsampling)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Candidate generator – several analytic families + a sparse one
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator,
                   base: np.ndarray | None) -> np.ndarray:
    """Create a random non‑negative vector of length n."""
    fam = rng.integers(0, 8)        # 0…7 (7 = new sparse family)
    if fam == 0 and base is not None:
        a = _upsample_seq(base, n)
    elif fam == 1:
        a = rng.random(n)                               # uniform
    elif fam == 2:
        rate = rng.uniform(0.92, 0.9995)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) * 0.5                    # symmetric geometric
    elif fam == 3:
        p = rng.uniform(0.2, 4.5)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p                # symmetric power‑law
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)                 # beta‑like
    elif fam == 5:
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p   # sinusoidal power
    elif fam == 6:
        scale = rng.uniform(0.02, 0.15) * n
        mid = (n - 1) / 2.0
        xs = np.arange(n, dtype=float)
        left = 1.0 / (1.0 + np.exp((xs - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - xs) / scale))
        a = left + right                                 # logistic bump
    else:   # fam == 7  – sparse random support
        target = max(2, int(np.sqrt(2.0 * n) *
                           rng.uniform(0.8, 1.2)))     # around √(2n)
        idx = rng.choice(n, size=target, replace=False)
        weights = rng.random(target)
        weights /= weights.sum()
        a = np.zeros(n, dtype=float)
        a[idx] = weights
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.05
    max_alpha = 800.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        # gradient of the soft‑max (conv weighted by w)
        # compute via convolution of a with reversed w
        w_rev = w[::-1]
        conv_len = 3 * n - 2
        fft_len = 1 << (conv_len - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        Fw = np.fft.rfft(w_rev, n=fft_len)
        full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = 0.5 * (a + a[::-1])

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient descent
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the selected entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = 0.5 * (a + a[::-1])
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.18:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        if it % 9000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb (local refinement)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = 0.5 * (cand + cand[::-1])
        cand_score = _peak_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based improvement: push mass into directions that respect the
# current maximal convolution bound M.
# ----------------------------------------------------------------------
def _lp_improve(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 10) -> tuple[np.ndarray, float]:
    """
    Starting from a0, each iteration solves a small LP:
        maximise   sum(g)
        subject to  f * g <= M   (only the K largest k‑indices)
                    g >= 0
    The solution g is scaled to the proper ℓ₁‑mass and mixed with
    the current vector (mixing factor t = 0.3).  The peak never increases.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a)
        M = conv.max()

        # pick the most critical constraints (top K values)
        K = min(200, conv.size)
        idx = np.argpartition(-conv, K - 1)[:K]

        # build dense constraint matrix A (K × n)
        A = np.zeros((len(idx), n), dtype=float)
        for row, k in enumerate(idx):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            A[row, i_range] = a[k - i_range]

        # linear program: maximise sum(g)  ==  minimise -∑g
        c = -np.ones(n, dtype=float)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=bounds, method='highs')
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # scale g to the required ℓ₁‑mass
        g = g * (S / sum_g)

        # mixing step (t can be tuned; 0.3 works well)
        t = 0.3
        a_new = (1 - t) * a + t * g
        # keep the exact simplex (tiny numerical drift)
        a_new = _project_simplex(a_new, S)

        new_score = _peak_convolution(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm‑start from a previously found sequence if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – cheap exploration, keep the K elite vectors
    # ------------------------------------------------------------------
    K = 8                               # number of elites
    elite = []                          # list of (score, vector)
    attempts = 0
    phase0_deadline = start_time + 0.08 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1

        # choose a length
        if base_seq is not None and rng.random() < 0.6:
            scale = rng.uniform(0.8, 1.2)
            n = max(600, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(2000, 12001))

        a = _gen_candidate(n, rng, base_seq)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        tot = a.sum()
        if tot == 0.0:
            a = np.full(n, 1.0)
            tot = a.sum()
        a = a / tot * S

        # occasional forced symmetry (helps the landscape)
        if rng.random() < 0.45:
            a = 0.5 * (a + a[::-1])

        cur_score = _peak_convolution(a)

        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    # fallback (should never happen)
    if not elite:
        n = 5000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        elite.append((_peak_convolution(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – per‑elite refinement (Adam → sub‑grad → local)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    time_per_elite = remaining / len(elite)

    best_overall_seq = None
    best_overall_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        # allocate sub‑budgets for the three stages
        ada_dead = now + 0.55 * time_per_elite
        sub_dead = ada_dead + 0.30 * time_per_elite
        loc_dead = sub_dead + 0.15 * time_per_elite

        a_adam, sc_adam = _optimize_adam(vec, ada_dead, rng, enforce_sym=False)
        a_sub, sc_sub = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=False)
        a_loc, sc_loc = _local_swap_search(a_sub, loc_dead, rng, enforce_sym=False)

        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[refine {idx}] improved to {best_overall_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – upsample the current best to larger lengths
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 15000
        up_factors = [1.2, 1.5, 2.0]
        for fac in up_factors:
            new_n = int(best_overall_seq.shape[0] * fac)
            if new_n > max_len or new_n < 2:
                continue
            a_up = _upsample_seq(best_overall_seq, new_n)

            S_new = np.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * S_new

            now = time.time()
            if now >= deadline:
                break
            short_dead = min(deadline, now + 0.07 * (deadline - now))
            a_up_opt, sc_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=False)

            if sc_up < best_overall_score:
                best_overall_score = sc_up
                best_overall_seq = a_up_opt.copy()
                print(f"[upsample] factor={fac:.2f} new_n={new_n} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final tiny local search
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_final, sc_final = _local_swap_search(best_overall_seq, deadline, rng, enforce_sym=False)
        if sc_final < best_overall_score:
            best_overall_score = sc_final
            best_overall_seq = a_final.copy()
            print(f"[final‑local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – LP boost (uses any remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_lp, sc_lp = _lp_improve(best_overall_seq, deadline, rng)
        if sc_lp < best_overall_score:
            best_overall_score = sc_lp
            best_overall_seq = a_lp.copy()
            print(f"[lp‑improve] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (store globally for possible warm‑restarts)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
