def _init_bimodal(n, rng):
    """Generate two Gaussian-like bumps at random positions."""
    raw = np.full(n, rng.uniform(0.0, 0.05))  # low background
    # parameters for first bump
    cent1 = rng.integers(0, n)
    width1 = rng.integers(max(5, n//200), max(10, n//100))

"""
Improved step‑function optimiser.

Algorithm
---------
1. **Diverse random seeding** – generate many candidates from a
   collection of parametric families (Dirichlet, Gamma, power‑law,
   exponential, sparse bumps and a new bi‑modal bump).  Each candidate
   is normalised to ℓ₁‑norm √(2 n) so that the internal objective is the
   peak of the self‑convolution.

2. **Multi‑scale Adam optimisation** – the best *k* candidates are
   iteratively refined on a schedule of growing lengths.  Adam works on
   the simplex by a log‑softmax parametrisation.  A soft‑max
   approximation of the `max` is used initially; after the temperature
   τ drops below 1e‑5 the exact max‑gradient is employed (mask = 1 on the
   peak indices).  Several hyper‑parameter settings are tried per
   scale.

3. **Final simulated‑annealing fine‑tuning** – a cheap mass‑transfer
   walk on the simplex with Metropolis acceptance removes any remaining
   shallow local minima.

The best sequence is stored globally as ``height_sequence_1`` and
returned as a plain list.  All helper functions are top‑level, no
closures or lambdas are used, and the routine respects the 1000 s

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector using FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Convolution gradient (soft‑max or exact)
# ----------------------------------------------------------------------
def _convolution_max_and_gradient(a: np.ndarray, tau: float):
    """
    Return (max_b, grad) where max_b = max(a*a) and grad = ∂ max_b / ∂a.
    If tau>0 a τ‑soft‑max is used, otherwise the exact sub‑gradient.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-12:
        mask_bool = (b == max_b)
        mask = mask_bool.astype(float)
        cnt = mask.sum()
        if cnt > 0:
            mask /= cnt
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return max_b, g

    ‖a‖₁ = √(2 n).  Returns the best vector encountered and its hard
    peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                        # target ℓ₁‑norm

    # initialise logits from the feasible start vector (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                             # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initial temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient (soft‑max or exact) ----
        peak, g = _convolution_max_and_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S                     # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                     # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- annealing schedules ----
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

# Random families for the exploration phase

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones (will be re‑scaled)."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # background mixing weight
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))    # low background
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two Gaussian‑like bumps placed at random locations on a weak background."""
    raw = np.full(n, rng.uniform(0.0, 0.05))   # tiny background
    for _ in range(2):
        cent = rng.integers(0, n)
        width = rng.integers(max(5, n // 200), max(10, n // 100))
        height = rng.uniform(0.5, 3.0)
        lo = max(0, cent - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.25 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.full(n, 1.0)
        a = a * (S / a.sum())
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'bimodal'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    else:   # bimodal
        raw = _init_bimodal(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (cheap, used only if time left)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with Metropolis acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score        # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]   # move up to 30 % of mass at i
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # re‑normalise (keeps ℓ₁‑norm exactly S)
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm follows random exploration → multi‑scale Adam refinement
    → optional simulated‑annealing fine‑tuning.
    The final construction is stored in the global ``height_sequence_1`` and
    also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # Warm‑start from an existing global solution if available
    # --------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # --------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 25 % of budget)
    # --------------------------------------------------------------
    explore_deadline = end_time - 350.0       # reserve the rest for refinement
    best_seq = None
    best_score = np.inf
    best_n = None
    top_k = 8
    top_candidates = []                      # list of (score, seq)

    attempt = 0
    while time.time() < explore_deadline:
        attempt += 1
        # Choose length: majority in the 2k‑5k range, some larger
        if rng.random() < 0.65:
            n = int(rng.integers(1800, 5000))
        else:
            n = int(rng.integers(5000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain top‑k list
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # Safeguard – ensure at least one candidate exists
    # --------------------------------------------------------------
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # --------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement
    # --------------------------------------------------------------
    max_len = min(40000, best_n * 8)           # upper length bound
    lengths = [best_n]
    # build a gentle progression of lengths
    for factor in (1.5, 2.0, 3.0, 5.0):
        cand = int(best_n * factor)
        if best_n < cand < max_len:
            lengths.append(cand)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    # iterate over shortlisted candidates (most promising first)
    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_n = len(cur_seq)

        for target_n in lengths:
            remaining = end_time - time.time()
            if remaining < 40.0:
                break

            # interpolate and renormalise to the new length
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining budget (but not too long)
            slice_budget = min(remaining * 0.4, 200.0)

            # run a few Adam instances with randomised hyper‑parameters
            runs = rng.integers(1, 3) + 1   # 1‑3 runs per scale
            best_local = a_scaled
            best_local_peak = _max_convolution(a_scaled)

            for _ in range(runs):
                lr0 = rng.choice([0.40, 0.45, 0.50])
                tau_decay = rng.choice([0.9992, 0.9994, 0.9996])
                refined, peak = _adam_optimize(best_local,
                                               time.time() + slice_budget,
                                               rng,
                                               lr0=lr0,
                                               tau_decay=tau_decay,
                                               lr_decay=0.9999,
                                               jitter_every=400)
                if peak < best_local_peak:
                    best_local = refined
                    best_local_peak = peak

            # update global best if improved
            if best_local_peak < best_score:
                best_score = best_local_peak
                best_seq = best_local.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            # use refined vector as seed for the next larger scale
            cur_seq = best_local
            cur_n = target_n

        # stop early if time is exhausted
        if time.time() >= end_time - 30.0:
            break

    # --------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning if time permits
    # --------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 30.0:
        final_seq, final_peak = _anneal(best_seq,
                                        time.time() + remaining,
                                        rng)
        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_seq.copy()
            print(f"[anneal] final annealed score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity check via the external evaluator
    # --------------------------------------------------------------
    final_list = best_seq.tolist()
    external = evaluate_sequence(final_list)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_list)}")

    globals()['height_sequence_1'] = final_list
    return final_list
