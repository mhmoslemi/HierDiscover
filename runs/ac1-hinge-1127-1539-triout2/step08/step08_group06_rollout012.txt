def _mutate_swap(a, rng, step):

    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    # hyper-parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    alpha = 0.01
    alpha_mul = 1.07
    max_alpha = 2000
    best_a = a.copy()
    best_score = _max_conv(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # compute weights via smooth max
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # project onto simplex
        a = _project_simplex(a, S)
        # schedule update
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995
    return best_a, best_score

"""
Hybrid evolutionary + gradient search for a step‑function with a minimal
self‑convolution peak.

Approach
--------
* Phase 0 – random exploration of many analytic families (exponential,
  power‑law, linear, quadratic, piecewise‑constant, logistic, uniform).
  Keep a small elite of the best vectors.
* Phase 1 – evolve the elite using a simple (mass‑swap) mutation with a
  decreasing step size, maintaining the simplex constraint Σa_i = √(2·n).
  Occasionally refine the current best vector with a short Adam run on the
  smooth log‑sum‑exp approximation of the max‑convolution.
* Phase 2 – up‑sample the best vector to larger lengths (≈ 1.2×, 1.5×, 2×)
  and give each a brief Adam burst.
* Phase 3 – final local hill‑climb (tiny random swaps) using any remaining
  time budget.

All updates respect non‑negativity and the L₁‑normalisation, and the
convolution is evaluated via FFT for speed.  The returned sequence is stored
globally as ``height_sequence_1`` for possible warm‑starts in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full (linear) self‑convolution via FFT
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length = 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Max entry of the self‑convolution (the internal objective)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) probabilities over convolution entries
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Probability vector ∝ exp(alpha·conv)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of Σ w_i·(a*a)_i w.r.t. a (via convolution with reversed weights)
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇_a Σ_i w_i (a*a)_i ."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # mapping back to gradient indices
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Linear interpolation to a new length (preserves shape)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to length new_n."""
    if base.size == new_n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Generate a random candidate (many analytic families, optionally up‑sample)
# ----------------------------------------------------------------------
def _random_candidate(n: int, rng: np.random.Generator,
                      base: np.ndarray | None = None) -> np.ndarray:
    """
    Produce a non‑negative vector of length n.
    Families include random, exponential left/right, power‑law left/right,
    linear/quadratic decay, piecewise constant, and asymmetric logistic.
    """
    # Occasionally up‑sample an existing good vector
    if base is not None and rng.random() < 0.3:
        return _upsample_seq(base, n)

    fam = rng.integers(1, 9)   # 1…8 → distinct families
    i = np.arange(n, dtype=float)

    if fam == 1:
        # pure uniform random
        a = rng.random(n)

    elif fam == 2:
        # asymmetric exponential decay from both ends
        r1 = rng.uniform(0.93, 0.9999)
        r2 = rng.uniform(0.93, 0.9999)
        A1 = rng.random()
        A2 = rng.random()
        a = A1 * (r1 ** i) + A2 * (r2 ** (n - 1 - i))

    elif fam == 3:
        # asymmetric power‑law (i+1)^‑p1 + (n‑i)^‑p2
        p1 = rng.uniform(0.3, 3.5)
        p2 = rng.uniform(0.3, 3.5)
        a = (i + 1) ** (-p1) + (n - i) ** (-p2)

    elif fam == 4:
        # linear decreasing (high → low)
        a = (n - i)

    elif fam == 5:
        # quadratic decreasing (high → low faster)
        a = (n - i) ** 2

    elif fam == 6:
        # piecewise constant: high block of width w, low elsewhere
        w = rng.integers(max(1, int(0.05 * n)), max(2, int(0.35 * n)))
        h_high = rng.random() + 0.1
        h_low = rng.random() * h_high
        a = np.full(n, h_low, dtype=float)
        a[:w] = h_high

    elif fam == 7:
        # asymmetric logistic (different scales left/right)
        mid = (n - 1) / 2.0
        scaleL = rng.uniform(0.02, 0.15) * n
        scaleR = rng.uniform(0.02, 0.15) * n
        left = 1.0 / (1.0 + np.exp((i - mid) / scaleL))
        right = 1.0 / (1.0 + np.exp((mid - i) / scaleR))
        a = left + right

    else:   # fam == 8 (unused but kept for symmetry)
        # sinusoidal power (symmetric)
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (i + 0.5) / n) ** p

    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones(n, dtype=float)
    return a


# ----------------------------------------------------------------------
# Simple mass‑swap mutation (keeps total mass unchanged)
# ----------------------------------------------------------------------
def _mutate_swap(a: np.ndarray, rng: np.random.Generator,
                 max_frac: float = 0.2) -> np.ndarray:
    """Move a random fraction (≤max_frac) of mass from one index to another."""
    n = a.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or a[i] == 0.0:
        return a
    max_move = max_frac * a[i]
    if max_move <= 0.0:
        return a
    delta = rng.random() * max_move
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float,
              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a log‑sum‑exp approximation of the max‑convolution.
    Returns the best vector found before the deadline together with its peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temp schedule for the smooth max
    alpha = 0.02
    alpha_mul = 1.07
    max_alpha = 2000.0

    best_a = a.copy()
    best_score = _max_conv(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights
        w = _smooth_weights(conv, alpha)
        # gradient w.r.t. a
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex (keeps a ≥ 0 and Σa = S)
        a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb (final refinement)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray, deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that repeatedly swaps a small mass chunk between two positions.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _max_conv(a)

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection (mostly redundant but safe)
        cand = _project_simplex(cand, S)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    peak of its self‑convolution.  The best sequence found before the
    time limit is returned (as a Python list) and stored globally as
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we always finish before the external timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Phase 0 – cheap random exploration, keep an elite pool
    # ------------------------------------------------------------------
    elite_size = 15
    elite = []                     # list of (score, vector)
    attempt = 0
    phase0_dead = start_time + 0.12 * budget_s
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    while time.time() < phase0_dead:
        attempt += 1
        # pick a length in a reasonable range
        n = int(rng.integers(4000, 13001))
        a = _random_candidate(n, rng, base_seq)
        # put onto the simplex Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        a = a / a.sum() * S
        score = _max_conv(a)
        if len(elite) < elite_size or score < elite[-1][0]:
            elite.append((score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > elite_size:
                elite.pop()
            print(f"[explore] attempt={attempt} n={n} score={score:.6f}")

    if not elite:                     # fallback (should never happen)
        n = 8000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        elite.append((_max_conv(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – evolutionary loop (swap mutation + occasional Adam)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:   # keep ~30 s for up‑sample + final
        # probability to pick a better elite more often
        scores = np.array([s for s, _ in elite])
        probs = 1.0 / (scores + 1e-12)
        probs /= probs.sum()
        idx = rng.choice(len(elite), p=probs)
        parent = elite[idx][1]

        # mutation step size shrinks with time
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.2 * (1.0 - elapsed / total) + 0.01

        child = _mutate_swap(parent, rng, max_frac=max_frac)

        # occasional multiplicative noise (helps escape plateaus)
        if rng.random() < 0.25:
            noise = rng.normal(0.0, 0.02, size=child.shape)
            child = child * np.exp(noise)

        # re‑project onto the simplex
        n = child.shape[0]
        child = _project_simplex(child, np.sqrt(2.0 * n))

        child_score = _max_conv(child)
        if child_score < elite[-1][0]:
            elite.append((child_score, child.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:elite_size]
            print(f"[evolve] new best={child_score:.6f} n={n}")

        # every now and then sharpen the current global best with Adam
        if rng.random() < 0.015:
            best_vec = elite[0][1].copy()
            sub_dead = min(deadline, time.time() + 0.8)   # short burst
            refined_vec, refined_score = _adam_opt(best_vec, sub_dead, rng)
            if refined_score < elite[0][0]:
                elite[0] = (refined_score, refined_vec.copy())
                elite.sort(key=lambda x: x[0])
                print(f"[adam] elite improved to {refined_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample the current best to larger lengths
    # ------------------------------------------------------------------
    best_vec, best_score = elite[0][1].copy(), elite[0][0]
    max_len = 15000
    for factor in (1.2, 1.5, 2.0):
        new_n = int(best_vec.shape[0] * factor)
        if new_n > max_len or new_n < 4:
            continue
        up = _upsample_seq(best_vec, new_n)
        up = up / up.sum() * np.sqrt(2.0 * new_n)

        sub_dead = min(deadline, time.time() + 1.2)
        up_opt, up_score = _adam_opt(up, sub_dead, rng)
        if up_score < best_score:
            best_score = up_score
            best_vec = up_opt.copy()
            print(f"[upsample] factor={factor:.2f} new_n={new_n} improved to {up_score:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final tiny random‑swap hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_vec, final_score = _local_swap_search(best_vec, deadline, rng)
        if final_score < best_score:
            best_score = final_score
            best_vec = final_vec.copy()
            print(f"[final‑local] improved to {final_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (store globally for possible warm‑restarts)
    # ------------------------------------------------------------------
    final_seq = best_vec.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempt}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
