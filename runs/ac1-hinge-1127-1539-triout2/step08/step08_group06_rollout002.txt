"""
Propose a step‐function (non‑negative vector) that minimises the maximum
entry of its self‑convolution, i.e. the value returned by
`evaluate_sequence`.  The algorithm consists of

1. **Extensive random exploration** – many analytic families (U‑shaped,
   exponential, power‑law, beta, logistic, mixtures) are sampled with a
   broad range of parameters and lengths.  The best K vectors (lowest
   convolution peak) are kept as elites.

2. **Refinement of the elites** – each elite is optimised three times:
   * Adam on a log‑sum‑exp (smooth‐max) approximation,
   * Projected sub‑gradient descent on the true max,
   * Tiny random‑swap hill‑climb.
   Optionally a brief simulated‑annealing phase is added.

3. **Upsampling** – the current best vector is linearly up‑sampled to
   larger lengths (up to ≈18000).  Each up‑sample receives a short Adam
   burst and a final local search.

4. **Final local search** – the remaining time budget is spent on
   random swaps.

All updates respect the simplex constraint
    Σ a_i = √(2·n) , a_i ≥ 0
by Euclidean projection onto the simplex after every move.

The function returns the best sequence found before the time limit and
stores it globally as `height_sequence_1` for possible warm‑starts.
"""

import time

import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length = 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Peak of the self‑convolution (the internal objective)
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Mirror symmetry helper (optional)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear interpolation to a new length (preserves shape)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Candidate families – return a raw non‑negative vector (not normalised)
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    fam = rng.integers(0, 13)
    idx = np.arange(n)
    mid = (n - 1) / 2.0
    d = np.minimum(idx, n - 1 - idx)               # distance to nearest edge
    x = idx / (n - 1)                               # normalised coordinate

    if fam == 0 and base is not None:
        # Upsample previous best
        a = _upsample_seq(base, n)

    elif fam == 1:
        # Pure uniform random
        a = rng.random(n)

    elif fam == 2:
        # Symmetric geometric decay from the edges (U‑shape)
        rate = rng.uniform(0.5, 0.999)               # smaller rates give stronger U‑shape
        a = rate ** d

    elif fam == 3:
        # Symmetric exponential decay from the edges
        beta = rng.uniform(0.0005, 0.02) * n          # scale with length
        a = np.exp(-beta * d)

    elif fam == 4:
        # Polynomial growth from the edges: a_i ∝ (d+1)^p
        p = rng.uniform(1.0, 5.0)
        a = (d + 1) ** p

    elif fam == 5:
        # Beta‑like shape (x^p1 * (1‑x)^p2) – can produce both U‑shapes and bumps
        p1 = rng.uniform(0.05, 5.0)
        p2 = rng.uniform(0.05, 5.0)
        a = (x ** p1) * ((1 - x) ** p2)

    elif fam == 6:
        # Logistic “U‑shape” – high on the edges, low in the centre
        scale = rng.uniform(0.02, 0.15) * n
        left = 1.0 / (1.0 + np.exp((idx - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - idx) / scale))
        a = left + right

    elif fam == 7:
        # Sinusoidal power (central bump) – for completeness
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (idx + 0.5) / n) ** p

    elif fam == 8:
        # Symmetric power‑law of the distance from centre (inverse U‑shape)
        p = rng.uniform(0.5, 3.0)
        a = (np.abs(idx - mid) + 1) ** p

    elif fam == 9:
        # Mixture of two random families (convex combination)
        a1 = _gen_candidate(n, rng, None)
        a2 = _gen_candidate(n, rng, None)
        w = rng.random()
        a = w * a1 + (1 - w) * a2

    elif fam == 10:
        # Simple linear ramp (low‑high) – may help asymmetric solutions
        a = idx.astype(float)

    elif fam == 11:
        # Quadratic bump in the centre (opposite of U‑shape)
        a = -( (idx - mid) ** 2 ) + (mid ** 2) + 1.0
        a = np.clip(a, 0, None)

    else:
        # Plain constant (will be normalised later)
        a = np.ones(n, dtype=float)

    # Ensure non‑negativity
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Temperature schedule for the soft‑max
    alpha = 0.2
    alpha_mul = 1.08
    max_alpha = 1000.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_shifted = np.exp(scaled - max_scaled)
        w = exp_shifted / exp_shifted.sum()

        # gradient of weighted sum of convolutions
        # (a * w_rev) convolution trick
        w_rev = w[::-1]
        n_conv_len = n + (2 * n - 1) - 1               # 3n‑2
        fft_len = 1 << (n_conv_len - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        Fw = np.fft.rfft(w_rev, n=fft_len)
        full = np.fft.irfft(Fa * Fw, n=fft_len)[:n_conv_len]
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true maximum entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # Sub‑gradient of the selected entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it + 1.0)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random swap
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.22:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        if it % 12000 == 0 and best_score == cur_score:
            # no progress for a while – break early
            break

    return best_a, best_score

def _local_swap_search(a0: np.ndarray,

# Simple simulated‑annealing search (adds a stochastic escape mechanism)
# ----------------------------------------------------------------------
def _annealing_search(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    start = time.time()
    total = deadline - start
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)
    # initial temperature – a fraction of the current peak
    T0 = max(best_score * 0.25, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = max(T0 * (1.0 - frac), 1e-6)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.05
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)
        cand = _project_simplex(cand, S)

        cand_score = _peak_convolution(cand)
        delta_score = cand_score - best_score
        if delta_score < 0 or rng.random() < math.exp(-delta_score / T):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        # occasional aggressive jitter
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.12
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):

    Search for a step‑function with minimal self‑convolution peak.
    Returns a list of floats (the heights).  The best sequence found
    before the time limit is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm start – reuse previous best if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – cheap exploration, keep the K best vectors
    # ------------------------------------------------------------------
    K = 8                       # number of elite candidates kept
    elite = []                  # list of (score, vector)
    attempts = 0
    phase0_deadline = start_time + 0.30 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1

        # Choose a length: favour moderate sizes, but also try extremes
        if base_seq is not None and rng.random() < 0.55:
            scale = rng.uniform(0.85, 1.20)
            n = max(500, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(2000, 18001))

        a = _gen_candidate(n, rng, base_seq)

        # Normalise onto the simplex Σa = √(2·n)
        S = math.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # Randomly enforce symmetry (helps reduce the peak)
        if rng.random() < 0.5:
            a = _symmetrize(a)

        cur_score = _peak_convolution(a)

        # Keep elite list sorted by score (ascending)
        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing survived (should not happen) fall back to a uniform vector
    # ------------------------------------------------------------------
    if not elite:
        n = 6000
        a = rng.random(n)
        a = a / a.sum() * math.sqrt(2.0 * n)
        elite.append((_peak_convolution(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – refine the elite candidates
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # focus on the top few elites (more time per elite)
    elites_to_refine = min(3, len(elite))
    best_overall_seq = None
    best_overall_score = np.inf

    for e_idx in range(elites_to_refine):
        init_score, vec = elite[e_idx]

        # allocate a slice of the remaining budget
        # each elite receives an equal share of the still‑available time
        share_deadline = now + remaining / (elites_to_refine - e_idx)

        # Adam phase (soft‑max)
        adam_dead = now + 0.55 * (share_deadline - now)
        a_adam, sc_adam = _optimize_adam(vec, adam_dead, rng, enforce_sym=False)

        # Sub‑gradient phase (true max)
        sub_dead = adam_dead + 0.30 * (share_deadline - adam_dead)
        a_sub, sc_sub = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=False)

        # Local hill‑climb
        local_dead = sub_dead + 0.10 * (share_deadline - sub_dead)
        a_loc, sc_loc = _local_swap_search(a_sub, local_dead, rng, enforce_sym=False)

        # Simulated annealing boost (optional, uses the rest of the share)
        if time.time() < share_deadline:
            a_anneal, sc_anneal = _annealing_search(a_loc, share_deadline, rng, enforce_sym=False)
        else:
            a_anneal, sc_anneal = a_loc, sc_loc

        # Record the best found so far
        if sc_anneal < best_overall_score:
            best_overall_score = sc_anneal
            best_overall_seq = a_anneal.copy()
            print(f"[refine {e_idx}] improved to {best_overall_score:.6f}")

        # Update timing for the next elite
        now = time.time()
        remaining = deadline - now
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – upsample the current best to larger lengths
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 18000
        up_factors = [1.2, 1.35, 1.5, 1.8, 2.0]
        for fac in up_factors:
            new_n = int(best_overall_seq.shape[0] * fac)
            if new_n > max_len or new_n < 200:
                continue
            a_up = _upsample_seq(best_overall_seq, new_n)

            # Re‑normalise to the new length
            S_new = math.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * S_new

            now = time.time()
            if now >= deadline:
                break
            # short Adam burst (≈ 10 % of the remaining time)
            short_dead = min(deadline, now + 0.10 * (deadline - now))
            a_up_opt, sc_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=False)

            if sc_up < best_overall_score:
                best_overall_score = sc_up
                best_overall_seq = a_up_opt.copy()
                print(f"[upsample] factor={fac:.2f} n={new_n} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final random‑swap local search with any remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline and best_overall_seq is not None:
        a_final, sc_final = _local_swap_search(best_overall_seq, deadline, rng, enforce_sym=False)
        if sc_final < best_overall_score:
            best_overall_score = sc_final
            best_overall_seq = a_final.copy()
            print(f"[final‑local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (store globally for possible warm‑starts)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
