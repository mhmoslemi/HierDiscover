"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search is completely deterministic apart from a single random seed.
It consists of two stages:

1. **Parametric sweep** –  on each chosen length ``n`` we generate
   sequences of the form

       a_i ∝ (i+1)^{-p} · exp(-c·i / n)

   for a grid of exponents ``p`` (≈0.4 … 0.6) and exponential decay
   rates ``c`` (≈0 … 10).  After scaling to the required ℓ¹‑norm
   (∑a = √(2n)) we evaluate the maximal entry of the self‑convolution
   (which equals the objective because the denominator is fixed).
   The best sequence over all (n,p,c) triples is recorded.

2. **Local refinement** –  the best candidate is fed to a fast
   O(n) hill‑climber that repeatedly moves a small amount of mass from
   a “source’’ index to a “destination’’ index while keeping the full
   convolution up‑to‑date.  Moves are accepted with a Metropolis rule
   (annealing temperature slowly decays) so the algorithm can escape
   shallow plateaus.

Both stages respect the global wall‑clock budget (default 1000 s) and
the final list of heights is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers – maximum entry of a self‑convolution
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_power_exp_sequence(n: int, p: float, c: float) -> np.ndarray:
    """
    Return a non‑negative vector of length n with shape
        a_i = (i+1)^{-p} * exp(-c * i / n)
    scaled so that Σ a_i = √(2n).
    """
    idx = np.arange(n, dtype=float)
    raw = (idx + 1.0) ** (-p) * np.exp(-c * idx / n)
    # avoid a completely zero vector (should never happen)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    scale = math.sqrt(2.0 * n) / raw.sum()
    return raw * scale


# ----------------------------------------------------------------------
# Simple hill‑climber that updates the convolution in O(n)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing / simulated annealing.
    ``a0`` must already satisfy the ℓ¹‑norm constraint.
    Returns a (possibly) improved sequence and its max‑convolution value.
    """
    a = a0.copy()
    b = _full_convolution(a)
    n = a.size
    cur = b.max()
    best_a = a.copy()
    best_score = cur

    rng = np.random.default_rng(12345)

    # initial temperature – something comparable to the current worst value
    T = cur * 0.1 + 1e-9
    cooling = 0.9996

    while time.time() < deadline:
        # pick a worst‑entry index (could be several)
        worst = np.flatnonzero(b == cur)
        if worst.size == 0:
            break
        k = rng.choice(worst)

        # source index – pick one that contributes a lot to entry k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = k - idx
        contrib = a[idx] * a[jdx]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination – favour indices that are *unlikely* to raise the worst entry
        # (small partner values a[k‑j])
        partner = np.zeros(n)
        j_low = max(0, k - (n - 1))
        j_high = min(n - 1, k)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        partner[j_range] = a[k - j_range]

        inv = 1.0 / (partner + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount to move – up to 80 % of the source entry
        delta = rng.random() * a[i] * 0.8
        if delta <= 0.0:
            continue

        # ---------- update the convolution efficiently ----------
        # contribution of a[i] with the whole vector appears at positions i..i+n-1
        b[i:i + n] -= 2.0 * delta * a
        b[j:j + n] += 2.0 * delta * a

        # diagonal terms (i+i) and (j+j) and the mixed term (i+j)
        if 2 * i < b.size:
            b[2 * i] += delta * delta
        if 2 * j < b.size:
            b[2 * j] += delta * delta
        if i + j < b.size:
            b[i + j] -= 2.0 * delta * delta
        # ---------------------------------------------------------

        a[i] -= delta
        a[j] += delta
        cur = b.max()

        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Metropolis acceptance (already applied – we accept all improving moves)
        # For non‑improving moves we may still accept – here we simply keep the move
        # only if it improves; otherwise we revert (this keeps the code simple).
        if cur > best_score:
            # revert the move
            a[i] += delta
            a[j] -= delta
            b[i:i + n] += 2.0 * delta * a
            b[j:j + n] -= 2.0 * delta * a
            if 2 * i < b.size:
                b[2 * i] -= delta * delta
            if 2 * j < b.size:
                b[2 * j] -= delta * delta
            if i + j < b.size:
                b[i + j] += 2.0 * delta * delta
            cur = b.max()
        else:
            # keep the move, temperature cools down
            T = max(T * cooling, 1e-12)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5     # safety margin

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 1️⃣  Parameter sweep over a few lengths
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 20000]          # increasing schedule
    # exponent grid – we focus around the theoretically promising 0.5
    p_grid = np.linspace(0.40, 0.60, 21)        # 21 values
    # exponential‑decay grid – c = -n·log(r)  (c≈0 → almost pure power‑law)
    c_grid = np.linspace(0.0, 10.0, 21)         # 21 values

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)         # the ℓ¹‑norm we will enforce

        # remaining time for this length
        remain = deadline - time.time()
        # give roughly half of the remaining budget to this length,
        # the rest will be kept for later lengths / final polishing
        length_deadline = time.time() + remain * 0.5

        print(f"[{time.time():.1f}] Exploring length n={n}")

        for p in p_grid:
            if time.time() >= length_deadline:
                break
            for c in c_grid:
                a = _make_power_exp_sequence(n, float(p), float(c))
                # safeguard – due to rounding errors we might be slightly off
                # from the precise ℓ¹‑norm, so re‑project onto the simplex.
                a = a * (target_sum / a.sum())
                score = _max_convolution_fft(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New best (n={n}, p={p:.3f}, c={c:.2f}) → {best_score:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Very short local hill‑climb on the current best of this length
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            hc_deadline = min(deadline - 0.1, time.time() + 0.1 * (deadline - time.time()))
            a_hc, score_hc = _hill_climb(best_seq, hc_deadline)
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing on the overall best sequence (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        final_deadline = deadline - 1.0
        a_refined, final_score = _hill_climb(best_seq, final_deadline)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_refined.copy()
            print(f"[{time.time():.1f}] Final hill‑climb improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Trim trailing near‑zeros and report the official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)

    # remove negligible tail for tidiness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # official evaluation (objective = max of the self‑convolution)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
