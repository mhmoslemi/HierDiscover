"""
search for a non‑negative step function that minimises the peak of its
self‑convolution (the evaluation function is 2·n·max(conv)/(sum^2)).
The algorithm consists of three stages:

1. **LP mass‑spreading** – starting from a random shape (or the previous
   best) we repeatedly solve a linear program:
        max Σ b_i   s.t. b_i ≥ 0,  (f * b)_k ≤ M   for all k,
   where M = max(f * f).  The solution spreads the mass while keeping the
   convolution peak ≤ M.  Mixing f ← (1‑t)·f + t·b (with decreasing t)
   strictly reduces the peak.

2. **Upsampling** – the obtained shape is linearly interpolated to a larger
   length (≈2×) and renormalised.  Larger sequences allow the peak to be
   reduced further.

3. **Adam refinement + optional annealing** – a projected Adam optimiser
   minimises a soft‑max approximation of the convolution peak on the
   upscaled sequence; a short simulated‑annealing hill‑climb can polish the
   result.

All steps respect the non‑negativity and sum‑normalisation constraints,
and the final sequence is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, diags

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1‑i]."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Soft‑max helpers used by the Adam optimiser
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (soft‑max approximation of max conv)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 400.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.03
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climber (optional polishing)
# ----------------------------------------------------------------------
def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    cur_max = best_score
    start = time.time()
    T0 = max(0.1 * cur_max, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.05 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming mass‑spreading step
# ----------------------------------------------------------------------
def _lp_spread_step(f: np.ndarray, M: float,
                    rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve  max Σ b_i   s.t. b_i ≥ 0, (f * b)_k ≤ M.
    Returns the optimal b (unscaled).  If the LP fails, returns None.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build sparse convolution matrix: (f * b)_k = Σ_j f_{k-j} b_j
    # each diagonal offset k (0…n‑1) contains the constant f[k]
    data = [np.full(n, f_k) for f_k in f]
    offsets = np.arange(n, dtype=int)          # positive offsets -> lower triangle
    A = diags(data, offsets, shape=(rows, n), format='csr')

    b_ub = np.full(rows, M, dtype=float)
    c = -np.ones(n)                           # maximise Σ b_i

    # No time‑limit option for older SciPy – we rely on the outer deadline
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    return res.x

    start_time = time.time()

        if base_seq.size != init_n:

    else:

        rate = rng.uniform(0.94, 0.998)
        left = rate ** np.arange(init_n)
        f = (left + left[::-1])
        f = np.clip(f, 0.0, None)

    # Normalise Σf = √(2·n)
    S = np.sqrt(2.0 * init_n)
    f = f / f.sum() * S

    best_f = f.copy()
    best_score = np.max(_full_convolution(f))

    print(f"[init] length={init_n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. LP mass‑spreading iterations (≈30 % of total time)
    # --------------------------------------------------------------
    lp_deadline = start_time + 0.30 * budget_s
    t_mix = 0.3                     # initial mixing factor
    iteration = 0
    while time.time() < lp_deadline:
        iteration += 1
        conv_f = _full_convolution(f)
        M = conv_f.max()

        # Solve the LP
        b = _lp_spread_step(f, M, rng)
        if b is None:
            # LP failed – fall back to random jitter and continue
            i, j = rng.integers(0, init_n, size=2)
            if i != j and f[i] > 0:
                delta = 0.1 * f[i] * rng.random()
                f[i] -= delta
                f[j] += delta
                f = _project_simplex(f, S)
            continue

        sum_b = b.sum()
        if sum_b == 0:

        b = b / sum_b * S

        # Mix and (optionally) enforce symmetry

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
