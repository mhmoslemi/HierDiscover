"""
Adversarial step‑function optimizer – fourth generation.

Goal
----
Search for a non‑negative sequence `a` (length `n`) that minimises
``evaluate_sequence`` which, after normalising the ℓ₁‑norm to √(2 n),
simply equals the maximum entry of the self‑convolution `a*a`.
The optimum is the smallest possible `max(a*a)`.

Method
------
1. **Warm‑start** – reuse the best sequence found in earlier runs
   (`height_sequence_1`), if it exists.
2. **Random exploration** – draw a handful of short candidates
   (n≈4 k–9 k) from a few simple parametric families (power‑law,
   exponential, sparse bumps).  Keep the five best according to the
   exact max‑convolution.
3. **Scale‑up & Adam refinement** – each retained candidate is linearly
   interpolated to a large target length (`N≈40 000`).  On the simplex
   (non‑negative, ℓ₁‑norm = √(2 N)) we run a log‑softmax‑parametrised
   Adam optimiser that minimises a τ‑soft‑max approximation of
   `max(a*a)`.  The optimiser runs for a time‑slice proportional to the
   remaining budget, using a modest learning rate, aggressive τ‑decay
   and occasional Gaussian jitter.
4. **Optional simulated‑annealing fine‑tuning** – if time permits a very
   cheap mass‑transfer walk on the simplex is performed to possibly
   escape shallow minima.
5. **Return** – the best sequence seen is stored as ``height_sequence_2``
   and returned as a plain Python list.  The external score is printed
   for verification.

The whole procedure respects the 1000 s wall‑clock limit and never
returns an empty or invalid list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return the maximum entry of the full self‑convolution of ``a``."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of τ‑soft‑max (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # convolution of mask with reversed a (size 2n‑1 + n‑1)
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]

    # extract the slice that corresponds to ∂b/∂a
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.35,
                   tau_decay: float = 0.9996,
                   lr_decay: float = 0.9999,
                   jitter_every: int = 300) -> tuple[np.ndarray, float]:
    """
    Minimise the exact max‑convolution of ``a`` under the ℓ₁‑norm
    constraint (‖a‖₁ = √(2 n)).  Returns the best vector seen and its
    *hard* peak (max‑convolution).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0  (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution & gradient (soft max) -------
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuner (very cheap, used only if time left)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score        # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score

def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # amount of flat background
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The best sequence found is stored in the global variable `height_sequence_2`
    and also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if present
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 30 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 400.0       # reserve the rest for refinement/anneal
    top_k = 5
    top_candidates = []   # list of (score, seq)

    while time.time() < explore_deadline:
        n = int(rng.integers(4000, 9000))
        fam = rng.choice(['powerlaw', 'exponential', 'sparse'])
        if fam == 'powerlaw':
            raw = _init_left_powerlaw(n, rng)
        elif fam == 'exponential':
            raw = _init_left_exponential(n, rng)
        else:
            raw = _init_sparse_bumps(n, rng)

        raw = np.clip(raw, 0.0, None)
        S = np.sqrt(2.0 * n)
        seq = raw * (S / raw.sum())
        score = _max_convolution(seq)

        if len(top_candidates) < top_k:
            top_candidates.append((score, seq.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif score < top_candidates[-1][0]:
            top_candidates[-1] = (score, seq.copy())
            top_candidates.sort(key=lambda x: x[0])

    # Ensure we have at least one candidate (fallback to uniform)
    if not top_candidates:
        n0 = 5000
        seq0 = np.full(n0, np.sqrt(2.0 * n0) / n0)
        top_candidates = [(_max_convolution(seq0), seq0.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – scale‑up & Adam refinement (≈ 60 % of budget)
    # ------------------------------------------------------------------
    TARGET_LEN = 40000          # final length we optimise at
    best_seq = None
    best_score = np.inf

    for idx, (_, init_seq) in enumerate(top_candidates):
        # Scale to the large target length
        a = _scale_sequence(init_seq, TARGET_LEN)
        a = np.clip(a, 0.0, None)
        a = a * (np.sqrt(2.0 * TARGET_LEN) / a.sum())

        # Allocate a time slice for this candidate
        remaining = end_time - time.time()
        # leave at least ~30 s for possible annealing
        if remaining < 60.0:
            break
        slice_budget = min(remaining * 0.33, 180.0)   # generous slice but not too long

        refined, peak = _adam_optimize(a,
                                       time.time() + slice_budget,
                                       rng,
                                       lr0=0.3,
                                       tau_decay=0.9996,
                                       lr_decay=0.9999,
                                       jitter_every=300)

        if peak < best_score:
            best_score = peak
            best_seq = refined.copy()
            print(f"[refine] init#{idx+1} → len={TARGET_LEN}  new best={best_score:.6f}")

        # early exit if we are out of budget
        if time.time() >= end_time - 35.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning if time permits
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if best_seq is not None and remaining > 25.0:
        final_seq, final_peak = _anneal(best_seq,
                                        time.time() + remaining,
                                        rng)
        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_seq.copy()
            print(f"[anneal] final annealed score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen) – uniform sequence of moderate size
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 5000
        best_seq = np.full(n_fallback, np.sqrt(2.0 * n_fallback) / n_fallback)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    globals()['height_sequence_2'] = final_seq
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    return final_seq
