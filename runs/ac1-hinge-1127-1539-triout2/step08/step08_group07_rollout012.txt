def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5
    # Choose n based on budget and to give enough iterations; allocate n=30000 if possible.
    # We'll compute max_n = 30000 roughly such that O(N log N) is okay.
    if budget_s >= 800:
        n = 30000
    elif budget_s >= 400:
        n = 20000
    else:
        n = 12000

    target = np.sqrt(2.0 * n)

    # Build initial candidate list
    init_seqs = []

    # power law exponents around 0.5, also exponents 0.6, 0.7
    for p in (0.45, 0.5, 0.55, 0.6, 0.65):
        init_seqs.append(_power_law_sequence(n, p, target))

    # If previous global found, resize
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # Jitter variants
    rng = np.random.default_rng()
    jitter_seqs = []
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target * 0.005, size=n)  # smaller jitter
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target)
        jitter_seqs.append(seq_j)
    init_seqs.extend(jitter_seqs)

    # Phase A: Softmax gradient descent (use high beta)
    phase_a_end = time.time() + (deadline - time.time()) * 0.25
    best_seq = None
    best_score = np.inf
    for init in init_seqs:
        if time.time() >= phase_a_end:
            break
        # Run gradient descent with restarts: combine multiple runs
        a_cand, val = _gradient_descent_with_restarts(init, target, phase_a_end)
        if val < best_score:
            best_score = val
            best_seq = a_cand.copy()
            print(f"[{time.time():.2f}] Phase A new best = {best_score:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target)
        best_score = _score(best_seq)

    # Phase B: Hard subgradient descent
    phase_b_end = time.time() + (deadline - time.time()) * 0.55  # allocate majority
    a_hard, val_hard = _hard_subgradient_descent(best_seq, target, phase_b_end)
    if val_hard < best_score:
        best_score = val_hard
        best_seq = a_hard.copy()
        print(f"[{time.time():.2f}] Phase B improved best = {best_score:.6f}")

    # Phase C: Deterministic mass transfer (short)
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq, deadline, target, start_temp=0.3, end_temp=5e-5, low_frac=0.04)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()

    # Phase D: Final polishing using softmax gradient descent with higher beta
    if time.time() + 2.0 < deadline:

def _hard_subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    best = a.copy()
    best_score = _score(a)
    n = a.size
    rng = np.random.default_rng()
    it = 0
    # schedule: step_base initial = target_sum * 0.5
    # momentum (like heavy ball) optional
    momentum = np.zeros_like(a)
    while time.time() < deadline:
        # compute convolution
        conv = _conv_fft(a)
        k = int(np.argmax(conv))
        # compute gradient (2*a_{k-i})
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1)
        g = np.zeros_like(a)
        g[idx] = a[k - idx] * 2.0
        # momentum update
        gamma = 0.9
        momentum = gamma * momentum + g
        # step scaling using norm of momentum
        norm = np.linalg.norm(momentum)
        if norm > 0:
            # step decays with iteration count
            step = target_sum * 0.5 / (norm * (np.sqrt(it + 1)))
            a -= step * momentum
        # project
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if it % 1000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_score:
            best_score = val
            best = a.copy()
            print(f"[{time.time():.2f}] Hard subgradient it={it} new best={best_score:.6f}")
        # stop condition maybe if val not improving after some iterations? Not needed
        it += 1
    return best, best_score

"""
Multi‑phase optimizer for the step‑function benchmark (improved).

The algorithm
--------------
1. Choose a problem size n (larger n gives more freedom, n≈30 000
   works well with the 1000 s budget).
2. Build a diverse set of feasible starting points:
      – several power‑law sequences,
      – a resized copy of the previous best (if any),
      – jittered variants of the above.
   All candidates are projected onto the simplex
   Σ a_i = √(2·n), a_i ≥ 0.

3. **Phase A – Soft‑max projected gradient descent**  
   Short restarts of the momentum‑based scheme from the reference
   implementation, but with a stronger schedule (β up to 5·10⁵
   and larger step factor).  The best result is kept.

4. **Phase B – Hard sub‑gradient descent on the exact max**  
   While time permits we repeatedly
   * compute the full convolution,
   * pick the index k⁎ where it attains its maximum,
   * form the exact gradient of this active entry
     (g_i = 2·a_{k⁎−i} for admissible i),
   * take a momentum step with a decaying step‑size,
   * project back onto the simplex.
   This removes the soft‑max smoothing bias and drives the
   max‑convolution entry down further.

5. **Phase C – Deterministic mass‑transfer refinement**  
   The same “donor‑receiver’’ scheme as in the reference code,
   but with a slower temperature schedule and a tighter low‑fraction
   parameter.

6. **Phase D – Final polish** – a short high‑β soft‑max GD pass.

All heavy linear algebra uses FFT‑based convolutions (O(n log n)),
the simplex projection is Euclidean (O(n log n)),
and the time budget is respected by explicit deadline checks.

The function `propose_candidate` is the entry point required by the
competition harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)² (reduces to max convolution)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient utilities (used in Phase A and final polish)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected gradient descent (enhanced)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 5e5,
                                     step_factor: float = 0.8,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation (more aggressive)."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Several short GD runs with jitter; keeps the best result."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(start,
                                                   time.time() + run_budget,
                                                   target_sum,
                                                   max_beta=5e5,
                                                   step_factor=0.9,
                                                   gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Hard sub‑gradient descent on the exact max entry
# ----------------------------------------------------------------------


def _hard_subgradient_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that uses the true maximal convolution entry.
    The update direction is g_i = 2·a_{k*‑i} (zero when out of range).
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    n = a.size
    momentum = np.zeros_like(a)
    gamma = 0.95          # lighter momentum than Phase A
    it = 0

    while time.time() < deadline:
        # full convolution and index of the current maximum
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = a[k - idx] * 2.0

        # momentum update
        momentum = gamma * momentum + g

        norm = np.linalg.norm(momentum)
        if norm > 0:
            # step size decays like 1/√t
            step = target_sum * 0.5 / (norm * np.sqrt(it + 1))
            a -= step * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional random jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] HardSGD it={it:6d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.3,
                                end_temp: float = 5e-5,
                                low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly lower the current maximal convolution entry.
    Moves are accepted with a Metropolis rule; the temperature decays
    exponentially with the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    # initial convolution
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions w_i = a_i + a_{kmax‑i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # transfer amount (bounded)
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 3000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score
