def propose_candidate(seed=..., budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    # init: either use existing global or generate new via families of length ~8000-15000
    best_seq = None
    best_score = np.inf

    # Generate initial candidates (maybe 2000 attempts) within first 10% of time
    # length randomly in [4000,15000]
    while time.time() < start_time + 0.1*budget_s:
        n = int(rng.integers(4000, 20001))
        a = _sample_family(rng, n, None)
        # symmetrize
        a = _symmetrize(a)
        # normalize sum = sqrt(2n)
        S = np.sqrt(2.0*n)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S
        score = np.max(_full_convolution(a))
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[init] n={n} score={score:.6f}")

    # if nothing found use global height_sequence_1 else uniform
    if best_seq is None:
        if 'height_sequence_1' in globals():
            try:
                base = np.asarray(height_sequence_1, dtype=float)
            except:
                base = None
        else:
            base = None
        if base is not None:
            best_seq = base.copy()
        else:
            n = 5000
            best_seq = np.full(n, 1.0)
        best_score = np.max(_full_convolution(best_seq))

    # Main outer loop: progressively upsample and refine
    # Determine target max n 30000
    target_n = 30000

    while time.time() < deadline:
        # compute remaining time
        now = time.time()
        remaining = deadline - now
        if remaining < 5:
            break

        cur_n = best_seq.shape[0]
        # decide next n: increase by factor up to target
        if cur_n < target_n:
            # choose scaling factor between 1.2-1.5 depending on time
            factor = min(1.5, max(1.2, (target_n / cur_n) ** (remaining / (deadline - start_time))))
            new_n = min(target_n, int(cur_n * factor))
        else:
            new_n = cur_n

        # none if same
        if new_n == cur_n:
            # run refinement only
            t_end = now + remaining * 0.9
            a_ref, sc_ref = _optimize_adam(best_seq, t_end, rng,
                                           max_alpha=800., enforce_sym=True,
                                           smooth_lambda=0.02)
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[refine] n={new_n} score={sc_ref:.6f}")
            break

        # upsample
        a_big = _upsample_seq(best_seq, new_n)
        # re-normalize sum
        S_big = np.sqrt(2.0 * new_n)
        a_big = a_big / a_big.sum() * S_big
        # symmetrize
        a_big = _symmetrize(a_big)

        # refine a_big via Adam for a fraction of remaining time (maybe 30% of remaining)
        t_end = now + 0.3 * remaining
        a_ref, sc_ref = _optimize_adam(a_big, t_end, rng,
                                       max_alpha=800., enforce_sym=True,
                                       smooth_lambda=0.02)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[upsample] from {cur_n} -> {new_n}, score={sc_ref:.6f}")

        # optional subgrad step if time left > ...
        now = time.time()
        remaining = deadline - now
        if remaining > 5:
            t_sub_end = now + 0.2 * remaining
            a_sub, sc_sub = _optimize_subgrad(best_seq, t_sub_end, rng,
                                              enforce_sym=True)
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.copy()
                print(f"[sub] n={best_seq.shape[0]} score={sc_sub:.6f}")

        # continue loop

    # final evaluation and return
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} length={len(final_seq)}")
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq

"""
Propose a step‑function with a low self‑convolution peak.

The algorithm proceeds in several stages:
  • Generate many symmetric random candidates (various analytic families)
    and keep the best.
  • Iteratively up‑sample the current best solution (up to 30 000 points),
    each time re‑normalising and refining it with a projected Adam optimiser.
    The Adam optimiser minimises a smooth approximation of the max‑convolution
    (log‑sum‑exp) and includes a small ‑‑‑smoothness regulariser that encourages
    a smooth bell‑shaped sequence.  Symmetry is enforced after every projection.
  • After each up‑sampling step a short sub‑gradient descent and an optional
    random perturb‑refine are performed.
  • Finally the best sequence is evaluated with the provided
    evaluate_sequence() function, printed and stored globally as
    ``height_sequence_1``.
"""

import time
import numpy as np
from scipy import sparse  # noqa: F401  (kept for compatibility with original code)

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the scaled probability simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families for the initial sampling stage
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a variety of families."""
    fam = rng.integers(0, 12)
    if fam == 0 and base_seq is not None:
        # Upsample an existing good solution
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                 # symmetric power‑law (i·(n‑i))^p
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                # beta‑like skewed
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                 # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        # monotone decreasing power‑law
        p = rng.uniform(0.3, 3.0)
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        # exponential decay
        lam = rng.uniform(0.001, 0.02) * n
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        # logistic S‑curve
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    elif fam == 9:
        # Beta‑distribution shape with random a,b >0 (bell‑shaped)
        a_param = rng.uniform(0.5, 5.0)
        b_param = rng.uniform(0.5, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
    elif fam == 10:
        # Piecewise linear tent with random apex height
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])
    else:
        # Randomly shuffled decreasing exponential (adds extra variability)
        lam = rng.uniform(0.001, 0.02) * n
        a = np.exp(-lam * np.arange(n) / n)
        rng.shuffle(a)
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 600.0,
                   enforce_sym: bool = False,
                   smooth_lambda: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a smooth approximation of the convolution peak.
    An optional quadratic smoothness regulariser (smooth_lambda) encourages
    a bell‑shaped profile.  Symmetry can be enforced after each projection.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper-parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.03

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # smoothness regulariser: penalise (a[i]-a[i+1])^2
        if smooth_lambda > 0.0:
            grad_smooth = np.zeros_like(a)
            # boundary contributions
            grad_smooth[0] = 2.0 * (a[0] - a[1])
            grad_smooth[-1] = 2.0 * (a[-1] - a[-2])
            # interior contributions
            grad_smooth[1:-1] = 2.0 * (2.0 * a[1:-1] - a[0:-2] - a[2:])
            grad = grad + smooth_lambda * grad_smooth

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional random jitter to escape flat regions
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient (single‑term version)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional aspirational random move
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.3:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit if stagnating
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that swaps tiny masses.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    cur_max = best_score
    start = time.time()
    T0 = max(0.1 * cur_max, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.05 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises the self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Initialise from previous global if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈10 % of total time)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.10 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1
        n = int(rng.integers(4000, 20001))
        a = _sample_family(rng, n, base_seq)
        a = _symmetrize(a)                     # enforce symmetry early
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing generated
    # ------------------------------------------------------------------
    if best_seq is None:
        if base_seq is not None:
            best_seq = base_seq.copy()
        else:
            n = 5000
            best_seq = np.full(n, 1.0)
        best_score = np.max(_full_convolution(best_seq))

    # ------------------------------------------------------------------
    # Main optimisation loop – progressive up‑sampling + refinement
    # ------------------------------------------------------------------
    TARGET_N = 30000
    while time.time() < deadline:
        now = time.time()
        remaining = deadline - now
        if remaining < 8.0:
            break

        cur_n = best_seq.shape[0]

        # Decide next length
        if cur_n < TARGET_N:
            # grow by a factor between 1.2 and 1.5, limited by TARGET_N
            factor = rng.uniform(1.2, 1.5)
            new_n = min(TARGET_N, int(cur_n * factor))
        else:
            new_n = cur_n

        if new_n == cur_n:
            # No further up‑sampling possible – perform a final refinement
            t_end = now + 0.75 * remaining
            a_ref, sc_ref = _optimize_adam(best_seq, t_end, rng,
                                           max_alpha=800.0,
                                           enforce_sym=True,
                                           smooth_lambda=0.02)
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[final‑refine] n={new_n} score={sc_ref:.6f}")
            break

        # Upsample
        a_big = _upsample_seq(best_seq, new_n)
        S_big = np.sqrt(2.0 * new_n)
        a_big = a_big / a_big.sum() * S_big
        a_big = _symmetrize(a_big)

        # Adam refinement (use about 20 % of remaining time for this stage)
        t_adam_end = now + 0.20 * remaining
        a_ref, sc_ref = _optimize_adam(a_big, t_adam_end, rng,
                                       max_alpha=800.0,
                                       enforce_sym=True,
                                       smooth_lambda=0.02)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[upsample‑adam] {cur_n}->{new_n} score={sc_ref:.6f}")

        # Optional short sub‑gradient descent
        now = time.time()
        remaining = deadline - now
        if remaining > 6.0:
            t_sub_end = now + 0.10 * remaining
            a_sub, sc_sub = _optimize_subgrad(best_seq, t_sub_end, rng,
                                              enforce_sym=True)
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.copy()
                print(f"[subgrad] n={best_seq.shape[0]} score={sc_sub:.6f}")

        # Random perturb‑refine occasionally (helps escape local minima)
        now = time.time()
        remaining = deadline - now
        if remaining > 5.0 and rng.random() < 0.35:
            # Gaussian perturbation scaled to a few percent of the mean height
            scale = 0.03 * np.mean(best_seq)
            pert = best_seq + rng.normal(scale=scale, size=best_seq.shape)
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert, np.sqrt(2.0 * best_seq.shape[0]))
            pert = _symmetrize(pert)
            t_pert_end = now + 0.08 * remaining
            a_pert, sc_pert = _optimize_adam(pert, t_pert_end, rng,
                                             max_alpha=600.0,
                                             enforce_sym=True,
                                             smooth_lambda=0.015)
            if sc_pert < best_score:
                best_score = sc_pert
                best_seq = a_pert.copy()
                print(f"[perturb‑refine] score={sc_pert:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and output
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
