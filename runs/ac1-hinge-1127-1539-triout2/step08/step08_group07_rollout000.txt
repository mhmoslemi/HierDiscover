def projected_gradient_descent(a0, target_sum, deadline):
    a = a0 copy, project onto simplex.
    best = a, best_score = evaluate score (maxconv after scaling sum? But since sum=target so objective = maxconv, but we can use _score.)

    while time.time() < deadline:
        conv = compute conv, maxc = max(conv)
        max_idx = argmax conv
        # compute gradient w:
        lo = max(0, max_idx-(n-1))
        hi = min(max_idx, n-1)
        idx_range = np.arange(lo, hi+1)
        w = a[idx_range] + a[max_idx - idx_range]
        grad = np.zeros_like(a)
        grad[idx_range] = w

        # maybe use step size as alpha = step factor * current maxconv / (norm of grad)^2 ? Or we can use simple rule: α = step_size / norm(grad) where step_size diminishes.
        norm = np.linalg.norm(grad)

        base_lr = 0.5 * target_sum / (t+1e-6) # maybe gradually decreasing.

"""
Propose a low‑score step‑function for the benchmark.

Algorithm
---------
* Choose a problem size n (12000 for the full budget).
* The ℓ¹‑norm is fixed to √(2·n); after each update the vector is projected
  onto the simplex {x≥0, Σx = √(2·n)}.
* Generate a diverse pool of initial shapes:
      - power‑law a_i ∝ (i+1)^{-p}   with p scanned in [0.35,0.85]
      - exponential a_i ∝ exp(-c·i)   with c scanned in [3e‑4, 3e‑3]
      - triangular (linear decay) shapes.
  Every candidate is normalised to the target sum.
* Keep the best few candidates (by the true objective) and run a
  **projected sub‑gradient descent** on the exact maximal convolution
  entry.  The gradient at the current maximal index k* is
      g_i = a_i + a_{k*‑i}  (zero outside the support of the sum).
  A simple diminishing step size and Euclidean simplex projection are used.
* With the remaining time perform a **deterministic mass‑transfer**
  refinement that moves mass from indices contributing most to the current
  maximal convolution entry to indices contributing least.  Moves are
  accepted only if they improve (or by a tiny Metropolis rule to escape plateaus).
* Finally a short high‑β soft‑max gradient descent polish is applied.
* The best sequence found is trimmed of trailing zeros and returned.

The whole routine respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective (for a fixed target sum the denominator is constant)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Benchmark score 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex with sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, c: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-c·i), normalised to target_sum."""
    a = np.exp(-c * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, peak_frac: float, target_sum: float) -> np.ndarray:
    """
    Linear decay from a peak (at index 0) to zero.
    peak_frac controls the fraction of entries that are non‑zero.
    """
    m = int(np.ceil(peak_frac * n))
    a = np.concatenate([np.linspace(1.0, 0.0, m, endpoint=False),
                        np.zeros(n - m, dtype=np.float64)])
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact max‑convolution gradient
# ----------------------------------------------------------------------


def _gradient_maxconv(a: np.ndarray):
    """
    Returns gradient of the maximal convolution entry w.r.t. a,
    together with the current maximal convolution value and its index.
    """
    conv = _conv_fft(a)
    k = int(np.argmax(conv))
    max_val = float(conv[k])

    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    # derivative ∂conv[k]/∂a_i = a_i + a_{k-i}
    grad[idx] = a[idx] + a[k - idx]
    return grad, max_val, k


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convex objective
# ----------------------------------------------------------------------


def _pgd_maxconv(a0: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 max_iter: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent using the exact max‑conv gradient.
    The step size decays roughly as 1/√t.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        grad, max_val, _ = _gradient_maxconv(a)

        # step size schedule (1/√t) scaled to the typical magnitude of a
        t = (time.time() - start) / (deadline - start + 1e-12)
        # early phase bigger steps, then shrink
        lr = (0.5 * target_sum) / (np.linalg.norm(grad) + 1e-12) * (1.0 - t) ** 0.5

        a -= lr * grad
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] PGD it={it:5d} score={best_val:.6f}")

        # occasional tiny jitter to escape flat regions
        if it % 1000 == 0 and it > 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.0005,
                                                   size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         start_temp: float = 0.3,
                         end_temp: float = 1e-4,
                         low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Greedy move of mass from the highest‑contributing index (w.r.t. the
    current maximal convolution entry) to a low‑contributing index.
    Acceptance follows a simple annealed Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    # initial convolution
    conv = _conv_fft(a)

    while time.time() < deadline:
        # temperature schedule (exponential annealing)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution weights w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        # pick a low‑weight receiver among the lowest low_frac fraction
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        # bound the transferred mass
        max_delta = a[donor] * 0.4
        weight_gap = w.max() - w.min()
        if weight_gap > 0:
            max_delta = min(max_delta, weight_gap * 0.3)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # propose move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to avoid being stuck
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max gradient (high‑β final polish, adapted from the reference)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the β‑soft‑max approximation to max(convolution).
    Returns (grad, conv) where grad has same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # soft‑max weights, sum to 1

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _softmax_polish(a0: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    max_beta: float = 2e5,
                    step_factor: float = 0.6,
                    gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    High‑β projected gradient descent on the soft‑max approximation of max‑conv.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0003 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Softmax polish it={it:5d} β={beta:.0f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.8, max_beta)

        it += 1

    return best_a, best_val
