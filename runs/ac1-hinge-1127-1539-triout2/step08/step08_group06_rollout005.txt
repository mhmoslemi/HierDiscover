"""
Propose a step‑function (non‑negative vector) that minimises the
self‑convolution peak evaluated by ``evaluate_sequence``.
--------------------------------------------------------------------
Idea
----
* Use a fairly long sequence (≈20 000 entries) – the average
  contribution per convolution index is then about 1, so the optimum
  cannot be far below this bound.
* Initialise the vector from a heavy‑tailed Dirichlet (Gamma) distribution.
  The distribution is then scaled so that Σa = √(2·n) which makes the
  external score equal to the internal convolution peak.
* Perform a simulated‑annealing / hill‑climbing loop that repeatedly
  transfers a small amount of mass from one index to another (or swaps
  two values).  After each move the full convolution peak is recomputed
  using an FFT‑based linear convolution.
* Accept improving moves always; accept worsening moves with a
  probability that decays with a temperature schedule.
* Every few thousand steps the vector is re‑projected onto the simplex
  {a≥0, Σa = √(2·n)} to counter numerical drift.
* When the time budget allows, the best vector is linearly up‑sampled
  to a larger length (up‑to 40 000) and a short annealing phase is run
  again – longer sequences often admit a lower peak.
* The final best vector is returned (and stored globally as
  ``height_sequence_1`` for warm‑restarts).
--------------------------------------------------------------------
The algorithm is deliberately simple, uses only NumPy and runs well
within the 1000 s budget.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of total mass s."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based linear convolution (full length = 2*n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Linear interpolation to a new length (preserves shape)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, new_len: int) -> np.ndarray:
    if base.size == new_len:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, new_len)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose an initial length.
    # Larger lengths give more “bins” for the convolution, often
    # allowing a lower peak, but also cost more per FFT.
    # ------------------------------------------------------------------
    min_n, max_n = 18000, 30000
    n = int(rng.integers(min_n, max_n + 1))
    S = math.sqrt(2.0 * n)                     # required L1‑norm

    # ------------------------------------------------------------------
    # Initialise the vector.
    # If a previous construction exists, up‑sample it; otherwise draw
    # from a heavy‑tailed Dirichlet (Gamma) distribution.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            a = _upsample_seq(base, n)
            a = a / a.sum() * S
        except Exception:
            base = None
    if 'base' not in locals():
        # shape parameter <1 gives many tiny entries and a few larger ones
        a = rng.gamma(shape=0.25, scale=1.0, size=n)
        a = a / a.sum() * S

    # ------------------------------------------------------------------
    # Keep track of the best vector seen so far.
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_score = _max_convolution(best_vec)

    cur_vec = a.copy()
    cur_score = best_score

    # Simple temperature that controls the size of the transferred mass.
    temperature = S * 0.4

    iteration = 0
    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- propose a move ------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # With probability 0.9 we move a fraction of mass from i → j,
        # otherwise we simply swap the two entries.
        if rng.random() < 0.9:
            if cur_vec[i] <= 0.0:
                continue
            # transfer up to 20 % of the source mass, scaled by temperature
            max_delta = 0.20 * cur_vec[i]
            delta = max_delta * rng.random() * (temperature / S)
            delta = min(delta, cur_vec[i])
            cand = cur_vec.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            cand = cur_vec.copy()
            cand[i], cand[j] = cur_vec[j], cur_vec[i]

        # ----- evaluate the candidate ---------------------------------------
        cand_score = _max_convolution(cand)

        # ----- acceptance test -----------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis rule
            prob = math.exp(-(cand_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_vec = cand.copy()
                best_score = cand_score
                print(f"[iter {iteration}] new best = {best_score:.6f}")

        # ----- temperature / periodic re‑projection --------------------------
        temperature *= 0.99997                     # very gentle cooling
        if iteration % 3000 == 0:
            # numeric drift may break the exact L1‑norm; re‑project.
            cur_vec = _project_simplex(cur_vec, S)
            cur_score = _max_convolution(cur_vec)
            if cur_score < best_score:
                best_vec = cur_vec.copy()
                best_score = cur_score
                print(f"[reproj {iteration}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling phase (if time permits).  Larger n often yields
    # a lower peak because the average per convolution bin stays 1.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        up_factors = [1.2, 1.5, 2.0]
        max_len = 40000
        for fac in up_factors:
            new_n = int(best_vec.shape[0] * fac)
            if new_n < 2 or new_n > max_len:
                continue
            up_vec = _upsample_seq(best_vec, new_n)
            up_vec = up_vec / up_vec.sum() * math.sqrt(2.0 * new_n)

            # give each up‑sampled version a short annealing burst
            short_dead = min(deadline, time.time() + 0.07 * (deadline - time.time()))
            # reuse the same annealing loop but with a tighter iteration budget
            cur = up_vec.copy()
            cur_sc = _max_convolution(cur)
            temp = math.sqrt(2.0 * new_n) * 0.3
            while time.time() < short_dead:
                i, j = rng.integers(0, new_n, size=2)
                if i == j:
                    continue
                if rng.random() < 0.9:
                    if cur[i] <= 0.0:
                        continue
                    max_delta = 0.15 * cur[i]
                    delta = max_delta * rng.random() * (temp / math.sqrt(2.0 * new_n))
                    delta = min(delta, cur[i])
                    cand = cur.copy()
                    cand[i] -= delta
                    cand[j] += delta
                else:
                    cand = cur.copy()
                    cand[i], cand[j] = cur[j], cur[i]
                cand_sc = _max_convolution(cand)
                if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (temp + 1e-12)):
                    cur = cand
                    cur_sc = cand_sc
                    if cand_sc < best_score:
                        best_vec = cand.copy()
                        best_score = cand_sc
                        print(f"[upsample {fac:.2f}] improved to {best_score:.6f}")
                temp *= 0.99995
            # after the short burst, keep the best version
            best_vec = best_vec
            # continue with whatever time remains

    # ------------------------------------------------------------------
    # Final tiny hill‑climb (pure swaps) using any remaining budget.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i, j = rng.integers(0, best_vec.shape[0], size=2)
        if i == j:
            continue
        cand = best_vec.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_sc = _max_convolution(cand)
        if cand_sc < best_score:
            best_vec = cand
            best_score = cand_sc
            print(f"[final swap] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result and store it globally for possible warm‑restart.
    # ------------------------------------------------------------------
    final_seq = best_vec.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} iterations={iteration}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
