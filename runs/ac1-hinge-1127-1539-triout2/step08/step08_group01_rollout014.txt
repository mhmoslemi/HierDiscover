"""
Propose a candidate step‑function that (approximately) minimises
``evaluate_sequence``.  The method is completely different from the
previous gradient‑descent / hill‑climbing pipeline:

* Start from the provided ``height_sequence_1`` (or a random seed) and
  normalise it to the required ℓ¹‑norm  √(2 n).
* Iteratively solve a *linear programme* :

      maximise    Σ g_j
      subject to  (f * g)_k ≤ M   for all k
                  g_j ≥ 0

  where ``f`` is the current sequence, ``M = max(f*f)`` and ``*`` is the
  linear convolution.  The LP is a pure linear problem (the convolution
  is linear in the unknown vector ``g``) and can be solved with SciPy’s
  HiGHS solver in a few seconds for n≈2000.

* The LP gives a vector ``g`` with a *much larger* ℓ¹‑norm than ``f``.
  Scaling it back to the normalisation (``target = √(2 n)``) produces a
  new candidate ``g̃``.  Because the convolution bound is linear in the
  scaling factor, ``max(g̃*g̃)`` is typically smaller than the old value
  ``M``.
* Accept the scaled vector if it strictly improves the objective,
  otherwise stop.  A few mixing steps are tried as a fallback.
* After the LP phase a short Adam‑gradient run is performed for a final
  polish.

The whole routine respects the 1000 s wall‑clock budget, prints
intermediate scores and finally returns the best sequence found as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (simplex projection, FFT max‑convolution, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi‑et‑al. (2008) implementation.
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (returned array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# Random‑initialiser (used only if no seed is present)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    # Dirichlet (uniform) is a good generic starter
    a = rng.dirichlet(alpha=np.ones(n)) * target
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (core of the new algorithm)
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ x = f * x (linear convolution).
    Column j of A contains the vector f shifted by j positions.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # fill by columns – each column j contributes to rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_improve(f: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Perform a single LP improvement step.
    Returns the possibly‑improved vector (or the original if no improvement).
    """
    from scipy.optimize import linprog

    n = f.size
    M = _max_convolution_fft(f)

    # Build the constraint matrix A (size (2n‑1) × n)
    A = _build_convolution_matrix(f)

    # RHS for all constraints – the current worst convolution value
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximise Σ g_j  →  minimise −Σ g_j
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # time budget for the solver – leave a safety margin
    time_left = deadline - time.time() - 0.2
    opts = {"presolve": True}
    if time_left > 0.0:
        opts["time_limit"] = time_left

    # Solve the LP
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options=opts)

    if not res.success:
        # Solver failed – return the input unchanged
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return f

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    # Scale back to the required ℓ¹‑norm
    scale = target_sum / sum_g
    g_scaled = g * scale

    # Evaluate the new candidate
    new_M = _max_convolution_fft(g_scaled)

    # Accept only if we strictly improve the max‑convolution
    if new_M + 1e-12 < M:
        print(f"[{time.time():.1f}] LP step improved M: {M:.6f} → {new_M:.6f}"
              f"   sum_g={sum_g:.2f} scale={scale:.4f}")
        return g_scaled
    else:
        # No improvement – try a small convex combination as a fall‑back
        t = 0.2
        mixed = (1.0 - t) * f + t * g_scaled
        mixed = _project_to_simplex(mixed, target_sum)
        mixed_M = _max_convolution_fft(mixed)
        if mixed_M + 1e-12 < M:
            print(f"[{time.time():.1f}] LP mixing improved M: {M:.6f} → {mixed_M:.6f}")
            return mixed
        # Nothing helped – keep the old vector
        print(f"[{time.time():.1f}] LP step gave no improvement (M={M:.6f})")
        return f

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smooth‑max surrogate (log‑sum‑exp) of a*a."""
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """Projected Adam on the smooth‑max objective (few seconds max)."""
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning rate – tuned to the ℓ¹‑radius
    base_lr = 0.2 * target_sum

    beta = 2.0
    beta_max = 1e5
    beta_factor = 1.3

    t = 0
    rng = np.random.default_rng(12345)

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 150 == 0:
            # slight jitter to avoid stagnation
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

            best_f = f_candidate.copy()
            f = f_candidate               # continue from the improved point
            print(f"[{time.time():.1f}] New best after LP iteration {it}: {best_score:.6f}")
        else:
            # No improvement – exit the LP loop
            print(f"[{time.time():.1f}] LP loop stagnated after {it} iterations")
            break

    # ------------------------------------------------------------------
    # Brief Adam polishing (optional, runs only if time permits)
    # ------------------------------------------------------------------
    time_left = deadline - time.time() - 1.0
    if time_left > 10.0:
        print(f"[{time.time():.1f}] Starting final Adam polishing (≈{time_left:.1f}s left)")
        polished = _adam_optimize(best_f, target, time.time() + time_left)
        polished_score = _max_convolution_fft(polished)
        if polished_score < best_score:
            best_score = polished_score
            best_f = polished.copy()
            print(f"[{time.time():.1f}] Polishing improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    best_f = np.asarray(best_f, dtype=float)
    nz = np.flatnonzero(best_f > 1e-12)
    if nz.size:
        best_f = best_f[: nz[-1] + 1]
    else:
        best_f = best_f[:1]

    # ------------------------------------------------------------------
    # Final evaluation (official objective)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_f.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f}"
              f"  (length={best_f.size})")
    except Exception as exc:                         # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}", file=sys.stderr)

    return best_f.tolist()
```

This implementation introduces a **linear‑programming based improvement step** that is not present in the previous gradient‑descent/hill‑climbing code.  By repeatedly solving the LP that maximises the total mass under the current convolution bound and scaling back to the required ℓ¹‑norm, the algorithm strictly reduces the maximal self‑convolution value in practice.  A short Adam polishing stage is added for a final tweak, and the whole search respects the 1000 s wall‑clock limit.
