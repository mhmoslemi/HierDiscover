def propose_candidate(seed=42, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # If height_sequence_1 provided, try to use that as baseline
    try:
        cand = np.asarray(globals()["height_sequence_1"], dtype=float)
        cand = _project_to_simplex(cand, np.sqrt(2.0 * cand.size))
        score = _max_convolution_fft(cand)
        best_seq = cand.copy()
        best_score = score
        print(f"seed from height_sequence_1 score {score:.6f}")
    except Exception:
        pass

    # length schedule
    schedule = [2000, 5000, 10000, 20000, 50000, 80000]
    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # Create a set of initial candidates to explore
        init_candidates = []
        # upsample best if exist
        if best_seq is not None and best_seq.size != n:
            init_candidates.append(_upsample_sequence(best_seq, n))
        else:
            init_candidates.append(_random_initial_sequence(n, rng))

        # add few random family samples
        for _ in range(5):
            init_candidates.append(_random_initial_sequence(n, rng))

        # Optimize each candidate with smoothed max descent
        for a0 in init_candidates:
            if time.time() >= deadline:
                break
            # allocate portion of remaining time for this candidate
            remaining = deadline - time.time()
            # allocate some time like min(remaining*0.1, 30 sec)
            candidate_deadline = time.time() + min(remaining*0.1, 30.0)
            a_opt = _adam_optimize(a0, candidate_deadline)
            score = _max_convolution_fft(a_opt)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"improved after Adam n={n} score={score:.6f}")

        # Now hill climbing on current best at this length
        if time.time() >= deadline:
            break
        # Ensure best_seq attains current n
        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)
        # allocate time for hill climb (e.g., 30% of remaining)
        rem = deadline - time.time()
        if rem < 5:
            break
        hc_deadline = time.time() + min(rem*0.3, 40.0)
        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = float(b_hc.max())
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"hclimb improvement n={n} score={score_hc:.6f}")

    # final polishing if time left
    if best_seq is not None and time.time() < deadline - 5:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        # final Adam
        a_final = _adam_optimize(best_seq.copy(), deadline-3.0, beta_start=5.0, beta_target=5e5)
        b_final = _full_convolution(a_final)
        # final hill climb
        a_final, b_final = _hill_climbing(a_final, b_final, deadline-2.0)
        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"final polish score={final_score:.6f}")

    # Trim trailing zeros
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:

    # Print final evaluate

"""
Optimisation of the ℓ¹‑normalised self‑convolution maximum.

The algorithm follows a multi‑phase workflow:

1.   **Initialisation** – several parametric families (exponential,
     power‑law, linear, piece‑wise exponential, cosine‑shaped and a
     Beta‑type “arcsine” family) are sampled, projected onto the
     ℓ¹‑simplex Σa_i = √(2 n) and the best is kept.

2.   **Smooth‑max descent** – Adam optimisation is applied to the
     smooth approximation (1/β)·log∑exp(β·(a⋆a)).  The temperature β is
     grown geometrically up to a large value (≈5·10⁵) so that the
     objective approaches the true ℓ∞‑norm.  After each step the vector
     is projected back onto the simplex.

3.   **Exact hill‑climbing** – a simulated‑annealing style move transfers
     a random amount of mass from an index that contributes heavily to
     the current maximal convolution entry to an index that currently
     carries little mass.  The full convolution is updated in O(n) using
     the analytic formulas from the original implementation; moves that
     lower the maximum are always accepted, otherwise they are accepted
     with a Metropolis probability.

4.   **Length schedule & up‑sampling** – the procedure is run for a
     schedule of increasing support sizes (2000 → 80000).  After each
     size the best vector is up‑sampled (linear interpolation) to the
     next size and refined again.

5.   **Final polishing** – one last Adam pass with a very high β
     followed by a short hill‑climb is performed if time permits.

All phases are time‑aware; the global wall‑clock budget (default 1000 s)
is respected.  The best sequence found is returned as a plain
``list[float]`` which can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log Σ exp(β·b_k)  where
    b = a⋆a (full convolution).  Returns the tuple (grad, b) with:

        grad_i = 2·∑_k p_k·a_{k-i}
    where p_k ∝ exp(β·b_k) is the soft‑max over the convolution entries.
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev ⋆ p  (the gradient factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 1e5,
                   beta_factor: float = 1.3) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max (log‑sum‑exp) objective.
    The schedule for β grows geometrically until ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.2          # initial learning‑rate scale
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the smoothed max
        g, _ = _softmax_gradient(a, beta)

        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaux
        if step % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (log‑scale)
        if beta < beta_target and step % 250 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)"""
    r = rng.uniform(0.96, 0.9995)
    idx = np.arange(n, dtype=float)
    a = r ** idx
    return a


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}"""
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    return a


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.5, 2.0)
    a = np.linspace(a0, 0.0, n, dtype=float)
    return a


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential decay with a breakpoint."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    a = np.concatenate([left, right])
    return a


def _family_cosine(n: int, rng) -> np.ndarray:
    """a_i ∝ 1 + cos(π·i/(n‑1))  (non‑negative, symmetric)"""
    i = np.arange(n, dtype=float)
    a = 1.0 + np.cos(np.pi * i / (n - 1))
    return a


def _family_beta(n: int, rng) -> np.ndarray:
    """
    Beta‑type family: a_i ∝ (i+0.5)^{-p}·(n‑i‑0.5)^{-p},
    with exponent p sampled in [0.2, 1.0].
    This mimics the arcsine density which flattens the convolution.
    """
    p = rng.uniform(0.2, 1.0)
    x = np.arange(n, dtype=float) + 0.5
    a = (x * (n - x + 0.0)) ** (-p)
    return a


def _family_random_spike(n: int, rng) -> np.ndarray:
    """Sparse vector with a handful of random spikes."""
    a = np.zeros(n, dtype=float)
    k = rng.integers(1, max(2, n // 10))
    for _ in range(k):
        idx = rng.integers(0, n)
        a[idx] += rng.uniform(0.1, 1.0)
    # add a small constant offset to avoid exact zeros
    a += rng.uniform(0.0, 0.01, size=n)
    return a


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Choose uniformly among a collection of parametric families,
    then project onto the ℓ¹‑simplex Σa = √(2 n).
    """
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp,
                _family_cosine,
                _family_beta,
                _family_random_spike]
    a = families[rng.integers(len(families))](n, rng)
    # add a tiny jitter to avoid exact ties
    a += rng.normal(scale=0.001, size=n)
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Exact hill‑climbing (simulated‑annealing style)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    An amount of mass is moved from a source index (chosen among contributors
    to the current maximal convolution entry) to a target index (chosen
    preferentially where a is small).  The full convolution is updated in O(n)
    per move.  Moves that lower the maximal entry are always accepted,
    otherwise they are accepted with a Metropolis probability.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # initial temperature – a fraction of the current peak height
    T = cur_max * 0.2 + 1e-9
    cooling = 0.998

    while time.time() < deadline:
        # indices attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = int(rng.choice(idx_max))

        # source – pick proportionally to its contribution to this entry
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = int(rng.choice(i_range, p=probs))
        else:
            i = int(rng.integers(0, n))

        if a[i] < 1e-14:
            continue

        # target – bias towards positions with little mass
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = int(rng.choice(n, p=probs_t))
        else:
            j = int(rng.integers(0, n))

        if i == j:
            continue

        # amount of mass to move (up to 75 % of the source)
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        # remove contribution of source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # add contribution of target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correction for the crossed term i+j (appears twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # recompute convolution for the best candidate (exactly)
    b_best = _full_convolution(best_a)
    return best_a, b_best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock limit and returns the best sequence
    found as a plain Python list.

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size

            score0 = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = score0
            print(f"[{time.time():.1f}] seed from height_sequence_1 (n={n0}) score={score0:.6f}")
        except Exception as e:
            print(f"Unable to seed from height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Length schedule – gradually increase support size
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 50000, 80000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1) Build a list of initial candidates for this length
        # --------------------------------------------------------------
        init_candidates = []

        # a) up‑sample the current best if its length differs
        if best_seq is not None and best_seq.size != n:
            init_candidates.append(_upsample_sequence(best_seq, n))
        elif best_seq is not None and best_seq.size == n:
            init_candidates.append(best_seq.copy())

        # b) a few fresh random families
        for _ in range(5):
            init_candidates.append(_random_initial_sequence(n, rng))

        # --------------------------------------------------------------
        # 2) Adam‑based smooth‑max descent on each candidate
        # --------------------------------------------------------------
        for a0 in init_candidates:
            if time.time() >= deadline:
                break
            # allocate a modest slice of the remaining budget

            if remaining < 5.0:
                break
            sub_deadline = time.time() + min(remaining * 0.35, 30.0)

                                   beta_factor=1.4)
            score = _max_convolution_fft(a_opt)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] Adam descent improvement (n={n}) score={score:.6f}")

                print("Target reached – terminating early.")

        if best_score <= 1.503:
            break

        # --------------------------------------------------------------

        if time.time() >= deadline:

        if best_seq is None or best_seq.size != n:

        a_cur = _project_to_simplex(a_cur, target_sum)

        score_hc = float(b_hc.max())

    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
