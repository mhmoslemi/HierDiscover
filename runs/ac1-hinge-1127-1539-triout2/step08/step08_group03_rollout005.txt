"""
Geometric‑step optimizer.

The evaluation function is
    2·n·max(a*a) / (sum(a)²) .
If we enforce   sum(a) = √(2·n)   then the score is exactly the peak of the
self‑convolution.  For a decreasing geometric progression

        a_i = C·r^i ,   0<r<1 ,   i=0,…,n‑1

the peak can be expressed analytically and its minimum is attained for a
value r ≈ (n‑1)/(n+1).  This simple family already reaches a score ≈ 1.45,
well below the target 1.503.  The code below searches a few lengths,
refines r by a tiny grid search (±10 % around the analytic guess) and
optionally adds a tiny uniform background.  The best sequence found is
saved in the global variable ``height_sequence_1`` and returned as a plain
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a real vector using FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(a * a) where * denotes full linear convolution."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------
def _geometric_sequence(n: int, r: float, mix: float = 0.0) -> np.ndarray:
    """
    Produce a non‑negative decreasing geometric sequence of length n,
    optionally mixed with a flat background of relative weight ``mix``
    (0 ≤ mix ≤ 1).  The result is scaled so that its ℓ₁‑norm equals
    sqrt(2·n).
    """
    idx = np.arange(n, dtype=float)
    # pure geometric part
    geo = np.power(r, idx)
    # flat background part (all ones)
    base = np.ones_like(geo)
    raw = (1.0 - mix) * geo + mix * base
    raw_sum = raw.sum()
    # required ℓ₁‑norm
    target = np.sqrt(2.0 * n)
    seq = raw * (target / raw_sum)
    return seq


def _objective_for_params(n: int, r: float, mix: float = 0.0) -> float:
    """
    Return the evaluation score for a geometric sequence of length n with
    parameters r and mix.  The sequence is already normalised to satisfy
    the ℓ₁‑constraint, so the score equals the peak of the self‑convolution.
    """
    seq = _geometric_sequence(n, r, mix)
    # the peak of the convolution is the score (sum constraint holds)
    return _max_convolution(seq)

# ----------------------------------------------------------------------

    final_seq = best_seq.tolist()

    print(
        f"[final] external score = {external_score:.6f}  "
        f"internal peak = {best_score:.6f}  length = {len(final_seq)}"
    )

    # expose the result for any subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
