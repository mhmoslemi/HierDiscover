from scipy.optimize import linprog

def solve_lp(f, M, target_sum):
    n = f.shape[0]
    mask = ((np.arange(2*n-1)[:, None] - np.arange(n) >= 0) &
            (np.arange(2*n-1)[:, None] - np.arange(n) < n))
    A = mask * f   # shape (2n-1, n)
    # linprog wants A_ub @ x <= b_ub
    # Since we want each row <= M, we set b_ub = np.full(2*n-1, M)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=(0, None), method='highs', options={'presolve': True})
    if res.success:
        b_opt = res.x
    else:
        return None
    # note: objective is -sum(b), maximizing sum(b)
    return b_opt
```

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = np.max(b)
    exps = np.exp(beta*(b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1: n-1 + n]
    return grad, b
```

def _annealed_gradient_descent(a0, deadline, target_sum, max_beta=2e5, step_factor=0.7, gamma=0.99):

"""
Propose a low‐score step function.

The algorithm consists of three phases:

1. **Small‑scale shaping (LP phase).**  
   Work with a modest length `n_small ≈ 2000`.  Starting from a
   power‑law shape we iteratively solve a linear program

        max  Σ b_i
        s.t. b_i ≥ 0,
             (f * b)_k ≤ M   for all k,

   where `f` is the current sequence and `M = max(f*f)`.  The LP
   pushes mass into positions that keep the convolution under the
   current maximum.  The solution is renormalised to the required
   ℓ¹‑norm and mixed into `f` with a decreasing step size `t`.
   This produces a very favourable shape for the final sequence.

2. **Upscaling & polishing (gradient phase).**  
   The best small sequence is linearly interpolated to a large
   length (`n≈12000` for a 1000 s budget) and projected onto the
   simplex `{x ≥ 0, Σx = √(2·n)}`.  A projected (momentum) gradient
   descent on a soft‑max approximation of the maximal convolution
   entry is then run with an annealing schedule for the soft‑max
   temperature `β`.  The final pass uses an extreme `β` (≈4·10⁵) to
   drive the solution close to the true max‑convolution optimum.

3. **Final trimming.**  
   Trailing zeros are removed and the list of floats is returned.

The score is `2·n·max(a*a)/(sum(a)²)`.  Since `sum(a)` is fixed,
minimising the maximal convolution entry suffices.

All helper functions are top‑level; the only external dependency is
`scipy.optimize.linprog` (HiGHS backend) which solves the dense LPs
efficiently.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)² (reduced to max convolution)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent (with momentum and annealed soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short gradient‑descent runs with random jitter.
    Keeps the best result found before the given deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start,
                                            time.time() + run_budget,
                                            target_sum,
                                            max_beta=1e5,
                                            step_factor=0.8,
                                            gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming shaping (small scale)
# ----------------------------------------------------------------------


def _build_mask(n: int) -> np.ndarray:
    """Boolean mask M[k,i] = 1 iff 0 ≤ k‑i < n."""
    k = np.arange(2 * n - 1)[:, None]
    i = np.arange(n)
    return ((k - i >= 0) & (k - i < n)).astype(np.float64)


def _lp_shaping(f: np.ndarray, M_target: float, mask: np.ndarray) -> np.ndarray | None:
    """
    Solve  max Σ b_i   s.t.  b_i ≥ 0,  (f * b)_k ≤ M_target.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = f.shape[0]
    # Linear constraints matrix A_ub @ b ≤ b_ub
    A = mask * f  # broadcasting: each column i scaled by f_i
    b_ub = np.full(2 * n - 1, M_target, dtype=np.float64)

    # Objective: maximise Σ b_i  → minimise -Σ b_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c,
                  A_ub=A,
                  b_ub=b_ub,
                  bounds=(0, None),
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None
    return np.asarray(res.x, dtype=np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm consists of:
        1. Small‑scale LP shaping.
        2. Upscaling to the final size.
        3. Projected gradient descent polishing.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose problem sizes according to the total budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_large = 12000
    elif budget_s >= 400:
        n_large = 8000
    else:
        n_large = 5000

    target_large = np.sqrt(2.0 * n_large)

    # small problem (for LP shaping)
    n_small = min(2000, max(500, n_large // 5))
    target_small = np.sqrt(2.0 * n_small)

    # --------------------------------------------------------------
    # Initial small sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_big = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        f_small = _resize_sequence(prev_big, n_small)
    else:
        f_small = _power_law_sequence(n_small, 0.55, target_small)

    # a quick GD warm‑start on the tiny problem (optional, cheap)
    warm_deadline = start_time + 0.08 * budget_s
    if warm_deadline > deadline:
        warm_deadline = deadline
    f_small, _ = _gradient_descent_with_restarts(f_small, target_small, warm_deadline)

    # --------------------------------------------------------------
    # Phase 1 – LP shaping on the small problem
    # --------------------------------------------------------------
    mask_small = _build_mask(n_small)

    # allocate roughly 30 % of the total budget to the LP phase
    lp_deadline = time.time() + 0.30 * (deadline - time.time())
    best_small = f_small.copy()
    best_score_small = _score(best_small)

    iteration = 0
    while time.time() < lp_deadline:
        M_current = _max_convolution_fft(f_small)

        b_opt = _lp_shaping(f_small, M_current, mask_small)
        if b_opt is None:
            print(f"[{time.time():.2f}] LP failed at iteration {iteration}")
            break

        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            break

        # Rescale to preserve the ℓ¹‑norm
        g = b_opt * (target_small / sum_b)

        # Mixing step with a decreasing step size
        t = max(0.05, 0.30 * (1.0 - iteration * 0.07))
        f_small = (1.0 - t) * f_small + t * g
        f_small = _project_to_simplex(f_small, target_small)

        cur_score = _score(f_small)
        if cur_score < best_score_small:
            best_score_small = cur_score
            best_small = f_small.copy()
            print(f"[{time.time():.2f}] LP iter {iteration:3d} best small score={best_score_small:.6f}")

        iteration += 1
        # a tiny safety break – we do not want to spend all time here
        if iteration >= 30:
            break

    # --------------------------------------------------------------
    # Phase 2 – Upscale to the final size
    # --------------------------------------------------------------
    f_large = _resize_sequence(best_small, n_large)
    f_large = _project_to_simplex(f_large, target_large)

    # --------------------------------------------------------------
    # Phase 3 – Gradient descent polishing on the full size
    # --------------------------------------------------------------
    gd_deadline = deadline - 10.0  # keep a little time for final polishing
    f_large, best_val = _gradient_descent_with_restarts(f_large, target_large, gd_deadline)

    # Final high‑β polish (few seconds)
    if time.time() + 2.0 < deadline:
        f_large, val2 = _annealed_gradient_descent(f_large,
                                                   deadline,
                                                   target_large,
                                                   max_beta=4e5,
                                                   step_factor=0.6,
                                                   gamma=0.995)
        if val2 < best_val:
            best_val = val2
            f_large = f_large.copy()
            print(f"[{time.time():.2f}] Final polish improved score={best_val:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing zeros and convert to list
    # --------------------------------------------------------------
    arr = np.array(f_large, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
