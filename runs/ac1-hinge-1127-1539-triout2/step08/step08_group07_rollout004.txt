"""
Propose a step‑function (non‑negative sequence) whose benchmark score
2·n·max(conv(a,a)) / (∑a)² is as small as possible.

Algorithm
---------
1. Choose a problem size `n` (12000 for a full 1000 s budget) and fix the
   ℓ¹‑norm to `sqrt(2·n)`.  The objective is homogeneous, therefore fixing the
   sum does not affect the optimal value but makes projection cheap.

2. **Rich initial pool** – generate many simple families:
   * pure power laws a[i] ∝ (i+1)^{-p}
   * offset power laws a[i] ∝ (i+offset)^{-p}
   * exponentials a[i] ∝ exp(-λ·i)
   * a resized previous best (if available) and jittered copies.
   All candidates are projected onto the simplex {x≥0, Σx=√(2·n)} and
   quickly evaluated with an FFT based convolution.  The best few (≤5)
   are kept.

3. **Adam‑based projected gradient descent** – for each retained candidate
   we run a momentum optimiser on a soft‑max approximation of the
   maximal convolution entry.  The temperature β is exponentially
   increased up to `max_beta≈2·10⁵`.  After each step the iterate is
   projected onto the simplex.  The best point over the whole run is
   recorded.

4. **Deterministic mass‑transfer refinement** – using the current maximal
   convolution index `k*` we compute the contribution
   `w_i = a_i + a_{k*-i}`.  Mass is moved from the index with largest
   `w_i` to one with smallest `w_i`.  Moves are accepted with a
   Metropolis rule whose temperature decays with the remaining time.
   This directly attacks the bottleneck entry of the convolution.

5. **Final high‑β polishing** – a short Adam run with a very large β
   (≈2·10⁵) removes any remaining irregularities.

All heavy linear‑algebra uses FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s budget and prints progress information.

The returned list is truncated after the last entry larger than 1e‑12,
as required by the benchmark harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)²."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a[i] = (i+1)^(-p), normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _offset_power_law_sequence(n: int, p: float, offset: float,
                               target_sum: float) -> np.ndarray:
    """a[i] = (i+1+offset)^(-p), normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float,
                          target_sum: float) -> np.ndarray:
    """a[i] = exp(-lam*i), normalised."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(conv(a,a))
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where grad is the gradient of the soft‑max
    approximation of max(conv(a, a)).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # numerically stable soft‑max weights
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) ⊛ p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           beta_mult: float = 2.0,
                           beta_step: int = 200,
                           beta_start: float = 0.5,
                           beta_growth: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex.
    The temperature β of the soft‑max is increased exponentially.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    beta = beta_start

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # global step size – scales inversely with β and the gradient norm
        step = step_factor / ((beta + 1.0) * (np.linalg.norm(m_hat) + 1e-12))

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β occasionally
        if it % beta_step == 0 and beta < max_beta:
            beta = min(beta * beta_growth, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (the “DMT” phase)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift mass from positions that contribute heavily to the
    current maximal convolution entry towards positions that contribute little.
    Moves are accepted with a Metropolis rule whose temperature decays
    exponentially with the remaining time.
    """
    a = a0.copy()
    n = a.size

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    conv = _conv_fft(a)                     # current convolution

    it = 0
    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution of each index

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bound the transferred amount
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

    # resized previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies to increase diversity
    rng = np.random.default_rng()
    jittered = []
    for seq in init_seqs:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    init_seqs.extend(jittered)

    # --------------------------------------------------------------
    # quick evaluation of the pool; keep the best few
    # --------------------------------------------------------------
    scores = []
    for seq in init_seqs:
        val = _score(seq)
        scores.append(val)
    # sort indices by score (ascending)
    sorted_idx = np.argsort(scores)
    top_k = 5
    best_candidates = [init_seqs[i] for i in sorted_idx[:top_k]]
    best_score = scores[sorted_idx[0]]
    best_seq = best_candidates[0].copy()
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam based projected gradient descent on top candidates
    # --------------------------------------------------------------
    # allocate roughly 45 % of the remainder to this phase
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.45
    for i, cand in enumerate(best_candidates):
        if time.time() >= phase_a_deadline:
            break
        # give each candidate a slice of the remaining budget
        remaining = phase_a_deadline - time.time()
        sub_deadline = time.time() + remaining * 0.3   # generous slice
        a_opt, val_opt = _adam_gradient_descent(cand,
                                                sub_deadline,
                                                target_sum,
                                                max_beta=1e5,
                                                step_factor=0.9,
                                                beta_step=250,
                                                beta_start=0.5,
                                                beta_growth=2.0)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after GD = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-4,
                                                      low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After DMT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Final high‑β polishing using Adam
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=2e5,
                                                step_factor=0.6,
                                                beta_step=250,
                                                beta_start=0.5,
                                                beta_growth=2.0)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # final post‑processing – drop trailing near‑zeros
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation using the provided harness function
    try:
        eval_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {eval_val:.6f}")
    except Exception:
        pass

    return final_seq
