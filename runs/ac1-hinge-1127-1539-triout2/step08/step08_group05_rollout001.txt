def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    def max_conv(a):
        return np.max(_full_convolution(a))

    # load base_seq if exist
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None; best_score = np.inf

    # Stage 1: random initial with many lengths up to 30000
    while time.time() < start_time + 0.1*budget_s:
        # choose length n
        if base_seq is not None and rng.random() < 0.3:
            n = int(min(30000, max(2000, base_seq.shape[0] * rng.uniform(0.8, 1.2))))
        else:
            n = rng.integers(5000, 30001)
        # choose family based on rng
        family_id = rng.integers(0, 10)
        if family_id == 0:
            # random uniform
            a = rng.random(n)
        elif family_id == 1:
            # Gaussian-like: exp(-((i-(n-1)/2)^2)/(2*sigma^2))
            sigma = rng.uniform(0.1, 0.3) * n
            x = np.arange(n) - (n-1)/2
            a = np.exp(-0.5 * (x/sigma)**2)
        elif family_id == 2:
            # Symmetric exponential decay
            lam = rng.uniform(0.001, 0.02) * n
            a = np.exp(-lam * np.arange(n))
            a = a + a[::-1]
        elif family_id == 3:
            # Power law
            p = rng.uniform(0.3, 3.0)
            a = (np.arange(1, n+1)**p)[::-1]  # decreasing
        elif family_id == 4:
            # Beta-like
            p1 = rng.uniform(0.1,5.0)
            p2 = rng.uniform(0.1,5.0)
            x = np.linspace(0, 1, n)
            a = x**p1 * (1-x)**p2
        elif family_id == 5:
            # sinusoidal power
            p = rng.uniform(0.5,4.0)
            a = np.sin(np.pi*(np.arange(n)+0.5)/n)**p
        elif family_id == 6 and base_seq is not None:
            # upsample base seq
            a = _upsample_seq(base_seq, n)
        else:
            # random piecewise linear (tent)
            apex = rng.integers(1,n-1)
            left = np.linspace(0,1,apex,endpoint=False)
            right = np.linspace(1,0,n-apex)
            a = np.concatenate([left,right])

        a = np.clip(a, 0, None)
        # Normalize to sum sqrt(2n)
        S = np.sqrt(2.0 * n)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0)
            a_sum = a.sum()
        a = a / a_sum * S

        if rng.random() < 0.5:
            a = _symmetrize(a)

        score = max_conv(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[init] n={n} score={score:.6f}")

    # Stage 2: long Adam on best_seq (increase length gradually)
    # We'll potentially upsample then Adam.

    def run_adam_on(seq, time_budget, enforce_sym=False):
        return _optimize_adam(seq, time.time() + time_budget, rng, max_alpha=800.0, enforce_sym=enforce_sym)

    # Maybe we try progressive upsampling: start with current n, then upsample by factor 1.5, 2 etc.

    # Define list of upsample factors
    while time.time() < deadline - 30:  # reserve time for final eval
        cur_n = best_seq.shape[0]
        # decide factor
        factor = rng.uniform(1.2, 1.6)
        new_n = int(min(30000, cur_n * factor))
        # upsample
        up_seq = _upsample_seq(best_seq, new_n)
        # normalize
        S_big = np.sqrt(2.0 * new_n)
        up_seq = up_seq / up_seq.sum() * S_big
        # optionally symmetrize
        if rng.random() < 0.5:
            up_seq = _symmetrize(up_seq)
        # allocate time for Adam
        time_left = deadline - time.time()
        # try to use a chunk for Adam: maybe 0.2 * time_left
        adam_time = min(30.0, 0.2 * time_left)
        a_opt, sc_opt = _optimize_adam(up_seq, time.time() + adam_time, rng, max_alpha=800.0, enforce_sym=False)
        # after Adam, maybe a small anneal
        if time.time() < deadline - 5:
            a_opt2, sc_opt2 = _anneal_hillclimb(a_opt, deadline, rng, enforce_sym=False)
            if sc_opt2 < sc_opt:
                a_opt = a_opt2
                sc_opt = sc_opt2
        # update best
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[upsampled] n={new_n} score={sc_opt:.6f}")

    # final evaluation
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal={best_score:.6f} length={len(final_seq)}")
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq

def _focused_anneal(a0, deadline, rng, enforce_sym=False):
    a = a0.copy(); n = len(a); S = sqrt(2n); best = a; best_score = max_conv(a)
    while time.time() < deadline:
        conv = _full_convolution(a)
        k = np.argmax(conv)
        # contributions: compute list of (i, product)
        i_low = max(0, k - (n-1))
        i_high = min(n-1, k)
        indices = np.arange(i_low, i_high+1)
        j = k - indices
        products = a[indices] * a[j]
        # pick i with max product
        max_idx = np.argmax(products)
        i = int(indices[max_idx])
        j_idx = int(j[max_idx])
        # choose target index l not in {i, j}. maybe random.
        l = rng.integers(0, n)
        while l == i or l == j_idx:
            l = rng.integers(0, n)
        # move delta from i or j whichever bigger
        source = i if a[i] > a[j_idx] else j_idx
        target = l
        if a[source] <= 0:
            continue
        delta = rng.random() * 0.05 * a[source]
        cand = a.copy()
        cand[source] -= delta
        cand[target] += delta
        cand = _project_simplex(cand, S) if enforce_sym else cand # Actually maintain sum: we moved mass, sum unchanged; but need ensure non-negativity; projection maybe not needed.
        # evaluate
        new_score = np.max(_full_convolution(cand))
        # accept if better or with probability
        if new_score <= best_score or rng.random() < 0.1:
            a = cand
            if new_score < best_score:
                best_score = new_score
                best = cand.copy()
    return best, best_score

"""
Improved step‑function optimizer.

Main ideas
----------
1.  Sample a diverse set of initial shapes (uniform, Gaussian, power‑law,
    beta‑like, exponential, sinusoidal, up‑sampled previous best, random
    tent).  Lengths are chosen up to 30 000 – larger sequences give a
    finer “mass distribution” and often lower autocorrelation peaks.

2.  A long projected‑Adam phase minimises a soft‑max approximation of the
    convolution peak.  The temperature (α) is increased slowly so that the
    optimiser first sees a smooth landscape and finally focuses on the
    true maximum.

3.  Progressive up‑sampling – after a promising solution is found the
    sequence is linearly interpolated to a longer length and Adam is run
    again.  This “refinement by smoothing” frequently lowers the peak.

4.  A focused simulated‑annealing / hill‑climb step that deliberately
    reduces the most offending convolution lag by moving a small amount
    of mass from the largest contributing pair to a random location.

All steps respect the constraints `a_i ≥ 0` and `∑ a_i = √(2·n)`.  The
final sequence is stored in the global variable `height_sequence_1` and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the scaled simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    `w` must be a probability vector (soft‑max weights) of length 2·n‑1.
    """
    n = a.shape[0]
    # reverse weights because convolution is commutative and we need
    # the gradient w.r.t. the first argument.
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that correspond to a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce exact mirror symmetry: v[i] = v[-1‑i]."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.shape[0] == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.shape[0])
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 500.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value) where the peak value is the *true*
    maximum convolution entry (not the smoothed one).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)          # required sum

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.035

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights and gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        # projection onto {a ≥ 0, Σa = S}
        a = _project_simplex(a, S)

        # occasional jitter to escape flat spots
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998

    return best_a, best_score


# ----------------------------------------------------------------------
# Focused simulated‑annealing hill‑climb that attacks the current peak
# ----------------------------------------------------------------------
def _focused_anneal(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Reduce the highest convolution entry by targeted mass transfers."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        conv = _full_convolution(a)
        k = int(np.argmax(conv))                     # index of the current peak

        # indices participating in this lag
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k - idx_i

        # contribution of each pair to the peak
        contrib = a[idx_i] * a[idx_j]
        max_pair = int(np.argmax(contrib))
        i = int(idx_i[max_pair])
        j = int(idx_j[max_pair])

        # choose a target index far from both i and j
        target = rng.integers(0, n)
        while target == i or target == j:
            target = rng.integers(0, n)

        # move a small fraction of mass from the larger of (i,j) to target
        src = i if a[i] >= a[j] else j
        if a[src] <= 0:
            continue
        delta = rng.random() * 0.04 * a[src]          # up to 4 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[target] += delta

        if enforce_sym:
            cand = _symmetrize(cand)

        # ensure feasibility (non‑negativity is already satisfied)
        cand = _project_simplex(cand, S) if enforce_sym else cand

        cand_score = np.max(_full_convolution(cand))
        cur_score = best_score

        # accept if improves, otherwise accept with a small probability
        if cand_score < cur_score or rng.random() < 0.07:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function with a minimal self‑convolution peak.
    Returns a list of floats (the heights) before the time limit expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0        # keep a small safety margin

    # ------------------------------------------------------------------
    # Retrieve previous best if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap diverse initial sampling (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start + 0.10 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # pick a length
        if base_seq is not None and rng.random() < 0.35:
            n = int(max(2000, min(30000,
                                 base_seq.shape[0] * rng.uniform(0.8, 1.2))))
        else:
            n = int(rng.integers(5000, 30001))

        # random family selection
        fam = rng.integers(0, 10)
        if fam == 0:
            a = rng.random(n)                               # plain uniform
        elif fam == 1:
            # Gaussian‑like bump
            sigma = rng.uniform(0.09, 0.22) * n
            x = np.arange(n) - (n - 1) / 2.0
            a = np.exp(-0.5 * (x / sigma) ** 2)
        elif fam == 2:
            # Symmetric exponential decay
            lam = rng.uniform(0.001, 0.02) * n
            a = np.exp(-lam * np.arange(n))
            a = a + a[::-1]
        elif fam == 3:
            # Power‑law decreasing
            p = rng.uniform(0.3, 3.0)
            a = (np.arange(1, n + 1) ** p)[::-1]
        elif fam == 4:
            # Beta‑like shape
            p1 = rng.uniform(0.1, 5.0)
            p2 = rng.uniform(0.1, 5.0)
            x = np.linspace(0, 1, n)
            a = x ** p1 * (1 - x) ** p2
        elif fam == 5:
            # Sinusoidal power
            p = rng.uniform(0.5, 4.0)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        elif fam == 6 and base_seq is not None:
            a = _upsample_seq(base_seq, n)                 # up‑sample old best
        elif fam == 7:
            # Logistic‑like S‑curve
            mid = rng.uniform(0.35, 0.65) * n
            steep = rng.uniform(5, 20)
            a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
        elif fam == 8:
            # Piecewise linear tent with random apex
            apex = rng.integers(1, n - 1)
            left = np.linspace(0, 1, apex, endpoint=False)
            right = np.linspace(1, 0, n - apex)
            a = np.concatenate([left, right])
        else:
            # Random mixture of two exponentials
            lam1 = rng.uniform(0.001, 0.015) * n
            lam2 = rng.uniform(0.001, 0.015) * n
            a = np.exp(-lam1 * np.arange(n)) + np.exp(-lam2 * (np.arange(n)[::-1]))

        a = np.clip(a, 0.0, None)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # optional forcing of symmetry
        if rng.random() < 0.4:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long projected‑Adam optimisation on the current best
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a modest uniform sequence
        n = 8000
        a = np.full(n, 1.0)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_seq = a
        best_score = np.max(_full_convolution(a))

    # progressive up‑sampling + Adam refinement
    while time.time() < deadline - 30.0:            # keep ~30 s for final polishing
        cur_n = best_seq.shape[0]
        # decide a modest enlargement (20‑50 %)
        factor = rng.uniform(1.2, 1.5)
        new_n = int(min(30000, cur_n * factor))
        up = _upsample_seq(best_seq, new_n)

        # re‑normalise to the new length
        S_big = np.sqrt(2.0 * new_n)
        up = up / up.sum() * S_big

        # occasional symmetry enforcement (helps the convolution shape)
        if rng.random() < 0.5:
            up = _symmetrize(up)

        # time slice for this Adam run
        remaining = deadline - time.time()
        adam_time = min(20.0, 0.20 * remaining)   # at most 20 s or 20 % of the rest
        a_opt, sc_opt = _optimize_adam(up,
                                        time.time() + adam_time,
                                        rng,
                                        max_alpha=800.0,
                                        enforce_sym=False)

        # quick focused anneal to shave the peak further
        if time.time() < deadline - 5.0:
            a_opt, sc_opt = _focused_anneal(a_opt,
                                            time.time() + 5.0,
                                            rng,
                                            enforce_sym=False)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[refine] upsampled to {new_n} score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final focussed annealing on the best length we have
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_final, sc_final = _focused_anneal(best_seq,
                                            deadline - 5.0,
                                            rng,
                                            enforce_sym=False)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[final‑anneal] score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Store & return the result
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
