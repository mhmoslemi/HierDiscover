"""
Step‑function optimizer for the self‑convolution bound.

The search is divided into several phases that together respect a 1000 s
budget:

1. **Random sampling** – generate candidates from a collection of analytic
   families (geometric, power‑law, sinusoidal, logistic, etc.) and keep the
   best one.

2. **Projected Adam** – smooth‑max approximation of the convolution peak.
   Several Adam runs are performed with different initialisations,
   learning rates and optional symmetry enforcement.

3. **Greedy local optimisation** – directly attacks the current maximal
   convolution entry.  At each iteration the algorithm moves a small amount of
   mass from an index that contributes strongly to the peak toward an index
   that contributes weakly (a simple sub‑gradient descent with Metropolis
   acceptance).

4. **Simulated‑annealing hill‑climb** – a short “anneal‑&‑climb” phase that
   can escape shallow local minima.

5. **Upsampling + short Adam refinement** – increase the sequence length
   (up to 30 k) by linear interpolation and polish the result.

All updates keep the sum of the coefficients equal to √(2 n)
(i.e. the simplex with sum = √(2 n)), which makes the internal objective
exactly the convolution maximum and therefore identical to the public
`evaluate_sequence` score.  The implementation uses FFT‑based full
convolution for speed and works entirely with NumPy (no external I/O).

The function `propose_candidate` returns the best sequence found.
"""
import time
import numpy as np
from scipy import fftpack  # used only for its FFT length helper (optional)

# ----------------------------------------------------------------------
# Projection onto the scaled simplex {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the non‑negative simplex with total sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (linear convolution a*a)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power‑of‑two length for efficient FFT
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) utilities – used by Adam
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv) with respect to a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = 3 * n - 2               # length of a * w_rev
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)      # indices that correspond to a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1‑i]."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Candidate families for the initial random stage
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a variety of analytic families."""
    fam = rng.integers(0, 9)
    if fam == 0 and base_seq is not None:
        # up‑sample an existing good solution
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                 # symmetric power‑law
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                # beta‑like skewed
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                 # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)                 # monotone decreasing power‑law
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n        # exponential decay
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n           # logistic‑like S‑curve
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    else:
        apex = rng.integers(1, n - 1)             # piecewise linear “tent”
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Projected Adam minimising a soft‑max approximation of the peak
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 400.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam optimiser; returns (best_vector, best_peak)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.03

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local optimisation – moves mass from high‑gradient to low‑gradient
# ----------------------------------------------------------------------
def _optimize_greedy(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator,
                     enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Simple greedy descent on the true max‑convolution entry (Metropolis acceptance)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    cur_max = best_score

    start = time.time()
    T0 = max(0.02 * best_score, 1e-3)   # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))  # linear cooling

        conv = _full_convolution(a)
        cur_max = conv.max()
        k_max = int(np.argmax(conv))

        # Build the gradient‑like weights w_i = a[k_max - i] where defined
        idx_i = np.arange(n)
        idx_j = k_max - idx_i
        mask = (idx_j >= 0) & (idx_j < n)
        w = np.zeros_like(a)
        w[mask] = a[idx_j[mask]]

        # source index – large contribution (and non‑zero mass)
        src_candidates = np.where((w == w.max()) & (a > 0))[0]
        if src_candidates.size == 0:
            src = np.argmax(w)
        else:
            src = rng.choice(src_candidates)

        # target index – preferably small contribution
        # pick among the 20 % lowest‑weight positions
        low_thresh = np.percentile(w, 20)
        low_idxs = np.where(w <= low_thresh)[0]
        if low_idxs.size > 0:
            tgt = rng.choice(low_idxs)
        else:
            tgt = rng.integers(0, n)

        if src == tgt:
            continue

        # amount to transfer (2‑8 % of source mass)
        delta_frac = rng.uniform(0.02, 0.08)
        delta = min(delta_frac * a[src], a[src])
        if delta <= 0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        if enforce_sym:
            cand = _symmetrize(cand)
        cand = _project_simplex(cand, S)

        cand_score = np.max(_full_convolution(cand))
        diff = cand_score - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        # occasional random perturbation to improve exploration
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta2 = 0.05 * a[i] * rng.random()
                cand2 = a.copy()
                cand2[i] -= delta2
                cand2[j] += delta2
                cand2 = _project_simplex(cand2, S)
                score2 = np.max(_full_convolution(cand2))
                if score2 < cur_max or rng.random() < 0.15:
                    a = cand2
                    cur_max = score2
                    if score2 < best_score:
                        best_score = score2
                        best_a = cand2.copy()

    return best_a, best_score

def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Simulated‑annealing hill‑climb that swaps tiny masses."""
    a = a0.copy()

    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load any previously stored sequence
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0
    MAX_LEN = 30000
    MIN_LEN = 2000

    # ------------------------------------------------------------------
    # Helper: compute the internal score (max convolution) for a normalized vector
    # ------------------------------------------------------------------
    def _internal_score(v):
        return np.max(_full_convolution(v))

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈12 % of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.12 * budget_s
    while time.time() < phase1_end:
        attempts += 1
        # length selection
        if base_seq is not None and rng.random() < 0.5:
            n = max(MIN_LEN, int(base_seq.shape[0] * rng.uniform(0.9, 1.1)))
        else:
            n = int(rng.integers(MIN_LEN, MAX_LEN + 1))

        a = _sample_family(rng, n, base_seq)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        if rng.random() < 0.4:
            a = _symmetrize(a)

        score = _internal_score(a)
        if score < best_score:
            best_score, best_seq = score, a.copy()
            print(f"[phase1] attempt={attempts} n={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multiple projected‑Adam runs (≈45 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    num_adam_runs = 3
    for run_id in range(num_adam_runs):
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget
        run_deadline = time.time() + 0.3 * remaining
        # initialise a0
        if best_seq is not None and rng.random() < 0.7:
            # sometimes up‑sample / down‑sample the current best
            if rng.random() < 0.6:
                new_n = min(MAX_LEN, int(best_seq.shape[0] * rng.uniform(1.0, 1.3)))
                a0 = _upsample_seq(best_seq, new_n)
            else:
                new_n = max(MIN_LEN, int(best_seq.shape[0] * rng.uniform(0.7, 1.0)))
                start_idx = rng.integers(0, max(1, best_seq.shape[0] - new_n + 1))
                a0 = best_seq[start_idx:start_idx + new_n].copy()
            a0 = a0 + 0.01 * rng.normal(size=a0.shape)   # slight jitter
            a0 = np.clip(a0, 0, None)
        else:
            n = int(rng.integers(MIN_LEN, MAX_LEN + 1))
            a0 = rng.random(n)

        S0 = np.sqrt(2.0 * a0.shape[0])
        a0 = a0 / a0.sum() * S0
        enforce_sym = rng.random() < 0.5

        a_opt, sc_opt = _optimize_adam(a0, run_deadline, rng,
                                      max_alpha=600.0, enforce_sym=enforce_sym)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[adam{run_id}] improved to {best_score:.6f}")

        # Optional short greedy refinement inside the same time slice
        now2 = time.time()
        if now2 < run_deadline - 4.0:
            g_end = run_deadline - 2.0
            a_g, sc_g = _optimize_greedy(a_opt, g_end, rng, enforce_sym=enforce_sym)
            if sc_g < best_score:
                best_score, best_seq = sc_g, a_g.copy()
                print(f"[adam{run_id}+greedy] improved to {best_score:.6f}")

        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # Phase 3 – standalone greedy refinement (≈25 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 5.0:
        greedy_end = deadline - 3.0
        a_g, sc_g = _optimize_greedy(best_seq, greedy_end, rng, enforce_sym=False)
        if sc_g < best_score:
            best_score, best_seq = sc_g, a_g.copy()
            print(f"[greedy] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (≈10 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 4.0:
        a_hc, sc_hc = _anneal_hillclimb(best_seq, deadline, rng, enforce_sym=False)
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[hillclimb] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – up‑sample & short Adam refinement (≈8 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 6.0:
        cur_n = best_seq.shape[0]
        new_n = min(MAX_LEN, int(cur_n * 1.2))
        a_big = _upsample_seq(best_seq, new_n)
        S_big = np.sqrt(2.0 * new_n)
        a_big = a_big / a_big.sum() * S_big
        refine_deadline = deadline - 2.0
        a_ref, sc_ref = _optimize_adam(a_big, refine_deadline, rng,
                                      max_alpha=800.0, enforce_sym=False)
        if sc_ref < best_score:
            best_score, best_seq = sc_ref, a_ref.copy()
            print(f"[upsample+adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
