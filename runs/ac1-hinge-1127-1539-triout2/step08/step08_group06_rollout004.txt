def _lp_step(f, M, S, t):
    n = len(f)
    # Build sparse A matrix (2n-1 x n)
    diags = []
    offsets = []
    for s in range(n):
        diags.append(f)  # shape n
        offsets.append(s)
    A = scipy.sparse.diags(diags, offsets, shape=(2*n-1, n), format='csc')
    b = np.full(2*n-1, M)
    # solve maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return f
        return f
    g = res.x
    # scale to sum = S
    g_sum = g.sum()
    if g_sum == 0:
        g_scaled = f
    else:
        g_scaled = g * (S / g_sum)
    # combine
    new_f = (1-t) * f + t * g_scaled
    # optionally symmetrize
    new_f = _symmetrize(new_f)
    # project onto simplex
    new_f = _project_simplex(new_f, S)
    return new_f
```

"""
Propose a non‑negative step‑function that minimizes the self‑convolution peak.

Algorithm
---------
1. **Initial pool** – generate a few symmetric candidates of modest length
   (n≈2000) using analytic families (uniform, geometric, beta, sinusoid…).
   If a previous solution exists it is down‑sampled and used as one candidate.

2. **Linear‑programming refinement** – for each candidate repeat:
        * M = max (f∗f)
        * Solve the LP
              maximise   Σ g_j
              s.t.       g_j ≥ 0
                         (f∗g)_k ≤ M   for all k
          The convolution constraint is linear; it is built as a sparse
          Toeplitz matrix (2n‑1 × n) and solved with SciPy’s HiGHS driver.
        * Rescale g to the simplex (∑g = √(2n)) and replace
              f ← (1‑t)·f + t·g   (t≈0.2).
          After each step we symmetrize, project onto the simplex and
          recompute M.  A few (≈10) LP steps usually reduce the peak
          substantially.

3. **Smooth optimisation** – the best LP‑refined vector is further
   improved by three stages using only fast FFT‑based convolutions:
        * Adam on a log‑sum‑exp (soft‑max) surrogate of the max‑convolution.
        * Projected sub‑gradient descent on the true max entry.
        * Tiny random‑swap hill‑climbing.
   Time is split proportionally among the three stages.

4. **Upsampling** – the current best vector is linearly interpolated to
   larger lengths (≈1.2×, 1.5×, 2×, capped at 15000).  Each up‑sampled
   version receives a short Adam burst; improvements are kept.

5. **Final local search** – a remaining budget is spent on a last
   random‑swap hill‑climb.

6. Store the final sequence globally as ``height_sequence_1`` and return it.

All internal evaluations use an FFT‑based convolution (O(n log n)).
Only the final candidate is sent to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sparse

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Peak of the self‑convolution (internal metric)
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Simple symmetry helper (optional)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear interpolation (preserves shape)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families for cheap exploration
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    fam = rng.integers(0, 7)
    if fam == 0 and base is not None:
        a = _upsample_seq(base, n)
    elif fam == 1:
        a = rng.random(n)
    elif fam == 2:
        rate = rng.uniform(0.92, 0.9995)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) * 0.5
    elif fam == 3:
        p = rng.uniform(0.2, 4.5)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    else:
        scale = rng.uniform(0.02, 0.15) * n
        mid = (n - 1) / 2.0
        xs = np.arange(n, dtype=float)
        left = 1.0 / (1.0 + np.exp((xs - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - xs) / scale))
        a = left + right
    return np.clip(a, 0.0, None)


# ----------------------------------------------------------------------
# Smooth‑max helpers for Adam stage
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.05
    max_alpha = 800.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.18:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        if it % 9000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb (local refinement)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _peak_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximising Σg subject to f∗g ≤ M)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               t: float,
               iters: int,
               deadline: float,
               rng: np.random.Generator) -> np.ndarray:
    """
    Perform a few LP steps starting from f.
    Returns the refined vector (still on the simplex).
    """
    n = f.shape[0]
    S = math.sqrt(2.0 * n)

    M = _peak_convolution(f)

    for _ in range(iters):
        if time.time() >= deadline:
            break

        # Build sparse Toeplitz matrix for the convolution constraint f∗g ≤ M
        A = sparse.diags([f] * n, offsets=range(n),
                         shape=(2 * n - 1, n), format='csc')
        b = np.full(2 * n - 1, M, dtype=float)
        c = -np.ones(n, dtype=float)          # maximise Σg ⇒ minimise -Σg

        res = opt.linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})
        if not res.success:
            # If the LP fails we abort further refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g > 0:
            g = g * (S / sum_g)

        # convex combination
        f = (1 - t) * f + t * g

        # optional symmetrisation for stability
        if rng.random() < 0.5:
            f = _symmetrize(f)

        f = _project_simplex(f, S)
        M_new = _peak_convolution(f)

        # stop early if the peak does not improve
        if M_new >= M - 1e-9:
            M = M_new
            continue
        M = M_new

    return f


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from previous best if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – build a small elite pool of cheap candidates (n≈2000)
    # ------------------------------------------------------------------
    n0 = 2000
    K = 4
    elite = []
    for i in range(K):
        if base_seq is not None and i == 0:
            a = _upsample_seq(base_seq, n0)
        else:
            a = _gen_candidate(n0, rng, base_seq)
        S0 = math.sqrt(2.0 * n0)
        if a.sum() == 0:
            a = np.full(n0, 1.0)
        a = a / a.sum() * S0
        if rng.random() < 0.7:
            a = _symmetrize(a)
        score = _peak_convolution(a)
        elite.append((score, a))
        print(f"[init] idx={i} n={n0} score={score:.6f}")

    elite.sort(key=lambda x: x[0])
    best_overall_seq = None
    best_overall_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – LP refinement + smooth optimiser for each elite member
    # ------------------------------------------------------------------
    for idx, (init_score, vec) in enumerate(elite):
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now
        # split remaining time among the still‑to‑process elites
        per_elite = remaining / (len(elite) - idx)

        # ---- LP refinement (≈25% of per_elite) ----
        lp_dead = now + 0.25 * per_elite
        f = _lp_refine(vec, t=0.2, iters=12, deadline=lp_dead, rng=rng)

        # ---- Adam → Sub‑grad → Local swap (≈70%) ----
        now = time.time()
        if now >= deadline:
            break
        time_for_opt = min(deadline, now + 0.70 * per_elite)

        # Adam stage
        ada_dead = now + 0.55 * (time_for_opt - now)
        a_adam, _ = _optimize_adam(f, ada_dead, rng, enforce_sym=False)

        # Sub‑gradient stage
        sub_dead = ada_dead + 0.30 * (time_for_opt - ada_dead)
        a_sub, _ = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=False)

        # Tiny swap stage
        swap_dead = sub_dead + 0.15 * (time_for_opt - sub_dead)
        a_loc, loc_score = _local_swap_search(a_sub, swap_dead, rng, enforce_sym=False)

        cand_score = _peak_convolution(a_loc)
        if cand_score < best_overall_score:
            best_overall_score = cand_score
            best_overall_seq = a_loc.copy()
            print(f"[elite {idx}] new best internal={best_overall_score:.6f} n={n0}")

    # ------------------------------------------------------------------
    # Phase 2 – Upsampling to larger lengths and quick Adam pass
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 15000
        up_factors = [1.2, 1.5, 2.0]
        for fac in up_factors:
            now = time.time()
            if now >= deadline:
                break
            new_n = int(best_overall_seq.shape[0] * fac)
            if new_n > max_len or new_n < 10:
                continue
            a_up = _upsample_seq(best_overall_seq, new_n)
            Snew = math.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * Snew

            # short Adam burst (≈8% of remaining time)
            remaining = deadline - now
            short_dead = now + min(0.08 * remaining, 5.0)
            a_up_opt, _ = _optimize_adam(a_up, short_dead, rng, enforce_sym=False)

            up_score = _peak_convolution(a_up_opt)
            if up_score < best_overall_score:
                best_overall_score = up_score
                best_overall_seq = a_up_opt.copy()
                print(f"[upsample] factor={fac:.2f} n={new_n} new internal={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final local hill‑climb with any remaining budget
    # ------------------------------------------------------------------
    if best_overall_seq is not None and time.time() < deadline:
        a_fin, fin_score = _local_swap_search(best_overall_seq, deadline, rng, enforce_sym=False)
        if fin_score < best_overall_score:
            best_overall_score = fin_score
            best_overall_seq = a_fin.copy()
            print(f"[final] improved internal={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return result (store globally for warm‑restarts)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[return] external={external:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)}")
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
