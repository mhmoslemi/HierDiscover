def propose_candidate(budget_s=1000, **kwargs):
    import time, numpy as np, math, random
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine n based on remaining time
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)
    # Use warm start: if height_sequence_1 exists, use it resized else uniform (or power law)
    if "height_sequence_1" in globals():
        a = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if a.size != n:
            a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a.size), a)
    else:
        a = np.full(n, target_sum / n, dtype=np.float64)
    # Ensure sum exactly target_sum
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)
    print(f"[{time.time():.2f}] initial score {best_val:.6f}")

    # Simulated annealing parameters
    temp0 = best_val * 0.5
    # scaling factor for mass move
    base_frac = 0.1  # maximum fraction of donor mass to move

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # linearly decreasing temperature
        t = (deadline - time.time()) / (deadline - start)  # fraction left
        temp = temp0 * t

        # Pick donor and receiver
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Compute delta: random fraction of donor's mass up to base_frac
        max_delta = a[i] * base_frac
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # propose new a
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # compute new score
        val = _score(a_candidate)
        # acceptance
        if val < best_val or rng.random() < math.exp(-(val - best_val) / (temp + 1e-12)):
            a = a_candidate
            if val < best_val:
                best_val = val
                best = a.copy()
                print(f"[{time.time():.2f}] it {it} new best {best_val:.6f}")
        it += 1
        # occasional rebalancing to keep sum exact (floating errors)
        if it % 1000 == 0:
            cur_sum = a.sum()
            if abs(cur_sum - target_sum) > 1e-10:
                a = a * (target_sum / cur_sum)

    # Final cleanup: truncate trailing near-zero
    nz = np.where(best > 1e-12)[0]
    if nz.size:
        best = best[: nz[-1] + 1]

    final_seq = best.tolist()
    # store for warm start
    globals()["height_sequence_1"] = final_seq
    # evaluate
    try:
        sc = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {sc:.6f}")
    except Exception as e:
        print("Eval error:", e)
    return final_seq

"""Propose a step‑function with a very low benchmark score.

The method builds on the best sequence found in previous runs
(`height_sequence_1` if present) and then performs a temperature‑controlled
random walk (simulated annealing).  In each iteration we

1. compute the current full self‑convolution,
2. identify the index `k_max` where the convolution attains its maximum,
3. measure the contribution of each coefficient to that entry,
4. move a small fraction of mass from a high‑contributing index to a
   low‑contributing one,
5. accept the move if it improves the score or, with a probability that
   decreases with a linear cooling schedule, if it is slightly worse.

The move preserves the L¹‑norm exactly, respects non‑negativity, and
the temperature schedule lets the algorithm escape shallow local minima.
When the time budget expires the best sequence seen so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Evaluation function used by the benchmark:
        2·n·max(a*a) / (sum(a)²)
    The sequence is assumed to be non‑negative; the routine is cheap
    (single FFT) and therefore suitable for many inner‑loop calls.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a small safety margin

    # --------------------------------------------------------------
    # Choose problem size (the benchmark is scale‑invariant, but n≈12000
    # is a good trade‑off between solution quality and speed).
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)                # the conventional normalisation

    # --------------------------------------------------------------
    # Warm‑start: either the previously saved sequence or a simple
    # power‑law (which is already a decent starting point).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(globals()["height_sequence_1"], dtype=np.float64)
        a = _resize_sequence(a, n)
    else:
        # a_i = (i+1)^(-0.55) normalised to the target sum
        i = np.arange(1, n + 1, dtype=np.float64)
        a = i ** (-0.55)
        a = a * (target_sum / a.sum())

    # Exact normalisation (preserves the value of the objective)
    a = a * (target_sum / a.sum())

    best_seq = a.copy()
    best_val = _score(best_seq)
    print(f"[{time.time():.2f}] initial score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing loop
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    max_frac = 0.03                                 # at most 3 % of donor mass per move
    it = 0
    # initial temperature – a fraction of the current score
    T0 = best_val * 0.15 if best_val > 0 else 0.1

    while time.time() < deadline:
        # --- compute current convolution and locate the worst entry
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # --- contribution of each index to that entry
        #    (a[i] participates only if 0 ≤ k_max‑i < n)
        weight = np.zeros(n, dtype=np.float64)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        weight[idx] = a[idx] * a[k_max - idx]

        # --- pick a donor (high contribution) and a receiver (low contribution)
        total_w = weight.sum()
        if total_w > 0.0:
            donor = rng.choice(n, p=weight / total_w)
        else:
            donor = rng.integers(n)

        low_thr = np.percentile(weight, 20)          # 20 % percentile → “low” contributors
        low_candidates = np.where(weight <= low_thr)[0]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --- move a small amount of mass
        max_delta = a[donor] * max_frac
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        # --- evaluate
        cand_val = _score(a_candidate)

        # --- temperature schedule (linear cooling)
        remaining = deadline - time.time()
        total = deadline - start_time
        T = T0 * max(remaining / total, 0.0)

        # --- Metropolis acceptance
        if cand_val < best_val or rng.random() < math.exp(-(cand_val - best_val) / (T + 1e-12)):
            a = a_candidate
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a.copy()
                print(f"[{time.time():.2f}] it {it:6d} new best = {best_val:.6f}")

        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    # --------------------------------------------------------------
    # Final clean‑up  – drop trailing zeros (they do not affect the score)
    # --------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    final_seq = best_seq.tolist()

    # Store for warm‑starts in a later call
    globals()["height_sequence_1"] = final_seq

    # --------------------------------------------------------------
    # Final evaluation (for logging only – the judge calls evaluate_sequence
    # separately as well)
    # --------------------------------------------------------------

        print(f"[{time.time():.2f}] final evaluated score = {final_score:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] evaluation error: {exc}")

    return final_seq
