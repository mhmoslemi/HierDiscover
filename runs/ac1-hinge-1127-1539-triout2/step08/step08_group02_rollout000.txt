"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Broad random exploration** – many short runs using a variety of
   parametric families (power‑law, exponential, linear, gamma,
   U‑shape, edge‑spikes, Dirichlet).  Lengths range from a few thousand
   up to ~50 000.  The best candidate (lowest max self‑convolution) is
   kept.

2. **Geometric length growth** – the current best sequence is repeatedly
   interpolated to a larger length (factor≈1.5) and refined with a
   dedicated Adam optimiser that works directly on the simplex via a
   log‑softmax parametrisation.  Each growth step gets a modest time slice
   (up to a few minutes).  The process stops when the time budget or a
   maximal length (≈50 000) is reached.

3. **Final long Adam run** – the best length found is optimised for the
   remaining time budget.

4. **Cheap stochastic hill‑climb** – a few seconds of random mass
   transfers between positions are performed; any improvement is kept.

5. **Sanity check** – the final sequence is evaluated with the provided
   ``evaluate_sequence`` function and stored in the global variable
   ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the τ‑soft‑max of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b               # ≤0
        exps = np.exp(diff / tau)      # stable because diff≤0
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of max self‑convolution.
    a0 must already satisfy the ℓ₁‑norm ‖a0‖₁ = sqrt(2·n) and be non‑negative.
    Returns the best vector seen and its (hard) peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # target ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                           # aggressive start

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)            # start with mild smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient (τ‑soft‑max)
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability (scale down only if really large)
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 10.0 * S:
            grad_p = grad_p * (10.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules (slow decay)
        tau *= 0.9998                # keep a little smoothing longer
        lr *= 0.9999                 # gentle decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_edge_spikes(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Two pronounced spikes at the edges, low background elsewhere.
    """
    left = rng.uniform(0.5, 6.0)
    right = rng.uniform(0.5, 6.0)
    interior = rng.uniform(0.0, 0.2)
    raw = np.full(n, interior)
    raw[0] = left
    raw[-1] = right
    # random small wiggle in the interior
    wiggle = rng.uniform(0.0, 0.1, size=n - 2)
    raw[1:-1] = interior + wiggle
    return raw


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    # simple Dirichlet‑like random start
    raw = rng.random(n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from the globally best sequence,
    otherwise a random parametric family is drawn.
    """
    S = np.sqrt(2.0 * n)

    # Warm start
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    styles = ['powerlaw', 'exponential', 'linear',
              'gamma', 'ushape', 'edge_spikes',
              'dirichlet', 'uniform']
    style = rng.choice(styles)

    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'edge_spikes':
        raw = _init_edge_spikes(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:   # uniform
        raw = _init_uniform(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list and stores it in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    # keep a 5 s safety margin for the final sanity check
    end_time = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – random exploration (~65 % of total budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 250.0   # reserve ~250 s for refinement
    while time.time() < explore_deadline:
        attempts += 1
        # length distribution – favour a few thousand, but occasionally try very
        # large sizes (up to 50 k).  Larger sizes are slower, so they are rarer.
        r = rng.random()
        if r < 0.70:
            n = int(rng.integers(1500, 8000))
        elif r < 0.95:
            n = int(rng.integers(8000, 20000))
        else:
            n = int(rng.integers(20000, 50000))
        if n < 2:
            n = 2

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – geometric length growth with Adam refinement
    # ------------------------------------------------------------------
    max_n = 50000
    factor = 1.5
    while time.time() < end_time - 120.0:   # keep ~2 min for final phases
        remaining = end_time - time.time()
        # decide next target length
        target_n = int(min(best_n * factor, max_n))
        if target_n <= best_n:
            break

        # prepare scaled start vector
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

        # allocate a slice of time (up to 120 s) for this length
        slice_time = min(remaining * 0.20, 120.0)
        a_opt, peak = _adam_optimize(a_scaled,
                                      time.time() + slice_time,
                                      rng)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[grow] target_n={target_n:5d}  improved score={best_score:.6f}")

        # continue growing
        best_n = target_n

    # ------------------------------------------------------------------
    # Phase 3 – final long Adam optimisation on the best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:   # leave time for hill‑climb & check
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final adam] length={best_n:5d}  best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount

        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:   # accept strict improvement

            best_seq = a.copy()
            print(f"[hill] improved score={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()

    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
