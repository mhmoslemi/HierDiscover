def max_conv(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    return conv.max()

    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    rng = np.random.default_rng()
    # we may precompute zero positions to limit swaps; but we can random over all.
    # Temperature schedule: use exponential decay based on progress.
    while time.time() < deadline:

        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time)
        # start temperature maybe 0.2, end 1e-4

        end_temp = 1e-4

def _alternating_order(seq):
    sorted_seq = np.sort(seq)[::-1]  # descending
    n = len(seq)
    out = np.empty_like(seq)
    left = 0
    right = n-1
    toggle = True
    for val in sorted_seq:
        if toggle:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
        toggle = not toggle
    return out

"""
Propose a low‑score step function.

The algorithm combines several complementary optimisation stages:

1. **Initial pool** – a few power‑law sequences, a resized previous best,
   and random jittered copies.  All candidates are projected onto the ℓ¹‑simplex
   {x ≥ 0, Σx = √(2 n)}.

2. **Projected gradient descent** – a momentum based optimiser on a soft‑max
   approximation of the convolution maximum.  The soft‑max temperature β is
   annealed up to 1e5 and the step size is adapted to the gradient norm.
   Several short restarts with jitter are performed and the best result is
   retained.

3. **Deterministic mass‑transfer** – a Metropolis style greedy move that
   transfers a small amount of mass from a “donor’’ index (large contribution
   to the current maximal convolution entry) to a “receiver’’ index (small
   contribution).  This directly attacks the peak of the convolution.

4. **Re‑ordering heuristics** – the multiset of values is reordered by
   alternating large/small entries (and a few random shuffles).  Since the
   self‑convolution depends on the ordering, this step can already lower the
   peak.

5. **Permutation annealing** – simulated‑annealing over swaps of two
   positions.  Swaps are accepted if they improve the peak or with a
   probability that decays exponentially with a temperature schedule.
   This explores the combinatorial space of orderings.

6. **Final polishing** – a last high‑β projected gradient descent (β up to
   3·10⁵) cleans up any remaining irregularities.

All heavy linear‑algebra uses FFT‑based convolutions (O(n log n)) and the
algorithm respects the 1000 s time budget.  The returned list contains the
non‑negative heights of the step function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective: 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this reduces to max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected gradient descent (enhanced)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short gradient‑descent runs with random jitter.
    Keeps the best result found before the given deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# Phase B – Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly target the current maximal
    convolution entry.  Moves are accepted with a Metropolis rule whose
    temperature decays exponentially with the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    # initial convolution
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # transfer amount (bounded)
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder a sequence by placing the largest value at the leftmost position,
    the second largest at the rightmost, the third largest next to the left,
    and so on – i.e. an alternating “big–small’’ pattern.
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over swaps of two indices.  The sum is invariant,
    therefore only the ordering is altered.  The current temperature decays
    exponentially from `start_temp` to `end_temp`.  The best ordering found
    is returned.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    # annealing schedule parameters
    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full reshuffle to avoid stagnation
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm proceeds through gradient descent, deterministic
    mass‑transfer, ordering heuristics, permutation annealing and a final
    polishing gradient pass.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build initial candidates
    # --------------------------------------------------------------
    init_seqs = []

    # a few power‑law shapes
    for p in (0.45, 0.5, 0.55, 0.6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # resized previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered variants of the above (adds diversity)
    rng = np.random.default_rng()
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # random permutations of a couple of power‑law candidates
    for p in (0.5, 0.55):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # --------------------------------------------------------------
    # Phase A – Projected gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.30

    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        a_cand, val = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
        if val < best_score:
            best_score = val
            best_seq = a_cand.copy()
            print(f"[{time.time():.2f}] New global best after GD phase = {best_score:.6f}")

    if best_seq is None:  # fallback
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum, start_temp=0.5, end_temp=1e-4, low_frac=0.05
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Re‑ordering heuristics (alternating large/small)
    # --------------------------------------------------------------
    # Try a few simple reorderings that keep the multiset unchanged.
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    # a couple of random shuffles for good measure
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation annealing (swap‑based simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(
            best_seq, target_sum, perm_deadline, init_score=best_score
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing gradient descent
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=3e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing (trim trailing zeros) and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for possible later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
