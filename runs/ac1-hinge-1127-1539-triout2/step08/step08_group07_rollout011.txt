def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose n
    n = 12000 if budget_s>=800 else 8000 if budget_s>=400 else 5000
    target_sum = np.sqrt(2*n)

    # initial candidates
    candidates = []
    for p in [0.45,0.5,0.55,0.6,0.65]:
        candidates.append(_power_law_sequence(n,p,target_sum))
    # exponential shape
    for c in [0.001,0.0025,0.005]:
        seq = np.exp(-c*np.arange(n))
        seq *= target_sum/seq.sum()
        candidates.append(seq)
    # uniform small
    seq = np.full(n, target_sum/n)
    candidates.append(seq)
    # add jittered variations
    rng = np.random.default_rng()
    for seq in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        cand = np.maximum(seq + jitter, 0)
        cand = _project_to_simplex(cand, target_sum)
        candidates.append(cand)

    best_seq = None
    best_val = np.inf

    # Phase A: softmax gradient descent with increased runs
    gd_end = time.time() + (deadline - time.time()) * 0.45
    for init in candidates:
        if time.time() >= gd_end:
            break
        a, val = _gradient_descent_with_restarts_mod(init, target_sum, gd_end)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f'New best after GD: {best_val:.6f}')

    # Phase D: subgradient descent (exact max) on best_seq
    if time.time() < deadline - 1.0:
        a, val = _subgradient_descent_exact(best_seq, target_sum, deadline)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f'New best after subgrad: {best_val:.6f}')

    # Phase B: deterministic mass transfer (maybe again)
    if time.time() < deadline - 1.0:
        a,best_val2 = _deterministic_mass_transfer(best_seq, deadline, target_sum)
        if best_val2 < best_val:
            best_val = best_val2
            best_seq = a.copy()
            print(f'Best after mass transfer: {best_val:.6f}')

    # Phase C: final polish with higher beta gradient descent
    if time.time() + 2 < deadline:
        a, val = _annealed_gradient_descent_v2_ext(best_seq, deadline, target_sum,
                                                   max_beta=5e5, step_factor=0.5, gamma=0.99)
        if val < best_val:
            best_val=val
            best_seq=a.copy()
            print(f'Best after final polish: {best_val:.6f}')

    final_seq = best_seq[:np.argmax(best_seq>1e-12)+1].tolist()

    try:
        eva = evaluate_sequence(final_seq)
        print(f'Final evaluate: {eva}')
    except Exception as e:
        pass

    return final_seq

def _gaussian_sequence(n, sigma, target_sum):
    i = np.arange(n)
    mu = (n - 1) / 2.0
    a = np.exp(- ((i - mu) ** 2) / (2.0 * sigma ** 2))
    a *= target_sum / a.sum()
    return a

def _exp_sequence(n, c, target_sum):
    a = np.exp(-c * np.arange(n))
    a *= target_sum / a.sum()
    return a

def _triangular_sequence(n, target_sum):
    a = np.minimum(np.arange(1, n+1), np.arange(n,0,-1))
    a = a.astype(np.float64)
    a *= target_sum / a.sum()
    return a

exponentials, Gaussians, triangular, uniform), refines them with
projected Adam, then directly attacks the active convolution entries
with a sub‑gradient step, and finally cleans the result with a
high‑β gradient pass.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities – FFT based for speed
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a)/(∑a)² (reduces to max‐convolution)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps Σa = const, a_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of candidate step functions
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, c: float, target_sum: float) -> np.ndarray:
    a = np.exp(-c * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(np.float64)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a_rev ⋆ p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_beta: float = 1e5,
                  base_lr: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam with Euclidean simplex projection.
    Uses the soft‑max surrogate of the max‑convolution and
    exponentially increases β (temperature) to approach the true max.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # scalar step size that adapts to the magnitude of v̂
        step = base_lr * target_sum / (np.sqrt(v_hat).mean() + 1e-12)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # β schedule
        if t % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


def _gradient_descent_with_adam_restarts(init: np.ndarray,
                                         target_sum: float,
                                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short Adam runs with random jitter.
    Keeps the best result seen before the deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(25.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_descent(start,
                               target_sum,
                               time.time() + run_budget,
                               max_beta=2e5,
                               base_lr=0.12)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution (β → ∞)
# ----------------------------------------------------------------------


def _subgradient_descent_exact(a0: np.ndarray,
                               target_sum: float,
                               deadline: float,
                               eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Uses the active set of indices attaining the current maximum.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    step = target_sum * 0.5          # initial step size (will decay)
    rng = np.random.default_rng()
    it = 0

    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_rev = a[::-1]                  # will be updated every iteration

    while time.time() < deadline:
        # convolution and active set
        conv = np.fft.irfft(np.fft.rfft(a, n=size) *
                            np.fft.rfft(a, n=size), n=size)[: 2 * n - 1]
        M = conv.max()
        active = conv >= M - eps
        if not np.any(active):
            active[np.argmax(conv)] = True

        # gradient = 2 * (a_rev ⋆ active)
        p = active.astype(np.float64)
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = step / norm
        else:
            lr = 0.0

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        new_val = _score(a_new)

        # simple acceptance rule (always accept improvements)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        # Metropolis‑like occasional acceptance to avoid stagnation
        if new_val <= _score(a) or rng.random() < 0.01:
            a = a_new

        # decay step size slowly
        step *= 0.9997
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (as in the original code)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly target the current maximal convolution entry.
    Uses a simple Metropolis acceptance rule with exponential temperature decay.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            # occasional global jitter
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Final polishing – high‑β annealed gradient descent (from original)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 2e5,
                                     step_factor: float = 0.6,
                                     gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation,
    meant as a final polishing phase with a very large β.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_polish it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The algorithm:
      1. Build a diverse pool of initial shapes (power‑law, exponential,
         Gaussian, triangular, uniform) and jittered copies.
      2. Phase A – Adam‑based soft‑max gradient descent with many short
         restarts.
      3. Phase B – Exact sub‑gradient descent on the true max convolution.
      4. Phase C – Deterministic mass‑transfer refinement.
      5. Phase D – Final high‑β polishing.
    Returns the best sequence found before the time budget expires.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------- problem size -------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------- generate candidates -------------------
    candidates = []

    # power‑law
    for p in (0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # exponentials
    for c in (0.0005, 0.001, 0.002, 0.003, 0.005):
        candidates.append(_exp_sequence(n, c, target_sum))

    # Gaussians (σ as a fraction of n)
    for frac in (0.10, 0.12, 0.14, 0.16, 0.18, 0.20):
        sigma = max(1.0, n * frac)
        candidates.append(_gaussian_sequence(n, sigma, target_sum))

    # triangular and uniform
    candidates.append(_triangular_sequence(n, target_sum))
    candidates.append(np.full(n, target_sum / n))

    # jittered copies to increase diversity
    rng = np.random.default_rng()
    jittered = []
    for seq in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    candidates.extend(jittered)

    # ------------------- Phase A: Adam gradient descent -------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.45
    best_seq = None
    best_score = np.inf

    for init in candidates:
        if time.time() >= phase_a_deadline:
            break
        a_cand, val = _gradient_descent_with_adam_restarts(init,
                                                          target_sum,
                                                          phase_a_deadline)
        if val < best_score:
            best_score = val
            best_seq = a_cand.copy()
            print(f"[{time.time():.2f}] Global best after Phase A = {best_score:.6f}")

    if best_seq is None:                     # sanity fallback
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # ------------------- Phase B: exact sub‑gradient descent -------------------
    if time.time() < deadline - 2.0:
        phase_b_deadline = time.time() + (deadline - time.time()) * 0.20
        a_sub, val_sub = _subgradient_descent_exact(best_seq,
                                                    target_sum,
                                                    phase_b_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] New best after sub‑gradient = {best_score:.6f}")

    # ------------------- Phase C: deterministic refinement -------------------
    if time.time() < deadline - 2.0:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-4,
                                                      low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Best after DMT = {best_score:.6f}")

    # ------------------- Phase D: final high‑β polishing -------------------
    if time.time() + 1.5 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2_ext(best_seq,
                                                            deadline,
                                                            target_sum,
                                                            max_beta=4e5,
                                                            step_factor=0.5,
                                                            gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polished best = {best_score:.6f}")

    # ------------------- post‑processing -------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
