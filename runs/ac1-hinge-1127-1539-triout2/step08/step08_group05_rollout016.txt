import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, csc_matrix

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the scaled probability simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1‑i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Family sampler – used for the initial random candidate
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Generate a random sequence using a variety of analytic families."""
    fam = rng.integers(0, 10)
    if fam == 0 and base_seq is not None:
        # up‑sample an existing good solution (or down‑sample if n < len(base_seq))
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                 # symmetric power‑law
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    else:   # fam == 9 – Gaussian‑like bump
        sigma = rng.uniform(0.08, 0.22) * n
        mid = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n) - mid) / sigma) ** 2)

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Build Toeplitz‑style convolution matrix for the LP step
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> csc_matrix:
    """
    Return a (2·n‑1) × n sparse matrix A such that A @ g = (a*g) (full linear conv).
    Each diagonal offset k (0 ≤ k < n) contains the constant a[k].
    """
    n = a.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    # each diagonal is a constant vector a[k] of length n
    data = [np.full(n, a_k, dtype=np.float64) for a_k in a]
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max approximation of the convolution peak
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 2000.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the smooth (soft‑max) approximation of the
    convolution peak.  Returns the best sequence seen and its peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.04

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()

        # gradient of the smooth objective (see original code)
        # we compute conv of a with reversed weight vector
        w_rev = w[::-1]
        fft_len = 1 << ((3 * n - 2) - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        Fw = np.fft.rfft(w_rev, n=fft_len)
        full = np.fft.irfft(Fa * Fw, n=fft_len)[:3 * n - 2]
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to shake out flat regions
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # temperature schedule
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point – orchestrates the whole search
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising evaluate_sequence().
    Returns the best list of heights found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # 5‑second safety margin

    # ------------------------------------------------------------------
    # Load a previously found sequence (if any) and maybe down‑sample it
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – initialise a modest‑size candidate (n ≈ 2000‑2500)
    # ------------------------------------------------------------------
    n0 = int(rng.integers(1900, 2501))
    if base_seq is not None:
        a = _upsample_seq(base_seq, n0)               # interpolation works both ways
    else:
        a = _sample_family(rng, n0, None)

    # normalise to Σa = √(2·n)
    a = _project_simplex(a, np.sqrt(2.0 * n0))
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    print(f"[init] n={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP refinement (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    t_blend = 0.5
    lp_iters = 0
    while time.time() < phase1_deadline and lp_iters < 60:
        lp_iters += 1

        # current peak and convolution matrix
        conv = _full_convolution(a)
        M = conv.max()
        if M < best_score:
            best_score = M
            best_a = a.copy()
            print(f"[lp] iter={lp_iters} n={n0} M={M:.6f}")

        # build the linear constraints A g ≤ M
        A = _build_conv_matrix(a)
        b_ub = np.full(2 * n0 - 1, M, dtype=np.float64)
        c = -np.ones(n0, dtype=np.float64)          # maximise Σg ⇔ minimise -Σg
        bounds = [(0, None) for _ in range(n0)]

        # solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'disp': False})
        if not res.success:
            print("[lp] linprog failed – abort LP stage")
            break

        g = res.x
        # blend and re‑project onto the simplex
        a = (1.0 - t_blend) * a + t_blend * g
        a = _symmetrize(a)               # symmetry helps the peak
        a = _project_simplex(a, np.sqrt(2.0 * n0))

        # gradually shrink the blending factor
        t_blend = max(t_blend * 0.93, 0.05)

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement on the short sequence (≈25 % of budget)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        t_adam_end = now + 0.25 * (deadline - now)
        a_adam, sc_adam = _optimize_adam(best_a, t_adam_end, rng,
                                         max_alpha=2000.0, enforce_sym=True)
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam.copy()
            print(f"[adam‑short] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Upsample to a larger length and perform a final Adam run
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 5.0:
        cur_n = best_a.shape[0]
        target_n = min(30000, int(cur_n * 1.5))
        a_big = _upsample_seq(best_a, target_n)
        a_big = _project_simplex(a_big, np.sqrt(2.0 * target_n))

        # final Adam refinement (the remaining time)
        t_final_end = deadline
        a_final, sc_final = _optimize_adam(a_big, t_final_end, rng,
                                           max_alpha=2500.0, enforce_sym=True)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"[adam‑final] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Prepare the final answer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)}")

    # expose globally for possible warm‑starts in future runs
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
