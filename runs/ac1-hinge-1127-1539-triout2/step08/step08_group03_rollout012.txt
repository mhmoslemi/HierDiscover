Implementation:

def project_onto_simplex(v, S):
    # v is numpy array
    # shift and clamp
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    end_time = time.time() + budget_s - 5

    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # Phase 1: random exploration
    top_k = 10
    top_candidates = []  # list of (score, seq)
    best_score = np.inf
    best_seq = None

    attempt = 0
    # allocate about 30% of time for random
    explore_deadline = end_time - (budget_s * 0.55)  # roughly 45% for refinement
    while time.time() < explore_deadline:
        attempt += 1
        # pick length with preference for moderate lengths
        if rng.random() < 0.7:
            n = int(rng.integers(2000, 8000))
        else:
            n = int(rng.integers(8000, 20000))
        a0 = _initial_vector_with_extra(n, rng, warm_seq)  # incorporate new families
        peak = _max_convolution(a0)
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()

# -*- coding: utf-8 -*-
"""
Improved step‑function optimizer.

The algorithm proceeds in three phases:

1️⃣ Random exploration of a rich family of parametric sequences.
2️⃣ Multi‑scale Adam refinement (log‑softmax parametrisation) with
   diversified hyper‑parameters.
3️⃣ Exact sub‑gradient descent on the simplex (Polyak step) for
   fine‑tuning, followed by a cheap simulated‑annealing walk.

The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum x = S}."""
    # based on the algorithm from
    #  https://arxiv.org/abs/1101.6081
    n = v.shape[0]
    if S < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.2, 5.0)
    b = rng.uniform(0.2, 5.0)
    scale = rng.uniform(0.2, 5.0)
    raw = rng.beta(a, b, size=n) * scale
    return raw


def _init_loguniform(n: int, rng: np.random.Generator) -> np.ndarray:
    lo = np.log(rng.uniform(0.01, 0.2))
    hi = np.log(rng.uniform(1.0, 10.0))
    raw = np.exp(rng.uniform(lo, hi, size=n))
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With some probability we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'beta', 'loguniform'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'beta':
        raw = _init_beta(n, rng)
    else:   # log‑uniform
        raw = _init_loguniform(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Gradient of τ‑soft‑max (used by Adam)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Minimise the exact max‑convolution of ``a`` under the ℓ₁‑norm
    constraint.  Returns the best vector seen and its *hard* peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0  (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution & gradient (soft max) -------
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (very cheap, used only if time left)
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score        # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        # transfer a random fraction of the mass at i to j
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # re‑normalise (keeps ℓ₁‑norm exactly S)
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimizer (Polyak step) on the simplex
# ----------------------------------------------------------------------


def _subgradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the simplex for the exact max‑convolution.
    Returns the best vector seen and its peak value.
    """
    a = a_start.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # exact convolution and maximal value
        b = _full_convolution_fft(a)
        max_b = b.max()

        # mask for active (maximal) entries
        mask = (b == max_b).astype(float)

        # exact gradient of the max entry (or average if many)
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]   # ∇ max_b

        # Polyak step towards the theoretical lower bound (≈1)
        norm2 = np.dot(g, g) + 1e-12
        step = (max_b - 1.0) / norm2
        # protect against too tiny / too large steps
        step = max(min(step, 5.0), 1e-6)

        a_new = a - step * g
        a_new = _project_onto_simplex(a_new, S)

        new_val = _max_convolution(a_new)

        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()

        # accept if we improved, otherwise keep the old point
        if new_val < max_b:
            a = a_new
        else:
            # occasional jitter to avoid stagnation
            if it % 250 == 0:
                a = a + rng.normal(scale=0.001, size=n)
                a = np.maximum(a, 0.0)
                a = _project_onto_simplex(a, S)

    return best_a, best_val

    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()

            a_scaled = np.clip(a_scaled, 0.0, None)
            S_target = np.sqrt(2.0 * target_n)
            a_scaled = a_scaled * (S_target / a_scaled.sum())

            # allocate a slice of the remaining budget (but never longer than 180 s)
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.45, 180.0)

            # run Adam with a random choice of hyper‑parameters
            lr0 = rng.choice([0.42, 0.48, 0.55])
            tau_decay = rng.choice([0.9993, 0.9995, 0.9997])
            refined, peak = _adam_optimize(
                a_scaled,
                time.time() + slice_budget,
                rng,
                lr0=lr0,
                tau_decay=tau_decay,
                lr_decay=0.99985,
                jitter_every=rng.choice([300, 500, 800])
            )

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved={best_score:.6f}")

            cur_seq = refined

        if time.time() >= end_time - 120.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent (if time permits)
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:
        remaining = end_time - time.time()
        sub_deadline = time.time() + remaining * 0.6
        sg_seq, sg_val = _subgradient_optimize(best_seq, sub_deadline, rng)
        if sg_val < best_score:
            best_score = sg_val
            best_seq = sg_seq.copy()
            print(f"[subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap simulated‑annealing fine‑tuning (if any time left)
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        remaining = end_time - time.time()
        final_seq, final_val = _anneal(best_seq, time.time() + remaining, rng)
        if final_val < best_score:
            best_score = final_val
            best_seq = final_seq.copy()
            print(f"[anneal] final improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal={best_score:.6f} len={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
