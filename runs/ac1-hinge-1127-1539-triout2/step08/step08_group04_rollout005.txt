"""
Propose a non‑negative step sequence with a low benchmark score.

Algorithm
---------
1. Choose problem size n based on the time budget.
2. Warm‑start:
   * If a previous sequence exists, resize it.
   * Otherwise use a power‑law shape.
   * Additionally run a cheap 2‑parameter grid search on a reduced
     problem (size ≈ n/4) and interpolate the best shape.
3. Sub‑gradient descent on the true L∞‑objective.
   * At each iteration compute the convolution, locate the index k*
     attaining the maximum, and form the exact sub‑gradient
     g_j = 2·a_{k*−j} (zero outside the overlapping region).
   * Perform a mirrored (exponential) update a ← a·exp(−η·g) and
     renormalise onto the ℓ¹‑simplex Σa = √(2n).
   * Accept any improvement; otherwise inject a small random jitter.
4. Optional soft‑max (entropy‑prox) descent for additional polishing
   (the same routine used in the reference code).
5. Deterministic mass‑transfer refinement:
   * Repeatedly move a modest fraction of mass from indices that most
     contribute to the current maximal convolution entry to indices
     with the smallest contribution, accepting only moves that improve
     the score.
6. Truncate trailing zeros, store the result for warm‑starts and return
   it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Parametric warm‑start on a reduced problem
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Search a simple two‑parameter family
        a_i = (i+1)^(-p) * exp(-λ·i / small_n)
    on a reduced problem.  Return the best sequence found.
    """
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (used in optional refinement)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with exponential updates using a soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Descent using the exact sub‑gradient of max(a*a) (the L∞ term).
    Updates are performed with an entropic mirror step and projection onto
    the ℓ¹‑simplex.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    eta = 0.5          # initial step size for the mirror update
    decay = 0.995      # geometric decay of eta
    it = 0

    while time.time() < deadline:
        # full convolution and the index of the current maximal entry
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # assemble the sub‑gradient g_j = 2 * a_{k‑j} on the overlap interval
        n = a.size
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        g = np.zeros_like(a)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]

        # try a few decreasing step sizes and keep the first improvement
        improved = False
        for factor in (1.0, 0.5, 0.25):
            step = eta * factor
            a_new = a * np.exp(-step * g)
            a_new = a_new * (target_sum / a_new.sum())
            val = _score(a_new)
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
                a = a_new
                improved = True
                print(f"[{time.time():.2f}] Subgrad it={it} step={step:.4f} score={best_val:.6f}")
                break

        if not improved:
            # noisy escape move
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        eta *= decay
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Move small fractions of mass from highly contributing indices
    (for the current maximal convolution entry) to low‑contributing
    indices, accepting only improvements.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # overlapping region for the maximal entry
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution weight = a[i] * a[k-i]
        contrib = a[idx] * a[k - idx]

        if contrib.sum() == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=contrib / contrib.sum())

        # build a full‑length weight vector (zero outside overlap)
        full_weight = np.zeros(n, dtype=np.float64)
        full_weight[idx] = contrib

        # receiver candidates: indices with low contribution (bottom 20%)
        low_thr = np.percentile(full_weight, 20)
        low_idxs = np.where(full_weight <= low_thr)[0]
        if low_idxs.size:
            receiver = rng.choice(low_idxs)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # transfer a modest fraction of the donor's mass
        max_frac = 0.03
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        # renormalise (positivity already guaranteed)
        a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")

    return best_a, best_val

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # problem size according to the available budget
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial sequence (fallback / warm‑start)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(20.0,
                                       (deadline - time.time()) * 0.10)
    best_small_seq, _ = _grid_param_search(small_n, small_target,
                                            param_deadline)
    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient descent on the true max‑convolution term
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.45
        a_sub, val_sub = _subgradient_descent(init, target_sum,
                                               sub_deadline)
        best_seq = a_sub
        best_score = val_sub
        print(f"[{time.time():.2f}] Sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional soft‑max (entropy) refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        soft_deadline = time.time() + (deadline - time.time()) * 0.30
        a_soft, val_soft = _exponential_descent(best_seq,
                                                target_sum,
                                                soft_deadline)
        if val_soft < best_score:
            best_seq = a_soft
            best_score = val_soft
        print(f"[{time.time():.2f}] Soft‑max refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – deterministic mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, val_final = _mass_transfer_refine(best_seq,
                                                   target_sum,
                                                   deadline)
        if val_final < best_score:
            best_seq = a_final
            best_score = val_final
        print(f"[{time.time():.2f}] Final mass‑transfer score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
