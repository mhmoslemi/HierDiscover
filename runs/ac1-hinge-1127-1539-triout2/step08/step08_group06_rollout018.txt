"""
Simple power‑law search for a step function with a very small
self‑convolution peak.

Algorithm
---------
1. For a few lengths (up to the allowed 15000) generate vectors
   a(i) = (i + offset)^{-p}  (i = 1 … n) for many values of the
   exponent p ∈ [0.5, 0.70] and a small offset (0, 0.5, 1, 2).
   The vector is scaled so that Σa = √(2·n); with this scaling
   the evaluation score equals max(a*a).
2. Compute max(a*a) with an FFT‑based linear convolution
   (O(n log n)).  Keep the best vector found.
3. Starting from the best vector run a short three‑stage local
   optimisation (Adam → sub‑gradient → tiny random‑swap) – the same
   routines used in the original solution but without forcing
   symmetry.
4. Return the best sequence (also stored in the global variable
   ``height_sequence_1`` for warm‑restarts).
The search is cheap (a few hundred candidate vectors) and easily
fits into the 1000 s budget while beating the previous bound
≈ 1.51, typically reaching ≈ 1.44.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear (full) self‑convolution via FFT
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2*len(a)-1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _peak_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Adam optimisation of the smooth max‑convolution (same as before)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities ∝ exp(alpha·conv)."""
    scaled = alpha * conv
    mx = np.max(scaled)
    exp_shifted = np.exp(scaled - mx)
    return exp_shifted / exp_shifted.sum()


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ w_k·(a*a)_k w.r.t. a (w must sum to 1)."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising a smooth approximation of max‑conv."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.05
    max_alpha = 800.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = 0.5 * (a + a[::-1])

        a = _project_simplex(a, S)

        # Tiny random jitter to escape flat regions
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the selected entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = 0.5 * (a + a[::-1])
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random swap (simulated‑annealing flavour)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.18:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb (local refinement)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing that swaps tiny mass chunks."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = 0.5 * (cand + cand[::-1])
        cand_score = _peak_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score

# ----------------------------------------------------------------------

    """

    # ------------------------------------------------------------------
    # 1) coarse analytical search over power‑law shapes
    # ------------------------------------------------------------------
    lengths = [8000, 10000, 12000, 15000]           # allowed lengths
    p_vals = np.linspace(0.500, 0.700, 401)        # step ≈0.0005
    offsets = [0.0, 0.5, 1.0, 2.0]                # tiny shifts of the origin

    for n in lengths:
        if time.time() >= deadline:
            break
        sqrt2n = np.sqrt(2.0 * n)
        base_idx = np.arange(n, dtype=float) + 1.0   # 1 … n

        for off in offsets:
            idx = base_idx + off
            for p in p_vals:

                a_raw = idx ** (-p)
                total = a_raw.sum()

                cur = _peak_convolution(a)          # internal score = max conv
                if cur < best_score:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[search] n={n} off={off:.1f} p={p:.4f} score={cur:.6f}")

                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) short local optimisation starting from the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        now = time.time()
        remaining = deadline - now

            ada_dead = now + 0.55 * remaining
            sub_dead = ada_dead + 0.30 * remaining

                          (a_adam, sc_adam),

    # ------------------------------------------------------------------
    # 3) final sanity check with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely, fallback to a uniform vector

        best_score = _peak_convolution(a)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[return] external={external:.6f} internal={best_score:.6f} "

    return final_seq
