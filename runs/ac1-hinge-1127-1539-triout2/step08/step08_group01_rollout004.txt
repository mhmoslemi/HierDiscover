"""
Propose a candidate step‑function minimising ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – use the provided ``height_sequence_1`` if it exists,
   otherwise draw a random non‑negative vector on the ℓ¹‑simplex
   Σa = √(2 n) (default n = 2000).

2. **Linear‑programming boost** – for the current sequence `a`
   solve a linear program

        maximise   Σ g_j
        subject to  (a * g)_k ≤ M   for all k,
                    g_j ≥ 0,

   where `M = max(a*a)` is the current worst convolution value.
   The solution `g` is scaled to the required ℓ¹‑norm and mixed with `a`
   (convex combinations with several mixing coefficients).  Any improvement
   is kept.

3. **Local refinement** – a short projected Adam optimisation on a
   smooth approximation of the max‑convolution is run, followed by a
   biased hill‑climb that updates the full self‑convolution in O(n) per
   move.

4. **Iterate** – steps 2‑3 are repeated while time is available (a few
   iterations are enough for a noticeable drop).  The best sequence found
   is retained.

5. **Final polish** – a last Adam pass and hill‑climb are performed,
   the result is trimmed of trailing near‑zeros and returned.

All helpers are top‑level; the routine respects the 1000 s wall‑clock
budget and prints progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation on the smooth‑max objective.
    The learning‑rate is scaled to the simplex radius.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # step‑size base (empirically tuned)
    base_lr = 0.25 * target_sum

    # smoothed‑max parameters
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the soft‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (biased version)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The destination is biased towards indices that contribute little to the
    current worst convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9997                    # slower cooling for finer search

    while time.time() < deadline:
        # pick one of the currently worst convolution positions
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: bias towards small counterpart a[M-j]
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Starting point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a = np.asarray(globals()["height_sequence_1"], dtype=float)
            n = a.size
            target_sum = math.sqrt(2.0 * n)
            a = _project_to_simplex(a, target_sum)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n})")
        except Exception as e:
            print(f"Seeding error: {e}")
            n = 2000
            a = _random_initial_sequence(n, rng)
    else:
        n = 2000
        a = _random_initial_sequence(n, rng)

    target_sum = math.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_seq = a.copy()
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.1f}] Initial max = {best_score:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # Main optimisation loop (LP + local search)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0 and iteration < 20:
        iteration += 1
        n = a.size
        target_sum = math.sqrt(2.0 * n)

        # current worst convolution value
        M = _max_convolution_fft(a)

        # ----- Linear programme -------------------------------------------------
        rows = 2 * n - 1
        # Build the (dense) convolution matrix A where column i is a[i] on rows i..i+n-1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = a[i]

        c = -np.ones(n)                     # maximise Σ g_j  → minimise -Σ g_j
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as exc:
            print(f"[{time.time():.1f}] linprog exception: {exc}")
            break

        if not res.success:
            print(f"[{time.time():.1f}] linprog failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            print(f"[{time.time():.1f}] LP returned (near) zero solution")
            break

        # Scale g to the required ℓ¹‑norm
        g = g * (target_sum / sum_g)

        # ----- Try several convex combinations ----------------------------------
        improved = False
        for t in (0.2, 0.4, 0.6, 0.8):
            cand = (1.0 - t) * a + t * g
            cand = _project_to_simplex(cand, target_sum)
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_score - 1e-7:
                best_seq = cand.copy()
                best_score = cand_score
                a = cand.copy()
                improved = True
                print(f"[{time.time():.1f}] LP‑mix (t={t:.2f}) new best {best_score:.6f}")
                break

        if not improved:
            # No improvement from LP mixing – we stop the outer loop
            print(f"[{time.time():.1f}] No LP‑mix improvement in iteration {iteration}")
            break

        # ----- Short Adam refinement -------------------------------------------
        if time.time() < deadline - 5.0:
            gd_deadline = min(deadline - 1.0, time.time() + 2.0)
            a = _adam_gradient_descent(a, target_sum, gd_deadline)
            cand_score = _max_convolution_fft(a)
            if cand_score < best_score - 1e-7:
                best_seq = a.copy()
                best_score = cand_score
                print(f"[{time.time():.1f}] Post‑ADAM improvement {best_score:.6f}")

        # ----- Biased hill climbing ---------------------------------------------
        if time.time() < deadline - 5.0:
            b = _full_convolution(a)
            hc_deadline = min(deadline - 1.0, time.time() + 2.0)
            a_hc, b_hc = _hill_climbing_improved(a, b, hc_deadline)
            cand_score = b_hc.max()
            if cand_score < best_score - 1e-7:
                best_seq = a_hc.copy()
                best_score = cand_score
                a = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 1.0
        best_seq = _adam_gradient_descent(best_seq, target, final_deadline)
        # one last quick hill‑climb
        b_final = _full_convolution(best_seq)
        if time.time() < deadline - 0.5:
            best_seq, _ = _hill_climbing_improved(best_seq, b_final,
                                                  deadline - 0.5)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Final polishing score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
