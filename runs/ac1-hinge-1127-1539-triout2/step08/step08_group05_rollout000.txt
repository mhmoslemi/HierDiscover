"""
Propose a step‑function sequence with a minimal self‑convolution peak.

Algorithm
---------
1.  Initialise a collection of candidate shapes from a few simple analytic
    families (exponential, Gaussian, Beta‑like, power‑law and uniform).
    For each length in a predefined set we sample a handful of random
    parameters, normalise the vector to Σa = √(2·n) and keep the
    best (smallest convolution maximum) found so far.
2.  Refine the best candidate with a projected Adam optimiser that
    minimises a soft‑max approximation of the convolution peak.
    Symmetry is enforced at every iteration and the vector is projected
    onto the scaled simplex {a ≥ 0, Σa = √(2·n)}.
3.  Optionally increase the length by up‑sampling (≈30 % larger, capped at 30 000)
    and run a short Adam refinement.
4.  A final, fast simulated‑annealing hill‑climb fine‑tunes the solution.
5.  The resulting sequence is stored globally as ``height_sequence_1`` and
    returned as a plain list.  The external score is printed for reference.

The code respects the 1000 s budget, performs all heavy work using
FFT‑based convolutions, and never writes to the file system.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    mask = rho_arr > 0
    if not np.any(mask):
        theta = 0.0
    else:
        rho = np.max(np.where(mask)[0])
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return np.max(_full_convolution(a))


# ----------------------------------------------------------------------
# Analytic families (parameterised by a random generator)
# ----------------------------------------------------------------------
def _make_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    # a_i = exp(-scale * |i - centre| / n)
    centre = (n - 1) / 2.0
    scale = rng.uniform(0.05, 5.0)          # relative scale
    d = np.abs(np.arange(n) - centre)
    a = np.exp(-scale * d / n)
    return a


def _make_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = (n - 1) / 2.0
    sigma_rel = rng.uniform(0.05, 0.5)      # relative sigma
    sigma = sigma_rel * n
    d = np.abs(np.arange(n) - centre)
    a = np.exp(-0.5 * (d / sigma) ** 2)
    return a


def _make_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    # a_i = x^p1 * (1-x)^p2  with x = i/(n-1)
    p1 = rng.uniform(0.1, 5.0)
    p2 = rng.uniform(0.1, 5.0)
    x = np.linspace(0.0, 1.0, n)
    a = (x ** p1) * ((1.0 - x) ** p2)
    return a


def _make_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    # a_i = (|i - centre| + 1)^(-p)
    centre = (n - 1) / 2.0
    p = rng.uniform(0.2, 3.0)
    d = np.abs(np.arange(n) - centre) + 1.0
    a = d ** (-p)
    return a


def _make_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.ones(n, dtype=float)


_FAMILY_FUNCS = {
    'exp': _make_exponential,
    'gauss': _make_gaussian,
    'beta': _make_beta,
    'pow': _make_powerlaw,
    'uniform': _make_uniform,
}


# ----------------------------------------------------------------------
# Optimiser: Projected Adam on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (log‑sum‑exp) w.r.t. a.
    w must be a probability vector of length 2*n‑1.
    """
    n = a.shape[0]
    # reverse weights because convolution gradient uses flipped kernel
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # indices that combine to a*a (see original code)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 1500.0) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed convolution peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.04

    best_a = a.copy()
    best_score = _max_conv(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # symmetry & simplex projection
        a = _symmetrize(a)
        a = _project_simplex(a, S)

        # occasional jitter to escape plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.08 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998  # very slow decay

    return best_a, best_score


def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing hill‑climb on the true max."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _max_conv(a)
    cur_score = best_score

    start = time.time()
    T0 = max(0.1 * cur_score, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = 0.05 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _symmetrize(cand)          # keep symmetry
        cand_score = _max_conv(cand)
        diff = cand_score - cur_score

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Load previous solution if it exists (may give a good starting point)
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – systematic sampling from analytic families
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000, 10000, 12000, 16000, 20000, 25000, 30000]
    families = list(_FAMILY_FUNCS.keys())
    # how many random draws per (length, family) – adapt to remaining time
    samples_per = 25

    for n in lengths:
        S = np.sqrt(2.0 * n)
        for fam in families:
            gen_fun = _FAMILY_FUNCS[fam]
            for _ in range(samples_per):
                if time.time() >= deadline:
                    break
                attempts += 1
                a = gen_fun(n, rng)
                # enforce symmetry (most families already symmetric, but cheap)
                a = _symmetrize(a)
                # clip tiny negatives that may appear due to numeric noise
                a = np.maximum(a, 0.0)
                # normalise to the prescribed sum
                s = a.sum()
                if s == 0.0:
                    a = np.full(n, 1.0)
                    s = a.sum()
                a = a / s * S
                cur = _max_conv(a)
                if cur < best_score:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[family] fam={fam} n={n} attempt={attempts} score={cur:.6f}")

    # ------------------------------------------------------------------
    # If we have a previous global solution, also test it (maybe short)
    # ------------------------------------------------------------------
    if base_seq is not None:
        cur = _max_conv(base_seq)
        if cur < best_score:
            best_score = cur
            best_seq = base_seq.copy()
            print(f"[base] imported previous solution with score={cur:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: a simple exponential at n=8000
        n = 8000
        S = np.sqrt(2.0 * n)
        a = _make_exponential(n, rng)
        a = _symmetrize(a)
        a = a / a.sum() * S
        best_seq = a
        best_score = _max_conv(a)

    now = time.time()
    if now < deadline:
        t_adam = now + 0.55 * (deadline - now)
        a_adam, sc_adam = _optimize_adam(best_seq, t_adam, rng)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(f"[adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_anneal, sc_anneal = _anneal_hillclimb(best_seq, deadline, rng)
        if sc_anneal < best_score:
            best_score = sc_anneal
            best_seq = a_anneal.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Optional up‑sampling & short Adam refinement
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 8.0:   # keep a few seconds for final printing
        cur_n = best_seq.shape[0]
        new_n = min(30000, int(cur_n * 1.30))
        # linear interpolation to new length
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, new_n)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = _symmetrize(a_big)
        S_big = np.sqrt(2.0 * new_n)
        a_big = a_big / a_big.sum() * S_big

        t_refine = now + 0.10 * (deadline - now)
        a_ref, sc_ref = _optimize_adam(a_big, t_refine, rng, max_alpha=2000.0)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[upsample‑adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation and output
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
