def lp_improve(f, M):
    n = len(f)
    # Build sparse convolution matrix
    rows = []
    cols = []
    data = []
    # For each i (0..n-1), for each k from i to i+n-1:
    for i in range(n):
        # row indices = i to i+n-1
        row_start = i
        row_end = i + n  # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    if not res.success:
        return None
    g = res.x
    return g

def build_conv_matrix(f):
    n = len(f)
    row_idx = np.empty(n * n, dtype=np.int32)
    col_idx = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    pos = 0
    for i in range(n):
        rows = np.arange(i, i + n, dtype=np.int32)
        cols = np.full(n, i, dtype=np.int32)
        row_idx[pos:pos+n] = rows
        col_idx[pos:pos+n] = cols
        data[pos:pos+n] = f[i]
        pos += n
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    return A

"""
Hybrid adversarial step‑function optimizer.

The algorithm proceeds in five stages:

1. **Broad random search** – many short trials drawn from diverse
   parametric families.  The best candidate (lowest peak of the
   self‑convolution) is kept.
2. **Geometric scaling & Adam refinement** – the current best sequence
   is interpolated to several larger lengths and each copy is polished
   with a soft‑max‑based Adam optimiser that directly minimises the
   peak of the convolution.
3. **Linear‑programming boosting** – for a moderate length (≤2000) we
   solve a small LP that maximises the ℓ₁‑norm of a sequence `g` under
   the constraint `f*g ≤ max(f*f)`.  Mixing the LP solution back into
   `f` provably reduces the peak.
4. **Simulated‑annealing hill‑climb** – a cheap stochastic search that
   moves small amounts of mass between entries and accepts occasional
   uphill moves while the temperature slowly cools.
5. **Final sanity check** – the resulting sequence is stored in the
   global ``height_sequence_1`` and returned as a plain list.

All steps respect the overall 1000 s budget; the code leaves a small
margin so the harness does not terminate the process prematurely.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient for Adam optimiser
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2*(mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ→0).
    Returns the best sequence found and the corresponding peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                           # aggressive start

    # temperature schedule (start a bit larger than before)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)            # start with mild smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient ----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (slow decay) ----
        tau *= 0.9998                # keep a little smoothing longer
        lr *= 0.9999                 # very gentle decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    raw = rng.random(n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    “U‑shape’’ – large values at the edges, strongly suppressed in the centre.
    a_i ∝ ((i+1)*(n-i))^{-p} with optional uniform mixing.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑shaped distribution (flexible edge vs centre mass)."""
    a = rng.uniform(0.3, 5.0)
    b = rng.uniform(0.3, 5.0)
    x = np.linspace(0.0, 1.0, n)
    raw = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    # avoid all‑zero vectors
    if raw.max() == 0:
        raw = np.ones(n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # pick a family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'beta'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'beta':
        raw = _init_beta(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def _build_conv_matrix(v: np.ndarray) -> sp.csr_matrix:
    """
    Build the dense convolution matrix A such that A @ g = f * g
    (full linear convolution of f=v with g).  Shape of A is (2n‑1)×n.
    """
    n = v.shape[0]
    # total number of non‑zeros = n * n
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = v[i]
        pos += n

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_refine_small(f: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    LP‑based refinement for a sequence of length ≤ 2000.
    Returns a (possibly) improved sequence and its peak.
    """
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    # Normalise to the canonical ℓ₁‑norm
    f = np.maximum(f, 0.0)
    if f.sum() == 0.0:
        f = np.full(n, 1.0)
    f = f * (S / f.sum())

    best_f = f.copy()
    best_M = _max_convolution(best_f)

    # Upper bound on each entry (clamp used by the evaluator)
    ub = 1000.0

    # We perform a handful of LP‑updates – each reduces the peak if possible.
    while time.time() < deadline:
        # Build the convolution matrix for the current best_f
        A = _build_conv_matrix(best_f)
        b = np.full(2 * n - 1, best_M)

        # Maximise sum(g)  → minimise -sum(g)
        c = -np.ones(n)

        res = opt.linprog(c, A_ub=A, b_ub=b,
                         bounds=[(0.0, ub)] * n,
                         method='highs', options={'presolve': True})

        if not res.success:
            # LP failed (should be rare); exit the refinement loop.
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # Normalise g to the same ℓ₁‑norm S
        g_norm = g * (S / sum_g)

        improved = False
        # Try a few convex‑combination step sizes.
        for t in (0.5, 0.3, 0.15, 0.07, 0.03):
            cand = (1.0 - t) * best_f + t * g_norm
            cand = np.clip(cand, 0.0, None)   # safety
            M_cand = _max_convolution(cand)
            if M_cand < best_M - 1e-12:
                best_f = cand
                best_M = M_cand
                improved = True
                break
        if not improved:
            break   # no further improvement possible

    return best_f, best_M


def _lp_refine(seq: np.ndarray,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Wrapper that guarantees the LP step works on a moderate length.
    If the input length exceeds 2000 the sequence is down‑scaled,
    refined, and then up‑scaled again.
    """
    n = len(seq)
    MAX_N = 2000
    if n <= MAX_N:
        return _lp_refine_small(seq, deadline, rng)

    # Down‑scale, refine, then up‑scale.
    target = MAX_N
    f_small = _scale_sequence(seq, target)
    # give a proportional slice of the remaining budget
    sub_deadline = time.time() + min(30.0, max(0.0, deadline - time.time()))
    refined_small, _ = _lp_refine_small(f_small, sub_deadline, rng)

    # Upscale back to the original length and re‑normalise.
    up = _scale_sequence(refined_small, n)
    up = up * (seq.sum() / up.sum())
    # one final quick Adam polishing (optional but cheap)
    up_opt, peak = _adam_optimize(up, time.time() + 5.0, rng)
    return up_opt, peak

# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing with a linear temperature schedule.
    Returns the best sequence seen and its peak.
    """
    a = seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = _max_convolution(a)

    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_M

    # initial temperature proportional to the current peak
    T0 = max(best_M * 0.07, 1e-4)

    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / total)

        # propose a move: transfer a fraction from i → j
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        amount = a[i] * frac
        cand = a.copy()
        cand[i] -= amount
        cand[j] += amount

        M_cand = _max_convolution(cand)
        delta = M_cand - best_M
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = cand
            if M_cand < best_M - 1e-12:
                best_a = cand
                best_M = M_cand

    return best_a, best_M
