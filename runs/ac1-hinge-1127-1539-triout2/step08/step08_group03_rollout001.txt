"""
Improved adversarial step‑function optimiser.

The optimiser works in three stages:

1️⃣  **Broad random sampling** – a few thousand candidates are drawn from
    a rich collection of parametric families (exponential, power‑law,
    mixed‑exponential, sparse bumps, Dirichlet‑like).  All candidates are
    normalised to ℓ₁‑norm √(2 n) so that the evaluation reduces to the
    peak of the self‑convolution.  The best few are kept.

2️⃣  **Mirror‑descent (Adam) on the simplex** – each retained candidate
    is repeatedly refined on a schedule of increasing lengths (≈ 2 k → 5 k
    → 12 k → max_len).  The optimisation uses a log‑softmax
    parametrisation (which implicitly enforces positivity and the ℓ₁‑norm)
    together with a τ‑soft‑max approximation of the max‑convolution.
    The temperature τ is annealed exponentially; once τ becomes tiny the
    gradient of the true max is used.  Adam’s learning‑rate also decays.
    Jitter is added periodically to escape plateaus.

3️⃣  **Fine‑grained simulated‑annealing** – any remaining time is spent
    on a cheap mass‑transfer walk on the simplex.  This helps to
    polish the solution and can occasionally discover tiny improvements.

The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum value of the full self‑convolution of a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of τ‑soft‑max (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    # soft‑max over the convolution entries
    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient w.r.t. a: 2 * conv(mask * a, a_rev)
    a_rev = a[::-1]
    conv_len = mask.shape[0] + a.shape[0] - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]

    # take the central n entries (indices n‑1 … 2n‑2)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.9985,
                   jitter_every: int = 300) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution of ``a`` under the ℓ₁‑norm
    constraint.  Returns the best vector seen and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                       # required ℓ₁‑norm

    # initialise logits from a feasible a0 (log‑softmax)
    eps = 1e-12
    probs = np.maximum(a0 / S, eps)
    probs = probs / probs.sum()
    p = np.log(probs)                          # logits

    # Adam moments
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise τ from the current peak
    tau = max(_max_convolution(a0), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- gradient (τ‑soft‑max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t logits (chain rule) -----
        mu = np.dot(a, g) / S                # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                # shape (n,)

        # optional clipping for stability
        max_abs = np.max(np.abs(grad_p))
        if max_abs > 5.0 * S:
            grad_p = grad_p * (5.0 * S / max_abs)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- annealing schedules -----
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape flat regions
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.007, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones (flat)."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 4.0)               # exponent
    offset = rng.uniform(0.0, 12.0)          # horizontal shift
    mix = rng.uniform(0.0, 1.0)              # mix weight for the background
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.0005, 0.04)         # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_double_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two exponentials with independent rates."""
    idx = np.arange(n)
    lam1 = rng.uniform(0.0005, 0.02)
    lam2 = rng.uniform(0.02, 0.08)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * np.exp(-lam1 * idx) + (1.0 - mix) * np.exp(-lam2 * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random rectangular bumps on a low‑level flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.12))
    nbumps = rng.integers(1, max(2, n // 400))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 80))
        height = rng.uniform(0.7, 4.5)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ~0.25 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # 25 % chance to warm‑start from the best known construction
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    # pick a random family
    style = rng.choice([
        'dirichlet', 'gamma', 'uniform',
        'left_powerlaw', 'left_exponential',
        'double_exponential', 'sparse'
    ])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'double_exponential':
        raw = _init_double_exponential(n, rng)
    else:  # sparse bumps
        raw = _init_sparse_bumps(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (cheap)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with Metropolis acceptance."""
    n = seq.shape[0]
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.13 * cur_score  # a modest start

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # re‑normalise to keep the ℓ₁‑norm exactly S
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99993
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found (as a plain list).  The global variable
    ``height_sequence_1`` is also updated.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 3.0   # small safety margin

    # ------------------------------------------------------------------
    # Load warm start if available
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 10 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 750.0   # leave ~75 % for refinement
    top_k = 5
    top_candidates = []   # list of (peak, vector)

    attempt = 0
    while time.time() < explore_deadline:
        attempt += 1
        # pick a length – two ranges give a good spread
        if rng.random() < 0.55:
            n = int(rng.integers(1800, 4000))
        else:
            n = int(rng.integers(4000, 15000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        # maintain a small elite set
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

        if attempt % 500 == 0:
            print(f"[explore] attempt={attempt:5d} n={n:5d}  best={top_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Guard against empty exploration (should never happen)
    # ------------------------------------------------------------------
    if not top_candidates:
        n = 2500
        a0 = np.full(n, np.sqrt(2.0 * n) / n)
        top_candidates = [( _max_convolution(a0), a0.copy() )]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement
    # ------------------------------------------------------------------
    # Define a length schedule that grows geometrically up to a safe ceiling
    best_candidate_peak, best_candidate_seq = top_candidates[0]
    best_n = best_candidate_seq.shape[0]

    max_len = min(40000, best_n * 8)   # cap to keep FFTs cheap
    lengths = [best_n]
    # progressive scaling factors
    for factor in (1.4, 2.0, 3.0, 5.0):
        cand = int(best_n * factor)
        if cand > max_len:
            break
        lengths.append(cand)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    best_overall_seq = best_candidate_seq.copy()
    best_overall_peak = best_candidate_peak

    for init_idx, (init_peak, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_n = cur_seq.shape[0]

        for target_n in lengths:
            remaining = end_time - time.time()
            if remaining < 45.0:
                break

            # interpolate to the new length and renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining budget (but not too long)
            slice_budget = min(remaining * 0.45, 170.0)

            # hyper‑parameter diversity
            lr0 = rng.choice([0.46, 0.50, 0.55])
            tau_decay = rng.choice([0.9993, 0.9995, 0.9997])
            refined, peak = _adam_optimize(
                a_scaled,
                time.time() + slice_budget,
                rng,
                lr0=lr0,
                tau_decay=tau_decay,
                lr_decay=0.9987,
                jitter_every=rng.integers(200, 600)
            )

            if peak < best_overall_peak:
                best_overall_peak = peak
                best_overall_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  new best={best_overall_peak:.6f}")

            # use the refined vector as seed for the next scale
            cur_seq = refined
            cur_n = target_n

            # early exit if the global deadline is close
            if time.time() >= end_time - 30.0:
                break
        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        final_seq, final_peak = _anneal(best_overall_seq, time.time() + remaining, rng)
        if final_peak < best_overall_peak:
            best_overall_peak = final_peak
            best_overall_seq = final_seq.copy()
            print(f"[anneal] final annealed best={best_overall_peak:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check through the provided black‑box evaluator
    # ------------------------------------------------------------------
    final_list = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_list)
    print(f"[final] external score = {external_score:.6f}  "
          f"internal max = {best_overall_peak:.6f}  length = {len(final_list)}")

    # store globally for warm‑starts in subsequent runs
    globals()['height_sequence_1'] = final_list
    return final_list
