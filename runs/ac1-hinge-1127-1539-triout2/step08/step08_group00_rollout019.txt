from scipy.optimize import linprog
def _lp_step(f, M, target_sum):
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2*n-1, M, dtype=float)
    c = -np.ones(n)  # maximize sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 == 0:
        return None
    g = g0 * (target_sum / sum_g0)
    return g
```

"""
Propose a candidate sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

under non‑negative and per‑element caps.  The method consists of three
phases:

1.  **Parametric warm‑start** –  a cheap random search over pure &
    piece‑wise exponential families (the shape that is known to be close to
    optimal).  The best sequence is kept.

2.  **Exact hill‑climbing** – a simulated‑annealing style move that transfers a
    random amount of mass from a “source’’ index (chosen among those that
    contribute to the current maximal convolution entry) to a “target’’ index.
    The full convolution is updated in O(n) per move, allowing millions of
    moves within the time budget.

3.  **Linear‑programming refinement** – with the current sequence `f`,
    solve the LP

        maximise   Σ g_i
        subject to f * g ≤ M   (componentwise)
                   g_i ≥ 0

    where `M` is the current maximal convolution entry.  The feasible `g`
    is scaled to the required ℓ¹‑sum and mixed with `f` (convex combination).
    Repeating a few LP → mixing steps drives the maximal entry down
    further.

The whole routine respects the global wall‑clock deadline (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Exponential families
# ----------------------------------------------------------------------


def _exp_sequence(alpha: float, n: int) -> np.ndarray:
    """
    a_i ∝ r^i   with  r = exp(‑α / n)  (α>0).
    Returns the sequence projected onto the ℓ¹‑simplex
    Σ a_i = sqrt(2·n).
    """
    r = np.exp(-alpha / n)
    idx = np.arange(n, dtype=float)
    raw = r ** idx
    target = np.sqrt(2.0 * n)
    # geometric series sum = (1‑rⁿ)/(1‑r)
    scale = target * (1.0 - r) / (1.0 - r ** n)
    return raw * scale


def _piecewise_exp(alpha1: float, alpha2: float, split: int, n: int) -> np.ndarray:
    """
    Two‑segment exponential: first segment length `split` uses α1,
    second segment uses α2.  The whole vector is projected onto the simplex.
    """
    r1 = np.exp(-alpha1 / n)
    r2 = np.exp(-alpha2 / n)
    left = r1 ** np.arange(split, dtype=float)
    right = r2 ** np.arange(n - split, dtype=float)
    raw = np.concatenate([left, right])
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    return raw * scale


# ----------------------------------------------------------------------
# Hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a source index to a target index
    while keeping the full convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9             # initial temperature
    cooling = 0.9995                     # slower cooling for a longer run

    while time.time() < deadline:
        # index(es) attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – weighted by its contribution to b_k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias toward low‑mass positions
        if rng.random() < 0.4:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        # removing contribution from source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # adding contribution from target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correction for the cross term at i+j (appears twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # final accurate convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that (A @ g)[k] = (f * g)[k].
    Rows correspond to convolution indices k.
    """
    n = f.size
    # rows[i * n + j] = i + j
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    One linear‑programming step.
    Solve  maximise Σ g_i   subject to   f * g ≤ M , g ≥ 0.
    Scale the feasible solution to the required ℓ¹‑sum.
    Returns the scaled vector or None on failure.
    """
    n = f.size
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=float)

    # maximise Σ g_i  → minimise  -Σ g_i
    c = -np.ones(n, dtype=float)

    res = linprog(c,
                  A_ub=A,
                  b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None

    g0 = res.x
    s0 = g0.sum()
    if s0 <= 0.0:
        return None
    g = g0 * (target_sum / s0)
    return g


def _lp_refine(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeated LP → mixing steps until the wall‑clock runs out.
    Each iteration mixes the current sequence with the LP‑solution
    using a convex combination (t = 0.3 by default) and projects back
    onto the simplex.
    """
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    cur = a.copy()
    cur_max = _max_convolution_fft(cur)
    best = cur.copy()
    best_max = cur_max
    t = 0.3  # mixing weight

    while time.time() < deadline - 2.0:
        M = cur_max
        g = _lp_step(cur, M, target_sum)
        if g is None:
            break
        # convex combination and re‑project
        cur = (1.0 - t) * cur + t * g
        cur = _project_to_simplex(cur, target_sum)
        cur_max = _max_convolution_fft(cur)

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best = cur.copy()
            print(f"[{time.time():.1f}] LP refinement improvement: {best_max:.6f}")

        # if progress stalls, break
        if cur_max / best_max > 0.9999:
            break

    return best
