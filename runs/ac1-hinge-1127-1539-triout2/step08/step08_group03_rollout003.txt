def _expgrad_tune(seq, deadline, rng, eta_initial=0.01):
    a = seq.copy()
    S = a.sum()  # should be sqrt(2n)
    n = len(a)
    it = 0
    while time.time() < deadline:
        it+=1
        # compute convolution
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))
        # gradient w.r.t a: for each i that contributes to conv_k: 2*a_{k-i}
        # Compute reversed a
        rev_a = a[::-1]
        # For each i, the needed index is k_max - i; we can vectorize:
        # g[i] = 2*a[k_max - i] if within 0..n-1 else 0
        # Equivalent to aligning reversed a with offset.
        # Use shift: we can compute g = np.zeros_like(a)
        # For i range max(0,k_max-(n-1)) to min(n-1,k_max)
        lo = max(0, k_max - (n-1))
        hi = min(n-1, k_max)
        # For i in lo..hi:
        # idx = k_max - i
        # So g[i] = 2*a[idx]
        # We'll compute using slicing.
        # Prepare slice
        i_range = slice(lo, hi+1)
        idx_range = slice(k_max - lo, k_max - hi - 1, -1)  # reversed order.
        # Actually compute directly vectorized:
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[idx_range]
        # exponentiated gradient
        # adapt eta gradually
        eta = eta_initial / np.sqrt(it)
        # update multiplicatively
        a = a * np.exp(-eta * g)
        # renormalize to sum S (maybe enforce non-neg by clipping)
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a == 0:
            break
        a = a * (S / sum_a)
    # final conv peak
    final_peak = _full_convolution_fft(a).max()
    return a, final_peak

"""
Adversarial step‑function optimizer – fourth generation.

Core ideas
==========

1. **Diversified random families** – In addition to the families used
   before, we add a symmetric “V‑shape” (large values near the two
   endpoints) and a flexible beta‑distribution family.  This gives the
   optimiser many more promising seeds.

2. **Larger elite set** – The top‑k list is extended to 8 candidates,
   allowing the refinement stage to start from a broader pool.

3. **Two‑stage refinement** –  
   *Stage A* runs Adam (log‑softmax parametrisation) on a sequence of
   increasing lengths (up to ~30 000).  Hyper‑parameters are sampled
   per run.  
   *Stage B* performs a cheap **exponentiated‑gradient** routine that
   directly follows the sub‑gradient of the *hard* max‑convolution.
   This is an efficient way to push mass away from the indices that
   dominate the current peak.

4. **Dynamic time budgeting** – 55 % of the wall‑time is spent on
   random exploration, ~35 % on the Adam refinements and the remaining
   time on the final exponentiated‑gradient fine‑tuning.

The function returns the best sequence found within the 1000 s limit
and stores it in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """Minimise the exact max‑convolution under the ℓ₁‑norm constraint."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0  (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient (soft max)
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # annealing schedules
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Exponentiated‑gradient fine‑tuner (direct sub‑gradient on the hard max)
# ----------------------------------------------------------------------
def _expgrad_tune(seq: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  eta0: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simple exponentiated‑gradient loop that follows the sub‑gradient of
    the *hard* max‑convolution.  Each iteration moves mass away from the
    positions that contribute most to the current peak.
    """
    a = seq.copy()
    S = a.sum()               # should already be √(2 n)
    n = len(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        # i in [lo, hi]  →  partner index = k_max - i
        i_slice = slice(lo, hi + 1)
        partner_slice = slice(k_max - lo, k_max - hi - 1, -1)  # reversed order
        g[i_slice] = 2.0 * a[partner_slice]

        # learning‑rate schedule (decrease like 1/√t)
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)

        # renormalise to the required ℓ₁‑norm
        sum_a = a.sum()
        if sum_a == 0.0:
            break
        a = a * (S / sum_a)

    final_peak = _full_convolution_fft(a).max()
    return a, final_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)          # horizontal shift
    mix = rng.uniform(0.0, 1.0)              # how much background
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))    # low background
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_V_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric V‑shape: large values near the two endpoints, decaying towards the centre."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)               # distance to nearest endpoint
    lam = rng.uniform(0.001, 0.02)                  # decay rate
    mix = rng.uniform(0.0, 1.0)                     # background mix
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * d)
    return raw


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑distribution‑like shape on the discrete grid."""
    a = rng.uniform(0.5, 5.0)
    b = rng.uniform(0.5, 5.0)
    x = np.linspace(0.0, 1.0, n)
    # add a tiny epsilon to avoid exact zeros when a,b<1
    raw = (np.maximum(x, 1e-12) ** (a - 1)) * (np.maximum(1 - x, 1e-12) ** (b - 1))
    # mix with a uniform component for robustness
    mix = rng.uniform(0.0, 0.4)
    raw = mix * np.ones_like(raw) + (1.0 - mix) * raw
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.22 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'V_shape', 'beta'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'V_shape':
        raw = _init_V_shape(n, rng)
    else:   # beta
        raw = _init_beta(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm follows three stages:

    1. Random exploration of diverse parametric families.
    2. Multi‑scale Adam refinement (log‑softmax parametrisation).
    3. Final exponentiated‑gradient fine‑tuning that follows
       the true max‑convolution sub‑gradient.

    The best found construction is stored in ``height_sequence_1`` and
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0      # safety margin

    # --------------------------------------------------------------
    # Warm‑start from a previously found construction, if present
    # --------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # --------------------------------------------------------------
    # Phase 1 – random exploration (≈ 55 % of budget)
    # --------------------------------------------------------------
    explore_deadline = end_time - 0.35 * budget_s   # reserve ~35 % for later stages
    top_k = 8
    top_candidates = []          # (score, seq)

    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # Length selection – a mixture of medium and larger supports
        if rng.random() < 0.55:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # keep elite list
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # Fallback in case exploration failed (should never happen)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # --------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement
    # --------------------------------------------------------------
    # Length schedule – gradually grow up to a safe ceiling
    max_len = min(30000, best_n * 6)
    lengths = [best_n]
    for factor in (1.4, 2.0, 3.0):
        cand = int(best_n * factor)
        if cand < max_len:
            lengths.append(cand)
    lengths.append(int(max_len))
    lengths = sorted(set(lengths))

    # iterate over elite candidates (most promising first)
    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_n = len(cur_seq)

        for target_n in lengths:
            remaining = end_time - time.time()
            if remaining < 30.0:
                break

            # interpolate to target length and normalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining budget (but never longer than 150 s)
            slice_budget = min(remaining * 0.45, 150.0)

            # hyper‑parameters sampled per run
            lr0 = rng.choice([0.38, 0.44, 0.50, 0.55])
            tau_decay = rng.choice([0.9992, 0.9994, 0.9996])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.99985,
                                           jitter_every=400)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
