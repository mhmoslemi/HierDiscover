def evaluate_sequence(sequence):
    # security checks: ensure list, not empty, each x is int/float, not bool, no NaN or Inf.
    # cast to list of floats
    # protect non-negative: [max(0, x) ...]
    # clamp each element to max 1000.0
    n = len(sequence)

    max_b = max(b_seq)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

"""
Improved step‑function optimizer.

The search proceeds in several stages:
  1. Wide random sampling of diverse analytic families (including new
     “U‑shaped’’ families that concentrate mass near the ends).
  2. Projected Adam optimisation on a smooth log‑sum‑exp approximation
     of the convolution peak; symmetry is enforced throughout.
  3. Sub‑gradient descent on the true maximum convolution entry.
  4. Simulated‑annealing hill‑climb with pairwise mass swaps.
  5. Length‑up‑sampling (≈30 % larger) followed by a short Adam polish.

All iterates are projected onto the simplex
   {a_i ≥ 0, Σ a_i = √(2 n)}.
The best sequence found is stored in the global name `height_sequence_1`
and returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the scaled probability simplex (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families for the initial sampling stage (including new U‑shaped ones)
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a diverse set of families."""
    fam = rng.integers(0, 13)
    if fam == 0 and base_seq is not None:
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                 # symmetric power‑law (i·(n‑i))^p
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                # beta‑like skewed
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                 # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)                 # monotone decreasing power‑law
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n        # exponential decay
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n           # logistic‑like S‑curve
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    elif fam == 9:
        # U‑shaped inverse power law (mass near the ends)
        p = rng.uniform(0.5, 3.0)
        eps = 1e-3
        idx = np.arange(n) + 0.5
        a = ( (idx) * (n + 0.5 - idx) + eps ) ** (-p)
    elif fam == 10:
        # Piecewise constant: high at the edges, low in the middle
        edge_frac = rng.uniform(0.05, 0.2)
        high = rng.uniform(0.5, 2.0)
        low = rng.uniform(0.1, 0.5)
        cut = max(1, int(edge_frac * n))
        a = np.full(n, low)
        a[:cut] = high
        a[-cut:] = high
    elif fam == 11:
        # Mixed beta + inverse‑power shape
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0.001, 0.999, n)
        a = (x ** p1) * ((1 - x) ** p2) + \
            0.5 * ((x * (1 - x) + 1e-4) ** (-rng.uniform(0.5, 2.0)))
    elif fam == 12:
        # Random convex combination of two simple families
        c = rng.random()
        a1 = rng.random(n) ** rng.uniform(0.5, 2.0)
        a2 = np.exp(-rng.uniform(0.001, 0.02) * np.arange(n) / n)
        a = c * a1 + (1 - c) * a2
    else:
        a = rng.random(n)

    # Ensure non‑negativity
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 800.0,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """Projected Adam minimising a smooth approximation of the convolution peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature schedule
    alpha = 0.1
    alpha_mul = 1.04
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it + 1.0)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass exchange
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # slight early‑stop check
        if it % 5000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """Simulated‑annealing hill‑climb with pairwise mass swaps."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    cur_max = best_score
    start = time.time()
    T0 = max(0.1 * cur_max, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.05 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load previous construction if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random families (≈12 % of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.12 * budget_s
    while time.time() < phase1_end:
        attempts += 1
        if base_seq is not None and rng.random() < 0.45:
            scale = rng.uniform(0.9, 1.1)
            n = max(2000, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(3000, 25001))

        a = _sample_family(rng, n, base_seq)
        S = np.sqrt(2.0 * n)
        a = a / (a.sum() + 1e-16) * S
        a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] att={attempts} n={n} score={cur_score:.6f}")

        # early exit if we have already beaten the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback in the unlikely case Phase 1 found nothing
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_seq = a
        best_score = np.max(_full_convolution(a))

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement (≈60 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    t_adam_end = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam(best_seq, t_adam_end, rng,
                                    max_alpha=900.0, enforce_sym=True)
    if sc_adam < best_score:
        best_score = sc_adam
        best_seq = a_adam.copy()
        print(f"[adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent (≈20 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        t_sub_end = now + 0.20 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_seq, t_sub_end, rng,
                                          enforce_sym=True)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing hill‑climb (≈5 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_ann, sc_ann = _anneal_hillclimb(best_seq, deadline, rng,
                                          enforce_sym=True)
        if sc_ann < best_score:
            best_score = sc_ann
            best_seq = a_ann.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Upsample + short Adam polish (≈10 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 4.0:
        cur_n = best_seq.shape[0]
        new_n = min(30000, int(cur_n * 1.35))
        a_big = _upsample_seq(best_seq, new_n)
        S_big = np.sqrt(2.0 * new_n)
        a_big = a_big / (a_big.sum() + 1e-16) * S_big
        a_big = _symmetrize(a_big)

        t_ref_end = now + 0.10 * (deadline - now)
        a_ref, sc_ref = _optimize_adam(a_big, t_ref_end, rng,
                                       max_alpha=1100.0, enforce_sym=True)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[upsample‑adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
```

This implementation expands the family pool, enforces symmetry throughout the optimisation, allocates more time to the powerful Adam stage, adds a modest up‑sampling‑and‑polish step, and keeps track of the best construction found before the time budget expires.
