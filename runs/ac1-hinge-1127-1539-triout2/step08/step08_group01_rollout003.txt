def _solve_lp_max_sum(f, M, target_sum):
    n = f.shape[0]
    # build A_ub
    m = 2 * n - 1
    A_ub = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # j indices inclusive
        js = np.arange(j_start, j_end + 1)
        i = k - js
        A_ub[k, j_start:j_end + 1] = f[i]
    b_ub = np.full(m, M, dtype=np.float64)
    # bounds for g: [0, 1000]
    bounds = [(0, 1000.0) for _ in range(n)]
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        # If linprog not available or fails, return None
        return None
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    factor = target_sum / s
    return g * factor

"""
Propose a non‑negative step‑function (list of heights) that minimises
evaluate_sequence.  The algorithm combines three complementary ideas:

1.  Projected Adam optimisation on a smooth (log‑sum‑exp) surrogate of the
    max‑convolution.  Several restarts are performed for robustness.

2.  A linear‑programming refinement (only for moderate lengths, n ≤ 2000):
    given the current shape f and its worst convolution value M,
    maximise Σ g_j subject to f * g ≤ M and g ≥ 0.  After scaling g to the
    required ℓ¹‑norm, f is mixed with g.  This step provably never increases
    the objective and often yields a noticeable drop.

3.  A fast O(n) biased hill‑climbing that moves mass from indices that
    contribute heavily to the current worst convolution entry towards
    indices that alleviate it.

A length schedule (2000 → 5000 → 20000 → 50000) is used; at each stage the
best candidate from the previous stage is up‑sampled, optimised, and
refined.  The whole search respects a 1000 s wall‑clock budget and prints
intermediate scores.  The final sequence is trimmed of trailing near‑zero
entries before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy linear programming import
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation on the smooth‑max objective.
    The learning‑rate is scaled to the simplex radius.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # step‑size base (empirically tuned)
    base_lr = 0.25 * target_sum

    # smoothed‑max parameters
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the soft‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The destination is biased towards indices that contribute little to the
    current worst convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9997                    # slower cooling for finer search

    while time.time() < deadline:
        # pick one of the currently worst convolution positions
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: bias towards small counterpart a[M-j]
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Linear‑programming refinement (only for moderate n)
# ----------------------------------------------------------------------


def _solve_lp_max_sum(f: np.ndarray, M: float, target_sum: float):
    """
    Solve
        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k,
                    g_j ≥ 0,
                    g_j ≤ 1000   (caps to avoid unboundedness).
    Returns a scaled version of the optimal g (scaled to have ℓ¹‑norm
    equal to target_sum).  If the LP fails, ``None`` is returned.
    """
    if linprog is None:
        return None
    n = f.shape[0]
    m = 2 * n - 1

    # Build the matrix A_ub where A_ub[k, j] = f_{k-j} if index valid
    A_ub = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1, dtype=int)
        is_ = k - js
        A_ub[k, j_start:j_end + 1] = f[is_]

    b_ub = np.full(m, M, dtype=np.float64)

    # Bounds for each variable: 0 ≤ g_j ≤ 1000 (the same clipping used in evaluate_sequence)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # Objective: maximize Σ g_j  →  minimize -Σ g_j
    c = -np.ones(n, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:  # pragma: no cover
        return None

    if not res.success:
        return None

    g = res.x
    total = g.sum()
    if total <= 0.0:
        return None
    factor = target_sum / total
    return g * factor


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            prev = _project_to_simplex(prev, math.sqrt(2.0 * n0))
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – allocate most effort to larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 20000, 50000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Allocate time slices for this dimension
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 30.0:
            break  # not enough time for a useful pass
        gd_budget = remaining * 0.35
        lp_budget = remaining * 0.10 if (linprog is not None and n <= 2000) else 0.0
        hc_budget = remaining * 0.35
        extra_budget = remaining - gd_budget - lp_budget - hc_budget

        # --------------------------------------------------------------
        # Phase A – Projected Adam gradient descent (multiple restarts)
        # --------------------------------------------------------------
        n_restarts = 6
        per_restart = gd_budget / max(1, n_restarts)
        local_best_a = a_init.copy()
        local_best_score = _max_convolution_fft(local_best_a)

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.1, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_gradient_descent(start_a, target_sum, this_deadline)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] GD improvement for n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – Linear‑programming refinement (only for small n)
        # --------------------------------------------------------------
        if lp_budget > 5.0 and linprog is not None and n <= 2000:
            lp_deadline = min(deadline - 0.1, time.time() + lp_budget)
            f = best_seq.copy()
            while time.time() < lp_deadline:
                M = _max_convolution_fft(f)
                g_scaled = _solve_lp_max_sum(f, M, target_sum)
                if g_scaled is None:
                    break
                # Mix the old and new shapes (the mix factor can be tuned)
                t_mix = 0.5
                f_new = (1.0 - t_mix) * f + t_mix * g_scaled
                f_new = _project_to_simplex(f_new, target_sum)
                new_score = _max_convolution_fft(f_new)

                if new_score < best_score - 1e-12:
                    best_seq = f_new.copy()
                    best_score = new_score
                    f = f_new
                    print(f"[{time.time():.1f}] LP improvement for n={n} score={best_score:.6f}")
                else:
                    # No improvement – stop the LP refinement for this length
                    break

        # --------------------------------------------------------------
        # Phase C – Biased hill‑climbing
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            hc_deadline = min(deadline - 0.1, time.time() + hc_budget)
            a_cur = best_seq.copy()
            b_cur = _full_convolution(a_cur)
            a_hc, b_hc = _hill_climbing_improved(a_cur, b_cur, hc_deadline)
            score_hc = b_hc.max()
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Optional extra gradient descent after hill‑climb
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0 and extra_budget > 5.0:
            extra_deadline = deadline - 1.0
            a_refined = _adam_gradient_descent(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution_fft(a_refined)
            if score_ref < best_score - 1e-12:
                best_seq = a_refined.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC GD improvement for n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the best sequence (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # Long Adam run
        final_deadline = deadline - 2.0
        best_seq = _adam_gradient_descent(best_seq, target, final_deadline)

        # Short hill‑climb
        b_final = _full_convolution(best_seq)
        best_seq, b_final = _hill_climbing_improved(best_seq, b_final, deadline - 1.0)

        final_score = b_final.max()
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
