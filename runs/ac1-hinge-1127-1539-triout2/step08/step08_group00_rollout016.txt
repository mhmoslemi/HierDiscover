"""
A fast convex‐optimization based search for a step‑function that minimises
``evaluate_sequence``.  The objective is the maximum entry of the
self‑convolution (the evaluation is scale‑invariant, so we keep the
ℓ¹‑norm fixed to √(2 n) and stay on the simplex).

The algorithm consists of three stages:

1. **Parametric warm‑start** – use a simple exponential family and
   up‑sample an existing solution (if any) to the current length.
2. **Exponentiated‑gradient descent on the exact max‑convolution**
   (projected onto the simplex).  At each iteration we compute the
   sub‑gradient of the (non‑smooth) maximum via the locations where the
   current convolution attains its maximum, then update
   ``a_i ← a_i·exp(‑η·g_i)`` and renormalise.
3. **Exact hill‑climbing** (simulated annealing) as a polishing step.

All work is bounded by the global wall‑clock deadline (default 1000 s)
and the best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a⋆a (full, length 2·n‑1) via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (FFT, O(n log n))."""
    return float(_full_convolution(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns (g, M) where M = max(conv) and g is a sub‑gradient of M
    with respect to a (the sum of the gradients of all maximal entries).
    """
    n = a.shape[0]
    M = conv.max()
    # tolerance to collect all "almost maximal" entries
    tol = max(M * 1e-12, 1e-15)
    idx_max = np.where(conv >= M - tol)[0]

    grad = np.zeros_like(a)
    for k in idx_max:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        j = np.arange(lo, hi + 1, dtype=int)
        # contribution 2*a_{k-j}
        grad[j] += 2.0 * a[k - j]
    return grad, M


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _exp_gradient_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient on the non‑smooth objective
    max(a⋆a).  The iterate stays in the positive orthant and is
    re‑normalised to the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    n = a.size
    # learning‑rate schedule (multiplicative)
    eta = 0.8
    decay = 0.9995
    min_eta = 1e-5
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()
    while time.time() < deadline and eta > min_eta:
        conv = _full_convolution(a)
        grad, cur_max = _subgradient_max(a, conv)

        # exponentiated update
        w = np.exp(-eta * grad)
        a = a * w
        # re‑normalise to the ℓ¹‑simplex of required sum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if rng.random() < 0.02:

            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # keep the best seen so far
        val = _max_convolution_fft(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] exp‑grad improvement: {best_val:.6f}")

        # step‑size decay
        eta *= decay

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing (exact O(n) updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill‑climbing.
    Moves a random amount of mass from a source index to a target index
    while keeping the convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9
    cooling = 0.998

    while time.time() < deadline:
        # pick a position attaining the current maximum
        max_pos = np.flatnonzero(b == cur_max)
        if max_pos.size == 0:
            break
        k = rng.choice(max_pos)

        # source index – weighted by contribution to this entry
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias towards small entries
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    b_best = _full_convolution(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Misc helpers
# ----------------------------------------------------------------------


def _interpolate_and_project(a: np.ndarray, new_n: int, target_sum: float) -> np.ndarray:
    """Linear interpolation of a to length new_n followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return _project_to_simplex(a_new, target_sum)


def _random_exponential(n: int, rng) -> np.ndarray:
    """Random exponential family a_i ∝ r^i with r∈(0.95,0.999)."""
    r = rng.uniform(0.95, 0.999)
    a = r ** np.arange(n, dtype=float)
    a = np.maximum(a, 0.0)
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock limit and returns the best sequence
    found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from historic solution if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            best_seq = a0.copy()
            best_score = _max_convolution_fft(a0)
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Failed to use height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Length schedule – gradually increase support size
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 50000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----- Initialise the candidate for this length -----
        if best_seq is not None and best_seq.size != n:
            a_start = _interpolate_and_project(best_seq, n, target_sum)
        else:
            # start from a random exponential shape
            a_start = _random_exponential(n, rng)
            a_start = _project_to_simplex(a_start, target_sum)

        # ----- Exponentiated‑gradient descent stage -----
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        gd_deadline = time.time() + remaining * 0.55   # roughly half the remaining budget

        a_opt, score_opt = _exp_gradient_descent(a_start, target_sum, gd_deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] exp‑grad improvement (n={n}) score={score_opt:.6f}")

        # ----- Exact hill‑climbing refinement -----
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        hc_deadline = time.time() + remaining * 0.35

        # make sure the vector matches the current length
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = float(b_hc.max())
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb improvement (n={n}) score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (use any remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # a short final exponentiated‑gradient run
        a_final, _ = _exp_gradient_descent(best_seq.copy(),
                                           target_sum,
                                           deadline - 3.0)

        # final hill‑climb
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 2.0)

        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for tidy output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never be needed)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    # drop trailing entries that are essentially zero
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
