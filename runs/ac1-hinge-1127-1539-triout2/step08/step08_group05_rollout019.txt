def _optimize_expgd(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.5
    it=0
    while time.time() < deadline:
        it+=1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        k = int(np.argmax(conv))
        # compute gradient g_i = 2 * a[k - i] if valid
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2 * a[j[mask]]
        # compute probability p = a / S
        p = a / S
        # multiplicative update: p *= np.exp(-eta * g / S)
        eta = eta0 / np.sqrt(it)  # schedule
        update_factor = np.exp(-eta * g / S)
        p = p * update_factor
        # renormalize
        p = p / p.sum()
        a = p * S

        if enforce_sym:
            a = _symmetrize(a)

        cur_max2 = np.max(_full_convolution(a))
        if cur_max2 < best_score:
            best_score = cur_max2
            best_a = a.copy()
    return best_a, best_score

"""
Step‑function optimizer for the self‑convolution bound.

The algorithm proceeds in several stages:

1. **Random‑family seeding** – generate many candidates from a rich
   collection of analytic families (uniform, power‑law, beta, arcsine,
   logistic, mixtures, …) and keep the best one.

2. **Linear‑programming “tightening”** – on a modest length (≈2000) we
   solve a small LP that maximises the total mass of a new distribution
   while respecting the current convolution peak.  Mixing the LP solution
   with the current vector provably cannot increase the peak and often
   reduces it.

3. **Upsampling** – the best small solution is interpolated to a much
   larger length (up to 30 000).  The sum is re‑normalised to
   √(2 n) (the scaling used by `evaluate_sequence`).

4. **Projected Adam** – a soft‑max (log‑sum‑exp) approximation of the
   max‑convolution is minimised with Adam; the iterates are projected
   onto the simplex (non‑negative, fixed ‑ℓ₁‑norm).

5. **Exponentiated‑gradient refinement** – a multiplicative update,
   natural for probability vectors, is applied to the exact max‑convolution
   sub‑gradient.  This efficiently moves mass from indices that contribute
   most to the peak to indices that contribute least.

6. **Short simulated‑annealing hill‑climb** – a final stochastic phase
   cleans up any remaining local irregularities.

All phases respect the time budget (`budget_s`) and the result is stored
in the global variable `height_sequence_1` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x ≥ 0, sum x = S}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto the scaled probability simplex."""
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Symmetry / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1‑i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families (more diverse than the original code)
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using many families."""
    fam = rng.integers(0, 13)          # extended set
    if fam == 0 and base_seq is not None:
        # up‑sample an existing good solution
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # uniform random
    elif fam == 2:
        # symmetric geometric decay
        rate = rng.uniform(0.93, 0.999)
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        # symmetric power‑law (i·(n‑i))^p
        p = rng.uniform(0.2, 4.0)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        # beta‑like skewed
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        # sinusoidal power
        p = rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        # monotone decreasing power‑law
        p = rng.uniform(0.3, 3.0)
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        # exponential decay
        lam = rng.uniform(0.001, 0.02) * n
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        # logistic S‑curve
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    elif fam == 9:
        # arcsine‑like U‑shape: 1/√(x(1‑x))
        x = (np.arange(n) + 0.5) / n
        eps = 1e-12
        a = 1.0 / np.sqrt(x * (1.0 - x) + eps)
    elif fam == 10:
        # mixture of two random families (weighted)
        w = rng.random()
        a1 = _sample_family(rng, n, None)
        a2 = _sample_family(rng, n, None)
        a = w * a1 + (1 - w) * a2
    elif fam == 11:
        # piecewise linear “tent” with random apex
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])
    else:
        # flat‑top plateau
        width = rng.integers(int(0.2 * n), int(0.5 * n))
        start = (n - width) // 2
        a = np.zeros(n)
        a[start:start + width] = 1.0

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser for the soft‑max approximation of the peak
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 800.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising a log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.03

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_shift = np.exp(scaled - max_scaled)
        w = exp_shift / exp_shift.sum()

        # gradient of the smooth objective
        # (see original code – convolution of a with reversed weights)
        n = a.shape[0]
        w_rev = w[::-1]
        conv_len = n + (2 * n - 1) - 1          # = 3n‑2
        fft_len = 1 << (conv_len - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        Fw = np.fft.rfft(w_rev, n=fft_len)
        full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional random jitter
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_expgd(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Multiplicative updates on the probability vector, driven by the exact max‑conv sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)          # target ℓ₁‑norm

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k = int(np.argmax(conv))
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # exact sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]      # factor 2 accounts for both (i,j) and (j,i)

        # work with probability vector p = a / S
        p = a / S
        eta = 0.5 / np.sqrt(it)           # decreasing step size
        factor = np.exp(-eta * grad / S)  # multiplicative factor
        p = p * factor
        p = p / p.sum()                   # re‑normalise to keep sum 1
        a = p * S

        if enforce_sym:
            a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (kept from the original code)
# ----------------------------------------------------------------------
def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    cur_max = best_score
    start = time.time()
    T0 = max(0.1 * cur_max, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.05 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming tightening step (small n)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 5) -> np.ndarray:
    """
    Given a feasible vector f (sum = √(2n)), perform one LP tightening:
    maximise the total mass of a new vector g subject to (f * g) ≤ M,
    where M = max(f * f).  The solution g is mixed back into f.
    """
    n = f.shape[0]
    target_sum = np.sqrt(2.0 * n)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        M = np.max(_full_convolution(f))

        # Build the dense convolution matrix A where (A @ g)[k] = (f * g)[k]
        conv_len = 2 * n - 1
        A = np.zeros((conv_len, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        c = -np.ones(n)                     # maximise sum(g)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(conv_len, M), bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # LP failed – fall back to the current vector
            break

        g = res.x
        s = g.sum()
        if s == 0:
            break
        g = g * (target_sum / s)            # renormalise to the exact ℓ₁‑norm

        # mix a fraction of the LP solution back into f
        mix = 0.2
        f = (1.0 - mix) * f + mix * g
        f = _project_simplex(f, target_sum)  # keep numerical safety

    return f


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Find a non‑negative step function that minimises the self‑convolution peak.
    The result is returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # 1) Random‑family seeding (≈20 % of the budget)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    phase1_deadline = start_time + 0.20 * budget_s

    while time.time() < phase1_deadline:
        n = int(rng.integers(1500, 3500))
        a = _sample_family(rng, n, base_seq=None)
        # normalise to Σa = √(2 n)
        target_sum = np.sqrt(2.0 * n)
        a = a / a.sum() * target_sum
        if rng.random() < 0.35:
            a = _symmetrize(a)
        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[seed] n={n} score={cur:.6f}")

    # --------------------------------------------------------------
    # 2) LP tightening on a modest size (≈20 % of the remaining budget)
    # --------------------------------------------------------------
    if best_seq is not None:
        # shrink / enlarge to a fixed small size for the LP step
        small_n = 2000
        if best_seq.shape[0] != small_n:
            small_seq = _upsample_seq(best_seq, small_n)
            small_seq = small_seq * (np.sqrt(2.0 * small_n) / small_seq.sum())
        else:
            small_seq = best_seq.copy()

        lp_deadline = time.time() + 0.20 * (deadline - time.time())
        small_seq = _lp_improve(small_seq, lp_deadline, rng, max_iters=5)
        cur = np.max(_full_convolution(small_seq))
        if cur < best_score:
            best_score = cur
            best_seq = small_seq.copy()
            print(f"[lp] n={small_n} score={cur:.6f}")

    # --------------------------------------------------------------
    # 3) Upsample to a large length (≈30 % of the remaining budget)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback uniform vector
        n_big = 8000
        a_big = rng.random(n_big)
        a_big = a_big * (np.sqrt(2.0 * n_big) / a_big.sum())
    else:
        cur_n = best_seq.shape[0]
        n_big = min(30000, int(cur_n * 8))          # expand aggressively but stay ≤30k
        a_big = _upsample_seq(best_seq, n_big)
        a_big = a_big * (np.sqrt(2.0 * n_big) / a_big.sum())

    # --------------------------------------------------------------
    # 4) Adam refinement (≈30 % of the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        t_adam = time.time() + 0.35 * (deadline - time.time())
        a_adam, sc_adam = _optimize_adam(a_big, t_adam, rng,
                                         max_alpha=800.0,
                                         enforce_sym=False)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(f"[adam] improved to {sc_adam:.6f}")

    # --------------------------------------------------------------
    # 5) Exponentiated‑gradient fine‑tuning (≈10 % of the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        t_exp = time.time() + 0.10 * (deadline - time.time())
        a_exp, sc_exp = _optimize_expgd(best_seq, t_exp, rng, enforce_sym=False)
        if sc_exp < best_score:
            best_score = sc_exp
            best_seq = a_exp.copy()
            print(f"[expgd] improved to {sc_exp:.6f}")

    # --------------------------------------------------------------
    # 6) Short simulated‑annealing polishing (≈5 % of the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _anneal_hillclimb(best_seq, deadline, rng,
                                              enforce_sym=False)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[anneal] final improved to {sc_final:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)}")

    # store globally for possible reuse in a later run
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
