"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence() = max_{k} (a⋆a)_k .
The search consists of three phases that repeat for increasing support
sizes:

1. **Broad random sampling** – generates many candidates from a
   collection of simple parametric families (exponential, power‑law,
   linear, piece‑wise exponential and a Dirichlet draw) and keeps the
   best one.

2. **Adam on a smoothed max** – a fast gradient descent on the smooth
   approximation  (1/β)·log∑exp(β·b_k)  with a geometrically growing β.
   The gradient is obtained analytically via FFTs.  After each step the
   vector is projected onto the ℓ¹‑simplex {x ≥ 0, Σx = √(2·n)}.

3. **Exact hill‑climbing** – a simulated‑annealing style move that
   transfers a small amount of mass from a “source” index to a “target”
   index while keeping the full convolution up‑to‑date in O(n).  The
   temperature is cooled very slowly (factor 0.9995) and the procedure
   is restarted a few times to escape local minima.

The three phases are applied to a schedule of lengths
[5000, 10000, 20000, 40000, 80000, 150000]; the budget (default 1000 s)
is split among them.  After the last length a final polishing pass
(Adam + hill‑climb) is performed.  The best sequence found is returned
as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a using FFT; returns 2·n‑1 entries."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a⋆a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max and its gradient (Adam optimiser)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b_k) where b = a⋆a.
    Returns (grad, b) with
        grad_i = 2·∑_k p_k·a_{k-i}
    where p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities (stable)
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev ⋆ p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 4e5,
                   beta_factor: float = 1.3) -> np.ndarray:
    """
    Adam optimiser for the smoothed‑max objective.
    The schedule for β grows geometrically until beta_target.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.15          # a bit more conservative than before
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)

        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid plateaux
        if step % 300 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase β
        if beta < beta_target and step % 200 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)"""
    r = rng.uniform(0.94, 0.9999)
    idx = np.arange(n, dtype=float)
    return r ** idx


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}"""
    p = rng.uniform(0.1, 1.8)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.4, 2.5)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential with a breakpoint."""
    b = rng.integers(int(0.25 * n), int(0.75 * n))
    r1 = rng.uniform(0.93, 0.999)
    r2 = rng.uniform(0.80, 0.97)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _family_dirichlet(n: int, rng) -> np.ndarray:
    """Draw from a Dirichlet distribution (equivalent to normalised gamma)."""
    # shape parameter > 0; using 0.5 gives a fairly spread distribution
    alpha = rng.uniform(0.3, 2.0, size=n)
    draw = rng.gamma(alpha, 1.0)
    return draw


_FAMILIES = [_family_exponential,
             _family_power_law,
             _family_linear,
             _family_piecewise_exp,
             _family_dirichlet]


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a random parametric family, add tiny jitter and project onto simplex."""
    a = _FAMILIES[rng.integers(len(_FAMILIES))](n, rng)
    a += rng.normal(scale=1e-3, size=n)
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float,
                   restarts: int = 2) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    The routine is executed `restarts` times (each with a fresh random
    jitter) and the best result among them is returned.
    """
    n = a0.size
    best_a = a0.copy()
    best_max = b0.max()

    rng = np.random.default_rng()

    for restart in range(restarts):
        # start from the current best and add a small random perturbation
        a = best_a.copy()
        if restart > 0:
            # jitter of order 1 % of sqrt(2·n) to explore a new basin
            jitter = rng.normal(scale=0.01 * np.sqrt(2.0 * n), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))

        b = _full_convolution(a) if restart > 0 else b0.copy()
        cur_max = b.max()
        T = cur_max * 0.15 + 1e-12          # start temperature
        cooling = 0.9995                   # slower cooling → deeper search

        while time.time() < deadline:
            # pick one of the positions attaining the current maximum
            max_idxs = np.flatnonzero(b == cur_max)
            if max_idxs.size == 0:
                break
            k = rng.choice(max_idxs)

            # pick source i weighted by its contribution to this max entry
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            j_range = k - i_range
            contrib = a[i_range] * a[j_range]
            total = contrib.sum()
            if total > 0.0:
                probs = contrib / total
                i = rng.choice(i_range, p=probs)
            else:
                i = rng.integers(0, n)

            if a[i] < 1e-14:
                continue

            # target index – bias towards small entries but keep some randomness
            if rng.random() < 0.4:
                inv = 1.0 / (a + 1e-12)
                probs_t = inv / inv.sum()
                j = rng.choice(n, p=probs_t)
            else:
                j = rng.integers(0, n)

            if i == j:
                continue

            # amount of mass to move
            max_delta = a[i] * 0.70
            delta = rng.random() * max_delta
            if delta <= 0.0:
                continue

            # ----- update convolution (exact O(n)) -----
            # remove contribution of source i
            b[i:i + n] -= 2.0 * delta * a
            if 2 * i < b.size:
                b[2 * i] += delta * delta

            # add contribution of target j
            b[j:j + n] += 2.0 * delta * a
            if 2 * j < b.size:
                b[2 * j] += delta * delta

            # correct cross term at i+j (appears twice)
            kij = i + j
            if kij < b.size:
                b[kij] -= 2.0 * delta * delta

            new_max = b.max()

            # Metropolis acceptance
            accept = False
            if new_max < cur_max:
                accept = True
            else:
                if rng.random() < np.exp(-(new_max - cur_max) / T):
                    accept = True

            if accept:
                a[i] -= delta
                a[j] += delta
                cur_max = new_max
                if cur_max < best_max - 1e-12:
                    best_max = cur_max
                    best_a = a.copy()
                    # occasional progress report
                    print(f"[{time.time():.1f}] hill‑climb restart {restart} improvement: {best_max:.6f}")

            T *= cooling

    # recompute exact convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock limit and returns the best sequence
    found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a historic construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            score0 = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = score0
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) score={score0:.6f}")
        except Exception as e:
            print(f"Unable to seed from height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Length schedule – we concentrate effort on larger supports
    # ------------------------------------------------------------------
    schedule = [5000, 10000, 20000, 40000, 80000, 150000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1) Random parametric sampling (more exhaustive)
        # --------------------------------------------------------------
        # number of samples is adaptive: more samples for smaller n,
        # guaranteeing roughly the same total effort.
        n_samples = max(20, int(100000 / n))
        for _ in range(n_samples):
            if time.time() >= deadline:
                break
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] random family improvement (n={n}) score={score:.6f}")

        # --------------------------------------------------------------
        # 2) Adam optimisation (smoothed‑max descent)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate roughly 40 % of the remaining budget to Adam
        gd_time = min(remaining * 0.40, 300.0)
        gd_deadline = time.time() + gd_time

        # initialise from current best (upsampled if needed)
        if best_seq is not None and best_seq.size != n:
            a_start = _upsample_sequence(best_seq, n)
        else:
            a_start = _random_initial_sequence(n, rng)

        a_opt = _adam_optimize(a_start, gd_deadline,
                               beta_start=2.0,
                               beta_target=5e5,
                               beta_factor=1.3)
        score_opt = _max_convolution_fft(a_opt)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Adam descent improvement (n={n}) score={score_opt:.6f}")

        # --------------------------------------------------------------
        # 3) Exact hill‑climbing (simulated annealing)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        hc_time = min(remaining * 0.30, 250.0)
        hc_deadline = time.time() + hc_time

        # make sure we have a vector of the right length
        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline, restarts=3)
        score_hc = float(b_hc.max())
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb improvement (n={n}) score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – any leftover time is spent on a last Adam + HC
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final Adam (sharper β)
        a_final = _adam_optimize(best_seq.copy(),
                                 deadline - 3.0,
                                 beta_start=4.0,
                                 beta_target=8e5,
                                 beta_factor=1.2)

        # final hill‑climb (more restarts)
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final,
                                          deadline - 2.0,
                                          restarts=5)
        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy return value
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 5000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (max‑convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
