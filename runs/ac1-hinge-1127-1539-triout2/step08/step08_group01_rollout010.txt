def solve_lp(a, M, target_sum):
    n = a.size
    # Precompute row and col indices static for given n; maybe caching outside.
    # But for simplicity recalc each call.
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)

    if not res.success:
        # fallback maybe return a copy of a

    return res.x

"""
Improved step‑function design for ``evaluate_sequence``.

Algorithm
---------

1.  Initialise the candidate from the historic global ``height_sequence_1``
    (or a random simplex point if unavailable).  The ℓ¹‑norm is forced to
    √(2 n) so the evaluation equals the maximal entry of the self‑convolution.

2.  Repeatedly *linear‑program* a new direction:
        maximise   Σ g_i
        subject to  g_i ≥ 0 , (a * g)_k ≤ M    for all k ,
    where ``a`` is the current sequence and ``M = max(a*a)``.
    The convolution with a fixed ``a`` is linear in ``g``, therefore the
    problem is an LP.  The solution ``g`` spreads mass towards indices that
    are “safe’’ with respect to the current bound.

3.  Rescale ``g`` to the required ℓ¹‑norm and replace
        a ← (1‑t)·a + t·g    (t ∈ (0,1) decreasing over time).
    This convex move never increases the maximal convolution value,
    and usually reduces it.

4.  When the LP loop stops (time limit or stagnation) run a few
    projected Adam gradient‑descent steps on a smooth max‑approximation
    followed by the deterministic hill‑climb from the previous code.
    This refines the solution locally.

5.  Trim trailing near‑zeros, report the final objective and return the
    sequence as a plain list.

The implementation respects the 1000 s wall‑clock budget, uses only
NumPy and SciPy (the LP solver *highs*), and prints progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic geometry helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns 2 n‑1 array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and Adam optimiser (kept from the previous run)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.25 * target_sum
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Random initialiser (kept for possible restarts)
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative point on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)
    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)
    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target
    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )
    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a onto a new length, followed by ℓ¹‑projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing (biased version from the previous run)
# ----------------------------------------------------------------------
def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves mass while keeping the
    full self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9997                    # slower cooling

    while time.time() < deadline:
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # Update convolution efficiently (O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Linear‑programming helper (template for the Toeplitz constraint matrix)
# ----------------------------------------------------------------------
def _constraint_matrix_template(n: int):
    """
    Return pre‑computed row/col index arrays for the matrix
        A_{k,i} = a_i   (for rows k = i … i+n‑1).
    The data array must be supplied separately for each new vector a.
    """
    if not hasattr(_constraint_matrix_template, "cache"):
        _constraint_matrix_template.cache = {}
    if n not in _constraint_matrix_template.cache:
        col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
        row_idx = col_idx + np.tile(np.arange(n, dtype=np.int32), n)
        _constraint_matrix_template.cache[n] = (row_idx, col_idx)
    return _constraint_matrix_template.cache[n]


def _solve_lp_step(a: np.ndarray, M: float, target_sum: float,
                   time_limit: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to g_i ≥ 0, (a * g)_k ≤ M   ∀k,
    with variable upper bounds set to ``target_sum``.
    Returns the optimal g (or the current a if the LP fails).
    """
    n = a.size
    row_idx, col_idx = _constraint_matrix_template(n)
    data = np.repeat(a, n)                     # a_i repeated n times
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)                            # minimise -sum ⇒ maximise sum
    bounds = [(0.0, target_sum) for _ in range(n)]

    # SciPy's HiGHS accepts a ``time_limit`` option (seconds)
    # Older versions ignore it – we simply break later if we run out of time.
    options = {"presolve": True}
    try:
        # recent SciPy versions accept ``time_limit`` in the options dict
        options["time_limit"] = max(0.0, time_limit)
    except Exception:
        pass

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options=options)

    if not res.success:
        # LP failed – fall back to the current point
        return a.copy()
    return res.x

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a = np.asarray(globals()["height_sequence_1"], dtype=float)
            n = a.size

        except Exception as e:

            n = 2000
            a = _random_initial_sequence(n, rng)
    else:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    mix_factor = 0.5               # initial convex‑combination weight
    iteration = 0
    while time.time() < deadline - 5.0:

        # Allocate a tiny slice for the LP solver (so we respect the global budget)

        # Rescale to the required ℓ¹‑norm

        # Convex blend

        new_score = _max_convolution_fft(a_new)

        if new_score < best_score - 1e-9:
            best_seq, best_score = a_new, new_score
            print(f"[{time.time():.1f}] LP‑mix iter {iteration}: improvement to {best_score:.6f}")
            # shrink the step size a little as we approach optimum
            mix_factor = max(0.05, mix_factor * 0.93)
        else:
            # no progress – reduce the mixing weight more aggressively
            mix_factor = max(0.05, mix_factor * 0.80)

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final refinement: Adam + hill‑climb (as in the previous script)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # Adam phase

        best_seq = _adam_gradient_descent(best_seq, target_sum, ad_deadline)
        best_score = _max_convolution_fft(best_seq)

        # Hill‑climbing phase

            b_cur = _full_convolution(best_seq)
            hc_deadline = deadline - 1.0
            best_seq, b_cur = _hill_climbing_improved(best_seq, b_cur, hc_deadline)

    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
