"""
Advanced local‐search optimizer for the step‑function benchmark.

The algorithm follows four phases:

1. **Parametric warm‑start** – a quick grid / random search on a reduced
   problem (power‑law × exponential shape) and interpolation to the final
   size.
2. **Exponential (mirror) descent** – minimizes a soft‑max surrogate of
   the maximal convolution entry.
3. **Projected gradient descent with restarts** – refines the solution
   obtained in phase 2.
4. **Intensive max‑reduction refinement** – a custom hill‑climbing loop
   that repeatedly moves a small amount of mass from indices that
   contribute most to the current maximal convolution entry to indices
   that contribute least, using an exact incremental update of the
   convolution.  This phase is the main source of improvement over the
   previous baseline and aims to push the score below the target 1.503.

All steps respect the non‑negativity and ℓ¹‑norm constraint
∑a_i = √(2 n).  The final sequence is returned as a plain list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Search a two‑parameter family a_i = (i+1)^(-p) · exp(−λ·i / small_n)
    on a reduced problem.
    """
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – exponential (multiplicative) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – projected gradient descent with restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – intensive max‑reduction refinement (hill climbing)
# ----------------------------------------------------------------------


def _max_reduction_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing loop that moves mass from indices contributing heavily
    to the current maximal convolution entry to indices contributing
    little.  The convolution is updated incrementally (O(n) per move)
    to keep the procedure fast.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    b = _conv_fft(a)                     # full convolution, length 2·n‑1
    best_a = a.copy()
    best_M = b.max()
    best_score = 2.0 * n * best_M / (target_sum ** 2)   # equals best_M for our normalisation
    rng = np.random.default_rng()
    it = 0
    stagnant = 0
    max_stagnant = 1500
    last_full = time.time()

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        M = b[k_max]

        # second‑largest value (ignoring the current maximum)
        # cheap O(n) operation – acceptable for n≈12000
        M2 = np.max(np.concatenate([b[:k_max], b[k_max + 1:]]))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        partner_idx = k_max - idx
        sens = a[idx] * a[partner_idx]

        donor = idx[np.argmax(sens)]
        receiver = idx[np.argmin(sens)]
        if donor == receiver:
            # pick any other index as receiver
            cand = idx[idx != donor]
            if cand.size == 0:
                continue
            receiver = rng.choice(cand)

        part_d = a[k_max - donor]
        part_r = a[k_max - receiver]
        diff = part_d - part_r
        if diff <= 0:
            # try a slightly different receiver (low‑percentile candidates)
            low_cand = idx[sens < np.percentile(sens, 10)]
            low_cand = low_cand[low_cand != donor]
            if low_cand.size == 0:
                continue
            receiver = rng.choice(low_cand)
            part_r = a[k_max - receiver]
            diff = part_d - part_r
            if diff <= 0:
                continue

        # reasonable step size – aim to lower the current peak towards M2
        max_delta = a[donor] * 0.3                     # never move more than 30 % of donor
        if M > M2:
            delta_target = (M - M2) / (2.0 * diff) * 0.9
            delta = min(max_delta, max(delta_target, max_delta * 0.05))
        else:
            delta = max_delta * 0.05

        if delta < 1e-12:
            delta = a[donor] * 0.02

        # backup current state (cheap copies, ~300 KB total)
        a_backup = a.copy()
        b_backup = b.copy()

        # ---- apply move -------------------------------------------------
        # 1) remove δ from donor (using the *old* a)
        b[donor:donor + n] -= 2.0 * delta * a
        a[donor] -= delta
        # 2) add δ to receiver (using the *updated* a)
        b[receiver:receiver + n] += 2.0 * delta * a
        a[receiver] += delta
        # -----------------------------------------------------------------

        new_M = b.max()
        if new_M < M - 1e-12:
            # successful improvement
            if new_M < best_M:
                best_M = new_M
                best_a = a.copy()
                best_score = 2.0 * n * best_M / (target_sum ** 2)
                print(f"[{time.time():.2f}] Iter {it:6d} best M={best_M:.6f}")
                stagnant = 0
            else:
                stagnant += 1
        else:
            # reject – restore previous state
            a = a_backup
            b = b_backup
            stagnant += 1

        # periodic full recomputation to avoid drift
        if it % 800 == 0:
            b = _conv_fft(a)

        # if we have been stagnant for a while, inject a small random jitter
        if stagnant >= max_stagnant:
            i, j = rng.choice(n, size=2, replace=False)
            eps = min(a[i] * 0.1, a[i])
            if eps > 0:
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                stagnant = 0

        # keep exact ℓ¹‑norm (tiny numerical drift may appear)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)
            b = _conv_fft(a)

        it += 1
        # safety – if many iterations without progress, break early
        if it >= 20000:
            break

    # final normalisation (in case of tiny drift)
    best_a = best_a * (target_sum / best_a.sum())
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initial sequence (fallback)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init.shape[0] != n:
            init = _resize_sequence(init, n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # --------------------------------------------------------------
    # Phase 1 – cheap parametric search on a reduced problem
    # --------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(12.0, (deadline - time.time()) * 0.07)
    best_small_seq, _ = _grid_param_search(small_n, small_target, param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # --------------------------------------------------------------
    # Phase 2 – exponential descent (≈20 % of remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.20
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")
    else:
        best_seq = init
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase 3 – projected gradient descent with restarts (≈30 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.30
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – intensive max‑reduction refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _max_reduction_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After intensive refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean up and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # evaluate with the provided benchmark (may raise if not imported)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
