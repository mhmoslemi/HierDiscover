"""
Enhanced step‑function search.

The algorithm works in three stages:

1. **Random pool generation** – draw many candidates from a collection of
   analytic families (geometric, power‑law, beta‑like, sinusoidal, etc.).
   Each candidate is projected onto the (scaled) probability simplex
   Σa = √(2 n) and its internal peak max(conv) is recorded.  The best
   `top_k` candidates are kept in a pool.

2. **Evolutionary refinement** – repeatedly select two parents from the pool
   (biased toward lower peaks), create an offspring by linear blending or
   splicing, optionally up‑sample, jitter and smooth it, renormalise, and
   evaluate.  If the offspring looks promising a short projected‑Adam run
   (soft‑max smoothing of the peak) is performed; better offspring replace
   the worst pool members.  Occasionally the current global best is refined
   with a longer Adam run.

3. **Final Adam refinement** – after the time budget is almost exhausted a
   final Adam optimisation is applied to the best sequence.

All sequences are kept non‑negative and are automatically scaled so that
`evaluate_sequence` returns exactly the internal peak value (the objective
`max(conv)`).  The implementation uses only NumPy and runs within the
provided 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the probability simplex scaled by s (non‑negative)."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families for the initial sampling stage
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int,
                   base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a variety of families."""
    fam = rng.integers(0, 9)
    if fam == 0 and base_seq is not None:
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = (left + left[::-1])
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                 # symmetric power‑law
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                # beta‑like skewed
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                 # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)                 # monotone decreasing power‑law
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    else:
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max approximation of the convolution peak
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 400.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.03
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998

    return best_a, best_score

    phase0_end = start + 0.12 * budget_s

    while time.time() < phase0_end:
        n = int(rng.integers(2500, 30001))
        a = _sample_family(rng, n, None)

        # occasional forced symmetry (does not dominate the search)
        if rng.random() < 0.15:
            a = _symmetrize(a)

        # Scale to Σa = √(2 n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        score = np.max(_full_convolution(a))

        if len(pool) < top_k:
            pool.append((score, a.copy()))
            pool.sort(key=lambda x: x[0])
        else:
            if score < pool[-1][0]:
                pool[-1] = (score, a.copy())
                pool.sort(key=lambda x: x[0])

    attempts = 0
    best_score, best_seq = pool[0]

    # ------------------------------------------------------------------
    # Phase 1 – evolutionary refinement with occasional Adam polishing
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # ----- parent selection (biased towards low scores) -----
        scores = np.array([s for s, _ in pool])
        prob = np.exp(-scores)      # higher weight for smaller score
        prob /= prob.sum()
        i1, i2 = rng.choice(len(pool), size=2, replace=False, p=prob)
        p1 = pool[i1][1]
        p2 = pool[i2][1]

        # ----- produce offspring ---------------------------------
        w = rng.random()
        child = w * p1 + (1.0 - w) * p2

        # occasionally splice instead of blending
        if rng.random() < 0.4:
            cut = rng.integers(1, child.size)
            child = np.concatenate((p1[:cut], p2[cut:]))

        # random target length
        target_n = int(rng.integers(2000, 30001))
        if target_n != child.size:
            child = _upsample_seq(child, target_n)

        # jitter – move a tiny amount of mass
        if rng.random() < 0.35:
            i, j = rng.integers(0, target_n, size=2)
            if i != j and child[i] > 0.0:
                delta = 0.07 * child[i] * rng.random()
                child[i] -= delta
                child[j] += delta

        # light smoothing (optional)
        if rng.random() < 0.2:
            kern = np.array([0.25, 0.5, 0.25])
            child = np.convolve(child, kern, mode='same')

        # normalise again
        S = np.sqrt(2.0 * target_n)
        child = child / child.sum() * S

        # ----- quick evaluation -----
        cur_score = np.max(_full_convolution(child))

        # ----- short Adam refinement if promising -----
        need_refine = (cur_score < pool[-1][0]) or (rng.random() < 0.07)
        if need_refine and (deadline - time.time() > 2.0):
            t_refine_end = min(time.time() + 1.2, deadline)
            child, cur_score = _optimize_adam(child, t_refine_end, rng,
                                             max_alpha=500.0,
                                             enforce_sym=False)

        # ----- insert into pool if better -----
        if cur_score < pool[-1][0]:
            pool[-1] = (cur_score, child.copy())
            pool.sort(key=lambda x: x[0])
            if cur_score < best_score:
                best_score, best_seq = cur_score, child.copy()
                print(f"[evo] new global best {best_score:.6f} n={target_n} attempts={attempts}")

        # ----- occasional intensive Adam on the current best -----
        if rng.random() < 0.02 and (deadline - time.time() > 3.0):
            t_refine_end = min(time.time() + 2.5, deadline)
            refined_seq, refined_score = _optimize_adam(best_seq, t_refine_end,
                                                       rng,
                                                       max_alpha=600.0,
                                                       enforce_sym=False)
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_seq.copy()
                # replace worst pool entry with this improved version
                pool[-1] = (best_score, best_seq.copy())
                pool.sort(key=lambda x: x[0])
                print(f"[best‑adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final longer Adam refinement on the best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_seq, best_score = _optimize_adam(best_seq, deadline, rng,
                                             max_alpha=800.0,
                                             enforce_sym=False)

    # ------------------------------------------------------------------
    # Return the final result (store globally)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
