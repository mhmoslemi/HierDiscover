"""
Propose a step‑function that minimises the benchmark score.

Algorithm
---------
1. Choose a problem size `n` based on the time budget.
2. Warm‑start by searching two 2‑parameter families
        a_i = (i+1)^(-p) * exp(-λ·i/n)                     (edge‑decay)
        a_i = (i+1)^(-p1) * (n-i)^(-p2) * exp(-λ·i/n)     (U‑shaped)
   on a reduced problem (≈ n/4).  The best candidate is interpolated
   to the target length and normalised to Σa = √(2n).
3. Adam‑style multiplicative mirror descent on the soft‑max surrogate
   of `max(conv)`.  The temperature β grows exponentially (β←2β) so the
   optimisation first smoothes the objective and then focuses on the
   true maximum.  The iterate is always projected onto the ℓ¹‑simplex
   to keep the sum exactly `target_sum`.
4. Deterministic mass‑transfer + simulated‑annealing refinement:
   repeatedly move a small fraction of mass from indices that most
   contribute to the current maximum convolution entry to indices that
   contribute least, occasionally accepting uphill moves with a cooling
   temperature.
5. Truncate trailing near‑zero entries, evaluate once and return the
   best sequence found.

All steps respect non‑negativity and the ℓ¹‑norm constraint, so the
result can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score = 2·n·max(conv) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max gradient of the maximal convolution entry
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where conv = a*a and grad = ∂softmax_max/∂a.
    The gradient is 2·(reverse(a) * p) on the overlap region,
    where p is the soft‑max distribution over convolution entries.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv.max()
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # reverse(a) convolution with p
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    rev_conv = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * rev_conv[n - 1: n - 1 + n]  # the overlapping window
    return grad, conv


# ----------------------------------------------------------------------
# Parameteric families for warm‑start
# ----------------------------------------------------------------------


def _family_edge_decay(i: np.ndarray, p: float, lam: float, n: int):
    """a_i = (i+1)^(-p) * exp(-λ·i/n)."""
    return (i + 1) ** (-p) * np.exp(-lam * i / n)


def _family_U_shape(i: np.ndarray, p1: float, p2: float, lam: float, n: int):
    """a_i = (i+1)^(-p1) * (n-i)^(-p2) * exp(-λ·i/n)."""
    return (i + 1) ** (-p1) * (n - i) ** (-p2) * np.exp(-lam * i / n)


def _normalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Rescale a non‑negative vector to have ℓ¹‑norm = target_sum."""
    s = seq.sum()
    if s == 0:
        return seq
    return seq * (target_sum / s)


# ----------------------------------------------------------------------
# Grid + random search on a reduced problem
# ----------------------------------------------------------------------


def _grid_search_small(
    small_n: int,
    target_sum: float,
    deadline: float,
):
    """Search the two families on a problem of size `small_n`."""
    i = np.arange(small_n, dtype=np.float64)
    rng = np.random.default_rng()
    best_seq = None
    best_sc = np.inf

    # deterministic grid for both families
    p_vals = np.linspace(0.2, 1.2, 15)
    lam_vals = np.linspace(0.0, 6.0, 13)
    p2_vals = np.linspace(0.2, 1.2, 9)

    # ---- edge‑decay family ----
    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = _family_edge_decay(i, p, lam, small_n)
            seq = _normalise(seq, target_sum)
            sc = _score(seq)
            if sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid edge p={p:.3f} λ={lam:.3f} → {sc:.6f}")

    # ---- U‑shaped family ----
    for p1 in p_vals:
        for p2 in p2_vals:
            for lam in lam_vals:
                if time.time() > deadline:
                    break
                seq = _family_U_shape(i, p1, p2, lam, small_n)
                seq = _normalise(seq, target_sum)
                sc = _score(seq)
                if sc < best_sc:
                    best_sc = sc
                    best_seq = seq.copy()
                    print(f"[{time.time():.2f}] Grid U p1={p1:.3f} p2={p2:.3f} λ={lam:.3f} → {sc:.6f}")

    # Random refinements until time runs out
    while time.time() < deadline:
        p = rng.uniform(0.15, 1.3)
        p2 = rng.uniform(0.15, 1.3)
        lam = rng.exponential(1.0) * 3.0
        use_u = rng.random() < 0.5
        if use_u:
            seq = _family_U_shape(i, p, p2, lam, small_n)
        else:
            seq = _family_edge_decay(i, p, lam, small_n)
        seq = _normalise(seq, target_sum)
        sc = _score(seq)
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            typ = "U" if use_u else "edge"
            print(f"[{time.time():.2f}] Random {typ} p={p:.3f} p2={p2:.3f} λ={lam:.3f} → {sc:.6f}")

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Adam‑style mirror descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
):
    """
    Multiplicative Adam on the soft‑max approximate objective.
    Returns the best sequence found.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    beta_temp = 1.0
    max_beta = 65536.0

    # learning‑rate base (scale with target_sum)
    base_lr = 0.35 * target_sum

    rng = np.random.default_rng()
    it = 1

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta_temp)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # multiplicative update in log‑space
        update = -base_lr * m_hat / (np.sqrt(v_hat) + eps) / (beta_temp + 1.0)
        a = a * np.exp(update)

        # re‑normalise to exact ℓ¹ target (preserves positivity)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} β={beta_temp:.1f} score={val:.6f}")

        # temperature growth
        if it % 600 == 0 and beta_temp < max_beta:
            beta_temp = min(beta_temp * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Mass‑transfer + simulated‑annealing refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
):
    """
    Repeatedly move a small fraction of mass away from indices that
    contribute most to the current maximal convolution entry towards
    low‑contributing indices.  Occasionally accepts uphill moves with a
    cooling temperature.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size

    best_a = a.copy()
    best_val = _score(a)

    conv = _conv_fft(a)
    rng = np.random.default_rng()

    # annealing schedule
    T0 = 0.02 * target_sum
    T_end = 0.0005 * target_sum
    total_time = deadline - time.time()
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = max(T_end, T0 * (1.0 - elapsed / total_time))

        # index of maximal convolution entry
        k_max = int(np.argmax(conv))

        # sensitivities: how much each a[i] participates in that entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = np.zeros_like(a)
        sens[idx] = a[k_max - idx]          # a[i] * a[k_max-i] contribution

        # choose donor with probability proportional to a[i]·sens[i]
        weight = a * sens
        wsum = weight.sum()
        if wsum > 0:
            donor = rng.choice(n, p=weight / wsum)
        else:
            donor = rng.integers(n)

        # choose a receiver among low‑weight indices
        low_thr = np.percentile(weight, 15)
        candidates = np.where(weight <= low_thr)[0]
        receiver = rng.choice(candidates) if candidates.size else rng.integers(n)

        if donor == receiver:
            continue

        # propose moving a modest fraction of donor's mass
        max_frac = 0.07
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        # recompute convolution only locally?  Simpler: recompute full
        conv = _conv_fft(a)
        val = _score(a)

        # accept if improvement or with Metropolis probability
        improve = val < best_val
        if improve or rng.random() < np.exp(-(val - best_val) / max(T, 1e-12)):
            if improve:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Transfer accept improve score={val:.6f}")
        else:
            # reject – revert move
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # occasional re‑normalisation (numerical drift)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val
