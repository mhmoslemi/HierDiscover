def propose_candidate(seed=42,budget_s=1000):
    import time, math, numpy as np
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    rng = np.random.default_rng(seed)

    lengths = [2000, 4000, 8000]  # maybe 2000 only
    for n in lengths:
        if time.time() >= deadline: break
        # generate initial from families
        # families: exponential, offset exponential, power law offset, mixed
        # sample many candidates per n
        trials = 2000
        for _ in range(trials):
            if time.time() >= deadline: break
            # choose family
            family = rng.integers(0,5)
            # generate sequence
            i = np.arange(n, dtype=float)
            if family == 0:
                # exp
                lam = rng.uniform(0.0003, 0.002)  # λ
                a = np.exp(-lam * i)
            elif family == 1:
                # exp + offset
                lam = rng.uniform(0.0003, 0.002)
                eps = rng.uniform(0.0, 0.3)
                a = eps + np.exp(-lam * i)
            elif family == 2:
                # power law
                p = rng.uniform(0.05, 0.6)
                a = (i + 1) ** (-p)
            elif family == 3:
                # power law + offset
                p = rng.uniform(0.05, 0.6)
                gamma = rng.uniform(0.1, 5.0)
                a = (i + gamma) ** (-p)
            else:
                # mixed
                p = rng.uniform(0.05, 0.6)
                lam = rng.uniform(0.0003, 0.002)
                a = (i + 1) ** (-p) * np.exp(-lam * i)

            # sometimes add small jitter for variation
            # maybe normalize by scaling factor? Not needed

            # Evaluate
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} family={family} val={val:.6f}")

        # after random search, run hill climbing on best_seq for this n
        if best_seq is not None and best_seq.size == n:
            b = _full_convolution(best_seq)
            hill_deadline = deadline - 1.0
            best_seq, b = _hill_climbing_improved(best_seq, b, hill_deadline)
            val = evaluate_sequence(best_seq.tolist())
            if val < best_val:
                best_val = val
                print(f"[{time.time():.1f}] Post hill climb improvement: {val:.6f}")
    # after loops, final polish with gradient descent maybe
    if best_seq is not None:
        # final gradient descent with higher iterations.
        # Use _adam_gradient_descent with larger deadline maybe.
        gd_deadline = deadline - 1.0
        if gd_deadline > time.time():
            best_seq = _adam_gradient_descent(best_seq, best_seq.sum(), gd_deadline)
            best_val = evaluate_sequence(best_seq.tolist())
            print(f"[{time.time():.1f}] Final GD val {best_val:.6f}")

    # trim near-zero trailing
    # ...
    return best_seq.tolist()

"""
Propose a candidate step‑function that (approximately) minimises
``evaluate_sequence``.  The search is based on a cheap objective
``obj(a) = 2·n·max(a*a)/(sum(a)**2)`` computed with an FFT, and explores
a few simple parametric families (exponential, power‑law, mixed and a
regularised arcsine shape).  For each family many random draws are tried,
the best candidate is refined with a fast O(n) hill‑climber that moves
mass between two positions while keeping the full self‑convolution up‑to‑date.
The whole routine respects a 1000 s wall‑clock budget and returns the
best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale invariant)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return the value of evaluate_sequence without the security checks."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    # protect from division by zero (should never happen in our usage)
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Fast hill‑climbing (exact convolution updates, O(n) per move)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-12               # initial temperature
    cooling = 0.9995                          # gentle cooling

    while time.time() < deadline:
        # pick a currently maximal convolution position
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index – choose proportionally to its contribution to position M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index – bias towards indices that currently contribute little
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]
        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        delta = rng.random() * a[i] * 0.8
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Length schedule – a few moderate sizes are enough for the budget
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]   # larger lengths give more flexibility

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = None   # we do not need an explicit L1 normalisation

        # --------------------------------------------------------------
        # Random sampling of simple parametric families
        # --------------------------------------------------------------
        trials = 2000   # reduced when time runs low
        for _ in range(trials):
            if time.time() >= deadline:
                break

            i = np.arange(n, dtype=float)

            # pick a family
            family = rng.integers(0, 5)

            if family == 0:                # pure exponential
                lam = rng.uniform(0.0003, 0.003)
                a = np.exp(-lam * i)

            elif family == 1:              # exponential + small constant offset
                lam = rng.uniform(0.0003, 0.003)
                eps = rng.uniform(0.0, 0.5)
                a = eps + np.exp(-lam * i)

            elif family == 2:              # power‑law
                p = rng.uniform(0.05, 0.6)
                a = (i + 1.0) ** (-p)

            elif family == 3:              # offset power‑law
                p = rng.uniform(0.05, 0.6)
                gamma = rng.uniform(0.1, 5.0)
                a = (i + gamma) ** (-p)

            elif family == 4:              # mixed power‑law × exponential
                p = rng.uniform(0.05, 0.6)
                lam = rng.uniform(0.0003, 0.003)
                a = (i + 1.0) ** (-p) * np.exp(-lam * i)

            else:                          # regularised arcsine‑like shape
                gamma = rng.uniform(0.1, 5.0)
                a = 1.0 / np.sqrt((i + gamma) * (n - 1 - i + gamma))

            # protect against NaNs or negative values (should not happen)
            a = np.maximum(a, 0.0)

            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} family={family} val={val:.6f}")

        # --------------------------------------------------------------
        # Refine the current best sequence for this length with hill climbing
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            b = _full_convolution(best_seq)
            # give the hill‑climber most of the remaining time for this n
            hc_deadline = min(deadline - 0.1, time.time() + (deadline - time.time()) * 0.3)
            best_seq, b = _hill_climbing_improved(best_seq, b, hc_deadline)
            val = _objective(best_seq)
            if val < best_val:
                best_val = val
                print(f"[{time.time():.1f}] post‑HC improvement n={n} val={val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        # one more short hill‑climb on the overall best
        b = _full_convolution(best_seq)
        final_deadline = deadline - 0.5
        best_seq, b = _hill_climbing_improved(best_seq, b, final_deadline)
        best_val = _objective(best_seq)
        print(f"[{time.time():.1f}] final polishing val={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never be used)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
