"""
Optimiser for the step‑function “max‐convolution’’ problem.

The goal is to minimise
    evaluate_sequence(a) = 2·n·max(a*a)/(sum(a)²)

If we keep the ℓ₁‑norm ‖a‖₁ = √(2·n) the factor 2·n/‖a‖₁² equals 1,
hence the problem reduces to minimising the peak of the self‑convolution.
The optimiser works in five phases:

1️⃣  Random exploration of many parametric families (including several
    monotone – i.e. non‑symmetric – families).  The best few candidates
    are kept.

2️⃣  For each of those candidates we run an Adam optimiser on the
    simplex (log‑softmax parametrisation).  The gradient is the exact
    sub‑gradient of the current maximal convolution entry (τ‑soft‑max
    with τ→0).  Warm‑starts from the best candidate are repeatedly
    scaled to larger lengths.

3️⃣  A long final Adam run on the currently best length.

4️⃣  A cheap stochastic hill‑climb that moves small fractions of mass
    between random indices while preserving the ℓ₁‑norm.

5️⃣  Final sanity check via the supplied ``evaluate_sequence`` and
    storage of the result in the global ``height_sequence_1`` variable.

The code respects a 1000 s budget, leaves a few seconds safety margin,
and prints a short log of the progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT (fast)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Gradient of (soft) max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    # exact sub‑gradient when τ is effectively zero
    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * conv(mask, rev(a))
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    # extract the n entries that correspond to the derivative
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the *hard* peak of a*a.
    a0 must already satisfy the ℓ₁‑norm of sqrt(2·n).
    Returns the best sequence found and its peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # Initialise logits from a0 (log‑softmax)
    eps = 1e-12
    prob = np.maximum(a0 / S, eps)
    prob = prob / prob.sum()
    p = np.log(prob)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.55                     # a little more aggressive than before

    # Temperature for the τ‑soft‑max – start high, decay to 0
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 2.0)      # a gentle start

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- recover a from logits ----------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S                    # a satisfies the ℓ₁‑norm exactly

        # ---------- convolution & gradient ----------
        g, peak = _softmax_gradient(a, tau)

        # keep the best *hard* peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---------- gradient w.r.t. logits ----------
        mu = np.dot(a, g) / S           # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)           # shape (n,)

        # optional clipping for numerical stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 8.0 * S:
            grad_p = grad_p * (8.0 * S / clip_val)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- schedules ----------
        tau *= 0.9995                    # keep a little smoothing for longer
        lr *= 0.9999                     # very gentle decay

        # occasional jitter to escape tiny plateaus
        if it % 1000 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak

# Parametric families for random initialisation

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric edge‑heavy power‑law (used previously)."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric edge‑heavy exponential decay."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shape – large values at the edges, penalised centre."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_monotonic_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law (breaks symmetry)."""
    idx = np.arange(n)
    p = rng.uniform(0.5, 4.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    dec = (idx + 1 + offset) ** (-p)
    base = np.full(n, rng.uniform(0.0, 0.5))
    raw = mix * dec + (1.0 - mix) * base
    return raw


def _init_monotonic_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.03)
    mix = rng.uniform(0.0, 1.0)
    dec = np.exp(-lam * idx)
    base = np.full(n, rng.uniform(0.0, 0.5))
    raw = mix * dec + (1.0 - mix) * base
    return raw


def _init_ramp(n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear ramp – large at the left, decreasing to (almost) zero."""
    max_h = rng.uniform(0.5, 6.0)
    raw = np.linspace(max_h, 0.0, n)
    base = np.full(n, rng.uniform(0.0, 0.3))
    mix = rng.uniform(0.0, 1.0)
    raw = mix * raw + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries."""
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    raw = rng.random(n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence,
    otherwise draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)   # required ℓ₁‑norm

    # Warm start probability ≈ 0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # Choose a family
    families = [
        _init_powerlaw,
        _init_exponential,
        _init_ushape,
        _init_monotonic_powerlaw,
        _init_monotonic_exponential,
        _init_ramp,
        _init_gamma,
        _init_uniform,
        _init_dirichlet,
    ]
    init_fn = rng.choice(families)
    raw = init_fn(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]
