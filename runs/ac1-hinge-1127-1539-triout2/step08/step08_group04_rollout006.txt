"""

Algorithm
---------
1. Choose a problem size n (≈12000 for a full 1000 s budget).
2. Warm‑start the sequence:
   * If a previous sequence is stored in the global variable
     ``height_sequence_1`` it is interpolated to length n.

3. Adam‑based mirror descent on the *soft‑max* surrogate of the maximal
   convolution entry.
   * Work in log‑space (x_i = log a_i) to keep positivity.
   * The soft‑max temperature β is increased slowly (β←β·1.02) up to a large
     value (≈2⁶), so the optimisation first sees a smooth landscape and
     later focuses on the true max.
   * After each update the vector is renormalised to the exact ℓ¹‑norm
     Σa = √(2n) (the benchmark score then equals max(a*a)).
   * The best vector seen so far is recorded.
4. Deterministic mass‑transfer refinement:
   * Repeatedly move a small amount of mass from an index that contributes
     most to the current maximal convolution entry to an index that
     contributes the least, accepting only improvements.
   * This step uses exact convolutions (FFT) and runs for the remaining
     time budget.
5. Trim trailing near‑zero entries, store the result in the global

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (convolution a*a)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------

    Returns (grad, conv) where
        conv = a*a  (full linear convolution, length 2·n‑1)
        grad = ∂/∂a  softmax_β(conv)

    The soft‑max approximation is
        S_β(a) = log Σ_k exp(β·conv_k)  (up to an additive constant),
    whose gradient equals 2·(reverse(a) ⊛ p) with
        p_k = exp(β·(conv_k‑max(conv))) / Σ_j exp(β·(conv_j‑max(conv))).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # a*a via FFT
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_c = np.max(conv)
    # β·(conv‑max) ≤ 0, safe from overflow
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)

    # gradient = 2·reverse(a) ⊛ p   (only the central n entries are needed)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # central n components
    return grad, conv


# ----------------------------------------------------------------------
# Simple sequence generators / utilities
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _grid_param_search(small_n: int, target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Cheap two‑parameter grid (p, λ) on a reduced problem."""
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq *= target_sum / seq.sum()
            score = _max_convolution_fft(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    # random refinements
    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq *= target_sum / seq.sum()
        score = _max_convolution_fft(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam‑based mirror descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on log a with a slowly increasing soft‑max temperature.
    Returns the best sequence (exact max‑convolution value) found before
    ``deadline``.
    """
    a = a0.copy()
    n = a.size
    # work in log‑space; add small constant to avoid log(0)
    log_a = np.log(a + 1e-30)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                     # base learning rate for log‑space
    t = 0

    # soft‑max temperature schedule
    beta = 1.0
    max_beta = 65536.0
    beta_mul = 1.02

    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    while time.time() < deadline:
        t += 1

        # gradient of soft‑max surrogate and exact convolution
        grad_a, conv = _gradient_softmax(a, beta)

        # update temperature
        if beta < max_beta:
            beta = min(beta * beta_mul, max_beta)

        # gradient w.r.t. log‑variables (chain rule)
        grad_log = grad_a * a

        # Adam biased first/second moments
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update (descent)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # keep log values in a reasonable range to avoid overflow/underflow
        log_a = np.clip(log_a, -40.0, 40.0)

        # back‑to a, re‑normalise exactly
        a = np.exp(log_a)
        cur_sum = a.sum()
        if cur_sum <= 0.0:
            # fallback to uniform if something went terribly wrong
            a = np.full_like(a, target_sum / n)
            cur_sum = target_sum
        a *= target_sum / cur_sum

        # exact maximal convolution entry (objective value)
        cur_score = float(np.max(conv))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={t:5d} β={beta:6.1f} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (exact max‑convolution)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly shift a modest fraction of mass from indices that
    contribute most to the current maximal autocorrelation entry
    to indices that contribute least.  Accepts only strict improvements.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    rng = np.random.default_rng()
    max_frac = 0.03            # maximal fraction of donor's mass per move

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # contributions w_i = a_i + a_{k_max-i} (where defined)
        w = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w[idx] = a[idx] + a[k_max - idx]

        donor = int(np.argmax(w))
        receiver = int(np.argmin(w))

        if donor == receiver:
            break

        # random fraction of donor's mass, limited by max_frac and overall scale
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        new_score = _max_convolution_fft(a)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            # occasional jitter to escape plateaus
            if rng.random() < 0.001:
                jitter = rng.normal(scale=target_sum * 5e-4, size=n)
                a = np.maximum(a + jitter, 0.0)
                a *= target_sum / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n = more expressive but a bit slower)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial sequence (warm‑start)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        init *= target_sum / init.sum()
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – very cheap parametric search on a down‑scaled problem
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(12.0, (deadline - time.time()) * 0.07)
    small_seq, _ = _grid_param_search(small_n, small_target, param_deadline)
    if small_seq is not None:
        init = _resize_sequence(small_seq, n)
        init *= target_sum / init.sum()
        print(f"[{time.time():.2f}] Warm‑start from parametric search (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – Adam based mirror descent (major share of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        gd_deadline = time.time() + remaining * 0.78   # ~78 % of what is left
        best_seq, best_score = _adam_gradient_descent(init,
                                                      target_sum,
                                                      gd_deadline)
    else:
        best_seq = init
        best_score = _max_convolution_fft(best_seq)

    print(f"[{time.time():.2f}] After AdamGD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – deterministic mass‑transfer refinement (rest of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _deterministic_mass_transfer(best_seq,
                                                            target_sum,
                                                            deadline)
        print(f"[{time.time():.2f}] After transfer refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: trim trailing zeros, store for future warm‑starts
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final evaluation (may raise if evaluate_sequence is not available)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Keep the result for possible warm‑starts in a later call
    globals()["height_sequence_1"] = final_seq

    return final_seq
