def _subgrad_optimize(a0, deadline, rng, lr0=0.5, lr_decay=0.999):
    n = len(a0)
    S = sqrt(2*n)
    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)
    t = 0
    while time.time() < deadline:
        t += 1
        # compute conv and gradient
        conv = _full_convolution_fft(a)
        max_b = conv.max()
        # record best
        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()
        # identify ks naive using np.where:
        ks = np.flatnonzero(conv >= max_b - 1e-12) # might catch near ties
        # compute subgradient 
        g = np.zeros_like(a)
        for k in ks:
            idx = np.arange(n)
            valid = (k - idx >= 0) & (k - idx < n)
            g[valid] += 2 * a[k - idx[valid]]
        # average over ks: divide by len(ks)
        if len(ks) > 1:
            g /= len(ks)
        # update step size
        lr = lr0 * (lr_decay ** t)
        # step
        a_new = a - lr * g
        # project onto simplex
        a = _project_simplex(a_new, S)
        # optional jitter occasionally
        if t % 500 == 0:
            # add random small noise to avoid stagnation
            noise = rng.normal(scale=0.01 * S / n, size=n)
            a = _project_simplex(a + noise, S)
    return best_a, best_peak

Note: The projection onto simplex algorithm: we can implement:

def _project_simplex(v, z):
    # v: array; z: sum to enforce >0.
    # algorithm: sort v+descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _subgrad_optimize(a0, deadline, rng, lr0=0.3):
    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)
    t = 0
    while time.time() < deadline:
        t += 1
        # conv
        conv = _full_convolution_fft(a)
        max_b = conv.max()
        # update best
        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()
        # ks
        # Using tolerance relative maybe 1e-12 relative; we can use np.isclose
        ks = np.where(conv >= max_b - 1e-12)[0]  # salvage ties
        # gradient accumulate
        n = len(a)
        g = np.zeros_like(a)
        for k in ks:
            idx = np.arange(n)
            valid = (k - idx >= 0) & (k - idx < n)
            g[valid] += 2.0 * a[k - idx[valid]]
        if len(ks) > 1:
            g /= len(ks)
        # step size
        lr = lr0 / np.sqrt(t)
        # step
        a = a - lr * g
        # project simple
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 0.0)
            a[:] = (np.sqrt(2.0 * n) / n)  # uniform
            total = a.sum()
        a = a * (np.sqrt(2.0 * n) / total)
        # occasional jitter
        if t % 500 == 0:
            noise = rng.normal(scale=0.005 * np.sqrt(2.0 * n) / n, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = a * (np.sqrt(2.0 * n) / a.sum())
    return best_a, best_peak

"""
Propose candidate step‑function using:
1️⃣  Random exploration of several parametric families.
2️⃣  Exact sub‑gradient descent on the ℓ₁‑simplex (projected by simple
    clipping+renormalisation) to directly minimise the peak of the
    self‑convolution.
3️⃣  A cheap mass‑transfer simulated‑annealing fine‑tuner for the last
    seconds.

The optimiser works with the normalisation ∑a = √(2 n);  under this
normalisation the external score equals the maximal convolution value,
so we only need to minimise that quantity.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # background proportion
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.05)           # larger range than before
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))    # low background
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best, otherwise draws
    from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ~0.25
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential', 'sparse'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    else:  # sparse bumps
        raw = _init_sparse_bumps(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      lr0: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient minimisation of the maximal self‑convolution.
    The simplex constraint (non‑negative, ℓ₁‑norm = √(2 n)) is enforced
    by clipping and renormalising after each step.
    Returns the best vector seen and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution and peak
        conv = _full_convolution_fft(a)
        max_b = conv.max()

        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

        # indices attaining the maximum (allow a tiny tolerance)
        ks = np.where(conv >= max_b - 1e-12)[0]

        # exact sub‑gradient (average over ties)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in ks:
            valid = (k - idx >= 0) & (k - idx < n)
            g[valid] += 2.0 * a[k - idx[valid]]
        if len(ks) > 1:
            g /= len(ks)

        # diminishing step size
        lr = lr0 / np.sqrt(it)

        # gradient step
        a = a - lr * g

        # simple projection: clip and renormalise
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            # fallback to uniform if everything vanished
            a = np.full(n, S / n)
        else:
            a = a * (S / total)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=0.005 * S / n, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = a * (S / a.sum())

    return best_a, best_peak


# ----------------------------------------------------------------------
# Cheap simulated‑annealing fine‑tuner (mass‑transfer walk)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing on the simplex."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score
