def build_conv_constraint_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i, fi in enumerate(f):
        if fi == 0:
            continue
        row_start = i
        row_end = i + n
        # For each j, row index = i + j, column = j, coefficient = fi
        rows.extend(range(row_start, row_end))
        cols.extend(range(n))
        data.extend([fi] * n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A
```

"""
propose_candidate – search for a non‑negative step function with a small
self‑convolution supremum.

Algorithm
---------
1. Initialise a length‑2000 sequence (either the historic best or a random
   parametric family) and project it onto the ℓ¹‑simplex
   Σa_i = √(2 n).

2. A short Adam optimisation (smooth max of the convolution) improves the
   shape quickly.

3. Main refinement loop (LP‑mix):
   * With the current sequence f compute M = max(f⋆f).
   * Solve a linear programme
         maximise   Σ g_i
         subject to  f⋆g ≤ M   (2 n‑1 linear constraints)
                     g_i ≥ 0 .
     The LP finds a “candidate’’ g that can carry as much ℓ¹‑mass as
     possible without breaking the current maximal convolution value.
   * Normalise g to the required ℓ¹‑sum and evaluate its own max‑convolution.
     If it improves the best value we replace f by this normalised g.
   * Otherwise we gently mix the current f with the normalised g
     (mixing factor t) and project back onto the simplex.
   * The loop repeats while time permits, adapting the mixing factor.

4. A final hill‑climbing (simulated‑annealing) phase performs exact O(n)
   updates on the convolution matrix to polish any remaining irregularities.

All steps respect the global time budget (default 1000 s).  The function
returns the best sequence found as a plain Python list; its evaluation
equals the maximal entry of its self‑convolution after ℓ¹‑normalisation.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log Σ exp(β·b_k)  where
    b = a⋆a (full convolution).  Returns (grad, b) with:

        grad_i = 2·∑_k p_k·a_{k-i}   (p ∝ exp(β·b_k))
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a ⋆ p (the gradient factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 1e5,
                   beta_factor: float = 1.3) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max objective.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        a = _project_to_simplex(a, target_sum)

        if step % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if beta < beta_target and step % 250 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Random initialisers – a handful of simple parametric families
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    r = rng.uniform(0.96, 0.9995)
    idx = np.arange(n, dtype=float)
    return r ** idx


def _family_power_law(n: int, rng) -> np.ndarray:
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Exact hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9
    cooling = 0.998

    while time.time() < deadline:
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (exact O(n) slices)
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    b_best = _full_convolution(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Linear‑programming helper (mixing step)
# ----------------------------------------------------------------------


def _build_constraint_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Construct the sparse matrix A such that (A @ g) == f ⋆ g.
    Shape is (2*n-1, n).
    """
    n = f.size
    nz = np.where(f > 1e-14)[0]
    if nz.size == 0:
        # degenerate – return a zero matrix
        return scipy.sparse.csr_matrix((2 * n - 1, n), dtype=float)

    # Repeat each non‑zero index n times (once for every column of g)
    rows = np.repeat(nz, n) + np.tile(np.arange(n), nz.size)
    cols = np.tile(np.arange(n), nz.size)
    data = np.repeat(f[nz], n)

    return scipy.sparse.csr_matrix((data, (rows, cols)),
                                   shape=(2 * n - 1, n), dtype=float)


def _lp_maximum_mass(f: np.ndarray, M: float,
                    deadline: float) -> np.ndarray | None:
    """
    Solve
        maximise   Σ g_i
        subject to  f ⋆ g ≤ M
                    g_i ≥ 0 .
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f.size
    A = _build_constraint_matrix(f)
    b_vec = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)          # minimise –Σ g_i
    bounds = [(0, None) for _ in range(n)]

    # time check before solving
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return None

    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_vec,
                                    bounds=bounds, method='highs')
    except Exception as e:
        print("LP exception:", e)
        return None

    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        # LP was infeasible or unbounded; just skip this iteration
        print("LP not successful:", res.message)
        return None
