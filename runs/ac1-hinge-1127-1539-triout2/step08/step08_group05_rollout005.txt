propose_candidate – improved multi‑stage search for a minimal

The algorithm works in several phases:

1. **Diverse analytic families** – Randomly generate step functions
   from a large collection of parametric shapes (beta‑like, exponential,

6. **Upsampling & refinement** – The best solution is up‑sampled
   (≈+30 %) to a larger length (capped at 30 000) and a short Adam
   refinement is performed.

All steps respect the constraints of `evaluate_sequence`:

maximum convolution value).  The routine is single‑threaded, uses only

# Simplex projection – Euclidean projection onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the scaled probability simplex."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    The weight vector w must be a probability distribution over the
    (2n‑1) convolution entries.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices picking a*a terms
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1‑i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families for the initial sampling stage
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int,
                   base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a diverse set of families."""
    fam = rng.integers(0, 12)
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0

    if fam == 0 and base_seq is not None:
        # up‑sample an existing good solution
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # plain uniform random
    elif fam == 2:
        # symmetric exponential decay from the centre
        lam = rng.uniform(0.001, 0.02) * n
        a = np.exp(-lam * np.abs(i - center))
    elif fam == 3:
        # beta‑like shape (i+1)^{α‑1} (n‑i)^{β‑1}
        alpha = rng.uniform(0.1, 5.0)
        beta = rng.uniform(0.1, 5.0)
        a = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
    elif fam == 4:
        # symmetric power‑law (i·(n‑1‑i))^p
        p = rng.uniform(0.2, 4.0)
        a = (i * (n - 1 - i) + 1) ** p
    elif fam == 5:
        # sinusoidal power
        p = rng.uniform(0.5, 5.0)
        a = np.sin(np.pi * (i + 0.5) / n) ** p
    elif fam == 6:
        # Gaussian centred at the middle
        sigma = rng.uniform(0.05, 0.4) * n
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    elif fam == 7:
        # cosine‑power shape, peaked in the centre
        p = rng.uniform(0.5, 6.0)
        a = (np.cos(np.pi * (i - center) / n) + 1) ** p
    elif fam == 8:
        # logistic S‑curve (fast rise, slow decay)
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (i - mid) / n))
    elif fam == 9:
        # piecewise linear “tent” with random apex location
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])
    elif fam == 10:
        # mixed beta + exponential (product)
        alpha = rng.uniform(0.2, 4.0)
        beta = rng.uniform(0.2, 4.0)
        lam = rng.uniform(0.001, 0.015) * n
        a = ((i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)) * np.exp(-lam * np.abs(i - center))
    else:
        # random convex combination of two simple families
        w = rng.random()
        a1 = rng.random(n) ** rng.uniform(0.5, 3.0)
        lam = rng.uniform(0.001, 0.02) * n
        a2 = np.exp(-lam * np.abs(i - center))
        a = w * a1 + (1 - w) * a2

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 2000.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the soft‑max approximation of the peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.03
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.025:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998   # very slow decay

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: each term a_i contributes a_{k-i}
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional aspirational random transfer
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early stop if no progress for a while
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Simulated‑annealing hill‑climb with pairwise mass swaps."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    cur_max = best_score

    start = time.time()
    T0 = max(0.1 * cur_max, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.07 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Load previous best if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0
    TOP_K = 5
    top_candidates = []   # list of (score, seq)

    # ------------------------------------------------------------------
    # Phase 1 – rich family sampling (≈20 % of total time)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # Choose length: favour larger sizes but stay ≤30 000
        if base_seq is not None and rng.random() < 0.45:
            scale = rng.uniform(0.9, 1.1)
            n = max(2000, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(3000, 25001))

        a = _sample_family(rng, n, base_seq)

        # Normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        # Randomly enforce symmetry (helps in later stages)
        if rng.random() < 0.4:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

        # Keep a short list of top‑K candidates for later refinement
        if len(top_candidates) < TOP_K:
            top_candidates.append((cur_score, a.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if cur_score < top_candidates[-1][0]:
                top_candidates[-1] = (cur_score, a.copy())
                top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Phase 1b – jitter around the current best (≈5 % of total time)
    # ------------------------------------------------------------------
    jitter_deadline = time.time() + 0.05 * budget_s
    while time.time() < jitter_deadline and best_overall_seq is not None:
        attempts += 1
        a = best_overall_seq.copy()
        n = a.shape[0]
        # perform a few random mass‑transfer moves
        for _ in range(3):
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.08 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
        a = _project_simplex(a, np.sqrt(2.0 * n))
        if rng.random() < 0.5:
            a = _symmetrize(a)
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
            print(f"[jitter] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam optimisation (≈45 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if best_overall_seq is None:
        # fallback: uniform of medium size
        n = 5000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_overall_seq = a
        best_overall_score = np.max(_full_convolution(a))

    adam_deadline = now + 0.45 * remaining
    # try both symmetric and non‑symmetric runs, keep the better
    for enforce_sym in (True, False):
        a_adam, sc_adam = _optimize_adam(best_overall_seq,
                                         adam_deadline,
                                         rng,
                                         max_alpha=2000.0,
                                         enforce_sym=enforce_sym)
        if sc_adam < best_overall_score:
            best_overall_score = sc_adam
            best_overall_seq = a_adam.copy()
            print(f"[adam] sym={enforce_sym} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent (≈20 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        subgrad_deadline = now + 0.20 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_overall_seq,
                                          subgrad_deadline,
                                          rng,
                                          enforce_sym=False)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing hill‑climb (≈10 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        anneal_deadline = now + 0.10 * (deadline - now)
        a_anneal, sc_anneal = _anneal_hillclimb(best_overall_seq,
                                                anneal_deadline,
                                                rng,
                                                enforce_sym=False)
        if sc_anneal < best_overall_score:
            best_overall_score = sc_anneal
            best_overall_seq = a_anneal.copy()
            print(f"[anneal] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Upsample & final Adam refinement (≈10 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 8.0:   # keep a few seconds for final output
        cur_n = best_overall_seq.shape[0]
        new_n = min(30000, int(cur_n * 1.30))
        a_big = _upsample_seq(best_overall_seq, new_n)

        # re‑normalise for the larger length
        S_big = np.sqrt(2.0 * new_n)
        a_big = a_big / a_big.sum() * S_big

        # short Adam refinement (symmetric)
        refine_deadline = now + 0.10 * (deadline - now)
        a_ref, sc_ref = _optimize_adam(a_big,
                                       refine_deadline,
                                       rng,
                                       max_alpha=2500.0,
                                       enforce_sym=True)
        if sc_ref < best_overall_score:
            best_overall_score = sc_ref
            best_overall_seq = a_ref.copy()
            print(f"[upsample‑adam] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    # store for future runs
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
