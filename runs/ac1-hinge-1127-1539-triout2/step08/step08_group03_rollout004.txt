def _subgradient_max_convolution(a):
    b = _full_convolution_fft(a)
    max_b = b.max()
    # mask as float
    mask = (b == max_b).astype(float)
    count = mask.sum()
    if count > 0:
        mask /= count
    # convolution mask * a_rev
    # compute via FFT as before
    a_rev = a[::-1]
    conv_len = len(mask) + len(a) - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[len(a)-1:len(a)-1+len(a)]
    return g, max_b

"""
Adversarial step‑function optimizer – fourth generation.

Goal
----
Search for a non‑negative sequence `a` (length n) minimising
    evaluate_sequence(a) = 2·n·max(a*a) / (sum(a)^2).

The objective is invariant under global scaling, so we enforce the
ℓ₁‑normalisation  sum(a) = √(2 n).  Under this normalisation the score
equals the peak of the self‑convolution, i.e.  max(a*a).

Strategy
--------
1. **Broad random exploration** – draw many candidates from a richer
   collection of parametric families (including double power‑law and
   Gaussian bumps).  Keep the best `top_k` for later refinement.

2. **Multi‑scale Adam refinement** – for each retained candidate
   interpolate it to a schedule of increasing lengths and run a
   projected‑Adam optimiser on the exact max‑convolution.  The Adam
   optimiser works on a log‑softmax parametrisation (automatically
   respecting non‑negativity and the ℓ₁‑constraint).  The “τ‑soft‑max”
   approximation of the max is annealed very slowly, and once τ
   becomes tiny we switch to the exact sub‑gradient (mask of the
   maximal convolution entries).

3. **Simulated‑annealing fine‑tuning** – a cheap mass‑transfer walk on
   the simplex helps escape shallow local minima.

4. **Final long Adam run** – if time permits, a last Adam optimisation
   on the best sequence (with a very small τ) is performed.

All phases are guarded by strict time checks so the function always
returns a feasible sequence before the 1000 s budget expires.

The resulting sequence is stored in the global variable
`height_sequence_1` and also returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient (τ‑soft‑max) – used in the early Adam iterations
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
        # when several positions share the max, average them
        cnt = mask.sum()
        if cnt > 0:
            mask /= cnt
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Exact sub‑gradient (mask of the maximal convolution entries)
# ----------------------------------------------------------------------
def _exact_subgradient(a: np.ndarray):
    """
    Sub‑gradient of the exact max‑convolution (τ = 0).  Returns
    (grad, max_b) where grad is one element of the sub‑gradient set
    obtained by averaging over all maximal positions.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    mask = (b == max_b).astype(float)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    a_rev = a[::-1]
    conv_len = len(mask) + len(a) - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[len(a) - 1:len(a) - 1 + len(a)]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Minimise the exact max‑convolution of ``a`` under the ℓ₁‑norm
    constraint using Adam on a log‑softmax parametrisation.
    The τ‑soft‑max approximation is used while τ > 1e‑12;
    afterwards the exact sub‑gradient is employed.
    Returns the best vector seen and its *hard* peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution & (approx) gradient -------
        if tau > 1e-12:
            g, peak = _softmax_gradient(a, tau)
        else:
            g, peak = _exact_subgradient(a)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule -------
        mu = np.dot(a, g) / S                # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # how much background
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))    # low background
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_double_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Two‑segment decreasing power‑law:
        a[i] = (i+offset1)^(-p1)   for i < split
        a[i] = c * (i+offset2)^(-p2) for i ≥ split
    """
    idx = np.arange(n)
    split = rng.integers(n // 4, 3 * n // 4)
    p1 = rng.uniform(0.2, 2.5)
    p2 = rng.uniform(0.2, 2.5)
    offset1 = rng.uniform(0.0, 5.0)
    offset2 = rng.uniform(0.0, 5.0)
    c = rng.uniform(0.2, 1.5)
    left = (idx[:split] + offset1) ** (-p1)
    right = c * (idx[split:] + offset2) ** (-p2)
    raw = np.concatenate([left, right])
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """A (possibly asymmetric) Gaussian‑shaped bump."""
    idx = np.arange(n)
    mu = rng.uniform(0, n - 1)
    sigma = rng.uniform(n * 0.02, n * 0.4)
    raw = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    # add a tiny uniform floor to keep strict positivity after log‑softmax
    raw += rng.uniform(0.0, 0.05)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ~0.22 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'double_powerlaw', 'gaussian'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'double_powerlaw':
        raw = _init_double_powerlaw(n, rng)
    else:  # gaussian
        raw = _init_gaussian(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (very cheap, used only if time left)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score        # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        # transfer a random fraction of the mass at i to j
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # renormalise (keeps ℓ₁‑norm exactly S)
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score

def propose_candidate(seed: int = 42,

    """

    globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)

    end_time = start_time + budget_s - 3.0

    # Warm‑start from a previously found construction, if present
    # ------------------------------------------------------------------
    warm_seq = None

        try:

        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    explore_deadline = start_time + budget_s * 0.45

    attempt = 0

        # pick a length – favour medium‑size vectors but keep some very large ones

    final_seq = best_seq.tolist()

    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
