def propose_candidate(seed=0, budget_s=1000):
    """Search for a non-negative step function minimizing the max self-convolution.
    The algorithm combines random initialisation, projected Adam on a smooth
    max‐approximation, and a long simulated‑annealing hill‑climb.  It works
    with up‑sampling to larger lengths and spends most of the budget on a
    deterministic hill‑climb on the largest dimension."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose a target length that is a power of two for efficient FFT
    # Use 32768 (2^15) which fits comfortably in memory and gives many
    # degrees of freedom.

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – random sampling (≈5 % of the budget)
    # ------------------------------------------------------------------
    random_time = (budget_s * 0.05)
    rand_deadline = time.time() + random_time

    best_score = np.inf

    while time.time() < rand_deadline:

        a = np.maximum(a, 1e-12)

            best_a = a.copy()
            print(f"[{time.time():.1f}] Random init score {best_score:.6f}")

    if best_a is None:
        best_a = np.full(n, target_sum / n)
        best_score = _max_convolution_fft(best_a)

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (≈15 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        adam_deadline = time.time() + min(remaining * 0.15, 30.0)
        a_adam = _adam_gradient_descent(best_a, target_sum, adam_deadline)
        score_adam = _max_convolution_fft(a_adam)
        if score_adam < best_score:
            best_score = score_adam
            best_a = a_adam.copy()
            print(f"[{time.time():.1f}] Adam improvement score {best_score:.6f}")

    # Phase 2 – Long hill‑climbing (≈75 % of the budget)

"""
search for a non‑negative step function that minimises
evaluate_sequence (i.e. the maximum entry of the self‑convolution
when the ℓ¹‑norm is fixed to sqrt(2 n)).

The algorithm proceeds in three phases:

0️⃣  If a previous construction is available in the global variable
    `height_sequence_1`, it is up‑sampled to the target length and used
    as the first seed.

1️⃣  Random initialisation (≈5 % of the budget).  A handful of Dirichlet,
    power‑law and exponential families are sampled, projected onto the
    ℓ¹‑simplex Σa = √(2 n) and the best candidate (lowest max‑convolution)
    is kept.

2️⃣  A projected Adam optimiser works on a smooth log‑sum‑exp proxy of the
    max‑convolution.  The β‑parameter is annealed up to 1e⁶, providing a
    strong descent direction while respecting the simplex constraint.

3️⃣  The main workhorse – a simulated‑annealing style hill‑climber that
    updates the full self‑convolution in O(n) per move.  The move
    transfers a random amount of mass from a source index that contributes
    heavily to the current worst convolution entry to a destination that
    contributes little, with a Metropolis acceptance rule.  The temperature
    is slowly cooled, allowing the search to escape shallow local minima.

The bulk of the 1000 s wall‑clock budget is spent on phase 3 at a
relatively large dimension (n = 2¹⁵ = 32768).  After hill‑climbing a
short Adam polish is performed.  The final sequence is trimmed of
trailing near‑zeros and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Implementation follows Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the complete array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns a pair (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a ⋆ p (a reversed)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation of the smooth‑max surrogate.
    The learning‑rate is scaled by the simplex radius.
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.25 * target_sum

    beta = 2.0               # start smoothing
    beta_max = 1e6
    beta_factor = 1.5

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if t % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 5e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are mixed to provide diversity.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform plus jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length,
    followed by projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (biased version)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill‑climb that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The destination is biased towards indices that contribute little to the
    current worst convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9997                   # slow cooling

    while time.time() < deadline:
        # pick a worst convolution index
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index – weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination – bias towards small counterpart a[M‑j]
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]
        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s)
    and returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a working dimension.  A power of two makes the FFT fast.
    # --------------------------------------------------------------
    n = 32768                     # 2**15
    target_sum = math.sqrt(2.0 * n)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optional seeding from a previously stored construction.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            if prev.size != n:
                a0 = _upsample_sequence(prev, n)
            else:
                a0 = prev.copy()
            a0 = _project_to_simplex(a0, target_sum)
            score0 = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = score0
            print(f"[{time.time():.1f}] Seeded from height_sequence_1, score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # --------------------------------------------------------------
    # Phase 0 – random sampling (≈5 % of the budget)
    # --------------------------------------------------------------
    random_budget = min(60.0, (deadline - time.time()) * 0.05)
    rand_deadline = time.time() + random_budget

    while time.time() < rand_deadline:
        a = _random_initial_sequence(n, rng)
        a = np.maximum(a, 1e-12)                     # avoid exact zeros
        score = _max_convolution_fft(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Random candidate improvement: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – projected Adam on smooth max (≈15 % of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        adam_deadline = time.time() + min(remaining * 0.15, 40.0)
        a_adam = _adam_gradient_descent(best_seq, target_sum, adam_deadline)
        score_adam = _max_convolution_fft(a_adam)
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.1f}] Adam improvement: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – intensive hill‑climbing (≈80 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_hc, b_hc = _hill_climbing_improved(a_cur, b_cur, deadline - 2.0)
        best_score = b_hc.max()
        best_seq = a_hc
        print(f"[{time.time():.1f}] Hill‑climb final score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional final Adam polish (short, if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        final_deadline = deadline - 1.0
        a_final = _adam_gradient_descent(best_seq, target_sum, final_deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final
            print(f"[{time.time():.1f}] Final Adam polish improvement: {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # Report the official evaluation
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
