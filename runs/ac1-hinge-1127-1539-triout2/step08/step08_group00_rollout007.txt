"""
Improved search for a non‑negative step function that minimises the
maximum entry of its self‑convolution (the value returned by
``evaluate_sequence``).

The algorithm combines several ideas that were not present in the
reference implementation:

* a richer collection of parametric families (offset exponentials,
  double exponentials, stretched exponentials, Gaussian‑like bumps);
* a much larger random‑sampling phase per support size;
* a projected *sub‑gradient* descent that directly uses the true
  maximal convolution entry (instead of a soft‑max surrogate);
* a short Adam optimisation of the smooth‑max objective (as before)
  but with an aggressive learning‑rate schedule;
* an enhanced hill‑climbing move set (source selection weighted by
  contribution to the current peak, target bias towards low‑mass
  locations);
* a final polishing stage that alternates Adam → sub‑gradient → hill‑climb.

All phases respect the global wall‑clock budget (default 1000 s).  The
best sequence encountered is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a using FFT (returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a⋆a via FFT."""
    return float(_full_convolution_fft(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to length new_n, then simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _family_exponential(n: int, rng) -> np.ndarray:
    r = rng.uniform(0.96, 0.9995)
    return r ** np.arange(n, dtype=float)


def _family_power_law(n: int, rng) -> np.ndarray:
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _family_offset_exponential(n: int, rng) -> np.ndarray:
    offset = rng.integers(0, n // 2)
    r = rng.uniform(0.96, 0.9995)
    tail = r ** np.arange(n - offset, dtype=float)
    return np.concatenate([np.zeros(offset, dtype=float), tail])


def _family_double_exponential(n: int, rng) -> np.ndarray:
    r1 = rng.uniform(0.94, 0.999)
    r2 = rng.uniform(0.94, 0.999)
    w = rng.uniform(0.0, 1.0)
    idx = np.arange(n, dtype=float)
    return w * (r1 ** idx) + (1.0 - w) * (r2 ** idx)


def _family_stretched_exponential(n: int, rng) -> np.ndarray:
    tau = rng.uniform(0.2 * n, 1.2 * n)
    p = rng.uniform(0.3, 2.0)
    idx = np.arange(n, dtype=float)
    return np.exp(- (idx / tau) ** p)


def _family_gaussian_bump(n: int, rng) -> np.ndarray:
    mu = rng.uniform(0.0, n - 1.0)
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    idx = np.arange(n, dtype=float)
    return np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Pick a random parametric family, instantiate it and project onto the
    ℓ¹‑simplex of radius sqrt(2n).
    """
    f = rng.choice(_FAMILY_FUNCS)
    a = f(n, rng)
    # add a tiny jitter to avoid exact ties
    a += rng.normal(scale=0.001, size=n)

    a = _project_to_simplex(a, target)
    # Clip to the range enforced by evaluate_sequence (optional safety)
    a = np.clip(a, 0.0, 1000.0)
    return a

# Adam optimisation of the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of  (1/β)·log Σ exp(β·b_k)   where b = a⋆a.
    Returns (grad, b) with  grad_i = 2·∑_k p_k·a_{k-i}.
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (a_rev ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1      # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 1e5,
                   beta_factor: float = 1.3) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max surrogate.
    The schedule for β grows geometrically until ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.25
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if step % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β schedule
        if beta < beta_target and step % 250 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         init_lr: float = 0.02) -> np.ndarray:
    """
    Projected sub‑gradient descent that uses the *exact* maximal
    convolution entry (instead of a soft‑max).  The step size decays
    geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    lr = init_lr
    rng = np.random.default_rng()
    last_improve = time.time()

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        cur_max = b.max()
        # all indices attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # gradient of the (single) max entry
        g = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # vectorised computation of the gradient
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        # where i != j we add a_j, where i == j we add 2*a_i
        same = i_range == j_range
        g[i_range[~same]] = a[j_range[~same]]
        g[i_range[same]] = 2.0 * a[i_range[same]]

        # step and projection
        a -= lr * g
        a = _project_to_simplex(a, target_sum)

        # simple learning‑rate decay
        lr *= 0.9995

        # occasional reset of lr if no improvement for a while
        if cur_max < _max_convolution_fft(a):
            # we made it worse – restore a tiny bit of the previous step size
            lr = max(lr, init_lr * 0.1)

        # early exit check
        if time.time() - last_improve > 30.0:
            # no improvement for half a minute – stop early
            break

    return a


# ----------------------------------------------------------------------
# Hill‑climbing (simulated annealing) – same as reference but with a
# slightly larger temperature decay factor.
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.25 + 1e-9
    cooling = 0.992  # slower cooling → more thorough exploration

    while time.time() < deadline:
        # pick a location attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # choose a source index i that contributes to b_k (probability ∝ a_i a_{k-i})
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias towards low‑mass locations
        if rng.random() < 0.4:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        # removal from source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # addition to target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correction for the cross term i+j (appears twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb new best max={best_max:.6f}")

        T *= cooling

    # recompute exact convolution for the best candidate
    b_best = _full_convolution_fft(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects a wall‑clock budget and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional warm‑start from a historic construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            score0 = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = score0
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) score={score0:.6f}")
        except Exception as e:
            print(f"Failed to seed from height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Length schedule – start small, finish with a larger support
    # ------------------------------------------------------------------
    schedule = [500, 1000, 2000, 4000, 8000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1) Random parametric sampling (much larger than before)
        # --------------------------------------------------------------
        n_samples = 80                     # a generous budget for random families
        for _ in range(n_samples):
            if time.time() >= deadline:
                break
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] random family improvement (n={n}) score={score:.6f}")

        # --------------------------------------------------------------
        # 2) Adam descent on the smooth‑max surrogate
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        gd_time = min(remaining * 0.35, 200.0)   # allocate a fraction of the budget
        gd_deadline = time.time() + gd_time

        # initialise from the current best (upsampled if needed)
        if best_seq is not None and best_seq.size != n:
            a_start = _upsample_sequence(best_seq, n)
        else:
            a_start = _random_initial_sequence(n, rng)

        a_adam = _adam_optimize(a_start, gd_deadline,
                                beta_start=4.0,
                                beta_target=2e5,
                                beta_factor=1.2)
        score_adam = _max_convolution_fft(a_adam)
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.1f}] Adam improvement (n={n}) score={score_adam:.6f}")

        # --------------------------------------------------------------
        # 3) Sub‑gradient descent on the true max entry
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 4.0:
            break
        sg_time = min(remaining * 0.25, 150.0)
        sg_deadline = time.time() + sg_time

        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()

        a_sg = _subgradient_descent(a_cur, sg_deadline, init_lr=0.02)
        score_sg = _max_convolution_fft(a_sg)
        if score_sg < best_score:
            best_score = score_sg
            best_seq = a_sg.copy()
            print(f"[{time.time():.1f}] Sub‑gradient improvement (n={n}) score={score_sg:.6f}")

        # --------------------------------------------------------------
        # 4) Exact hill‑climbing (simulated annealing)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 4.0:
            break
        hc_time = min(remaining * 0.30, 200.0)
        hc_deadline = time.time() + hc_time

        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution_fft(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = float(b_hc.max())
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – use any leftover budget for a last Adam → SG → HC pass
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final Adam optimisation
        a_final = _adam_optimize(best_seq.copy(),
                                 deadline - 3.0,
                                 beta_start=8.0,
                                 beta_target=3e5,
                                 beta_factor=1.15)

        # short sub‑gradient refinement
        a_final = _subgradient_descent(a_final, deadline - 2.0, init_lr=0.015)

        # final hill‑climb
        b_final = _full_convolution_fft(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 1.0)

        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy return value
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: simple exponential with modest length
        n = 2000
        best_seq = _project_to_simplex(_family_exponential(n, rng), np.sqrt(2.0 * n))

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (max‑convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
