"""Search for step function minimizing evaluate_sequence.

Implements a multi‑stage optimisation:
1. Random parametrised start.
2. Projected Adam on a smoothed max (log‑sum‑exp) objective.
3. Optional exact hill‑climbing (simulated annealing) on moderate sizes.
The algorithm concentrates most of the time budget on large support sizes,
where the self‑convolution can be made almost flat and the objective
approaches its theoretical lower bound ≈1.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    return float(_full_convolution(a).max())

def _softmax_gradient(a: np.ndarray, beta: float):
    """Gradient of (1/β)·log Σ exp(β·(a⋆a)). Returns (grad, b)."""
    n = a.shape[0]

    # convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2·a ⋆ p (restricted to length n)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 5e5,
                   beta_factor: float = 1.15,
                   lr_scale: float = 0.2) -> np.ndarray:
    """Projected Adam optimisation of the smoothed‑max of a⋆a."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * lr_scale
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the smoothed max
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter (helps escape plateaux)
        if step % 500 == 0:
            jitter = rng.normal(scale=target_sum * 3e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (geometric growth)
        if beta < beta_target and step % 200 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a

def _family_exponential(n: int, rng) -> np.ndarray:
    r = rng.uniform(0.96, 0.9995)
    idx = np.arange(n, dtype=float)
    return r ** idx

def _family_power_law(n: int, rng) -> np.ndarray:
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)

def _family_linear(n: int, rng) -> np.ndarray:
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)

def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])

def _family_cosine_bell(n: int, rng) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / (n - 1)))
    return a

def _family_gaussian(n: int, rng) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.1, 0.3) * n
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp,
                _family_cosine_bell,
                _family_gaussian]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)   # tiny jitter
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a

def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to new length followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """Simulated‑annealing hill climbing (O(n) per accepted move)."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9
    cooling = 0.998

    while time.time() < deadline:
        # pick a position attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – proportional to contribution to this max entry
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – biased towards small‑mass positions
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=prob_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to transfer
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- exact O(n) convolution update -----
        # Remove contribution from source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # Add contribution from target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # Correct the cross term at i+j
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # Final accurate convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best

"""Search for a step function minimising ``evaluate_sequence``.

The algorithm proceeds in stages:

1. **Initialisation** – draw a candidate from several parametric families
   (exponential, power‑law, linear, piece‑wise exponential, cosine bell,
   Gaussian) and project it onto the ℓ¹‑simplex {x_i ≥ 0, Σx_i = √(2 n)}.

2. **Projected Adam optimisation** – minimise a smooth approximation of the
   maximal entry of the self‑convolution, namely
   (1/β)·log ∑ exp(β·(a ⋆ a)).  The schedule for β grows geometrically up to
   ≈10⁶ while the iterates are projected back onto the simplex after each
   update.

3. **Exact hill‑climbing (simulated annealing)** – for moderate support
   sizes (n ≤ 20 000) we perform an O(n)‑per‑move local search that transfers
   a random amount of mass between two indices and updates the whole
   convolution exactly.  Moves that lower the current maximum are always kept;
   otherwise they are accepted with probability exp(−Δ/T) and the temperature
   T is slowly cooled.

The search concentrates most of the time budget on large support sizes
(≥10⁴), where the self‑convolution can be made almost flat and the objective
approaches its theoretical lower bound ≈1. The best sequence found is trimmed
of trailing near‑zeros and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log ∑ exp(β·b_k)  where
    b = a⋆a (full convolution).  Returns (grad, b) with

        grad_i = 2·∑_k p_k·a_{k-i}   (where p_k ∝ exp(β·b_k))
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities (stable)
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev ⋆ p  (gradient factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 5e5,
                   beta_factor: float = 1.15,
                   lr_scale: float = 0.2) -> np.ndarray:
    """
    Projected Adam optimiser for the smoothed‑max objective.
    ``beta`` grows geometrically from ``beta_start`` to ``beta_target``.
    ``lr_scale`` controls the initial learning‑rate magnitude.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * lr_scale
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the smoothed max
        g, _ = _softmax_gradient(a, beta)

        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaux
        if step % 500 == 0:
            jitter = rng.normal(scale=target_sum * 3e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (geometric growth)
        if beta < beta_target and step % 200 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Random families for initialisation
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)"""
    r = rng.uniform(0.96, 0.9995)
    return r ** np.arange(n, dtype=float)


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}"""
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a₀ to 0."""
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential decay."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _family_cosine_bell(n: int, rng) -> np.ndarray:
    """Cosine‑shaped bump (non‑negative)."""
    i = np.arange(n, dtype=float)
    return 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / (n - 1)))


def _family_gaussian(n: int, rng) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.1, 0.3) * n
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a family at random, add tiny jitter, and project onto the simplex."""
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp,
                _family_cosine_bell,
                _family_gaussian]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Exact hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a “source” index to a “target”
    index while keeping the full convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9
    cooling = 0.998

    while time.time() < deadline:
        # pick one of the positions attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – weighted by its contribution to the max entry
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias towards positions with little current mass
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----
        # remove source contribution
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # add target contribution
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correct the cross term i+j (appears twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # final accurate convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best
