def project_simplex(v, S):
    # Based on algorithm of Duchi et al. (2008)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

   {a≥0, Σa = √(2 n)}.  The loss is a convex combination of a
   *smoothed max* gradient (τ‑soft‑max) and a *variance* gradient that
   flattens the whole convolution profile.  The τ‑temperature and the
   learning rate decay slowly.

3. **Length scaling & further refinement** – the current best sequence
   is linearly interpolated to several larger lengths (up to ≈25 000)
   and the same Adam routine is applied again.  Longer sequences tend
   to lower the peak because the convolution can be spread more evenly.

4. **Stochastic hill‑climbing** – a cheap local search that repeatedly
   moves a small fraction of mass from one index to another, keeping
   any improvement.  The final sequence is stored in the global variable
   ``height_sequence_1`` and returned as a plain Python list.

All operations respect the 1000 s time budget; the code leaves a small
margin for the final ``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, scaling, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convoluted with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = S}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if S < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_hyperbolic(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edges heavy: 1/(i+1)^α + 1/(n‑i)^α."""
    i = np.arange(n)
    α = rng.uniform(0.3, 2.2)
    raw = (1.0 / (i + 1) ** α) + (1.0 / (n - i) ** α)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Frequently warm‑starts from the current global best, otherwise draws
    from a random family and projects onto the simplex Σa = √(2 n).
    """
    S = np.sqrt(2.0 * n)

    # occasional warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'hyperbolic', 'ushape', 'uniform',
                        'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'hyperbolic':
        raw = _init_hyperbolic(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (combined max‑soft‑max + variance)
# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser directly on the heights (not in log‑space).
    The loss is a convex combination of a τ‑soft‑max of the convolution
    (focus on the current peak) and a variance term that flattens the
    whole convolution profile.
    Returns the best sequence encountered and its hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # start from a feasible point
    a = a0.copy()
    a = _project_onto_simplex(a, S)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                               # relatively aggressive start

    # temperature for the soft‑max (start mildly smoothed)
    b0 = _full_convolution_fft(a)
    tau = max(b0.max(), 1.0)

    best_a = a.copy()
    best_peak = b0.max()
    it = 0
    ramp = 2000.0                           # iterations after which we focus completely on the max

    while time.time() < deadline:
        it += 1

        # ----- gradients -------------------------------------------------
        # softened max gradient
        g_max, peak = _softmax_gradient(a, tau)

        # variance‑gradient (flattens the whole convolution)
        b = _full_convolution_fft(a)
        mu = b.mean()
        diff = b - mu
        a_rev = a[::-1]
        conv_len = len(diff) + n - 1
        f_len = _next_pow2(conv_len)
        Fdiff = np.fft.rfft(diff, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fdiff * Fa_rev, n=f_len)[:conv_len]
        g_var = 2.0 * conv[n - 1:n - 1 + n]

        # schedule: early iterations emphasise variance, later ones max
        alpha = min(1.0, it / ramp)
        g = (1.0 - alpha) * g_var + alpha * g_max

        # ----- best‑so‑far tracking ------------------------------------
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- Adam update on the raw vector ---------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_onto_simplex(a, S)       # enforce feasibility

        # ----- schedules ------------------------------------------------
        tau *= 0.9998                         # keep a touch of smoothing
        lr *= 0.9999                          # gentle decay

        # occasional jitter to escape tiny plateaus
        if it % 800 == 0:
            a += rng.normal(scale=0.006 * S, size=n)
            a = _project_onto_simplex(a, S)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search combines random exploration, projected Adam refinement,
    length scaling and a final stochastic hill‑climb.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0    # safety margin for the final evaluator

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    K = 8                                      # size of the elite pool
    elite = []                                 # list of (score, seq)

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 55 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0
    while time.time() < explore_deadline:
        # length bias towards a few thousand entries
        if rng.random() < 0.7:
            n = int(rng.integers(1800, 6200))
        else:
            n = int(rng.integers(6200, 19000))

        a0 = _initial_vector(n, rng, warm_seq)
        score = _max_convolution(a0)

        if len(elite) < K:
            elite.append((score, a0.copy()))
            elite.sort(key=lambda x: x[0])
        elif score < elite[-1][0]:
            elite[-1] = (score, a0.copy())
            elite.sort(key=lambda x: x[0])
        # occasional progress output
        if rng.random() < 0.0005:
            print(f"[explore] n={n:5d}  current elite best={elite[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refinement of elite candidates (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 180.0   # keep room for scaling & hill‑climb
    if remaining <= 0:
        remaining = 0.0
    per_candidate = max(5.0, remaining / max(1, len(elite)))
    refined = []

    for score, seq in elite:
        deadline = time.time() + per_candidate
        if time.time() >= deadline:
            break
        a_opt, new_score = _refine_sequence(seq, deadline, rng)
        refined.append((new_score, a_opt))
        if new_score < score:
            print(f"[refine] length={len(seq):5d}  improved {score:.6f} → {new_score:.6f}")

    # keep the best refined sequence
    if refined:
        refined.sort(key=lambda x: x[0])
        best_score, best_seq = refined[0]
        best_n = best_seq.shape[0]
    else:
        # fallback to the best from the elite pool
        best_score, best_seq = elite[0]
        best_n = best_seq.shape[0]

    # ------------------------------------------------------------------
    # Phase 3 – scaling to larger lengths and further Adam runs
    # ------------------------------------------------------------------
    scaling_factors = [1.0, 1.5, 2.0, 2.5, 3.0]
    max_len = 25000
    for factor in scaling_factors:
        if time.time() >= end_time - 120.0:
            break
        target_n = int(best_n * factor)
        if target_n < 2000 or target_n > max_len:
            continue

        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        deadline = time.time() + min(60.0, (end_time - time.time()) * 0.15)
        a_opt, new_score = _refine_sequence(a_scaled, deadline, rng)

        if new_score < best_score:
            print(f"[scale‑refine] n={target_n:5d}  new best {best_score:.6f} → {new_score:.6f}")
            best_score = new_score
            best_seq = a_opt.copy()
            best_n = target_n

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (remaining time)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.14)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount

        new_peak = _max_convolution(a)
        if new_peak + 1e-9 < best_score:   # strict improvement
            best_score = new_peak
            best_seq = a.copy()
            print(f"[local] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final evaluation via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
