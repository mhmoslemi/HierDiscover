def _simulated_annealing(a, deadline, rng):
    n = a.shape[0]
    # compute initial M
    best_a = a.copy()
    best_M = _peak_convolution(a)
    cur_a = a.copy()
    cur_M = best_M

    total_time = deadline - time.time()
    # initial temperature maybe (best_M - 1.0) maybe 0.5 test
    T0 = max(0.1, (best_M - 1.0) * 0.5) # adjust
    # bounding factor for move size
    # We allow move_frac up to maybe 0.2.
    while time.time() < deadline:
        elapsed = time.time() - (deadline - total_time)
        frac = elapsed / total_time
        T = T0 * (1 - frac)  # linear cooling to zero
        # adapt move size: bigger early
        max_frac = 0.2 * (1 - frac) + 0.02
        # pick source i where value > small
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * max_frac
        # apply move
        cur_a[i] -= delta
        cur_a[j] += delta

        # compute M
        new_M = _peak_convolution(cur_a)

        # accept?
        if new_M < cur_M or (T > 0 and rng.random() < np.exp(-(new_M - cur_M) / T)):
            cur_M = new_M
            if new_M < best_M:
                best_M = new_M
                best_a = cur_a.copy()
        else:
            # revert move
            cur_a[i] += delta
            cur_a[j] -= delta

        # occasionally random permutation of two elements (swap values) triggers bigger changes
        if rng.random() < 0.01:
            i2, j2 = rng.integers(0, n, size=2)
            if i2 != j2:
                # swap values (preserves sum)
                cur_a[i2], cur_a[j2] = cur_a[j2], cur_a[i2]
                new_M2 = _peak_convolution(cur_a)
                if new_M2 < cur_M or (T > 0 and rng.random() < np.exp(-(new_M2 - cur_M) / T)):
                    cur_M = new_M2
                    if new_M2 < best_M:
                        best_M = new_M2
                        best_a = cur_a.copy()
                else:
                    # swap back
                    cur_a[i2], cur_a[j2] = cur_a[j2], cur_a[i2]

    return best_a, best_M

def _simulated_annealing(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    cur_M = _max_convolution(a)
    best_M = cur_M
    total_time = deadline - time.time()
    if total_time <= 0:
        return best_a, best_M
    start = time.time()
    # set initial temperature based on current M
    T0 = max(0.01, (cur_M - 1.0) * 0.4)  # maybe
    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start
        frac = elapsed / total_time
        T = max(0.0, T0 * (1.0 - frac))
        # move fraction: larger early, smaller later
        max_move_frac = 0.15 * (1.0 - frac) + 0.01
        i = rng.integers(0, n)
        if a[i] <= 0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = a[i] * rng.random() * max_move_frac
        # apply
        a[i] -= delta
        a[j] += delta
        # compute M
        new_M = _max_convolution(a)
        # acceptance
        accept = False
        if new_M < cur_M:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_M - cur_M) / T)
                if rng.random() < prob:
                    accept = True

"""
Propose a step‑function (non‑negative vector) whose self‑convolution peak
is as small as possible.

Algorithm
---------
1. **Warm start** – if a previous sequence is available (`height_sequence_1`)
   it is used, otherwise a random vector is generated.
2. **Iterative search** – while time permits the following loop is repeated:
       * pick a length `n` (either up‑sample the current best or choose a
         fresh random length ≤ 15000);
       * generate an initial vector (`a0`) and normalise it to
         Σa = √(2·n) (the simplex constraint);
       * run a **simulated‑annealing** optimiser for a short time slice.
         The optimiser moves a small amount of mass from one index to another
         (preserving the sum) and occasionally swaps two entries, accepting
         worse moves with probability exp(‑Δ/ T).  Temperature and move size
         decay linearly to zero.
       * keep the best vector ever seen.
3. **Final evaluation** – the best vector is returned as a plain list and
   stored globally as `height_sequence_1` for possible warm‑restarts.

The optimiser works directly on the objective
    M(a) = max_k (a * a)_k   (linear convolution),
which equals the external score because the sum is fixed to √(2·n).

The code is self‑contained, uses only `numpy` and the provided
`evaluate_sequence` routine, and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution (full length = 2*n-1) via FFT
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a * a)_k using an FFT based linear convolution."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


# ----------------------------------------------------------------------
# Linear interpolation (used for up‑sampling)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `base` to length `new_n` by linear interpolation."""
    if base.size == new_n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Normalisation helper – forces Σa = √(2·n)
# ----------------------------------------------------------------------
def _normalise_to_sum(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    total = a.sum()
    if total <= 0:
        a = np.full_like(a, target / n, dtype=float)
    else:
        a = a * (target / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected simulated‑annealing that moves tiny masses between indices.
    Returns the best vector encountered and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_M = _max_convolution(a)
    best_a = a.copy()
    best_M = cur_M

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # Initial temperature – a fraction of the current excess over the theoretical lower bound (≈1)
    T0 = max(0.02, (cur_M - 1.0) * 0.4)

    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start
        frac = elapsed / total_time          # 0 → 1
        T = max(0.0, T0 * (1.0 - frac))      # linear cooling

        # step size shrinks with temperature
        max_move_frac = 0.15 * (1.0 - frac) + 0.01   # 0.16 → 0.01

        # pick a source with non‑zero mass
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        delta = a[i] * rng.random() * max_move_frac   # amount moved from i to j
        a[i] -= delta
        a[j] += delta

        new_M = _max_convolution(a)

        accept = False
        if new_M < cur_M:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0.0:
                prob = np.exp(-(new_M - cur_M) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_M = new_M
            if new_M < best_M:
                best_M = new_M
                best_a = a.copy()
        else:
            # revert move
            a[i] += delta
            a[j] -= delta

        # occasional pure index‑swap (no mass change) to explore permutations
        if rng.random() < 0.005:
            p, q = rng.integers(0, n, size=2)
            if p != q:
                a[p], a[q] = a[q], a[p]
                swap_M = _max_convolution(a)
                accept_swap = False
                if swap_M < cur_M:
                    accept_swap = True
                elif T > 0.0 and rng.random() < np.exp(-(swap_M - cur_M) / T):
                    accept_swap = True

                if accept_swap:
                    cur_M = swap_M
                    if swap_M < best_M:
                        best_M = swap_M
                        best_a = a.copy()
                else:
                    # revert swap
                    a[p], a[q] = a[q], a[p]

    return best_a, best_M

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    external = evaluate_sequence(final_list)
    print(f"[return] external={external:.6f} internal={best_M:.6f} "
          f"length={len(final_list)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_list, dtype=float)
    return final_list
