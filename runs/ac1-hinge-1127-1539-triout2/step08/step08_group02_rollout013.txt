def solve_lp_for_g(f, M, S):
    n = len(f)
    num_rows = 2*n - 1
    # Build A
    A = np.zeros((num_rows, n), dtype=np.float64)
    idx = np.arange(n)
    for i in range(n):
        rows = i + idx
        A[rows, idx] = f[i]
    # b vector
    b = np.full(num_rows, M, dtype=np.float64)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    # bounds: g_i >= 0
    bounds = [(0, None)] * n
    # Solve:
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: maybe return original f?
        return f.copy()
    g0 = res.x
    # Rescale to sum S
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f.copy()
    g = g0 * (S / sum_g0)
    return g

def lp_iterative_opt(f, S, t=0.2, max_iters=30, tol=1e-6):
    best_f = f.copy()
    best_M = _max_convolution(best_f)
    for it in range(max_iters):

        f_new = (1 - t) * best_f + t * g

            best_f = f_new

"""
Step‑function optimizer (2026 edition).

The optimizer builds a candidate by three complementary ideas:

1. **Broad random exploration** – many parametric families are sampled,
   the best raw shape is kept (the evaluation is cheap thanks to FFT).

2. **Gradient based refinement** – Adam on the log‑softmax parametrisation
   of the vector minimises the exact maximum of the auto‑convolution.
   The optimizer uses a τ‑soft‑max that is driven to zero so the
   gradient eventually becomes a true sub‑gradient of the max‑operator.

3. **Linear‑programming polishing** – for moderate lengths (≤ 2500) the
   classic “LP‑step” from the paper is performed:
   given the current sequence f, solve
       maximise   Σ g_i
       subject to (f∗g)_k ≤ M   (M = max(f∗f))   and g_i ≥ 0 .
   The solution g is rescaled to the required ℓ₁‑norm and mixed with f.
   Repeating this step strictly reduces the peak value.
   A short simulated annealing hill‑climb finishes the run.

The whole procedure respects the 1000 s budget.  The final sequence is
stored in the global variable ``height_sequence_1`` and returned as a
plain Python list.
"""

import time
import numpy as np

# optional LP import – if unavailable the LP phase is skipped
try:
    from scipy.optimize import linprog
    _HAS_LP = True
except Exception:                     # pragma: no cover
    _HAS_LP = False

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, scaling, soft‑max gradient, Adam)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # exact sub‑gradient of the hard max
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / max(tau, 1e-30))
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the true maximum of the auto‑convolution.

    Returns (best_sequence, best_peak) where best_peak = max(a*a).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits from the supplied direction a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.55                           # a bit larger than the baseline

    # start temperature a little above the first peak
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # --- recover a from logits (log‑softmax) ---
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # --- gradient of (soft) max ---
        grad_a, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # --- gradient w.r.t. logits (chain rule) ---
        mu = np.dot(a, grad_a) / S                # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (grad_a - mu)                # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 8.0 * S:
            grad_p = grad_p * (8.0 * S / clip_val)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- schedules ---
        tau *= 0.9995                # faster cooling (more exact sub‑gradient)
        lr *= 0.9998                 # gentle decay

        # occasional jitter to escape plateaus
        if it % 1200 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    raw = rng.random(n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    “U‑shape’’ – large values at the edges, strongly suppressed in the centre.
    a_i ∝ ((i+1)*(n-i))^{-p} with optional uniform mixing.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sinusoid_mix(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    A sinusoidal mixture – the sine curve is flipped (U‑like) and mixed
    with a small uniform background.
    """
    i = np.arange(n)
    sin_part = (np.sin(np.pi * (i + 0.5) / n)) ** (-1.0)   # peaks at the ends
    base = np.ones(n)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * sin_part + (1.0 - mix) * base
    raw = np.maximum(raw, 0.0)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best sequence,
    otherwise draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'ushape', 'sinusoid'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sinusoid':
        raw = _init_sinusoid_mix(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

# Linear‑programming polishing (the “LP” step from the paper)
# ----------------------------------------------------------------------


def _solve_lp_for_g(f: np.ndarray, M: float, S: float) -> np.ndarray:
    """
    Solve  maximise Σ g_i   s.t.   (f ∗ g)_k ≤ M   (∀k),   g_i ≥ 0.
    Returns the scaled solution g with Σ g_i = S.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A (rows × n)
    # A[k, j] = f_{k-j}   when 0 ≤ k‑j < n, else 0
    A = np.zeros((rows, n), dtype=np.float64)
    idx = np.arange(n)
    for i in range(n):
        A[i + idx, idx] = f[i]

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  → minimise -Σ g_i
    bounds = [(0, None)] * n

    # Solve the LP (highs is the default modern solver)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # LP failed – fall back to the current vector
        return f.copy()

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        return f.copy()
    g = g0 * (S / sum_g0)
    return g


def _lp_refine(f: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_iters: int = 20,
               step: float = 0.18) -> tuple[np.ndarray, float]:
    """
    Repeated LP polishing.  Returns (best_sequence, best_peak).
    """
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    best_f = f.copy()
    best_M = _max_convolution(best_f)

    it = 0
    while time.time() < deadline and it < max_iters:
        M = _max_convolution(best_f)
        g = _solve_lp_for_g(best_f, M, S)
        # convex combination
        f_new = (1.0 - step) * best_f + step * g
        M_new = _max_convolution(f_new)

        if M_new + 1e-12 < best_M:
            best_M = M_new
            best_f = f_new
            print(f"[LP] iter={it+1:2d}  M={best_M:.6f}")
        else:
            # no progress – stop early
            break
        it += 1
    return best_f, best_M

        a = best_seq.copy()
        n = a.shape[0]
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # move a random fraction of mass from i to j
        frac = rng.uniform(0.02, 0.14)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-12 < best_val:
            best_val = peak
            best_seq = a.copy()
            print(f"[hill] improved peak={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)          # type: ignore[name-defined]
    print(f"[final] external score = {external:.6f}  "
          f"internal max = {best_val:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
