"""
Propose a sequence of non‑negative heights that minimizes the
`evaluate_sequence` objective.

The algorithm improves on the previous implementation by

* exploring a richer set of parametric families (exponential,
  double‑exponential, power‑law, logistic, truncated Gaussian);
* performing a coarse grid search over the exponential decay parameter;
* allocating a longer schedule of lengths (up to 200 000);
* running several Adam restarts with a more aggressive β‑schedule;
* using a slower‑cooling hill‑climber and allowing several restarts;
* giving more time to the later (larger) problem sizes.

All internal evaluations of the max‑convolution use FFT (O(n log n)),
while the final score is obtained with the provided `evaluate_sequence`
function.  The code respects a global wall‑clock budget (default 1000 s)
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log Σ exp(β·b_k)  where
    b = a⋆a (full convolution). Returns (grad, b).

    grad_i = 2 * Σ_k p_k * a_{k-i},
    where p_k ∝ exp(β·b_k) and the sum is over the full convolution range.
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the convolution entries
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev ⋆ p   (a_rev is a reversed)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 0.5,
                   beta_target: float = 1e6,
                   beta_factor: float = 1.4) -> np.ndarray:
    """
    Adam optimiser for the smoothed‑max objective.
    The β‑parameter grows geometrically from ``beta_start`` to ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.3          # a slightly larger initial step size
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the smoothed max
        g, _ = _softmax_gradient(a, beta)

        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection onto the ℓ¹‑simplex (keeps scale invariant)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if step % 500 == 0:
            jitter = rng.normal(scale=target_sum * 2e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if beta < beta_target and step % 200 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a

def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)"""
    r = rng.uniform(0.94, 0.9999)
    idx = np.arange(n, dtype=float)
    return r ** idx


def _family_double_exponential(n: int, rng) -> np.ndarray:
    """Two‑segment exponential with a breakpoint."""
    b = rng.integers(int(0.2 * n), int(0.8 * n))
    r1 = rng.uniform(0.94, 0.999)
    r2 = rng.uniform(0.85, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}"""
    p = rng.uniform(0.1, 2.0)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_logistic(n: int, rng) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑mid)/scale))"""
    mid = rng.uniform(0, n - 1)
    scale = rng.uniform(0.05 * n, 0.3 * n)
    i = np.arange(n, dtype=float)
    return 1.0 / (1.0 + np.exp((i - mid) / scale))


def _family_gaussian(n: int, rng) -> np.ndarray:
    """Truncated Gaussian shape."""
    mu = rng.uniform(0, n - 1)
    sigma = rng.uniform(0.05 * n, 0.3 * n)
    i = np.arange(n, dtype=float)
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


def _family_piecewise_linear(n: int, rng) -> np.ndarray:
    """Linear rise to a peak then linear decay."""
    peak = rng.integers(int(0.2 * n), int(0.8 * n))
    left = np.linspace(0.0, 1.0, peak, endpoint=False)
    right = np.linspace(1.0, 0.0, n - peak, endpoint=False)
    return np.concatenate([left, right])


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Choose randomly among several parametric families,
    add a tiny jitter and project onto the ℓ¹‑simplex.
    """
    families = [_family_exponential,
                _family_double_exponential,
                _family_power_law,
                _family_logistic,
                _family_gaussian,
                _family_piecewise_linear]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)          # tiny jitter
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _geometric_grid_search(n: int, rng, trials: int = 30) -> tuple[np.ndarray, float]:
    """
    Search a single‑parameter exponential family a_i = C·r^i for the best
    max‑convolution value.  Returns the best vector and its score.
    """
    best_a = None
    best_score = np.inf
    target = np.sqrt(2.0 * n)
    for _ in range(trials):
        # sample r densely near 1 (log‑uniform on (1‑r))
        u = rng.uniform(-7, -0.5)          # u = log10(1‑r)
        r = 1.0 - 10 ** u
        raw = r ** np.arange(n, dtype=float)
        C = target / raw.sum()
        a = raw * C
        score = _max_convolution_fft(a)
        if score < best_score:
            best_score = score
            best_a = a
    return best_a, best_score


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a “source” index to a “target”
    index while keeping the full convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.4 + 1e-9
    cooling = 0.9995

    # limit number of iterations to avoid endless loops on very large n
    max_iter = int(5e5)        # generous upper bound; real stop is time‑limit
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # pick one index attaining the current maximum (could be several)
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # pick a source index i that contributes to this maximum entry
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # pick a target index j (prefer low‑mass positions)
        if rng.random() < 0.4:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # choose delta – we use a fraction of the source mass
        max_delta = a[i] * 0.7
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----
        # remove contribution of source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # add contribution of target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correction for the cross term i+j (appears twice in the two updates)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # recompute the convolution of the best candidate for safety
    b_best = _full_convolution(best_a)
    return best_a, b_best

            a_start = _random_initial_sequence(n, rng)

        # run a few Adam restarts (2 for larger n, more for smaller n)
        n_restarts = 2 if n >= 50000 else 3
        for r in range(n_restarts):
            if time.time() >= gd_deadline:
                break
            a_opt = _adam_optimize(a_start, gd_deadline,
                                   beta_start=0.5,
                                   beta_target=1e6,
                                   beta_factor=1.5)
            score_opt = _max_convolution_fft(a_opt)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] Adam restart {r+1}/{n_restarts} improvement (n={n}) score={score_opt:.6f}")

        # --------------------------------------------------------------
        # 4) Exact hill‑climbing (simulated annealing)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        hc_time = min(remaining * 0.30, 200.0)
        hc_deadline = time.time() + hc_time

        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = float(b_hc.max())
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb improvement (n={n}) score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – use any leftover budget for one more Adam + HC
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final Adam pass (longer schedule)
        a_final = _adam_optimize(best_seq.copy(),
                                 deadline - 3.0,
                                 beta_start=1.0,
                                 beta_target=2e6,
                                 beta_factor=1.4)

        # short hill‑climb to clean up remaining rough edges
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 2.0)
        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy return value
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely fallback – uniform distribution of modest length
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (max‑convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
