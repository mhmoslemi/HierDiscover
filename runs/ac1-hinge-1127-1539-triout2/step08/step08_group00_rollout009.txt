```python
"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence (i.e. the maximal entry of its self‑convolution).

Algorithm
---------
1.  Initialise from a previously discovered sequence (height_sequence_1)
    if available, otherwise from a small random family.
2.  Follow a length schedule that gradually grows the support up to
    ≈200 k entries.  For each length:
       •  sample a few random parametric families and keep the best;
       •  run Adam optimisation on the smooth log‑sum‑exp surrogate;
       •  run a mirror‑descent optimiser (exponential‑gradient);
       •  finish with an exact hill‑climbing (simulated annealing) move set.
3.  Use any remaining time for a final polishing pass (Adam + hill‑climb).
4.  Trim trailing near‑zeros and return the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log Σ exp(β·b_k)  where
    b = a⋆a (full convolution).  Returns (grad, b) with

        grad_i = 2·∑_k p_k·a_{k-i}   (p_k ∝ exp(β·b_k))
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities (numerically stable)
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev ⋆ p  (grad factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 1e5,
                   beta_factor: float = 1.3) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max objective.
    The schedule for β grows geometrically until ``beta_target``.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)

        a -= lr * m_hat
        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaux
        if step % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if beta < beta_target and step % 250 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      beta_start: float = 2.0,
                      beta_target: float = 2e5,
                      beta_factor: float = 1.5,
                      eta: float = 0.4) -> np.ndarray:
    """
    Mirror‑descent / exponentiated‑gradient optimiser.
    Uses the same smoothed‑max gradient but performs a multiplicative
    update a_i ← a_i·exp(−η·g_i) followed by simplex projection.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    beta = beta_start

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        # multiplicative update
        a = a * np.exp(-eta * g)
        a = _project_to_simplex(a, target_sum)

        # increase smoothing parameter
        if beta < beta_target:
            beta = min(beta * beta_factor, beta_target)

    return a


# ----------------------------------------------------------------------
# Random families (used for cheap exploration)
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)"""
    r = rng.uniform(0.96, 0.9995)
    idx = np.arange(n, dtype=float)
    return r ** idx


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}"""
    p = rng.uniform(0.2, 1.5)
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential decay with a breakpoint."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a family uniformly at random, jitter a little and project to the simplex."""
    families = [_family_exponential,
                _family_power_law,
                _family_linear,
                _family_piecewise_exp]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, np.sqrt(2.0 * n))
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = _project_to_simplex(a_new, np.sqrt(2.0 * new_n))
    return a_new


# ----------------------------------------------------------------------
# Exact hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a “source” index to a “target” index,
    while maintaining the full convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # initial temperature proportional to current max
    T = cur_max * 0.2 + 1e-9
    cooling = 0.998

    while time.time() < deadline:
        # pick a position attaining the current maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – choose among contributors to this entry
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias towards small‑mass positions
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 90 % of source)
        max_delta = a[i] * 0.9
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- exact convolution update (O(n)) -----
        # remove contribution of source i
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # add contribution of target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correct the cross term at i+j (was counted twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # recompute exact convolution for the best candidate
    b_best = _full_convolution(best_a)
    return best_a, b_best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock limit and returns the best sequence
    found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) Seed from a historic construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            if a0.size > 0:
                n0 = a0.size
                a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
                val0 = _max_convolution_fft(a0)
                best_seq = a0.copy()
                best_val = val0
                print(f"[{time.time():.1f}] seed from height_sequence_1 (n={n0}) val={val0:.6f}")
        except Exception as e:
            print(f"Unable to seed from height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # 2) Fallback random initialisation (if seeding failed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 2000
        best_seq = _random_initial_sequence(n0, rng)
        best_val = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] random init (n={n0}) val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Length schedule – grow the support gradually
    # ------------------------------------------------------------------
    length_schedule = [2000, 4000, 8000, 16000, 32000, 64000, 128000, 200000]

    for n in length_schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Upsample current best to the new length (if needed)
        # --------------------------------------------------------------
        if best_seq.size != n:
            a_start = _upsample_sequence(best_seq, n)
        else:
            a_start = best_seq.copy()

        # --------------------------------------------------------------
        # Phase A – random parametric families (cheap)
        # --------------------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _random_initial_sequence(n, rng)
            val = _max_convolution_fft(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] random family improvement (n={n}) val={val:.6f}")

        # --------------------------------------------------------------
        # Phase B – Adam optimisation on smoothed max
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        gd_deadline = time.time() + min(remaining * 0.40, 250.0)
        a_adam = _adam_optimize(a_start, gd_deadline)
        val_adam = _max_convolution_fft(a_adam)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.1f}] Adam improvement (n={n}) val={val_adam:.6f}")

        # --------------------------------------------------------------
        # Phase C – Exponential‑gradient (mirror descent) optimisation
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 2.0:
            eg_deadline = time.time() + min(remaining * 0.25, 120.0)
            a_eg = _expgrad_optimize(a_start, eg_deadline)
            val_eg = _max_convolution_fft(a_eg)
            if val_eg < best_val:
                best_val = val_eg
                best_seq = a_eg.copy()
                print(f"[{time.time():.1f}] Exp‑grad improvement (n={n}) val={val_eg:.6f}")

        # --------------------------------------------------------------
        # Phase D – Exact hill‑climbing (simulated annealing)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        hc_deadline = time.time() + min(remaining * 0.30, 200.0)

        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)
        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        val_hc = float(b_hc.max())
        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) val={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 4) Final polishing on the best length found
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final Adam pass (more aggressive smoothing)
        a_final = _adam_optimize(best_seq.copy(),
                                 deadline - 3.0,
                                 beta_start=4.0,
                                 beta_target=3e5)

        # final hill‑climb
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 2.0)
        final_val = float(b_final.max())
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: val={final_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Trim trailing near‑zeros for a tidy return value
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 6) Report the official evaluation
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence -> {eval_val:.6f} (len={best_seq.size})")
    except Exception as exc:        # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```

The function `propose_candidate` respects the 1000‑second budget, explores a broad family of candidates, uses two complementary gradient‑based optimisers, and finishes with an exact hill‑climbing phase. It returns the best sequence it discovers as a plain list of floats.
