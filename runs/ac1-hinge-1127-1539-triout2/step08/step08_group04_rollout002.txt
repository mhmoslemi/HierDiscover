def _softmax_gradient(a, beta):
    n=a.shape[0];
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = conv.max()
    # compute probabilities; safe scaling
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

"""
Propose a step‑function (non‑negative sequence) whose self‑convolution
has a small maximal entry.  The algorithm

1. Chooses a problem size n (larger n → more degrees of freedom).
2. Generates several sensible initial sequences (power‑law, exponential,
   Hann window, triangular, random and the previously found best).
3. Runs an Adam‑style projected gradient descent on a soft‑max
   approximation of the maximal convolution entry.  The temperature β
   is increased geometrically so the surrogate approaches the true max.
   After each update the vector is projected onto the simplex
   {x≥0, Σx=S} to keep the ℓ¹‑norm fixed (S=√(2n)).
4. Interleaves occasional jitter and a periodic “mass‑transfer”
   refinement that moves a tiny fraction of mass from indices that
   contribute most to the current peak toward indices that contribute
   least, accepting only improving moves.
5. Keeps the globally best sequence seen and returns it as a plain list.
   The final sequence is evaluated with the supplied ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=S this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv, max_conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv.max()
    # soft‑max weights (all ≤ 1, numerical stability ensured)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv, max_b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best sequence and its score found before *deadline*.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())           # exact normalisation
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 1
    beta = 1.0
    max_beta = 32768.0
    base_alpha = 0.08 * target_sum          # base step size (tuned empirically)
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, conv, max_b = _softmax_gradient(a, beta)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_alpha / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat

        # projection onto the simplex Σa = target_sum, a ≥ 0
        a = _project_to_simplex(a_new, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase temperature β
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} β={beta:6.1f} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from indices that contribute most to the
    current maximal convolution entry towards indices that contribute least.
    Accept only moves that improve the objective.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        # sensitivities for the current maximal entry (grad components)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        # donation weights proportional to contribution a[i]*sens[i]
        weight = a * sens
        total_w = weight.sum()
        if total_w > 0:
            donor = rng.choice(n, p=weight / total_w)
        else:
            donor = rng.integers(n)

        # pick a low‑contribution receiver (bottom 10 % of weight)
        low_thr = np.percentile(weight, 10)
        low_idx = np.where(weight <= low_thr)[0]
        receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

        if donor == receiver:
            continue

        # transfer a modest fraction (≤3 %)
        max_frac = 0.03
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # re‑normalise occasionally (numeric drift)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val
