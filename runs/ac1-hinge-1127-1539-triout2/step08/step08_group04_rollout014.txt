"""
Propose a step‑function (non‑negative sequence) with a low benchmark score.

Algorithm
---------
1. **Problem size** – choose the length `n` according to the time budget
   (larger `n` gives a richer search space).
2. **Warm start** – grab the best sequence found so far (global
   `height_sequence_1`) or build a simple power‑law shape.
3. **Cheap parametric search on a reduced scale** – explore the two‑parameter
   family   a_i = (i+1)^(-p)·exp(-λ·i / small_n)   on a small problem,
   keep the best shape and interpolate it to the target size.
4. **Hard‑max mirror descent** – repeatedly minimise the current maximal
   convolution entry using the exact sub‑gradient
   (∂(a*a)[k]/∂a_i = 2·a_{k-i} when the index is inside the array).
   An exponential update `a ← a·exp(‑η·g)` preserves non‑negativity;
   after every update we renorm the vector to the required ℓ¹‑norm.
5. **Local mass‑transfer refinement** – move a small fraction of mass
   from indices that contribute heavily to the current peak to indices
   that contribute little.  Each trial is accepted only if it improves the
   peak value; the step size is chosen from a set of fractions.
6. **Final clean‑up** – truncate trailing zeros, store the result for later
   warm‑starts and return it as a plain list.

All stages respect the simplex constraint `a_i ≥ 0, Σa_i = √(2n)`,
so the returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# Simple helpers

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Search a two‑parameter family
        a_i = (i+1)^(-p) * exp(-λ·i / small_n)
    on a small problem.  Returns the best sequence found.
    """
    p_vals = np.linspace(0.30, 0.90, 13)
    lam_vals = np.linspace(0.0, 8.0, 17)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    # deterministic grid
    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    # random refinement until time runs out
    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.30, 0.90)
        lam = rng.exponential(1.0) * 4.0   # typical range up to ~8
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score

def _hard_max_descent(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the exact sub‑gradient of the maximal convolution entry.
    Multiplicative update keeps the vector non‑negative.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # full convolution and its argmax
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max entry
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # exponential (mirror) step; decreasing η helps stability
        eta = 0.6 / (1.0 + it * 1e-4)
        a = a * np.exp(-eta * g)

        # renormalise to keep the ℓ¹‑norm exact
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] HardMax it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------


def _local_transfer_refine(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from “heavy” to “light” indices.
    Only accept moves that strictly improve the peak of the convolution.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    conv = _conv_fft(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # current peak location
        k_max = int(np.argmax(conv))

        # sensitivities w.r.t. the current peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = np.zeros(n, dtype=np.float64)
        sens[idx] = a[k_max - idx]

        # weight = a * sens ; donor proportional to this weight
        weight = a * sens
        total_w = weight.sum()
        if total_w > 0.0:
            donor = rng.choice(n, p=weight / total_w)
        else:
            donor = rng.integers(n)

        # receiver – pick from the lower 20 % of the weight distribution
        low_thr = np.percentile(weight, 20)
        low_candidates = np.where(weight <= low_thr)[0]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = a[donor]
        # try a few fractions of the donor mass
        improved = False
        for frac in (0.35, 0.20, 0.10, 0.05):
            delta = max_delta * frac
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[donor] -= delta
            a_new[receiver] += delta

            new_conv = _conv_fft(a_new)
            new_val = new_conv.max()
            if new_val < best_val:
                # accept
                a = a_new
                conv = new_conv
                best_val = new_val
                best_a = a.copy()
                improved = True
                it += 1
                print(f"[{time.time():.2f}] Transfer it={it:5d} d→r {donor}->{receiver} "
                      f"δ={delta:.4e} score={best_val:.6f}")
                break  # move on after first improvement
        if not improved:
            # occasionally try a pure random jitter to avoid stagnation
            if rng.random() < 0.001:
                jitter = rng.normal(scale=target_sum * 0.0005, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = a * (target_sum / a.sum())
                conv = _conv_fft(a)
        # keep loop running
        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # a tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 12000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial sequence (fallback)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a small problem
    # ------------------------------------------------------------------
    small_n = max(2000, n // 5)
    small_target = math.sqrt(2.0 * small_n)
    phase1_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.07)
    best_small_seq, _ = _grid_param_search(small_n, small_target, phase1_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – hard‑max mirror descent (≈55 % of remaining budget)
    # ------------------------------------------------------------------
    phase2_deadline = time.time() + (deadline - time.time()) * 0.55
    a_hard, val_hard = _hard_max_descent(init, target_sum, phase2_deadline)
    best_seq = a_hard
    best_score = val_hard
    print(f"[{time.time():.2f}] After hard‑max descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – local mass‑transfer refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_transfer_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After transfer refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:

    final_seq = arr.tolist()

    # Evaluate with the provided benchmark (may raise if not imported)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
