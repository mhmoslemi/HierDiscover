def _adam_exact_optimize(a0, deadline, rng):
    n = len(a0)
    S = sqrt(2*n)
    # initial logits from a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p); v = np.zeros_like(p)
    beta1 = 0.9; beta2 = 0.999; eps_adam = 1e-8
    lr = 0.2   # smaller steps for exact gradient

    best_a = a0.copy()
    best_peak = _max_convolution(a0)

    it = 0
    while time.time() < deadline:
        it += 1
        # compute a
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # full convolution and max
        b = _full_convolution_fft(a)
        max_b = b.max()

        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

        # compute mask: 1 for positions within tolerance (e.g., max - epsilon)
        eps_mask = 1e-12
        mask = (b >= max_b - eps_mask).astype(float)  # can be multiple entries

        # compute gradient: g = 2 * conv(mask, a_rev)[:n][::-1]
        a_rev = a[::-1]
        # Convolution of mask and a_rev:
        conv = _convolve_fft(mask, a_rev)  # need new function
        g = 2.0 * conv[:n][::-1]

        # Chain rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # optionally add jitter
        if it % 1000 == 0:
            p += rng.normal(scale=0.005, size=n)

    return best_a, best_peak

def _init_beta(n, rng):
    p = rng.uniform(0.2, 3.0) # shape for i+1

"""
Advanced step‑function optimizer.

The algorithm proceeds in four stages:

1. **Broad random search** – many parametric families (power‑law,
   exponential, triangular, beta, sine, U‑shape, etc.) are sampled
   with a variety of lengths.  The best candidate (lowest hard
   convolution peak) is kept.
2. **Geometric scaling & gradient refinement** – the current best
   sequence is linearly interpolated to several larger lengths and
   each scaled copy is optimised with two Adam runs:
   * a “soft‑max” Adam run (smooth approximation of the max)
   * an “exact‑max” Adam run (sub‑gradient of the true peak).
3. **Final long Adam optimisation** on the best length.
4. **Cheap stochastic hill‑climb** and a final symmetry projection.

All updates respect the non‑negativity and ℓ₁‑norm constraint
‖a‖₁ = √(2·n); after the optimisation the sequence is symmetrised,
renormalised and returned as a plain Python list stored in the global
``height_sequence_1``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = rfft(a, n=f_len)
    conv = irfft(A * A, n=f_len)[:total_len]
    return conv


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = rfft(x, n=f_len)
    Y = rfft(y, n=f_len)
    conv = irfft(X * Y, n=f_len)[:n]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Simple geometry helpers
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient (τ‑soft‑max) – used in the first Adam phase
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = rfft(mask, n=f_len)
    Fa_rev = rfft(a_rev, n=f_len)
    conv = irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser – soft‑max version
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex (log‑softmax parametrisation).
    Uses the τ‑soft‑max gradient (smooth surrogate of the max‑convolution).
    Returns the best vector found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                           # fairly aggressive start

    # start temperature – a bit larger than the first peak
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient & exact peak
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain‑rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules – keep a little smoothing longer
        tau *= 0.9998
        lr *= 0.9999

        # occasional jitter
        if it % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Adam optimiser – exact‑max version (sub‑gradient of the hard peak)
# ----------------------------------------------------------------------
def _adam_exact_optimize(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex using the true sub‑gradient of
    max(a*a).  The gradient is computed from the mask of peak locations.
    Returns best vector and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.22                     # slightly smaller than soft‑max phase

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # hard convolution and mask of maximisers
        b = _full_convolution_fft(a)
        peak = b.max()

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        mask = (b >= peak - 1e-12).astype(float)   # sub‑gradient mask

        # compute sub‑gradient of the hard max:
        # g_i = 2 * sum_k mask[k] * a_{k-i}
        a_rev = a[::-1]
        conv = _convolve_fft(mask, a_rev)          # length = 3n-2
        g = 2.0 * conv[n - 1:n - 1 + n]             # slice of length n

        # chain‑rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # jitter every now and then
        if it % 1000 == 0:
            p += rng.normal(scale=0.005, size=n)

    return best_a, best_peak

# Random parametric families for initialisation

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    raw = rng.random(n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shape – large values at the edges, strongly suppressed in the centre."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)^{p-1} (n-i)^{q-1} with optional mixing."""
    i = np.arange(1, n + 1)                     # 1 … n
    rev = np.arange(n, 0, -1)                   # n … 1
    p = rng.uniform(0.2, 3.0)
    q = rng.uniform(0.2, 3.0)
    mix = rng.uniform(0.0, 1.0)
    beta_shape = (i ** (p - 1)) * (rev ** (q - 1))
    base = np.ones(n)
    raw = mix * beta_shape + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sine‑type shape – sin(π·(i+½)/n) raised to a power, mixed with uniform."""
    i = np.arange(n)
    p = rng.uniform(0.5, 5.0)
    mix = rng.uniform(0.0, 1.0)
    sine_part = np.sin(np.pi * (i + 0.5) / n) ** p
    base = np.ones(n)
    raw = mix * sine_part + (1.0 - mix) * base
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    20 % of the time a warm‐start (scaled former best) is used,
    otherwise a random parametric family is sampled.
    """
    S = np.sqrt(2.0 * n)

    # Warm‑start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        tot = a.sum()
        if tot == 0.0:
            a = np.full(n, 1.0)
            tot = n
        a = a * (S / tot)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'ushape', 'beta', 'sine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'beta':
        raw = _init_beta(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    tot = raw.sum()
    if tot == 0.0:
        raw = np.full(n, 1.0)
        tot = n
    a = raw * (S / tot)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best found sequence as a plain Python list and stores it in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin for final checks

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 60 % of the total budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.60 * budget_s
    while time.time() < explore_deadline:
        attempt += 1

        # bias length towards a few thousand, occasionally sample larger ones
        if rng.random() < 0.78:
            n = int(rng.integers(1800, 8000))
        else:
            n = int(rng.integers(8000, 24000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & refined Adam optimisation on several lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of moderate size
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # Geometrically increasing lengths (up to the safe bound 25000)
    lengths = [best_n]
    factors = [1.2, 1.5, 2.0, 3.0, 4.0]
    for f in factors:
        cand = int(best_n * f)
        if 2000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 30.0:
            break
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 200.0)          # bounded per length

        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # soft‑max Adam run
        a_opt, peak = _adam_optimize(a_scaled,
                                      time.time() + slice_time * 0.6,
                                      rng)

        # exact‑max Adam run (finer improvement)
        a_opt2, peak2 = _adam_exact_optimize(a_opt,
                                             time.time() + slice_time * 0.4,
                                             rng)

        # keep the better of the two
        if peak2 < peak:
            a_opt, peak = a_opt2, peak2

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final long Adam optimisation on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 12.0:
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining * 0.6,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final soft] len={best_n:5d}  best score={best_score:.6f}")

        # follow up with exact‑max refinement

                                            time.time() + remaining * 0.4,
                                            rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final exact] len={best_n:5d}  best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (remaining few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.03, 0.18)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[local] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final symmetry projection and sanity check
    # ------------------------------------------------------------------
    # Symmetry (a_i ≈ a_{n-1-i}) often improves the bound.
    a_sym = (best_seq + best_seq[::-1]) * 0.5
    S = np.sqrt(2.0 * a_sym.shape[0])
    a_sym = a_sym * (S / a_sym.sum())
    final_seq = a_sym.tolist()

    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
