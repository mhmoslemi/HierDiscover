def _deterministic_mass_transfer_modified(a0, deadline, target_sum,
                                         start_temp=0.5, end_temp=1e-4,
                                         low_frac=0.05):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # recompute conv
        conv = _conv_fft(a)

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # product weight
        w = a[idx] * a[kmax - idx]
        if w.max() == 0:
            # all zero? break
            break
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        # maximum delta limited by donor amount, and not exceed receiver capacity
        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)

"""
Optimising a non‑negative step function for the benchmark
---------------------------------------------------------

The score for a sequence `a` of length *n* is

    2·n·max(a*a) / (∑a)²

where `a*a` is the linear convolution of `a` with itself.
If we enforce ∑a = √(2·n) the factor 2·n/∑a² equals 1, i.e.
the score is simply the maximal convolution entry.
The algorithm below therefore tries to minimise that peak.

Main ideas
~~~~~~~~~~
1. **Rich initial pool** – power‑law, exponential and jittered
   variants, together with a resized previous best (if any).
2. **Adam‑style projected gradient descent** on a soft‑max
   approximation of the max‑convolution.  The gradient is obtained
   analytically via FFT (`grad = 2·rev(a) ★ p`, where `p` are the
   soft‑max weights).  After each step the vector is projected onto
   the simplex {x≥0, Σx = √(2·n)}.
3. **Deterministic mass‑transfer** – we repeatedly look at the index
   `k*` where the convolution attains its maximum and move a small
   amount of mass from an index with a large product
   `a_i·a_{k*-i}` (the main contributor to the peak) to an index
   with a small product.  Moves are accepted with a Metropolis rule
   and the temperature decays exponentially.
4. **Final polishing** – a short high‑β Adam run cleans up the
   remaining irregularities.

All operations are O(n log n) via FFT, the total running time respects
the 1000‑second budget.  The returned sequence is trimmed of trailing
zeros.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score – with the normalisation Σa = √(2n) this equals max‑conv."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Exponential shape a_i = C·lam**i  (0<lam<1), C chosen for target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = lam ** i
    if lam == 1.0:
        sum0 = float(n)
    else:
        sum0 = (1.0 - lam ** n) / (1.0 - lam)
    a *= target_sum / sum0
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = np.max(conv)
    # soft‑max weights
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)

    # grad = 2 * rev(a) ★ p   (valid only on the central n entries)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    rev_conv = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * rev_conv[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max loss
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           max_beta: float = 1e5,
                           step_factor: float = 0.6,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Momentum + adaptive step (Adam‑like) on the soft‑max surrogate,
    with projection onto the simplex after each step.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5               # temperature for the soft‑max
    m = np.zeros_like(a)     # 1st moment
    v = np.zeros_like(a)     # 2nd moment
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam update (scalar step, but element‑wise normalisation)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # element‑wise adaptive learning rate, then a global scaling
        adaptive_lr = (target_sum * step_factor) / (np.sqrt(v_hat) + eps)
        a -= adaptive_lr * m_hat

        a = _project_to_simplex(a, target_sum)

        # temperature schedule – double every few hundred steps
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        # occasional jitter to shake out of plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.2f} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (product‑based donor choice)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly attack the current maximal convolution entry.
    The donor is chosen as the index with the largest product a_i·a_{k*-i},
    the receiver among the low‑product indices.
    Moves are accepted via a Metropolis rule with a decaying temperature.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product that actually contributes to the peak
        w = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(w)]
        # choose a low‑product receiver
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        # transfer amount – bounded by donor’s current value
        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with minimal benchmark score.
    The routine builds a diverse initial pool, runs Adam‑style projected
    gradient descent, refines the result with a deterministic mass‑transfer
    phase and ends with a high‑β polishing pass.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # choose problem size (more points give a richer optimisation space)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    elif budget_s >= 300:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # build initial candidate pool
    # --------------------------------------------------------------
    candidates = []

    # power‑law shapes
    for p in (0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.65, 0.70):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # exponential shapes – sweep λ (as lam = exp(-eps))
    for eps in np.logspace(-4, -1.5, 10):   # eps = 1‑λ approx
        lam = np.exp(-eps)
        candidates.append(_exp_sequence(n, lam, target_sum))

    # jittered variants of everything above (adds stochastic diversity)
    rng = np.random.default_rng()
    for seq in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        candidates.append(seq_jit)

    # resized previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # --------------------------------------------------------------
    # quick evaluation – keep the few best as starting points
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:
        val = _score(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])
    top_k = 5
    best_seq = scored[0][1].copy()
    best_val = scored[0][0]
    print(f"[{time.time():.2f}] Initial best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam gradient descent (≈ 40 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.40
    for val, seq in scored[:top_k]:
        if time.time() >= phase_a_deadline:
            break
        a_opt, val_opt = _adam_gradient_descent(seq,
                                                target_sum,
                                                phase_a_deadline,
                                                max_beta=1e5,
                                                step_factor=0.7)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-4,
                                                      low_frac=0.05)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After refinement best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase C – Final polishing with a higher‑β Adam run
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                target_sum,
                                                deadline,
                                                max_beta=5e5,
                                                step_factor=0.4)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polished best={best_val:.6f}")

    # --------------------------------------------------------------
    # post‑processing – trim trailing near‑zeros
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (the harness may ignore the return value)
    try:
        eval_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {eval_score:.6f}")
    except Exception:
        pass

    return final_seq
