def _mass_transfer(a, target_sum, deadline, top_k=10, low_frac=0.1, start_temp=0.5, end_temp=1e-4):
    # similar to previous _deterministic_mass_transfer but uses topK weight scheme.
    a = a.copy(); best = a.copy(); best_score = _score(a)
    start_time = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        # TopK indices
        top_idx = np.argpartition(conv, -top_k)[-top_k:]   # unsorted
        # Compute weights for each i:
        w = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            idx = np.arange(lo, hi+1, dtype=int)
            w[idx] += a[idx] + a[k - idx]
        # select donor (max w) and receiver (min w)
        donor = np.argmax(w)
        # To avoid trivial donor=receiver or donor small, select low indices set:
        low_cnt = max(1, int(len(w)*low_frac))
        receivers = np.argsort(w)[:low_cnt]
        receiver = np.random.choice(receivers)
        if donor == receiver:
            continue
        # transfer amount:
        max_delta = a[donor] * 0.5
        # can also bound by weight difference
        weight_diff = w[donor] - w[receiver]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff)
        delta = np.random.rand() * max_delta
        if delta <= 0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)
        new_score = _score(a)
        # schedule temperature:
        elapsed = time.time() - start_time
        tau = elapsed/(deadline-start_time)
        temp = start_temp * (end_temp/start_temp)**tau
        # accept if improves else possibly accept with prob
        if new_score < best_score or np.random.rand() < np.exp(-(new_score - best_score)/max(temp,1e-12)):
            best_score = new_score
            best = a.copy()
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)
    return best, best_score

"""
Improved step function optimizer.

The algorithm consists of three phases:
1) ADAM-based projected gradient descent on a soft‑max approximation of the
   maximum self‑convolution, with a schedule that drives the temperature β
   from 0.5 to 1e6.
2) Deterministic mass‑transfer moves that target the top‑K convolution
   entries, guided by a temperature schedule.
3) A short high‑β ADAM polish.

The code is written without closures, using only numpy and the standard
library.
"""

import time
import numpy as np

def _next_pow_two(x: int) -> int:
    return 1 << ((x-1).bit_length())

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    size = _next_pow_two(2*n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = _next_pow_two(2*n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x>=0, sum x = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _softmax_grad(a: np.ndarray, beta: float):
    """Gradient of soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = _next_pow_two(2*n - 1)
    # convolution
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    # soft‑max probabilities
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    # gradient = 2 * (a_rev convolve p)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, b

def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   beta_start: float = 0.5,
                   beta_end: float = 1e6,
                   beta_steps: int = 200,
                   lr_base: float = 1.0,
                   eps: float = 1e-8):
    """
    ADAM with projected simplex and a schedule for the temperature β.
    Returns the best sequence seen and its score.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start
    it = 0
    best_a = a.copy()
    best_val = _score(a)
    # pre‑compute growth factor for β
    if beta_steps > 0:
        beta_factor = (beta_end / beta_start) ** (1.0 / beta_steps)
    else:
        beta_factor = 1.0

    beta1 = 0.9
    beta2 = 0.999

    while time.time() < deadline:
        grad, _ = _softmax_grad(a, beta)
        it += 1
        # ADAM moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size scaled to target_sum and current β
        step = lr_base * target_sum / (np.sqrt(v_hat) + eps) / (beta + 1.0)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # update β periodically
        if it % beta_steps == 0 and beta < beta_end:
            beta = min(beta * beta_factor, beta_end)

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ADAM it={it:5d} β={beta:6.1f} score={best_val:.8f}")

    return best_a, best_val

def _mass_transfer_optimize(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           top_k: int = 8,
                           low_frac: float = 0.05,
                           start_temp: float = 0.5,
                           end_temp: float = 1e-4):
    """
    Greedy mass‑transfer moves that look at the K largest convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # temperature schedule (simulated annealing)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # compute full convolution once per iteration
        conv = _conv_fft(a)

        # pick top K indices (largest values)
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # np.argpartition returns unordered top K
            top_idx = np.argpartition(conv, -top_k)[-top_k:]

        # compute contribution weights
        w = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            w[idx] += a[idx] + a[k - idx]

        donor = int(np.argmax(w))
        # choose receiver among low‑weight indices
        low_cnt = max(1, int(len(w) * low_frac))
        low_idxs = np.argsort(w)[:low_cnt]
        receiver = int(rng.choice(low_idxs))

        if donor == receiver:
            continue

        # bound the amount we move
        max_delta = a[donor] * 0.5
        weight_diff = w[donor] - w[receiver]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.8f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to avoid dead‑ends
        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate a_i = (i+1)^(-p) and scale to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n: int, c: float, target_sum: float) -> np.ndarray:
    """Generate a_i = exp(-c*i) and scale to target_sum."""
    a = np.exp(-c * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a different length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

"""
Improved optimizer for the step‑function benchmark.

The algorithm proceeds in three phases:

1. **Projected ADAM on a soft‑max approximation of the max
   self‑convolution** – β (the soft‑max temperature) is increased from a
   small value up to ~10⁶, while the iterates are projected onto the simplex
   {x≥0, Σx = √(2·n)}.

2. **Deterministic mass‑transfer refinement** – the K largest convolution
   entries are identified; a weight w_i that measures how much a_i
   participates in those entries is built and a small amount of mass is
   transferred from a high‑weight index to a low‑weight one.  The moves are
   accepted with a Metropolis rule whose temperature decays to 10⁻⁴.

3. **Final high‑β ADAM polish** – a short ADAM run with β already very
   large (≈10⁶) cleans up any remaining irregularities.

All functions are top‑level, use only NumPy and the standard library, and
respect the 1000 s time budget.  The returned list is trimmed of trailing
zeros and is ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective (equivalent to the maximum convolution)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of a soft‑max approximation of max(conv)
# ----------------------------------------------------------------------


def _softmax_grad(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where grad is the gradient of
    log‑sum‑exp_{β}(a*a) and conv is the exact convolution a*a.
    """
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)

    # exact self‑convolution
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    p = exps / exps.sum()

    # gradient = 2 * (a_rev convolve p)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    return grad, conv


# ----------------------------------------------------------------------
# ADAM optimizer with simplex projection
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.5,
    beta_end: float = 1e6,
    beta_steps: int = 200,
    lr_base: float = 1.0,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected ADAM on the soft‑max objective.
    Returns the best sequence seen before the deadline and its score.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    n = a.shape[0]

    # ADAM state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # β schedule
    beta = beta_start
    if beta_steps > 0:
        beta_factor = (beta_end / beta_start) ** (1.0 / beta_steps)
    else:
        beta_factor = 1.0

    beta1, beta2 = 0.9, 0.999
    it = 0
    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        grad, _ = _softmax_grad(a, beta)
        it += 1

        # ADAM moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate – scaled by the current β
        step = lr_base * target_sum / (np.sqrt(v_hat) + eps) / (beta + 1.0)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # increase β periodically
        if beta_steps > 0 and it % beta_steps == 0 and beta < beta_end:
            beta = min(beta * beta_factor, beta_end)

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ADAM it={it:5d} β={beta:6.1f} score={best_val:.8f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (focus on top‑K conv entries)
# ----------------------------------------------------------------------


def _mass_transfer_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    top_k: int = 8,
    low_frac: float = 0.05,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that attack the K largest convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # temperature schedule (simulated annealing)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # full convolution
        conv = _conv_fft(a)

        # indices of the K largest entries
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -top_k)[-top_k:]

        # build a weight vector measuring participation in those entries
        w = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            w[idx] += a[idx] + a[k - idx]

        donor = int(np.argmax(w))
        low_cnt = max(1, int(len(w) * low_frac))
        low_idxs = np.argsort(w)[:low_cnt]
        receiver = int(rng.choice(low_idxs))

        if donor == receiver:
            it += 1
            continue

        # bound the transferred mass
        max_delta = a[donor] * 0.5
        weight_diff = w[donor] - w[receiver]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.8f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to break stagnation
        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^{-p} normalized to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, c: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-c·i) normalized to target_sum."""
    a = np.exp(-c * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low benchmark score.
    The algorithm consists of three phases:
      1. ADAM on a soft‑max max‑conv surrogate.
      2. Deterministic top‑K mass transfer.
      3. High‑β ADAM polish.
    """
    start = time.time()
    deadline = start + budget_s - 0.2  # safety margin

    # -------------------- problem size --------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    # -------------------- generate diverse starts --------------------
    init_seqs = []

    # power‑law family
    for p in (0.35, 0.40, 0.45, 0.5, 0.55, 0.6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # exponential family
    for c in (0.0005, 0.001, 0.002):
        init_seqs.append(_exponential_sequence(n, c, target_sum))

    # previous best (if provided by the harness)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies for extra diversity
    rng = np.random.default_rng()
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # -------------------- Phase 1: ADAM with restarts --------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.35
    best_seq = None
    best_score = np.inf

    while time.time() < phase1_deadline:
        # pick a random starter and jitter it a little
        starter = rng.choice(init_seqs)
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_seq = np.maximum(starter + jitter, 0.0)
        start_seq = _project_to_simplex(start_seq, target_sum)

        remaining = phase1_deadline - time.time()
        run_budget = min(30.0, remaining)  # at most 30 s per ADAM run
        if run_budget < 5.0:
            break

        cand_seq, cand_score = _adam_optimize(
            start_seq,
            target_sum,
            time.time() + run_budget,
            beta_start=0.5,
            beta_end=5e5,
            beta_steps=250,
            lr_base=0.9,
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] Global best after ADAM = {best_score:.8f}")

    # fallback (should never happen)
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.5, target_sum)
        best_score = _score(best_seq)

    # -------------------- Phase 2: Mass‑transfer refinement --------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + (deadline - time.time()) * 0.45
        a_ref, score_ref = _mass_transfer_optimize(
            best_seq,
            target_sum,
            phase2_deadline,
            top_k=8,
            low_frac=0.08,
            start_temp=0.5,
            end_temp=1e-4,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best={best_score:.8f}")

    # -------------------- Phase 3: Final high‑β ADAM polish --------------------
    if time.time() + 2.0 < deadline:
        a_fin, fin_score = _adam_optimize(
            best_seq,
            target_sum,
            deadline,
            beta_start=1e4,
            beta_end=2e6,
            beta_steps=150,
            lr_base=0.6,
        )
        if fin_score < best_score:
            best_score = fin_score
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.8f}")

    # -------------------- post‑processing --------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional sanity check (the harness may already call evaluate_sequence)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.8f}")
    except Exception:
        pass

    return final_seq
