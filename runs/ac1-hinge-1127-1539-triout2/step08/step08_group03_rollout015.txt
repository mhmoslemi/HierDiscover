"""
Improved step‑function optimizer (≈ 1000 s budget).

Algorithm
---------
1.  Random exploration – draw many candidates from a variety of
    parametric families (Dirichlet‑like, Beta, exponential‑decay,
    power‑law, sparse bumps).  All vectors are scaled to the
    ℓ₁‑norm S = √(2 n) (the evaluation then equals the peak
    of the self‑convolution).  The best *k* candidates are retained.

2.  Projected sub‑gradient descent – for each retained candidate we
    enlarge it (linear interpolation) to a large target length
    (up to 80 000).  At each iteration we locate the shift with the
    current maximal convolution value, compute the exact sub‑gradient
    of this peak (g_i = 2·a_{k‑i} when the index exists), take a
    diminishing step and re‑project onto the simplex {a≥0, Σa=S}.
    The best vector seen during the run is kept.

3.  Optional Adam refinement – a short Adam run with a τ‑soft‑max
    surrogate (identical to the previous generation) may be applied
    if time permits.

4.  Cheap simulated‑annealing “mass‑transfer’’ walk over the simplex
    finishes the search.

The global variable ``height_sequence_1`` stores the final sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, x ≥ 0, Σx = s)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # shift to make all entries non‑negative before sorting
    # (the algorithm works for any real v)
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)                # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Interpolation / scaling of sequences
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `seq` to length `new_n` by linear interpolation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.2, 5.0)
    b = rng.uniform(0.2, 5.0)
    return rng.beta(a, b, size=n)


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.uniform(0.001, 0.02)
    idx = np.arange(n)
    return np.exp(-lam * idx)


def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.3, 3.0)
    idx = np.arange(n)
    offset = rng.uniform(0.0, 10.0)
    return (idx + offset) ** (-p)


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length `n` with ℓ₁‑norm √(2 n).
    Occasionally re‑uses the best known construction (`warm_seq`);
    otherwise draws from one of the random families.
    """
    S = np.sqrt(2.0 * n)

    # occasional warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'beta', 'exponential',
                        'powerlaw', 'sparse'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'beta':
        raw = _init_beta(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    else:
        raw = _init_sparse_bumps(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the peak of the self‑convolution
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          step0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Minimise `max(a*a)` under the ℓ₁‑norm constraint.
    Returns the best vector seen and its (hard) peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # full convolution
        b = _full_convolution_fft(a)
        k = int(np.argmax(b))          # shift where the peak occurs
        peak = b[k]
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # sub‑gradient of the hard max
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing step size (1/√t)
        eta = step0 / np.sqrt(it)
        a_new = a - eta * g

        # project back onto the simplex {a≥0, Σa=S}
        a = _simplex_projection(a_new, S)

        # occasional random jitter to escape plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=0.005 * S, size=n)
            a = _simplex_projection(a + jitter, S)

    return best_a, best_peak

# Adam optimiser (soft‑max surrogate) – kept for extra polishing
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """Adam optimisation of the exact max‑convolution (as in the
       previous generation).  Returns the best vector and its hard peak."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        g, peak = _softmax_gradient(a, tau)
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        tau *= tau_decay
        lr *= lr_decay

        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Cheap simulated‑annealing fine‑tuner (mass‑transfer walk)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple mass‑transfer walk on the simplex with Metropolis acceptance."""
    n = seq.shape[0]
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The final construction is stored in the global variable ``height_sequence_1``
    and also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – random exploration (≈ 40 % of time)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 400.0          # keep ~400 s for refinement & anneal
    top_k = 12
    top_candidates = []   # list of (peak, seq)
    attempts = 0
    best_seq = None
    best_peak = np.inf

    while time.time() < explore_deadline:
        attempts += 1
        # pick a length – a mix of medium and long supports
        if rng.random() < 0.55:
            n = int(rng.integers(2000, 6000))
        else:
            n = int(rng.integers(6000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_peak:
            best_peak = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] #{attempts:5d} n={n:5d}  new best={best_peak:.6f}")

        # maintain a sorted list of the best candidates
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Ensure we have at least one candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_peak = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_peak, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on enlarged supports
    # ------------------------------------------------------------------
    # Target length schedule – go to a relatively large support
    max_len = min(80000, max(best_n, 20000) * 4)
    lengths = [best_n]
    # geometric growth up to max_len
    cur_len = best_n
    while cur_len < max_len:
        cur_len = int(cur_len * 1.75)
        if cur_len > max_len:
            cur_len = max_len
        lengths.append(cur_len)
    lengths = sorted(set(lengths))

    # Run refinement for each stored candidate (most promising first)
    for idx, (init_peak, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_n = len(cur_seq)

        for target_n in lengths:
            if time.time() >= end_time - 120.0:   # reserve ~2 min for annealing
                break

            # enlarge / interpolate to the target length and renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            S = np.sqrt(2.0 * target_n)
            a_scaled = a_scaled * (S / a_scaled.sum())

            # allocate a time slice (max 180 s, but usually far less)
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.45, 180.0)

            # sub‑gradient descent
            refined, peak = _subgradient_optimize(
                a_scaled,
                time.time() + slice_budget,
                rng,
                step0=0.9)

            # occasional Adam polishing if time permits
            if time.time() + 20.0 < end_time:
                refined, peak = _adam_optimize(
                    refined,
                    time.time() + 12.0,
                    rng,
                    lr0=rng.choice([0.38, 0.44, 0.52]),
                    tau_decay=rng.choice([0.9992, 0.9995, 0.9997]),
                    lr_decay=0.99985,
                    jitter_every=600)

            if peak < best_peak:
                best_peak = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] cand#{idx+1} → len={target_n:5d}  peak={best_peak:.6f}")

            cur_seq = refined
            cur_n = target_n

        if time.time() >= end_time - 120.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning if time permits
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 30.0:
        final_seq, final_peak = _anneal(best_seq,
                                        time.time() + remaining,
                                        rng)
        if final_peak < best_peak:
            best_peak = final_peak
            best_seq = final_seq.copy()
            print(f"[anneal] final peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_peak:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
