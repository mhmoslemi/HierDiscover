"""
Search for a step‑function minimizing evaluate_sequence.
Uses:
  – diverse random parametric families (including a two‑parameter Beta shape)
  – aggressive Adam optimisation on the τ‑soft‑max of the self‑convolution
  – geometric scaling to longer lengths (up to ~30 k)
  – multi‑restart Adam on each length
  – final stochastic hill‑climb.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using real FFT (zero‑padded to next power of two)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv

def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()

def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves ordering)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iter: int = None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the probability simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution via a τ‑soft‑max surrogate with τ→0.
    Returns the best sequence found (as a NumPy array) and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁ norm

    # initialise logits from feasible a0 (avoid 0 entries)
    eps = 1e-14
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.55                           # slightly higher initial learning rate

    # temperature start – proportional to current max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0) * 0.5     # start a bit softer

    best_a = a0.copy()
    best_peak = b0.max()

    it = 0
    while time.time() < deadline:
        it += 1
        if max_iter is not None and it > max_iter:
            break

        # recover a from logits (log‑softmax) – guarantees positivity & sum=S
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient of τ‑soft‑max
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient wrt logits (chain rule)
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # clip extremely large gradients for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 8.0 * S:
            grad_p = grad_p * (8.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules – very gentle decay
        tau *= 0.9997
        lr *= 0.9999

        # occasional jitter to avoid stagnation
        if it % 1200 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw

def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw

def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decay from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw

def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw

def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start (will be normalised later)."""
    return np.full(n, 1.0)

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like random start."""
    raw = rng.random(n)
    return raw

def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shape – large at the edges, suppressed in the centre."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw

def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Beta‑type shape: a_i ∝ (i+1)^{α‑1} (n‑i)^{β‑1}.
    Allows asymmetric edge‑heavy or centre‑heavy profiles.
    """
    i = np.arange(n)
    # draw exponents in log‑space to cover wide range
    log_alpha = rng.uniform(-2.0, 2.0)   # α in [~0.01, ~100]
    log_beta = rng.uniform(-2.0, 2.0)
    alpha = 10 ** log_alpha
    beta = 10 ** log_beta
    # protect against overflow: work in log domain
    log_raw = (alpha - 1) * np.log(i + 1.0) + (beta - 1) * np.log(n - i + 0.0)
    # subtract max to avoid overflow, then exponentiate
    log_raw -= np.max(log_raw)
    raw = np.exp(log_raw)
    # mix with a small uniform component to keep numerical stability
    mix = rng.uniform(0.0, 0.5)
    raw = mix * raw + (1.0 - mix) * np.ones(n)
    return raw

def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ~0.25 we warm‑start from the global best.
    Otherwise we draw from a randomly chosen parametric family.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.25:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            a = a * (S / total)
            return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'ushape', 'beta'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'beta':
        raw = _init_beta(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The procedure consists of:
      1) Random exploration over many parametric families (incl. a Beta family).
      2) Geometric scaling to larger lengths (up to ~30 k) and multiple
         Adam refinements with independent restarts.
      3) A long final Adam optimisation on the best length.
      4) A cheap stochastic hill‑climb to squeeze out last gains.
    The final sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # keep a few seconds for final sanity

    # ------------------------------------------------------------------
    # Warm‑start from previously found construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 15 % of total budget)
    # ------------------------------------------------------------------
    explore_deadline = deadline - 300.0   # retain at least ~5 min for refined work
    while time.time() < explore_deadline:
        attempts += 1
        # Choose length; bias towards a few thousand entries but allow up to 30k
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 8000))
        else:
            n = int(rng.integers(8000, 30000))
        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & multi‑restart Adam on a ladder of lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of moderate size
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # build geometric ladder of lengths
    factors = [1.0, 1.3, 1.6, 2.0, 2.8, 4.0, 6.0, 9.0]
    lengths = []
    for f in factors:
        cand = int(best_n * f)
        if 2000 <= cand <= 30000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= deadline - 80.0:   # leave time for final phase & hill‑climb
            break
        # allocate a slice of the remaining time (but no more than 180 s)
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.12, 180.0)

        # start from scaled version of current best
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

        # run a few independent Adam restarts on this length
        restarts = 3
        best_local_seq = a_scaled
        best_local_peak = _max_convolution(a_scaled)
        for r in range(restarts):
            # give each restart a fair share of the slice_time
            sub_deadline = time.time() + slice_time / restarts
            a_opt, peak = _adam_optimize(best_local_seq, sub_deadline, rng)
            if peak < best_local_peak:
                best_local_peak = peak
                best_local_seq = a_opt.copy()
        if best_local_peak < best_score:
            best_score = best_local_peak
            best_seq = best_local_seq.copy()
            best_n = target_n
            print(f"[scale‑refine] n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deep Adam optimisation on the current best length
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        remaining = deadline - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final‑adam] length={best_n:5d}  best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------

"""
Search for a step‑function minimizing the provided ``evaluate_sequence``.

Algorithm
---------
1. **Random exploration** – many candidates are drawn from a collection of
   parametric families (power‑law, exponential, linear, gamma, uniform,
   Dirichlet, U‑shape and a new asymmetric *Beta* family).  The best
   candidate (lowest hard max of the self‑convolution) is kept.
2. **Geometric scaling & multi‑restart Adam** – the current best vector is
   interpolated to a ladder of larger lengths (up to ≈30 k).  For each new
   length several independent Adam runs (log‑soft‑max surrogate of the
   max‑convolution) are performed and the best result kept.
3. **Long Adam refinement** – the currently best length receives a final
   prolonged Adam optimisation.
4. **Stochastic hill‑climb** – a cheap random mass‑transfer loop makes tiny
   local improvements.
5. The final sequence is stored in the global ``height_sequence_1`` and
   returned.

The implementation respects the 1000 s budget, uses only NumPy and
no external files, and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves ordering)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (smooth approximation of max(b) with b = a*a)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a and
    max_b = max(b).  ``b`` is the full self‑convolution of ``a``.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iter: int = None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation that minimises the hard max of the self‑convolution.
    The optimiser works in log‑softmax space, guaranteeing non‑negativity
    and exact ℓ₁‑normalisation (∑a = √(2 n)).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁ norm

    # initialise logits (log‑softmax) from the feasible start
    eps = 1e-14
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.55                     # a slightly higher start than before

    # initialise temperature (τ) from the current max value
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0) * 0.5

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1
        if max_iter is not None and it > max_iter:
            break

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of τ‑soft‑max of the convolution
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to obtain gradient wrt logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clip extreme gradients for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 8.0 * S:
            grad_p = grad_p * (8.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # gentle schedule decay
        tau *= 0.9997
        lr *= 0.9999

        # occasional jitter to leave plateaus
        if it % 1200 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shape – large at the edges, strongly suppressed in the centre."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Asymmetric Beta‑type shape:
        a_i ∝ (i+1)^{α‑1} (n‑i)^{β‑1}
    α,β are drawn in log‑space to span many orders of magnitude.
    """
    i = np.arange(n)
    log_alpha = rng.uniform(-2.0, 2.0)   # α in [10⁻², 10²]
    log_beta = rng.uniform(-2.0, 2.0)
    alpha = 10 ** log_alpha
    beta = 10 ** log_beta

    # work in log‑domain to avoid overflow
    log_raw = (alpha - 1) * np.log(i + 1.0) + (beta - 1) * np.log(n - i + 0.0)
    log_raw -= np.max(log_raw)          # normalise for numerical stability
    raw = np.exp(log_raw)

    # mix a little uniform component to keep the vector well‑behaved
    mix = rng.uniform(0.0, 0.5)
    return mix * raw + (1.0 - mix) * np.ones(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Return a feasible (non‑negative, ℓ₁‑norm = √(2 n)) start vector."""
    S = np.sqrt(2.0 * n)

    # occasional warm‑start from the globally best sequence
    if warm_seq is not None and rng.random() < 0.25:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            return a * (S / total)

    # pick a random parametric family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'ushape', 'beta'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'beta':
        raw = _init_beta(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    return raw * (S / total)

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm follows the phases described in the module docstring.
    The final sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # keep a few seconds for final check

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 15 % of the total budget)
    # ------------------------------------------------------------------
    explore_deadline = deadline - 300.0   # keep at least ~5 min for refinements
    while time.time() < explore_deadline:
        attempts += 1
        # Length bias: mostly 2‑4 k, occasionally up to 30 k
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 8000))
        else:
            n = int(rng.integers(8000, 30000))
        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & multi‑restart Adam on a ladder of lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback uniform vector (should hardly happen)
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # geometric ladder of lengths (up to ≈30 k)
    factors = [1.0, 1.3, 1.6, 2.0, 2.8, 4.0, 6.0, 9.0]
    lengths = []
    for f in factors:
        cand = int(best_n * f)
        if 2000 <= cand <= 30000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= deadline - 80.0:   # reserve time for final phases
            break
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.12, 180.0)   # ≤180 s per length

        # start from the current best, interpolated to the new size
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

        # run a few independent Adam restarts for robustness
        restarts = 3
        best_local_seq = a_scaled
        best_local_peak = _max_convolution(a_scaled)
        for _ in range(restarts):
            sub_deadline = time.time() + slice_time / restarts
            a_opt, peak = _adam_optimize(best_local_seq, sub_deadline, rng)
            if peak < best_local_peak:
                best_local_peak = peak
                best_local_seq = a_opt.copy()

        if best_local_peak < best_score:
            best_score = best_local_peak
            best_seq = best_local_seq.copy()
            best_n = target_n
            print(f"[scale‑refine] n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deep Adam optimisation on the current best length
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        remaining = deadline - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final‑adam] length={best_n:5d}  best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.025, 0.12)    # move a modest fraction of mass
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:

    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = "
          f"{best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
