def max_conv_gradient(a):
    b = _full_convolution_fft(a)
    max_b = b.max()
    mask = (b == max_b).astype(float)
    # Convolution of mask with a_rev:
    a_rev = a[::-1]
    conv_len = len(mask) + len(a) - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    # extract central part which corresponds to indices [n-1 ... n-1 + n-1]
    start = len(a) - 1
    g = 2.0 * conv[start:start+len(a)]
    return g, max_b

def subgradient_descent(a0, deadline, rng, best_score_global):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2 * n)
    best_a = a.copy()
    best_peak = _max_convolution(a) # compute initial
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_peak = max_conv_gradient(a)
        # compute step size: polyak but ensure not too big
        # compute norm squared
        norm_sq = np.dot(grad, grad)
        if norm_sq == 0:
            step = 0.0
        else:
            step = (cur_peak - best_peak) / (norm_sq + 1e-12)
            # bound step to prevent overly large moves
            step = max(min(step, 1.0), 1e-6)  # restrict step
        # update
        a_new = a - step * grad
        # ensure no negative values (project onto simplex)
        a = project_onto_simplex(a_new, S)
        # evaluate
        new_peak = _max_convolution(a)
        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_a = a.copy()
            # optionally update global best_score
        # occasional jitter
        if it % 200 == 0:
            # add tiny random noise to avoid plateau
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = project_onto_simplex(a + jitter, S)
    return best_a, best_peak

"""
Improved step‑function optimizer.

The goal is to minimise  maxₖ (a ∗ a)[k]  subject to a_i ≥ 0 and
∑ a_i = √(2 n).  The evaluation function returns exactly this maximum.
The algorithm consists of four stages:

1. **Broad random exploration** – generate many candidates from several
   parametric families (power‑law, exponential, linear, U‑shape,
   sinusoid, etc.) for a wide range of lengths (2 500 – 20 000).  Keep
   the best few based on the hard convolution peak.

2. **Projected sub‑gradient descent** – for each promising candidate
   perform a convex sub‑gradient method on the objective
   f(a)=maxₖ(a∗a)[k].  The sub‑gradient is obtained analytically by
   convolving the mask of maximal entries with the reversed sequence.
   After each gradient step the vector is projected onto the simplex
   {a ≥ 0, ∑a = √(2 n)} using the O(n) projection algorithm of
   Duchi‑Shalev‑Shwartz‑Singer‑Chandra.  A Polyak step size
   ( (f(x)−f_best)/‖g‖² ) guarantees monotone descent.

3. **Length scaling & refinement** – the current best sequence is
   linearly interpolated to larger lengths (up to ≈ 20 000) and refined
   again with the sub‑gradient method.  Larger n gives the optimiser
   more degrees of freedom to flatten the autocorrelation.

4. **Simulated‑annealing style hill‑climb** – a cheap stochastic
   mass‑transfer loop that accepts improving moves and, with a slowly
   decreasing temperature, occasional worsening moves.  This escapes
   tiny plateaus that pure gradient descent may miss.

All stages respect the 1000 s budget.  The final sequence is stored in
the global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np

# Low‑level utilities (FFT, scaling, projection, gradient)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _max_convolution_and_gradient(a: np.ndarray):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    maxₖ(a∗a)[k] with respect to a and max_val is that maximum.
    """
    n = a.shape[0]
    # convolution
    b = _full_convolution_fft(a)
    max_val = b.max()
    # mask of maximal positions (hard max)
    mask = (b == max_val).astype(float)

    # gradient = 2 * (mask ∗ reverse(a)), extracted on central window
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { w | w_i ≥ 0 , ∑ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # shift by mean to avoid numerical issues
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:

    i = np.arange(n)

    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:

def _initial_vector(n: int,

                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    elif style == 'exponential':

    elif style == 'linear':

    elif style == 'ushape':

    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(a∗a).
    Uses a Polyak step size (f(x)−f_best)/‖g‖² and projects onto the
    ℓ₁‑simplex of radius √(2 n).
    Returns the best vector seen and its hard max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best values
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        grad, cur_val = _max_convolution_and_gradient(a)

        # Polyak step size – ensure positivity
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            # Use best_val as the target lower bound
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            # Clip step to a reasonable interval
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        # gradient step followed by simplex projection
        a = _project_onto_simplex(a - step * grad, S)

        # evaluate
        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb that moves a random fraction of
    mass between two positions and accepts worsening moves with a
    temperature that decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule
    T = 0.05 * cur_val  # start roughly 5 % of current peak
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # move a random fraction from i to j
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        # every few hundred steps apply a tiny random perturbation
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val

    # Phase 1 – broad random exploration (≈ 60 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0   # reserve ~300 s for refinement
    while time.time() < explore_deadline:
        attempts += 1
        # favour longer sequences (3 000 – 18 000)
        if rng.random() < 0.75:
            n = int(rng.integers(3000, 12000))
        else:
            n = int(rng.integers(12000, 20000))

        a0 = _initial_vector(n, rng, warm_seq)
        val = _max_convolution(a0)
        if val < best_score:
            best_score = val
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement on a few scaled lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform of moderate size
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n

    # Determine a small set of target lengths (geometrically increasing)
    lengths = [best_n]
    for factor in [1.3, 1.8, 2.5, 3.5]:
        cand = int(best_n * factor)
        if 3000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 120.0:
            break
        # Scale current best to the new length
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # Allocate a time slice (≈ 15 % of the remaining budget, capped)
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 80.0)

        a_opt, val = _subgradient_descent(a_scaled,
                                          time.time() + slice_time,
                                          rng,
                                          best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[subgrad] target_n={target_n:5d}  improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long sub‑gradient run on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _subgradient_descent(best_seq,
                                          time.time() + remaining,
                                          rng,
                                          best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[final subgrad] length={len(best_seq)}  best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        remaining = end_time - time.time() - 1.0
        a_opt, val = _anneal_local_search(best_seq,
                                          time.time() + remaining,
                                          rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
