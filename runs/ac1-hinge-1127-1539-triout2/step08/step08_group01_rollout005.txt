"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  

The method is a multi‑stage search:

1.  If a previously found construction ``height_sequence_1`` exists,
    use it as a seed (projected onto the ℓ¹‑simplex of the required
    ℓ¹‑norm).

2.  For a series of increasing lengths ``schedule``:
      * up‑sample the current best seed to the new length,
      * run several **Adam** runs on a smooth approximation of the
        max‑convolution (log‑sum‑exp surrogate) – many restarts,
      * improve the best candidate with a custom **simulated‑annealing
        hill‑climb** that updates the full self‑convolution in O(n) per
        move,
      * keep the globally best vector seen so far.

3.  A final polishing phase (long Adam → hill‑climb) uses the remaining
    time budget.

All vectors are projected onto the simplex
``Σ a_i = sqrt(2·n)`` (the same normalisation used by
``evaluate_sequence``), so the objective value equals the maximum entry
of the self‑convolution.  The code respects the wall‑clock limit, prints
progress, and returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation on the smooth‑max surrogate.
    The steps are projected to the ℓ¹‑simplex Σa = target_sum.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # step‑size (scaled to the simplex radius)
    base_lr = 0.25 * target_sum

    # smoothed‑max schedule
    beta = 2.0                     # start with a very smooth approximation
    beta_max = 1e6
    beta_factor = 1.5

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the smooth‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise)
    are sampled.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass move
# ----------------------------------------------------------------------


def _apply_move_incremental(a: np.ndarray, b: np.ndarray,
                           i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a copy of *b* after moving ``delta`` mass from index *i* to *j*.
    The update is performed in O(n) using the formulas from the
    original hill‑climbing implementation.
    """
    n = a.shape[0]
    b_new = b.copy()

    # remove the contribution generated by the source index i
    b_new[i:i + n] -= 2.0 * delta * a

    # add the contribution generated by the destination index j
    b_new[j:j + n] += 2.0 * delta * a

    # correct the three self‑pair terms
    k2i = 2 * i
    if k2i < b_new.size:
        b_new[k2i] += delta * delta
    k2j = 2 * j
    if k2j < b_new.size:
        b_new[k2j] += delta * delta
    kij = i + j
    if kij < b_new.size:
        b_new[kij] -= 2.0 * delta * delta

    return b_new


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (more aggressive than the reference)
# ----------------------------------------------------------------------


def _hill_climb_simulated_annealing(a0: np.ndarray, b0: np.ndarray,
                                    deadline: float,
                                    temp_factor: float = 0.2,
                                    cooling: float = 0.9999,
                                    max_delta_frac: float = 0.5) -> tuple[np.ndarray,
                                                                         np.ndarray]:
    """
    Simulated‑annealing hill‑climb that moves a random chunk of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.

    The move is biased:
        – source index is chosen proportionally to its contribution to
          the currently worst convolution entry,
        – destination index is biased towards positions whose counterpart
          in that worst entry is small.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * temp_factor + 1e-12
    while time.time() < deadline:
        # 1) pick a worst convolution position M
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # 2) source index – weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs_src = contrib / total_contrib
            i = rng.choice(idx, p=probs_src)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # 3) destination index – bias towards small counterpart a[M‑j]
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]
        inv = 1.0 / (counterpart + 1e-12)
        probs_dst = inv / inv.sum()
        j = rng.choice(n, p=probs_dst)

        if i == j:
            continue

        # 4) amount of mass to move
        delta = rng.random() * max_delta_frac * a[i]
        if delta <= 0.0:
            continue

        # 5) compute the new convolution
        b_new = _apply_move_incremental(a, b, i, j, delta)
        new_score = b_new.max()

        # 6) Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            # accept
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()

        # temperature decrease
        T *= cooling

    return best_a, best_b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock limit (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf                     # this stores max_b (objective)

    # ------------------------------------------------------------------
    # 1) seed from a previous construction, if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0})  max_b={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2) Length schedule – larger dimensions get the lion’s share of time
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for the current length
        # --------------------------------------------------------------
        if best_seq is not None:
            # always start from the current best (upsampled if necessary)
            a_init = _upsample_sequence(best_seq, n) if best_seq.size != n else best_seq.copy()
            # add a tiny jitter to avoid staying exactly on a plateau
            a_init += rng.normal(scale=target_sum * 0.0008, size=n)
            a_init = np.maximum(a_init, 0.0)
            a_init = _project_to_simplex(a_init, target_sum)
        else:
            a_init = _random_initial_sequence(n, rng)

        # --------------------------------------------------------------
        # Time allocation for this dimension
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 12.0:          # not enough time for a meaningful run
            break
        gd_budget = remaining * 0.38   # Adam gradient descent
        hc_budget = remaining * 0.38   # hill‑climb
        # (roughly 24 % is left for later dimensions / final polish)

        # --------------------------------------------------------------
        # 2.A – Multiple Adam restarts on the smooth‑max surrogate
        # --------------------------------------------------------------
        n_restarts = 5
        per_restart = gd_budget / n_restarts
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_gradient_descent(start_a, target_sum,
                                           min(deadline - 0.1,
                                               time.time() + per_restart))
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam improvement for n={n}  max_b={best_score:.6f}")

        # --------------------------------------------------------------
        # 2.B – Simulated‑annealing hill climb
        # --------------------------------------------------------------
        if time.time() >= deadline - 5.0:
            break
        # ensure we are working with the current length
        a_cur = best_seq.copy() if best_seq.size == n else _upsample_sequence(best_seq, n)
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        hill_deadline = min(deadline - 0.1, time.time() + hc_budget)
        a_hc, b_hc = _hill_climb_simulated_annealing(a_cur, b_cur, hill_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = score_hc
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n}  max_b={best_score:.6f}")

        # --------------------------------------------------------------
        # 2.C – Optional quick Adam pass after hill‑climb (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            extra_deadline = deadline - 1.0
            a_refined = _adam_gradient_descent(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution_fft(a_refined)
            if score_ref < best_score - 1e-12:
                best_seq = a_refined.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC Adam improvement for n={n}  max_b={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing (use whatever time is left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # a long Adam run
        final_deadline = deadline - 2.0
        best_seq = _adam_gradient_descent(best_seq, target, final_deadline)

        # a short, intensive hill‑climb
        b_final = _full_convolution(best_seq)
        best_seq, b_final = _hill_climb_simulated_annealing(
            best_seq, b_final, deadline - 1.0)

        final_score = b_final.max()
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: max_b={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Clean‑up – drop trailing near‑zeros for a neat result
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Report the official objective value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
