def _adam_gradient_descent(a0, target_sum, deadline, max_beta=5e5, init_lr=0.5, beta_schedule_factor=1.5):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = init_lr * target_sum   # scale
    beta = 0.5
    it = 0
    best_val = _score(a)
    best_a = a.copy()
    while time.time() < deadline:
        grad, b = _gradient_softmax(a, beta)
        it += 1
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # learning rate schedule: step / (sqrt(v_hat) + epsilon)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)
        # update best
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it} β={beta:.1f} score={best_val:.6f}")

        # schedule beta
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * beta_schedule_factor, max_beta)
        # jitter occasionally
        if it % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        # maybe reduce LR slowly
        if it % 1000 == 0:
            step *= 0.9
    return best_a, best_val

def _deterministic_mass_transfer2(a0, deadline, target_sum):
    a = a0.copy()
    n = a.size
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)   # compute conv
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi+1)
        # product contributions
        w = a[idx] * a[kmax - idx]
        # pick donor with largest w and receiver smallest w (maybe restrict to extremes)
        donor = idx[np.argmax(w)]
        receiver = idx[np.argmin(w)]
        # compute delta: limited by donor value and weight diff
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        max_delta = a[donor] * 0.5
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff / (a[kmax - donor] + 1e-12) * 0.5)  # heuristics
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)
        # compute new score
        new_score = _score(a)
        # accept if better else possibly accept with small probability
        if new_score < cur_score or rng.random() < 0.001:  # random occasional acceptance
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT2 improved score={best_score:.6f}")
        else:
            # revert change
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)
        # schedule maybe jitter
        if time.time() - start_time > (deadline - start_time) * 0.8:
            # after certain time reduce transfers
            pass
    return best_a, best_score

"""
Propose a low‑max‑convolution step function.

The algorithm works in three stages:

1. **Initialisation** – generate a diverse pool of candidates
   (power‑law, Gaussian, triangular, a resized previous best and jittered
   variants) and project each onto the simplex {x≥0, Σx = √(2·n)}.

2. **Adam‑based soft‑max gradient descent** – minimise a smooth
   approximation of maxₖ (a*a)ₖ using the soft‑max with an exponentially
   increasing temperature β.  The Adam optimiser supplies an adaptive
   step size; after each update the vector is projected back onto the
   simplex.

3. **Deterministic mass‑transfer refinement** – while the current maximal
   convolution entry k\* is known, move mass from an index that contributes
   strongly to (a*a)_{k\*} (large product a_i·a_{k\*-i}) to an index with a
   weak contribution.  Moves are accepted with a Metropolis rule whose
   temperature decays exponentially.  The best solution found is returned.

All heavy work (convolutions, gradients) uses FFT‑based O(n log n)
operations.  The time budget (default 1000 s) is split roughly 45 % for
gradient descent, 35 % for mass‑transfer and the remaining time for a final
polishing pass.  The code respects the required API
`propose_candidate(budget_s: int) -> list[float]`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective 2·n·max(a*a) / (∑a)².
    For ∑a = √(2·n) this equals max(a*a), but we keep the full expression.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^{-p} shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, target_sum: float) -> np.ndarray:
    """Centered Gaussian, normalised to target_sum."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = n / 8.0
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Triangle (peak in the centre), normalised to target_sum."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(idx - centre) / centre
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    # stable soft‑max
    scaled = beta * (conv - max_b)
    scaled = np.clip(scaled, -700, 700)           # avoid overflow
    exps = np.exp(scaled)
    p = exps / np.sum(exps)

    # gradient: 2 * (reverse(a) ⋆ p)
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    grad_rev = np.fft.irfft(ar_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * grad_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           max_beta: float = 5e5,
                           beta_factor: float = 1.5,
                           beta_update_freq: int = 200,
                           base_lr_factor: float = 0.1,
                           jitter_freq: int = 300,
                           epsilon: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex.
    The temperature β of the soft‑max increases exponentially.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.size

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    it = 0
    beta = 0.5
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    # step size before the β‑scaling
    base_lr = base_lr_factor * target_sum

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        it += 1

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate decays with β
        lr = base_lr / (beta + 1.0)

        # Adam update
        a -= lr * (m_hat / (np.sqrt(v_hat) + epsilon))
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % jitter_freq == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # β schedule
        if it % beta_update_freq == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

        # keep the best solution found so far
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (product‑based)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_product(a0: np.ndarray,
                                        deadline: float,
                                        target_sum: float,
                                        start_temp: float = 0.5,
                                        end_temp: float = 1e-4,
                                        low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly target the current maximal
    convolution entry.  The contribution w_i = a_i·a_{k*‑i} is used (instead
    of the sum used previously).  Moves are accepted with a Metropolis rule
    whose temperature decays exponentially with the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau

        # current maximal convolution entry
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # indices that appear in the sum defining conv[kmax]
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        opp = kmax - idx

        # product contributions
        w = a[idx] * a[opp]

        donor = idx[np.argmax(w)]
        # choose a receiver among the low‑contributing indices
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        # bound the amount of mass to move
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            opp_donor = opp[np.argmax(w)]
            max_delta = min(max_delta,
                            weight_diff / (a[opp_donor] + 1e-12) * 0.5)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # propose move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT‑product new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake a possible plateau
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Initial candidate pool
# ----------------------------------------------------------------------


def _initial_candidates(n: int, target_sum: float):
    """Build a diverse list of starting points."""
    seqs = []

    # power‑law family
    for p in (0.45, 0.5, 0.55, 0.6, 0.65):
        seqs.append(_power_law_sequence(n, p, target_sum))

    # Gaussian and triangular shapes
    seqs.append(_gaussian_sequence(n, target_sum))
    seqs.append(_triangular_sequence(n, target_sum))

    # previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seqs.append(_resize_sequence(prev, n))

    # jittered variants for extra diversity
    rng = np.random.default_rng()
    jittered = []
    for s in seqs:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        s_jit = s + jitter
        s_jit = np.maximum(s_jit, 0.0)
        s_jit = _project_to_simplex(s_jit, target_sum)
        jittered.append(s_jit)
    seqs.extend(jittered)

    return seqs


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm combines: (i) Adam‑based soft‑max gradient descent,
    (ii) deterministic product‑based mass transfer, and (iii) a final
    polishing pass.  It respects the supplied time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size depending on the available budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 24000
    elif budget_s >= 500:
        n = 18000
    elif budget_s >= 300:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build initial pool
    # ------------------------------------------------------------------
    candidates = _initial_candidates(n, target_sum)

    # ------------------------------------------------------------------
    # Phase A – Adam gradient descent on soft‑max surrogate
    # ------------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.45
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # allocate roughly equal time slices to the candidates (but stop early)
    for init in candidates:
        now = time.time()
        if now >= phase_a_deadline - 5.0:
            break
        # give each run at most 30 s (or the remaining phase‑A time)
        remaining = phase_a_deadline - now
        run_deadline = now + min(30.0, remaining)
        a, val = _adam_gradient_descent(
            init,
            target_sum,
            run_deadline,
            max_beta=5e5,
            beta_factor=1.7,
            base_lr_factor=0.08,
        )
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New global best after Phase A = {best_score:.6f}")

    # fallback in the unlikely case that no candidate succeeded
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase B – Deterministic product‑based mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer_product(
            best_seq,
            deadline,
            target_sum,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.07,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After Phase B best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Final polishing (high β)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            target_sum,
            deadline,
            max_beta=1e6,
            beta_factor=1.5,
            base_lr_factor=0.05,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
