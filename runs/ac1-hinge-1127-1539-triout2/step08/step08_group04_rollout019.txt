"""
Propose a step‑function using a symmetric‑product parametric family, 
followed by Adam‑style gradient descent on a excess‑loss, then
mass‑transfer refinement.  Aims to drive the maximum self‑convolution
below 1.503 within the 1000 s budget.
"""

import time
import numpy as np

def _conv_full(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a):
    conv = _conv_full(a)
    return float(conv.max())

def _score(a):
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v, s):
    # Euclidean projection onto simplex {x>=0, sum=x = s}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    rho = np.nonzero(rho_arr)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _sym_product_shape(n, p, lam):
    # a_i ∝ ((i+1)*(n-i))^{-p} * exp(-lam * ((i- (n-1)/2)**2) / n**2)
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    base = prod ** (-p)
    if lam != 0.0:
        mid = (n - 1) / 2.0
        # Gaussian envelope centred, width ~ n / sqrt(lam) (if lam large, narrow)
        env = np.exp(-lam * ((i - mid) ** 2) / (n ** 2))
        base *= env
    return base

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _gradient_excess(a, avg):
    # gradient of loss = sum_{k} max(0, b_k-avg)^2
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    excess = np.maximum(b - avg, 0.0)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    excess_fft = np.fft.rfft(excess, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * excess_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _adam_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    avg = target_sum * target_sum / (2 * n - 1)   # average convolution value
    best_a = a.copy()
    best_val = _score(a)
    print(f"[{time.time():.2f}] Adam start score={best_val:.6f}")

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 1
    beta_temp = 1.0
    max_beta = 16384.0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute gradient of excess loss (more direct)
        grad_ex, _ = _gradient_excess(a, avg)
        # optionally mix with softmax gradient for robustness
        grad_soft, _ = _gradient_softmax(a, beta_temp)
        grad = grad_ex + 0.1 * grad_soft   # small contribution from softmax

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = 0.5 * target_sum / (np.sqrt(v_hat) + eps) * (1.0 / (beta_temp + 1.0))

        a = a - lr * m_hat
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it} beta={beta_temp:.1f} score={best_val:.6f}")

        # schedule beta increase (temperature for softmax part)
        if it % 300 == 0 and beta_temp < max_beta:
            beta_temp = min(beta_temp * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _mass_transfer_refine(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    conv = _conv_full(a)
    while time.time() < deadline:
        # index of current maximal convolution entry
        k_max = int(np.argmax(conv))

        # sensitivity vector for this k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = a[k_max - idx]     # shape matches idx

        weight = a[idx] * sens
        total_w = weight.sum()
        if total_w > 0:
            donor = rng.choice(idx, p=weight / total_w)
        else:
            donor = rng.integers(n)

        # choose a receiver among indices with low weight
        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_frac = 0.02
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        # apply move with O(n) slice updates
        # store original a slice for revert if needed
        a[donor] -= delta
        a[receiver] += delta

        # update convolution efficiently using slice operations
        # original a before move (we saved a_old copy?) we need original a for shift contributions.
        # Simpler: recompute conv fully (costly) but we only do this after accepted moves,
        # acceptance is based on new score, we compute new conv anyway.
        conv = _conv_full(a)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_full(a)

        # renormalise periodically
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

def _parametric_search(n, target_sum, deadline):
    # search both asymmetric power‑law and symmetric product families
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()
    # Grid for power law + exponential (as before)
    p_vals = np.linspace(0.2, 1.2, 15)
    lam_vals = np.linspace(0.0, 8.0, 15)
    i = np.arange(n, dtype=np.float64)

"""
Propose a step‑function using a symmetric‑product parametric family,
followed by Adam‑style excess‑loss descent and a deterministic mass‑transfer
refinement.  The aim is to drive the benchmark score (the maximal
self‑convolution) well below 1.503 within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_full(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    rho = np.nonzero(rho_arr)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _sym_product_shape(n: int, p: float, lam: float) -> np.ndarray:
    """
    Symmetric product shape:
        a_i ∝ ((i+1)*(n-i))^{-p} · exp(-lam·((i-mid)²)/n²)
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)                     # (i+1)*(n-i) > 0
    base = prod ** (-p)
    if lam != 0.0:
        mid = (n - 1) / 2.0
        env = np.exp(-lam * ((i - mid) ** 2) / (n ** 2))
        base *= env
    return base


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max surrogate of max(convolution).
    Returns (grad, b) where b is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_excess(a: np.ndarray, avg: float):
    """
    Gradient of ∑_k max(0, b_k‑avg)²,
    where b = a*a (full convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    excess = np.maximum(b - avg, 0.0)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    excess_fft = np.fft.rfft(excess, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * excess_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Adam‑style descent on the excess loss

def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the excess‑loss (plus a tiny soft‑max term).
    Keeps the iterates on the ℓ¹‑simplex.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    avg = target_sum * target_sum / (2 * n - 1)      # average entry of the convolution
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 1
    beta_temp = 1.0
    max_beta = 16384.0
    rng = np.random.default_rng()

    print(f"[{time.time():.2f}] Adam start score={best_val:.6f}")

    while time.time() < deadline:
        # two complementary gradients
        grad_ex, _ = _gradient_excess(a, avg)
        grad_soft, _ = _gradient_softmax(a, beta_temp)
        grad = grad_ex + 0.1 * grad_soft                 # soft‑max stabilises early steps

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step size – scaled by the target ℓ¹‑norm for robustness
        lr = 0.5 * target_sum / (np.sqrt(v_hat) + eps) * (1.0 / (beta_temp + 1.0))

        a = a - lr * m_hat
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta_temp:6.1f} score={best_val:.6f}")

        # increase the soft‑max inverse temperature slowly
        if it % 300 == 0 and beta_temp < max_beta:
            beta_temp = min(beta_temp * 2.0, max_beta)

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a modest fraction of mass from indices that
    heavily contribute to the current maximal convolution entry to
    indices that contribute little, keeping the best sequence found.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    n = a.shape[0]
    rng = np.random.default_rng()
    conv = _conv_full(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = a[k_max - idx]                     # a_{k‑i}
        weight = a[idx] * sens
        total_w = weight.sum()
        if total_w > 0:
            donor = rng.choice(idx, p=weight / total_w)
        else:
            donor = rng.integers(n)

        # receiver among indices that currently have low contribution
        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_frac = 0.02
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        # recompute full convolution (costly but safe)
        conv = _conv_full(a)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # reject – revert the move
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_full(a)

        # renormalise occasionally (numerical drift)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val
