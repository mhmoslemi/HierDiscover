"""
search_algorithm_v2.py

A long‑running, multi‑phase optimizer for the step‑function benchmark.

Key ideas
---------
1. **Problem size** – choose n≈12000 for a full budget.
2. **Rich initial pool** – pure power‑law sequences, a *piecewise* power‑law
   (two different exponents) and jittered variants of the previous best.
3. **Phase A – aggressive projected sub‑gradient descent**  
   A momentum based descent on the soft‑max approximation of the maximal
   convolution entry.  The temperature (β) is doubled every few hundred
   iterations and allowed to reach 5·10⁵, i.e. the gradient becomes almost
   the exact sub‑gradient of the max‑norm.  Most of the total time budget
   (≈80 %) is spent here with many restarts and small random perturbations.
4. **Phase B – deterministic mass‑transfer refinement**  
   Same deterministic “donor → receiver” moves as the reference implementation,
   but with a slower exponential temperature schedule and a larger number of
   iterations (the whole remaining budget after Phase A).
5. **Phase C – final polish** – a short high‑β descent (β up to 2·10⁶) to
   clean up any residual spikes.

All heavy linear algebra uses FFT‑based convolutions (O(n log n)) and the
Euclidean simplex projection (O(n log n)).  The routine respects the
provided time limit; if the deadline is missed the harness will give
negative points.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)² (reduces to the max convolution)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _piecewise_power_sequence(
    n: int,
    p1: float,
    p2: float,
    split_frac: float,
    target_sum: float,
) -> np.ndarray:
    """
    Two‑segment power law:
        a_i = C * (i+1)^(-p1)                for i < m
        a_i = C * m**(p2 - p1) * (i+1)^(-p2) for i ≥ m
    where m = round(split_frac * n).  The whole vector is normalised to
    target_sum.
    """
    if not (0.0 < split_frac < 1.0):
        split_frac = 0.5
    m = max(1, min(n - 1, int(round(split_frac * n))))

    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)

    # first segment
    a[:m] = idx[:m] ** (-p1)

    # second segment – continuity enforced by the factor m**(p2-p1)
    factor = m ** (p2 - p1)
    a[m:] = factor * idx[m:] ** (-p2)

    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used as a smooth sub‑gradient)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.7,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max of the maximal
    convolution entry.  The temperature β doubles periodically and can
    reach `max_beta`, turning the update into an (almost) exact sub‑gradient
    step for the max‑norm.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:8.2f} score={best_val:.6f}")

        # increase temperature slowly
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
    low_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly attack the current maximal
    convolution entry.  Moves are accepted with a Metropolis rule whose
    temperature decays exponentially with the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # transfer amount (bounded)
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm proceeds in three phases:
      * Phase A – long projected gradient descent (soft‑max, high β)
      * Phase B – deterministic mass‑transfer refinement
      * Phase C – short high‑β polishing pass
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------- problem size -------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------- generate initial candidates -------------------
    init_seqs = []

    # pure power‑law seeds
    for p in (0.44, 0.45, 0.46, 0.48, 0.50, 0.52):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # piecewise power‑law seeds (two exponents)
    for split in (0.3, 0.5, 0.7):
        for p1, p2 in ((0.45, 1.2), (0.48, 1.5), (0.50, 2.0)):
            init_seqs.append(_piecewise_power_sequence(n, p1, p2, split, target_sum))

    # previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies to enlarge diversity
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # ------------------- Phase A – long gradient descent -------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.80  # 80 % of budget
    best_seq = None
    best_score = np.inf

    # first run each seed for a modest chunk (≈30 s or less)
    for seed in init_seqs:
        if time.time() >= phase_a_deadline:
            break
        run_deadline = min(phase_a_deadline, time.time() + 30.0)
        cand, val = _annealed_gradient_descent(
            seed,
            run_deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New global best after seed GD = {best_score:.6f}")

    # continue with random restarts around the current best
    while time.time() < phase_a_deadline - 5.0:
        # jitter around the current best
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start = best_seq + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        run_deadline = min(phase_a_deadline, time.time() + 20.0)
        cand, val = _annealed_gradient_descent(
            start,
            run_deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Improved global best (restart) = {best_score:.6f}")

    # --------------------------------------------------------------------
    # Safety fallback (should never happen)
    # --------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.48, target_sum)
        best_score = _score(best_seq)

    # ------------------- Phase B – deterministic refinement -------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            deadline,
            target_sum,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # ------------------- Phase C – final high‑β polishing -------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------- post‑processing ----------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (non‑essential for the harness)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
