"""
Proposes a step‑function (non‑negative sequence) with a low benchmark score.

Algorithm
---------
1. **Choose problem size** – n is selected from the time budget.
2. **Warm‑start** – a fast parametric search on a small problem finds a
   shape a_i ≈ (i+1)^(-p)·exp(-λ·i/small_n) possibly cut off after a
   fraction of the indices.  The best shape is interpolated to the full
   length n and normalised to Σa = √(2n).
3. **Adam soft‑max descent** – minimise a smooth Log‑Sum‑Exp surrogate of
   max(a∗a) with Adam updates and a projection onto the simplex (keeps
   non‑negativity and the ℓ¹‑norm).  The temperature β of the soft‑max
   is doubled regularly to sharpen the objective.
4. **Deterministic mass‑transfer refinement** – repeatedly locate the
   current maximal convolution entry k_max, move a small fraction of mass
   from the index that contributes most to that entry to an index that
   contributes little, keep moves only if they improve the score.
5. **Final clean‑up** – truncate trailing near‑zero entries, evaluate,
   store for warm‑starts and return the list.

All stages respect the constraints required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to ``new_len`` entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to ``target_sum``."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the Log‑Sum‑Exp (soft‑max) surrogate of max(convolution).
    Returns (grad, conv) where ``conv`` is the full self‑convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

    lam_vals = np.linspace(0.0, 20.0, 13)
    cut_facs = np.linspace(0.5, 1.0, 6)

    for p in p_vals:
        for lam in lam_vals:

def _adam_softmax_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max surrogate of max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lr = 0.08
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # temperature schedule
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass away from indices that
    contribute most to the current maximal convolution entry.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # contributions to the maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[idx] = a[idx] * a[k_max - idx]

        donor = int(np.argmax(contrib))
        low_thr = np.percentile(contrib, 5)     # bottom 5 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = int(rng.choice(low_idx))

        if donor == receiver:
            it += 1
            continue

        # transfer a modest fraction (2 % of donor's mass)
        delta = a[donor] * 0.02
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta

        # renormalise (keeps the exact ℓ¹‑norm)
        a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve it={it:5d} score={best_val:.6f}")
        else:
            # reject the move
            a[donor] += delta
            a[receiver] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Searches for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget and returns the best sequence found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (parametric search on a reduced scale)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init.size != n:
            init = _resize_sequence(init, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑start from saved sequence (len={init.size})")
    else:
        small_n = max(2000, n // 4)
        small_target = np.sqrt(2.0 * small_n)
        param_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.10)
        best_small, _ = _parametric_search_small(small_n, small_target, param_deadline)
        if best_small is not None:
            init = _resize_sequence(best_small, n)
            init = init * (target_sum / init.sum())
            print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")
        else:
            init = _power_law_sequence(n, 0.55, target_sum)
            print(f"[{time.time():.2f}] Fallback power‑law start")

    # ------------------------------------------------------------------
    # Phase 1 – Adam soft‑max descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.45
        a_adam, val_adam = _adam_softmax_descent(init, target_sum, adam_deadline)
        best_seq = a_adam
        best_score = val_adam
        print(f"[{time.time():.2f}] After Adam descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _mass_transfer_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After transfer refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Evaluate (may raise if evaluate_sequence is not in the namespace)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
