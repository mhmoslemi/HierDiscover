def _gen_candidate_family(n, rng, base=None):
    f = rng.integers(0, 9)
    if f ==0: # uniform random
        a = rng.random(n)
    elif f==1: # geometric decreasing
        rate = rng.uniform(0.9, 0.9999)
        a = rate ** np.arange(n)
    elif f==2: # geometric increasing
        rate = rng.uniform(0.9, 0.9999)
        a = rate ** np.arange(n-1, -1, -1)
    elif f==3: # linear decreasing
        a = np.linspace(1, 0.1, n)
    elif f==4: # linear increasing
        a = np.linspace(0.1, 1, n)
    elif f==5: # beta-like asym
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0,1,n)
        a = (x**p1) * ((1-x)**p2)
    elif f==6: # random permutation of sorted values
        base_shape = np.exp(-rng.uniform(0.5, 3.0) * np.arange(n))
        a = rng.permutation(base_shape)
    elif f==7: # mixture of two exponentials
        rate1 = rng.uniform(0.9, 0.9999)
        rate2 = rng.uniform(0.9, 0.9999)
        split = rng.integers(1, n-1)
        a = np.concatenate([rate1**np.arange(split), rate2**np.arange(n-split)])
    elif f==8: # from previous best with random jitter
        if base is not None:
            a = base.copy()
            # jitter: add small noise then proj
            noise = rng.normal(scale=0.05, size=n)
            a = np.maximum(a + noise, 0)
        else:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0, None)
    return a

                break
            short_dead = min(deadline, now + 0.07 * (deadline - now))
            a_up_opt, s_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=False)
            if s_up < best_score:
                best_score = s_up
                best_seq = a_up_opt.copy()
                print(f"[upsample] factor={factor:.2f} n={new_n} improved to {best_score:.6f}")
    # Phase 3: Simulated Annealing final local
    now = time.time()
    if best_seq is not None and now < deadline:
        a_final, s_final = _simulated_annealing(best_seq, deadline, rng)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
            print(f"[SA final] improved to {best_score:.6f}")
    # Return result
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[return] external={external:.6f} internal={best_score:.6f} length={len(final_seq)} attempts={attempts}")
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq

def _simulated_annealing(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)
    fft_len = 1 << ((2*n - 1) - 1).bit_length()
    cur_score = _max_convolution(a, fft_len)
    best_a = a.copy()
    best_score = cur_score
    start = time.time()
    total = deadline - start
    # initial temperature as fraction of best_score
    T0 = max(best_score * 0.05, 0.001)
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total
        T = T0 * (1.0 - progress)  # linear cooling
        # choose move type
        if rng.random() < 0.6:
            # transfer mass
            # bias source towards larger values
            # sample i with probability proportional to a_i
            probs = a / a.sum()
            i = rng.choice(n, p=probs)
            # ensure i has enough mass
            if a[i] < 1e-12:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            max_delta = min(a[i], 0.3 * a[i])
            delta = max_delta * rng.random()
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        elif rng.random() < 0.4:
            # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            # random shuffle of a small subset
            idx = rng.choice(n, size=min(5, n), replace=False)
            a_new = a.copy()
            shuffled = rng.permutation(a_new[idx])
            a_new[idx] = shuffled
        # No need to project because sum unchanged; ensure non-negative.
        # small negative due to rounding treat:
        a_new = np.maximum(a_new, 0.0)
        # compute score
        new_score = _max_convolution(a_new, fft_len)
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # occasional large perturbation
        if rng.random() < 0.001:
            # reinitialize a fraction of entries
            mask = rng.random(n) < 0.02
            a[mask] = rng.random(mask.sum())
            a = a / a.sum() * S
            cur_score = _max_convolution(a, fft_len)
    return best_a, best_score

"""
Algorithm: Random exploration of asymmetric families followed by
gradient based refinement (Adam + sub‑gradient) and a final simulated‑annealing
search.  The sequence is always projected onto the simplex
Σa_i = √(2 n) and kept non‑negative.  The goal is to minimise the peak
of the self‑convolution, which equals the objective returned by
`evaluate_sequence` when the simplex constraint holds.
"""
import math, time, numpy as np

# Simplex projection
def _project_simplex(v, s):
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Full convolution (linear) via FFT – also used to get max entry
def _max_convolution(a, fft_len=None):
    n = a.shape[0]
    conv_len = 2 * n - 1
    if fft_len is None:
        fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return float(conv.max())

def _full_convolution(a):
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

# Soft‑max weights for Adam
def _smooth_weights(conv, alpha):
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w

def _conv_with_weights(a, w):
    # gradient of Σ w_k (a*a)_k
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad

def _symmetrize(v):
    return 0.5 * (v + v[::-1])

def _upsample_seq(base, n):
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)

# ----------------------------------------------------------------------
# Candidate families (asymmetric)
# ----------------------------------------------------------------------
def _gen_candidate_family(n, rng, base=None):
    fam = rng.integers(0, 9)
    if fam == 0 and base is not None:
        a = _upsample_seq(base, n)
    elif fam == 1:
        a = rng.random(n)                     # pure uniform random
    elif fam == 2:
        rate = rng.uniform(0.90, 0.9995)
        a = rate ** np.arange(n)              # decreasing geometric
    elif fam == 3:
        rate = rng.uniform(0.90, 0.9995)
        a = rate ** np.arange(n-1, -1, -1)     # increasing geometric
    elif fam == 4:
        a = np.linspace(1.0, 0.1, n)           # linear decreasing
    elif fam == 5:
        a = np.linspace(0.1, 1.0, n)           # linear increasing
    elif fam == 6:
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)        # beta‑like (asymmetric)
    elif fam == 7:
        # mixture of two exponentials with a random split point
        split = rng.integers(1, n-1)
        rate1 = rng.uniform(0.90, 0.9995)
        rate2 = rng.uniform(0.90, 0.9995)
        part1 = rate1 ** np.arange(split)
        part2 = rate2 ** np.arange(n - split)
        a = np.concatenate([part1, part2])
    elif fam == 8:
        # random permutation of a decreasing exponential – encourages asymmetry
        rate = rng.uniform(0.90, 0.9995)
        vals = rate ** np.arange(n)
        rng.shuffle(vals)
        a = vals
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    return a

# ----------------------------------------------------------------------
# Optimisers – same as in the reference implementation, slightly tuned
# ----------------------------------------------------------------------
def _optimize_adam(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.07
    max_alpha = 800.0

    best_a = a.copy()
    best_score = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.9999

    return best_a, best_score

def _optimize_subgrad(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _max_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _max_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random swap / hill‑climb flavour
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _max_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.18:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score

def _local_swap_search(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _max_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _max_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score

# ----------------------------------------------------------------------
# Simulated annealing – pure non‑gradient local search
# ----------------------------------------------------------------------
def _simulated_annealing(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)
    fft_len = 1 << ((2 * n - 1) - 1).bit_length()
    cur_score = _max_convolution(a, fft_len)
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    total = deadline - start
    T0 = max(best_score * 0.04, 0.001)

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total
        T = T0 * (1.0 - progress)

        # move selection
        if rng.random() < 0.6:
            # transfer mass from i → j
            i = rng.integers(0, n)
            if a[i] <= 1e-12:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            max_delta = min(a[i], 0.3 * a[i])
            delta = max_delta * rng.random()
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        elif rng.random() < 0.4:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            # small local shuffle of up to 5 entries
            k = min(5, n)
            idx = rng.choice(n, size=k, replace=False)
            a_new = a.copy()
            a_new[idx] = rng.permutation(a_new[idx])

        a_new = np.maximum(a_new, 0.0)   # safeguard
        # sum stays constant, but round‑off may drift
        a_new = _project_simplex(a_new, S)

        new_score = _max_convolution(a_new, fft_len)

        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # occasional large random perturbation to escape deep traps
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            a[mask] = rng.random(mask.sum())
            a = _project_simplex(a, S)
            cur_score = _max_convolution(a, fft_len)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # previous best (warm start)
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – exploration, keep a few elite candidates
    # ------------------------------------------------------------------
    K = 8
    elite = []
    attempts = 0
    phase0_dead = start_time + 0.12 * budget_s
    while time.time() < phase0_dead:
        attempts += 1
        # Choose length
        if base_seq is not None and rng.random() < 0.5:
            scale = rng.uniform(0.85, 1.20)
            n = max(2000, int(base_seq.shape[0] * scale))
            base = _upsample_seq(base_seq, n)
        else:
            n = int(rng.integers(3000, 15001))
            base = None

        a = _gen_candidate_family(n, rng, base_seq if base is None else base)

        # Normalise to Σa = √(2·n)
        S = math.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # occasional random shuffle for extra asymmetry
        if rng.random() < 0.35:
            rng.shuffle(a)

        cur_score = _max_convolution(a)

        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    if not elite:
        n = 5000
        a = np.full(n, 1.0)
        S = math.sqrt(2.0 * n)
        a = a / a.sum() * S
        elite.append((_max_convolution(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – refinement of each elite (Adam → sub‑grad → local swap)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    per_elite = remaining / max(1, len(elite))

    best_overall_seq = None
    best_overall_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        ada_dead = now + 0.55 * per_elite
        sub_dead = ada_dead + 0.30 * per_elite
        loc_dead = sub_dead + 0.15 * per_elite

        a_adam, sc_adam = _optimize_adam(vec, ada_dead, rng, enforce_sym=False)
        a_sub, sc_sub = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=False)
        a_loc, sc_loc = _local_swap_search(a_sub, loc_dead, rng, enforce_sym=False)

        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[refine {idx}] improved to {best_overall_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – up‑sampling to larger lengths (still ≤ 15000)
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 15000
        for factor in (1.2, 1.5, 2.0):
            new_n = int(best_overall_seq.shape[0] * factor)
            if new_n > max_len or new_n < 1000:
                continue
            a_up = _upsample_seq(best_overall_seq, new_n)
            S_new = math.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * S_new

            now = time.time()
            if now >= deadline:
                break
            short_dead = min(deadline, now + 0.07 * (deadline - now))
            a_up_opt, sc_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=False)

            if sc_up < best_overall_score:
                best_overall_score = sc_up
                best_overall_seq = a_up_opt.copy()
                print(f"[upsample] factor={factor:.2f} n={new_n} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated annealing on the current best
    # ------------------------------------------------------------------
    now = time.time()
    if best_overall_seq is not None and now < deadline:
        a_final, sc_final = _simulated_annealing(best_overall_seq, deadline, rng)
        if sc_final < best_overall_score:

"""
Algorithm: Random exploration of asymmetric families followed by
gradient‑based refinement (Adam → sub‑gradient → local swaps) and a final
simulated‑annealing search.  The sequence is always projected onto the
simplex Σ a_i = √(2 n) and kept non‑negative.  The objective is the
peak of the self‑convolution, which equals the value returned by
`evaluate_sequence` when the simplex constraint holds.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution (via FFT) – also used for the max entry
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _max_convolution(a: np.ndarray, fft_len: int | None = None) -> float:
    """Return max_i (a*a)_i using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    if fft_len is None:
        fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return float(conv.max())


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers for Adam
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_k w_k (a*a)_k  w must sum to 1."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1-i]."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Asymmetric candidate families
# ----------------------------------------------------------------------
def _gen_candidate_family(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    fam = rng.integers(0, 9)
    if fam == 0 and base is not None:
        a = _upsample_seq(base, n)
    elif fam == 1:
        a = rng.random(n)                        # pure uniform random
    elif fam == 2:
        rate = rng.uniform(0.90, 0.9995)
        a = rate ** np.arange(n)                 # decreasing geometric
    elif fam == 3:
        rate = rng.uniform(0.90, 0.9995)
        a = rate ** np.arange(n - 1, -1, -1)      # increasing geometric
    elif fam == 4:
        a = np.linspace(1.0, 0.1, n)              # linear decreasing
    elif fam == 5:
        a = np.linspace(0.1, 1.0, n)              # linear increasing
    elif fam == 6:
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)          # beta‑like (asymmetric)
    elif fam == 7:
        split = rng.integers(1, n - 1)
        rate1 = rng.uniform(0.90, 0.9995)
        rate2 = rng.uniform(0.90, 0.9995)
        part1 = rate1 ** np.arange(split)
        part2 = rate2 ** np.arange(n - split)
        a = np.concatenate([part1, part2])
    elif fam == 8:
        rate = rng.uniform(0.90, 0.9995)
        vals = rate ** np.arange(n)
        rng.shuffle(vals)
        a = vals                                 # random permutation of a decay
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.07
    max_alpha = 800.0

    best_a = a.copy()
    best_score = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape tiny plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.9999

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _max_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _max_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random swap / hill‑climb flavour
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _max_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.18:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb (local refinement)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _max_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _max_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing – pure non‑gradient local search
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)
    fft_len = 1 << ((2 * n - 1) - 1).bit_length()
    cur_score = _max_convolution(a, fft_len)
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    total = deadline - start
    T0 = max(best_score * 0.04, 0.001)
    step_scale = 0.3  # max fraction of source mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total
        T = T0 * (1.0 - progress)

        move_type = rng.random()
        if move_type < 0.6:
            # mass transfer i → j
            i = rng.integers(0, n)
            if a[i] <= 1e-14:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            max_delta = min(a[i], step_scale * a[i])
            delta = max_delta * rng.random()
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        elif move_type < 0.9:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            # small shuffle of ≤5 entries
            k = min(5, n)
            idx = rng.choice(n, size=k, replace=False)
            a_new = a.copy()
            a_new[idx] = rng.permutation(a_new[idx])

        a_new = np.maximum(a_new, 0.0)
        # renormalise only if drifting
        if abs(a_new.sum() - S) > 1e-8:
            a_new = a_new / a_new.sum() * S

        new_score = _max_convolution(a_new, fft_len)

        accept = False
        if new_score <= cur_score:
            accept = True
        elif T > 0:
            prob = math.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # occasional bigger random perturbation
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            a[mask] = rng.random(mask.sum())
            a = a / a.sum() * S
            cur_score = _max_convolution(a, fft_len)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm start from previous run, if any
    # ------------------------------------------------------------------
    base_seq = None
    if "height_sequence_1" in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – cheap exploration, keep a small elite set
    # ------------------------------------------------------------------
    K = 8
    elite = []          # (score, vector)
    attempts = 0
    phase0_dead = start_time + 0.12 * budget_s
    while time.time() < phase0_dead:
        attempts += 1

        # Choose a length – occasionally rescale the previous best
        if base_seq is not None and rng.random() < 0.5:
            scale = rng.uniform(0.85, 1.20)
            n = max(2000, int(base_seq.shape[0] * scale))
            base = _upsample_seq(base_seq, n)
        else:
            n = int(rng.integers(3000, 15001))
            base = None

        a = _gen_candidate_family(n, rng,
                                  base_seq if base is None else base)

        # Normalise to Σa = √(2 n)
        S = math.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # Occasionally randomise the order for extra asymmetry
        if rng.random() < 0.35:
            rng.shuffle(a)

        cur_score = _max_convolution(a)

        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    if not elite:
        n = 5000
        a = np.full(n, 1.0)
        S = math.sqrt(2.0 * n)
        a = a / a.sum() * S
        elite.append((_max_convolution(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – refinement of each elite vector (Adam → sub‑grad → swaps)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    per_elite = remaining / max(1, len(elite))

    best_overall_seq = None
    best_overall_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        ada_dead = now + 0.55 * per_elite
        sub_dead = ada_dead + 0.30 * per_elite
        loc_dead = sub_dead + 0.15 * per_elite

        a_adam, sc_adam = _optimize_adam(vec, ada_dead, rng, enforce_sym=False)
        a_sub, sc_sub = _optimize_subgrad(a_adam, sub_dead, rng, enforce_sym=False)
        a_loc, sc_loc = _local_swap_search(a_sub, loc_dead, rng, enforce_sym=False)

        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[refine {idx}] improved to {best_overall_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample the current best to larger lengths (≤15 000)
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 15000
        for factor in (1.2, 1.5, 2.0):
            new_n = int(best_overall_seq.shape[0] * factor)
            if new_n > max_len or new_n < 1000:
                continue
            a_up = _upsample_seq(best_overall_seq, new_n)
            S_new = math.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * S_new

            now = time.time()
            if now >= deadline:
                break
            short_dead = min(deadline, now + 0.07 * (deadline - now))
            a_up_opt, sc_up = _optimize_adam(a_up, short_dead, rng, enforce_sym=False)

            if sc_up < best_overall_score:
                best_overall_score = sc_up
                best_overall_seq = a_up_opt.copy()
                print(f"[upsample] factor={factor:.2f} n={new_n} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing refinement on the best vector
    # ------------------------------------------------------------------
    now = time.time()
    if best_overall_seq is not None and now < deadline:
        a_final, sc_final = _simulated_annealing(best_overall_seq, deadline, rng)
        if sc_final < best_overall_score:
            best_overall_score = sc_final
            best_overall_seq = a_final.copy()
            print(f"[final‑SA] improved to {best_overall_score:.6f}")

    # Final projection onto the simplex to remove any tiny drift
    if best_overall_seq is not None:
        n_final = best_overall_seq.shape[0]
        best_overall_seq = _project_simplex(best_overall_seq,
                                            math.sqrt(2.0 * n_final))

    # ------------------------------------------------------------------
    # Store and return
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()["height_sequence_1"] = np.array(final_seq, dtype=float)
    return final_seq
