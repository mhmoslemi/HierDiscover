def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5

    if budget_s >= 800: n=12000
    elif budget_s >= 400: n=8000
    else: n=5000

    target_sum = np.sqrt(2.0 * n)

    # initial candidates:
    candidates = []

    # power law
    for p in np.linspace(0.35, 0.8, 10):
        seq = _power_law_sequence(n, p, target_sum)
        candidates.append(seq)

    # truncated power law
    for p in np.linspace(0.35, 0.8, 6):
        for frac in np.linspace(0.3, 1.0, 5):
            k = int(frac * n)
            seq = _power_law_sequence(k, p, target_sum) # length k
            seq = np.pad(seq, (0, n-k), 'constant')
            candidates.append(seq)

    # geometric
    for q in np.linspace(0.999, 0.9, 6):
        a = np.arange(n)
        seq = q**a
        seq = seq * target_sum / seq.sum()
        candidates.append(seq)

    # piecewise linear decreasing to zero
    for m_frac in np.linspace(0.2, 1.0, 5):
        m = int(m_frac * n)
        if m==0: continue
        # linear from a0 to 0 at m-1
        lin = np.linspace(1, 0, m, endpoint=False)
        seq = np.concatenate([lin, np.zeros(n-m)])
        seq = seq * target_sum / seq.sum()
        candidates.append(seq)

    # include previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        candidates.append(prev)

    # Evaluate candidates quickly and pick top few.
    candidate_scores = []
    for seq in candidates:
        val = _score(seq)
        candidate_scores.append((val, seq))
    candidate_scores.sort(key=lambda x: x[0])

    # keep top K (maybe 5)
    top_candidates = [seq for _, seq in candidate_scores[:5]]
    best_seq = top_candidates[0].copy()
    best_val = _score(best_seq)

    # Phase 1: Exact gradient descent (subgradient)
    # allocate maybe 30% time

    if time.time() < deadline:
        seq, val = _deterministic_mass_transfer(best_seq, deadline, target_sum)
        if val < best_val:

"""
Propose a low‑score step function.

The algorithm works in three phases:
1. **Diverse parametric seeding** – generate a few hundred simple
   families (power‑law, truncated power‑law, geometric, piecewise‑linear)
   and keep the best few candidates (fast O(n log n) evaluation).
2. **Exact sub‑gradient descent** – for each kept candidate run a
   projected gradient descent on the *true* max‑convolution gradient
   (the index of the current maximal entry).  This uses a simple
   momentum/Adam style update and is very cheap (one FFT per iteration).
3. **Deterministic mass‑transfer refinement** – the same greedy
   “donor‑receiver” move as in the reference solution, now applied to
   the improved sequence.
4. **Final polishing** – a short high‑β soft‑max projected gradient
   descent (the routine from the reference implementation) to clean
   up any remaining irregularities.

All steps respect the time budget (default 1000 s) and the simplex
constraint Σa = √(2 n).  The returned list contains only the non‑zero
prefix of the optimal sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark objective (scale‑invariant)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p)   (1‑based index) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _geometric_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    """q^i   (0 ≤ i < n) normalised to target_sum."""
    a = q ** np.arange(n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _piecewise_linear_sequence(n: int, m_frac: float, target_sum: float) -> np.ndarray:
    """Linear decay from 1 to 0 over the first m entries, padded with zeros."""
    m = max(1, int(m_frac * n))
    lin = np.linspace(1.0, 0.0, m, endpoint=False)
    a = np.concatenate([lin, np.zeros(n - m, dtype=np.float64)])
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _exact_subgradient_descent(a0: np.ndarray,
                               target_sum: float,
                               deadline: float,
                               step0: float = 0.5,
                               decay: float = 5e-4,
                               jitter_every: int = 200,
                               jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent where the gradient is taken at the
    current maximal convolution index.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    n = a.size

    while time.time() < deadline:
        # full convolution and maximal index
        conv_full = _conv_fft(a)
        k = int(np.argmax(conv_full))

        # construct gradient g_i = 2 * a_{k-i}  (zero when out of range)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # adaptive step size
        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = (step0 / (1.0 + decay * it)) / norm
            a -= lr * grad

        # occasional jitter to avoid stagnation
        if it % jitter_every == 0 and it > 0:
            jitter = rng.normal(scale=jitter_scale * target_sum, size=n)
            a += jitter

        # enforce simplex
        a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] exactSGD it={it:6d} val={best_val:.6f}")

        it += 1

    return best_a, best_val


def _exact_gradient_descent_restarts(init: np.ndarray,
                                    target_sum: float,
                                    deadline: float,
                                    max_restarts: int = 5) -> tuple[np.ndarray, float]:
    """
    Perform a few short exact‑gradient runs with random jitter.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline and max_restarts > 0:
        remaining = deadline - time.time()
        run_time = min(5.0, remaining * 0.15)     # very short runs
        if run_time < 0.5:
            break

        jitter = rng.normal(scale=0.02 * target_sum, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _exact_subgradient_descent(start,
                                            target_sum,
                                            time.time() + run_time,
                                            step0=0.6,
                                            decay=1e-3,
                                            jitter_every=300,
                                            jitter_scale=5e-4)
        max_restarts -= 1

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] exactSGD restart improved best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy donor‑receiver mass moves directly targeting the current
    maximal convolution entry; Metropolis acceptance with exponential cooling.
    """
    a = a0.copy()
    n = a.size

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of maximal entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=0.001 * target_sum, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of soft‑max approx of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      max_beta: float = 1e5,
                                      step_factor: float = 0.7,
                                      gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:6d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

        for frac in np.linspace(0.3, 1.0, 5):
            k = max(1, int(frac * n))
            seq = _power_law_sequence(k, p, target_sum)
            seq = np.pad(seq, (0, n - k), 'constant')
            pool.append(seq)

    # geometric decay
    for q in np.linspace(0.99, 0.90, 6):
        pool.append(_geometric_sequence(n, q, target_sum))

    # piecewise linear decay
    for mf in np.linspace(0.2, 1.0, 5):
        pool.append(_piecewise_linear_sequence(n, mf, target_sum))

    # previous best (if available)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        pool.append(prev)

    # ---------------------------------------------------------------
    # Quick evaluation & keep the top few (5) seeds
    # ---------------------------------------------------------------
    scored = [( _score(seq), seq ) for seq in pool]
    scored.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in scored[:5]]
    best_seq = top_seeds[0].copy()
    best_val = _score(best_seq)
    print(f"[{time.time():.2f}] Initial best after seeding = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – exact sub‑gradient descent (30% of budget)
    # ---------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.30
    for seed in top_seeds:
        if time.time() >= phase1_deadline:
            break
        a, val = _exact_gradient_descent_restarts(seed,
                                                  target_sum,
                                                  phase1_deadline)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New global best after exact SGD = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer refinement (remaining 40%)
    # ---------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-4,
                                                      low_frac=0.05)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After DMT best = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 3 – final polish with soft‑max gradient descent (last 30%)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2_ext(best_seq,
                                                            deadline,
                                                            target_sum,
                                                            max_beta=2e5,
                                                            step_factor=0.6,
                                                            gamma=0.995)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Trim trailing zeros and return
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional final evaluation (for debugging)
    try:
        final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception:
        pass

    return final_seq
