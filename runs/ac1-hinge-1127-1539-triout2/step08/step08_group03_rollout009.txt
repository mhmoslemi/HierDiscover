def _adam_subgrad_optimize(a0, deadline, rng, lr0=0.45, lr_decay=0.99985, jitter_every=500):
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    # init logits
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)
    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0
    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0
    while time.time() < deadline:
        it += 1
        # get a
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S
        # compute gradient g and peak
        g, peak = _max_convolution_subgradient(a)
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
        # compute mu
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)
        # gradient clip
        max_abs = np.max(np.abs(grad_p))
        if max_abs > 5.0 * S:
            grad_p = grad_p * (5.0 * S / max_abs)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # Jitter
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)
        # decay lr
        lr = lr * lr_decay
    return best_a, best_peak

    Each candidate is normalised so that its ℓ₁‑norm equals √(2 n);
    the evaluation then reduces to the peak of the self‑convolution.

2️⃣  Multi‑scale refinement.
    For the best few candidates we grow the support (interpolation)
    and run an Adam optimiser that works directly on the *exact*
    max‑convolution sub‑gradient (no τ‑soft‑max smoothing).  The
    simplex constraint is enforced via a log‑softmax parametrisation,
    which guarantees positivity and the required ℓ₁‑norm.

3️⃣  Cheap simulated‑annealing fine‑tuning (mass‑transfer walk on the
    simplex) if time remains.

The whole pipeline fits comfortably into the 1000 s budget and
stores the final construction in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _max_convolution_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns (g, max_b) where g = ∂ (max_k (a*a)[k]) / ∂a  (a sub‑gradient)
    and max_b = max_k (a*a)[k].
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    # active set – all indices attaining the maximum (within a tiny tolerance)
    eps = 1e-12 * max(1.0, max_b)
    mask = (b >= max_b - eps).astype(float)
    total = mask.sum()
    if total > 0:
        mask /= total                     # average over the active set
    else:
        mask[:] = 0.0

    # gradient = 2 * (mask * a) * reversed a  (convolution)
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1 : n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation) using the
# exact sub‑gradient of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_subgrad_optimize(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator,
                           lr0: float = 0.45,
                           lr_decay: float = 0.99985,
                           jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Minimises the peak of the self‑convolution while keeping
    sum(a) = √(2 n).  Returns the best vector seen and its *hard* peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from a0  (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- recover a from logits ----------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---------- exact sub‑gradient ----------
        g, peak = _max_convolution_subgradient(a)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---------- gradient w.r.t. logits (chain rule) ----------
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        max_abs = np.max(np.abs(grad_p))
        if max_abs > 5.0 * S:
            grad_p = grad_p * (5.0 * S / max_abs)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- schedules ----------
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric centred Gaussian."""
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    sigma = rng.uniform(0.05 * n, 0.20 * n)
    raw = np.exp(- (idx - centre) ** 2 / (2.0 * sigma ** 2))
    return raw


def _init_cosine_bell(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine‑bell (raised cosine) symmetric about the centre."""
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    width = rng.uniform(0.3, 0.9) * (n - 1)      # full width where the bell is positive
    lo = centre - width / 2.0
    hi = centre + width / 2.0
    mask = (idx >= lo) & (idx <= hi)
    raw = np.where(mask,
                   np.cos(np.pi * (idx - centre) / width) ** 2,
                   0.0)
    raw = raw ** rng.uniform(0.5, 2.0)           # tweak sharpness
    return raw


def _init_sym_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric power‑law decreasing from the centre."""
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    offset = rng.uniform(1.0, 10.0)
    p = rng.uniform(0.5, 3.0)
    raw = 1.0 / ((np.abs(idx - centre) + offset) ** p)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.22 we warm‑start from the globally best
    construction; otherwise we draw from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    styles = ['dirichlet', 'gamma', 'uniform',
              'left_powerlaw', 'left_exponential', 'sparse',
              'gaussian', 'cosine', 'sym_powerlaw']
    style = rng.choice(styles)

    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'cosine':
        raw = _init_cosine_bell(n, rng)
    else:   # sym_powerlaw
        raw = _init_sym_powerlaw(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Cheap simulated‑annealing fine‑tuner (used only if time permits)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  The algorithm consists of:

    1. Random exploration of many vectors from diverse parametric families.
    2. Multi‑scale Adam refinement using the *exact* sub‑gradient of the
       max‑convolution.
    3. Optional simulated‑annealing fine‑tuning.

    The final construction is stored in the global variable
    ``height_sequence_1`` and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if present
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 55 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0          # reserve ~5 min for refinement
    best_seq = None
    best_score = np.inf
    top_k = 7
    top_candidates = []   # list of (score, seq)

    attempt = 0
    while time.time() < explore_deadline:
        attempt += 1
        # length uniformly drawn on a logarithmic scale
        if rng.random() < 0.55:
            n = int(rng.integers(1800, 5000))
        else:
            n = int(rng.integers(5000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # keep a small list of the best candidates for refinement
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Guarantee at least one candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale refinement (Adam with exact sub‑gradient)
    # ------------------------------------------------------------------
    max_len = int(min(50000, best_n * 8))          # a generous ceiling
    lengths = [best_n]
    factor_steps = [1.4, 2.0, 3.0, 5.0]
    for fac in factor_steps:
        cand = int(best_n * fac)
        if best_n < cand < max_len:
            lengths.append(cand)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_n = len(cur_seq)

        for target_n in lengths:
            if time.time() >= end_time - 30.0:
                break

            remaining = end_time - time.time()
            if remaining < 40.0:
                break

            # interpolate to target length and normalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining budget (max 200 s)
            slice_budget = min(remaining * 0.45, 200.0)

            # hyper‑parameters are jittered per run
            lr0 = rng.choice([0.38, 0.44, 0.52, 0.60])
            lr_decay = rng.choice([0.99980, 0.99985, 0.99990])
            refined, peak = _adam_subgrad_optimize(a_scaled,
                                                   time.time() + slice_budget,
                                                   rng,
                                                   lr0=lr0,
                                                   lr_decay=lr_decay,
                                                   jitter_every=500)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            cur_seq = refined
            cur_n = target_n

        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning if time permits
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 25.0:
        final_seq, final_peak = _anneal(best_seq,
                                        time.time() + remaining,
                                        rng)
        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_seq.copy()
            print(f"[anneal] final annealed score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
