def _deterministic_subgradient_descent(a, target_sum, deadline):
    a = a.copy()
    n = a.size
    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        # compute active indices
        active = np.flatnonzero(conv == max_b)
        # compute gradient contributions
        g = np.zeros_like(a)
        # loop through active entries:
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            idx = np.arange(i_low, i_high + 1)
            g[idx] += 2.0 * a[k - idx]  # gradient contribution
        # move mass
        i_max = np.argmax(g)
        i_min = np.argmin(g)
        if g[i_max] <= g[i_min] + 1e-12:
            # gradient uniform, maybe optimal
            break
        # compute delta
        delta = min(a[i_max] * 0.1, target_sum * 0.001)  # fraction of source
        if delta <= 1e-12:
            break
        # attempt move
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # due to floating errors, ensure nonneg
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)  # keep sum exactly
        conv_new = _full_convolution(a_new)
        max_b_new = conv_new.max()
        # accept if reduces max
        if max_b_new < max_b - 1e-12:
            a = a_new
        else:
            # try smaller delta
            delta *= 0.5
            if delta < 1e-12:
                break
            # retry with smaller delta (maybe need loop)

            a_new[i_max] -= delta

"""
Step‑function optimisation for ``evaluate_sequence``.

The objective for a normalised sequence (∑a = √(2·n)) is exactly the
maximum entry of its self‑convolution.  The algorithm therefore
optimises the max‑convolution directly.

Approach
--------
1.  Initialise from the supplied global ``height_sequence_1`` (if any);
    otherwise use a random power‑law shape.
2.  Run a length schedule (2000 → 5000 → 10 000).  For each length
    * up‑sample the current best seed,
    * perform a long projected Adam optimisation on a smooth
      log‑sum‑exp approximation of the max‑convolution,
    * polish the result with a deterministic sub‑gradient descent that
      moves a small amount of mass from the “most responsible’’ index
      to the “least responsible’’ index while keeping the full
      convolution up‑to‑date (O(n log n) per move).
3.  The best sequence seen so far is returned (trailing near‑zeros are
    trimmed).  The routine respects the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT based)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (empirically chosen)
    base_lr = 0.25 * target_sum

    # smoothed‑max temperature schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 1.7

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the log‑sum‑exp surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 2e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # temperature annealing
        if t % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled for diversity.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent (exact max‑convolution updates)
# ----------------------------------------------------------------------


def _deterministic_subgradient_descent(a0: np.ndarray,
                                      target_sum: float,
                                      deadline: float) -> np.ndarray:
    """
    Repeatedly move a small amount of mass from the index that contributes
    most to the current worst convolution entry towards the index that
    contributes least.  The full convolution is recomputed via FFT each
    move (O(n log n)).  Stops when no improvement is possible or the time
    budget expires.
    """
    a = a0.copy()
    n = a.size
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        # active set (could be more than one index)
        active = np.flatnonzero(conv == max_b)

        # compute sub‑gradient of max(b) (2·∑_{k∈active} a_{k‑i})
        g = np.zeros_like(a)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # indexes with largest / smallest gradient contribution
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if i_max == i_min:
            break

        # move a fraction of the source mass
        delta = a[i_max] * 0.07          # 7 % of the source amount
        if delta < 1e-12:
            break

        # trial move
        a_trial = a.copy()
        a_trial[i_max] -= delta
        a_trial[i_min] += delta

        # maintain feasibility (simplex) – projection corrects tiny rounding errors
        a_trial = _project_to_simplex(a_trial, target_sum)

        # evaluate new max‑convolution
        new_max = _max_convolution_fft(a_trial)

        if new_max < max_b - 1e-12:
            a = a_trial
            continue

        # if the move did not help, shrink delta and try again (up to 3 attempts)
        improved = False
        for _ in range(3):
            delta *= 0.5
            if delta < 1e-12:
                break
            a_trial = a.copy()
            a_trial[i_max] -= delta
            a_trial[i_min] += delta
            a_trial = _project_to_simplex(a_trial, target_sum)
            new_max = _max_convolution_fft(a_trial)
            if new_max < max_b - 1e-12:
                a = a_trial
                improved = True
                break
        if not improved:
            # No profitable move found – perturb a little to escape possible flat region
            jitter = rng.normal(scale=target_sum * 5e-5, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Seed from a previously stored construction, if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # --------------------------------------------------------------
    # Length schedule – larger dimensions give more flexibility
    # --------------------------------------------------------------
    schedule = [2000, 5000, 10000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ----- initialise candidate for this length -----
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # ----- allocate time slices for this dimension -----
        remaining = deadline - time.time()
        if remaining < 30.0:
            break
        # 60 % for Adam, 35 % for deterministic sub‑gradient, rest for polishing
        adam_budget = remaining * 0.60
        dg_budget = remaining * 0.35

        # ----- Phase A – long Adam optimisation (single long run) -----
        adam_deadline = min(deadline - 0.5, time.time() + adam_budget)
        a_adam = _adam_gradient_descent(a_init, target_sum, adam_deadline)
        score_adam = _max_convolution_fft(a_adam)
        if score_adam < best_score - 1e-12:
            best_seq = a_adam.copy()
            best_score = score_adam
            print(f"[{time.time():.1f}] Adam improvement for n={n} score={best_score:.6f}")

        # ----- Phase B – deterministic sub‑gradient refinement -----
        if time.time() < deadline - 1.0:
            dg_deadline = min(deadline - 0.5, time.time() + dg_budget)
            a_refined = _deterministic_subgradient_descent(best_seq, target_sum, dg_deadline)
            score_ref = _max_convolution_fft(a_refined)
            if score_ref < best_score - 1e-12:

                best_score = score_ref
                print(f"[{time.time():.1f}] Sub‑gradient improvement for n={n} "

        # ----- optional extra short Adam pass if time permits -----
        if time.time() < deadline - 5.0:
            extra_deadline = deadline - 2.0
            a_extra = _adam_gradient_descent(best_seq, target_sum, extra_deadline)
            score_extra = _max_convolution_fft(a_extra)
            if score_extra < best_score - 1e-12:
                best_seq = a_extra.copy()
                best_score = score_extra
                print(f"[{time.time():.1f}] Extra Adam polishing for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the very best sequence (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        final_deadline = deadline - 1.0
        # final Adam
        best_seq = _adam_gradient_descent(best_seq, target, final_deadline)

        # final deterministic sub‑gradient pass (short)
        final_deadline = deadline - 0.5
        best_seq = _deterministic_subgradient_descent(best_seq, target, final_deadline)

        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Final polishing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:

    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
