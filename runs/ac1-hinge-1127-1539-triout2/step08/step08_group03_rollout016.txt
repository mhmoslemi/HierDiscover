def _project_simplex(v, S):
    # v: array like size n
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        # all non-positive after shift, return zero vector simple? But we need sum S. set uniform.
        return np.full(n, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _pgd_optimize(a0, deadline, rng, step0=0.2, momentum=0.9, decay=0.9999, jitter_every=1000, active_eps=0.0):
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_peak = _max_convolution(a)
    step = step0
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a)
        peak = conv.max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
        # active set
        if active_eps == 0.0:
            mask = (conv == peak)
        else:
            mask = (conv >= peak - active_eps)
        ks = np.nonzero(mask)[0]
        # compute gradient as sum over active ks
        g = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            g[valid] += 2.0 * a[idx[valid]]
        # momentum update
        v = momentum * v - step * g
        a_new = a + v
        # project onto simplex
        a = _project_simplex(a_new, S)
        # step decay
        step *= decay
        # jitter
        if jitter_every and it % jitter_every == 0:
            # add small random noise
            a += rng.normal(scale=step*0.1, size=n)
            a = _project_simplex(a, S)
    return best_a, best_peak

# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return the maximal entry of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Projection of v onto {x ≥ 0, sum(x)=S} (O(n log n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – fall back to uniform vector
        return np.full(n, S / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Return a feasible start vector of length n (ℓ₁‑norm = √(2 n))."""
    S = np.sqrt(2.0 * n)

    # occasional warm‑start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential', 'sparse'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    else:
        raw = _init_sparse_bumps(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Projected‑gradient descent on the *hard* peak of self‑convolution
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  step0: float = 0.2,
                  momentum: float = 0.9,
                  decay: float = 0.9998,
                  jitter_every: int = 800,
                  active_eps: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Minimise the maximal entry of a*a while keeping sum(a)=√(2n).
    Returns the best vector observed and its (hard) peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    v = np.zeros_like(a)           # momentum term
    best_a = a.copy()
    best_peak = _max_convolution(a)

    step = step0
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a)
        peak = conv.max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # --- active set (indices achieving the current maximum) ---
        if active_eps == 0.0:
            mask = (conv == peak)
        else:
            mask = (conv >= peak - active_eps)
        ks = np.nonzero(mask)[0]

        # --- exact gradient of the (selected) max‑convolution ---
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in ks:
            src = k - idx
            ok = (src >= 0) & (src < n)
            g[ok] += 2.0 * a[src[ok]]

        # --- momentum + projection step ---
        v = momentum * v - step * g
        a = _project_onto_simplex(a + v, S)

        # --- schedule / jitter ---
        step *= decay
        if jitter_every and (it % jitter_every == 0):
            a += rng.normal(scale=step * 0.5, size=n)
            a = _project_onto_simplex(a, S)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (cheap, used only if time permits)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with Metropolis acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


# ----------------------------------------------------------------------
# Primary entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The final construction is stored in ``height_sequence_1`` and also returned.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from any previously found construction
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 45 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0           # keep ~5 min for refinement
    top_k = 8
    top_candidates = []     # (peak, seq)

    # if a warm sequence exists, seed the list immediately
    if warm_seq is not None:
        seq = warm_seq.astype(float)
        n0 = len(seq)
        seq = seq * (np.sqrt(2.0 * n0) / seq.sum())
        top_candidates.append((_max_convolution(seq), seq.copy()))

    attempt = 0
    while time.time() < explore_deadline:
        attempt += 1
        # choose a length – give a broad spread
        if rng.random() < 0.55:
            n = int(rng.integers(1800, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        # keep a short sorted list of the best candidates
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

        if attempt % 500 == 0:
            print(f"[explore] attempt={attempt:5d} n={n:5d}  peak={peak:.6f}")

    # ------------------------------------------------------------------
    # Safety net – guarantee at least one candidate
    # ------------------------------------------------------------------
    if not top_candidates:
        n = 3000
        a0 = np.full(n, 1.0)
        a0 = a0 * (np.sqrt(2.0 * n) / n)
        top_candidates = [(_max_convolution(a0), a0.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Projected‑Gradient Descent refinement
    # ------------------------------------------------------------------
    best_seq = top_candidates[0][1]
    best_peak = top_candidates[0][0]

    for init_idx, (init_peak, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_n = len(cur_seq)

        # schedule of lengths (gradually grow)
        max_len = min(30000, cur_n * 4)
        lengths = [cur_n]
        if cur_n * 1.4 < max_len:
            lengths.append(int(cur_n * 1.4))
        if cur_n * 2.0 < max_len:
            lengths.append(int(cur_n * 2.0))
        if cur_n * 3.0 < max_len:
            lengths.append(int(cur_n * 3.0))
        lengths.append(int(max_len))
        lengths = sorted(set(lengths))

        for target_n in lengths:
            remaining = end_time - time.time()
            if remaining < 40.0:
                break

            if target_n != cur_n:
                # upscale (linear interpolation) and renormalise
                a_scaled = _scale_sequence(cur_seq, target_n)
                a_scaled = np.clip(a_scaled, 0.0, None)
                a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())
                cur_seq = a_scaled
                cur_n = target_n

            # allocate a slice of the remaining budget (max 150 s)
            slice_budget = min(remaining * 0.45, 150.0)

            refined, peak = _pgd_optimize(cur_seq,
                                          time.time() + slice_budget,
                                          rng,
                                          step0=0.25,
                                          momentum=0.9,
                                          decay=0.9997,
                                          jitter_every=600,
                                          active_eps=1e-12)

            if peak < best_peak:
                best_peak = peak
                best_seq = refined.copy()
                print(f"[pgd] init#{init_idx+1} n={target_n:5d}  new best={best_peak:.6f}")

            cur_seq = refined

        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning if time permits
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        final_seq, final_peak = _anneal(best_seq,
                                        time.time() + remaining,
                                        rng)
        if final_peak < best_peak:
            best_peak = final_peak
            best_seq = final_seq.copy()
            print(f"[anneal] final annealed peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score={external:.6f}  internal peak={best_peak:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
