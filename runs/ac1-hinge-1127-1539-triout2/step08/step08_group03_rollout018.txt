"""
Adversarial step‑function optimizer – fourth generation.

The goal is to minimise ``evaluate_sequence`` which (for a
ℓ₁‑normalised vector) returns the peak of the self‑convolution.
The algorithm consists of three stages:

1️⃣  **Focused random sampling** –  we draw vectors from a few
    tightly‑controlled families that are known to give low peaks
    (pure decreasing power‑laws, pure decreasing exponentials and a
    power‑law/exponential mixture).  The families contain no uniform
    background and allow a very small offset (to keep the first term
    finite).  The best few candidates are retained.

2️⃣  **Long‑run Adam optimisation** – the currently best candidate is
    repeatedly interpolated to larger supports (up to 50 000 entries)
    and refined with an Adam optimiser that works on the exact
    τ‑soft‑max approximation of the peak.  The optimiser runs for the
    bulk of the time budget, allowing many Adam steps on the large
    problem.

3️⃣  **Cheap simulated‑annealing walk** – if any time is left, a
    mass‑transfer walk on the simplex is performed to escape possible
    shallow minima.

The final construction is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of ``a*a`` (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # grad = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Minimise the exact max‑convolution of ``a`` under the ℓ₁‑norm
    constraint.  Returns the best vector seen and its *hard* peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initial temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- recover a from logits ----------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---------- convolution & soft‑max gradient ----------
        g, peak = _softmax_gradient(a, tau)

        # keep the best *hard* peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---------- gradient wrt logits (chain rule) ----------
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- annealing ----------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for the exploration phase (purely decreasing)
# ----------------------------------------------------------------------
def _init_powerlaw_strict(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pure decreasing power‑law a_i = (i+offset)^(-p)."""
    p = rng.uniform(0.5, 0.85)                     # exponent
    offset = 10 ** rng.uniform(-4, -0.2)           # tiny offset ∈ [1e‑4,≈0.63]
    raw = (np.arange(n, dtype=float) + offset) ** (-p)
    return raw


def _init_exponential_strict(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pure decreasing exponential a_i = exp(-λ·i)."""
    lam = rng.uniform(0.0005, 0.025)               # decay rate
    raw = np.exp(-lam * np.arange(n, dtype=float))
    return raw


def _init_powerlaw_exp_mix(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law multiplied by a gentle exponential decay."""
    p = rng.uniform(0.5, 0.8)
    lam = rng.uniform(0.001, 0.015)
    offset = 10 ** rng.uniform(-4, -0.2)
    idx = np.arange(n, dtype=float)
    raw = (idx + offset) ** (-p) * np.exp(-lam * idx)
    return raw


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like (Gamma) vector – kept for diversity."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑start from the global best; otherwise draw from
    a decreasing family.
    """
    S = np.sqrt(2.0 * n)

    # warm start with modest probability
    if warm_seq is not None and rng.random() < 0.18:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a * (S / a.sum())
        return a

    # pick a family – we bias toward the decreasing ones
    style = rng.choice(['powerlaw', 'exponential', 'mix', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw_strict(n, rng)
    elif style == 'exponential':
        raw = _init_exponential_strict(n, rng)
    elif style == 'mix':
        raw = _init_powerlaw_exp_mix(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (very cheap)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.12 * cur_score        # start a bit colder – peak is already low

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue

        delta = rng.uniform(0.0, 0.25) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99994
        if temperature < 1e-7:
            temperature = 1e-7

    return best, best_score

        else:
            n = int(rng.integers(8000, 30000))

        a0 = _initial_vector(n, rng,
                             warm_seq=globals().get('height_sequence_1', None))
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain a small sorted list of the best candidates
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # fallback – should never be needed
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # --------------------------------------------------------------
    # Phase 2 – long Adam refinement (≈ 65 % of the budget)
    # --------------------------------------------------------------
    refine_deadline = start_time + total_budget * 0.90   # leave 10 % for anneal
    max_len_cap = 50000

    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_n = len(cur_seq)

        # length schedule – grow roughly geometrically up to the cap
        lengths = [cur_n]
        for m in (1.5, 2.0, 3.0, 4.5):
            nxt = int(cur_n * m)
            if nxt < max_len_cap:
                lengths.append(nxt)
        lengths.append(min(max_len_cap, cur_n * 6))
        lengths = sorted(set(lengths))

        for target_n in lengths:
            if time.time() >= refine_deadline:
                break

            # interpolate, clip, and renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining refinement budget
            remain = refine_deadline - time.time()
            slice_budget = min(remain * 0.45, 180.0)

            # Adam hyper‑parameters are varied a little to increase diversity
            lr0 = rng.choice([0.42, 0.48, 0.53])
            tau_decay = rng.choice([0.9992, 0.9994, 0.9996])
            refined, peak = _adam_optimize(
                a_scaled,
                time.time() + slice_budget,
                rng,
                lr0=lr0,
                tau_decay=tau_decay,
                lr_decay=0.99985,
                jitter_every=400,
            )

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            # use refined vector as seed for the next scale
            cur_seq = refined
            cur_n = target_n

        if time.time() >= refine_deadline:
            break

    # --------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning (≈ 10 % of budget)
    # --------------------------------------------------------------
    if time.time() < end_time:
        final_seq, final_peak = _anneal(best_seq,
                                        end_time,
                                        rng)
        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_seq.copy()
            print(f"[anneal] final annealed score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
