"""
step_optimize.py

A multi‑phase optimizer for the step‑function benchmark.

Main ideas
----------
* Phase A – fast projected gradient descent on a soft‑max
  approximation of the max‑convolution (same idea as the reference
  implementation, but with a larger β‑budget and a tighter time split).

* Phase B – *masked sub‑gradient* refinement.
  At each iteration the current convolution is computed, the entries
  close to the maximal one are selected (a mask) and the exact
  sub‑gradient of the (soft) max‑convolution restricted to that mask
  is obtained via a single FFT convolution.  A diminishing step size
  and Euclidean projection onto the simplex keep the iterates feasible.
  This concentrates the updates on the truly problematic positions and
  reduces the peak more aggressively than random donor‑receiver moves.

* Phase C – final polishing, a short high‑β projected gradient run.

All heavy linear‑algebra is performed with FFTs (O(n log n)).  The code
automatically adapts the problem size to the available time budget and
stores the best sequence found so far.

The function `propose_candidate` is the entry point required by the
evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)² (reduces to max convolution)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (Phase A)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 1e5,
                                   step_factor: float = 0.7,
                                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short gradient‑descent runs with random jitter.
    Keeps the best result found before the given deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_ext(start,
                                                 time.time() + run_budget,
                                                 target_sum,
                                                 max_beta=1e5,
                                                 step_factor=0.8,
                                                 gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Masked sub‑gradient refinement
# ----------------------------------------------------------------------


def _masked_subgradient_refine(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               mask_frac_start: float = 0.02,
                               mask_frac_end: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Aggressive refinement that concentrates the sub‑gradient on the
    entries of the convolution that are close to the current maximum.
    The mask fraction shrinks linearly with time to gradually focus
    the updates.
    """
    a = a0.copy()
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    it = 0

    while time.time() < deadline:
        # current convolution
        conv = _conv_fft(a)
        cur_max = conv.max()

        # linearly anneal the mask width
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        mask_frac = mask_frac_start * (1 - tau) + mask_frac_end * tau

        # mask – keep the top‑fraction of entries
        thresh = cur_max * (1 - mask_frac)
        mask = (conv >= thresh).astype(np.float64)

        # sub‑gradient for the masked max‑convolution
        a_rev = a[::-1]
        A_fft = np.fft.rfft(a_rev, n=size)
        M_fft = np.fft.rfft(mask, n=size)
        rev = np.fft.irfft(A_fft * M_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * rev[n - 1:n - 1 + n]

        # step size (norm‑scaled, slowly decreasing)
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            lr = (target_sum * 0.5) / (gnorm * np.sqrt(it + 1))
            a -= lr * grad

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional global jitter to escape plateaus
        if it % 4000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # update best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Masked‑SG it={it:6d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm combines:
      * short projected‑gradient runs on a soft‑max surrogate,
      * a masked sub‑gradient refinement that directly attacks the peak,
      * a final high‑β polishing pass.
    """
    start = time.time()
    # safety margin so the harness does not cut us off exactly at the limit
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 15000                 # a bit larger than the reference 12000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)   # ∑a_i is fixed by the benchmark definition

    # ------------------------------------------------------------------
    # Build a small yet diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_seqs = []

    # power‑law families (the exponent determines how quickly the tail decays)
    for p in (0.45, 0.5, 0.55, 0.6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # previous best if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # a few jittered copies to increase diversity
    rng = np.random.default_rng()
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # ------------------------------------------------------------------
    # Phase A – projected‑gradient descent (soft‑max)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + 0.20 * (deadline - time.time())   # ~20 % of total time
    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        a_cand, val = _gradient_descent_with_restarts(init,
                                                      target_sum,
                                                      gd_deadline)
        if val < best_score:
            best_score = val
            best_seq = a_cand.copy()
            print(f"[{time.time():.2f}] New global best after GD phase = {best_score:.6f}")

    # fallback – should never happen
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase B – masked sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        b_seq, b_score = _masked_subgradient_refine(best_seq,
                                                    deadline,
                                                    target_sum,
                                                    mask_frac_start=0.02,
                                                    mask_frac_end=0.001)
        if b_score < best_score:
            best_score = b_score
            best_seq = b_seq.copy()
            print(f"[{time.time():.2f}] After masked refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – final polishing (high‑β soft‑max GD)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_ext(best_seq,
                                                        deadline,
                                                        target_sum,
                                                        max_beta=3e5,
                                                        step_factor=0.6,
                                                        gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing zeros and return a plain list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (purely for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
