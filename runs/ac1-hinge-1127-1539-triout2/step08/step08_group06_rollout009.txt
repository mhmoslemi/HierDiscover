"""
Propose a step‑function (non‑negative vector) that minimizes the peak of its
self‑convolution, i.e. the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a three‑phase pipeline:

1. **LP refinement** – start from a modest length (≈1500).  For the current
   vector `a` we solve a linear program that maximises the total mass of a
   new vector `b` under the constraints ``a * b ≤ M`` where `M` is the
   current maximal convolution entry.  Scaling the optimum to the required
   ℓ₁‑norm and mixing it with `a` reduces the peak.  This cheap LP step is
   repeated many times (fast for n≈1500).

2. **Gradient‑based polishing** – the best LP result is refined with the
   same three‑stage optimiser used previously (smooth‑max Adam → exact‑max
   sub‑gradient → tiny random‑swap hill‑climb).

3. **Upsampling & final local search** – the polished vector is linearly
   up‑sampled to larger lengths (≈1.2×, 1.5×, 2×, capped at 15000) and each
   version receives a short Adam burst; a final swap‑search exploits any
   remaining budget.

The whole routine respects the simplex constraint Σaᵢ = √(2·n) after every
update and never uses negative values.  It returns the best sequence found
before the time limit.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Helper that evaluates the peak of the self‑convolution (internal metric)
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    """Return max_i (a*a)_i."""
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Symmetrisation (optional)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear interpolation to a new length (preserves shape)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) probabilities over convolution entries
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return a probability vector proportional to exp(alpha·conv)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of weighted sum of convolution entries
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_i w_i (a*a)_i   (w must sum to 1).
    Uses the fact that gradient of (a*a)_i w.r.t a equals
    2·(a reversed shifted by i).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the soft‑max approximation of the convolution peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.05
    max_alpha = 800.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the genuine max‑convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the selected entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random swap
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.18:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        if it % 9000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb (local refinement)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing that swaps tiny mass chunks between random positions."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _peak_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Convolution‑matrix for the LP constraints  (a * b) ≤ M
# ----------------------------------------------------------------------
def _conv_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A (shape (2*n‑1, n)) such that
        A @ b == a * b   (full linear convolution)
    Each column j of A contains a copy of the vector a placed starting at row j.
    """
    n = a.shape[0]
    # each diagonal offset j (0 ≤ j < n) carries the full vector a
    offsets = np.arange(n, dtype=int)
    # diags copies the data for each diagonal, so we give a separate reference
    A = sp.diags([a] * n, offsets, shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Linear‑programming refinement (reduce max‑convolution peak)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 30,
               mix_factor: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Repeatedly:
      • solve LP   maximise Σ b_i   s.t. a * b ≤ M, b ≥ 0
      • scale the optimum to the required ℓ₁‑norm
      • mix the scaled vector into a   (a ← (1‑t)a + t g)
    Returns the best vector seen and its convolution peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = _peak_convolution(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a)
        M = conv.max()

        # build the (a * b) ≤ M constraints
        A = _conv_constraint_matrix(a)               # (2n‑1) × n
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # LP: maximise Σ b_i   ⇔   minimise −Σ b_i
        res = linprog(c=-np.ones(n, dtype=float),
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=(0, None),
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # If the LP fails we stop the refinement.
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # scale to the required ℓ₁‑norm and mix
        g = b * (S / sum_b)
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP] iter={it}  M={M:.6f}  score={cur_score:.6f}")

    return best_a, best_score

    # ------------------------------------------------------------------

            now = time.time()

            if sc_up < best_score:
