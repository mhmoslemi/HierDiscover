"""
Propose a step‑function (non‑negative vector) with a low self‑convolution peak.

Algorithm
---------
1. **Exploration** – generate many random families (uniform, geometric,
   exponential, power‑law, beta, logistic, up‑sample of a previous best).
   Keep the K best (lowest convolution peak) vectors.

2. **Intensive refinement** – on each elite vector run a long optimisation:
       * Adam on a smooth log‑sum‑exp (soft‑max) surrogate of the peak,
         with a temperature schedule that drives α up to several thousand.
       * Projected sub‑gradient descent on the true peak entry (exact max).
       * Tiny random‑swap hill‑climb (local perturbations).

3. **Upsampling** – linearly up‑sample the best vector to larger lengths
   (≈1.3×, 1.5×, 2×, bounded by 15000) and give each a short Adam burst.

4. **Final check** – evaluate the result with the provided `evaluate_sequence`,
   store it globally as `height_sequence_1` and return the list.

All updates respect the simplex constraint  Σ a_i = √(2·n) , a_i ≥ 0
by Euclidean projection after every step.  The code uses only numpy
(and the already‑imported `evaluate_sequence`) and stays well within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear self‑convolution using FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of Σ_i w_i (a*a)_i with respect to a (w must sum to 1)
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇_a Σ_i w_i (a*a)_i  (returns a vector of length len(a))."""
    n = a.shape[0]
    # reverse w because convolution formula is reversed
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]                 # factor 2 comes from the derivative
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) probabilities over convolution entries
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return a probability vector proportional to exp(alpha·conv)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Peak of the self‑convolution (the quantity we minimise)
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Linear interpolation to a new length (preserves overall shape)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, new_n: int) -> np.ndarray:
    if base.size == new_n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Candidate generators (different analytic families)
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    fam = rng.integers(0, 9)
    if fam == 0 and base is not None:
        # up‑sample previous best (maybe with a small random stretch)
        factor = rng.uniform(0.85, 1.15)
        a = _upsample_seq(base, int(max(2, int(n * factor))))
        a = a[:n] if a.shape[0] > n else np.pad(a, (0, n - a.shape[0]))
    elif fam == 1:
        # pure uniform random
        a = rng.random(n)
    elif fam == 2:
        # geometric decay from the left end (asymmetric)
        rate = rng.uniform(0.92, 0.9995)
        a = rate ** np.arange(n)
    elif fam == 3:
        # exponential decay
        lam = rng.uniform(0.0005, 0.02)
        a = np.exp(-lam * np.arange(n))
    elif fam == 4:
        # power‑law decay (a_i ∝ (n‑i)^p)
        p = rng.uniform(0.3, 5.0)
        a = (np.arange(n, 0, -1, dtype=float)) ** (-p)
    elif fam == 5:
        # beta‑like (asymmetric)
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0.0, 1.0, n)
        a = (x ** p1) * ((1 - x) ** p2)
        a = a[::-1]                     # flip to put mass near the left side
    elif fam == 6:
        # sinusoidal power (still symmetric but often useful)
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    else:
        # logistic bump (symmetric)
        scale = rng.uniform(0.02, 0.15) * n
        mid = (n - 1) / 2.0
        xs = np.arange(n, dtype=float)
        left = 1.0 / (1.0 + np.exp((xs - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - xs) / scale))
        a = left + right
    a = np.clip(a, 0.0, None)
    # add a tiny amount of Gaussian noise (helps escaping symmetry)
    if rng.random() < 0.3:
        noise = rng.normal(scale=0.02, size=n)
        a = np.maximum(a + noise, 0.0)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate of the peak
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 0.1
    alpha_mul = 1.07
    max_alpha = 5000.0

    best_a = a.copy()
    best_score = _peak_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)

        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep onto the feasible simplex
        a = _project_simplex(a, S)

        # occasional jitter to shake out of plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99993          # very gentle decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true maximum entry
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    eta0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k = int(conv.argmax())

        # gradient of the selected entry (factor 2 already included)
        g = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k - idx_i
        g[idx_i] = 2.0 * a[idx_j]               # factor 2 for the true derivative

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        a = _project_simplex(a, S)

        cur_score = _peak_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # light random swaps (simulated‑annealing flavour)
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = _peak_convolution(cand)
                if cand_score < cur_score or rng.random() < 0.15:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

        # early stall detection
        if it % 12000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _swap_hill_climb(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _peak_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand_score = _peak_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the timeout.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # safety margin to guarantee we finish before external kill
    deadline = start + budget_s - 4.0

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – cheap exploration, keep K elite candidates
    # ------------------------------------------------------------------
    K = 8
    elite = []                     # list of (score, vector)
    attempts = 0
    explore_deadline = start + 0.12 * budget_s
    while time.time() < explore_deadline:
        attempts += 1
        # pick a length – favour medium‑large vectors but keep some diversity
        if base_seq is not None and rng.random() < 0.5:
            scale = rng.uniform(0.85, 1.15)
            n = max(500, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(3000, 13001))

        a = _gen_candidate(n, rng, base_seq)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        cur_score = _peak_convolution(a)

        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts:5d} n={n:5d} score={cur_score:.6f}")

    if not elite:
        # fallback – simple decreasing geometric shape
        n = 8000
        a = np.exp(-0.001 * np.arange(n))
        a = a / a.sum() * np.sqrt(2.0 * n)
        elite.append((_peak_convolution(a), a))

    # ------------------------------------------------------------------
    # Phase 1 – intensive refinement of the elites
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # allocate roughly equal time slices to each elite (but always keep some headroom)
    per_elite = max(0.0, remaining / len(elite))

    best_seq = None
    best_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        # time budgets for the three sub‑phases
        ada_dead = now + 0.60 * per_elite
        sub_dead = ada_dead + 0.30 * per_elite
        loc_dead = sub_dead + 0.10 * per_elite

        # ---- Adam on smooth max ----
        a_adam, sc_adam = _adam_optimize(vec, ada_dead, rng)

        # ---- Sub‑gradient on true max ----
        a_sub, sc_sub = _subgrad_optimize(a_adam, sub_dead, rng)

        # ---- Tiny random‑swap hill‑climb ----
        a_loc, sc_loc = _swap_hill_climb(a_sub, loc_dead, rng)

        if sc_loc < best_score:
            best_score = sc_loc
            best_seq = a_loc.copy()
            print(f"[refine {idx}] new best score={best_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – upsample the current best to larger lengths (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None:
        max_len = 15000
        up_factors = [1.25, 1.5, 1.75, 2.0]
        for fac in up_factors:
            new_n = int(best_seq.shape[0] * fac)
            if new_n > max_len or new_n < 2:
                continue
            a_up = _upsample_seq(best_seq, new_n)

            # renormalise to the new length
            S_new = np.sqrt(2.0 * new_n)
            a_up = a_up / a_up.sum() * S_new

            now = time.time()
            if now >= deadline:
                break
            # short Adam burst (≈5 % of remaining time)
            short_dead = min(deadline, now + 0.05 * (deadline - now))
            a_up_opt, sc_up = _adam_optimize(a_up, short_dead, rng)

            if sc_up < best_score:
                best_score = sc_up
                best_seq = a_up_opt.copy()
                print(f"[upsample] factor={fac:.2f} n={new_n} improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final tiny local search with any leftover budget
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline and best_seq is not None:
        a_final, sc_final = _swap_hill_climb(best_seq, deadline, rng)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[final‑local] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (store globally for warm‑restarts)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[return] external={external_score:.6f} internal={best_score:.6f} "
          f"len={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
