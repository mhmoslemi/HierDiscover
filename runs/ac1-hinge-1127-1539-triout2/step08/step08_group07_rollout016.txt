"""
Improved search for a low‑value step function.

The objective that evaluate_sequence() finally returns (after normalising the
ℓ¹‑norm) is 2·n·max(conv(f,f)).  Hence we want a probability vector p (sum=1)
for which the self‑convolution p∗p has a small L∞‑norm.  The optimisation is
performed on the vector a = √(2·n)·p, i.e. a lives on the simplex
{a≥0, Σa = √(2·n)} – the scaling does not change the objective.

Algorithm
---------
1. Choose a problem size n according to the time budget (larger when we have
   many seconds).
2. Build a fairly large pool of *parametric* initial candidates:
      * power‑law a_i = (i+1)^{-p} for many p∈[0.45,0.65]
      * shifted power law (i+offset)^{-p}
      * exponential a_i = exp(-λ·i)
      * the previous best sequence (if any) resized to the current n.
   All candidates are projected onto the simplex.
3. Pick the 4‑5 best candidates (lowest raw score) and improve each of them
   with an Adam optimiser that minimises a soft‑max approximation of the
   maximal convolution entry.  The soft‑max temperature β is increased
   geometrically during the run.
4. The best result of step 3 is refined by a *deterministic mass‑transfer*
   moveset that explicitly reduces the current maximal convolution entry.
5. A final short Adam run with a very large β (≈2·10⁵) polishes the sequence.
6. Trailing zeros are stripped and the list is returned.

All heavy work uses FFT‑based convolutions (O(n log n)).  The code respects
the 1000 s budget thanks to explicit time checks.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)² (reduces to max convolution)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators (all return a vector already normalised to `target_sum`)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float,
                        offset: float = 0.0) -> np.ndarray:
    """a_i = (i+offset)^{-p}  (offset≥0), then normalise."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam·i), normalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max over the convolution entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)               # weights that sum to 1

    # gradient = 2 * (reverse(a) * p)  (linear convolution)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    # The relevant part of the convolution that aligns with a:
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution objective
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  max_beta: float = 1e6,
                  step_scale: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max (β‑controlled) approximation of max(conv).
    Returns the best vector encountered and its raw score.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    # Adam statistics
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    beta = 0.5

    while time.time() < deadline:
        # gradient of smooth objective
        g, _ = _gradient_softmax(a, beta)

        it += 1
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Classic Adam step with a simple global scaling
        lr = step_scale * target_sum / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # keep us on the simplex (non‑negativity + Σ = target_sum)
        a = _project_to_simplex(a, target_sum)

        # increase the soft‑max temperature gradually
        if it % 200 == 0:
            beta = min(beta * 1.5, max_beta)

        # occasional evaluation
        if it % 100 == 0:
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:8.2e} "
                      f"best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bound the amount that can be transferred
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Candidate creation – a fairly rich family of parametric seeds
# ----------------------------------------------------------------------


def _create_initial_pool(n: int, target_sum: float) -> list[np.ndarray]:
    """Generate a diversified set of feasible starting vectors."""
    pool = []

    # power‑law family (dense sweep of the exponent)
    for p in np.arange(0.45, 0.66, 0.01):
        pool.append(_power_law_sequence(n, float(p), target_sum))
        # also a shifted version to avoid a singular first element
        pool.append(_power_law_sequence(n, float(p), target_sum, offset=0.5))

    # exponential decays
    for lam in np.linspace(0.0001, 0.001, 5):
        pool.append(_exponential_sequence(n, float(lam), target_sum))

    # reverse of the power‑law (mass concentrated at the *end* of the array)
    for p in (0.5, 0.55, 0.6):
        seq = _power_law_sequence(n, p, target_sum)[::-1]
        pool.append(seq)

    # previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        resized = _resize_sequence(prev, n)
        resized = resized * target_sum / resized.sum()
        pool.append(resized)

    # jittered copies of the above to increase diversity
    rng = np.random.default_rng()
    for seq in list(pool):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        pool.append(seq_jit)

    return pool


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine builds a collection of parametric seeds, runs a few Adam
    optimisations, refines the best result by deterministic mass transfer
    and finishes with a short high‑β polishing pass.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # tiny safety margin

    # ------------------- choose problem size -------------------
    # Larger n gives a finer discretisation of the underlying continuous
    # optimisation problem.
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # ------------------- build a rich initial pool -------------
    pool = _create_initial_pool(n, target_sum)

    # ------------------- quick raw evaluation -----------------
    # keep only the few most promising seeds
    scored = [(seq, _score(seq)) for seq in pool]
    scored.sort(key=lambda x: x[1])
    top_candidates = [seq for seq, _ in scored[:5]]
    best_seq = top_candidates[0].copy()
    best_score = _score(best_seq)

    # ------------------- Phase A – Adam optimisation -----------
    # allocate roughly 60 % of the remaining budget to this phase
    phase_a_deadline = time.time() + 0.60 * (deadline - time.time())
    for cand in top_candidates:
        if time.time() >= phase_a_deadline:
            break
        a_opt, val = _adam_descent(cand,
                                   phase_a_deadline,
                                   target_sum,
                                   max_beta=3e5,
                                   step_scale=0.5)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after Adam = {best_score:.6f}")

    # ------------------- Phase B – deterministic refinement -----
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-4,
                                                      low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # ------------------- Phase C – final high‑β polish ---------
    if time.time() + 2.0 < deadline:
        a_final, val_final = _adam_descent(best_seq,
                                            deadline,
                                            target_sum,
                                            max_beta=2e5,
                                            step_scale=0.4)
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------- post‑processing -----------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
