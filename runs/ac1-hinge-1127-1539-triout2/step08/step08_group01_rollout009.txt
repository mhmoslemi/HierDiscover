```
    # schedule determined dynamically based on remaining time
    # target lengths

    # fractions of total budget (approx)

    fractions = [0.05, 0.07, 0.08, 0.1, 0.30, 0.40]  # sum=1

        if time.time() >= deadline:
            break
        # allocate time slice
        remaining = deadline - time.time()
        slice_time = remaining * frac
        if slice_time < 10.0:
            # if not enough time, use all remaining
            slice_time = remaining - 1.0
        # target sum
        target_sum = math.sqrt(2.0 * n)

        # upsample or random init
        if best_seq is not None and best_seq.size != n:
            a0 = _upsample_sequence(best_seq, n)
        else:
            a0 = _random_initial_sequence(n, rng)
        a0 = _project_to_simplex(a0, target_sum)

        # number of restarts depends on n: more restarts for smaller n where cheap
        if n <= 20000:

        elif n <= 50000:
            n_restarts = 6
        else:
            n_restarts = 4
        per_restart = slice_time * 0.6 / n_restarts  # 60% for GD

            if best_seq.size != n:

            else:
```

and the algorithm always normalises the ℓ¹‑norm to  

```
∑a = √(2·n)
```

the objective reduces to `max(a*a)`.  
The optimisation therefore consists of:

1. **Simplex projection** – keep the vector non‑negative and of fixed ℓ¹‑norm.  
2. **Smooth‑max Adam optimisation** – minimise a log‑sum‑exp (smooth) approximation of the
   maximum of the convolution.  The gradient is computed analytically by another convolution,
   all operations are performed with FFTs ( O(n log n) ).
3. **Length schedule** – start with a modest length (2000) and gradually increase up to
   200 000.  The current best vector is up‑sampled to the new length, a few random restarts are
   performed and the best result is kept.
4. **Bias‑guided hill‑climbing** – after the Adam phase a fast simulated‑annealing style
   hill‑climber moves a small amount of mass from one index to another while keeping the full
   convolution up‑to‑date in O(n) per move.
5. **Final polishing** – a longer Adam run and a short hill‑climb on the best vector found.
6. **Cleanup** – trailing near‑zero entries are trimmed.

The whole procedure respects the 1000 s wall‑clock limit, prints progress information and finally
returns the best sequence as a plain Python list.

```python
"""
Multi‑stage optimisation of a non‑negative step‑function.

The goal is to minimise 2·n·max(conv(a,a)) / (sum a)².
Because we keep the ℓ¹‑norm equal to √(2·n), the objective is simply
max(conv(a,a)).  The algorithm consists of:

1.  Simplex projection (non‑negative, fixed ℓ¹‑norm).
2.  Adam optimisation of a smoothed max (log‑sum‑exp) of the
    self‑convolution, gradient obtained via FFT‑based convolutions.
3.  Length schedule: 2 000 → 8 000 → 20 000 → 50 000 → 100 000 → 200 000.
    Each stage starts from the up‑sampled best vector, performs several
    random restarts and a biased hill‑climb.
4.  Final polishing with a long Adam run and a short hill‑climb.
5.  Trims trailing near‑zeros and returns the list.

All steps are timed so that the total wall‑clock budget (default 1000 s)
is never exceeded.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper routines
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).

    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           base_lr_factor: float = 0.10,
                           beta0: float = 2.0,
                           beta_max: float = 1e6,
                           beta_factor: float = 1.5) -> np.ndarray:
    """
    Projected Adam optimisation of the soft‑max surrogate of max(conv(a,a)).

    Parameters
    ----------
    a0 : np.ndarray
        Starting point (already on the simplex).
    target_sum : float
        Desired ℓ¹‑norm ( √(2·n) ).
    deadline : float
        Wall‑clock time after which the optimiser must stop.
    base_lr_factor : float
        Multiplicative factor for the base learning‑rate:
        base_lr = base_lr_factor * target_sum / n.
    beta0, beta_max, beta_factor
        Parameters governing the annealing of the smooth‑max temperature.

    Returns
    -------
    np.ndarray
        Optimised vector (projected onto the simplex).
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate scaled to the typical coordinate magnitude
    base_lr = base_lr_factor * target_sum / n

    beta = beta0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the smooth‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase temperature of the smooth‑max
        if t % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex  Σa = √(2 n).

    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # Dirichlet (sparse)
        alpha = rng.uniform(0.05, 0.5, size=n)
        a = rng.dirichlet(alpha=alpha) * target

    elif family == 5:                   # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(cut)),
                            np.exp(-lam2 * np.arange(cut, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length,
    followed by projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (biased version)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray,
                            b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The destination is biased towards indices that currently contribute little
    to the worst convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9997                    # slower cooling for finer search

    while time.time() < deadline:
        # pick one of the currently worst convolution positions
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index – weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index – bias towards small counterpart a[M‑j]
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seed from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            prev = _project_to_simplex(prev, math.sqrt(2.0 * n0))
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – larger dimensions get most of the remaining time
    # ------------------------------------------------------------------
    schedule = [2000, 8000, 20000, 50000, 100000, 200000]
    # Rough fractions of the *remaining* budget (they shrink as we go)
    fractions = [0.05, 0.07, 0.09, 0.12, 0.30, 0.37]

    for n, frac in zip(schedule, fractions):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        slice_time = remaining * frac
        if slice_time < 8.0:           # not enough time for a full stage
            slice_time = remaining - 1.0

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for the current length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Time split: 60 % Adam, 35 % hill‑climb, 5 % extra polishing
        # --------------------------------------------------------------
        gd_time = slice_time * 0.60
        hc_time = slice_time * 0.35
        extra_time = slice_time - gd_time - hc_time

        # --------------------------------------------------------------
        # Adam phase – several random restarts
        # --------------------------------------------------------------
        n_restarts = 6 if n <= 20000 else 4
        per_restart = gd_time / max(1, n_restarts)

        local_best = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)
            start_vec = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_gradient_descent(start_vec, target_sum,
                                           this_deadline,
                                           base_lr_factor=0.04)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best = a_opt.copy()

        if local_best is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] GD improvement for n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Hill‑climbing phase
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0 and hc_time > 5.0:
            if best_seq.size != n:
                a_cur = _upsample_sequence(best_seq, n)
            else:
                a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)

            hill_deadline = min(deadline - 0.2, time.time() + hc_time)
            a_hc, b_hc = _hill_climbing_improved(a_cur, b_cur, hill_deadline)
            score_hc = b_hc.max()
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Tiny extra Adam run (if any time left)
        # --------------------------------------------------------------
        if extra_time > 5.0 and time.time() < deadline - 5.0:
            a_ref = _adam_gradient_descent(best_seq, target_sum,
                                           deadline - 1.0,
                                           base_lr_factor=0.03)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Extra‑GD improvement for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the best sequence (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # long Adam run
        final_deadline = deadline - 2.0
        best_seq = _adam_gradient_descent(best_seq, target,
                                          final_deadline,
                                          base_lr_factor=0.02)

        # short hill‑climb
        b_final = _full_convolution(best_seq)
        best_seq, b_final = _hill_climbing_improved(best_seq, b_final,
                                                    deadline - 0.5)

        final_score = b_final.max()
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```
