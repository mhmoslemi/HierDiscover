def _hard_max_exponential_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()

        conv = _conv_fft(a)

        # Compute gradient for max index
        grad = np.zeros_like(a)

        idx = np.arange(lo, hi + 1, dtype=np.int64)

"""
Propose a step‑function with a low benchmark score.

Algorithm
---------
1. **Problem size** – n≈12000 (for a 1000 s budget).
2. **Warm‑start** – search a three‑parameter family  

       a_i = (i+1)^(-p) · exp(-λ·i / n) · (1 – i/(n‑1))^q

   on a reduced size (≈n/4).  The best shape is interpolated to the
   target length and renormalised so that Σa = √(2 n).

3. **Hard‑max exponential descent** – at each step the index `k_max`
   where the convolution `a*a` is maximal is located.
   The exact sub‑gradient of this entry is
       g_i = 2·a_{k_max‑i}   (zero outside the support).
   An exponentiated‑gradient step `a←a·exp(-η·g)` keeps the variables
   non‑negative; the ℓ¹‑norm is restored afterwards.  This aggressively
   lowers the current maximum.

4. **Soft‑max (smooth) exponential descent** – the same update as above
   but using a temperature‑controlled soft‑max approximation of the
   maximum (gradient obtained by `_gradient_softmax`).  The temperature
   β is doubled periodically, sharpening the objective.

5. **Deterministic mass‑transfer refinement** – repeatedly move a
   modest fraction of mass from indices that contribute heavily to the
   current maximal convolution entry to indices that contribute little.
   Only moves that improve the score are kept.

All steps respect non‑negativity and the ℓ¹‑norm constraint, so the
returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    With Σa = √(2 n) this equals max(a*a).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max (smooth) gradient utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 1 – extended parametric warm‑start on a reduced scale
# ----------------------------------------------------------------------


def _extended_grid_param_search(small_n: int,
                                target_sum: float,
                                deadline: float) -> tuple[np.ndarray, float]:
    """
    Search a three‑parameter family
        a_i = (i+1)^(-p) · exp(-λ·i/small_n) · (1 – i/(small_n‑1))^q
    on a reduced problem size.
    Returns the best sequence and its score.
    """
    p_vals = np.linspace(0.30, 1.30, 25)          # 25 values
    lam_vals = np.linspace(0.0, 12.0, 25)         # 25 values
    q_vals = np.linspace(0.0, 5.0, 13)            # 13 values

    best_seq = None
    best_score = np.inf
    idx = np.arange(small_n, dtype=np.float64)

    # deterministic grid
    for p in p_vals:
        for lam in lam_vals:
            for q in q_vals:
                if time.time() > deadline:
                    break
                seq = (idx + 1) ** (-p) * np.exp(-lam * idx / small_n) \
                      * ((1 - idx / (small_n - 1)) ** q)
                seq = seq * (target_sum / seq.sum())
                val = _score(seq)
                if val < best_score:
                    best_score = val
                    best_seq = seq.copy()
                    print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} q={q:.3f} score={best_score:.6f}")
            else:
                continue
            break
        else:
            continue
        break

    # random local refinement
    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.30, 1.30)
        lam = rng.exponential(scale=2.0) * 3.0   # typical range up to ~12
        q = rng.uniform(0.0, 5.0)
        seq = (idx + 1) ** (-p) * np.exp(-lam * idx / small_n) \
              * ((1 - idx / (small_n - 1)) ** q)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} q={q:.3f} score={best_score:.6f}")

    return best_seq, best_score

# Stage 2 – hard‑max exponential (mirror) descent
# ----------------------------------------------------------------------


def _hard_max_exponential_descent(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max(convolution) using the sub‑gradient
    of the active entry only.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    base_eta = 0.04

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k_max - idx]

        # decreasing step size
        eta = base_eta / (1.0 + it * 1e-4)
        a = a * np.exp(-eta * grad)

        # renormalise exactly
        a = a * (target_sum / a.sum())

        # occasional jitter to escape shallow plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Hard‑max it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Stage 3 – exponential descent on soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with multiplicative updates using a soft‑max
    approximation of the maximal convolution entry.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Stage 5 – enhanced deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _enhanced_mass_transfer(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Move a modest fraction of mass from indices that heavily
    contribute to the current maximal convolution entry towards indices
    that contribute little, accepting only improving moves.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        # sensitivities for the active entry (gradient = 2*a_{k_max-i})
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        # donor: large product a_i * sens_i
        weight = a * sens
        if weight.sum() > 0:
            donor = rng.choice(n, p=weight / weight.sum())
        else:
            donor = rng.integers(n)

        # receiver: preferably low weight (low contribution)
        low_thr = np.percentile(weight, 10)
        low_candidates = np.where(weight <= low_thr)[0]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # transfer a modest fraction (up to 10 % of donor's mass)
        max_frac = 0.10
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        # recompute convolution and evaluate
        conv = _conv_fft(a)
        val = _score(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # occasional renorm (numerical drift)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    """

    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on allotted time
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

                                         dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.60, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced problem
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)

                                                    small_target,
                                                    param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – hard‑max exponential descent (≈25 % of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:

                                                         target_sum,
                                                         hard_deadline)
        best_seq = a_hard
        best_val = val_hard

    else:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)

        print(f"[{time.time():.2f}] Evaluation error: {exc}")
