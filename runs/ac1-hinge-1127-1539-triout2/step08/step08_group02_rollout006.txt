"""
Improved step‑function optimizer.

Main ideas
-----------
* Enforce mirror symmetry at every stage – the optimal solution is
  believed to be symmetric, and a symmetric representation halves the
  effective dimension.
* More aggressive time allocation:
      – 300 s for cheap random exploration.
      – 200 s for scaling the current best to a few larger lengths and
        running short Adam refinements.
      – The rest for a long final Adam run on the best length.
* Adam with a *linear* temperature schedule that drives τ→0, i.e. a
  true τ‑soft‑max approximation of the peak of the self‑convolution.
* Projection of the logits onto the symmetric sub‑space after each
  Adam update.
* A cheap stochastic hill‑climb that respects symmetry.

The routine returns a list of non‑negative heights, stored in the global
``height_sequence_1`` and also printed together with the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parametric families for random initialisation (raw, then symmetrised)
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a symmetric, ℓ₁‑normalised start vector of length n.
    Occasionally warm‑starts from the globally best sequence.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with modest probability
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
    else:
        style = rng.choice(
            ['powerlaw', 'exponential', 'linear',
             'gamma', 'uniform', 'dirichlet', 'ushape'])
        if style == 'powerlaw':
            raw = _init_powerlaw(n, rng)
        elif style == 'exponential':
            raw = _init_exponential(n, rng)
        elif style == 'linear':
            raw = _init_linear(n, rng)
        elif style == 'gamma':
            raw = _init_gamma(n, rng)
        elif style == 'ushape':
            raw = _init_ushape(n, rng)
        elif style == 'uniform':
            raw = _init_uniform(n, rng)
        else:
            raw = _init_dirichlet(n, rng)
        a = raw

    # enforce non‑negativity
    a = np.clip(a, 0.0, None)

    # enforce exact mirror symmetry
    a = (a + a[::-1]) * 0.5

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = n
    a = a * (S / total)          # ℓ₁‑norm = √(2 n)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the symmetric simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the τ‑soft‑max approximation of the peak
    of the self‑convolution, constrained to the symmetric simplex:
        a_i ≥ 0,   Σ a_i = √(2 n),   a_i = a_{n‑1‑i}.
    Returns the best feasible sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                # logits
    p = (p + p[::-1]) * 0.5         # enforce symmetry of the logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5                       # start learning rate

    # temperature schedule (linear decay to a tiny value)
    b0 = _full_convolution_fft(a0)
    tau0 = max(b0.max(), 1.0)
    tau_min = 1e-5

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0
    start = time.time()

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S                     # a is automatically symmetric

        # ---- convolution & τ‑soft‑max mask ----
        b = _full_convolution_fft(a)
        max_b = b.max()

        # keep the best hard peak seen so far
        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

        # progress for schedules
        progress = (time.time() - start) / max(1e-12, (deadline - start))
        tau = tau0 * (1.0 - progress) + tau_min * progress

        # mask = soft‑max of b (used for the gradient)
        diff = b - max_b
        exps = np.exp(diff / max(tau, 1e-12))
        mask = exps / np.sum(exps)

        # ---- gradient w.r.t. a (τ‑soft‑max) ----
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]   # shape (n,)

        # ---- chain rule: gradient w.r.t. logits ----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 * max(0.1, 1.0 - progress)   # gentle linear decay, never below 0.1*lr0
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # re‑impose symmetry of the logits
        p = (p + p[::-1]) * 0.5

        # occasional jitter to escape shallow plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative symmetric step‑function that minimises
    ``evaluate_sequence``.  The algorithm follows the phases described
    above and stores the final construction in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap random exploration (≈300 s)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 300.0
    while time.time() < explore_deadline:
        attempt += 1
        # length biased towards a few thousand entries
        if rng.random() < 0.68:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 18000))

        a0 = _initial_vector(n, rng, warm_seq)   # symmetric by construction
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:4d} n={n:5d}  new best={best_score:.6f}")

    # fallback in the unlikely event that no candidate was produced
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – scaling & short Adam refinements (≈200 s)
    # ------------------------------------------------------------------
    scale_deadline = time.time() + 200.0
    # geometric growth candidates
    lengths = [best_n]
    for factor in [1.2, 1.5, 2.0, 2.5, 3.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= scale_deadline:
            break
        # perform a few short Adam runs for each target length
        for run_id in range(3):
            if time.time() >= scale_deadline:
                break
            slice_deadline = time.time() + min(30.0,
                                              (scale_deadline - time.time()) * 0.3)
            a_scaled = _scale_sequence(best_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = (a_scaled + a_scaled[::-1]) * 0.5
            S = np.sqrt(2.0 * target_n)
            a_scaled = a_scaled * (S / a_scaled.sum())
            a_opt, peak = _adam_optimize(a_scaled, slice_deadline, rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_opt.copy()
                best_n = target_n
                print(f"[scale] length={target_n:5d} run={run_id}  improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long Adam optimisation on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 10.0:
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[final] length={best_n:5d}  best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (respect symmetry)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # move a modest fraction of mass from i to j
        frac = rng.uniform(0.02, 0.15)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount

        # restore exact symmetry
        a = (a + a[::-1]) * 0.5

        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[hill] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[result] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
