def _lp_improve(a, M, target_sum, deadline):
    n = a.size
    rows = []
    cols = []
    data = []
    for i in range(n):
        start = i
        end = i + n
        rows.extend(range(start, end))
        cols.extend([i]*n)
        data.extend([a[i]]*n)
    A = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # convert to csr for linprog
    A_csr = A.tocsr()
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M, dtype=float)
    bounds = [(0, None)] * n
    # call linprog
    result = scipy.optimize.linprog(c, A_ub=A_csr, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        return None, None
    g0 = result.x
    s = g0.sum()
    if s <= target_sum * 1.000001: # no improvement
        return None, None
    # scale g
    scale = target_sum / s
    g = g0 * scale
    new_M = M * scale
    return g, new_M

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm follows a multi‑stage scheme:

1.   Use an existing construction (``height_sequence_1``) if available.
2.   Run a length schedule (2000 → 4000 → 8000 → 16000).  For each length
     * up‑sample the current best seed,
     * perform several restarts of projected Adam optimisation on a smooth
       approximation of the max‑convolution,
     * apply a linear‑programming “LP‑boost” (only for modest lengths) that
       expands the ℓ¹‑mass while keeping the current worst convolution entry
       bounded – this is the key new ingredient,
     * finish with a biased hill‑climb that updates the exact self‑convolution
       in O(n) per move.
     The time budget is split dynamically so that later (larger) lengths get
     most of the remaining time.
3.   A final polishing (Adam + hill‑climb) is performed on the best
     sequence found.
4.   Near‑zero trailing entries are trimmed and the list is returned.

All operations respect the 1000 s wall‑clock limit and avoid
closures, lambda‑expressions or file I/O.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    # we can compute by FFT but a reversed is easier
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation on the smooth‑max objective.
    The learning‑rate is scaled to the simplex radius.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Step‑size base (empirically tuned)
    base_lr = 0.20 * target_sum

    # Smoothed‑max parameters
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the soft‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0008 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (biased version)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The destination is biased towards indices that contribute little to the
    current worst convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    # temperature starts proportional to current score
    T = cur_score * 0.12 + 1e-12
    cooling = 0.9999

    while time.time() < deadline:
        # pick one of the worst positions
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index – bias towards small counterpart
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]
        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Linear‑programming “LP‑boost” (only for modest lengths)
# ----------------------------------------------------------------------


def _lp_improve(a: np.ndarray, M: float, target_sum: float,
                deadline: float) -> tuple[np.ndarray | None, float | None]:
    """
    Solve the LP:
        max   Σ g_i
        s.t.  g_i ≥ 0
              (a * g)_k ≤ M   for all k
    Return the scaled vector g (∥g∥₁ = target_sum) and the new bound
    M·target_sum/Σg_i, or (None, None) if the LP fails or gives no
    improvement.
    """
    if not _SCIPY_AVAILABLE:
        return None, None

    n = a.size
    # Build sparse constraint matrix A where A[k,i] = a_i  if 0≤k‑i<n else 0
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    pos = 0
    for i in range(n):
        start = i
        end = i + n
        rows[pos:pos + n] = np.arange(start, end, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = a[i]
        pos += n

    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

    c = -np.ones(n)                     # maximise Σg_i  → minimise -Σg_i
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # Run the LP – we rely on HiGHS which is fast for this size.
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
    except Exception:                  # pragma: no cover
        return None, None

    if not res.success:
        return None, None

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= target_sum * (1.0 + 1e-9):
        # No useful mass increase
        return None, None

    scale = target_sum / sum_g0
    g = g0 * scale
    new_M = M * scale
    return g, new_M


def _lp_iterative_improvement(a: np.ndarray, target_sum: float,
                              deadline: float, max_iters: int = 4) -> np.ndarray:
    """
    Repeatedly apply the LP‑boost while it yields improvement.
    Stops when time is out or no progress after ``max_iters`` attempts.
    """
    best = a.copy()
    best_score = _max_convolution_fft(best)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        M = _max_convolution_fft(best)
        g, new_M = _lp_improve(best, M, target_sum, deadline)
        if g is None:
            break
        new_score = _max_convolution_fft(g)
        if new_score < best_score - 1e-6:
            best = g
            best_score = new_score
            print(f"[{time.time():.1f}] LP‑boost improvement → {best_score:.6f}")
        else:
            break
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from previous construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            prev = _project_to_simplex(prev, math.sqrt(2.0 * n0))
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
                  f"score={best_score:.6f}")
        except Exception as e:          # pragma: no cover
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – larger dimensions get the lion's share of time
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Time partition for this stage
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 15.0:
            break
        # 40 % for gradient descent, 30 % for LP‑boost (if applicable),
        # 30 % for hill‑climb.
        gd_budget = remaining * 0.40
        lp_budget = remaining * 0.30 if (n <= 2500 and _SCIPY_AVAILABLE) else 0.0
        hc_budget = remaining * 0.30

        # --------------------------------------------------------------
        # Phase A – projected Adam gradient descent (multiple restarts)
        # --------------------------------------------------------------
        n_restarts = 5 if n <= 8000 else 3
        per_restart = gd_budget / max(n_restarts, 1)
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.1, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_gradient_descent(start_a, target_sum, this_deadline)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] GD improvement for n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – LP‑boost (only for modest lengths)
        # --------------------------------------------------------------
        if lp_budget > 5.0 and best_seq is not None and best_seq.size == n:
            lp_deadline = min(deadline - 0.1, time.time() + lp_budget)
            best_seq = _lp_iterative_improvement(best_seq, target_sum, lp_deadline)
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Post‑LP score for n={n} = {best_score:.6f}")

        if time.time() >= deadline - 5.0:
            break

        # --------------------------------------------------------------
        # Phase C – biased hill‑climbing
        # --------------------------------------------------------------
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        hill_deadline = min(deadline - 0.1, time.time() + hc_budget)
        a_hc, b_hc = _hill_climbing_improved(a_cur, b_cur, hill_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score - 1e-12:
            best_seq = a_hc.copy()
            best_score = score_hc
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Optional extra GD after hill‑climb (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            extra_deadline = deadline - 1.0
            a_refined = _adam_gradient_descent(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution_fft(a_refined)
            if score_ref < best_score - 1e-12:
                best_seq = a_refined.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC GD improvement for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the best sequence (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        final_deadline = deadline - 2.0
        best_seq = _adam_gradient_descent(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, b_final = _hill_climbing_improved(
            best_seq, b_final, deadline - 1.0)

        final_score = b_final.max()
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:                # fallback – uniform distribution
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:            # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
