def _gradient_max(a, tau):
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-10:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        # normalize
        mask_sum = mask.sum()
        if mask_sum > 0:
            mask = mask / mask_sum
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()
    # compute gradient via convolution of mask with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + len(a) - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[len(a)-1:len(a)-1+len(a)]
    return g, max_b

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Key ideas beyond the reference implementation
------------------------------------------------
* **Exact sub‑gradient** – after the soft‑max temperature drops,
  the gradient is computed from the true active set of convolution
  indices (all positions whose value is within a tiny tolerance of the
  maximum).  This yields a correct sub‑gradient of the hard max and
  improves convergence.
* **Richer random families** – add Gaussian bumps and half‑sine bumps
  to the exploration pool (in addition to Dirichlet, Gamma, power‑law,
  exponential, sparse‑bump).  This diversifies the starting points.
* **Longer support** – the refinement schedule now grows up to
  ``≈ 8·10⁴`` entries (instead of ~3·10⁴), giving the optimiser more
  freedom to lower the peak.
* **Aggressive Adam schedules** – larger learning‑rates, faster
  temperature decay and occasional jitter help escape plateaus.
* **Final greedy hill‑climbing** – after simulated‑annealing a cheap
  mass‑transfer walk is performed to capture any remaining tiny
  improvements.

The code respects the 1000 s budget, prints progress, stores the best
construction in the global ``height_sequence_1`` and returns it as a
plain Python list.

import time

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (hard‑max sub‑gradient)
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂ max(b) / ∂a
    and max_b = max(b) with b = a*a (full convolution).

    If ``tau`` is very small we fall back to the exact active set
    (all indices within a tiny tolerance of the maximum) and average
    their gradients; otherwise a soft‑max (τ‑soft‑max) is used.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    # --------------------------------------------------------------
    # Build the weighting mask
    # --------------------------------------------------------------
    if tau < 1e-12:
        # hard max – take all indices whose value is within tolerance
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # --------------------------------------------------------------
    # Gradient: 2 * (mask * a_rev) convolved with a_rev
    # --------------------------------------------------------------
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    # slice to obtain length‑n gradient
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm
    constraint (sum = sqrt(2 n)).  Returns the best vector seen
    and its hard peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # shift for stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- gradient of (soft‑)max convolution -------
        g, peak = _max_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # background weight
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Centered Gaussian bump (continuous) with random width."""
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n   # 5‑30 % of the length
    x = np.arange(n)
    raw = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return raw


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Half‑sine shape (zero at the ends, peak in the middle)."""
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    # random amplitude scaling and optional offset
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.22 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = math.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':

    else:  # half_sine

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:

        total = n
    a = raw * (S / total)
    return a

# Simulated‑annealing / greedy hill‑climbing fine‑tuner

def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score  # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        # transfer a random fraction of the mass at i to j
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())  # renormalise exactly
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy hill‑climber (strictly downhill moves only).
    """
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]  # move up to 15 % of mass
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score

            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain top‑k list
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Safety net – ensure at least one candidate exists
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement (≈70 % of budget)
    # ------------------------------------------------------------------
    # decide a generous upper bound for the length (up to ~8·10⁴)
    max_len = min(80000, best_n * 12)
    lengths = [best_n]
    cur_len = best_n
    # geometric growth schedule
    while cur_len * 1.5 < max_len:
        cur_len = int(cur_len * 1.5)
        lengths.append(cur_len)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    # iterate over shortlisted candidates (most promising first)
    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_len = len(cur_seq)

        for target_n in lengths:
            if time.time() >= end_time - 30.0:
                break

            # interpolate to the new length and renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining time (cap at 180 s)
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.45, 180.0)

            # hyper‑parameters – randomise a little for diversity
            lr0 = rng.choice([0.42, 0.48, 0.55, 0.60])
            tau_decay = rng.choice([0.9992, 0.9994, 0.9996, 0.9998])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.99985,
                                           jitter_every=400)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            # seed the next scale with the refined vector
            cur_seq = refined
            cur_len = target_n

        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic fine‑tuning (≈10 % of budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        # simulated annealing first
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.6 * remaining,
                                       rng)
        remaining = end_time - time.time()
        # then a strict greedy hill‑climber
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
