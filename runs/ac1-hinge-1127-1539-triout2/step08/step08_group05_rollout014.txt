"""
Propose a step‑function sequence with a low self‑convolution peak.

The algorithm follows a multi‑phase scheme:

1. **Random sampling** – many cheap candidate families (including
   geometric, power‑law, beta‑like, sinusoidal, logistic, etc.) are
   generated.  The best few (up to 5) are kept.  All candidates are
   projected onto the simplex Σa = √(2 n) and symmetrised.

2. **Projected Adam** – each stored candidate is refined with a
   projected Adam optimiser that minimises a soft‑max approximation of
   the convolution peak.  The temperature (α) is increased to a large
   value (≈2000) so the gradient closely follows the true
   max‑gradient.  Symmetry is enforced after every update.

3. **Multi‑sub‑gradient descent** – a deterministic sub‑gradient method
   that averages the sub‑gradients of the *k* largest convolution
   entries (k=3) is applied to the current best sequence.

4. **Simulated‑annealing hill‑climb** – tiny mass transfers between
   random positions are accepted with a Metropolis rule, allowing the
   optimiser to escape shallow local minima.

5. **Upsampling & final Adam** – the best sequence is linearly
   up‑sampled to the maximal allowed length (30 000) and a short Adam
   run finishes the search.

All steps respect the problem’s constraints (non‑negative entries,
fixed ℓ₁‑norm, clipping to [0,1000]) and run within the supplied time
budget.  The final sequence is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog  # imported only to keep the environment happy

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the non‑negative simplex of total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    The weight vector w must sum to one (soft‑max probabilities).
    """
    n = a.shape[0]
    # reverse w because of convolution definition
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1‑i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Families for cheap random generation
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a diverse set of families."""
    fam = rng.integers(0, 11)
    if fam == 0 and base_seq is not None:
        # Up‑sample a known good solution
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # Uniform random
    elif fam == 2:
        # Symmetric geometric decay
        rate = rng.uniform(0.93, 0.999)
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        # Symmetric power‑law (i·(n‑i))^p
        p = rng.uniform(0.2, 4.0)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        # Beta‑like skewed (x^p1 * (1‑x)^p2)
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        # Sinusoidal power
        p = rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        # Monotone decreasing power‑law
        p = rng.uniform(0.3, 3.0)
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        # Exponential decay
        lam = rng.uniform(0.001, 0.02) * n
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        # Logistic‑like S‑curve
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    elif fam == 9:
        # Cosine power shape (central bump)
        p = rng.uniform(1.0, 6.0)
        a = np.cos(np.pi * (np.arange(n) - (n - 1) / 2) / n) ** p
        a = np.clip(a, 0, None)
    else:
        # Piecewise linear tent with random apex
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])

    a = np.clip(a, 0.0, None)
    return a

# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 2000.0,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    alpha = 0.1
    alpha_mul = 1.02

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.08 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997   # slow decay

    return best_a, best_score


def _optimize_multi_subgrad(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            enforce_sym: bool = True,
                            top_k: int = 3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum convolution entry,
    averaging the sub‑gradients of the top‑k largest entries.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.25
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        # indices of the k largest entries
        top_idx = np.argpartition(-conv, top_k)[:top_k]

        g = np.zeros_like(a)
        for kmax in top_idx:
            i_low = max(0, kmax - (n - 1))
            i_high = min(n - 1, kmax)
            idx_i = np.arange(i_low, i_high + 1, dtype=int)
            idx_j = kmax - idx_i
            g[idx_i] += a[idx_j]
        if top_k > 1:
            g /= top_k

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random re‑balancing move
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.06 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        if it % 8000 == 0 and best_score == cur_max:
            # no progress for a while
            break

    return best_a, best_score


def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that swaps tiny masses.
    Accepts worsened moves with probability exp(-Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    cur_max = best_score
    start = time.time()
    T0 = max(0.12 * cur_max, 1e-3)   # a slightly larger initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # transfer up to 12 % of the source mass
        delta = 0.12 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Retrieve previous best if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # store up to five promising candidates
    best_pool = []          # list of (score, sequence)
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈15 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1
        # moderate to large lengths, up to the hard limit 30 000
        if base_seq is not None and rng.random() < 0.45:
            scale = rng.uniform(0.9, 1.1)
            n = max(2000, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(3000, 30001))

        a = _sample_family(rng, n, base_seq)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        # enforce symmetry
        a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))

        # keep the five best candidates seen so far
        if len(best_pool) < 5:
            best_pool.append((cur_score, a.copy()))
            best_pool.sort(key=lambda x: x[0])
        else:
            if cur_score < best_pool[-1][0]:
                best_pool[-1] = (cur_score, a.copy())
                best_pool.sort(key=lambda x: x[0])

        if cur_score < best_pool[0][0]:
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement for each pool element (≈55 % of remaining)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        raise RuntimeError("Time budget exhausted before Phase 2")

    # allocate the remaining time proportionally to the pool size
    for idx, (init_score, seq) in enumerate(best_pool):
        # fraction of remaining time for this element
        t_end = now + (remaining * (0.55 / len(best_pool)))
        a_ref, sc_ref = _optimize_adam(seq, t_end, rng,
                                       max_alpha=2000.0,
                                       enforce_sym=True)
        now = time.time()
        # replace entry if improved
        if sc_ref < init_score:
            best_pool[idx] = (sc_ref, a_ref.copy())
        else:
            best_pool[idx] = (init_score, seq.copy())
        # keep pool sorted
        best_pool.sort(key=lambda x: x[0])
        # update remaining budget
        remaining = deadline - now
        if remaining <= 0:
            break

    # best sequence after Phase 2
    best_overall_seq = best_pool[0][1].copy()
    best_overall_score = best_pool[0][0]

    # ------------------------------------------------------------------
    # Phase 3 – multi‑sub‑gradient descent (≈25 % of remaining)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        t_sub = now + 0.25 * (deadline - now)
        a_sub, sc_sub = _optimize_multi_subgrad(best_overall_seq, t_sub, rng,
                                                enforce_sym=True,
                                                top_k=3)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing hill‑climb (≈5 % of remaining)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_anneal, sc_anneal = _anneal_hillclimb(best_overall_seq, deadline, rng,
                                                enforce_sym=True)
        if sc_anneal < best_overall_score:
            best_overall_score = sc_anneal
            best_overall_seq = a_anneal.copy()
            print(f"[anneal] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – aggressive up‑sampling + final Adam (≈remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 7.0:   # keep a few seconds for final output
        cur_n = best_overall_seq.shape[0]
        new_n = min(30000, int(cur_n * 1.5))   # up‑sample a bit (max 30 k)
        a_big = _upsample_seq(best_overall_seq, new_n)

        # re‑normalise to the new length
        S_big = np.sqrt(2.0 * new_n)
        a_big = a_big / a_big.sum() * S_big
        a_big = _symmetrize(a_big)

        t_refine = now + 0.10 * (deadline - now)
        a_ref, sc_ref = _optimize_adam(a_big, t_refine, rng,
                                       max_alpha=2500.0,
                                       enforce_sym=True)
        if sc_ref < best_overall_score:
            best_overall_score = sc_ref
            best_overall_seq = a_ref.copy()
            print(f"[upsample‑adam] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
