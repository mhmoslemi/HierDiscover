def linear_step(f, M):
    n = len(f)
    # Build sparse convolution matrix
    from scipy.sparse import diags, csr_matrix
    A = diags(f, offsets=range(n), shape=(2*n-1, n), format='csr')
    # Upper bound constraints: A @ b <= M
    ub = np.full(2*n-1, M)

    return b_opt

def _lp_solve(f, M):
    n = len(f)
    from scipy.sparse import diags
    # Build matrix A with shape (2*n-1, n)
    diag_vals = [np.full(n, f[i]) for i in range(n)]
    A = diags(diag_vals, offsets=range(n), shape=(2*n-1, n), format='csr')
    ub = np.full(2*n-1, M)

    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=ub,

def _init_cosine(n, rng):
    """Cosine-shaped peaked at centre, zero at edges."""
    i = np.arange(n)
    x = (i / (n - 1)) * np.pi  # 0..π
    raw = np.cos(x)  # cos from 1 to -1. shift to nonnegative: (cos+1)/2
    raw = (raw + 1) / 2

                                          max_iters=30,
                                          deadline=refine_deadline)
        print(f"[lp refine] length={len(best_seq)} score={best_score:.6f}")

    # Phase 2b: scaling to longer lengths and LP refine
    lengths = [best_n]
    for factor in [1.4, 1.8, 2.2]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))
    for target_n in lengths[1:]:
        if time.time() >= end_time - 100.0:
            break
        slice_deadline = time.time() + min( (end_time - time.time()) * 0.2, 120.0)
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = _normalize(a_scaled)
        a_refined, new_peak = _lp_refine(a_scaled, t=0.12,
                                         max_iters=20,
                                         deadline=slice_deadline)
        if new_peak < best_score:
            best_score = new_peak
            best_seq = a_refined.copy()
            best_n = target_n
            print(f"[scale refine] n={target_n} new best={best_score:.6f}")

    # Phase 3: optional Adam fine‑tuning
    if time.time() < end_time - 30.0:
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[adam] length={best_n} score={best_score:.6f}")

    # Phase 4: cheap stochastic hill‑climb
    while time.time() < end_time - 2.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        frac = rng.uniform(0.01, 0.2)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[local] improved score={best_score:.6f}")

    # final check
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal={best_score:.6f} length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq

"""
Propose a step‑function with a lower bound on
evaluate_sequence().

Algorithm
---------
1. Randomly generate many candidates from a diverse set of
   parametric families (power‑law, exponential, triangular,
   gamma, uniform, Dirichlet, U‑shape and a cosine bump).
   Keep the best (lowest max‑convolution).

2. Refine the best candidate with a *linear‑program* based
   iteration:
      • For the current sequence f compute M = max(f*f).
      • Solve   maximise   Σ b_j
                subject to  f * b ≤ M   (elementwise),
                         0 ≤ b_j ≤ 1000 .
        (this is a small linear program; we use SciPy’s HiGHS
        LP solver).
      • Rescale b to the same ℓ₁‑norm as f and mix:
                f←(1‑t)·f + t·b   (t≈0.12).
      • Accept the mix only if the peak decreases; otherwise
        keep f and possibly reduce t.
      • Repeat for a limited number of iterations or until the
        time budget for this phase expires.

3. Upscale the current best sequence to a few larger lengths
   (geometric factors) and run the same LP refinement on each
   scaled copy.

4. Perform a short Adam optimisation (log‑softmax parametrisation)
   to fine‑tune the continuous variables.

5. Finish with a cheap stochastic hill‑climb that moves a random
   fraction of mass between two entries.

All steps respect the 1000 s overall budget; the final sequence
is stored in the global variable ``height_sequence_1`` and returned
as a plain Python list.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Sequence normalisation / scaling helpers
# ----------------------------------------------------------------------
def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector so that Σ seq = √(2·len(seq))."""
    n = len(seq)
    S = np.sqrt(2.0 * n)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0)
        total = n
    return seq * (S / total)


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parametric families for the random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine bump – values are (cos(π·i/(n‑1))+1)/2, possibly mixed."""
    i = np.arange(n)
    x = np.pi * i / (n - 1)
    raw = (np.cos(x) + 1.0) / 2.0           # 1 → 0 back to 1
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * raw + (1.0 - mix) * base
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Produce a feasible start vector of length n."""
    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = _normalize(a)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    a = _normalize(raw)
    return a


# ----------------------------------------------------------------------
# Linear‑program based refinement (step 2 of the algorithm)
# ----------------------------------------------------------------------
def _lp_solve(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   Σ b_j
        subject to conv(f, b) ≤ M      (elementwise)
                   0 ≤ b_j ≤ 1000
    Returns the optimal b or None if the solver fails.
    """
    n = f.shape[0]
    # Build the convolution matrix A (shape (2n‑1, n)).
    # Each diagonal i (offset i) contains the constant f[i].
    diag_vals = [np.full(n, f[i]) for i in range(n)]
    A = scipy.sparse.diags(diag_vals, offsets=range(n),
                           shape=(2 * n - 1, n), format='csr')
    ub = np.full(2 * n - 1, M)

    # Bounds for each b_j.
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # linprog solves a minimisation problem → minimise –Σb.
    res = scipy.optimize.linprog(c=-np.ones(n),
                                 A_ub=A,
                                 b_ub=ub,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True,
                                          'time_limit': 5.0})
    if res.success:
        return res.x
    return None


def _lp_refine(a0: np.ndarray,
               t: float,
               max_iters: int,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Iterative LP based improvement.
    Returns the best sequence found and its max‑convolution value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # Solve LP with current peak as the bound.
        b_opt = _lp_solve(a, best_peak)
        if b_opt is None:
            break

        sum_a = a.sum()
        sum_b = b_opt.sum()
        if sum_b == 0.0:
            break

        # Rescale b to the same ℓ₁‑norm as a.
        g = b_opt * (sum_a / sum_b)

        # Convex mix.
        a_new = (1.0 - t) * a + t * g
        a_new = _normalize(a_new)

        new_peak = _max_convolution(a_new)
        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            # keep step size
        else:
            # No improvement – reduce mixing factor to be more cautious.
            t = max(t * 0.5, 0.02)
    return best_a, best_peak


# ----------------------------------------------------------------------
# Adam optimiser (used as a final fine‑tune)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """τ‑soft‑max gradient of max(b) where b = a*a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parameterisation).
    Returns the best sequence seen and its exact max‑convolution.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                       # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # temperature schedule – start a bit larger than before
    tau = max(_full_convolution_fft(a0).max(), 1.0)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient ----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (slow decay) ----
        tau *= 0.9998
        lr *= 0.9999

        # occasional jitter
        if it % 800 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`. The method combines random exploration,
    a linear‑program refinement, scaling, Adam fine‑tuning and a
    short stochastic hill‑climb.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 65 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 250.0   # keep ~250 s for later phases
    while time.time() < explore_deadline:
        attempt += 1
        # Length bias – favour a few thousand entries
        if rng.random() < 0.68:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 18000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)   # with the required normalisation this *is* the score
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – linear‑program refinement on the current best length
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: simple uniform construction
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n

    lp_deadline = end_time - 150.0   # reserve time for scaling + Adam + hill‑climb
    if time.time() < lp_deadline:
        best_seq, best_score = _lp_refine(best_seq,
                                          t=0.12,
                                          max_iters=30,
                                          deadline=lp_deadline)
        print(f"[lp‑refine] length={len(best_seq)}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – scale to larger lengths and repeat LP refinement
    # ------------------------------------------------------------------
    lengths = [best_n]
    for factor in [1.3, 1.6, 2.0, 2.8]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths[1:]:
        if time.time() >= end_time - 100.0:
            break
        # give each scaled run a modest slice of the remaining budget
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 130.0)

        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = _normalize(a_scaled)

        refined_seq, refined_peak = _lp_refine(a_scaled,
                                               t=0.12,
                                               max_iters=20,
                                               deadline=time.time() + slice_time)
        if refined_peak < best_score:
            best_score = refined_peak
            best_seq = refined_seq.copy()
            best_n = target_n
            print(f"[scale‑refine] n={target_n:5d}  improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final Adam optimisation (≈ 30 s)
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:
        remaining = end_time - time.time() - 5.0
        a_opt, peak = _adam_optimize(best_seq,
                                      time.time() + remaining,
                                      rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = best_seq.shape[0]
            print(f"[adam] length={best_n}  best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.15)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[local] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
