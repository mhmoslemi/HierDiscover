Algorithm
---------
1. **Problem size** – pick n≈12000 (larger n gives a finer search space).
2. **Warm start** – cheap 2‑parameter grid search on a reduced problem

      * compute the full self‑convolution by FFT,

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). For the normalised sum this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Stage 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """
    Two‑parameter family:
        a_i = (i+1)^(-p) * exp(-λ·i / small_n)
    Returns best sequence and its score.
    """
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0   # typical range up to ~6
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a)=max_k (a*a)_k .
    Uses the exact sub‑gradient from the currently maximal index.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    base_step = 0.6          # initial step magnitude (tuned empirically)

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))
        n = a.size
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # diminishing step size
        step = base_step / np.sqrt(it + 1.0)
        a_candidate = a - step * g
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        val = _score(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()
            a = a_candidate
            # modestly increase step when we make progress
            base_step *= 1.02
        else:
            # no improvement – shrink step size a bit
            base_step *= 0.90

        # occasional jitter to escape flat plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Subgrad jitter it={it} score={best_val:.6f}")

        if it % 500 == 0:
            print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.6f} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – deterministic mass‑transfer refinement (more aggressive)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of mass from the index that contributes most to the
    current maximal convolution entry to the index that contributes the
    least, accepting only moves that improve the score.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    max_frac = 0.06          # fraction of donor mass moved at each step

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution weight for each index to the current max entry
        weight = a[idx] * a[k_max - idx]

        donor = idx[np.argmax(weight)]
        # choose a receiver among the indices with smallest weight (excluding donor)
        mask = idx != donor
        if not np.any(mask):
            break
        receiver = idx[mask][np.argmin(weight[mask])]

        # move a modest fraction of the donor's mass
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        # numerical drift correction
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a *= target_sum / cur_sum

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improve score={best_val:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            # tiny drift correction again
            cur_sum = a.sum()
            if abs(cur_sum - target_sum) > 1e-9:
                a *= target_sum / cur_sum

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the given second‑level time budget and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin for the grader

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)              # the mandatory ℓ¹‑normalisation

    # ------------------------------------------------------------------
    # Warm start (fallback = simple power‑law)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = _project_to_simplex(init, target_sum)
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – quick parametric grid search on a smaller instance
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(20.0,
                                       (deadline - time.time()) * 0.10)
    best_small_seq, _ = _grid_param_search(small_n, small_target, param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = _project_to_simplex(init, target_sum)
        print(f"[{time.time():.2f}] Warm‑start from parametric search (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.65
    a_sub, val_sub = _subgradient_descent(init, target_sum, subgrad_deadline)
    best_seq = a_sub
    best_score = val_sub
    print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – deterministic mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      target_sum,
                                                      deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Evaluate with the provided benchmark (the grader will have evaluate_sequence)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
