def _optimize_lp(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0*n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    # schedule parameters
    t = 0.2
    while time.time() < deadline:
        conv = _full_convolution(a)
        M = conv.max()
        # Build constraint matrix using current a (f)
        A = _build_conv_matrix(a) # shape (2n-1, n)
        # Solve LP max sum(b) s.t. A @ b <= M
        c = -np.ones(n)
        ub = M * np.ones(2*n - 1)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # fallback: break
            break
        b = res.x
        sum_b = b.sum()
        if sum_b < 1e-12:
            # nothing, break
            break
        # scale to sum S
        g = b * (S / sum_b)
        # combine
        # optionally random step size
        a = (1 - t) * a + t * g
        # optional symmetrize
        # a = _symmetrize(a)
        # update best
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # adapt t maybe decay
        t = max(0.01, t * 0.97)

"""
Propose a step‑function with a low self‑convolution peak.

The algorithm has two stages:

1. **Small‑scale linear‑programming refinement**  
   For a modest length (≈2000) we iteratively solve a linear program:
   maximise the ℓ₁‑norm of a new sequence `b` subject to the
   pointwise constraint `conv(f, b) ≤ M`, where `M` is the current
   peak of `conv(f,f)`.  After scaling `b` to the required ℓ₁‑norm we
   replace `f` by this `b`.  The process (a coordinate‑descent on the
   non‑convex problem) quickly drives the peak down.

2. **Upsample & continuous‑optimisation**  
   The best small sequence is linearly interpolated to a larger length
   (8 000–30 000).  After re‑normalising we run a projected Adam optimiser
   that minimises a smooth approximation of the convolution peak.
   Symmetrisation is enforced because the optimal solution is known to be
   mirror‑symmetric.

The whole pipeline respects the 1000 s budget, falling back to a uniform
sequence if needed.  The final sequence is stored in the global
`height_sequence_1` variable and returned as a list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (full length = 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry: v[i] = v[-1-i]."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _downsample_seq(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a smaller length n."""
    if seq.size == n:
        return seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, seq).astype(float)


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for random initialisation (same as in the previous code)
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a variety of analytic families."""
    fam = rng.integers(0, 9)
    if fam == 0 and base_seq is not None:
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                 # symmetric power‑law
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                # beta‑like skewed
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                 # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)                 # monotone decreasing power‑law
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n        # exponential decay
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    else:
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 400.0,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising a smooth approximation of the convolution peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.03
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # smooth weights (soft‑max) over convolution entries
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()

        # gradient of the smooth approximation
        # convolution of a with the reversed weight vector
        rev_w = w[::-1]
        fft_len = 1 << (3 * n - 2 - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        Fw = np.fft.rfft(rev_w, n=fft_len)
        full = np.fft.irfft(Fa * Fw, n=fft_len)[:3 * n - 2]
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998

    return best_a, best_score

def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Construct the matrix A such that A @ b = conv(a, b) (full length)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.empty((m, n), dtype=float)
    A.fill(0.0)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _optimize_lp(a0: np.ndarray,
                 deadline: float,
                 rng: np.random.Generator,
                 enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Iterative LP refinement:
    maximise Σb subject to conv(a, b) ≤ M, where M = max conv(a, a).
    Afterwards scale to the correct ℓ₁‑norm and replace a.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _full_convolution(a)
        M = conv.max()

        # Build the convolution matrix for the current a
        A = _build_conv_matrix(a)

        # Linear programme: maximise Σb  →  minimise -Σb
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # If LP fails we abort the LP phase
            print("[lp] linprog failed:", res.message)
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # Scale to the required ℓ₁‑norm
        g = b * (S / sum_b)
        if enforce_sym:
            g = _symmetrize(g)

        a = g
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[lp] iter={iteration} score={cur_score:.6f}")

        # Stop after a modest number of iterations – the improvement quickly stalls
        if iteration >= 30:
            break

    return best_a, best_score

    # Adam refinement – remaining time (leave a couple seconds for final I/O)
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_ref, sc_ref = _optimize_adam(a_big, deadline, rng,
                                       max_alpha=800.0, enforce_sym=True)
    else:
        a_ref, sc_ref = a_big, np.max(_full_convolution(a_big))

    # ------------------------------------------------------------------
    # Final output
    # ------------------------------------------------------------------
    final_seq = a_ref.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={sc_ref:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
