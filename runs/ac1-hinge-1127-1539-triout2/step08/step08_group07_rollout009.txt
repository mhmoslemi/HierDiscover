"""
Propose a step‐function (non‑negative sequence) with a low
max‑convolution value.  The objective is exactly the benchmark score
when the ℓ¹‑norm is fixed to √(2·n).

The algorithm
---------------
1. Choose a problem size n (≈12k–20k) depending on the time budget.
2. Build a diverse set of initial sequences:
   – power‑law a_i ∝ (i+1)^‑p for several p,
   – exponential a_i ∝ exp(‑λ·i),
   – a resized previous best if present,
   – jittered variants of the above.
3. Phase A – Adam‑style projected gradient descent on a soft‑max
   approximation of the max‑convolution.  β (softmax “temperature”) is
   annealed up to 5·10⁵ and the Adam learning‑rate is scaled by 1/(β+1).
   The best sequence over all restarts is kept.
4. Phase B – Deterministic mass‑transfer moves that directly attack the
   current maximal convolution entry.  The donor is the index with the
   largest contribution a_i·a_{k*‑i}, the receiver a low‑contributing one.
   Moves are accepted with a Metropolis rule whose temperature decays.
5. Phase C – Short high‑β Adam polishing.
6. Optional local smoothing / random perturbations are tried and the
   best result is returned.

All operations respect the simplex {x_i ≥ 0, Σx_i = √(2n)}.
FFT‑based convolutions give O(n log n) per iteration, comfortably
fitting into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using an FFT of given size."""
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * a.size - 1]
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using an FFT of given size."""
    return float(_full_convolution(a, size).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)⁻ᵖ, normalized to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalized to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float, size: int) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, max_conv) where grad has the same shape as a.
    """
    n = a.size
    # forward convolution (full)
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    max_conv = conv_full.max()

    # soft‑max probabilities
    exps = np.exp(beta * (conv_full - max_conv))
    p = exps / exps.sum()

    # gradient = 2 * (a_rev ⋆ p) restricted to the central n entries
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    rev_conv = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * rev_conv[n - 1 : n - 1 + n]
    return grad, max_conv


# ----------------------------------------------------------------------
# Phase A – Adam projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           max_beta: float = 5e5,
                           step_factor: float = 0.5,
                           beta_inc_interval: int = 200,
                           beta_mul: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    Returns the best sequence found and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    best_a = a.copy()
    best_score = _max_convolution(a, size)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    beta = 0.5
    rng = np.random.default_rng()
    step_base = target_sum * step_factor

    while time.time() < deadline:
        grad, cur_max = _gradient_softmax(a, beta, size)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate scaled by 1/(beta+1) – higher beta ⇒ smaller steps
        lr = step_base / (np.sqrt(v_hat) + eps) / (beta + 1.0)

        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter_std = target_sum * 0.001 / (beta + 1.0)
            a += rng.normal(scale=jitter_std, size=n)
            a = _project_to_simplex(a, target_sum)

        # evaluate (max‑convolution is the score because Σa = √(2n))
        cur_score = cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it:5d} β={beta:6.1f} score={best_score:.6f}")

        # increase soft‑max temperature
        if it % beta_inc_interval == 0 and beta < max_beta:
            beta = min(beta * beta_mul, max_beta)

    return best_a, best_score


def _gradient_descent_restarts(init: np.ndarray,
                               target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short Adam runs with random jitter; keep the best result.
    """
    best_a = init.copy()
    best_score = _max_convolution(best_a, 1 << ((2 * best_a.size - 1).bit_length()))
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = _project_to_simplex(np.maximum(start, 0.0), target_sum)

        a, val = _adam_gradient_descent(start,
                                         target_sum,
                                         time.time() + run_budget,
                                         max_beta=5e5,
                                         step_factor=0.6,
                                         beta_inc_interval=300,
                                         beta_mul=2.0)
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly reduce the maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    conv = _full_convolution(a, size)
    cur_score = conv.max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    it = 0

    while time.time() < deadline:
        # temperature schedule (exponential decay)
        tau = (time.time() - start_time) / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # actual contributions a_i * a_{kmax-i}
        contrib = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(contrib)]
        # choose a low‑contributing receiver
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(contrib)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bounded transfer amount
        max_delta = a[donor] * 0.4  # move at most 40 % of donor mass
        weight_diff = contrib[np.argmax(contrib)] - contrib[np.argmin(contrib)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.3)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _full_convolution(a, size)
        new_score = conv.max()

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to escape deep plateaus
        if it % 3000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(np.maximum(a, 0.0), target_sum)
            conv = _full_convolution(a, size)
            cur_score = conv.max()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Optional local smoothing / random perturbations
# ----------------------------------------------------------------------


def _smooth_and_project(a: np.ndarray,
                        target_sum: float,
                        sigma: float = 1.0) -> np.ndarray:
    """Gaussian smoothing (window size ≈6σ) followed by simplex projection."""
    radius = int(math.ceil(3.0 * sigma))
    x = np.arange(-radius, radius + 1, dtype=np.float64)
    kernel = np.exp(-(x ** 2) / (2.0 * sigma ** 2))
    kernel /= kernel.sum()
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.0)
    return _project_to_simplex(a_smooth, target_sum)


def _random_local_search(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Tiny random moves (hill climbing) for a few seconds."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, 1 << ((2 * a.size - 1).bit_length()))
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick a handful of indices and perturb them
        idx = rng.choice(n, size=5, replace=False)
        delta = rng.normal(scale=target_sum * 0.001, size=5)
        a_candidate = best_a.copy()
        a_candidate[idx] += delta
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        score = _max_convolution(a_candidate,
                                 1 << ((2 * n - 1).bit_length()))
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            print(f"[{time.time():.2f}] Random local improved score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the benchmark score.
    The routine respects the given time budget (seconds) and returns a list
    of heights (float).  It combines multi‑start Adam gradient descent,
    deterministic mass‑transfer refinement and a final polishing phase.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # build initial candidates
    # ------------------------------------------------------------------
    init_seqs = []

    # power‑law families
    for p in (0.44, 0.46, 0.48, 0.5, 0.52, 0.54, 0.56, 0.58, 0.6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # exponential families
    for lam in (0.0004, 0.0006, 0.001, 0.0015, 0.002):
        init_seqs.append(_exp_sequence(n, lam, target_sum))

    # resized previous best, if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered variants for diversity
    rng = np.random.default_rng()
    jitter_scale = target_sum * 0.008
    for seq in list(init_seqs):
        jitter = rng.normal(scale=jitter_scale, size=n)
        seq_jit = seq + jitter
        seq_jit = _project_to_simplex(np.maximum(seq_jit, 0.0), target_sum)
        init_seqs.append(seq_jit)

    # ------------------------------------------------------------------
    # Phase A – Adam gradient descent (multi‑restart)
    # ------------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.35
    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= phase_a_deadline:
            break
        cand, val = _gradient_descent_restarts(init, target_sum, phase_a_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New global best after GD phase = {best_score:.6f}")

    # fallback sanity
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.52, target_sum)
        best_score = _max_convolution(best_seq,
                                      1 << ((2 * n - 1).bit_length()))

    # ------------------------------------------------------------------
    # Phase B – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      target_sum,
                                                      deadline,
                                                      start_temp=0.5,
                                                      end_temp=1e-4,
                                                      low_frac=0.03)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – final Adam polishing (high β)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                target_sum,
                                                deadline,
                                                max_beta=5e5,
                                                step_factor=0.4,
                                                beta_inc_interval=250,
                                                beta_mul=2.0)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional smoothing / local random search (tiny remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # try a few smoothing scales
        for sigma in (0.5, 1.0, 1.5):
            a_smooth = _smooth_and_project(best_seq, target_sum, sigma=sigma)
            score = _max_convolution(a_smooth,
                                     1 << ((2 * a_smooth.size - 1).bit_length()))
            if score < best_score:
                best_score = score
                best_seq = a_smooth.copy()
                print(f"[{time.time():.2f}] Smoothing σ={sigma} improved best={best_score:.6f}")

        # short random local search
        a_loc, val_loc = _random_local_search(best_seq, target_sum,
                                               time.time() + min(remaining - 2.0, 12.0))
        if val_loc < best_score:
            best_score = val_loc
            best_seq = a_loc.copy()
            print(f"[{time.time():.2f}] Random local search final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # post‑processing: truncate trailing zeros for brevity
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (visible in the log, not required by harness)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
