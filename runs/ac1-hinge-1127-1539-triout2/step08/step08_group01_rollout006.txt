"""
A hybrid LP‑+‑hill‑climb optimiser for the max‑self‑convolution problem.

* Initialise from the previous construction (if available) or from a random
  on‑simplex seed.
* For a series of increasing lengths we:
    – up‑sample the current best seed,
    – run an LP‐based “mass spreading’’ iteration (the classic LP step from
      the reference paper) which repeatedly solves
        maximise Σ b_j  subject to  b≥0,  f∗b ≤ M,
      where M = max(f∗f) for the current f.
      The resulting b is rescaled to the original ℓ¹‑norm and mixed into the
      current f.
    – finish the stage with a cheap biased hill‑climb that keeps the full
      self‑convolution up‑to‑date in O(n) per move.
* A final polishing (LP + hill‑climb) is performed with any remaining time.
* The wall‑clock budget (default 1000 s) is respected; the best sequence found
  is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Optional CVXPY import (used for the linear‑programming step)
# ------------------------------------------------------------
try:
    import cvxpy as cp
except Exception:                     # pragma: no cover
    cp = None

# ------------------------------------------------------------
# Basic linear‑algebra helpers
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Draw a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(cut)),
                            np.exp(-lam2 * np.arange(cut, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length |a| to new_n, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ------------------------------------------------------------
# LP based “mass‑spreading’’ step
# ------------------------------------------------------------

def _solve_lp_for_g(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   subject to  b≥0  and  a∗b ≤ M
    (elementwise).  Returns the optimiser b (or None on failure).
    """
    if cp is None:                     # pragma: no cover
        return None

    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    a_const = cp.Constant(a.astype(float))
    conv_expr = cp.conv(a_const, b)          # length 2n‑1
    constraints = [conv_expr <= M]           # elementwise
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=500, abstol=1e-6, reltol=1e-6)
    except Exception as exc:                # pragma: no cover
        print(f"LP solve exception: {exc}")
        return None

    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)


def _lp_iterative_improvement(a0: np.ndarray,
                              max_iters: int,
                              deadline: float) -> np.ndarray:
    """
    Repeatedly apply the LP step described in the paper.
    Mixing parameter t decreases geometrically.
    """
    a = a0.copy()
    cur_M = _max_convolution_fft(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        b_opt = _solve_lp_for_g(a, cur_M)
        if b_opt is None:
            break

        sum_b = b_opt.sum()
        sum_a = a.sum()
        if sum_b <= 0 or sum_a <= 0:
            break

        # Rescale b to have the same ℓ¹‑norm as a
        g = sum_a * b_opt / sum_b

        # Decreasing mixing weight
        t = 0.4 * (0.9 ** it)
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)

        new_M = _max_convolution_fft(a_new)

        # Accept only if the bound truly drops
        if new_M + 1e-12 < cur_M:
            a = a_new
            cur_M = new_M
        else:
            # No improvement – keep the current a and continue with a smaller t
            if t < 1e-4:
                break                      # stagnation

    return a

def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that keeps the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # bias destination towards small counterpart
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b
