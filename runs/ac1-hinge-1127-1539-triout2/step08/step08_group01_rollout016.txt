"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Strategy
--------
1.  Work with a single length (the length of the provided seed, or 2000 if
    none).  A smaller length makes the inner optimisation cheap, allowing
    thousands of independent local searches within the 1000 s budget.
2.  For each trial we initialise a random non‑negative vector on the
    ℓ¹‑simplex Σa = √(2 n) (several families are used – exponential,
    power‑law, Dirichlet, piecewise exponential, etc.).
3.  A two‑phase local optimiser is applied:
    * **Phase A – Smoothed‑max Adam descent** – minimises a
      log‑sum‑exp surrogate of the max‑convolution.  The β‑parameter for
      the soft‑max is annealed up to 10⁶ so that the optimiser eventually
      targets the true maximum.
    * **Phase B – Biased hill‑climbing** – operates on the exact
      self‑convolution and repeatedly moves a small amount of mass from a
      source index (chosen proportionally to its contribution to the
      currently worst convolution entry) to a destination index that
      contributes little to that entry.  The move is accepted with a
      Metropolis rule; the temperature cools slowly.
4.  The best sequence over all trials is kept.  After the time budget is
    exhausted a final polishing pass (Adam + hill‑climb) is run on the
    current best.
5.  Trailing near‑zero entries are trimmed before returning.

The algorithm spends the majority of the wall‑clock time on cheap
local optimisation (≈ 5 s per trial), thus exploring thousands of starts
and typically beating the previous bound of 1.5083.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation on the smooth‑max surrogate.
    The learning‑rate is scaled to the simplex radius.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base step‑size (empirically chosen)
    base_lr = 0.30 * target_sum

    # smoothed‑max parameters
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of the soft‑max surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0004 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisers (various families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _perturb(a: np.ndarray, rng, scale: float) -> np.ndarray:
    """Add small Gaussian noise and re‑project onto the simplex."""
    n = a.size
    target = np.sum(a)            # already = √(2 n)
    a2 = a + rng.normal(scale=scale * target, size=n)
    a2 = np.maximum(a2, 0.0)
    a2 = _project_to_simplex(a2, target)
    return a2


# ----------------------------------------------------------------------
# Biased hill‑climbing on the exact convolution (O(n) updates)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9        # initial temperature (a bit higher)
    cooling = 0.9998                  # very slow cooling

    while time.time() < deadline:
        # --------  choose a worst convolution position ------------
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        M = rng.choice(max_idxs) if max_idxs.size else 0

        # -------- source index: proportionally to its contribution ------------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # -------- destination index: bias towards small counterpart ----------
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # -------- amount of mass to move ----------
        max_delta = a[i] * 0.85
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # -------- update convolution (O(n) slices) ----------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # -------- Metropolis acceptance ----------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Local optimiser – Adam followed by hill‑climb
# ----------------------------------------------------------------------


def _local_optimize(a_start: np.ndarray, target: float, budget: float) -> np.ndarray:
    """Run a short Adam optimisation then hill‑climbing."""
    now = time.time()
    if now + 0.5 >= budget:
        return a_start

    # split the allotted time (≈40 % Adam, 60 % hill‑climb)
    gd_time = (budget - now) * 0.4
    hc_time = (budget - now) - gd_time - 0.01
    gd_deadline = now + gd_time
    a_mid = _adam_gradient_descent(a_start, target, gd_deadline)

    # hill‑climbing on the result
    if time.time() + 0.1 < budget:
        b_mid = _full_convolution(a_mid)
        hc_deadline = min(budget - 0.05, time.time() + hc_time)
        a_end, _ = _hill_climbing_improved(a_mid, b_mid, hc_deadline)
        return a_end
    else:
        return a_mid

    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

            n = 2000
            target_sum = math.sqrt(2.0 * n)
            best_seq = _random_initial_sequence(n, rng)
            best_score = _max_convolution_fft(best_seq)
    else:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = _random_initial_sequence(n, rng)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] No seed – random start (n={n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main exploratory loop – many independent restarts
    # ------------------------------------------------------------------
    trial = 0
    min_batch_time = 5.0          # seconds allocated to each trial
    while time.time() < deadline - min_batch_time:
        trial += 1
        # initialise (sometimes perturb the current best, otherwise random)
        if rng.random() < 0.25:
            a0 = _perturb(best_seq, rng, scale=0.02)
        else:
            a0 = _random_initial_sequence(n, rng)

        # run a short local optimisation
        trial_deadline = min(deadline - 0.1, time.time() + min_batch_time)

        if cur_score < best_score - 1e-12:

            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Trial {trial:04d}: new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the best sequence with the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] Final polishing (remaining time ≈ {deadline - time.time():.1f}s)")
        # a longer Adam run
        final_deadline = deadline - 1.5
        best_seq = _adam_gradient_descent(best_seq, target_sum, final_deadline)
        # short hill‑climb
        b_final = _full_convolution(best_seq)
        best_seq, _ = _hill_climbing_improved(best_seq, b_final, deadline - 0.5)

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]

    # Report the official evaluation

    try:

        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:          # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
