```python
"""
Propose candidate for the convolution benchmark.

Algorithm:
1. Choose a problem size n depending on the time budget and fix the L1 norm
   to √(2·n) – then the benchmark score equals the maximal entry of the
   self‑convolution.
2. Generate a diverse set of initial sequences (power‑law, exponential,
   combined power‑law) and keep the best few.
3. Phase A – projected gradient descent on a soft‑max approximation of the
   maximum convolution (as in the previous state‑of‑the‑art).
4. Phase B – direct sub‑gradient descent on the true max‑convolution
   (the gradient of max (a*a) is simply 2·rev(a) shifted to the
   maximal index).  This aggressively reduces the worst entry.
5. Phase C – deterministic mass‑transfer refinement that equalises
   contributions w_i = a_i + a_{k−i}.
6. Phase D – a short high‑β projected gradient descent polishing pass.
All steps respect the simplex constraint via Euclidean projection.
The routine respects the 1000 s budget and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (2·n·max(a*a) / (∑a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^{-p} sequence normalised to target_sum."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, r: float, target_sum: float) -> np.ndarray:
    """Geometric decay r**i (0<r<1), normalised to target_sum."""
    a = r ** np.arange(n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _combined_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """
    Product of left‑ and right‑hand power laws:
        a_i = (i+1)^{-p} * (n-i)^{-q}
    Then normalised to target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * (n - i) ** (-q)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of a soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Projected gradient descent (with restarts)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_ext(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 1e5,
    step_factor: float = 0.7,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Several short gradient‑descent runs with random jitter.
    Keeps the best result found before the given deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer(

    temperature decays exponentially with the remaining time.

    conv = _conv_fft(a)

    it = 0

        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]

            it += 1

        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:

            if new_score < best_score:

            a[receiver] -= delta

            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score 2·n·max(a*a) / (∑a)²."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^{-p} sequence, normalised to target_sum."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, r: float, target_sum: float) -> np.ndarray:
    """Geometric decay r**i (0<r<1), normalised to target_sum."""
    a = r ** np.arange(n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _combined_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    """
    Product of left‑ and right‑hand power laws:
        a_i = (i+1)^{-p} * (n-i)^{-q},
    then normalised to target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * (n - i) ** (-q)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of a soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b

    for r in (0.985, 0.99, 0.995, 0.998):
        init_seqs.append(_exponential_sequence(n, r, target_sum))

    for p, q in ((0.6, 0.2), (0.4, 0.4), (0.2, 0.6)):
        init_seqs.append(_combined_sequence(n, p, q, target_sum))

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    rng = np.random.default_rng()
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # keep the few most promising seeds (cheap evaluation)
    scores = [_score(s) for s in init_seqs]
    keep = np.argsort(scores)[:5]
    init_seqs = [init_seqs[i] for i in keep]

    # ------------------------------------------------------------------
    # Phase A – soft‑max GD
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.25
    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        a_cand, val = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
        if val < best_score:
            best_score = val
            best_seq = a_cand.copy()
            print(f"[{time.time():.2f}] After GD phase best={best_score:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase B – hard‑max sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.30
        a_sub, val_sub = _hard_max_subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] After sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum, start_temp=0.5, end_temp=1e-4, low_frac=0.05
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After DMT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – final high‑β polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # make the sequence available for later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
