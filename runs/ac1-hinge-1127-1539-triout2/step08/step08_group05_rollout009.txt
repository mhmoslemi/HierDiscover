def _full_convolution(a):
    n = a.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv
```

def _upsample_seq(base, n):
    if base.size == n: return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)

"""
Propose a step‑function sequence with minimal self‑convolution peak.

Algorithm
---------
1.  Initialise a candidate by sampling from several analytic families
    (exponential, beta‑like, sinusoidal, logistic, etc.).  Length is
    chosen in the 20k‑30k range; symmetry is enforced with 60 % probability.
    The vector is normalised so that Σa = sqrt(2·n).

2.  Projected Adam optimisation on the smooth approximation of the max‑convolution.
    The soft‑max temperature α is gradually increased up to a large value
    (≈2000) while the Adam learning rate decays.  Symmetry and the simplex
    (non‑negative, fixed sum) are enforced after each update.

3.  Targeted local refinement (simulated‑annealing style):
    * Find the index k where the convolution is maximal.
    * Identify the pair (i,j) contributing most to that peak.
    * Transfer a small amount of mass from i (and its mirror) to a random
      index outside the peak’s support, keeping the total sum constant.
    * Accept the move if it lowers the peak, otherwise accept with a
      temperature‑controlled probability.  The temperature linearly cools
      to zero.

4.  Optional up‑sampling (≈25 % increase, capped at 30 000) and a short
    Adam + refinement pass to exploit the extra degrees of freedom.

5.  The best sequence found is returned (and stored in the global
    variable ``height_sequence_1``).  Its external score is printed.

The whole search respects the 1000 s budget and works with only NumPy
and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the non‑negative simplex with total sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(a*a) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = 3 * n - 2                      # len(a) + len(w_rev) - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)            # positions giving a*a terms
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i]=v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families for the initial sampling stage
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int, base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using several parameterised families."""
    fam = rng.integers(0, 10)
    if fam == 0 and base_seq is not None:
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                         # uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)            # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                 # symmetric power‑law
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                # beta‑like skew
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                 # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)                 # monotone decreasing power‑law
        a = (np.arange(n, 0, -1) ** p)
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n        # exponential decay
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n           # logistic S‑curve
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    else:
        # piecewise linear tent with random apex
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 2000.0,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1
    alpha_mul = 1.05
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                if enforce_sym:
                    a = _symmetrize(a)
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99999

    return best_a, best_score


def _local_peak_refine(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local refinement.
    Moves small mass from indices that contribute most to the current
    convolution peak to indices that do not, accepting worsening moves
    with probability exp(-Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        remaining = deadline - start
        conv = _full_convolution(a)
        cur_max = conv.max()
        kmax = int(conv.argmax())

        # indices that contribute to this peak
        i_low = max(0, kmax - (n - 1))
        i_high = min(n - 1, kmax)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = kmax - idx_i
        contrib = a[idx_i] * a[idx_j]

        # pick the index with the largest contribution
        max_pos = np.argmax(contrib)
        i = int(idx_i[max_pos])

        # choose a target index outside the contributing range
        left = np.arange(0, i_low) if i_low > 0 else np.empty(0, dtype=int)
        right = np.arange(i_high + 1, n) if i_high < n - 1 else np.empty(0, dtype=int)
        candidates = np.concatenate((left, right))
        if candidates.size == 0:
            break
        j = int(rng.choice(candidates))

        # amount of mass to transfer (a few percent of a[i])
        delta = a[i] * rng.uniform(0.005, 0.02)
        if delta > a[i]:
            delta = a[i] * 0.5
        if delta <= 0:
            continue

        cand = a.copy()
        i_sym = n - 1 - i
        j_sym = n - 1 - j

        # transfer from i (and its mirror) to j (and its mirror)
        cand[i] -= delta
        if i_sym != i:
            cand[i_sym] -= delta
        if cand[i] < 0 or cand[i_sym] < 0:
            continue
        cand[j] += delta
        if j_sym != j:
            cand[j_sym] += delta

        if enforce_sym:
            cand = _symmetrize(cand)

        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand, S)

        new_max = np.max(_full_convolution(cand))
        diff = new_max - cur_max

        # temperature schedule (linear cooling)
        T0 = max(0.1 * cur_max, 1e-4)
        T = T0 * max(0.0, 1.0 - elapsed / remaining)

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()

    return best_a, best_val

            print(f"[upsample‑adam] improved to {best_score:.6f}")

        # final local refinement on the up‑sampled vector
        now2 = time.time()
        if now2 < deadline:
            a_big_ref, sc_big_ref = _local_peak_refine(best_seq, deadline, rng,
                                                       enforce_sym=True)
            if sc_big_ref < best_score:
                best_score = sc_big_ref
                best_seq = a_big_ref.copy()
                print(f"[upsample‑refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the final result (store globally)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
