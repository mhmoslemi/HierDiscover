def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())

def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log Σ exp(β·b_k)  where
    b = a⋆a (full convolution).  Returns the tuple (grad, b) with:

        grad_i = 2·∑_k p_k·a_{k-i}   (where p_k ∝ exp(β·b_k))
    """
    n = a.shape[0]

    # full convolution
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a ⋆ p  (the gradient factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b

def _family_exponential(n: int, rng) -> np.ndarray:

    right = r2 ** np.arange(n - b, dtype=float)

def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a source index to a target index
    while keeping the full convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2 + 1e-9             # initial temperature
    cooling = 0.998

    while time.time() < deadline:
        # pick one of the positions that currently attain the maximum
        idx_max = np.flatnonzero(b == cur_max)
        if idx_max.size == 0:
            break
        k = rng.choice(idx_max)

        # source index – pick among contributors to this entry, weighted by product size
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # target index – bias towards positions with small current mass
        if rng.random() < 0.35:
            inv = 1.0 / (a + 1e-12)
            probs_t = inv / inv.sum()
            j = rng.choice(n, p=probs_t)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.75
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (exact O(n) slices) -----
        # Removing contribution from source i
        #   b[k']  -= 2*delta*a[k'-i]   for k' = i … i+n-1
        b[i:i + n] -= 2.0 * delta * a
        #   self‑term at 2i
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        # Adding contribution from target j
        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        # correction for the cross term at i+j (appears twice)
        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # final accurate convolution for the best candidate (re‑compute)
    b_best = _full_convolution(best_a)
    return best_a, b_best
