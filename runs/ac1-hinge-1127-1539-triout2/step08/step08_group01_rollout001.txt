    data = np.empty(size, dtype=np.float64)

        row_idx[pos:pos + n] = rows_i
        col_idx[pos:pos + n] = i

"""
Propose a candidate step‑function that (approximately) minimises
``evaluate_sequence``.

The algorithm is based on the LP‑based “kernel update’’ described in the
statement:

* start from a feasible non‑negative vector f of length n whose ℓ¹‑norm is
  √(2 n) (an exponential seed or the previous construction);
* compute M = max(f * f);
* solve the linear program  

      maximise   Σ g_j
      subject to 0 ≤ g_j
                  (f * g)_k ≤ M   for all k

  (the convolution is linear in g);
* rescale g to the required ℓ¹‑norm and replace f by a convex
  combination   f ← (1‑t) f + t g  (t≈0.1);
* repeat the LP‑step while time permits;
* finally run a short projected‑Adam smooth‑max descent and a biased
  hill‑climb to polish the result.

All steps respect the 1000 s wall‑clock budget.  The code prints progress
information and returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.

    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used for polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).

    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """Projected Adam minimisation of the smooth‑max surrogate."""
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.25 * target_sum
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_exponential_initial(n: int, rng) -> np.ndarray:
    """Exponential decay seed, projected onto the required simplex."""
    lam = rng.exponential(scale=0.02)
    a = np.exp(-lam * np.arange(n))
    target = math.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence onto a new length, then project."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Biased hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        # pick a worst convolution position
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: bias towards small counterpart a[M‑j]
        counterpart = np.zeros(n)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        inv = 1.0 / (counterpart + 1e-12)
        probs_j = inv / inv.sum()
        j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Linear program that maximises Σ g_j  under   (f * g)_k ≤ M
# ----------------------------------------------------------------------


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   Σ g_j
        subject to 0 ≤ g_j,
                   (f * g)_k ≤ M   for all k = 0,…,2n‑2

    Returns the optimal g (or None on failure).
    """
    n = f.size
    # Build the (2n‑1) × n constraint matrix.
    # For each column i we have the coefficient f[i] on rows i … i+n‑1.
    size = n * n
    row_idx = np.empty(size, dtype=np.int32)
    col_idx = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows_i = np.arange(i, i + n, dtype=np.int32)          # length n
        row_idx[pos:pos + n] = rows_i
        col_idx[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)           # maximise sum(g)  → minimise -sum
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})
        if res.success:
            return res.x
        else:
            print(f"[LP] failed: {res.message}")
            return None
    except Exception as e:
        print(f"[LP] exception: {e}")
        return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Choose a working length (larger n usually gives a lower bound,
    # but the LP becomes O(n²).  2000 is a comfortable compromise.
    # ------------------------------------------------------------------
    n = 2000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise f
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            f = _upsample_sequence(prev, n)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (len={prev.size})")
        except Exception as e:
            print(f"Seeding error: {e}")
            f = _random_exponential_initial(n, rng)
    else:
        f = _random_exponential_initial(n, rng)

    f = _project_to_simplex(f, target_sum)

    best_f = f.copy()
    best_M = _max_convolution_fft(best_f)
    print(f"[{time.time():.1f}] Initial max_b = {best_M:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Main LP‑driven iteration
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 5.0 and iteration < 150:
        iteration += 1
        M = _max_convolution_fft(f)

        # Solve LP to get a direction g
        g0 = _solve_lp_max_sum(f, M)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break

        # Rescale g to the required ℓ¹‑norm
        g = g0 * (target_sum / sum_g0)

        # Convex combination (small step)
        t = 0.1
        f_new = (1.0 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)

        M_new = _max_convolution_fft(f_new)

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new.copy()
            print(f"[{time.time():.1f}] LP iter {iteration:3d}: improved max_b = {best_M:.6f}")

        # Accept only non‑worsening updates (simple greedy policy)
        if M_new <= M:
            f = f_new
        else:
            # occasional probabilistic acceptance to escape flat regions
            T = max(best_M * 0.05, 1e-4)
            if rng.random() < math.exp(-(M_new - M) / T):
                f = f_new

    # ------------------------------------------------------------------
    # Polishing: Adam smooth‑max descent followed by hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # Adam (projected) optimisation
        extra_deadline = deadline - 2.0
        f_polished = _adam_gradient_descent(best_f, target_sum, extra_deadline)
        M_polished = _max_convolution_fft(f_polished)
        if M_polished < best_M:
            best_M = M_polished
            best_f = f_polished
            print(f"[{time.time():.1f}] Adam polish improvement: max_b = {best_M:.6f}")

    if time.time() < deadline - 5.0:
        b_cur = _full_convolution(best_f)
        hill_deadline = deadline - 1.0
        best_f, b_final = _hill_climbing_improved(best_f, b_cur, hill_deadline)
        best_M = b_final.max()
        print(f"[{time.time():.1f}] Hill‑climb final max_b = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    nz = np.where(best_f > 1e-12)[0]
    if nz.size:
        best_f = best_f[: nz[-1] + 1]
    else:
        best_f = best_f[:1]

    # ------------------------------------------------------------------
    # Final evaluation (the official score)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_f.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_f.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_f.tolist()
