def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.1

    if budget_s >= 800:
        n = 15000 maybe? But previous used 12000. Let's keep 12000.
    else: n = 8000 or 5000.

    target_sum = np.sqrt(2.0 * n)
    # Build grid candidates
    candidate_seqs = []

    # power law
    for p in np.arange(0.35, 0.85, 0.025):
        a = _power_law_sequence(n, p, target_sum)
        candidate_seqs.append(a)
    # shifted power law
    for shift in [0.0, 0.5, 1.0, 2.0]:
        p = 0.5
        a = (np.arange(1, n+1) + shift) ** (-p)
        a *= target_sum / a.sum()
        candidate_seqs.append(a)

    # exponential
    for lam in np.arange(0.0001, 0.0011, 0.0002):
        a = np.exp(-lam * np.arange(n))
        a *= target_sum / a.sum()
        candidate_seqs.append(a)

    # combination: mix two power laws
    for p1, p2, w in [(0.4, 0.7, 0.5), (0.45, 0.6, 0.7), (0.5, 0.8, 0.3)]:
        a1 = (np.arange(1, n+1) ** (-p1))
        a2 = (np.arange(1, n+1) ** (-p2))
        a = w * a1 + (1-w) * a2
        a *= target_sum / a.sum()
        candidate_seqs.append(a)

    # previous best if exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=np.float64)
        candidate_seqs.append(_resize_sequence(prev, n))

    # Evaluate each candidate quickly using _score (without evaluating evaluate_sequence)
    scores = []
    for a in candidate_seqs:
        sc = _score(a)
        scores.append(sc)
    # Choose top K best
    K_best = 5
    top_idx = np.argsort(scores)[:K_best]
    best_seq = None
    best_score = np.inf

    # Phase A: gradient descent for each top candidate
    # allocate fraction time
    remaining = deadline - time.time()
    gd_time = remaining * 0.55
    gd_deadline = time.time() + gd_time

    for idx in top_idx:
        if time.time() >= gd_deadline:
            break
        init_a = candidate_seqs[idx]
        # run gradient descent
        a_opt, sc_opt = _gradient_descent_with_restarts(init_a, target_sum, gd_deadline)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.2f}] New best after GD: {best_score:.6f}")

    # Phase B: deterministic mass transfer
    if time.time() < deadline:
        a_ref, sc_ref = _deterministic_mass_transfer(best_seq, deadline, target_sum,
                                                    start_temp=1.0, end_temp=1e-3,
                                                    low_frac=0.08)
        if sc_ref < best_score:
            best_score, best_seq = sc_ref, a_ref.copy()
            print(f"[{time.time():.2f}] New best after mass transfer: {best_score:.6f}")

    # Phase C: final polish with max gradient descent
    if time.time() + 2.0 < deadline:
        final_seq, final_score = _final_max_gradient_descent(best_seq, target_sum, deadline)
        if final_score < best_score:
            best_score, best_seq = final_score, final_seq.copy()
            print(f"[{time.time():.2f}] Final polish best: {best_score:.6f}")

    # Trim zeros
    arr = best_seq
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1]+1]
    final = arr.tolist()
    try:
        val = evaluate_sequence(final)
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final

def _max_gradient_descent(a0, target_sum, deadline, max_iter=5000, step_factor=0.1):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    best = a.copy()
    best_val = _score(a)
    it=0
    while time.time() < deadline and it < max_iter:
        # compute convolution and max index
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax = int(conv.argmax())
        # gradient for max entry
        n = a.shape[0]
        grad = np.zeros_like(a)
        # compute contributions only where valid
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)

        idx = np.arange(lo, hi+1)
        rev_idx = kmax - idx

        # step size: we can approximate Lipschitz ~ 2*max(a), but we can use diminishing step
        step = step_factor * target_sum / (it+1)**0.5

    return best, best_val

"""
Multi‑phase optimizer for the step‑function benchmark.

The algorithm proceeds in three main stages:

1. **Coarse shape search** – a cheap grid over a few parametric families
   (power‑law, shifted power‑law, exponential and simple mixtures) is
   evaluated at the target ℓ¹‑norm.  The best few shapes are kept as seeds.

2. **Projected gradient descent** – each seed is refined by a momentum
   projected‑gradient method that uses a soft‑max approximation of the
   max‑convolution.  The temperature (β) is annealed up to 5e5, the step
   size adapts to the momentum norm and occasional jitter prevents
   stagnation.  Several short restarts are performed within the allocated
   time budget.

3. **Deterministic mass‑transfer & max‑gradient polishing** – a greedy
   mass‑transfer scheme directly attacks the current maximal convolution
   entry, using the product contributions as a guide.  Finally a
   lightweight gradient descent on the *true* max‑convolution (β = ∞)
   further cleans the solution.

All iterates are projected onto the simplex
{ x≥0 , Σx = √(2·n) } after each update.  The routine respects the total
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring (objective = max convolution entry after ℓ¹‑normalisation)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective value = 2·n·max(a*a)/(Σa)² = max(a*a) after normalisation."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (for the coarse grid search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p), normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _shifted_power_law_sequence(n: int, p: float, shift: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1+shift)^(-p), normalised."""
    a = ((np.arange(1, n + 1, dtype=np.float64) + shift) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam * i), normalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _mix_two_power_laws(n: int, p1: float, p2: float, w: float, target_sum: float) -> np.ndarray:
    """Weighted mixture w·i^{-p1} + (1-w)·i^{-p2}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = w * (i ** (-p1)) + (1.0 - w) * (i ** (-p2))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (grad, conv) where grad = ∂_{a} softmax_β(conv)
    and conv is the full linear convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)                # soft‑max weights

    # grad = 2 * (a_rev) ⋆ p   (convolution with reversed a)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max (Phase A)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max of the convolution.
    Returns best sequence encountered and its score.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake off plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        # anneal temperature
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform several short gradient‑descent runs from jittered starts.
    Keeps the best result before deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start,
                                            time.time() + run_budget,
                                            target_sum,
                                            max_beta=5e5,
                                            step_factor=0.7,
                                            gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.8,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly target the current
    maximal convolution entry.  Acceptance follows a simple annealed
    Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    # initial convolution
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of current max convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributions w_i = a_i * a_{kmax-i}
        w = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(w)]
        # choose a receiver among the lowest contributors
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # amount to transfer – a fraction of the donor’s mass,
        # bounded also by the contribution gap
        max_delta = a[donor] * 0.5
        gap = w[np.argmax(w)] - w[np.argmin(w)]
        if gap > 0:
            max_delta = min(max_delta, gap * 0.3)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to escape deep plateaus
        if it % 3000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Final polishing with the true max‑gradient (β = ∞) (Phase C)
# ----------------------------------------------------------------------


def _max_gradient_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          step_factor: float = 0.03,
                          max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent where the gradient is taken
    from the *current* maximal convolution entry (no soft‑max).
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax = int(conv.argmax())

        n = a.size
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rev_idx = kmax - idx

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[rev_idx]          # sub‑gradient of the max entry

        # Diminishing step size
        step = step_factor * target_sum / np.sqrt(it + 1)

        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Max‑Grad it={it:5d} best={best_val:.6f}")

        # occasional tiny jitter to avoid exact stagnation
        if it % 1000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the given time budget and uses the multi‑phase
    strategy described above.
    """
    start = time.time()
    deadline = start + budget_s - 0.2          # small safety margin

    # --------------------------------------------------------------
    # Problem size selection
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Phase 0 – coarse grid of parametric families
    # --------------------------------------------------------------
    cand_seqs = []

    # power‑law family
    for p in np.arange(0.35, 0.81, 0.025):
        cand_seqs.append(_power_law_sequence(n, p, target_sum))

    # shifted power‑law
    for shift in (0.0, 0.5, 1.0, 2.0):
        cand_seqs.append(_shifted_power_law_sequence(n, 0.5, shift, target_sum))

    # exponential decays
    for lam in np.arange(0.0002, 0.0013, 0.0002):
        cand_seqs.append(_exponential_sequence(n, lam, target_sum))

    # mixtures of two power laws

        (0.45, 0.6, 0.6),
        (0.5, 0.8, 0.3),
        (0.42, 0.55, 0.7),

    for p1, p2, w in mixtures:

    # previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        cand_seqs.append(_resize_sequence(prev, n))

    # evaluate scores (very cheap)
    scores = [_score(seq) for seq in cand_seqs]
    top_k = 5
    top_idx = np.argsort(scores)[:top_k]

    # --------------------------------------------------------------

    remaining = deadline - time.time()
