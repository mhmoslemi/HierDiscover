def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3

    # Determine base length
    base_n = 8000
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=float)
        base_n = int(prev.shape[0] * 1.0)  # maybe keep same length
        base_seq = prev.copy()
    else:
        base_seq = None

    # Helper to create a random initial sequence of given length
    def random_seq():
        n = base_n
        a = _sample_family(rng, n, base_seq)
        a = _symmetrize(a)
        S = np.sqrt(2.0 * n)
        a = a / a.sum() * S
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        return a

    # Initialize population
    pop_size = 7
    pop = []
    pop_scores = []
    for i in range(pop_size):
        a = random_seq()
        score = np.max(_full_convolution(a))
        pop.append(a)
        pop_scores.append(score)
    # sort population by score
    sorted_indices = np.argsort(pop_scores)
    pop = [pop[i] for i in sorted_indices]
    pop_scores = [pop_scores[i] for i in sorted_indices]

    attempts = 0
    best_seq = pop[0]
    best_score = pop_scores[0]
    print(f"[init] best_score={best_score:.6f} n={base_n}")

    # Evolutionary loop
    while time.time() < deadline - 5.0:
        # generate child
        # select two parents from top half
        idx1 = rng.integers(0, max(1, pop_size // 2))
        idx2 = rng.integers(0, max(1, pop_size // 2))
        p1 = pop[idx1]
        p2 = pop[idx2]
        alpha = rng.random()
        child = alpha * p1 + (1 - alpha) * p2

        # mutation: move small mass
        n = child.shape[0]
        i_src = rng.integers(0, n)
        i_dst = rng.integers(0, n)
        if i_src != i_dst and child[i_src] > 0:
            delta = 0.03 * child[i_src] * rng.random()
            child[i_src] -= delta
            child[i_dst] += delta

        child = _symmetrize(child)

        # project to simplex
        S = np.sqrt(2.0 * n)
        child = child / child.sum() * S

        # local refinement: short Adam
        now = time.time()
        if now < deadline - 10:
            sub_dead = now + min(0.15, (deadline - now) * 0.1)
            child, child_score = _optimize_adam(child, sub_dead, rng,
                                                max_alpha=1500.0,
                                                enforce_sym=True)
        else:
            child_score = np.max(_full_convolution(child))

        attempts += 1

        # replace worst if better
        if child_score < pop_scores[-1]:
            pop[-1] = child
            pop_scores[-1] = child_score
            # re-sort
            sorted_idx = np.argsort(pop_scores)
            pop = [pop[i] for i in sorted_idx]
            pop_scores = [pop_scores[i] for i in sorted_idx]
            # update global best
            if child_score < best_score:
                best_score = child_score
                best_seq = child.copy()
                print(f"[improve] attempt={attempts} score={best_score:.6f}")

        # occasional full Adam on best
        if attempts % 200 == 0 and time.time() < deadline - 10:

            best_seq, new_score = _optimize_adam(best_seq, t_best_dead, rng,
                                                max_alpha=2000.0, enforce_sym=True)
            if new_score < best_score:
                best_score = new_score
                print(f"[best adam] attempt={attempts} new best {best_score:.6f}")

    # After loop, final refinement: upsample and Adam

"""
Propose a step‑function sequence that minimises the self‑convolution peak.
The method combines a small evolutionary population, symmetric Adam
optimisation of a smooth max‑approximation, occasional sub‑gradient
refinements and a final up‑sampling + long Adam run.
All sequences are forced to be symmetric and to satisfy
    Σ a_i = sqrt(2·n)               (⇒ evaluation = max convolution).
The algorithm respects the 1000 s budget and stores the best sequence
globally as ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (non‑negative, fixed L1 sum)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector of length 2·n‑1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices that give a*a term
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry: v[i] = v[-1‑i]."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Families for random initialisation (all subsequently symmetrised)
# ----------------------------------------------------------------------
def _sample_family(rng: np.random.Generator, n: int,
                   base_seq: np.ndarray | None) -> np.ndarray:
    """Return a random candidate of length n using a variety of analytic families."""
    fam = rng.integers(0, 10)
    if fam == 0 and base_seq is not None:
        a = _upsample_seq(base_seq, n)               # up‑sample existing good solution
    elif fam == 1:
        a = rng.random(n)                             # uniform random
    elif fam == 2:
        rate = rng.uniform(0.93, 0.999)               # symmetric geometric decay
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 3:
        p = rng.uniform(0.2, 4.0)                     # symmetric power‑law
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        p1 = rng.uniform(0.1, 5.0)                    # beta‑like skewed
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:
        p = rng.uniform(0.5, 4.0)                     # sinusoidal power
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:
        p = rng.uniform(0.3, 3.0)                     # monotone decreasing power‑law
        a = np.arange(n, 0, -1) ** p
    elif fam == 7:
        lam = rng.uniform(0.001, 0.02) * n            # exponential decay
        a = np.exp(-lam * np.arange(n) / n)
    elif fam == 8:
        mid = rng.uniform(0.3, 0.7) * n                # logistic S‑curve
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(steep * (np.arange(n) - mid) / n))
    else:
        # piecewise linear tent with random apex location
        apex = rng.integers(1, n - 1)
        left = np.linspace(0, 1, apex, endpoint=False)
        right = np.linspace(1, 0, n - apex)
        a = np.concatenate([left, right])

    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_alpha: float = 800.0,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a smooth approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature (smooth‑max) schedule
    alpha = 0.1
    alpha_mul = 1.025
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998   # slow decay

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum convolution entry.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random aspirational move
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit if stagnating
        if it % 7000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _anneal_hillclimb(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that swaps tiny masses.
    Accepts worse moves with probability exp(-Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    cur_max = best_score
    start = time.time()
    T0 = max(0.1 * cur_max, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.05 * a[i] * rng.random()
        delta = min(delta, a[i])

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_max = np.max(_full_convolution(cand))
        diff = cand_max - cur_max

        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_a = cand.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function (list of floats) that
    minimises the evaluation function. The method uses:
      • symmetric random initialisation from several analytic families,
      • a small evolutionary population with crossover & mutation,
      • projected Adam optimisation on a smooth max‑approximation,
      • occasional sub‑gradient refinement,
      • a final up‑sampling + long Adam run.
    The best sequence is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Determine base length (use previous solution if available)
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # If we have a previous solution, keep its length; otherwise pick a
    # moderate length that balances quality and speed.
    if base_seq is not None:
        base_n = max(4000, int(base_seq.shape[0]))
    else:
        base_n = 8000

    # ------------------------------------------------------------------
    # Helper to create a fresh random symmetric candidate of length n
    # ------------------------------------------------------------------
    def _random_candidate():
        a = _sample_family(rng, base_n, base_seq)
        a = _symmetrize(a)
        S = np.sqrt(2.0 * base_n)
        a = a / a.sum() * S
        return a

    # ------------------------------------------------------------------
    # Initialise a small population
    # ------------------------------------------------------------------
    pop_size = 7
    population = []
    pop_scores = []
    for _ in range(pop_size):
        a = _random_candidate()
        score = np.max(_full_convolution(a))
        population.append(a)
        pop_scores.append(score)

    # sort by score (ascending)
    order = np.argsort(pop_scores)
    population = [population[i] for i in order]
    pop_scores = [pop_scores[i] for i in order]

    best_seq = population[0].copy()
    best_score = pop_scores[0]
    attempts = 0
    print(f"[init] length={base_n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Evolutionary loop (crossover + short Adam refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:   # keep a few seconds for final phase
        # --- parent selection (top half of the sorted population) ---
        top = max(1, pop_size // 2)
        i1 = rng.integers(0, top)
        i2 = rng.integers(0, top)
        p1 = population[i1]
        p2 = population[i2]

        # --- crossover (convex combination) ---
        alpha = rng.random()
        child = alpha * p1 + (1.0 - alpha) * p2

        # --- mutation: transfer a tiny amount of mass ---
        n = child.shape[0]
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src != dst and child[src] > 0:
            delta = 0.03 * child[src] * rng.random()
            child[src] -= delta
            child[dst] += delta

        child = _symmetrize(child)
        S = np.sqrt(2.0 * n)
        child = child / child.sum() * S

        # --- a brief Adam refinement (≈0.12 s or 10 % of remaining time) ---
        now = time.time()
        if now < deadline - 6.0:
            sub_dead = now + min(0.12, (deadline - now) * 0.10)
            child, child_score = _optimize_adam(child, sub_dead,
                                                rng,
                                                max_alpha=1500.0,
                                                enforce_sym=True)
        else:
            child_score = np.max(_full_convolution(child))

        attempts += 1

        # --- replace worst if improvement ---
        if child_score < pop_scores[-1]:
            population[-1] = child
            pop_scores[-1] = child_score
            order = np.argsort(pop_scores)
            population = [population[i] for i in order]
            pop_scores = [pop_scores[i] for i in order]

            if child_score < best_score:
                best_score = child_score
                best_seq = child.copy()
                print(f"[improve] attempt={attempts} score={best_score:.6f}")

        # --- occasional deeper Adam on the current best ---
        if attempts % 250 == 0 and time.time() < deadline - 6.0:
            t_best = time.time() + min(0.4, (deadline - time.time()) * 0.25)
            best_seq, refined_score = _optimize_adam(best_seq, t_best,
                                                     rng,
                                                     max_alpha=2000.0,
                                                     enforce_sym=True)
            if refined_score < best_score:
                best_score = refined_score
                print(f"[best‑adam] attempt={attempts} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional sub‑gradient polishing (short window)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 4.0:
        t_sub = now + min(0.6, (deadline - now) * 0.4)
        best_seq, sub_score = _optimize_subgrad(best_seq, t_sub, rng,
                                                enforce_sym=True)
        if sub_score < best_score:
            best_score = sub_score
            print(f"[subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Upsampling + long Adam refinement (uses the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.5:
        # increase length by ≈30 % but not beyond 30000
        cur_n = best_seq.shape[0]
        new_n = min(30000, int(cur_n * 1.30))
        if new_n != cur_n:
            best_seq = _upsample_seq(best_seq, new_n)
            best_seq = _symmetrize(best_seq)
            S_big = np.sqrt(2.0 * new_n)
            best_seq = best_seq / best_seq.sum() * S_big

        # final Adam run on the enlarged sequence
        t_final = time.time() + remaining - 0.5
        best_seq, final_score = _optimize_adam(best_seq, t_final, rng,
                                               max_alpha=2500.0,
                                               enforce_sym=True)
        best_score = final_score

    # ------------------------------------------------------------------
    # Return the final result (and store globally)
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    external_score = evaluate_sequence(final_list)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_list)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_list, dtype=float)
    return final_list
