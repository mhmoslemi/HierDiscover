def _family_gaussian(n, rng):
    mu = rng.uniform(0.3*n, 0.7*n)
    sigma = rng.uniform(0.1*n, 0.4*n)
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - mu)/sigma)**2)
    return a

def _family_triangular(n, rng):
    peak = rng.uniform(0.3, 0.7)
    # create linear increase to peak then decrease
    # approximate: shape similar to triangle lying in [0,1]
    left_len = int(peak * n)
    right_len = n - left_len
    left = np.linspace(0, 1, left_len, endpoint=False)
    right = np.linspace(1, 0, right_len, endpoint=False)
    a = np.concatenate([left, right])
    return a

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Random parametric families** – generate many candidates (exponential,
   power‑law, linear, piece‑wise exponential, Gaussian, triangular, mixed‑exp)
   and project them onto the ℓ¹‑simplex {Σa_i = √(2n)}.

2. **Projected Adam** – minimise a smooth approximation of the max
   (`log‑sum‑exp`) with a geometric schedule for the smoothing parameter β.
   After each step the vector is projected back onto the simplex.

3. **Deterministic max‑descent** – repeatedly locate the index k where the
   self‑convolution attains its maximum and move a fraction of mass from a
   contributing source index (large a_{k‑i}) to a target index that does not
   affect the current maximal entry (i.e. where k‑j is out of range).  The
   vector is re‑projected onto the simplex after every move.

4. **Exact hill‑climbing (simulated‑annealing)** – a fast O(n) update of the
   full convolution while swapping a random amount of mass between a source
   and a target.  Accepted moves lower the maximum; otherwise they are
   accepted with probability exp(−Δ/T) with a slowly cooling temperature.

The time budget (default 1000 s) is split among the phases; the final
polishing gives the remaining time to a long run of deterministic descent
followed by a short hill‑climb.  The best sequence found (trimmed of trailing
zeros) is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (FFT, returns 2·n‑1 entries)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (computed via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed maximum  (1/β)·log Σ exp(β·b_k)  where
    b = a⋆a (full convolution).  Returns ``(grad, b)`` with::

        grad_i = 2·∑_k p_k·a_{k-i}
    """
    n = a.shape[0]

    # full convolution a⋆a
    size1 = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a ⋆ p  (the gradient factor)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    ar_f = np.fft.rfft(a_rev, n=size2)
    p_f = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_f * p_f, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 2.0,
                   beta_target: float = 1e5,
                   beta_factor: float = 1.25) -> np.ndarray:
    """
    Adam optimiser on the smoothed‑max objective.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = target_sum * 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    step = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        a = _project_to_simplex(a, target_sum)

        if step % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if beta < beta_target and step % 250 == 0:
            beta = min(beta * beta_factor, beta_target)

    return a

# Parametric families for random initialisation
# ----------------------------------------------------------------------


def _family_exponential(n: int, rng) -> np.ndarray:
    """a_i ∝ r^i   (0<r<1)"""
    r = rng.uniform(0.96, 0.9995)
    return r ** np.arange(n, dtype=float)


def _family_power_law(n: int, rng) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}"""
    p = rng.uniform(0.2, 1.5)
    return np.arange(1, n + 1, dtype=float) ** (-p)


def _family_linear(n: int, rng) -> np.ndarray:
    """Linear decay from a0 to 0."""
    a0 = rng.uniform(0.5, 2.0)
    return np.linspace(a0, 0.0, n, dtype=float)


def _family_piecewise_exp(n: int, rng) -> np.ndarray:
    """Two‑segment exponential decay."""
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    left = r1 ** np.arange(b, dtype=float)
    right = r2 ** np.arange(n - b, dtype=float)
    return np.concatenate([left, right])


def _family_gaussian(n: int, rng) -> np.ndarray:
    """Gaussian‑shaped bump."""
    mu = rng.uniform(0.3 * n, 0.7 * n)
    sigma = rng.uniform(0.08 * n, 0.25 * n)
    x = np.arange(n, dtype=float)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _family_triangular(n: int, rng) -> np.ndarray:
    """Triangular window, peak somewhere inside the support."""
    peak_frac = rng.uniform(0.3, 0.7)
    peak = int(peak_frac * n)
    left = np.linspace(0, 1, peak, endpoint=False)
    right = np.linspace(1, 0, n - peak, endpoint=False)
    return np.concatenate([left, right])


def _family_mixed_exp(n: int, rng) -> np.ndarray:
    """Weighted sum of two exponentials."""
    r1 = rng.uniform(0.95, 0.999)
    r2 = rng.uniform(0.90, 0.98)
    w = rng.uniform(0.3, 0.7)
    idx = np.arange(n, dtype=float)
    return w * (r1 ** idx) + (1 - w) * (r2 ** idx)


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Pick a random family, add tiny jitter, and project onto the simplex."""
    families = [_family_exponential, _family_power_law, _family_linear,
                _family_piecewise_exp, _family_gaussian,
                _family_triangular, _family_mixed_exp]
    a = families[rng.integers(len(families))](n, rng)
    a += rng.normal(scale=0.001, size=n)          # tiny jitter to avoid ties
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Deterministic max‑descent (sub‑gradient on the actual max)
# ----------------------------------------------------------------------


def _max_descent(a0: np.ndarray,
                 deadline: float,
                 move_fraction: float = 0.15) -> np.ndarray:
    """
    Reduce the maximal entry of the self‑convolution by repeatedly moving
    mass from a contributing source to an index that does not affect the
    current maximal entry.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        k = int(np.argmax(b))                     # index attaining the maximum

        # indices that contribute to b[k]
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range

        # pick source: index with largest partner a_{k-i}
        src_idx = i_range[np.argmax(a[j_range])]

        # pick a target outside the influencing interval (gradient = 0)
        out_left = np.arange(0, max(0, k - (n - 1)), dtype=int)
        out_right = np.arange(min(n, k + 1), n, dtype=int)
        out_candidates = np.concatenate([out_left, out_right])
        if out_candidates.size > 0:
            tgt_idx = int(rng.choice(out_candidates))
        else:
            tgt_idx = int(rng.integers(0, n))

        if src_idx == tgt_idx or a[src_idx] < 1e-14:
            continue

        # move a fraction of the source mass
        delta = move_fraction * a[src_idx]
        delta = min(delta, a[src_idx] * 0.9)

        a[src_idx] -= delta
        a[tgt_idx] += delta

        # re‑project onto the simplex to keep the ℓ¹‑norm exact
        a = _project_to_simplex(a, target_sum)

    return a

                j = int(rng.integers(0, n))

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- exact O(n) convolution update -----
        b[i:i + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] += delta * delta

        b[j:j + n] += 2.0 * delta * a
        if 2 * j < b.size:
            b[2 * j] += delta * delta

        kij = i + j
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_max = b.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling
        move_counter += 1

        # occasional full recomputation to keep the convolution exact
        if move_counter % recompute_interval == 0:
            b = _full_convolution(a)
            cur_max = b.max()

    b_best = _full_convolution(best_a)
    return best_a, b_best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock limit and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously discovered construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            score0 = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = score0
            print(f"[{time.time():.1f}] seed from height_sequence_1 (n={n0}) score={score0:.6f}")
        except Exception as e:
            print(f"Unable to seed from height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Length schedule – gradually increase support size
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000, 50000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1) Random parametric sampling
        # --------------------------------------------------------------
        n_samples = 20
        for _ in range(n_samples):
            if time.time() >= deadline:
                break
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] random family improvement (n={n}) score={score:.6f}")

        # --------------------------------------------------------------
        # 2) Adam optimisation on smoothed‑max
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        gd_time = min(remaining * 0.30, 200.0)
        gd_deadline = time.time() + gd_time

        if best_seq is not None and best_seq.size != n:
            a_start = _upsample_sequence(best_seq, n)
        else:
            a_start = _random_initial_sequence(n, rng)

        a_adam = _adam_optimize(a_start, gd_deadline,
                                beta_start=2.0,
                                beta_target=1e5,
                                beta_factor=1.3)
        score_adam = _max_convolution_fft(a_adam)
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.1f}] Adam improvement (n={n}) score={score_adam:.6f}")

        # --------------------------------------------------------------
        # 3) Deterministic max‑descent
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 3.0:
            break
        dd_time = min(remaining * 0.20, 150.0)
        dd_deadline = time.time() + dd_time

        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()
        a_dd = _max_descent(a_cur, dd_deadline,
                            move_fraction=0.12)
        score_dd = _max_convolution_fft(a_dd)
        if score_dd < best_score:
            best_score = score_dd
            best_seq = a_dd.copy()
            print(f"[{time.time():.1f}] max‑descent improvement (n={n}) score={score_dd:.6f}")

        # --------------------------------------------------------------
        # 4) Exact hill‑climbing (simulated annealing)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 4.0:
            break
        hc_time = min(remaining * 0.30, 250.0)
        hc_deadline = time.time() + hc_time

        if best_seq is None or best_seq.size != n:
            a_cur = _random_initial_sequence(n, rng)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = float(b_hc.max())
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb improvement (n={n}) score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – maximise the remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final deterministic descent (most of remaining time)
        a_final = _max_descent(best_seq.copy(),
                               deadline - 4.0,
                               move_fraction=0.08)

        # a short Adam run to smooth any remaining irregularities
        a_final = _adam_optimize(a_final,
                                 deadline - 3.0,
                                 beta_start=4.0,
                                 beta_target=2e5,
                                 beta_factor=1.5)

        # final brief hill‑climb
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final,
                                          deadline - 2.0)
        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy return value
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (max‑convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
