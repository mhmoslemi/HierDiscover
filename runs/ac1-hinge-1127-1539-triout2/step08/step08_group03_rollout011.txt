Pseudo:

def project_to_simplex(v, S):
    # v: array of real numbers
    # Return w >=0, sum w = S
    # If S>0 else all zero.
    n = len(v)
    # shift to make sum exactly S
    # According to algorithm: sort v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

but careful to handle floating.

"""
Improved adversarial step‑function optimizer.

The algorithm proceeds in three stages (all fit into the 1000 s budget).

1️⃣  Fast random exploration – many vectors are sampled from a variety of
    parametric families, projected onto the simplex 
    Σa_i = √(2 n).  The best few candidates are kept.

2️⃣  Multi‑scale refinement – the best candidate is up‑sampled through a
    schedule of lengths (up to ≈ 5·10⁴).  For each length a projected‑Adam
    optimiser works on a τ‑soft‑max approximation of the peak of the
    self‑convolution (the true objective after normalisation).

3️⃣  Hard‑max projected SGD – a cheap sub‑gradient method that uses the
    exact location of the maximal convolution value.  It runs until the
    time limit is almost exhausted, giving a final fine‑tuned
    construction.

The final list of heights is stored globally as ``height_sequence_1`` and
returned as a plain Python list.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT (a * a)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = rfft(a, n=f_len)
    conv = irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
    { x | x_i ≥ 0, Σ x_i = S }.
    """
    n = v.shape[0]
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # Duchi et al. algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of τ‑soft‑max (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = rfft(mask, n=f_len)
    Fa_rev = rfft(a_rev, n=f_len)
    conv = irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Minimise the exact max‑convolution of ``a`` under the ℓ₁‑norm
    constraint using a τ‑soft‑max approximation.
    Returns the best vector seen and its *hard* peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0  (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution & gradient (soft max) -------
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)          # horizontal shift
    mix = rng.uniform(0.0, 1.0)              # background weight
    raw = mix * np.ones(n) + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)          # decay rate
    mix = rng.uniform(0.0, 1.0)
    raw = mix * np.ones(n) + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))  # low background
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian_bump(n: int, rng: np.random.Generator) -> np.ndarray:
    """Single Gaussian‑like bump placed near the left side."""
    idx = np.arange(n)
    center = rng.integers(0, n // 3)
    sigma = rng.uniform(n / 40.0, n / 10.0)
    raw = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    # add a tiny constant background to keep positivity
    raw += rng.uniform(0.0, 0.02)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ~0.25 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = math.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    else:  # gaussian
        raw = _init_gaussian_bump(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Projected SGD on the exact max (hard‑max) – cheap final fine tuner
# ----------------------------------------------------------------------
def _projected_sgd_max(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       lr0: float = 0.5,
                       lr_decay: float = 0.99995,
                       beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true peak of a*a.
    The iterate stays on the simplex Σa_i = √(2 n) via Euclidean projection.
    Returns the best vector seen and its peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)

    # start from a feasible point
    a = _project_onto_simplex(a0.copy(), S)

    best = a.copy()
    best_val = _max_convolution(a)

    # momentum buffer
    v = np.zeros_like(a)

    lr = lr0
    it = 0
    cur_b = _full_convolution_fft(a)
    cur_val = cur_b.max()
    cur_k = int(np.argmax(cur_b))

    while time.time() < deadline:
        it += 1

        # ---- exact gradient of the current maximal entry ----
        lo = max(0, cur_k - (n - 1))
        hi = min(cur_k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[cur_k - idx]

        # ---- momentum step ----
        v = beta * v - lr * g
        a = a + v
        a = _project_onto_simplex(a, S)

        # ---- evaluate new point (one FFT per loop) ----
        cur_b = _full_convolution_fft(a)
        cur_val = cur_b.max()
        cur_k = int(np.argmax(cur_b))

        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()

        # decay learning rate
        lr *= lr_decay

        # occasional jitter to shake out of flat regions
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * S, size=n)
            a = _project_onto_simplex(best + jitter, S)
            v[:] = 0.0
            cur_b = _full_convolution_fft(a)
            cur_val = cur_b.max()
            cur_k = int(np.argmax(cur_b))

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows three stages (exploration → Adam refinement →
    hard‑max SGD).  The best construction is stored in the global variable
    ``height_sequence_1`` and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # a little safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if present
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 20 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.20 * budget_s
    best_seq = None
    best_score = np.inf
    top_k = 10
    top_candidates = []   # list of (score, seq)

    attempt = 0
    while time.time() < explore_deadline:
        attempt += 1
        # pick a length – give a bias to the ‘medium’ range
        if rng.random() < 0.7:
            n = int(rng.integers(2000, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain a small elite set
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Safety net – ensure we have at least one candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement on the current elite
    # ------------------------------------------------------------------
    max_len = min(50000, best_n * 8)   # allow a fairly large final support
    # geometric schedule – a few intermediate scales
    lengths = np.geomspace(best_n, max_len, num=7, dtype=int)
    lengths = sorted(set(lengths))

    # we will focus on the single best candidate (others are kept as fall‑backs)
    cur_seq = best_seq.copy()
    cur_n = len(cur_seq)

    for target_n in lengths:
        remaining = deadline - time.time()
        if remaining < 30.0:
            break

        # upscale / downscale to the target length
        a_scaled = _scale_sequence(cur_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

        # allocate a slice of the remaining budget (capped to keep some time for final SGD)
        slice_budget = min(remaining * 0.45, 120.0)

        # run Adam with a few different hyper‑parameter choices
        lr0 = rng.choice([0.38, 0.44, 0.52])
        tau_decay = rng.choice([0.9992, 0.9994, 0.9996])
        refined, peak = _adam_optimize(a_scaled,
                                       time.time() + slice_budget,
                                       rng,
                                       lr0=lr0,
                                       tau_decay=tau_decay,
                                       lr_decay=0.99985,
                                       jitter_every=400)

        if peak < best_score:
            best_score = peak
            best_seq = refined.copy()
            best_n = target_n
            print(f"[refine] len={target_n:5d}  improved score={best_score:.6f}")

        # prepare for the next scale
        cur_seq = refined
        cur_n = target_n

    # ------------------------------------------------------------------
    # Phase 3 – hard‑max projected SGD fine‑tuning (takes the remainder)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        final_seq, final_peak = _projected_sgd_max(best_seq,
                                                   time.time() + remaining,
                                                   rng,
                                                   lr0=0.5,
                                                   lr_decay=0.9999,
                                                   beta=0.9)
        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_seq.copy()
            print(f"[sgd] final SGD peak={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
