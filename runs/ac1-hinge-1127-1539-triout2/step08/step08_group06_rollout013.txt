def _hill_climb(a, deadline, rng):

    n = len(a)
    best = a.copy(); best_score = _peak_convolution(a)
    cur = a.copy()
    cur_score = best_score

    start = time.time()
    # parameters
    initial_temp = 0.2
    final_temp = 0.001
    total_time = deadline - start
    it = 0
    while time.time() < deadline:
        it += 1
        # compute current k_max (maybe reuse but we compute after each update)
        # choose move type
        if rng.random() < 0.75:
            # targeted proportional move: reduce high contribution
            conv = _full_convolution(cur)
            k_max = int(np.argmax(conv))
            # compute contributions
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            idx_i = np.arange(i_low, i_high+1)
            idx_j = k_max - idx_i
            products = cur[idx_i] * cur[idx_j]
            i_max = idx_i[products.argmax()]  # index with highest product
            # choose j random not equal i_max
            j = rng.integers(0, n)
            while j == i_max:
                j = rng.integers(0, n)
            # move delta from i_max to j
            if cur[i_max] > 0:
                delta = cur[i_max] * rng.uniform(0.01, 0.05)  # small fraction
            else:
                continue
        else:
            # random swap
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] == 0:
                continue
            delta = cur[i] * rng.uniform(0.01, 0.05)

        cand = cur.copy()
        cand[i_max] -= delta
        cand[j] += delta
        # project onto simplex (ensuring sum)
        S = np.sqrt(2.0 * n)
        cand = _project_simplex(cand, S)

        cand_score = _peak_convolution(cand)
        # acceptance
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # temperature schedule
            elapsed = time.time() - start
            T = initial_temp * (1 - elapsed / total_time) + final_temp * (elapsed / total_time)
            if rng.random() < math.exp(-(cand_score - cur_score) / T):
                cur = cand
                cur_score = cand_score

    return best, best_score

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using FFT (fast for the sizes we use)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Peak of the self‑convolution (the quantity we ultimately minimise)
# ----------------------------------------------------------------------
def _peak_convolution(a: np.ndarray) -> float:
    """Return max_i (a*a)_i."""
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Analytic one‑sided geometric sequence: a_i = C·r^i, 0<r<1.
# λ = -n·log(r) is the more convenient shape parameter.
# ----------------------------------------------------------------------
def _geom_one_sided(n: int, lam: float) -> np.ndarray:
    """Generate a monotone geometric vector of length n with shape λ."""
    # r = exp(-λ/n)  ∈ (0,1)
    r = math.exp(-lam / n)
    # sum_i r^i = (1 - r**n) / (1 - r)
    # we need Σ a_i = √(2·n) → C = √(2·n) / sum_i r^i
    if r == 1.0:               # degenerate (should never happen)
        C = 0.0
    else:
        sum_geom = (1.0 - r ** n) / (1.0 - r)
        C = math.sqrt(2.0 * n) / sum_geom
    return C * r ** np.arange(n)


# ----------------------------------------------------------------------
# Various families used during exploration
# ----------------------------------------------------------------------
def _gen_candidate(n: int, rng: np.random.Generator, base: np.ndarray | None) -> np.ndarray:
    """Generate a random non‑negative vector of length n using several families."""
    fam = rng.integers(0, 9)
    if fam == 0 and base is not None:
        # up‑sample previous best (linear interpolation)
        x_old = np.linspace(0.0, 1.0, base.size)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, base)
    elif fam == 1:
        # pure uniform random
        a = rng.random(n)
    elif fam == 2:
        # one‑sided geometric (λ sampled)
        lam = rng.uniform(0.8, 3.5)          # λ = -n·log r
        a = _geom_one_sided(n, lam)
    elif fam == 3:
        # symmetric geometric decay (as in the previous solution)
        rate = rng.uniform(0.92, 0.9995)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) * 0.5
    elif fam == 4:
        # symmetric power‑law (i·(n‑i))^p
        p = rng.uniform(0.2, 4.5)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 5:
        # beta‑like (x^p1 * (1‑x)^p2)
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 6:
        # sinusoidal power
        p = rng.uniform(0.5, 4.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    else:
        # symmetric logistic bump
        scale = rng.uniform(0.02, 0.15) * n
        mid = (n - 1) / 2.0
        xs = np.arange(n, dtype=float)
        left = 1.0 / (1.0 + np.exp((xs - mid) / scale))
        right = 1.0 / (1.0 + np.exp((mid - xs) / scale))
        a = left + right
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Targeted hill‑climbing / simulated annealing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that repeatedly moves a small chunk of mass from the index
    contributing most to the current maximum convolution entry to a random other
    index.  A temperature schedule allows occasional uphill moves.
    Returns the best vector seen and its peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # keep the best ever seen
    best = a.copy()
    best_score = _peak_convolution(a)

    cur_score = best_score
    start = time.time()
    total = deadline - start
    # temperature schedule (linear cooling)
    T0 = 0.15
    Tf = 0.0005

    while time.time() < deadline:
        # compute the index of the current maximum entry
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # the set of i that contribute to this entry
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        # contribution of each pair (i, j)
        contrib = a[idx_i] * a[idx_j]
        # pick the pair with the largest product
        i_max = idx_i[contrib.argmax()]

        # choose a random destination index different from i_max
        j = rng.integers(0, n)
        while j == i_max:
            j = rng.integers(0, n)

        # move a small fraction of the mass from i_max to j
        if a[i_max] <= 0.0:
            continue
        delta = a[i_max] * rng.uniform(0.01, 0.07)  # between 1 % and 7 %
        cand = a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)

        cand_score = _peak_convolution(cand)

        # acceptance test
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # temperature that decays linearly with elapsed time
            elapsed = time.time() - start
            T = T0 * (1 - elapsed / total) + Tf * (elapsed / total)
            if rng.random() < math.exp(-(cand_score - cur_score) / T):
                a = cand
                cur_score = cand_score

        # occasional random jitter to avoid dead ends
        if rng.random() < 0.012:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                d = a[i] * rng.uniform(0.02, 0.06)
                tmp = a.copy()
                tmp[i] -= d
                tmp[j] += d
                tmp = _project_simplex(tmp, S)
                s2 = _peak_convolution(tmp)
                if s2 < cur_score or rng.random() < 0.12:
                    a = tmp
                    cur_score = s2
                    if s2 < best_score:
                        best = tmp.copy()
                        best_score = s2

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with minimal self‑convolution peak.
    Returns a list of floats (the heights) before the allotted time expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # margin of a few seconds for the very final evaluation / bookkeeping
    deadline = start_time + budget_s - 4.0

    # --------------------------------------------------------------
    # Retrieve a previously saved sequence (if any) for warm‑starts
    # --------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # --------------------------------------------------------------
    # Phase 0 – cheap exploration, keep the K elite vectors
    # --------------------------------------------------------------
    K = 7                         # number of elite candidates kept
    elite = []                    # list of (score, vector)
    attempts = 0
    phase0_deadline = start_time + 0.18 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1

        # choose a length (we vary between 2k and 15k)
        if base_seq is not None and rng.random() < 0.55:
            # resize previous best a little
            scale = rng.uniform(0.85, 1.2)
            n = max(600, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(2000, 15001))

        a = _gen_candidate(n, rng, base_seq)

        # normalise to Σ a = √(2·n)
        S = math.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # optionally enforce a mild symmetry (helps some families)
        if rng.random() < 0.35:
            a = 0.5 * (a + a[::-1])

        cur_score = _peak_convolution(a)

        # keep elite list sorted by score (ascending)
        if len(elite) < K or cur_score < elite[-1][0]:
            elite.append((cur_score, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > K:
                elite.pop()
            print(f"[explore] attempt={attempts} n={n} score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Fallback if nothing survived (unlikely)
    # --------------------------------------------------------------
    if not elite:
        n = 8000
        a = rng.random(n)
        a = a / a.sum() * math.sqrt(2.0 * n)
        elite.append((_peak_convolution(a), a))

    # --------------------------------------------------------------
    # Phase 1 – refinement of each elite via hill‑climbing
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    time_per_elite = remaining / len(elite)

    best_overall_seq = None
    best_overall_score = np.inf

    for idx, (init_score, vec) in enumerate(elite):
        sub_deadline = now + time_per_elite
        refined_vec, refined_score = _hill_climb(vec, sub_deadline, rng)

        if refined_score < best_overall_score:
            best_overall_score = refined_score
            best_overall_seq = refined_vec.copy()
            print(f"[refine {idx}] improved to {best_overall_score:.6f}")

        now = time.time()
        if now >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – optional up‑sampling of the current best
    # --------------------------------------------------------------
    if best_overall_seq is not None:
        max_len = 15000
        for factor in (1.2, 1.5, 1.8, 2.0):
            new_n = int(best_overall_seq.shape[0] * factor)
            if new_n < 600 or new_n > max_len:
                continue
            # linear interpolation to new length
            x_old = np.linspace(0.0, 1.0, best_overall_seq.size)
            x_new = np.linspace(0.0, 1.0, new_n)
            up = np.interp(x_new, x_old, best_overall_seq)

            # renormalise
            S_new = math.sqrt(2.0 * new_n)
            up = up / up.sum() * S_new

            # short hill‑climb on the up‑sampled version
            now = time.time()
            if now >= deadline:
                break
            sub_dead = min(deadline, now + 0.07 * (deadline - now))
            up_ref, up_score = _hill_climb(up, sub_dead, rng)

            if up_score < best_overall_score:
                best_overall_score = up_score
                best_overall_seq = up_ref.copy()
                print(f"[upsample] factor={factor:.2f} n={new_n} improved to {best_overall_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final tiny local search with remaining budget
    # --------------------------------------------------------------
    now = time.time()
    if best_overall_seq is not None and now < deadline:
        final_vec, final_score = _hill_climb(best_overall_seq, deadline, rng)
        if final_score < best_overall_score:
            best_overall_score = final_score
            best_overall_seq = final_vec.copy()
            print(f"[final] improved to {best_overall_score:.6f}")

    # --------------------------------------------------------------
    # Return results (store globally for potential warm‑restart)
    # --------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)  # type: ignore  # provided by the harness
    print(f"[return] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
