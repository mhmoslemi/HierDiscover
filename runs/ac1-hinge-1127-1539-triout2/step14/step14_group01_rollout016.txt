"""
Hybrid optimisation for the minimum peak convolution problem.

Improvements over the previous version:
- Larger population and more aggressive DE parameters (F=0.8, CR=0.9).
- Adaptive population‐restart when progress stalls.
- Longer deterministic mass‑transfer phase and a final random‑local‑search phase.
- Additional seed families (Dirichlet, reversed, random Gaussian jitter).
- More re‑ordering candidates (including the reverse sequence).
The routine respects the 1000 s budget and stores the best sequence in
``height_sequence_1`` for warm‑starting the next run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """
    Objective 2·n·max(a*a) / (∑a)².
    When ∑a = target_sum = sqrt(2n) the value equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – forces Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed factories – diverse families of admissible sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, alpha: float, target_sum: float, rng) -> np.ndarray:
    # Dirichlet with identical concentration parameter α
    a = rng.gamma(alpha, 1.0, size=n)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _gaussian_jitter(seq: np.ndarray, scale: float, rng) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = seq + rng.normal(scale=scale, size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    return _project_to_simplex(noisy, np.sum(seq))


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Re‑ordering helpers (try to spread large values more uniformly)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    s = np.sort(seq)[::-1]
    n = len(seq)
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in s:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step that is coprime to n."""
    n = len(seq)
    s = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(s):
        out[(i * step) % n] = v
    return out


def _reverse_order(seq: np.ndarray) -> np.ndarray:
    return seq[::-1].copy()


# ----------------------------------------------------------------------
# Differential Evolution (projected onto ℓ¹‑simplex)
# ----------------------------------------------------------------------


def _de_optimize(population, target_sum, deadline,
                 F: float = 0.8, CR: float = 0.9,
                 stagnation_limit: int = 12) -> tuple[np.ndarray, float]:
    """
    Classic DE loop with projection onto the ℓ¹‑simplex.
    Restarts a fraction of the population after a period of stagnation.
    """
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind, target_sum) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    generations_without_improve = 0
    while time.time() < deadline:
        improvement_this_gen = False
        for i in range(len(population)):
            # Choose three distinct mates different from i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            # DE/current‑to‑best/1 mutation
            mutant = (population[i] +
                      F * (best_seq - population[i]) +
                      F * (population[a] - population[b]))
            mutant = _project_to_simplex(mutant, target_sum)

            # Binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial, target_sum)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                improvement_this_gen = True
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if improvement_this_gen:
            generations_without_improve = 0
        else:
            generations_without_improve += 1

        # Restart a fraction if stagnating
        if generations_without_improve >= stagnation_limit:
            # replace half of the worst individuals with fresh random simplex points
            sorted_idx = np.argsort(scores)[::-1]  # worst first
            for j in sorted_idx[: len(population) // 2]:
                pop = _random_simplex(n, target_sum, rng)
                population[j] = pop
                scores[j] = _score(pop, target_sum)
            generations_without_improve = 0

        # early exit margin
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (greedy & Metropolis)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-6,
                                low_frac: float = 0.1) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        # choose the larger of the symmetric pair as donor
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        # low‑contribution receiver: take the smallest *low_frac* of products
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = float(conv.max())

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (projected) on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = float(_conv_fft(a).max())
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]

        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 3e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = float(_conv_fft(a).max())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Random local perturbation phase (hill‑climbing + occasional acceptance)
# ----------------------------------------------------------------------


def _random_local_search(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly reshuffle a small random subset of coordinates.
    Accepts only improving moves (simple hill‑climbing).
    """
    a = a0.copy()
    cur_score = _max_convolution(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]

    # how many indices to reshuffle at once – grows with n
    block = max(2, int(n * 0.001))

    while time.time() < deadline:
        # pick a random block of indices
        idx = rng.choice(n, size=block, replace=False)
        # redistribute the total mass of the block uniformly (or random)
        total = a[idx].sum()
        # propose a random Dirichlet allocation inside the block
        proposal = rng.gamma(1.0, 1.0, size=block)
        proposal *= total / proposal.sum()
        a_new = a.copy()
        a_new[idx] = proposal
        # re‑project to ensure global sum (numerical safety)
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _max_convolution(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score

    # Differential Evolution (≈ 65 % of the budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.65 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer (≈ 15 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_dead = time.time() + min(30.0, deadline - time.time() - 8.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.45, end_temp=1e-6, low_frac=0.1)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] MT polished – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent (≈ 12 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sg_dead = time.time() + min(20.0, deadline - time.time() - 6.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_dead, target_sum,
            step_factor=0.48, gamma=0.995)
        if score_sg < best_score:
            best_seq, best_score = seq_sg, score_sg
            print(f"[{time.time():.2f}] Sub‑grad refined – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Random local refinement (use remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        rl_dead = deadline - 2.0
        seq_rl, score_rl = _random_local_search(best_seq, rl_dead, target_sum)
        if score_rl < best_score:
            best_seq, best_score = seq_rl, score_rl
            print(f"[{time.time():.2f}] Random local search – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Reordering – test a few permutations that often help
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        _reverse_order(best_seq)
    ]
    best_seq = min(candidates, key=_max_convolution)
    best_score = _score(best_seq, target_sum)
    print(f"[{time.time():.2f}] After reordering – final peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for warm‑starting the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
