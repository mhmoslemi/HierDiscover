"""
Propose a step‑function (non‑negative vector) with a small evaluation score.

The programme follows a multi‑phase strategy:

1.  Build a large pool of diverse seeds (power‑law, exponential, random Dirichlet,
    linear decreasing, previous best).  All seeds are projected onto the ℓ¹‑simplex
    Σa = √(2·n).

2.  Very fast evaluation of all seeds (with a deterministic alternating ordering)
    to obtain a good starting point.

3.  Long projected gradient descent on a smooth soft‑max approximation of the
    peak of the auto‑convolution.  The temperature β is annealed up to a large
    value (≈2·10⁶) and momentum (Adam‑style) is used.

4.  Upscale the current candidate to a larger length (≈12 k–15 k).  The
    alternating ordering is applied again – this spreads the biggest masses
    as far apart as possible.

5.  Deterministic mass‑transfer simulated annealing (many thousands of local
    moves) which moves a small amount of mass from a “donor” (large contribution
    to the current peak) to a “receiver” (small contribution).

6.  Targeted permutation annealing – swaps high‑contributing indices with
    low‑contributing ones.

7.  A short projected sub‑gradient descent on the exact max‑convolution
    (fast per‑iteration) to polish the final answer.

8.  Trim trailing zeros, store the result globally and return it.

The whole routine respects a 1000 s budget (the harness supplies this value
as `budget_s`).  All intermediate steps print progress information that will
appear in the log.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for large n)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective function used throughout the optimiser:
        2·n·max(a*a) / (Σa)²
    For a vector that is already projected onto the simplex Σa = √(2·n) this
    reduces to the maximal convolution entry.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, second‑largest, … placed alternately from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (smooth approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution over shifts

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Projected Adam‑style gradient descent on the soft‑max peak
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e6,
                               step_factor: float = 0.4,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on a smooth approximation of the peak.
    The temperature β is doubled periodically up to max_beta.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    m = np.zeros_like(a)          # first‑moment estimate
    v = np.zeros_like(a)          # second‑moment estimate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step proportional to the (momentum‑adjusted) gradient norm
        norm = np.linalg.norm(m_hat)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        # β schedule
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on maxᵢ (a*a)ᵢ."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (local annealing)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Local moves that transfer a small amount of mass from a donor (large
    contribution to the current peak) to a receiver (small contribution).
    Acceptance follows a temperature schedule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate current peak and its contributing indices
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # pick a low‑contributor among the bottom `low_frac` of the same window
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5          # we never move more than half the donor
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-6,
                                 top_frac: float = 0.08,
                                 low_frac: float = 0.08) -> tuple[np.ndarray, float]:
    """Swap an index that contributes a lot to the current peak with one that contributes little."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Perm it new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈1000 s) and returns a
    list of heights.  Intermediate progress is printed to stdout.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – basic problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 10000
    elif budget_s >= 500:
        n_expl = 8000
    else:
        n_expl = 6000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – build a diverse seed pool
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # power‑law family (wide range)
    for p in np.linspace(0.2, 2.0, 19):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.001, 0.015, 15):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like seeds
    for _ in range(12):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing seed
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.01, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Phase 2 – quick evaluation of seeds (find a decent starting point)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    explore_deadline = min(deadline - 200.0, start_time + budget_s * 0.2)
    while time.time() < explore_deadline:
        seed = rng.choice(init_pool)

        # try both raw order and alternating order
        for cand in (seed, _alternating_order(seed)):
            cand = _project_to_simplex(cand, target_expl)
            sc = _score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.2f}] Seed eval new best={best_score:.6f}")

        if time.time() >= explore_deadline:
            break

    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.8, target_expl)

    # ------------------------------------------------------------------
    # Phase 3 – long annealed gradient descent on the current best seed
    # ------------------------------------------------------------------
    gd_dead = time.time() + (deadline - time.time()) * 0.45
    a_gd, val_gd = _annealed_gradient_descent(
        best_seq,
        gd_dead,
        target_expl,
        max_beta=2e6,
        step_factor=0.4,
        gamma=0.99,
    )
    if val_gd < best_score:
        best_score, best_seq = val_gd, a_gd.copy()
        print(f"[{time.time():.2f}] After GD new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional upscale to a larger problem size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        # decide final length (use a larger n to gain extra degrees of freedom)
        if budget_s >= 800:
            n_final = 15000
        elif budget_s >= 500:
            n_final = 12000
        else:
            n_final = 10000
        target_final = np.sqrt(2.0 * n_final)

        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)

        # try a couple of systematic reorderings (they often help)
        candidates = [
            seq_big,
            _alternating_order(seq_big),
            _alternating_order(_alternating_order(seq_big)),
        ]
        best_seq = seq_big
        best_score = _score(seq_big)
        for cand in candidates[1:]:
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
        print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_score:.6f}")
        # swap names for later phases
        target_expl = target_final
        n_expl = n_final

    # ------------------------------------------------------------------
    # Phase 5 – intensive deterministic mass‑transfer annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.35
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-5,
            low_frac=0.04,
        )
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] After MT new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – targeted permutation annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        perm_dead = time.time() + (deadline - time.time()) * 0.30
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.5,
            end_temp=1e-6,
            top_frac=0.07,
            low_frac=0.07,
        )
        if val_perm < best_score:
            best_score, best_seq = val_perm, a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – short sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_dead = time.time() + min(4.0, deadline - time.time() - 0.5)
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_expl,
            max_iter=200_000,
        )
        if val_sub < best_score:
            best_score, best_seq = val_sub, a_sub.copy()
            print(f"[{time.time():.2f}] Final sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 8 – final trim and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional logging of the true evaluation
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
```

This implementation expands the search space, allocates more time to the most effective phases (high‑β gradient descent, deterministic mass‑transfer, and permutation annealing), upsizes the problem when the budget permits, and uses an Adam‑style update for more stable convergence.  All numbers are kept non‑negative, projected onto the ℓ¹‑simplex, and the routine respects the 1000 s budget.
