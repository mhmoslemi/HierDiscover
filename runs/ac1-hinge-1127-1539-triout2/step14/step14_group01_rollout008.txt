"""
Hybrid optimizer for the minimum peak convolution problem.

The main idea is to start from analytically‑motivated “Beta‑type’’ shapes
that concentrate mass near the ends of the interval (a_i ∝
[x(1‑x)]^{-p} with p∈(0,1)).  Those shapes are known to produce an almost
flat self‑convolution, i.e. a low peak.  A small grid‑search over p is
performed, the best shape is selected and then refined by a few
projected sub‑gradient steps.  The whole routine respects the 1000 s
budget, adapts the problem size n to the available time and stores the
result in the global variable ``height_sequence_1`` for warm‑starting the
next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used for the small corrective steps
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _uniform_sequence(n: int, target_sum: float) -> np.ndarray:
    """All entries equal – the trivial baseline."""
    a = np.full(n, target_sum / n, dtype=np.float64)
    return a


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (right‑hand tail) and renormalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) and renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _beta_shape_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """
    a_i ∝ [x_i (1‑x_i)]^{-p},  x_i = (i+0.5)/n,  0<p<1.
    This family contains the arcsine law (p=0.5) which gives an almost
    constant self‑convolution.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid exact zero or one – the +0.5 offset already ensures that.
    a = (x * (1.0 - x)) ** (-p)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Simple sub‑gradient descent on the max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (few thousand iterations).
    The objective is max_k (a*a)_k; the gradient is built from all
    indices attaining the maximum.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    n = a.shape[0]
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    # small jitter schedule to avoid stagnation
    def _maybe_jitter(vec: np.ndarray) -> np.ndarray:
        jitter = rng.normal(scale=target_sum * 5e-4, size=vec.shape)
        return _project_to_simplex(vec + jitter, target_sum)

    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices where the maximum is (within tiny tolerance)
        max_idx = np.where(conv >= max_val * (1 - 1e-12))[0]

        # sub‑gradient for the max‑convolution
        grad = np.zeros_like(a)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            partner = k - idx
            # distinct pairs contribute twice, the diagonal once
            grad[idx] += 2.0 * a[partner]
            # correct the diagonal (i=j) contribution
            diag = idx == partner
            if np.any(diag):
                grad[idx[diag]] -= a[partner[diag]]  # we added 2 instead of 1
        # momentum + step size
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 400 == 0:
            a = _maybe_jitter(a)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low peak of its
    self‑convolution.  The routine:
      1. Chooses a problem size n that respects the time budget.
      2. Generates a family of analytically motivated seed shapes
         (uniform, power‑law, exponential, Beta‑type with several exponents).
      3. Picks the best seed according to the exact max‑convolution value.
      4. Refines that seed with a short projected sub‑gradient descent.
      5. Trims trailing zeros and stores the result in ``height_sequence_1``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Choose n – larger n gives more freedom but costs a bit more time.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000           # a moderately large problem
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1.  Seed generation (all seeds are already ℓ¹‑normalised)
    # ------------------------------------------------------------------
    seeds = []

    # uniform baseline
    seeds.append(_uniform_sequence(n, target_sum))

    # power‑law family
    for p in (0.3, 0.6, 0.9, 1.2, 1.5):
        seeds.append(_power_law_sequence(n, p, target_sum))

    # exponential family
    for lam in (0.0005, 0.001, 0.003, 0.006, 0.012):
        seeds.append(_exponential_sequence(n, lam, target_sum))

    # Beta‑type (edge‑heavy) family – the core of the new approach
    for p in (0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85):
        seeds.append(_beta_shape_sequence(n, p, target_sum))

    # warm‑start from previous run if it exists (rescaled if needed)
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            # simple linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        seeds.append(prev)

    # ------------------------------------------------------------------
    # 2. Pick the best seed according to the exact max‑convolution.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in seeds:
        sc = _max_convolution_fft(seq)          # because Σseq = √(2n) ⇒ objective = max conv
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] best seed – peak = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient refinement – run only the time that remains.
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:                         # ensure at least a few seconds for refinement
        refine_deadline = time.time() + time_left * 0.85   # keep a small buffer
        seq_ref, sc_ref = _subgradient_descent(best_seq,
                                                refine_deadline,
                                                target_sum,
                                                step_factor=0.45,
                                                gamma=0.995)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = seq_ref.copy()
            print(f"[{time.time():.2f}] after sub‑grad – peak = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final tidy‑up: trim negligible tail entries.
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for warm‑starting the next invocation
    globals()["height_sequence_1"] = final_seq

    # ------------------------------------------------------------------
    # 5. Report the final score using the provided evaluator (optional).
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
