"""
Hybrid optimisation for the minimum peak auto‑convolution problem.

The search works in three stages:

1. **Seeding + Differential Evolution** – a small DE loop on the ℓ¹‑simplex
   (the ℓ¹‑norm is fixed to √(2 n)).  The DE phase produces a good
   global starting point.

2. **Smooth‑max gradient descent** – we minimise a smooth approximation of
   the peak (λ·log‑sum‑exp of the convolution) using a projected gradient
   method with momentum.  λ is increased from a small to a large value,
   turning the smooth objective into the true max.

3. **Polishing & reordering** – a short deterministic mass‑transfer
   refinement, followed by a few simple permutations (alternating,
   even‑spread) and trimming of trailing zeros.

All vectors are kept non‑negative and projected onto the simplex
∑a = √(2 n) after each update, so the evaluation score equals the
maximum entry of the auto‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with projection onto the ℓ¹‑simplex."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_max_convolution_fft(ind) for ind in population]  # max = objective
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # pick three distinct mates different from i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _max_convolution_fft(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        # early break to leave time for refinements
        if time.time() > deadline - 2.0:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (kept for a final polish)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = float(conv.max())

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (new component)
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    lam_start: float = 0.5,
                    lam_end: float = 40.0,
                    step_factor: float = 0.6,
                    gamma: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth surrogate
    L(a) = 1/λ · log Σ_k exp(λ·(a*a)_k).

    λ is increased from lam_start to lam_end during the run.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    start_time = time.time()

    # FFT size large enough for the cross‑correlation (a_rev * w)
    size = 1 << ((3 * n - 2).bit_length())

    momentum = np.zeros_like(a)

    while time.time() < deadline:
        # schedule λ
        tau = (time.time() - start_time) / (deadline - start_time)
        lam = lam_start * (lam_end / lam_start) ** tau

        # convolution a*a
        conv = _conv_fft(a)
        conv_max = conv.max()

        # smooth weights (soft‑max)
        exp_arg = lam * (conv - conv_max)          # shift for stability
        w_raw = np.exp(exp_arg)
        sum_w = w_raw.sum()
        w = w_raw / sum_w

        # cross‑correlation S[t] = Σ_k w_k·a_{k‑t}
        a_rev = a[::-1]
        fa = np.fft.rfft(a_rev, n=size)
        fw = np.fft.rfft(w, n=size)
        conv_rev = np.fft.irfft(fa * fw, n=size)[: (3 * n - 2)]
        start_idx = n - 1
        S = conv_rev[start_idx:start_idx + n]

        # gradient of the smooth surrogate
        w2t = w[2 * np.arange(n)]
        grad = 2.0 * S - w2t * a

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a = a - lr * momentum

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val
