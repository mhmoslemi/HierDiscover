def smooth_optimize(a_init, target_sum, deadline):
    a = a_init.astype(np.float64).copy()
    n = a.shape[0]
    # projection at start to ensure sum target
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 0
    max_iter = 100000  # should be high enough
    while time.time() < deadline and iteration < max_iter:
        conv = _conv_fft(a)
        max_conv = conv.max()
        # smoothing parameter schedule
        frac = iteration / max_iter
        beta = 0.5 + frac * (200.0 - 0.5)  # linear from 0.5 to 200
        if beta > 200.0:
            beta = 200.0
        # compute w
        shifted = conv - max_conv  # max shift for numerical stability
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        # compute gradient via convolution with reversed w
        w_rev = w[::-1]
        m_len = len(w_rev)
        size = 1 << ((m_len + n - 1).bit_length())
        W_rev = np.fft.rfft(w_rev, n=size)
        A_fft = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(W_rev * A_fft, n=size)[: m_len + n - 1]
        grad = 2.0 * conv_full[(m_len - 1) - np.arange(n)]
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        # learning rate schedule: maybe decreasing
        # base LR determined heuristically
        base_lr = 0.3
        lr = base_lr / (1.0 + 0.001 * iteration)  # decays slowly
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # ensure nonnegativity and project onto simplex
        a = _project_to_simplex(a, target_sum)
        # evaluate
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth opt iter={iteration:5d} score={best_score:.6f}")
        iteration += 1
    return best_a, best_score

"""
Hybrid optimizer for the max‑convolution problem.

The algorithm proceeds in three phases:
1️⃣  Build a small library of deterministic and random starts (power‑law,
    exponential, combined, symmetric beta, warm‑start).
2️⃣  Run a smoothed log‑sum‑exp gradient descent (Adam) on the true
    convolution while projecting onto the simplex {x ≥ 0, Σx = √(2n)}.
    The smoothing parameter β is increased from 0.5 to 200 during the run,
    turning the smooth objective into the true max.
3️⃣  A short aggressive simulated‑annealing “mass‑transfer’’ phase
    (identical to the one from the previous baseline) is applied to
    polish the best candidate.

The whole routine respects the given time budget (default 1000 s) and
stores the final sequence in the global variable ``height_sequence_1``
for warm‑starts in subsequent calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  With the normalisation Σa=√(2n) this
    equals the maximal convolution entry.
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n-1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) * exp(-lam * i/(n-1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape on (0,1)."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (alpha - 1.0)
    )
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp gradient descent (Adam)
# ----------------------------------------------------------------------


def _smooth_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Minimise the maximum entry of a*a using a smoothed objective.
    The method uses Adam optimisation and projects onto the simplex after
    each step.  The smoothing parameter β is increased linearly from 0.5
    to 200 throughout the run.
    """
    a = _project_to_simplex(a0.astype(np.float64).copy(), target_sum)
    n = a.shape[0]

    best_a = a.copy()
    best_val = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Fixed FFT size for the correlation w_rev * a
    corr_len = 3 * n - 2               # (2n-1) + n - 1
    size = 1 << (corr_len.bit_length())  # next power of two

    it = 0
    while time.time() < deadline:
        # ----- current convolution -----
        conv = _conv_fft(a)
        max_conv = conv.max()

        # ----- smoothing weight -----
        beta = 0.5 + (200.0 - 0.5) * (it / 100000.0)  # linear schedule, 100 k iters enough
        if beta > 200.0:
            beta = 200.0
        shifted = conv - max_conv                 # ≤ 0, safe for exp
        w = np.exp(beta * shifted)
        w /= w.sum()                              # w sums to 1

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]                           # length 2n‑1
        # convolution w_rev * a (linear, no wrap‑around)
        W_rev = np.fft.rfft(w_rev, n=size)
        A_fft = np.fft.rfft(a, n=size)
        corr_full = np.fft.irfft(W_rev * A_fft, n=size)[:corr_len]
        # grad_i = 2 * Σ_j w_{i+j} a_j = 2 * conv_rev[2n‑2‑i]
        grad = 2.0 * corr_full[(2 * n - 2) - np.arange(n)]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning rate – slowly decaying
        lr = 0.05 / (1.0 + 1e-4 * it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- feasibility projection -----
        a = _project_to_simplex(a, target_sum)

        # ----- bookkeeping -----
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth‑opt iter={it:6d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Fast O(n) rank‑one update simulated‑annealing that moves a little mass
    from “bad’’ indices (large contribution to the current peak) to
    “good’’ ones (small height).  This implementation is identical to the
    one used in the baseline.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact self‑convolution
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                     # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- O(n) update of the convolution -----
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # normalisation Σa = √(2n)

    # ------------------------------------------------------------------
    # 1️⃣  Build a palette of starting sequences
    # ------------------------------------------------------------------
    candidates = []

    # warm‑start if present
    if "height_sequence_1" in globals():
        w = np.array(globals()["height_sequence_1"], dtype=np.float64)
        w = _resize_sequence(w, n)
        w = w * (target_sum / w.sum())
        candidates.append(w)

    # deterministic families
    for p in np.linspace(0.45, 0.75, 7):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    for lam in np.linspace(0.5, 8.0, 7):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    for _ in range(6):
        p = np.random.random() * 0.8 + 0.1   # 0.1 … 0.9
        lam = np.random.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    for alpha in np.linspace(0.25, 0.9, 6):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    # a few random Dirichlet starts
    rng = np.random.default_rng()
    for _ in range(5):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        candidates.append(rnd)

    # ------------------------------------------------------------------
    # Evaluate and keep the best few candidates
    # ------------------------------------------------------------------
    scored = [( _score(c), c ) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_k = min(3, len(scored))
    best_a = scored[0][1].copy()
    best_score = scored[0][0]
    top_candidates = [c for _, c in scored[:top_k]]

    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed log‑sum‑exp optimisation (Adam)
    # ------------------------------------------------------------------
    while time.time() < deadline and top_candidates:
        remaining = deadline - time.time()
        per = remaining / len(top_candidates) - 0.05
        if per <= 0.0:
            break
        cand = top_candidates.pop(0)
        sub_deadline = time.time() + per
        a_opt, val_opt = _smooth_optimize(cand, target_sum, sub_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after smooth = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final aggressive annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, val_opt = anneal_mass_transfer(best_a, target_sum, deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after anneal = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up, truncation of trailing zeros and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
