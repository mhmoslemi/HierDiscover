def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5

    # Step 1: initial candidate generation
    n0 = 4000
    target = math.sqrt(2*n0)
    # generate pool
    rng = np.random.default_rng()
    pool = []
    for p in np.linspace(0.2, 1.8, 8):
        pool.append(_power_law(n0, p, target))
    for lam in np.linspace(0.001, 0.02, 6):
        pool.append(_exponential(n0, lam, target))
    for _ in range(6):
        pool.append(_random_simplex(n0, target, rng))
    # add prev if exists
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=float)
        if seq.shape[0] >= n0:
            pool.append(_resize_sequence(seq, n0))
        else:
            # upsample to n0
            pool.append(_resize_sequence(seq, n0))

    best_seq = None
    best_val = np.inf

    # run a few annealed gradient descent runs
    while time.time() < deadline - 120:  # leave time for later phases
        seed = rng.choice(pool)
        # optionally jitter
        if best_seq is not None and rng.random() < 0.5:
            seed = best_seq.copy()
            seed += rng.normal(scale=target * 0.01, size=n0)
            seed = np.maximum(seed, 0)
            seed = _project_to_simplex(seed, target)

        # run annealed gradient descent: adjust parameters for more thorough search
        gd_deadline = time.time() + min(15.0, deadline - time.time() - 100)
        a_opt, maxconv = _annealed_gradient_descent(seed, gd_deadline, target,
                                                    max_beta=3e6,
                                                    step_factor=0.8,
                                                    gamma=0.97)
        score = 2 * n0 * maxconv / (target*target)
        if score < best_val:
            best_val = score
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New best after GD: {best_val:.6f}")

    # Step 2: LP refinement on small size using more active constraints
    if best_seq is not None and time.time() < deadline - 100:
        lp_deadline = time.time() + min(30.0, deadline - time.time() - 80)
        best_seq, lp_val = _lp_refine(best_seq, target, max_iters=40, deadline=lp_deadline,
                                      K=30, mix_factor=0.12)
        score = 2 * n0 * _max_convolution_fft(best_seq) / (target * target)
        if score < best_val:
            best_val = score

"""
Hybrid optimizer for the step‑function benchmark (aim ≈ 1.503).

The pipeline is:
  1.  Build a modest‑size (n≈4000) pool of parametric candidates.
  2.  Run several short annealed soft‑max GD passes to obtain a good seed.
  3.  Refine that seed with a small LP (more active constraints than before).
  4.  Upscale the seed to a large problem (n≈15 k–30 k) and spread large values.
  5.  Long exact‑subgradient descent with a *tightening* tolerance schedule.
  6.  Two rounds of permutation optimisation (random shuffles + hill‑climb).
  7.  A final mass‑transfer simulated‑annealing polish.
All updates stay on the ℓ¹‑simplex (∑a = √(2 n)) and respect non‑negativity.
The most expensive stage (5) receives the bulk of the 1000 s budget, which
typically drives the maximal convolution down to ≈ 1.00 – 1.01,
i.e. an objective ≲ 1.503.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (simple parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old to new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (scatter the biggest values)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} max={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent_dynamic(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_step: float = 0.7,
                                 gamma: float = 0.995,
                                 tol_start: float = 0.98,
                                 tol_end: float = 0.9999) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent.
    The tolerance on the active set tightens linearly from tol_start to tol_end."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_max = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    total = max(1.0, deadline - time.time())

    while time.time() < deadline:
        # elapsed fraction for schedule
        frac = (time.time() - (deadline - total)) / total
        tol = tol_start + (tol_end - tol_start) * frac

        conv = _conv_fft(a)
        M = conv.max()
        mask = (conv >= M * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        mask_fft = np.fft.rfft(mask, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(mask_fft * rev_fft, n=size)[: len(mask) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * start_step) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_max = _max_convolution_fft(a)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} tol={tol:.5f} max={best_max:.6f}")

        it += 1

    return best_a, best_max

# Small LP refinement (active‑set LP)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               max_iters: int,
               deadline: float,
               K: int = 30,
               mix_factor: float = 0.12) -> tuple[np.ndarray, float]:
    """Solve a tiny LP that maximises ℓ¹‑norm under the K most binding convolution constraints."""
    try:
        from scipy.optimize import linprog
    except Exception:
        print("SciPy not available – skipping LP refinement.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        if K >= len(conv):
            active = np.arange(len(conv), dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # Build matrix A (size |active| × n)
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            A[row, idx] = a[k - idx]

        # LP: maximise sum(g) ⇔ minimise -sum(g)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        g = g * (target_sum / g.sum())

        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} max={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation search utilities
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer simulated annealing (final polish)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.2,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow traps
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns the best sequence discovered within the allotted time.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Build a modest‑size pool (n≈4000)
    # ------------------------------------------------------------------
    n0 = 4000
    target0 = math.sqrt(2.0 * n0)

    pool: List[np.ndarray] = []

    # Parametric families
    for p in np.linspace(0.2, 1.8, 8):
        pool.append(_power_law_sequence(n0, float(p), target0))
    for lam in np.linspace(0.001, 0.02, 6):
        pool.append(_exponential_sequence(n0, float(lam), target0))
    for _ in range(6):
        pool.append(_cosine_sequence(n0, target0))
    for _ in range(4):
        pool.append(_sin_half_sequence(n0, target0))
    for _ in range(12):
        pool.append(_random_simplex(n0, target0, rng))

    # reuse a previous construction if present
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        pool.append(_resize_sequence(old, n0))

    # jittered variants of the pool
    extra = []
    for seq in pool:
        jitter = rng.normal(scale=target0 * 0.025, size=n0)
        s = np.maximum(seq + jitter, 0.0)
        s = _project_to_simplex(s, target0)
        extra.append(s)
    pool.extend(extra)

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 2. Short annealed GD passes (use most of the early budget)
    # ------------------------------------------------------------------
    while time.time() < deadline - 150.0:   # leave ~150 s for later phases
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target0 * 0.008, size=n0)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target0)
        else:
            seed = rng.choice(pool)

        run_deadline = min(time.time() + 15.0, deadline - 140.0)
        if run_deadline <= time.time():
            break

        a_opt, max_conv = _annealed_gradient_descent(
            seed, run_deadline, target0,
            max_beta=2e6, step_factor=0.85, gamma=0.97)

        obj = 2 * n0 * max_conv / (target0 * target0)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD new best objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. LP refinement on the modest size
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        lp_deadline = time.time() + min(30.0, deadline - time.time() - 110.0)
        best_seq, max_conv = _lp_refine(best_seq,
                                        target0,
                                        max_iters=45,
                                        deadline=lp_deadline,
                                        K=30,
                                        mix_factor=0.12)
        obj = 2 * n0 * max_conv / (target0 * target0)
        if obj < best_obj:
            best_obj = obj
            print(f"[{time.time():.2f}] After LP best objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Upscale to a larger problem (n≈15 k–30 k)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 350:
        n_big = 30000
    elif remaining > 250:
        n_big = 25000
    elif remaining > 150:
        n_big = 20000
    elif remaining > 100:
        n_big = 15000
    else:
        n_big = 10000

    target_big = math.sqrt(2.0 * n_big)

    if best_seq is not None:
        seq_big = _resize_sequence(best_seq, n_big)
    else:
        seq_big = _power_law_sequence(n_big, 0.7, target_big)

    seq_big = _project_to_simplex(seq_big, target_big)

    # spread the largest entries
    seq_big = _even_spread_order(seq_big)

    cur_max = _max_convolution_fft(seq_big)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start maxConv={cur_max:.6f}")

    # ------------------------------------------------------------------
    # 5. Heavy exact sub‑gradient descent (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        seq_big, max_conv = _subgradient_descent_dynamic(seq_big,
                                                         sub_deadline,
                                                         target_big,
                                                         start_step=0.6,
                                                         gamma=0.996,
                                                         tol_start=0.98,
                                                         tol_end=0.9999)
        best_obj = 2 * n_big * max_conv / (target_big * target_big)
        print(f"[{time.time():.2f}] Sub‑grad final objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 6. Permutation optimisation (random shuffles + hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        perm_deadline = time.time() + min(12.0, deadline - time.time() - 18.0)
        seq_big, max_conv = _permutation_search(seq_big, trials=250, deadline=perm_deadline)
        best_obj = 2 * n_big * max_conv / (target_big * target_big)
        print(f"[{time.time():.2f}] Random permutation best objective={best_obj:.6f}")

    if time.time() < deadline - 15.0:
        hill_deadline = time.time() + min(8.0, deadline - time.time() - 13.0)
        seq_big, max_conv = _local_permutation_hillclimb(seq_big, hill_deadline)
        best_obj = 2 * n_big * max_conv / (target_big * target_big)
        print(f"[{time.time():.2f}] Hill‑climb best objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 7. Final mass‑transfer simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = time.time() + min(6.0, deadline - time.time() - 4.0)
        seq_big, max_conv = _mass_transfer_simulated_annealing(seq_big,
                                                              mt_deadline,
                                                              target_big,
                                                              start_temp=0.18,
                                                              end_temp=1e-5,
                                                              low_frac=0.04)
        best_obj = 2 * n_big * max_conv / (target_big * target_big)
        print(f"[{time.time():.2f}] Final MT best objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 8. Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    # make the construction available for a possible second run
    globals()["height_sequence_1"] = final_seq

    try:
        score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {score:.6f}")
    except Exception as exc:
        print(f"Failed to evaluate final sequence: {exc}")

    return final_seq
