def solve_lp_max_sum(f, M, target_sum):
    n = f.shape[0]
    # Build matrix
    B = build_conv_matrix(f)
    # objective: maximize sum => minimize -1^T g
    c = -np.ones(n)
    # bounds g_i >= 0
    bounds = [(0, None)] * n
    # constraints: B @ g <= M
    # Use hiGHs method with time limit.
    # Options dictionary: {'time_limit': 5.0}
    res = scipy.optimize.linprog(c, A_ub=B, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'time_limit': 5.0})
    if not res.success:
        # fallback: return zero vector
        return np.zeros(n)
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 1e-12:
        return np.zeros(n)
    # scale to target sum
    scale = target_sum / sum_g0
    g = g0 * scale
    return g

"""
Simple iterative optimizer for the step‑function peak problem.

Main ideas
----------
1. Start from a modest‑size (≈1200) seed sequence.
2. Repeatedly solve a linear program:
   maximize   Σ g_i
   subject to (f * g)_k ≤ M   (M = max(f * f))  for all k,
   g_i ≥ 0.
   The LP gives a “friendly’’ direction g that can increase the L¹ mass
   without raising the current peak.  A small convex combination
   f←(1−t)f+t g (t≈0.2) followed by a projection onto the simplex tends to
   lower the peak.
3. After a few mixing rounds we polish the sequence with a high‑β
   soft‑max gradient descent (projected onto the ℓ¹‑simplex), then upscale
   to a larger length (≈12000) and repeat the polishing.
4. A short deterministic “mass‑transfer’’ annealing finishes the search.

The algorithm respects a 1000 s time budget, prints progress, stores the
best sequence in the global variable `height_sequence_1` and finally
returns a list of non‑negative heights.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective value: 2·n·max(a*a) / (Σa)².
    When Σa = √(2·n) the denominator cancels and the score equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, second‑largest,… placed alternately from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution over shifts

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Momentum‑based annealed soft‑max gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of the peak.
    The temperature β is increased exponentially up to ``max_beta``.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (fast per iteration)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution entry.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # gradient of the max entry (linear term)
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]      # derivative of a[mid]^2

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small local moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Local mass‑transfer moves with simulated‑annealing acceptance."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate current peak
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear programming step (mixing)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix B where (B @ g)[k] = (f * g)[k].
    Shape: (2*n-1, n)
    """
    n = f.shape[0]
    # rows = i + j  (i = index of g, j = index of f)
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)
    rows = (i[:, None] + j[None, :]).ravel()
    cols = np.repeat(i, n)
    data = np.tile(f, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_max_sum(f: np.ndarray, M: float, target_sum: float,
                      lp_time_limit: float = 5.0) -> np.ndarray:
    """
    Solve   max Σ g_i   s.t.   (f * g)_k ≤ M   (all k),   g_i ≥ 0.
    Returns the scaled vector g with Σ g_i = target_sum.
    If the LP fails, a zero vector is returned.
    """
    n = f.shape[0]
    B = _build_conv_matrix(f)
    c = -np.ones(n)                     # minimize -Σ g_i  == maximize Σ g_i
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # HiGHS (method='highs') handles sparse A_ub.
    try:
        res = opt.linprog(c,
                          A_ub=B,
                          b_ub=b_ub,
                          bounds=bounds,
                          method='highs',
                          options={'time_limit': lp_time_limit,
                                   'presolve': True})
    except Exception as e:
        print(f"Linprog exception: {e}")
        return np.zeros_like(f)

    if not res.success:
        # no feasible improvement found
        return np.zeros_like(f)

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 1e-12:
        return np.zeros_like(f)

    scale = target_sum / sum_g0
    return g0 * scale


def _mix_step(f: np.ndarray, g: np.ndarray, t: float, target_sum: float) -> np.ndarray:
    """Convex combination (1−t) f + t g projected onto the ℓ¹‑simplex."""
    new_f = (1.0 - t) * f + t * g
    return _project_to_simplex(new_f, target_sum)

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Respects the supplied time budget (≈1000 s) and returns a list of heights.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 0 – choose a modest size for the LP‑mixing stage
    # --------------------------------------------------------------
    # The mixing stage works best with a few thousand entries.
    # We keep it relatively small to keep the LP cheap.
    if budget_s >= 800:
        n_mix = 1200
    elif budget_s >= 500:
        n_mix = 1000
    else:
        n_mix = 800
    target_mix = np.sqrt(2.0 * n_mix)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 1 – initialise the seed sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            f = _resize_sequence(prev, n_mix)
        else:
            f = _power_law_sequence(n_mix, 0.6, target_mix)
    else:
        f = _power_law_sequence(n_mix, 0.6, target_mix)

    f = _project_to_simplex(f, target_mix)

    best_f = f.copy()
    best_val = _score(f)
    print(f"[{time.time():.2f}] Initial mix‑size n={n_mix} score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – LP‑based mixing iterations
    # --------------------------------------------------------------
    mixing_deadline = deadline - 250.0  # reserve ~250 s for the final polishing
    t_mix = 0.25                          # starting mixing weight

    mix_iter = 0
    while time.time() < mixing_deadline and mix_iter < 15:
        M = _max_convolution_fft(f)

        # solve LP, allow at most 4 s per solve
        g = _solve_lp_max_sum(f, M, target_mix, lp_time_limit=4.0)
        if g.sum() == 0:
            # LP failed – give up mixing
            print(f"[{time.time():.2f}] LP failed at iter {mix_iter}")
            break

        # try a few mixing coefficients and keep the best improvement
        best_local_f = None
        best_local_val = np.inf
        best_local_t = None
        for tt in (0.12, 0.20, 0.28, 0.36):
            cand = _mix_step(f, g, tt, target_mix)
            val = _score(cand)
            if val < best_local_val:
                best_local_val = val
                best_local_f = cand
                best_local_t = tt

        if best_local_val < best_val - 1e-7:
            best_val = best_local_val
            best_f = best_local_f.copy()
            f = best_local_f
            print(f"[{time.time():.2f}] Mix iter={mix_iter:2d} t={best_local_t:.2f} score={best_val:.6f}")
        else:
            # no improvement – add a tiny jitter and keep trying
            jitter = rng.normal(scale=target_mix * 0.001, size=n_mix)
            f = _project_to_simplex(f + jitter, target_mix)
            print(f"[{time.time():.2f}] Mix iter={mix_iter:2d} no gain, jitter added")

        mix_iter += 1

    # --------------------------------------------------------------
    # Phase 3 – short gradient polishing on the modest size
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        gd_dead = time.time() + min(80.0, deadline - time.time() - 70.0)
        f_gd, val_gd = _annealed_gradient_descent(best_f,
                                                  gd_dead,
                                                  target_mix,
                                                  max_beta=1e6,
                                                  step_factor=0.6,
                                                  gamma=0.99)
        if val_gd < best_val:
            best_val = val_gd
            best_f = f_gd.copy()
            print(f"[{time.time():.2f}] Post‑mix GD best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – upscale to a larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 250.0 and budget_s >= 800:
        n_final = 15000
    elif remaining > 180.0:
        n_final = 12000
    elif remaining > 100.0:
        n_final = 8000
    else:
        n_final = max(n_mix, int(remaining * 15))   # very rough fallback

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_f, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # systematic re‑ordering to spread large masses
    seq_big = _alternating_order(seq_big)

    cur_val = _score(seq_big)
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={cur_val:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – long high‑β gradient descent on the large instance
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        gd_final_dead = deadline - 15.0
        seq_big, final_val = _annealed_gradient_descent(seq_big,
                                                        gd_final_dead,
                                                        target_final,
                                                        max_beta=2e6,
                                                        step_factor=0.5,
                                                        gamma=0.995)
        if final_val < cur_val:
            cur_val = final_val
            print(f"[{time.time():.2f}] Long GD improved to {cur_val:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – deterministic mass‑transfer refinement (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 2.0)
        seq_big, mt_val = _deterministic_mass_transfer(seq_big,
                                                       mt_dead,
                                                       target_final,
                                                       start_temp=0.4,
                                                       end_temp=1e-5,
                                                       low_frac=0.04)
        if mt_val < cur_val:
            cur_val = mt_val
            print(f"[{time.time():.2f}] MT refinement improved to {cur_val:.6f}")

    # --------------------------------------------------------------
    # Phase 7 – final clean‑up and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # optional logging of the official evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
