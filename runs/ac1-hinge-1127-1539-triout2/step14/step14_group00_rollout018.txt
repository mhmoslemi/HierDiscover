def propose_candidate(budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    # set small size n based on budget, maybe 1500 to allow many iterations
    n = 1500
    S = np.sqrt(2.0 * n)
    # initialize f: maybe power law or uniform random normalized.
    rng = np.random.default_rng()
    f = rng.random(n)
    f = f / f.sum() * S
    best_f = f.copy()
    best_M = _max_convolution_fft(f)
    # parameters
    t_step = 0.2
    # iteration loop
    while time.time() < deadline - 5.0:   # keep some margin for final polishing
        # compute current M
        M = _max_convolution_fft(f)
        # build convolution matrix A (shape (2n-1, n))
        c = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
        r = np.concatenate(([f[0]], np.zeros(n-1, dtype=f.dtype)))
        A = np.linalg.toeplitz(c, r)  # shape (2n-1, n)
        # solve LP
        # objective maximize sum(g) => minimize -sum(g)
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        # Using scipy.optimize.linprog
        res = linprog(c=c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(5.0, (deadline-time.time())*0.9)})
        if not res.success:
            print('LP failed, fallback to previous f')
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        # scale to target sum
        g = g0 * (S / sum_g0)
        # line search for t
        improved = False
        for t in [0.5, 0.2, 0.1, 0.05]:
            new_f = (1 - t) * f + t * g
            new_M = _max_convolution_fft(new_f)
            if new_M < M:
                f = new_f
                improved = True
                if new_M < best_M:
                    best_M = new_M
                    best_f = new_f.copy()
                break
        if not improved:
            # maybe accept even if same? degrade t small?
            f = (1-t_step) * f + t_step * g  # take a small step anyway
            # update best if improvement
            new_M = _max_convolution_fft(f)
            if new_M < best_M:
                best_M = new_M
                best_f = f.copy()
    # after loop, upsample to larger N if time remaining
    # choose upsample factor about 5 if still time
    remaining = deadline - time.time()
    if remaining > 10.0:
        up_n = 8000
        up_f = _resize_sequence(best_f, up_n)
        up_f = _project_to_simplex(up_f, np.sqrt(2.0 * up_n))
        # apply a short smooth gradient descent
        up_f, up_score = _smooth_gradient_descent(up_f,
                                                  time.time() + remaining - 5.0,
                                                  np.sqrt(2.0 * up_n),
                                                  alpha=6.0,
                                                  init_step_factor=0.5)
        best_f = up_f
        best_M = up_score
    # trim near-zero entries
    arr = np.array(best_f)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1] + 1]
    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq
    print('Final best max convolution =', best_M)

"""
Hybrid linear‑programming + gradient polishing optimizer.

Goal
----
Find a non‑negative vector `a` (step‑function heights) that minimises

    2·n·max(a*a) / (sum(a)**2)

where `a*a` is the linear convolution of `a` with itself.
For a vector normalised to Σa = √(2n) the objective equals simply
`max(a*a)`.  The theoretical lower bound of the objective is 1 (the
average of the convolution entries), while the current best known
value is ≈ 1.5068.  This implementation follows the LP‑based scheme
described in the problem statement:

* Given a current vector `f` with max‑convolution `M`,
  solve the linear programme

        maximise  Σ g_i
        subject   f * g ≤ M    (component‑wise)
                   g_i ≥ 0

  (here `*` denotes linear convolution, which is linear in `g`).

* The optimal `g₀` is scaled to the required ℓ¹‑norm,
  then blended with `f` using a small step size `t`.  Repeating this
  loop drives the peak `M` down.

Because the LP matrix is dense (size ≈ n²), we keep `n` modest
(≈ 1500) during the LP phase, then up‑sample the final vector to a
larger length (≈ 8000) and finish with a few iterations of smooth
gradient descent to polish the result.

The whole routine respects a 1000 s time budget and returns the best
sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for large n)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹‑norm = target_sum, non‑negative)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothing surrogate (log‑sum‑exp) + projected gradient descent
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise `log_sum_exp(alpha·(a*a)) / alpha` by projected gradient descent.
    The surrogate approaches the true maximum as `alpha → ∞`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    step_factor = init_step_factor
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)
        M = conv.max()
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        # gradient of the smooth max (see derivation in analysis)
        a_rev = a[::-1]
        conv_w_rev = np.fft.irfft(
            np.fft.rfft(w, n=1 << ((3 * n - 2).bit_length())) *
            np.fft.rfft(a_rev, n=1 << ((3 * n - 2).bit_length())),
            n=1 << ((3 * n - 2).bit_length())
        )[: n]               # first n elements correspond to gradient
        grad = a + conv_w_rev / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        cur = _max_convolution_fft(a_new)
        if cur < best_score:
            best_score = cur
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple linear interpolation for up‑sampling
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    LP‑based optimiser with final gradient polishing.
    Returns a list of non‑negative heights whose evaluation score
    is as low as possible.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final checks

    # --------------------------------------------------------------
    # 1️⃣  LP phase – work with a moderate dimension to keep each LP cheap
    # --------------------------------------------------------------
    n_lp = 1500                          # modest size for the LP loop
    target_sum = np.sqrt(2.0 * n_lp)

    rng = np.random.default_rng()
    # start from a random point on the ℓ¹‑simplex
    f = rng.random(n_lp)
    f = _project_to_simplex(f, target_sum)

    best_f = f.copy()
    best_M = _max_convolution_fft(f)

    # LP iteration parameters
    t_step = 0.3                         # blending factor (can be tuned)
    max_lp_iters = 30

    for it in range(max_lp_iters):
        if time.time() > deadline - 10.0:
            break

        M = _max_convolution_fft(f)

        # Build the convolution matrix A where (A @ g)[k] = (f * g)[k]
        # A is a Toeplitz matrix with first column = [f, 0,…,0] (length 2n‑1)
        # and first row = [f[0], 0,…,0] (length n)
        col = np.concatenate([f, np.zeros(n_lp - 1, dtype=f.dtype)])
        row = np.concatenate(([f[0]], np.zeros(n_lp - 1, dtype=f.dtype)))
        A = toeplitz(col, row)                 # shape (2n‑1, n)

        # Linear programme:
        # maximise Σ g_i   <=>   minimise -Σ g_i
        c_obj = -np.ones(n_lp)
        b_ub = np.full(2 * n_lp - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n_lp

        # solve (the HiGHS solver is fast and handles dense matrices reasonably well)
        res = linprog(c=c_obj,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(5.0,
                          deadline - time.time() - 2.0)})
        if not res.success:
            # If the LP fails for any reason we abort the LP phase.
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        g0 = np.maximum(res.x, 0.0)          # numerical safety
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # Rescale to the required ℓ¹‑norm
        g = g0 * (target_sum / sum_g0)

        # ------------------------------------------------------------------
        # 2️⃣  Blend f and g (small step) and keep improvement
        # ------------------------------------------------------------------
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            f_new = (1.0 - t) * f + t * g
            M_new = _max_convolution_fft(f_new)
            if M_new < M - 1e-12:          # accept any strict improvement
                f = f_new
                improved = True
                if M_new < best_M:
                    best_M = M_new
                    best_f = f_new.copy()
                break
        if not improved:
            # Take a tiny step to keep the algorithm moving
            f = (1.0 - t_step) * f + t_step * g
            M_new = _max_convolution_fft(f)
            if M_new < best_M:
                best_M = M_new
                best_f = f.copy()

        # optional early exit if we are already very close to the lower bound
        if best_M <= 1.02:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Upsample to a larger size (≈ 8000) and polish with smooth GD
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        up_n = 8000
        up_target = np.sqrt(2.0 * up_n)
        f_up = _resize_sequence(best_f, up_n)
        f_up = _project_to_simplex(f_up, up_target)

        # a short smooth gradient descent (surrogate max) to polish
        smooth_deadline = time.time() + min(20.0, deadline - time.time() - 2.0)
        f_up, score_up = _smooth_gradient_descent(f_up,
                                                  smooth_deadline,
                                                  up_target,
                                                  alpha=7.0,
                                                  init_step_factor=0.6)
        if score_up < best_M:
            best_M = score_up
            best_f = f_up

    # ------------------------------------------------------------------
    # 4️⃣  Trim trailing near‑zero entries, store globally, and return
    # ------------------------------------------------------------------
    arr = np.array(best_f, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    globals()["height_sequence_1"] = final_seq

    # final sanity check (the evaluation function is provided externally)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[final] objective = {val:.6f}")
    except Exception:
        pass

    return final_seq
