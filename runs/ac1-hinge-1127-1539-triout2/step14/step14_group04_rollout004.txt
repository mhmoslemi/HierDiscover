"""
Propose a low‑peak step‑function.

The algorithm builds a diverse pool of seed sequences (power‑law,
exponential, sparse Dirichlet, previous best), runs many short
gradient‑descent / sub‑gradient passes to obtain a strong candidate,
up‑scales it to a larger length and then spends the bulk of the time
doing a long annealed soft‑max gradient descent followed by
mass‑transfer, permutation‑annealing and a final sub‑gradient polish.
All vectors are kept on the ℓ¹‑simplex Σ a_i = √(2·n) so that the
evaluation score reduces to the maximum entry of the self‑convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(Σa)².  For Σa = √(2·n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    cond = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place sorted values alternately from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution over shifts

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Momentum‑based annealed soft‑max gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of the peak.
    β is increased exponentially up to max_beta.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β periodically
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (fast per iteration)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution entry.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # gradient of the max entry (linear term)
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]  # derivative of a[mid]^2

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small local moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Local mass‑transfer moves with simulated‑annealing acceptance."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # current peak location
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # choose a low contributor among the low_frac of the window
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing (swap high/low contributors)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that swaps a high‑contributing index with a low‑contributing one.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # swap masses
        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈1000 s) and returns
    a list of heights.  Intermediate progress is printed to stdout.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – size selection
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 7000
    elif budget_s >= 600:
        n_expl = 5000
    else:
        n_expl = 4000
    target_expl = math.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – build a diverse seed pool
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # power‑law family (wide range)
    for p in np.linspace(0.2, 1.2, 13):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0005, 0.03, 13):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # sparse Dirichlet‑like seeds (Gamma with small shape)
    for alpha in (0.5, 0.2, 0.1):
        for _ in range(3):
            a = rng.gamma(alpha, 1.0, size=n_expl)
            a = np.maximum(a, 0.0)
            a *= target_expl / a.sum()
            init_pool.append(a)

    # linear decreasing seed
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.01, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Phase 2 – exploration (≈40 % of total budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.40 * budget_s
    best_seq = None
    best_val = np.inf

    while time.time() < explore_deadline:
        # pick a seed (occasionally bias towards current best)
        if best_seq is not None and rng.random() < 0.45:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.003, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # short optimisation run (≈8 s)
        if rng.random() < 0.6:
            run_dead = time.time() + 8.0
            a_opt, val_opt = _annealed_gradient_descent(
                seed,
                run_dead,
                target_expl,
                max_beta=1e6,
                step_factor=0.6,
                gamma=0.99,
            )
        else:
            run_dead = time.time() + 8.0
            a_opt, val_opt = _subgradient_descent(
                seed,
                run_dead,
                target_expl,
                max_iter=120_000,
            )

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – up‑scale to a larger length
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.6, target_expl)

    remaining = deadline - time.time()
    # choose final length based on remaining time
    if remaining > 300 and budget_s >= 800:
        n_final = 18000
    elif remaining > 200:
        n_final = 15000
    elif remaining > 120:
        n_final = 12000
    else:
        n_final = max(n_expl, int(remaining * 30))

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few systematic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
    ]
    best_seq = seq_big
    best_val = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_val:
            best_val = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – long high‑β gradient descent (majority of remaining time)
    # ------------------------------------------------------------------
    time_rem = deadline - time.time()
    if time_rem > 30.0:
        gd_dead = time.time() + time_rem - 30.0
        a_gd, val_gd = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=5e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] Long GD best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    time_rem = deadline - time.time()
    if time_rem > 10.0:
        mt_dead = time.time() + min(10.0, time_rem)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.5,
            end_temp=1e-5,
            low_frac=0.07,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] MT after GD best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – targeted permutation simulated annealing
    # ------------------------------------------------------------------
    time_rem = deadline - time.time()
    if time_rem > 5.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.07,
            low_frac=0.07,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Perm after MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – final short sub‑gradient pass
    # ------------------------------------------------------------------
    time_rem = deadline - time.time()
    if time_rem > 2.0:
        sub_dead = time.time() + min(3.0, time_rem)
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
            max_iter=150_000,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Final sub‑grad best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 8 – final deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    time_rem = deadline - time.time()
    if time_rem > 1.0:
        mt_dead = time.time() + min(4.0, time_rem)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_val:
            best_val = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 9 – trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation for logging
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as e:
        print("Evaluation error:", e)

    globals()["height_sequence_1"] = final_seq
    return final_seq
