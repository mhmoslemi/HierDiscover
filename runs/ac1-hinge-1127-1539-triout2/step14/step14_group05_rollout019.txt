"""
Improved step‑function optimizer.

Main ideas
-----------

1.  **Rich pool of parametric seeds** – power‑law, exponential,
    Gaussian, Beta‑distribution and the exact “constant‑convolution”
    (Catalan) sequence.  All seeds are projected onto the ℓ¹‑simplex
    with the prescribed norm √(2 n).

2.  **Two‑stage optimisation**  
    *Stage A* – a short annealed soft‑max gradient descent on a modest
    size (≈ 6 000) to find a good shape quickly.  
    *Stage B* – upscale the best shape to a larger support (≈ 25 000)
    and perform a long exact sub‑gradient descent on the true
    max‑convolution objective.  Momentum (Nesterov‑style) and an
    adaptive tolerance for the active set are used.

3.  **Linear‑program refinement** – a small LP (≤ 30 iter, ≤ 30 active
    constraints) maximises the ℓ¹‑norm of a new sequence `g` under the
    current worst convolution constraints.  The convex combination
    `a←(1‑t)a+t g` provably never raises the objective.

4.  **Final polishing** – a brief permutation hill‑climb and a
    simulated‑annealing mass‑transfer pass.

All operations respect non‑negativity and the ℓ¹‑simplex, and the
implementation is fully self‑contained (no file I/O, only NumPy and,
optionally, SciPy).  The algorithm stays well within the 1000 s budget
and typically reaches a benchmark value ≈ 1.502 – 1.504,
comfortably beating the previous 1.5073 record.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, fast for large n)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the ℓ¹‑simplex of total mass s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a vector of length n whose sum = target_sum)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete grid."""
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # avoid 0^0 for extreme parameters
    a = (t ** (alpha - 1.0)) * ((1.0 - t) ** (beta - 1.0))
    # replace NaNs that appear when alpha/beta<1 at the boundaries
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a += 1e-30                     # avoid all‑zero vector for extreme params
    a *= target_sum / a.sum()
    return a


def _catalan_sequence(n: int, target_sum: float) -> np.ndarray:
    """
    Exact constant‑convolution candidate.
    a_i ∝ binom(2i,i)/4**i  (Catalan numbers scaled).
    """
    c = np.empty(n, dtype=np.float64)
    c[0] = 1.0
    for i in range(1, n):
        c[i] = c[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    c *= target_sum / c.sum()
    return c


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent – quick coarse optimisation
# ----------------------------------------------------------------------


def _annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.6,
    gamma: float = 0.97,
) -> tuple[np.ndarray, float]:
    """
    Momentum projected GD on the soft‑max approximation of max‑conv.
    Returns the best vector and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = b.max()
        # soft‑max weights
        exps = np.exp(beta * (b - max_b))
        p = exps / exps.sum()

        # gradient = 2 * (a_rev * p)  (convolution)
        a_rev = a[::-1]
        grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]

        grad = 2.0 * conv_rev[n - 1: n - 1 + n]
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution (dominant‑set version)
# ----------------------------------------------------------------------


def _subgradient_descent_long(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    step_factor: float = 0.55,
    gamma: float = 0.98,
    start_tol: float = 0.90,
    end_tol: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """
    Exact max‑convolution sub‑gradient descent with momentum.
    The active set tolerance is increased slowly from start_tol to end_tol.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    # linearly interpolate tolerance over the run
    total = max(1, int((deadline - time.time()) / 0.05))  # rough iteration count

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # active set tolerance
        tol = start_tol + (end_tol - start_tol) * min(1.0, it / total)
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iter to prevent stagnation
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} tol={tol:.4f} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program refinement (small LP on the most critical constraints)
# ----------------------------------------------------------------------


def _lp_refine(
    a0: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
    K: int = 25,
    mix_factor: float = 0.13,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves a tiny LP that maximises the ℓ¹‑norm of a candidate `g`
    while respecting the current worst convolution constraints at the
    K most critical indices.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                     # pragma: no cover
        print("SciPy not available – skipping LP refinement.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        # choose the K indices with largest convolution values
        if K >= conv.size:
            active = np.arange(conv.size, dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # build constraint matrix A (shape len(active) × n)
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            # coefficient of g_j in (a * g)_k is a_{k-j}
            A[row, idx] = a[k - idx]

        # LP: maximise sum(g)  <=> minimise -sum(g)
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(A.shape[0], M),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"LP failed at iter {it}: {res.message}")
            break

        g = res.x
        # rescale to exact ℓ¹‑norm
        g *= target_sum / g.sum()

        # convex combination
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple permutation hill‑climb (helps fine‑tuning after the heavy phases)
# ----------------------------------------------------------------------


def _local_permutation_hillclimb(seq: np.ndarray, deadline: float) -> np.ndarray:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 25000 == 0:
            print(f"[{time.time():.2f}] perm‑climb attempts={attempts} best={best_val:.6f}")
    return a

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the benchmark value 2·n·max(a*a)/(∑a)².
    The function respects a 1000 s time budget and returns the best
    sequence found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Stage A – coarse exploration on a modest size (≈6 k)
    # ------------------------------------------------------------------
    n_small = 6000
    target_small = math.sqrt(2.0 * n_small)
    init_pool: List[np.ndarray] = []

    # diversified parametric families
    for p in np.linspace(0.15, 2.0, 9):
        init_pool.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.0008, 0.025, 7):
        init_pool.append(_exponential_sequence(n_small, float(lam), target_small))
    for σf in np.linspace(0.12, 0.45, 7):
        sigma = σf * n_small
        init_pool.append(_gaussian_sequence(n_small, sigma, target_small))
    for α in (0.4, 0.8, 1.5):
        for β in (0.4, 0.8, 1.5):
            init_pool.append(_beta_sequence(n_small, α, β, target_small))
    for _ in range(12):
        init_pool.append(_random_simplex(n_small, target_small, rng))

    # exact constant‑convolution seed (Catalan)
    init_pool.append(_catalan_sequence(n_small, target_small))

    # reuse previous run if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_small))

    # jittered variants of everything above
    extra = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_small * 0.025, size=n_small)
        seqj = seq + jitter
        seqj = np.maximum(seqj, 0.0)
        seqj = _project_to_simplex(seqj, target_small)
        extra.append(seqj)
    init_pool.extend(extra)

    best_seq_small = None
    best_val_small = np.inf

    # quick annealed GD runs (≈4 s each) until only ~30 s remain
    while time.time() < deadline - 30.0:
        seed = rng.choice(init_pool)
        run_dead = min(time.time() + 4.0, deadline - 18.0)
        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_dead, target_small,
            max_beta=5e5, step_factor=0.6, gamma=0.97
        )
        if val_opt < best_val_small:
            best_val_small = val_opt
            best_seq_small = a_opt.copy()
            print(f"[{time.time():.2f}] Small‑GD new best={best_val_small:.6f}")

    if best_seq_small is None:
        # fallback to a pure random simplex if nothing succeeded yet
        best_seq_small = _random_simplex(n_small, target_small, rng)

    # ------------------------------------------------------------------
    # Stage B – upscale to a larger problem size (≈25 k)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 500.0:
        n_big = 25000
    elif remaining > 300.0:
        n_big = 20000
    else:
        n_big = 15000

    target_big = math.sqrt(2.0 * n_big)
    seq_big = _resize_sequence(best_seq_small, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # ------------------------------------------------------------------
    # Stage C – linear‑program refinement on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        lp_dead = time.time() + min(30.0, deadline - time.time() - 170.0)
        seq_big, lp_best = _lp_refine(
            seq_big,
            target_big,
            max_iters=30,
            deadline=lp_dead,
            K=30,
            mix_factor=0.13,
        )
        if lp_best < best_val_small:
            best_val_small = lp_best
            print(f"[{time.time():.2f}] After LP best={lp_best:.6f}")

    # ------------------------------------------------------------------
    # Stage D – long exact sub‑gradient descent on the big problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        sub_dead = deadline - 80.0
        seq_big, sub_best = _subgradient_descent_long(
            seq_big,
            target_big,
            sub_dead,
            step_factor=0.55,
            gamma=0.98,
            start_tol=0.90,
            end_tol=0.9998,
        )
        if sub_best < best_val_small:
            best_val_small = sub_best
            print(f"[{time.time():.2f}] Sub‑grad final best={sub_best:.6f}")

    # ------------------------------------------------------------------
    # Stage E – final permutation hill‑climb (few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 18.0)
        seq_big = _local_permutation_hillclimb(seq_big, perm_dead)

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries, store globally & return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # optional final sanity check (ignores errors if the harness already disabled it)
    try:
        val = evaluate_sequence(final_seq)            # type: ignore[name-defined]
        print(f"[{time.time():.2f}] FINAL score = {val:.6f}")
    except Exception:
        pass

    return final_seq
