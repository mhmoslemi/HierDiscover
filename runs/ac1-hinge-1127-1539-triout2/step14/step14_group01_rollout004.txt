def _gradient_smooth(a, alpha):
    n = a.shape[0]
    conv = _conv_fft(a)  # length 2n-1
    # compute weights
    # to avoid overflow:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shift = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_shift)
    w = exp_shift / sum_exp
    # compute gradient
    a_rev = a[::-1]
    # compute convolution w * a_rev
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    # use FFT rfft with real input
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(fw * fa_rev, n=size)[:len(w) + len(a_rev) - 1]
    # slice for i in [0,n-1], offset = n-1
    offset = n - 1
    grad = 2.0 * grad_full[offset: offset + n]
    # return gradient
    return grad
```

def _permutation_sa(a, target_sum, deadline, rng):
    # a: np.ndarray
    best = a.copy()
    best_val = _max_convolution_fft(a)
    cur = a.copy()
    cur_val = best_val
    T_start = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T_start * (1 - frac) + T_end * frac
        # propose swap or small mass transfer
        i, j = rng.integers(0, len(cur), size=2)
        if i == j:
            continue
        # propose swap
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        # projection not needed because sum unchanged and nonneg remain.
        new_val = _max_convolution_fft(new)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
    return best, best_val
```

"""
Hybrid optimization for the “minimum peak convolution” problem.

The routine combines several complementary heuristics that respect the
ℓ¹‑simplex (non‑negative entries with fixed sum √(2 n)):

1. **Seeding** – generate many diverse candidates (power‑law, exponential,
   random simplex and the previous best) and project them onto the simplex.

2. **Differential Evolution** – a classic DE loop with simplex projection.
   This quickly yields a good global starting point.

3. **Smooth‑gradient descent** – minimise a smooth approximation of the
   max‑convolution using the log‑sum‑exp (temperature α).  The gradient is
   computed efficiently by two FFT‑based convolutions.  α is increased
   gradually so the objective becomes sharper while the iterates stay in
   a basin of attraction.

4. **Permutation simulated‑annealing** – large‑scale reordering of the
   entries (random swaps) can lower the peak further because the
   convolution depends heavily on the placement of large values.  Swaps are
   accepted with a Metropolis rule governed by a linear cooling schedule.

5. **Deterministic re‑ordering** – finally the three simple permutations
   (original, alternating, even‑spread) are tested and the best kept.

The algorithm respects the 1000 s budget, automatically allocating time to
each stage.  The best sequence is stored in ``globals()["height_sequence_1"]``
for warm‑starting the next round.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE on the ℓ¹‑simplex, minimising the peak of a*a."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    # initial scores
    scores = [_max_convolution_fft(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # select three distinct mates
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _max_convolution_fft(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        # early exit guard
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑gradient descent on log‑sum‑exp approximation
# ----------------------------------------------------------------------


def _gradient_smooth(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·conv) w.r.t. a.
    The result is 2 * (w ⊛ rev(a))[i + n - 1] where w are normalized weights.
    """
    n = a.shape[0]
    conv = _conv_fft(a)                     # length 2n‑1
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    # stable softmax weights
    exp_shift = np.exp(scaled - max_scaled)
    w = exp_shift / np.sum(exp_shift)

    # convolution w * rev(a)
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    fw = np.fft.rfft(w, n=size)
    fa = np.fft.rfft(a_rev, n=size)
    full = np.fft.irfft(fw * fa, n=size)[: len(w) + len(a_rev) - 1]
    offset = n - 1
    grad = 2.0 * full[offset: offset + n]
    return grad


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed max‑convolution.
    Alpha (temperature) starts low (smooth) and grows, sharpening the objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution_fft(a)
    start_time = time.time()
    # schedule for alpha
    alpha = 1.0
    max_alpha = 500.0

    while time.time() < deadline:
        # compute gradient at current alpha
        grad = _gradient_smooth(a, alpha)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        # step size heuristic
        lr = 0.2 * target_sum / (norm + 1e-12)

        # try a few step lengths (line search)
        improved = False
        cur_peak = _max_convolution_fft(a)
        for factor in (1.0, 0.5, 0.25):
            step = lr * factor
            a_new = _project_to_simplex(a - step * grad, target_sum)
            new_peak = _max_convolution_fft(a_new)
            if new_peak < cur_peak:
                a = a_new
                cur_peak = new_peak
                improved = True
                break

        if not improved:
            # increase smoothness (larger alpha) to escape flat regions
            alpha = min(alpha * 1.5, max_alpha)
        else:
            # slowly increase alpha to sharpen the objective
            alpha = min(alpha * 1.1, max_alpha)
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = a.copy()

        # occasional tiny jitter to avoid stagnation
        if rng.random() < 0.02:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Permutation simulated annealing (swap based)
# ----------------------------------------------------------------------


def _permutation_sa(a: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random swaps of indices with a Metropolis acceptance rule.
    The sum and non‑negativity are preserved (swaps do not change them).
    """
    n = a.shape[0]
    cur_seq = a.copy()
    cur_peak = _max_convolution_fft(cur_seq)

    best_seq = cur_seq.copy()
    best_peak = cur_peak

    start = time.time()
    T0 = max(1e-4, cur_peak * 0.2)
    T_end = 1e-6

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * (1 - frac) + T_end * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        new_seq = cur_seq.copy()
        new_seq[i], new_seq[j] = new_seq[j], new_seq[i]

        new_peak = _max_convolution_fft(new_seq)

        if new_peak < cur_peak or rng.random() < np.exp(-(new_peak - cur_peak) / max(T, 1e-12)):
            cur_seq = new_seq
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_seq = cur_seq.copy()

    return best_seq, best_peak


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with minimal evaluation score.
    The routine runs within the supplied time budget and stores the best
    sequence in the global variable ``height_sequence_1`` for warm‑starting.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives richer search space but higher cost.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 500:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    seeds = []
    for p in np.linspace(0.1, 1.5, 12):
        seeds.append(_power_law_sequence(n, float(p), target_sum))
    for lam in np.linspace(0.001, 0.05, 10):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # --------------------------------------------------------------
    # Initialise DE population
    # --------------------------------------------------------------
    pop_size = max(30, int(n / 250))  # ≈30‑40 individuals
    population = []

    best_seed = min(seeds, key=_max_convolution_fft)
    population.append(best_seed.copy())
    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.random(n), target_sum))

    # --------------------------------------------------------------
    # Differential Evolution phase (≈30 % of the budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.30 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_peak = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best peak = {best_peak:.6f}")

    # --------------------------------------------------------------
    # Smooth‑gradient descent on a few promising candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate roughly 60 % of the remaining time to gradient descent
        grad_deadline_total = time.time() + 0.60 * remaining
        # pick three candidates (the DE result + two random seeds)
        cand_list = [best_seq]
        while len(cand_list) < 3:
            cand_list.append(seeds[rng.integers(len(seeds))])

        # split remaining time evenly among them
        per_cand_deadline = (grad_deadline_total - time.time()) / len(cand_list)
        for i, cand in enumerate(cand_list):
            cand_dead = time.time() + per_cand_deadline
            seq_g, peak_g = _smooth_gradient_descent(cand, target_sum, cand_dead, rng)
            print(f"[{time.time():.2f}] Gradient descent #{i} – peak = {peak_g:.6f}")
            if peak_g < best_peak:
                best_peak = peak_g
                best_seq = seq_g

    # --------------------------------------------------------------
    # Permutation simulated annealing (swap based)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_deadline = deadline - 2.0
        seq_p, peak_p = _permutation_sa(best_seq, target_sum, perm_deadline, rng)
        if peak_p < best_peak:
            best_peak = peak_p
            best_seq = seq_p
        print(f"[{time.time():.2f}] Permutation SA – final peak = {best_peak:.6f}")

    # --------------------------------------------------------------
    # Deterministic re‑orderings as a final safety check
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_peak = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] After deterministic reorder – peak = {best_peak:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for possible warm‑start later
    globals()["height_sequence_1"] = final_seq

    # sanity check using the provided evaluator (if available)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
