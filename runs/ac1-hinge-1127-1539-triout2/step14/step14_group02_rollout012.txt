"""
Improved step‑function search.

The algorithm runs several independent starts (warm‑start, deterministic
shapes and random vectors).  For each start it performs three phases
inside the allocated time slice:

1. Smooth‑max projected gradient descent – a soft‑max surrogate of the
   max‑convolution entry is minimised.  The temperature β is raised
   gradually so that the final solution approximates the true max.

2. Aggressive mass‑transfer simulated annealing – tiny amounts of mass
   are moved from indices that contribute strongly to the current peak
   towards indices with small height.  The convolution is updated in
   O(n) per move.

3. Peak‑gradient refinement – the exact gradient of the current peak
   (∂max / ∂a_i = 2·a_{k‑i}) is used in a back‑tracking line‑search
   projected onto the simplex.  This deterministic stage removes the
   last few percent of the peak.

Multiple restarts share the overall wall‑clock budget, guaranteeing a
solution is returned before the timer expires.  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_last = rho[-1]
        theta = cssv[rho_last] / (rho_last + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 25.0) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution entry.
    The approximation is (1/β)·log Σ exp(β·(a*a)_i).
    After each step the vector is projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    b = _conv_self_fft(a)
    best_val = float(2.0 * n * b.max() / (target_sum * target_sum))

    beta = beta_start
    step0 = target_sum * 0.8  # raw step size before normalisation
    it = 0

    while time.time() < deadline:
        # self‑convolution and soft‑max weights
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # log‑sum‑exp (stable)
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ a_rev) (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)          # length ≈ 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]    # central n entries

        # step size (decays with √it)
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # slowly increase β
        if it % 1200 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.4)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves tiny amounts of mass from ‘bad’
    indices (large contribution to the current peak) to ‘good’
    indices (small height).  The convolution is updated in O(n) for
    each move, exactly as in the reference implementation.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = 2.0 * n * b.max() / (target_sum * target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = max(1e-9, deadline - start)
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # square‑root cooling

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0.0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 3000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = 2.0 * n * b.max() / (target_sum * target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – deterministic peak‑gradient refinement
# ----------------------------------------------------------------------


def peak_gradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass in the exact direction that
    reduces the current peak a*a[k_max] (gradient = 2·a_{k_max‑i}).
    A back‑tracking line search guarantees a strict decrease of the
    max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_max = b.max()
    best_a = a.copy()
    best_val = cur_max

    while time.time() < deadline:
        k = int(b.argmax())

        # gradient of the peak entry (2·a_{k‑i} on the valid range)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        # initial step – a modest fraction of the total mass
        step = target_sum * 0.2 / norm

        improved = False
        for _ in range(30):
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new, target_sum)
            b_new = _conv_self_fft(a_new)
            new_max = b_new.max()
            if new_max < cur_max - 1e-12:
                a = a_new
                b = b_new
                cur_max = new_max
                improved = True
                if new_max < best_val:
                    best_val = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] peak‑grad improved best={best_val:.6f}")
                break
            else:
                step *= 0.5
                if step * norm < 1e-12:
                    break
        if not improved:
            # no further progress – quit early
            break

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)  # normalisation used in the benchmark

    # --------------------------------------------------------------
    # Build a collection of start vectors (warm‑start + deterministic)
    # --------------------------------------------------------------
    init_candidates = []

    # warm‑start from the previous run, if any
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()

    # deterministic shapes
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    init_candidates.append(uniform)

    # power‑law
    alpha = 0.55
    power = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
    power = power * (target_sum / power.sum())
    init_candidates.append(power)

    # sine shape (positive half‑wave)
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sine = sine * (target_sum / sine.sum())
    init_candidates.append(sine)

    # triangular (peak in the middle)
    tri = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 1:
        tri = np.concatenate((tri, tri[-2::-1]))
    else:
        tri = np.concatenate((tri, tri[::-1]))
    tri = tri[:n]
    tri = tri * (target_sum / tri.sum())
    init_candidates.append(tri)

    # a few random starts
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    # --------------------------------------------------------------
    # Run the optimisation phases for each start, sharing the time budget
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    remaining = len(init_candidates)
    for idx, init in enumerate(init_candidates):
        now = time.time()
        if now >= deadline - 5.0:          # keep a tiny reserve
            break
        # allocate equal share of the remaining wall‑clock time
        remain_time = deadline - now
        sub_deadline = now + remain_time / (remaining - idx)

        print(f"[{time.time():.2f}] ----- start #{idx + 1} (n={n}) -----")

        # ---- Phase 1 : smooth‑max projected gradient descent ------------
        phase1_dead = time.time() + (sub_deadline - time.time()) * 0.30
        a_smooth, val_smooth = smooth_max_descent(
            init, target_sum, phase1_dead,
            beta_start=0.08, beta_max=40.0
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ---- Phase 2 : aggressive annealing ----------------------------
        # give the bulk of the remaining budget to annealing
        if time.time() < sub_deadline:
            phase2_dead = sub_deadline - 3.0  # leave ~3 s for the final refinement
            a_anneal, val_anneal = anneal_mass_transfer(
                a_smooth, target_sum, phase2_dead
            )
            if val_anneal < best_score:
                best_score = val_anneal
                best_seq = a_anneal.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

        # ---- Phase 3 : deterministic peak‑gradient refinement ----------
        if time.time() < sub_deadline:
            a_peak, val_peak = peak_gradient_descent(
                best_seq if best_seq is not None else a_anneal,
                target_sum,
                sub_deadline
            )
            if val_peak < best_score:
                best_score = val_peak
                best_seq = a_peak.copy()
                print(f"[{time.time():.2f}] New global best after peak‑grad: {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – decreasing power‑law
        fallback = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = fallback * (target_sum / fallback.sum())

    arr = np.array(best_seq, dtype=np.float64)

    # trim trailing near‑zeros for a slightly shorter representation
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (outside of the optimisation loop)
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store the result for a possible warm‑start on the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
