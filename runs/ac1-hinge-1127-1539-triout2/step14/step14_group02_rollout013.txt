    start = time.time()

    elif budget_s >= 500:

    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    # warm start if available
    if "height_sequence_1" in globals():
        init_seq = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init_seq = _project_to_simplex(init_seq, target_sum)
        init_candidates.append(init_seq)

    # exponent families
    for p in [0.45, 0.5, 0.55, 0.6, 0.65]:
        seq = np.arange(1, n+1, dtype=np.float64) ** (-p)
        seq = _project_to_simplex(seq, target_sum)
        init_candidates.append(seq)

    # sparse uniform
    m = int(2.5 * np.sqrt(n))
    positions = np.linspace(0, n-1, m, dtype=int)
    seq = np.zeros(n, dtype=np.float64)

    # random dense
    rng = np.random.default_rng()
    for _ in range(2):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_val = np.inf

    remaining_candidates = len(init_candidates)
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break
        # allocate fraction of remaining time per remaining candidates
        now = time.time()
        remaining_time = deadline - now
        # allocate half to smooth, half to hill
        sub_deadline_smooth = now + remaining_time * 0.45
        a_smooth, val_smooth = smooth_max_descent(init, target_sum, sub_deadline_smooth,

        if cur_val < best_val:

            best_seq = a_smooth.copy()

def smooth_max_descent(a0, target_sum, deadline, beta_start=0.1, beta_max=30.0):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # compute initial max:
    b = _conv_self_fft(a)
    best_val = b.max()
    beta = beta_start
    it = 0
    step0 = target_sum * 0.8
    while time.time() < deadline:
        # compute b
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it} best={best_val:.6f}")
        # compute softmax weights
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # compute gradient
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)  # length (size) maybe >= n?
        # central slice length n
        g = 2.0 * conv_wa[n-1:n-1+n]   # central n entries
        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)
        a = a - eta * g
        # projection to simplex
        a = _project_to_simplex(a, target_sum)
        # update beta
        if it % 500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)
        it += 1
    return best_a, best_val

"""
Multi‑stage search for a low‑peak step‑function.

1.  Build a diverse set of initial sequences (warm‑start, power‑law,
    sparse uniform and random dense).
2.  Run a quick smooth‑max projected gradient descent on each candidate
    (soft‑max approximation of the convolution max).  The best candidate
    after this coarse stage is retained.
3.  Refine the retained candidate with a longer smooth‑max descent
    (larger β) followed by a deterministic hill‑climbing “mass‑transfer”
    phase that moves tiny amounts of mass from indices that contribute most
    to the current peak towards indices with the smallest height.
    The convolution is updated in O(n) per move, exactly as in the reference
    implementation.

All operations respect the simplex constraint Σa = √(2·n) and a_i ≥ 0,
so the returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, power‑of‑2 padding)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a*a) using projected gradient descent.
    The approximation is (1/β)·log Σ exp(β·(a*a)_i).
    The vector is projected onto the simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    b = _conv_self_fft(a)
    best_val = b.max()
    beta = beta_start
    it = 0
    step0 = target_sum * 0.8

    while time.time() < deadline:
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it} best={best_val:.6f}")

        # soft‑max weights (stable)
        b_shift = b - max_b                     # max(b_shift) == 0
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w * a_rev)  (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)           # length ≥ n
        g = 2.0 * conv_wa[n - 1: n - 1 + n]     # central n entries

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)
        a -= eta * g
        a = _project_to_simplex(a, target_sum)

        # increase β gradually
        if it % 500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – deterministic hill‑climbing mass transfer
# ----------------------------------------------------------------------


def hill_climb(a0: np.ndarray,
               target_sum: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer optimiser.
    In each iteration the index that contributes most to the current peak
    (donor) gives a small amount of mass to the smallest index (receiver).
    The convolution is updated in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = b.max()
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    it = 0

    while time.time() < deadline:
        # --- locate the current maximum of the convolution ----------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.max() > 0.0:
            donor = int(idx[np.argmax(contrib)])
        else:
            donor = int(rng.integers(lo, hi + 1))

        # receiver – index with smallest height (avoid trivial move)
        receiver = int(np.argmin(a))
        if receiver == donor:
            # pick a neighbour if all values equal
            receiver = (donor + 1) % n

        # --- propose a tiny transfer ------------------------------------
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            # nothing to move; break to avoid infinite loop
            break

        # temperature‑driven scaling (cooling)
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, (1.0 - np.sqrt(frac))) * cur_val
        delta = rng.random() * max_delta * temperature / (cur_val + 1e-12)
        if delta <= 0.0:
            continue

        # --- O(n) update of convolution ---------------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = b_new.max()

        # --- accept / reject --------------------------------------------
        accept = False
        if new_val <= cur_val:
            accept = True
        else:
            # occasional uphill moves (Metropolis)
            if temperature > 0.0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] hill improved best={best_val:.6f}")

        # periodic full recompute to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute best={best_val:.6f}")

        it += 1

    return best_a, best_val

                                         beta_start=0.1,
                                         beta_max=30.0)
        if val_s < best_score:
            best_score = val_s
            best_seq = a_s.copy()
            print(f"[{time.time():.2f}] New global best after coarse smooth: {best_score:.6f}")

    # --------------------------------------------------------------
    # Refine the current best candidate with a longer optimisation
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # longer smooth descent (larger β)
        sub_dead = time.time() + (deadline - time.time()) * 0.5
        a_ref, val_ref = smooth_max_descent(best_seq, target_sum,
                                            sub_dead,
                                            beta_start=0.5,
                                            beta_max=120.0)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] New global best after long smooth: {best_score:.6f}")

        # final hill‑climbing phase
        if time.time() < deadline:
            a_final, val_final = hill_climb(best_seq, target_sum, deadline)
            if val_final < best_score:
                best_score = val_final
                best_seq = a_final.copy()
                print(f"[{time.time():.2f}] New global best after hill climb: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Post‑process: trim trailing zeros and return as plain list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (outside the optimisation loop)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
