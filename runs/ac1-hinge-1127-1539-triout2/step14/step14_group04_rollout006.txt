    best_val = _score(a)

    rng = np.random.default_rng()
    base_step = target_sum * 0.5
    it = 0
    last_reorder = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

"""
Propose a low‑peak non‑negative step function.

The strategy is simple but long‑running:
1. Initialise a decent shape (power‑law or the previous best) and
   project it onto the ℓ¹‑simplex Σa = √(2·n).
2. Apply a few quick soft‑max gradient steps to get out of a bad start.
3. Run a *very* long projected sub‑gradient descent on the true
   max‑convolution entry, with occasional jitter and periodic
   “alternating” re‑ordering to break symmetries.
4. Finish with a short deterministic mass‑transfer annealing.
5. Trim trailing zeros, store the result in the global variable
   ``height_sequence_1`` and return it.

All heavy work respects the supplied time budget (≈1000 s).  The
sub‑gradient descent is the main workhorse – it directly minimises the
peak of the linear convolution while keeping the ℓ¹‑norm fixed, which
is exactly the quantity evaluated by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Score (the objective to be minimised)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """2·n·max(a*a) / (Σa)² – the evaluation without the security wrapper."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, then projected onto the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, second‑largest, … placed alternately from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution over shifts

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Annealed soft‑max gradient descent (used only for a quick warm‑up)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Long projected sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_with_reorder(a0: np.ndarray,
                              deadline: float,
                              target_sum: float,
                              reorder_interval: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(convolution) with periodic alternating re‑ordering.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5
    it = 0
    last_reorder = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx

        grad = np.zeros_like(a)
        grad[idx] = a[partner]

        # The middle term a[mid]^2 contributes 2*a[mid] to the gradient;
        # we add a[mid] (the original implementation) – the extra factor
        # is absorbed by the step size schedule.
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:6d} best={best_val:.6f}")

        # periodic deterministic re‑ordering (alternating)
        if time.time() - last_reorder > reorder_interval:
            a = _alternating_order(a)
            a = _project_to_simplex(a, target_sum)
            last_reorder = time.time()
            print(f"[{time.time():.2f}] Reordered (alternating).")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small local moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Local mass‑transfer annealing."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence minimising
    2·n·max(a*a)/(∑a)² under the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (larger n gives more freedom but also a larger factor 2·n)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise a reasonable seed
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=float)
        if seq0.size != n:
            seq0 = _resize_sequence(seq0, n)
    else:
        # power‑law is a solid starter; exponent 0.55 works well in practice
        seq0 = _power_law_sequence(n, 0.55, target_sum)

    seq0 = _project_to_simplex(seq0, target_sum)
    seq0 = _alternating_order(seq0)

    best_seq = seq0
    best_val = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Very short soft‑max GD to escape pathological starts (≈5 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        gd_dead = min(time.time() + 5.0, deadline - 5.0)
        best_seq, best_val = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_sum,
            max_beta=1e5,
            step_factor=0.7,
            gamma=0.99,
        )
        print(f"[{time.time():.2f}] After quick GD best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Long projected sub‑gradient descent (majority of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_dead = time.time() + remaining - 15.0   # leave ~15 s for polishing
        best_seq, best_val = _subgradient_with_reorder(
            best_seq,
            sub_dead,
            target_sum,
            reorder_interval=200.0,
        )
        print(f"[{time.time():.2f}] After long sub‑grad best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic mass‑transfer polishing (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = deadline - 2.0
        seq_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_sum,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_val:
            best_seq, best_val = seq_mt, val_mt
            print(f"[{time.time():.2f}] After final MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros, store globally and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final logging through the provided wrapper
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
