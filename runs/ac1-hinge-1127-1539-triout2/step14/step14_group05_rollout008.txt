def _adam_subgradient(a0, deadline, target_sum, step0=1.0, beta1=0.9, beta2=0.999, eps=1e-8, tol=0.999):

    a = _project_to_simplex(a0.copy(), target_sum)

    m = np.zeros_like(a)

    v = np.zeros_like(a)   # maybe not needed.

    best_a = a.copy()

    best_val = _max_convolution_fft(a)

    it = 0

    while time.time() < deadline:

        # compute conv, M, mask

        conv = _conv_fft(a)

        M = conv.max()

        # mask I: conv >= M * tol

        I = (conv >= M * tol).astype(np.float64)

        # gradient: 2 * conv of a_rev with I

        a_rev = a[::-1]

        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())

        I_fft = np.fft.rfft(I, n=size)

        rev_fft = np.fft.rfft(a_rev, n=size)

        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[:len(I)+len(a_rev)-1]

        grad = 2.0 * grad_full[len(a_rev)-1:len(a_rev)-1 + len(a)]

        # Adam update

        m = beta1 * m + (1 - beta1) * grad

        v = beta2 * v + (1 - beta2) * (grad**2)

        m_hat = m / (1 - beta1**(it+1))

        v_hat = v / (1 - beta2**(it+1))

        # step size schedule: base / (sqrt(v_hat) + eps)

        step = step0 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat

        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)

        if cur_val < best_val:

            best_val = cur_val

            best_a = a.copy()

            print(...)

        if it % 1000 == 0:

            # maybe jitter

            jitter = rng.normal(scale=target_sum*0.001/(it+1), size=a.shape)

            a += jitter

            a = _project_to_simplex(a, target_sum)

        it+=1

    return best_a, best_val

```python
"""
Hybrid optimizer for step function minimization.

The approach:
1. Build a diverse pool of candidate sequences at a modest size (n≈3000)
   using several parametric families (power‑law, exponential, sinusoid, beta‑like)
   and a random simplex. Keep the best.
2. Upscale the best candidate to a much larger support (n up to 50000)
   via linear interpolation and project onto the ℓ1‑simplex.
3. Run a long projected sub‑gradient descent on the exact max‑convolution
   (the objective equals max(conv) when the ℓ1‑norm is √(2n)).
   Momentum (Adam‑style) and occasional jitter keep the iterates moving.
4. Finish with a short simulated‑annealing mass‑transfer phase to
   fine‑tune the final distribution.
The final list is trimmed of trailing near‑zero entries and returned.
"""

# Imports
import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return the maximum entry of the linear convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = s, all non‑negative)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, Σx = s}."""
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find rho
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ i^{-p} (1‑based)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i) (i starts at 0)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π * i/(n‑1))^power, symmetric."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x) ** power
    a *= target_sum / a.sum()
    return a


def _beta_like_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{alpha} * (n-i)^{beta}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_len, dtype=np.float64)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Sub‑gradient descent on max‑convolution (Adam‑style)
# ----------------------------------------------------------------------


def _subgradient_adam(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      step0: float = 1.0,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      tol: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the exact max‑convolution.
    At each iteration the gradient is computed from the set of
    convolution entries whose value is at least `tol` times the current maximum.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    m = np.zeros_like(a)    # first moment
    v = np.zeros_like(a)    # second moment
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # convolution a*a
        conv = _conv_fft(a)
        M = conv.max()
        # mask of near‑max entries
        I = (conv >= M * tol).astype(np.float64)

        # gradient of M (sub‑gradient)
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        step = step0 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam subgrad it={it:6d} M={best_val:.6f}")

        # occasional jitter to escape flat regions
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001 / math.sqrt(it + 1), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer (fine‑tuning)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.2,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy pairwise mass transfer with Metropolis acceptance."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # find current maximum index
        kmax = int(np.argmax(conv))

        # feasible i such that 0 <= i <= n-1 and 0 <= kmax-i <= n-1
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributions for this kmax
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        # choose a receiver among indices with small product
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

"""
Hybrid optimizer for the step‑function benchmark.

Goal: minimise 2·n·max(conv)/(∑a)².  Because we keep the ℓ¹‑norm equal to
√(2n), the objective is simply max(conv).  The algorithm proceeds in
four stages:

1. **Exploration on a modest size (≈3000).**  A pool of parametric families
   (power‑law, exponential, half‑sine, beta‑like) and random simplices is
   generated, projected onto the simplex and the best (smallest max
   convolution) seed kept.

2. **Upscaling.**  The best seed is linearly interpolated to a large
   support (up to 50 000 entries) and again projected onto the required
   ℓ¹‑simplex.  Larger n gives the optimiser more freedom to flatten the
   autocorrelation.

3. **Long projected sub‑gradient descent.**  Using the exact
   max‑convolution sub‑gradient (via FFT) together with a momentum term
   (similar to Adam/Polyak) we run for almost the whole remaining budget.
   Occasional jitter avoids stagnation.

4. **Fine‑grained simulated‑annealing mass‑transfer.**  A greedy pairwise
   mass‑transfer move is accepted with a Metropolis rule; this polishing
   step yields the final low‑peak sequence.

The final sequence is trimmed of trailing near‑zero entries, stored in the
global variable ``height_sequence_1`` (so later runs can reuse it) and
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex (non‑negative, prescribed sum)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the initial search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ i^{-p} (1‑based)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π·i/(n‑1))^power."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x) ** power
    a *= target_sum / a.sum()
    return a


def _beta_like_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^α·(n‑i)^β."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len (preserves total mass)."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_len, dtype=np.float64)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (momentum) on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995,
                         tol: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Momentum projected sub‑gradient descent for max(conv).
    The sub‑gradient is taken from all convolution entries whose value is
    at least ``tol``·M, where M is the current maximum.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        I = (conv >= M * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:6d} max‑conv={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer (final polishing)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.15,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy pairwise mass transfer with a Metropolis acceptance rule.
    ``low_frac`` controls the size of the candidate set for the receiver
    (indices that contribute little to the current maximum).
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where the convolution attains its maximum
        kmax = int(np.argmax(conv))

        # feasible i such that i and kmax‑i lie in [0, n‑1]
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributions for this kmax
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake the system
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.

    expires.  All intermediate stages print progress messages.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n_small = 3000
    target_small = math.sqrt(2.0 * n_small)

    best_seq = None
    best_val = np.inf

    # power‑law family
    for p in np.linspace(0.2, 2.0, 10):

        val = _max_convolution_fft(seq)
        if val < best_val:

    # half‑sine raised to a power

        val = _max_convolution_fft(seq)
        if val < best_val:

    # beta‑like shapes
    for a_par, b_par in [(0.5, 0.5), (1.0, 0.5), (0.5, 1.0), (1.5, 1.5), (2.0, 2.0)]:
        seq = _beta_like_sequence(n_small, a_par, b_par, target_small)
        val = _max_convolution_fft(seq)

    # random points on the simplex

    print(f"[{time.time():.2f}] Stage 1 best max‑conv = {best_val:.6f}")

    # --------------------------------------------------------------
