"""
Low‑score non‑negative step sequence search.

Algorithm (≈1000 s budget)

1️⃣  **Initial families** – warm‑start from the previously stored best
    (`height_sequence_1`) or a handful of deterministic shapes
    (power‑law, exponential, combined, symmetric Beta).  The best of these
    seeds the optimisation.

2️⃣  **Smooth‑max gradient descent** – replace the true max in the
    objective by a smooth approximation
        M_β(a) = (1/β)·log∑ₖ exp(β·(a∗a)ₖ) .
    Its gradient is easily obtained via a single FFT‑based convolution.
    β is increased linearly from 0.5 to ≈10 so that early iterations move
    mass globally and later iterations focus on the true peak.

3️⃣  **Projected sub‑gradient descent** – fine‑tune the exact max
    (as in the reference implementation).

4️⃣  **Aggressive annealing** – a fast O(n) mass‑transfer simulated‑annealing
    that moves tiny amounts from “bad’’ indices (large contribution to the
    current peak) to “good’’ ones.

All intermediate vectors are projected onto the simplex
{ x≥0 , Σx = √(2·n) } to keep the denominator fixed.
The final sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear convolution) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary real vectors (FFT version)."""
    n = x.size + y.size - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


def _shape_ends(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: high at both ends, low in the middle."""
    i = np.arange(n, dtype=np.float64)
    left = (i + 1) ** (-p)
    right = (n - i) ** (-p)
    return left + right


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Gradient descent on a smooth approximation of max(a*a)."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    # schedule for β (smoothness) – from 0.5 to 10.0 linearly
    beta_start = 0.5
    beta_end = 10.0

    # step size (will be decayed)
    eta0 = 0.5

    while time.time() < deadline:
        # fraction of total planned iterations (we just use elapsed / total)
        elapsed = time.time() - start
        total = deadline - start
        frac = min(1.0, elapsed / total)

        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta0 * (1.0 - frac) + 0.01  # never below 0.01

        # self‑convolution and current score
        b = _conv_fft(a)
        cur_val = 2.0 * n * b.max() / (a.sum() ** 2)

        # keep the best seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth descent it={it:6d} β={beta:.2f} "
                  f"score={cur_val:.6f}")

        # ----- smooth max (soft‑max) -----
        shift = b.max()
        w = np.exp(beta * (b - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # ----- gradient of smooth max -----
        w_rev = w[::-1]
        conv_full = _conv_full_fft(a, w_rev)          # length 3n‑2
        len_w = w.shape[0]                           # = 2n‑1
        grad = 2.0 * conv_full[(len_w - 1) - np.arange(n)]  # shape (n,)

        # gradient step + projection onto simplex
        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 750 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_projected_descent(a0: np.ndarray, target_sum: float, deadline: float):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        # derivative of the active peak (2·a_{k−i})
        grad[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑proj it={it:6d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Phase 3 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = 2.0 * n * b.max() / (target_sum * target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak (a_i·a_{k-i})
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver favours small heights
        inv = 1.0 / (a + 1e-12)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue

        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ---- O(n) update of convolution (exact) ----
        b_new = b.copy()
        # linear part 2·a·Δ
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = 2.0 * n * b.max() / (target_sum * target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val
