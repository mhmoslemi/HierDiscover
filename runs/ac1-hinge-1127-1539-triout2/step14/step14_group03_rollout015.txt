"""
Improved search for a non‑negative step sequence that minimizes
max (a * a) under the constraint Σa = √(2·n).  The algorithm combines
three ideas:

1.  Deterministic shape families (including a symmetric power‑law)
    for a strong warm‑start.
2.  A smooth‑max gradient descent.  The true max is approximated by
    (1/β)·log Σ exp(β·(a*a)) and the gradient is computed via a single
    FFT‑based convolution.  β is increased gradually (continuation
    method) to approach the true max.
3.  The original sub‑gradient descent and aggressive simulated‑annealing
    (mass‑transfer) refinements are kept as final polishing steps.

All helper functions are top‑level; no closures or lambdas are used.
The routine respects the given time budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear self‑convolution)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Return the full linear self‑convolution a * a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (alpha - 1.0)
    )
    return a


def _shape_symmetric_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) + (n-i)^(-p)  (large mass at both ends)."""
    left = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    right = np.arange(n, 0, -1, dtype=np.float64) ** (-p)
    return left + right


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (β‑continuation)
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, b: np.ndarray, beta: float):
    """
    Gradient of the smoothed max Fβ(a) = (1/β)·log Σ exp(β·b_k)
    where b = a*a.  Returns (grad, Fβ(a)).
    """
    maxb = b.max()
    # weights proportional to exp(β·b_k)
    expb = np.exp(beta * (b - maxb))
    sumexp = expb.sum()
    p = expb / sumexp  # Σ p = 1

    # smooth objective value
    smooth_val = (np.log(sumexp) + beta * maxb) / beta

    # convolution p * a_rev  (a_rev[i] = a[n-1-i])
    a_rev = a[::-1]
    L = len(p) + len(a_rev) - 1
    size = 1 << (L.bit_length())
    fp = np.fft.rfft(p, n=size)
    fra = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(fp * fra, n=size)[:L]

    # gradient_i = 2 * Σ_k p_k a_{k-i}
    grad = 2.0 * conv[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, smooth_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_projected_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Classic sub‑gradient method with Euclidean projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_full(a).max()

    rng = np.random.default_rng()
    it = 0
    start = time.time()
    while time.time() < deadline:
        b = _conv_full(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        # derivative of max entry (2·a_{k–i})
        grad[idx] = 2.0 * a[k_max - idx]

        # simple diminishing stepsize
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = b.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Mass‑transfer simulated annealing (identical to the previous version)."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_full(a)
    cur_val = b.max()
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9   # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = b_new.max()

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_full(a)
            cur_val = b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Recalc improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low max‑convolution
    value.  The routine respects the time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # small safety margin

    # Choose a problem size depending on the allotted time.
    if budget_s >= 800:
        n = 24000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm start from previously found solution (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        val = _conv_full(warm).max()
        best_seq = warm.copy()
        best_val = val
        print(f"[{time.time():.2f}] Warm start score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Deterministic shape sweep (cheap but effective)
    # ------------------------------------------------------------------
    # power‑law
    for p in np.linspace(0.45, 0.75, 7):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _conv_full(a).max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} val={val:.6f}")

    # exponential
    for lam in np.linspace(0.5, 8.0, 7):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _conv_full(a).max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.3f} val={val:.6f}")

    # combined power‑law + exponential (random sweeps)
    rng = np.random.default_rng()
    for _ in range(8):
        p = rng.random() * 0.8 + 0.1   # 0.1 … 0.9
        lam = rng.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _conv_full(a).max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] PowerExp p={p:.3f} λ={lam:.3f} val={val:.6f}")

    # symmetric Beta family
    for alpha in np.linspace(0.25, 0.9, 7):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        val = _conv_full(a).max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.3f} val={val:.6f}")

    # symmetric power‑law (large mass at both ends)
    for p in np.linspace(0.4, 0.8, 5):
        a = _shape_symmetric_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _conv_full(a).max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] SymPower p={p:.3f} val={val:.6f}")

    # Random Dirichlet starts (very cheap)
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _conv_full(rnd).max()
        if val < best_val:
            best_val = val
            best_seq = rnd.copy()
            print(f"[{time.time():.2f}] Random start val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (β‑continuation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 40 % of the remaining budget to this phase
        grad_deadline = time.time() + 0.40 * remaining

        a = best_seq.copy()
        betas = [0.5, 1.0, 2.0, 5.0]          # increasing sharpness
        for beta in betas:
            phase_end = time.time() + (grad_deadline - time.time()) / len(betas)
            while time.time() < phase_end:
                b = _conv_full(a)
                grad, smooth_val = _smooth_gradient(a, b, beta)

                grad_norm2 = np.dot(grad, grad)
                # Polyak step towards the theoretical lower bound (1.0)
                eta = max((smooth_val - 1.0) / (grad_norm2 + 1e-12), 0.0)
                # guard against too large steps
                eta = min(eta, 0.2 * target_sum)

                a = a - eta * grad
                a = _project_to_simplex(a, target_sum)

                cur_val = b.max()
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] β={beta:.2f} smooth best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – classic sub‑gradient descent (more time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.30 * (deadline - time.time())
        a_sub, val_sub = subgradient_projected_descent(
            best_seq, target_sum, sub_deadline
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive annealing mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up, trimming trailing zeros
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final verification with the provided evaluator
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start of the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
