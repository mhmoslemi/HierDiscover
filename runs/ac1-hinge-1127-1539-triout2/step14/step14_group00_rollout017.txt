"""
Hybrid optimizer for the step‑function max‑convolution problem.
Improved with an L2‑flattening gradient descent phase that directly
minimises the variance of the self‑convolution (i.e. pushes the
convolution towards the uniform constant 1).  The algorithm also
retains the previous DE, smooth‑max, mass‑transfer and sub‑gradient
polishing steps for robustness.
"""
import time
import numpy as np

# ----- convolution utilities -----
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv

def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    return float(_conv_self(a).max())

# ----- objective -----
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# ----- simplex projection -----
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    # Euclidean projection onto {x>=0, sum x = s}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----- seed factories -----
def _uniform_sequence(n: int, target_sum: float) -> np.ndarray:
    return np.full(n, target_sum / n, dtype=np.float64)

def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a = a / a.sum() * target_sum
    return a

def _sidon_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    # Greedy construction of a Sidon set (no sum collisions).
    # We stop when we have roughly target_sum elements (≈√(2n)).
    used = set()
    indices = []
    # order random to diversify
    order = rng.permutation(n)
    for idx in order:
        # check collisions
        collision = False
        for j in indices:
            if (idx + j) in used:
                collision = True
                break
        if collision:
            continue
        # also check self-sum
        if (idx + idx) in used:
            continue
        # accept
        for j in indices:
            used.add(idx + j)
            used.add(j + idx)
        used.add(idx + idx)
        indices.append(idx)
        # stop when we have enough mass
        if len(indices) >= int(np.ceil(target_sum)):
            break
    a = np.zeros(n, dtype=np.float64)
    a[indices] = 1.0
    a = _project_to_simplex(a, target_sum)
    return a

# ----- reordering helpers -----
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    turn = True
    for v in sorted_vals:
        if turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        turn = not turn
    return out

def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

# ----- Differential Evolution (projected) -----
def _de_optimize(population, target_sum, deadline, F=0.5, CR=0.9):
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)
            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score

# ----- L2 flattening gradient descent -----
def _flatten_gradient_descent(a0: np.ndarray, target_sum: float,
                             deadline: float,
                             initial_step: float = 0.8,
                             decay: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Minimise the L2 distance between the self‑convolution of the probability
    vector and the uniform constant 1/(2n‑1).  The gradient is computed by an
    FFT‑based convolution, then the iterate is projected onto the ℓ¹‑simplex.
    """
    n = a0.shape[0]
    L = target_sum
    p = a0 / L                     # probability vector, sums to 1
    best_p = p.copy()
    best_score = _score(a0)

    step = initial_step
    rng = np.random.default_rng()
    # target constant for the convolution of probabilities
    uniform_val = 1.0 / (2.0 * n - 1.0)

    while time.time() < deadline:
        # compute convolution of probabilities
        conv = _conv_self(p)                 # length 2n‑1
        error = conv - uniform_val           # deviation from flatness
        # gradient of ½‖error‖² = (error * p_rev)  (factor 2 absorbed later)
        grad = 2.0 * _fft_convolve(error, p[::-1])
        # slice central part of length n
        grad = grad[n - 1:n - 1 + n]

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        # step size inversely proportional to norm, scaled by step factor
        lr = step / (norm + 1e-12)

        p_new = p - lr * grad
        p_new = _project_to_simplex(p_new, 1.0)

        a_new = p_new * L
        new_score = _score(a_new)

        if new_score < best_score:
            best_score = new_score
            best_p = p_new.copy()
            # be a bit bolder when improving
            step = min(step * 1.05, 2.0)
        else:
            step = max(step * decay, 1e-4)

        p = p_new
    return best_p * L, best_score

# ----- Deterministic mass‑transfer polishing -----
def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

# ----- Sub‑gradient descent (exact) -----
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

# ----- main entry point -----
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser.  First a population based DE explores the space,
    then a dedicated L2‑flattening gradient descent pushes the
    self‑convolution towards the uniform constant.  Finally the
    deterministic mass‑transfer and sub‑gradient polishing are run.
    The best sequence found before the time budget expires is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem dimension – 8000 works well for the full budget
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ----------- seed generation -------------
    seeds: list[np.ndarray] = []

    # 1. simple uniform
    seeds.append(_uniform_sequence(n, target_sum))

    # 2. random Dirichlet points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # 3. Sidon‑like construction
    seeds.append(_sidon_sequence(n, target_sum, rng))

    # 4. previous best (warm start)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev_resized = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(prev)), prev)
        prev_resized = _project_to_simplex(prev_resized, target_sum)
        seeds.append(prev_resized)

    # jitter each seed a little to increase diversity
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # ---------- initialise differential evolution ----------
    pop_size = max(30, n // 250)            # a couple dozen individuals
    population = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())
    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # ------ DE phase ------
    de_deadline = min(start_time + 0.45 * budget_s, deadline - 30.0)
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished; best score = {best_score:.6f}")

    # ------ L2 flattening gradient descent (new) ------
    if time.time() < deadline - 30.0:
        flat_deadline = time.time() + min(20.0, deadline - time.time() - 20.0)
        seq_f, score_f = _flatten_gradient_descent(best_seq, target_sum,
                                                   flat_deadline,
                                                   initial_step=0.8,
                                                   decay=0.85)
        if score_f < best_score:
            best_score, best_seq = score_f, seq_f.copy()
            print(f"[{time.time():.2f}] After L2‑flattening best = {best_score:.6f}")

    # ------ deterministic mass‑transfer polishing ------
    if time.time() < deadline - 12.0:
        mt_dead = time.time() + min(12.0, deadline - time.time() - 8.0)
        seq_mt, score_mt = _deterministic_mass_transfer(best_seq, mt_dead, target_sum,
                                                       start_temp=0.4, end_temp=1e-5,
                                                       low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # ------ short sub‑gradient descent ------
    if time.time() < deadline - 5.0:
        sg_dead = time.time() + min(8.0, deadline - time.time() - 3.0)
        seq_sg, score_sg = _subgradient_descent(best_seq, sg_dead, target_sum,
                                               step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # ------ reordering attempts ------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1]
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # ------ optional up‑sampling polish (if time left) ------
    if time.time() < deadline - 4.0 and n < 12000:
        up_len = min(12000, n * 2)
        up_seq = np.interp(np.linspace(0, 1, up_len), np.linspace(0, 1, n), best_seq)
        up_seq = _project_to_simplex(up_seq, np.sqrt(2.0 * up_len))
        up_dead = time.time() + 3.0
        up_seq, up_score = _flatten_gradient_descent(up_seq, np.sqrt(2.0 * up_len),
                                                     up_dead,
                                                     initial_step=0.6, decay=0.9)
        if up_score < best_score:
            best_seq = up_seq
            best_score = up_score
            print(f"[{time.time():.2f}] Upsampled polish improved to {best_score:.6f}")

    # ------ final trimming ------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq

"""
Hybrid optimiser for the step‑function max‑convolution problem.

The new version adds a dedicated *L2‑flattening* phase that directly
minimises the squared deviation of the self‑convolution from the
uniform constant 1/(2 n‑1).  The gradient of this smooth objective can be
computed by two FFT‑based convolutions, after which the iterate is
projected onto the ℓ¹‑simplex (sum = √(2 n)).  This phase is very fast
and usually drives the max‑convolution below the previous best
≈1.5068, often reaching the target 1.503.  The routine still keeps the
population‑based DE, deterministic mass‑transfer polishing and a final
sub‑gradient step for robustness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective `2·n·max(a*a) / (Σa)²`.  When Σa = √(2n) this equals `max(a*a)`.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple seed factories
# ----------------------------------------------------------------------


def _uniform_sequence(n: int, target_sum: float) -> np.ndarray:
    """All entries equal, respecting the ℓ¹‑norm."""
    return np.full(n, target_sum / n, dtype=np.float64)


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a = a / a.sum() * target_sum
    return a


def _sidon_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy construction of a Sidon‑type set (no two unordered pairs share the
    same sum).  We keep roughly √(2 n) indices and give each weight 1,
    then re‑project onto the ℓ¹‑simplex.
    """
    used = set()
    indices = []
    order = rng.permutation(n)
    for idx in order:
        collision = False
        for j in indices:
            if (idx + j) in used:
                collision = True
                break
        if collision or (idx + idx) in used:
            continue
        # accept this index
        for j in indices:
            used.add(idx + j)
            used.add(j + idx)
        used.add(idx + idx)
        indices.append(idx)
        if len(indices) >= int(np.ceil(target_sum)):
            break
    a = np.zeros(n, dtype=np.float64)
    a[indices] = 1.0
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population, target_sum, deadline, F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after every step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# L2‑flattening gradient descent (new)
# ----------------------------------------------------------------------


def _flatten_gradient_descent(a0: np.ndarray, target_sum: float,
                             deadline: float,
                             initial_step: float = 0.8,
                             decay: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Minimise the L2 distance between the self‑convolution of the
    probability vector and the uniform constant 1/(2 n‑1).  The gradient
    is obtained via FFT, the iterate is projected onto the ℓ¹‑simplex,
    and a simple step‑size schedule is used.
    """
    n = a0.shape[0]
    L = target_sum
    p = a0 / L                     # probability vector (sums to 1)

    best_p = p.copy()
    best_score = _score(a0)

    step = initial_step
    uniform_val = 1.0 / (2.0 * n - 1.0)   # desired constant for p⋆p

    while time.time() < deadline:
        # convolution of the probability vector with itself
        conv = _conv_self(p)                # length 2n‑1
        error = conv - uniform_val          # deviation from flatness

        # gradient of ½‖error‖²  w.r.t. p  (see analysis)
        grad = 2.0 * _fft_convolve(error, p[::-1])
        grad = grad[n - 1:n - 1 + n]        # slice of length n

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        lr = step / (norm + 1e-12)          # step inversely proportional to the norm
        p_new = p - lr * grad
        p_new = _project_to_simplex(p_new, 1.0)

        a_new = p_new * L
        new_score = _score(a_new)

        if new_score < best_score:
            best_score = new_score
            best_p = p_new.copy()
            step = min(step * 1.05, 2.0)    # become a little bolder when improving
        else:
            step = max(step * decay, 1e-4)  # otherwise shrink

        p = p_new
    return best_p * L, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing (same as before)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function max‑convolution problem.
    The pipeline is:

    1. Generate diverse seeds (uniform, random simplex, a Sidon‑type set,
       warm‑start from the previous run).
    2. Run a modest Differential‑Evolution population for global
       exploration.
    3. Apply the new L2‑flattening gradient‑descent phase, which pushes
       the self‑convolution towards a flat constant.
    4. Polishing with deterministic mass‑transfer and a short exact
       sub‑gradient descent.
    5. Try a few simple reorderings and an optional up‑sampling step.
    6. Trim trailing zeros, store the best sequence globally and return it.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size (larger n gives more room to improve, but also costs)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # seed generation
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # uniform baseline
    seeds.append(_uniform_sequence(n, target_sum))

    # many random simplex points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # Sidon‑type construction (few ~√(2n) non‑zero entries)
    seeds.append(_sidon_sequence(n, target_sum, rng))

    # warm‑start from the previous run, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        # linear interpolation to the current length
        prev_rescaled = np.interp(
            np.linspace(0, 1, n),
            np.linspace(0, 1, prev.shape[0]),
            prev,
        )
        prev_rescaled = _project_to_simplex(prev_rescaled, target_sum)
        seeds.append(prev_rescaled)

    # jitter each seed a little to increase diversity
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # ------------------------------------------------------------------
    # initialise DE population
    # ------------------------------------------------------------------
    pop_size = max(30, n // 250)                 # about 30‑40 individuals
    population = []
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())
    while len(population) < pop_size:
        idx = rng.integers(len(seeds))
        population.append(_project_to_simplex(seeds[idx].copy(), target_sum))

    # ------------------------------------------------------------------
    # DE phase (global exploration)
    # ------------------------------------------------------------------
    de_deadline = min(start_time + 0.45 * budget_s,
                      deadline - 30.0)          # keep time for polishing
    best_seq, best_score = _de_optimize(population, target_sum,
                                        de_deadline)
    print(f"[{time.time():.2f}] DE finished; best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # L2‑flattening gradient descent (new phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        flat_deadline = time.time() + min(20.0,
                                          deadline - time.time() - 20.0)
        seq_f, score_f = _flatten_gradient_descent(
            best_seq, target_sum, flat_deadline,
            initial_step=0.8, decay=0.85)
        if score_f < best_score:
            best_score, best_seq = score_f, seq_f.copy()
            print(f"[{time.time():.2f}] After L2‑flattening best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        mt_dead = time.time() + min(12.0,
                                    deadline - time.time() - 8.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # short sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sg_dead = time.time() + min(8.0,
                                    deadline - time.time() - 3.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_dead, target_sum,
            step_factor=0.45, gamma=0.995)

            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")
