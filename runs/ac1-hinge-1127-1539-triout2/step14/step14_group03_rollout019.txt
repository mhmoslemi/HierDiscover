#!/usr/bin/env python
"""
Propose a very low‑score non‑negative step sequence.

Method
------
1. **Deterministic grid search** – evaluate the two‑parameter family
   a_i ∝ (i+1)^(−p)·exp(−λ·i/(n−1)) for a dense grid of
   (p,λ).  The sum is forced to √(2·n) so that the benchmark score
   equals the maximal self‑convolution value.  The best pair is kept.
2. **Local refinement** – a small random walk in the (p,λ) plane
   starting from the grid optimum improves the score further.
3. **Fast simulated‑annealing mass transfer** – starting from the
   refined vector a, a classic O(n) mass‑transfer annealer (the same
   routine used in the previous baseline) is run for the remaining
   time budget to polish the solution.
4. Return the final sequence as a plain Python list.  All steps respect
   the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (same as the original baseline)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Return the benchmark score 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    # The caller guarantees s>0.01, but we keep the guard.
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i/(n-1))."""
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam·i/(n-1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    if n == 1:
        exp_part = np.ones(1, dtype=np.float64)
    else:
        exp_part = np.exp(-lam * np.arange(n, dtype=np.float64) / (n - 1))
    return idx ** (-p) * exp_part


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer refinement (identical to baseline)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Aggressive simulated‑annealing that moves tiny amounts of mass."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact self‑convolution
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                     # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) exact update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # problem size – larger n → finer granularity, but FFT costs grow slowly
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)                # the usual normalisation

    best_seq = None
    best_score = np.inf
    best_p = None
    best_lam = None

    # ------------------------------------------------------------------
    # Warm start (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        val = _score(warm)
        if val < best_score:
            best_score = val
            best_seq = warm.copy()
            print(f"[{time.time():.2f}] Warm start score = {val:.6f}")

    # ------------------------------------------------------------------
    # Grid search over the power‑law × exponential family
    # ------------------------------------------------------------------
    # coarse grid (covers the whole plausible region)
    p_grid = np.linspace(0.40, 0.60, 41)          # step ≈0.005
    lam_grid = np.linspace(0.6, 2.0, 71)         # step ≈0.02 (covers lnπ≈1.14)

    for p in p_grid:
        if time.time() >= deadline:
            break
        for lam in lam_grid:
            if time.time() >= deadline:
                break
            a = _shape_powerexp(n, p, lam)
            scale = target_sum / a.sum()
            a = a * scale
            val = _score(a)
            if val < best_score:
                best_score = val
                best_seq = a.copy()
                best_p, best_lam = p, lam
                print(f"[{time.time():.2f}] Grid p={p:.5f} lam={lam:.5f} score={val:.6f}")

    # ------------------------------------------------------------------
    # Fine‑grained local search around the best parameters
    # ------------------------------------------------------------------
    if best_p is not None and best_lam is not None:
        # three refinement rounds, each narrowing the window
        for radius in [0.02, 0.008, 0.003]:
            p_grid = np.linspace(max(0.1, best_p - radius),
                                 min(0.9, best_p + radius), 31)
            lam_grid = np.linspace(max(0.0, best_lam - radius * 5),
                                   best_lam + radius * 5, 31)
            for p in p_grid:
                if time.time() >= deadline:
                    break
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    a = _shape_powerexp(n, p, lam)
                    a = a * (target_sum / a.sum())
                    val = _score(a)
                    if val < best_score:
                        best_score = val
                        best_seq = a.copy()
                        best_p, best_lam = p, lam
                        print(f"[{time.time():.2f}] Refine p={p:.5f} lam={lam:.5f} score={val:.6f}")

    # ------------------------------------------------------------------
    # Small random walk on (p,λ) to escape any remaining local minima
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(2000):
        if time.time() >= deadline:
            break
        p = best_p + rng.normal(scale=0.003)
        lam = best_lam + rng.normal(scale=0.01)
        if not (0.1 <= p <= 0.9 and lam >= 0.0):
            continue
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            best_p, best_lam = p, lam
            print(f"[{time.time():.2f}] Random walk p={p:.5f} lam={lam:.5f} score={val:.6f}")

    # ------------------------------------------------------------------
    # Aggressive annealing refinement (uses remaining time budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] Anneal final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # drop trailing near‑zero entries – they do not affect the score
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final verification through the provided evaluate_sequence
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
