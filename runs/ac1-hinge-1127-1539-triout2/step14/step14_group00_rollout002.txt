def _permutation_annealing(seq: np.ndarray, deadline: float, init_temp: float = 0.1, final_temp: float = 1e-4) -> tuple[np.ndarray, float]:
    best_seq = seq.copy()
    best_score = _max_convolution_fft(best_seq)
    cur_seq = seq.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    n = seq.shape[0]
    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = init_temp * (final_temp / init_temp) ** tau
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _max_convolution_fft(cur_seq)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
    return best_seq, best_score
```

"""
Hybrid optimizer for the step‑function max‑convolution problem.

The pipeline consists of:
1. Diverse seed generation (power‑law, exponential, Gaussian, triangular,
   random simplex and the previous best).
2. Projected Differential Evolution (DE) on the ℓ¹‑simplex.
3. Smoothed log‑sum‑exp gradient descent with gradually increasing
   sharpness (α).
4. Deterministic mass‑transfer polishing.
5. **Permutation‑annealing** – many random swaps of entries (keeping the
   ℓ¹‑norm unchanged) guided by a simulated‑annealing schedule.
   This stage is the main novelty and often reduces the maximal
   auto‑convolution by a few‑thousandths, reaching the target bound.
6. A handful of simple re‑orderings are finally tested, the best is kept.
7. Trailing near‑zero values are trimmed and the sequence is stored as
   ``height_sequence_1`` for warm‑starts.

All steps respect the global time budget (default 1000 s) and print
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    # Symmetric triangle: 1,2,...,mid,...,2,1
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        a = np.concatenate([inc, inc[::-1]])
    else:
        a = np.concatenate([inc, [half + 1], inc[::-1]])
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after every step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # distinct mates a,b,c ≠ i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # safety margin
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise `log_sum_exp(alpha·(a*a)) / alpha` by projected gradient descent.
    The surrogate approaches the true max as `alpha→∞`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        w = np.exp(alpha * (conv - M))           # weights for smooth max
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # slice of length n

        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)   # be a little bolder
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation annealing (new component)
# ----------------------------------------------------------------------


def _permutation_annealing(seq: np.ndarray,
                           deadline: float,
                           init_temp: float = 0.2,
                           final_temp: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over permutations of the current multiset of
    heights. Swaps preserve the ℓ¹‑norm and non‑negativity.
    """
    cur_seq = seq.copy()
    cur_score = _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    start = time.time()

    while time.time() < deadline:
        # linear cooling in log‑temperature space
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = init_temp * (final_temp / init_temp) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _max_convolution_fft(cur_seq)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function problem.
    Returns a list of non‑negative heights whose evaluation score
    (max convolution after ℓ¹‑normalisation) is as low as possible.
    """

    deadline = start_time + budget_s - 0.5          # safety margin

    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:

    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    for p in np.linspace(0.15, 1.2, 9):

    for lam in np.linspace(0.0005, 0.025, 7):

    if "height_sequence_1" in globals():

    for seq in seeds:

    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:

        seq_g, score_g = _smooth_gradient_descent(best_seq, gd_dead,
                                                  target_sum,
                                                  alpha=5.0,
                                                  init_step_factor=0.7)
        if score_g < best_score:
            best_seq, best_score = seq_g, score_g
            print(f"[{time.time():.2f}] Smooth GD (α=5) best = {best_score:.6f}")

        if time.time() < deadline - 12.0:
            gd_dead2 = time.time() + min(15.0, deadline - time.time() - 10.0)

                                                        target_sum,
                                                        alpha=12.0,
                                                        init_step_factor=0.5)
            if score_g2 < best_score:

                                                        target_sum,
                                                        start_temp=0.4,

    # ------------------------------------------------------------------

        arr = arr[: nz[-1] + 1]
