"""
Hybrid optimiser for the step‑function max‑convolution problem with
a dedicated permutation‑search phase.

The pipeline:
  1. Generate a diverse set of seeds (power‑law, exponential,
     random simplex, previous best) and project them onto the ℓ¹‑simplex.
  2. Run a modest projected Differential Evolution (DE) to obtain a
     good continuous base sequence.
  3. Refine the sequence with a few passes of smoothed gradient descent
     on the log‑sum‑exp surrogate (increasing α).
  4. **Permutation search** – keep the multiset of values but
     explore many orderings (random shuffles, random‑step permutations,
     anti‑pair ordering) and a hill‑climbing swap phase.
  5. A short final gradient‑descent polish on the best ordering.
  6. Try a few deterministic re‑orderings and keep the best.
  7. Trim trailing near‑zero entries, store the result globally and
     return it as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective `2·n·max(a*a) / (Σa)²`.  When Σa = √(2n) this equals
    `max(a*a)`.  The routine is robust to near‑zero sums.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _step_permutation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random step permutation (requires step coprime to n)."""
    n = seq.shape[0]
    step = rng.integers(1, n)
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(seq):
        out[(i * step) % n] = v
    return out


def _anti_pair_order(seq: np.ndarray) -> np.ndarray:
    """
    Pair a large element with a small one at the mirrored index.
    Positions i and n‑1‑i receive consecutive entries from a descending
    sorted list, ensuring that the central convolution multiplies
    large with small values.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]   # descending
    out = np.empty_like(seq)
    left, right = 0, n - 1
    idx = 0
    while left < right:
        out[left] = sorted_vals[idx]
        out[right] = sorted_vals[idx + 1]
        idx += 2
        left += 1
        right -= 1
    if left == right and idx < n:
        out[left] = sorted_vals[idx]
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after every step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # distinct mates a,b,c ≠ i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # safety margin
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise `log_sum_exp(alpha·(a*a)) / alpha` by projected gradient descent.
    The surrogate approaches the true max as `alpha→∞`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2·n‑1
        M = conv.max()
        # weights for the smooth max (never overflow because (conv-M) ≤ 0)
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3·n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # slice of length n

        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new

    return best_a, best_score

# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

def _random_permutations_search(values: np.ndarray,
                                time_limit: float,
                                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Explore many orderings of `values` (which already sum to target_sum)
    and keep the arrangement with the smallest max‑convolution.
    """
    best_seq = values.copy()
    best_score = _max_convolution_fft(best_seq)
    n = values.shape[0]
    start = time.time()
    while time.time() - start < time_limit:
        # 50 % of candidates are plain shuffles, 50 % are random‑step permutations
        if rng.random() < 0.5:
            perm = rng.permutation(n)
            cand = values[perm]
        else:
            cand = _step_permutation(values, rng)
        sc = _max_convolution_fft(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    return best_seq, best_score


def _hill_climbing_swaps(seq: np.ndarray,
                         time_limit: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple pair‑swap hill climbing on a fixed multiset.
    Accepts a swap only if it strictly improves the max‑convolution.
    """
    best_seq = seq.copy()
    best_score = _max_convolution_fft(best_seq)
    n = seq.shape[0]
    start = time.time()
    while time.time() - start < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = best_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]
        sc = _max_convolution_fft(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser that mixes DE, smoothed gradient descent,
    a dedicated permutation search, and final polishing.
    Returns a list of non‑negative heights (floats) ready for evaluation.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # problem size (larger n gives more flexibility but slower ops)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # seed generation
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.15, 1.2, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # anti‑pair ordering of a power‑law seed (often good for the central lag)
    anti = _anti_pair_order(_power_law_sequence(n, 0.6, target_sum))
    seeds.append(anti)

    # previous best, if any (warm‑start)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed to increase diversity
    jittered: list[np.ndarray] = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # --------------------------------------------------------------
    # initialise DE population
    # --------------------------------------------------------------
    pop_size = max(30, n // 200)          # around 30‑40 individuals
    population: list[np.ndarray] = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # --------------------------------------------------------------
    # DE phase (≈ 45 % of the total budget)
    # --------------------------------------------------------------
    de_deadline = min(start_time + 0.45 * budget_s,
                      deadline - 30.0)          # leave time for later phases
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Smoothed gradient descent passes
    # --------------------------------------------------------------
    if time.time() < deadline - 25.0:
        # first pass, moderate α
        gd_dead = time.time() + min(20.0, deadline - time.time() - 15.0)
        seq_gd, sc_gd = _smooth_gradient_descent(best_seq, gd_dead,
                                                 target_sum,
                                                 alpha=4.0,
                                                 init_step_factor=0.7)
        if sc_gd < best_score:
            best_score, best_seq = sc_gd, seq_gd.copy()
            print(f"[{time.time():.2f}] After smooth α=4 best = {best_score:.6f}")

        # second pass, sharper α
        if time.time() < deadline - 12.0:
            gd_dead2 = time.time() + min(12.0,
                                         deadline - time.time() - 8.0)
            seq_gd2, sc_gd2 = _smooth_gradient_descent(best_seq, gd_dead2,
                                                       target_sum,
                                                       alpha=12.0,
                                                       init_step_factor=0.5)
            if sc_gd2 < best_score:
                best_score, best_seq = sc_gd2, seq_gd2.copy()
                print(f"[{time.time():.2f}] After smooth α=12 best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer polishing (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 5.0)
        seq_mt, sc_mt = _deterministic_mass_transfer(best_seq,
                                                     mt_dead,
                                                     target_sum,
                                                     start_temp=0.4,
                                                     end_temp=1e-5,
                                                     low_frac=0.07)
        if sc_mt < best_score:
            best_score, best_seq = sc_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sg_dead = time.time() + min(6.0, deadline - time.time() - 2.0)
        seq_sg, sc_sg = _subgradient_descent(best_seq, sg_dead,
                                             target_sum,
                                             step_factor=0.45,
                                             gamma=0.995)
        if sc_sg < best_score:
            best_score, best_seq = sc_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Permutation search (core new ingredient)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # keep the multiset of values but explore orderings
        values = np.sort(best_seq)[::-1]   # descending
        perm_time = (deadline - time.time()) * 0.6
        # random permutations
        seq_perm, sc_perm = _random_permutations_search(values, perm_time, rng)
        # hill‑climbing swaps on the best found permutation
        remaining = deadline - time.time()
        if remaining > 1.0:
            seq_hc, sc_hc = _hill_climbing_swaps(seq_perm, remaining, rng)
        else:
            seq_hc, sc_hc = seq_perm, sc_perm

        if sc_hc < best_score:
            best_score, best_seq = sc_hc, seq_hc.copy()
            print(f"[{time.time():.2f}] After permutation search best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final gradient polish (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 3.0:
        final_gd_dead = time.time() + min(4.0, deadline - time.time() - 1.0)
        seq_final, sc_final = _smooth_gradient_descent(best_seq,
                                                        final_gd_dead,
                                                        target_sum,
                                                        alpha=8.0,
                                                        init_step_factor=0.6)
        if sc_final < best_score:
            best_score, best_seq = sc_final, seq_final.copy()
            print(f"[{time.time():.2f}] After final smooth best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Reordering attempts (deterministic variants)
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        _step_permutation(best_seq, rng),
        best_seq[::-1]
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After deterministic reorder best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sampling & quick polish if time permits
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0 and n < 12000:
        up_len = min(12000, n * 2)
        up_seq = _resize_sequence(np.array(best_seq), up_len)
        up_seq = _project_to_simplex(up_seq, np.sqrt(2.0 * up_len))
        up_dead = time.time() + 3.0
        up_seq, up_score = _smooth_gradient_descent(up_seq,
                                                    up_dead,
                                                    np.sqrt(2.0 * up_len),
                                                    alpha=6.0,
                                                    init_step_factor=0.5)
        if up_score < best_score:
            best_seq = up_seq
            best_score = up_score
            print(f"[{time.time():.2f}] Upsampled polish improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # final trimming of trailing near‑zero entries
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # store globally for warm‑start and report final evaluation
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
