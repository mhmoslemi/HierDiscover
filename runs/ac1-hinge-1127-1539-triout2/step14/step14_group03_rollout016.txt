"""
Propose a low‑score non‑negative step sequence.

Algorithm
---------
1. **Parameter sweep of simple analytic families** – power‑law,
   exponential, combined power‑exponential, symmetric Beta and a sine‑window.
   Each family is sampled densely (hundreds of candidates).  The best
   candidate is kept as a warm‑start.

2. **Multi‑peak projected sub‑gradient descent** – instead of following a
   single active convolution entry we take the (soft) sum of the gradients
   of the K largest entries (K≈3).  After every step the vector is projected
   onto the simplex Σa = √(2·n) (the exact normalisation used by
   ``evaluate_sequence``).  This gives a more informative descent direction
   and escapes the local minima that the single‑peak sub‑gradient got stuck
   in.

3. **Mass‑transfer simulated annealing** – the classic rank‑one update used
   previously, but with a larger temperature at the start and a
   square‑root cooling schedule.  Tiny amounts of mass are moved from the
   indices that contribute most to the current maximal convolution entry to
   indices that are currently small.

The three phases share the total time budget (default 1000 s).  The final
sequence is clipped to the supported range by ``evaluate_sequence`` and stored
in ``height_sequence_1`` for warm‑starts in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT sized to the next power of two."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (scale‑invariant objective)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simple geometry helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Analytic shape families
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n‑1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam * i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    return (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (alpha - 1.0)
    )


def _shape_sine_power(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·i/(n‑1))^power – zero at the ends."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.sin(np.pi * x) ** power


# ----------------------------------------------------------------------
# Phase 1 – dense sampling of analytic families
# ----------------------------------------------------------------------


def _initial_sampling(n: int, target_sum: float, deadline: float):
    """Return the best sequence found among a large random/regular sweep of families."""
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # include warm start if present
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        val = _max_conv(warm)
        best_seq = warm.copy()
        best_score = val
        print(f"[{time.time():.2f}] warm‑start score={val:.6f}")

    # power‑law sweep
    for p in np.linspace(0.30, 1.00, 15):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _max_conv(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] power‑law p={p:.3f} -> {val:.6f}")

    # exponential sweep
    for lam in np.linspace(0.1, 6.0, 15):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _max_conv(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] exponential λ={lam:.3f} -> {val:.6f}")

    # combined power‑exp sweep (coarse random)
    for _ in range(80):
        p = rng.random() * 0.9 + 0.05      # 0.05 … 0.95
        lam = rng.random() * 6.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _max_conv(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] powerexp p={p:.3f}, λ={lam:.3f} -> {val:.6f}")

    # Beta sweep
    for alpha in np.linspace(0.20, 1.50, 13):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        val = _max_conv(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] beta α={alpha:.3f} -> {val:.6f}")

    # Sine‑window sweep
    for power in np.linspace(0.5, 4.0, 13):
        a = _shape_sine_power(n, power)
        a = a * (target_sum / a.sum())
        val = _max_conv(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] sine‑power γ={power:.3f} -> {val:.6f}")

    # a few random Dirichlet vectors
    for _ in range(12):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _max_conv(rnd)
        if val < best_score:
            best_score, best_seq = val, rnd.copy()
            print(f"[{time.time():.2f}] random Dirichlet -> {val:.6f}")

    # keep time aware – stop early if we already run out of budget
    if time.time() > deadline:
        print("[early exit] sampling phase exceeded time budget")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – multi‑peak projected sub‑gradient descent
# ----------------------------------------------------------------------


def _multi_peak_subgradient(
    a0: np.ndarray, target_sum: float, deadline: float, topk: int = 3
):
    """Descend using the sum of gradients of the top‑k convolution entries."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_conv(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        # indices of the K largest convolution entries
        top_idx = np.argpartition(b, -topk)[-topk:]

        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += a[k - idx]  # derivative part a_{k-i}
        grad = 2.0 * grad  # each term appears twice (i ↔ k−i)

        # simple linearly decreasing step size
        elapsed = time.time() - start
        total = deadline - start
        eta = 0.5 * (1.0 - elapsed / total) + 1e-4

        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_conv(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] sub‑grad it={it:6d} best={best_val:.6f}"
            )

        # occasional jitter to avoid stagnation
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Same scheme as the previous implementation, but with a slightly hotter start."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _max_conv(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak location
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.06, target_sum * 0.008)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of convolution (exact)
        b_new = b.copy()
        # linear contribution
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic pieces
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal new best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 3500 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _max_conv(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # problem size – pick a size that balances quality and speed
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)  # normalisation used by evaluate_sequence

    # ------------------------------------------------------------------
    # Phase 1 – dense analytic sweep
    # ------------------------------------------------------------------
    sweep_deadline = time.time() + 0.20 * (deadline - time.time())
    seq, best_val = _initial_sampling(n, target_sum, sweep_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.35 * (deadline - time.time())
        seq, val = _multi_peak_subgradient(seq, target_sum, sub_deadline, topk=3)
        if val < best_val:
            best_val, seq = val, seq.copy()
            print(f"[{time.time():.2f}] after sub‑grad best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _anneal_mass_transfer(seq, target_sum, deadline)
        if val < best_val:
            best_val, seq = val, seq.copy()
            print(f"[{time.time():.2f}] after anneal best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    arr = np.array(seq, dtype=np.float64)
    # drop trailing near‑zero entries (evaluate_sequence would do it anyway)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final sanity check using the provided evaluate_sequence
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] evaluation error: {exc}")

    # store for warm‑start in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
