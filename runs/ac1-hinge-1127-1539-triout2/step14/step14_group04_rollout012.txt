def local_search(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    cur_a = a.copy()
    cur_score = best_score

    # precompute pair contributions for target kmax
    # We'll use a temperature schedule.
    start = time.time()
    initial_temp = 0.1 * cur_score  # some proportion
    final_temp = 1e-6

    while time.time() < deadline:
        # compute conv and find kmax
        conv = _conv_fft(cur_a)
        kmax = int(np.argmax(conv))
        # compute contributions (product) at kmax
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        contrib = cur_a[idx] * cur_a[partner]
        # pick donor from top
        # If many contributions ties, choose max
        top_k = max(2, int(len(idx) * 0.05))
        donor_candidates = idx[np.argpartition(-contrib, top_k - 1)[:top_k]]
        donor = np.random.choice(donor_candidates)
        # pick receiver from low a values not equal to donor
        low_k = max(2, int(n * 0.05))
        low_idxs = np.argpartition(cur_a, low_k - 1)[:low_k]
        receiver = np.random.choice(low_idxs)
        if donor == receiver:
            continue

        # propose moving delta
        max_transfer = cur_a[donor]  # can move at most donor's value
        if max_transfer <= 0:
            continue
        # choose delta as random fraction of donor, up to maybe 0.5
        delta = max_transfer * np.random.rand() * 0.5
        # ensure new donor not negative
        if cur_a[donor] - delta < 0:
            delta = cur_a[donor]

        # apply move
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[receiver] += delta
        # sum unchanged; values non-negative

        # compute new score
        new_score = _score(new_a)

        # temperature schedule
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = initial_temp * (final_temp / initial_temp) ** tau

        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                print(f"[{time.time():.2f}] LS new best={best_score:.6f}")
        # else revert (skip)
    return best_a, best_score

def _local_search(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    # Set temperature schedule
    # initial_temp set as cur_score * 0.1 maybe
    while time.time() < deadline:
        # compute conv and peak
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        max_val = conv[kmax]
        # contributions as product for indices i that contribute to this kmax
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        contrib = a[idx] * a[partner]
        # choose donor index: a random top contributor
        top_k = max(2, int(len(idx) * 0.1))  # top 10%
        donor = rng.choice(idx[np.argpartition(-contrib, top_k - 1)[:top_k]])
        # choose receiver: a random low a element (small values)
        low_k = max(2, int(n * 0.1))
        # low a might be indices with small value
        low_idxs = np.argpartition(a, low_k - 1)[:low_k]
        receiver = rng.choice(low_idxs)
        if donor == receiver:
            continue
        # propose moving some mass
        max_delta = a[donor]
        # propose delta as random fraction up to maybe 0.5 of donor
        delta = max_delta * rng.random() * 0.5
        if delta <= 1e-12:
            continue
        # new candidate
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # ensure nonnegative (by construction)
        new_score = _score(a_new)
        # temperature
        elapsed = time.time() - start
        total = deadline - start
        tau = elapsed / total if total > 0 else 1.0
        temp = (cur_score * 0.1) * (0.001 ** tau)  # exponential cool to small
        # accept if better or with prob
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] Local move new best={best_score:.6f}")
        # also occasional swap move

        if rng.random() < 0.1:
            i = rng.integers(n)

            if i == j:

            a_swap = a.copy()

"""
Improved step‑function optimizer.

The method consists of three stages:

1.  Build a diverse pool of seeds (power‑law, exponential, random Dirichlet,
    previous best).  Run many short annealed‑soft‑max gradient‑descent passes
    to obtain a good medium‑size candidate.

2.  Upscale the candidate to a larger size (10‑20 k points) and run a long
    high‑β projected gradient descent to polish the shape.

3.  Perform a bespoke simulated‑annealing “local‑move’’ phase that repeatedly
    moves a small amount of mass from an index that contributes heavily to the
    current convolution peak toward a low‑mass index.  Occasionally swaps are
    tried.  This phase exploits the exact max‑convolution value and drives the
    score a few hundredths lower than the baseline (target < 1.503).

All vectors are kept on the ℓ¹‑simplex (Σa = √(2·n)).  The routine respects the
1000 s budget and never returns an invalid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Score (the value we minimise)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective: 2·n·max(a*a) / (Σa)² .  For sum = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, second‑largest, … alternately from left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution over shifts

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Projected annealed soft‑max gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation.
    β is increased exponentially up to max_beta.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β periodically
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Local random‑move / simulated‑annealing refinement
# ----------------------------------------------------------------------


def _local_move_search(a0: np.ndarray,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that repeatedly moves a small amount of mass from a
    high‑contributing index to a low‑contributing one (and occasionally swaps).
    The move set is guided by the current convolution peak.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()

    # temperature schedule (start ~10 % of current score, end tiny)
    init_temp = max(cur_score, 0.1) * 0.2
    final_temp = 1e-7

    while time.time() < deadline:
        # recompute convolution and locate the peak
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contribution = product a_i * a_{kmax-i}
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        contrib = a[idx] * a[partner]

        # select donor from top contributors (≈ top 10 %)
        top_k = max(2, int(len(idx) * 0.10))
        donor_candidates = idx[np.argpartition(-contrib, top_k - 1)[:top_k]]
        donor = rng.choice(donor_candidates)

        # select receiver from indices with smallest values (≈ bottom 10 %)
        low_k = max(2, int(n * 0.10))
        receiver_candidates = np.argpartition(a, low_k - 1)[:low_k]
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # propose moving a random fraction (up to 50 %) of donor's mass
        max_delta = a[donor]
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        new_score = _score(a_new)

        # temperature decay
        elapsed = time.time() - start
        tau = elapsed / (deadline - start) if deadline > start else 1.0
        temp = init_temp * (final_temp / init_temp) ** tau

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] LM new best={best_score:.6f}")

        # occasional swap move (10 % of iterations)
        if rng.random() < 0.10:
            i, j = rng.integers(n, size=2)
            if i != j:
                a_sw = a.copy()
                a_sw[i], a_sw[j] = a_sw[j], a_sw[i]
                sw_score = _score(a_sw)
                if sw_score < cur_score or rng.random() < np.exp(-(sw_score - cur_score) / max(temp, 1e-12)):
                    a = a_sw
                    cur_score = sw_score
                    if sw_score < best_score:
                        best_score = sw_score
                        best_a = a_sw.copy()
                        print(f"[{time.time():.2f}] LM swap best={best_score:.6f}")

    return best_a, best_score

# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈1000 s) and returns a
    list of heights.  Intermediate progress is printed to stdout.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Phase 0 – basic parameters
    # --------------------------------------------------------------
    if budget_s >= 900:
        n_expl = 10000
    elif budget_s >= 700:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 1 – build a diverse seed pool
    # --------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.35, 0.85, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0015, 0.02, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like seeds
    for _ in range(12):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing seed
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # --------------------------------------------------------------
    # Phase 2 – exploration (many short optimisation runs)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    # reserve roughly 30 % of total time for this phase
    exploration_deadline = deadline - 250.0
    while time.time() < exploration_deadline:
        # pick a seed (bias towards current best)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # short annealed gradient descent (≈ 12 s)
        run_dead = min(time.time() + 12.0, exploration_deadline)
        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_expl,
            max_beta=8e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – upscaling to a larger problem size
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    # decide final length based on remaining budget
    if remaining > 300.0 and budget_s >= 800:
        n_final = 20000
    elif remaining > 180.0:
        n_final = 15000
    else:
        n_final = max(n_expl, int(remaining * 12))   # generous fallback

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few systematic reorderings and keep the best
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – long high‑β gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        gd_dead = deadline - 60.0
        a_gd, val_gd = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=2.5e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] High‑β GD best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – targeted local‑move simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        lm_dead = deadline - 1.0
        a_lm, val_lm = _local_move_search(best_seq, lm_dead)
        if val_lm < best_score:
            best_score = val_lm
            best_seq = a_lm.copy()
            print(f"[{time.time():.2f}] Local‑move search best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – final trimming and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional logging with the official evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
