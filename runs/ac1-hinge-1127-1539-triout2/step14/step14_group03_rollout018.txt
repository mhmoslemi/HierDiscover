def propose_candidate(budget_s: int = 1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose sequence length
    if budget_s > 800:
        n = 12000
    elif budget_s > 500:
        n = 8000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    # Helper for scaling and clipping
    def normalize_and_clip(seq):
        seq = seq.astype(np.float64)
        if seq.max() > 0:
            seq = seq * (target_sum / seq.sum())
        seq = np.clip(seq, 0, 1000)  # enforce limit
        return seq

    # scoring function using fast FFT (no clipping)
    def fast_score(seq):
        nlen = seq.shape[0]
        # compute convolution full using FFT
        size = 1 << ((2*nlen - 1).bit_length())
        fa = np.fft.rfft(seq, n=size)
        conv = np.fft.irfft(fa*fa, n=size)[:2*nlen-1]
        max_b = conv.max()
        s = seq.sum()
        if s < 0.01:
            return np.inf
        return float(2.0 * nlen * max_b / (s*s))
    
    # also function to get exact evaluate_sequence
    # since evaluate_sequence is available, we can use it.
```

"""
search for a low‑score non‑negative step sequence.

The algorithm is deliberately simple and deterministic:

1. **Random shape sampling** – generate a large pool of candidate
   sequences from a variety of parametric families (power‑law,
   power‑law·exponential, symmetric beta, “two‑ended’’ power‑law,
   pure Dirichlet, etc.).  Each candidate is scaled to the usual
   normalisation  Σa = √(2·n)  and clipped to the hard bound 1000.
   The best candidate from this cheap global search is kept as a warm‑start.

2. **Greedy simulated‑annealing** – from the current best sequence we
   repeatedly move a small amount of mass from a “donor’’ index (chosen
   proportionally to the current height) to a “receiver’’ index
   (chosen inversely proportionally to the height).  The move size is
   controlled by a linear cooling schedule.  After every move the full
   autocorrelation is recomputed with an FFT (O(n log n)) and the new
   objective is evaluated.  Improvements are always accepted; worse
   moves are occasionally accepted according to a Metropolis rule.
   The process continues until the time budget expires.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_max(seq: np.ndarray) -> float:
    """Return the maximum entry of the linear self‑convolution seq*seq."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast version of the objective 2·n·max(conv) / (sum(seq)**2).
    No clipping or security checks – used only inside the optimiser.
    """
    n = seq.shape[0]
    max_b = _conv_max(seq)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape families (all return a non‑negative vector, not yet normalised)
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (i = 0,…,n‑1)."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p) * np.exp(-lam * np.arange(n, dtype=np.float64))


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the discrete grid."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


def _shape_two_ended_powerlaw(n: int, p_left: float, p_right: float) -> np.ndarray:
    """Heavy tails on both ends: a_i ∝ (i+1)^(-p_left)·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    rev = np.arange(n, 0, dtype=np.float64)
    return i ** (-p_left) * rev ** (-p_right)


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # Problem size – bigger sizes give finer granularity but cost more.
    # We keep it moderate (≈10 k) to allow many iterations.
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)             # usual normalisation used in the paper

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1) Global random shape search
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # optionally warm‑start from the previous run
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if warm.shape[0] != n:
            # simple linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, warm.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            warm = np.interp(new_x, old_x, warm)
        # clip to the 0‑1000 box and keep as a candidate
        warm = np.clip(warm, 0.0, 1000.0)
        val = _fast_score(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[{time.time():.2f}] warm‑start score = {best_val:.6f}")

    shape_budget = 0.35 * (deadline - start)   # time for the cheap sweep
    while time.time() < start + shape_budget:
        shape_type = rng.choice(
            ["powerlaw", "powerexp", "beta", "twoended", "exponential", "random"]
        )
        if shape_type == "powerlaw":
            p = rng.uniform(0.3, 1.2)
            a = _shape_powerlaw(n, p)
        elif shape_type == "powerexp":
            p = rng.uniform(0.1, 1.0)
            lam = rng.uniform(0.0, 8.0)          # lam on the true index scale
            a = _shape_powerexp(n, p, lam)
        elif shape_type == "beta":
            alpha = rng.uniform(0.2, 1.0)
            a = _shape_beta(n, alpha)
        elif shape_type == "twoended":
            pL = rng.uniform(0.2, 1.0)
            pR = rng.uniform(0.2, 1.0)
            a = _shape_two_ended_powerlaw(n, pL, pR)
        elif shape_type == "exponential":
            lam = rng.exponential(scale=0.001) * 5.0   # typical decay ~0‑5
            a = _shape_exponential(n, lam)
        else:   # random Dirichlet‑like start
            concentration = rng.uniform(0.01, 0.5)
            a = rng.dirichlet([concentration] * n)

        # normalise to the target sum, clip to the hard bound
        a = a * (target_sum / a.sum())
        a = np.clip(a, 0.0, 1000.0)

        val = _fast_score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] new global best (shape) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Greedy simulated‑annealing (local refinement)
    # ------------------------------------------------------------------
    if best_seq is None:                     # should never happen, but be safe
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _fast_score(best_seq)

    cur_seq = best_seq.copy()
    cur_val = best_val

    # temperature schedule – start a fraction of the current energy
    T0 = cur_val * 0.6
    total = deadline - time.time()

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - frac))

        # donor – pick index proportional to its current mass
        if cur_seq.sum() == 0:
            break
        prob_donor = cur_seq / cur_seq.sum()
        donor = rng.choice(n, p=prob_donor)

        # receiver – prefer indices that are currently small
        inv = 1.0 / (cur_seq + 1e-12)
        prob_recv = inv / inv.sum()
        receiver = rng.choice(n, p=prob_recv)

        if donor == receiver:
            continue

        # maximal admissible transfer (keep within the 0‑1000 box)
        max_delta = min(cur_seq[donor] * 0.15, 1000.0 - cur_seq[receiver])
        if max_delta <= 0.0:
            continue

        # annealed step size – larger when hot, tiny when cold
        delta = rng.random() * max_delta * max(temperature / T0, 0.02)

        # propose new sequence (two‑coordinate update)
        new_seq = cur_seq.copy()
        new_seq[donor] -= delta
        new_seq[receiver] += delta

        # evaluate – fast version (no security checks)
        new_val = _fast_score(new_seq)

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                print(f"[{time.time():.2f}] improved best = {best_val:.6f}")

        # occasional full re‑projection to avoid drift
        if it % 3000 == 0:
            cur_seq = _project_to_simplex(cur_seq, target_sum)
            cur_seq = np.clip(cur_seq, 0.0, 1000.0)
            cur_val = _fast_score(cur_seq)
            if cur_val < best_val:
                best_val, best_seq = cur_val, cur_seq.copy()
                print(f"[{time.time():.2f}] recompute improved best = {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final clean‑up and exact evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # trim trailing tiny values (they are irrelevant for the final answer)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final check using the provided security‑heavy evaluator
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] evaluation error: {exc}")

    # store for a possible warm‑start in a later invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
