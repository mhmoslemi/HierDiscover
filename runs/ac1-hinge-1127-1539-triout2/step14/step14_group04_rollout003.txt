"""
Propose a low‑peak step‑function.

Algorithm outline
-----------------
1. **Seed generation** – create a diverse pool of length‑n₀ sequences
   (power‑law, exponential, random Dirichlet and the previous best).
2. **Short local optimisation** – run a few seconds of projected sub‑gradient
   descent on each seed; keep the best‑scoring candidate.
3. **Upscaling** – linearly interpolate the best candidate to a larger length
   n₁ (≈15 k when enough time is left) and project onto the ℓ¹‑simplex
   (∑a = √(2·n₁)).
4. **Permutation annealing** – keep the mass values fixed and repeatedly
   swap two positions, accepting moves with a Metropolis rule that cools
   from a high temperature down to near‑zero.  The objective is the
   exact peak of the convolution a∗a (no smooth approximation).
5. **Fine‑grained refinement** – a short high‑β soft‑max gradient descent
   followed by a final sub‑gradient pass and a deterministic mass‑transfer
   (simulated‑annealing) polishing stage.
6. **Return** – trim trailing zeros, store the result globally and output it.

All steps respect the supplied time budget (≈1000 s).  The code uses only
NumPy and the supplied `evaluate_sequence` for final logging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value (exact peak of a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # the normalisation factor in evaluate_sequence is 2·n/(s²); we keep it.
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹‑simplex)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for seeding
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, second‑largest, … placed alternately from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Annealed gradient descent (projected)
# ----------------------------------------------------------------------
def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β exponentially
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true peak
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient for the active shift
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]

        # the diagonal term when i=j (appears once)
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small local moves)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of current peak
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        # choose a low contributor among the bottom low_frac of the window
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing
# ----------------------------------------------------------------------
def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈1000 s) and returns a
    list of heights.  Intermediate progress is printed to stdout.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – parameters
    # ------------------------------------------------------------------
    # A modest length for the quick exploration phase (fast sub‑grad runs)
    n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – build a diverse pool of seeds
    # ------------------------------------------------------------------
    init_pool = []

    # Power‑law family
    for p in np.linspace(0.35, 0.85, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # Exponential family
    for lam in np.linspace(0.0015, 0.02, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # Random Dirichlet‑like seeds
    for _ in range(8):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # Linear decreasing seed
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # Previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            init_pool.append(_resize_sequence(prev, n_expl))

    # Jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Phase 2 – quick local optimisation on the exploration pool
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    exploration_deadline = start_time + 0.2 * budget_s   # ~20 % for this phase
    while time.time() < exploration_deadline:
        # Pick a seed (occasionally jitter the current best)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # Run a short sub‑gradient descent (≈5 s)
        run_dead = min(time.time() + 5.0, exploration_deadline)
        a_opt, val_opt = _subgradient_descent(
            seed,
            run_dead,
            target_expl,
            max_iter=200_000,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

        # occasional longer GD attempt (≈6 s)
        if time.time() < exploration_deadline and rng.random() < 0.4:
            run_dead = min(time.time() + 6.0, exploration_deadline)
            a_opt, val_opt = _annealed_gradient_descent(
                seed,
                run_dead,
                target_expl,
                max_beta=5e5,
                step_factor=0.7,
                gamma=0.99,
            )
            if val_opt < best_score:
                best_score = val_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.2f}] Exploration (GD) new best={best_score:.6f}")

    # Fallback if nothing succeeded
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # ------------------------------------------------------------------
    # Phase 3 – moderate refinement of the best seed
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        refine_dead = time.time() + 20.0
        a_ref, val_ref = _annealed_gradient_descent(
            best_seq,
            refine_dead,
            target_expl,
            max_beta=1e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Moderate GD refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – upscaling to a larger problem size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 500.0 and budget_s >= 800:
        n_final = 18000
    elif remaining > 300.0:
        n_final = 14000
    else:
        n_final = max(n_expl, int(remaining * 12))   # very rough fallback

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Several deterministic reorderings – keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
        seq_big[::-1],                     # simple reversal
    ]
    best_seq = candidates[0]
    best_score = _score(best_seq)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – long permutation‑annealing (mass values fixed)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        perm_dead = deadline - 4.0
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.6,
            end_temp=1e-6,
            top_frac=0.07,
            low_frac=0.07,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – high‑β soft‑max GD polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        gd_dead = deadline - 2.0
        a_gd, val_gd = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=3e6,
            step_factor=0.45,
            gamma=0.995,
        )
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] High‑β GD polishing best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – short sub‑gradient pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        sub_dead = time.time() + min(3.0, deadline - time.time() - 1.0)
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
            max_iter=200_000,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Post‑GD sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 8 – final deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(3.0, deadline - time.time() - 0.5)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.06,
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT polishing best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 9 – trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # ------------------------------------------------------------------
    # (Optional) final evaluation for logging
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
