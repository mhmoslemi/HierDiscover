def subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)
    it=0
    while time.time() < deadline:
        b = _conv_self_fft(a)
        k_max = int(np.argmax(b))
        # gradient
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # g_i = 2*a_{k_max - i} for i in [lo, hi]
        g = np.zeros_like(a)
        idx = np.arange(lo, hi+1)
        g[idx] = 2 * a[k_max - idx]
        # compute step size
        norm = np.linalg.norm(g)
        if norm == 0:
            break
        step = (target_sum * 0.3) / (np.sqrt(it+1) * norm)
        a = a - step * g
        a = _project_to_simplex(a, target_sum)
        val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        it += 1
    return best_a, best_val
```

def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # Build candidates
    init_candidates = []

    # Warm start
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    # Exponent search: keep best
    best_exp_seq = None
    best_exp_score = np.inf
    # choose a range of exponents
    exponents = np.linspace(0.2, 1.2, 25)  # adjust
    for p in exponents:
        raw = np.arange(1, n+1, dtype=np.float64) ** (-p)
        seq = raw * (target_sum / raw.sum())
        # compute score quickly using self convolution
        b = _conv_self_fft(seq)
        score = _score_from_selfconv(b, n, target_sum)
        if score < best_exp_score:
            best_exp_score = score
            best_exp_seq = seq.copy()

    if best_exp_seq is not None:
        init_candidates.append(best_exp_seq)

    # Random Dirichlet / uniform candidates
    rng = np.random.default_rng()
    for _ in range(3):
        rnd = rng.random(n)
        seq = _project_to_simplex(rnd, target_sum)
        init_candidates.append(seq)

    best_seq = None
    best_score = np.inf

    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}] Starting candidate #{idx+1}")

        # allocate time fractions
        remaining = deadline - time.time()
        sub_time = remaining * 0.5   # 50% for gradient descent
        sub_deadline = time.time() + sub_time

        a_smooth, val_smooth = smooth_max_descent(init, target_sum, sub_deadline,
            beta_start=0.1, beta_max=80.0)

        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New best after smooth: {best_score:.6f}")

        # Subgradient descent (refine)
        if time.time() < deadline:
            remaining = deadline - time.time()
            sub2_deadline = time.time() + remaining * 0.2   # 20% for subgrad
            a_sub, val_sub = subgradient_descent(a_smooth, target_sum, sub2_deadline)
            if val_sub < best_score:
                best_score = val_sub
                best_seq = a_sub.copy()
                print(f"[{time.time():.2f}] New best after subgrad: {best_score:.6f}")

        # Annealing refinement on the best among smooth/subgrad
        if time.time() < deadline:
            a_anneal, val_anneal = anneal_mass_transfer(best_seq if best_seq is not None else a_smooth,
                                                       target_sum, deadline)
            if val_anneal < best_score:
                best_score = val_anneal
                best_seq = a_anneal.copy()
                print(f"[{time.time():.2f}] New best after anneal: {best_score:.6f}")

    # fallback
    if best_seq is None:
        best_seq = (np.arange(1, n+1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)
    # trim trailing near-zero
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1]+1]
    final_seq = arr.tolist()
    # final eval
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error: {e}")
    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Improved step‑function optimizer.

The procedure consists of four stages:

1. **Problem size selection** – a moderate length (8000‑18000) is chosen
   based on the time budget.

2. **Low‑dimensional launch pool** – a few deterministic starts
   (warm‑start, a power‑law family, a random Dirichlet point) are built.
   For the power‑law family an exhaustive sweep over the exponent
   (`a_i ∝ (i+1)^{-p}`) is performed; the best exponent is kept as a
   candidate.

3. **Smooth‑max projected gradient descent** – a soft‑max approximation
   of the max–convolution is minimised.  The temperature β is raised
   slowly up to a large value (β≈80) so the optimisation approaches the
   true non‑smooth objective.  After each step the iterate is projected
   onto the simplex `{a≥0, Σa = √(2n)}`.

4. **Refinement** – a short sub‑gradient descent on the exact max‑convolution
   followed by the aggressive O(n) simulated‑annealing mass‑transfer
   routine from the reference implementation.

All steps respect the global deadline (default 1000 s).  The best sequence
found is returned as a plain list of floats and cached in the global
variable `height_sequence_1` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact objective 2·n·max(b) / (target_sum²) – b must be a*a."""
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.1,
    beta_max: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution entry.
    The approximation is  (1/β)·log Σ exp(β·(a*a)_i).
    After each step the vector is projected onto the simplex Σa = target_sum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    step0 = target_sum * 0.8
    it = 0

    while time.time() < deadline:
        # self‑convolution and soft‑max weights
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # log‑sum‑exp (numerically stable)
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ a_rev)  (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)
        g = 2.0 * conv_wa[n - 1 : n - 1 + n]

        # step size (decaying with √it)
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # slowly increase β
        if it % 1500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.4)

        it += 1

    return best_a, best_val

def anneal_mass_transfer(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves tiny amounts of mass from
    indices contributing heavily to the current peak to indices that
    contain little mass.  The convolution is updated in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # location of the current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – weighted by contribution a[i]·a[k‑i]
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ---- exact O(n) update of the convolution ----
        b_new = b.copy()
        # linear part 2·a·Δ
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 3500 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the true max‑convolution value.
    The sub‑gradient at the peak index k* is g_i = 2·a_{k*‑i} (zero outside the
    support).  After each step we project back onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    it = 0
    step0 = target_sum * 0.6
    while time.time() < deadline:
        b = _conv_self_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient (only non‑zero on the support of the peak)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] subgrad improved best={best_val:.6f}")

        it += 1

    return best_a, best_val
