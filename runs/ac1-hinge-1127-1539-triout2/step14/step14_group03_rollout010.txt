    start = time.time()
    deadline = start + budget_s - 0.5

    # choose initial n maybe 12000 (mid) for quick conv and high resolution.
    n_initial = 12000  # less but still high.
    target_sum = np.sqrt(2*n_initial)

    best_a = None
    best_score = np.inf

    # if warm start exists, start from it (resize to n_initial maybe)

        warm = np.array(globals()['height_sequence_1'], dtype=np.float64)
        # resize to initial n
        warm = _resize_sequence(warm, n_initial)
        # normalize sum
        warm = warm * (target_sum / warm.sum())
        # compute score
        sc = _score(warm)
        best_a, best_score = warm, sc
        print(f"[{time.time():.2f}] Warm start initial score {sc:.6f}")

    # define a list of families functions and parameter ranges
    # We'll iterate random draws until time is up.

    rng = np.random.default_rng()

    while time.time() < deadline - 5:  # keep 5 secs for refinement

        family = rng.integers(0, 5)
        if family == 0:
            # power law
            p = rng.random() * 0.4 + 0.4  # [0.4,0.8]

        elif family == 1:
            lam = rng.random() * 6 + 0.5   # [0.5, 6.5]
            a = _shape_exponential(n_initial, lam)
        elif family == 2:
            p = rng.random() * 0.8 + 0.1   # [0.1,0.9]
            lam = rng.random() * 6 + 0.5   # [0.5,6.5]
            a = _shape_powerexp(n_initial, p, lam)
        elif family == 3:
            # beta
            alpha = rng.random() * 0.7 + 0.2  # [0.2,0.9]
            a = _shape_beta(n_initial, alpha)

    # Determine how many seconds left
    remaining = deadline - time.time()

        # use subgradient descent with momentum

        n = a.shape[0]
        target_sum = np.sqrt(2.0 * n)

"""
Improved step‑function search for the max‑convolution objective.

The algorithm proceeds in three time‑controlled phases:

1. **Warm‑start** – reuse the sequence from the previous call (if any) after
   resizing it to the current problem size.

2. **Systematic shape search** – explore a dense grid of the two‑parameter
   family   a_i ∝ (i+1)^{‑p}·exp(‑λ·i/(n‑1))  (the “power‑exp” family) as well
   as a few simple alternatives (pure power‑law, pure exponential and a
   symmetric Beta shape).  For each candidate the exact max‑convolution value
   is obtained via an FFT‑based convolution and the objective
   2 n·max(b)/(∑a)^2 is recorded.  The best candidate of this phase is kept
   for refinement.

3. **Refinement** – a projected momentum sub‑gradient descent on the true
   max‑convolution followed by the aggressive simulated‑annealing mass‑transfer
   routine from the reference implementation.  Both phases respect the
   remaining time budget and continually keep the best vector seen.

All helper functions are top‑level; no closures or lambdas are used.
The returned list is guaranteed to be non‑negative and will be stored in the
global variable ``height_sequence_1`` for warm‑starts in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution and scoring utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of a power‑of‑2 size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # smallest power of 2 ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)²).  No normalisation is performed.
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:                     # protects against division by zero
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Parametric families of non‑negative step sequences
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i/(n‑1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}·exp(‑lam·i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape on the discrete grid."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (alpha - 1.0)
    )
    return a


# ----------------------------------------------------------------------
# Phase 2 – Projected momentum sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_momentum_descent(a0: np.ndarray, target_sum: float,
                                 deadline: float):
    """
    Momentum based sub‑gradient descent on the current max‑convolution entry.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    # momentum buffer
    v = np.zeros_like(a)

    start = time.time()
    it = 0
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]                 # exact sub‑gradient

        # step size decays like 1/√t – a simple schedule that works well
        eta = 1.0 / (np.sqrt(it + 1) + 1.0)

        # Nesterov‑style momentum
        v = 0.9 * v - eta * grad
        a = a + v
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Momentum GD it={it:5d} "
                  f"score={best_val:.6f}")

        # occasional jitter to escape shallow plateaus
        if it % 2000 == 0:
            jitter = np.random.default_rng().normal(
                scale=target_sum * 2e-4, size=n
            )
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Simulated‑annealing mass‑transfer (identical to reference)
# ----------------------------------------------------------------------


def _anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Aggressive rank‑one mass‑transfer annealing.  Returns the best vector
    encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()
    rng = np.random.default_rng()

    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                              # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # favour small entries as receivers
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.07, target_sum * 0.009)
        if max_delta <= 0.0:
            it += 1
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            it += 1
            continue

        # exact O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # occasional full recompute to avoid drift
        if it % 3000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        best_seq = warm.copy()
        best_score = _score(warm)
        print(f"[{time.time():.2f}] Warm‑start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic grid / random shape search
    # ------------------------------------------------------------------
    phase1_deadline = start + 0.32 * (deadline - start)

    # ---- dense grid for the power‑exp family -------------------------
    p_grid = np.linspace(0.45, 0.78, 34)          # step ≈0.01
    lam_grid = np.linspace(0.5, 6.0, 28)         # step ≈0.2

    for p in p_grid:
        for lam in lam_grid:
            if time.time() >= phase1_deadline:
                break
            a = _shape_powerexp(n, p, lam)
            val = _score(a)
            if val < best_score:
                best_score, best_seq = val, a.copy()
                print(f"[{time.time():.2f}] PowerExp p={p:.4f} λ={lam:.3f} "
                      f"score={val:.6f}")

    # ---- pure power‑law ------------------------------------------------
    for p in np.linspace(0.45, 0.78, 34):
        if time.time() >= phase1_deadline:
            break
        a = _shape_powerlaw(n, p)
        val = _score(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] PowerLaw p={p:.4f} score={val:.6f}")

    # ---- pure exponential ---------------------------------------------
    for lam in np.linspace(0.5, 6.0, 28):
        if time.time() >= phase1_deadline:
            break
        a = _shape_exponential(n, lam)
        val = _score(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.3f} score={val:.6f}")

    # ---- symmetric Beta ------------------------------------------------
    for alpha in np.linspace(0.25, 0.90, 28):
        if time.time() >= phase1_deadline:
            break
        a = _shape_beta(n, alpha)
        val = _score(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.3f} score={val:.6f}")

    # ---- random Dirichlet starts (few) --------------------------------
    rng = np.random.default_rng()
    for _ in range(6):
        if time.time() >= phase1_deadline:
            break
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _score(rnd)
        if val < best_score:
            best_score, best_seq = val, rnd.copy()
            print(f"[{time.time():.2f}] Random start score={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – momentum sub‑gradient descent (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.35 * (deadline - time.time())
        a_sub, val_sub = _subgradient_momentum_descent(best_seq,
                                                       target_sum,
                                                       sub_deadline)
        if val_sub < best_score:
            best_score, best_seq = val_sub, a_sub.copy()
            print(f"[{time.time():.2f}] Post‑GD best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive annealing refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = _anneal_mass_transfer(best_seq,
                                               target_sum,
                                               deadline)
        if val_ann < best_score:
            best_score, best_seq = val_ann, a_ann.copy()
            print(f"[{time.time():.2f}] Post‑Anneal best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up & return
    # ------------------------------------------------------------------
    final_arr = np.clip(best_seq, 0.0, 1000.0)   # obey hard limits of evaluate_sequence
    final_seq = final_arr.tolist()

    # store for the next run
    globals()["height_sequence_1"] = final_seq

    # final verification (the competition harness will also call evaluate_sequence)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
