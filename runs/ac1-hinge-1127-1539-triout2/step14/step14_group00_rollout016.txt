def _mass_transfer_local(a, target_sum, deadline, rng, max_iters=None):
    n = a.shape[0]; L = 2*n-1
    conv = _conv_self(a)
    best_a = a.copy()
    best_conv = conv.copy()
    best_max = conv.max()
    it = 0
    while time.time() < deadline and (max_iters is None or it < max_iters):
        km = int(np.argmax(conv))
        lo = max(0, km - (n - 1))
        hi = min(km, n - 1)
        idxs = np.arange(lo, hi+1)
        contribs = a[idxs] * a[km - idxs]
        donor = idxs[np.argmax(contribs)]
        # choose receiver candidate:
        # find indices with low avalue to move mass to
        low_vals = np.percentile(a, 15)
        candidates = np.where(a <= low_vals)[0]
        if candidates.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = rng.choice(candidates)
        if donor == receiver:
            continue
        max_delta = a[donor] * 0.2  # restrict to 20% of donor mass maybe
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # update conv using original a before modification
        conv[donor:donor+n] -= 2 * delta * a
        conv[receiver:receiver+n] += 2 * delta * a
        # delta^2 terms
        conv[2*donor] += delta*delta
        conv[2*receiver] += delta*delta
        conv[donor + receiver] -= 2 * delta*delta
        # modify a
        a[donor] -= delta
        a[receiver] += delta
        # ensure nonnegative maybe adjust delta
        # compute new max
        cur_max = conv.max()
        # accept if improvement
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_conv = conv.copy()
        else:
            # revert move (optional)
            # For now revert to best state
            a = best_a.copy()
            conv = best_conv.copy()
            # continue
        it += 1
        # recompute conv occasionally
        if it % 500 == 0:
            conv = _conv_self(a)
            best_conv = conv.copy()
            best_max = conv.max()
    # after loop, return best_a and best_max
    return best_a, best_max

"""
Hybrid optimiser for the step‑function autocorrelation problem.

Main ideas
----------
1. Initialise a diverse pool of candidate sequences (power law,
   exponential, random simplex, equally‑spaced spikes and the previous
   best construction).  All candidates are projected onto the ℓ¹‑simplex
   Σa = √(2 n).

2. Successive smooth gradient‑descent phases with an increasing
   “soft‑max’’ temperature α.  The surrogate
        φ_α(a)=max(a*a)+log‑sum‑exp(α·(a*a‑max))/α
   is differentiable; we minimise it by projected gradient descent
   (the gradient is obtained via two FFT‑based convolutions).

3. A fast deterministic “mass‑transfer’’ local search that moves a
   small amount of mass from the index contributing most to the peak
   autocorrelation to a low‑mass index.  The convolution is updated
   in O(n) using the analytical rank‑one update, so millions of moves
   are cheap.

4. Optional up‑sampling: double the length, re‑optimise briefly and
   keep the better of the two lengths.

5. Re‑ordering (alternating, even‑spread, reversal) and trimming of
   trailing zeros.

The whole routine respects a global time budget (default 1000 s) and
stores the final sequence in the global variable `height_sequence_1`
so that later runs can warm‑start from it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s,  x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Re‑ordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate of max(a*a) using projected gradient descent.
    The surrogate becomes tighter as `alpha` grows.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        # weights for the smooth max
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # slice of length n

        grad = a + Tg / Z                        # gradient of the surrogate
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        lr = step_factor * target_sum / (norm + 1e-12)
        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _max_convolution_fft(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer local search
# ----------------------------------------------------------------------


def _mass_transfer_local(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = None) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing moves of a small amount of mass from a donor index
    (large contribution to the current peak) to a receiver index
    (low value).  Convolution is updated analytically in O(n) per move.
    Only improving moves are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    L = 2 * n - 1
    conv = _conv_self(a)           # current full autocorrelation
    best_a = a.copy()
    best_conv = conv.copy()
    best_max = conv.max()

    it = 0
    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break

        # current peak location
        kmax = int(np.argmax(conv))

        # indices that actually contribute to this peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a[i] * a[kmax-i]
        contrib = a[idxs] * a[kmax - idxs]
        donor = idxs[np.argmax(contrib)]

        # pick a low‑mass receiver (20‑percentile)
        low_thr = np.percentile(a, 20)
        low_cands = np.where(a <= low_thr)[0]
        if low_cands.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = rng.choice(low_cands)

        if donor == receiver:
            continue

        # size of the move – a modest fraction of the donor mass
        max_delta = a[donor] * 0.20
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # ----- update convolution (using the *old* a values) -----
        # subtraction of donor contribution
        conv[donor:donor + n] -= 2.0 * delta * a
        # addition of receiver contribution
        conv[receiver:receiver + n] += 2.0 * delta * a
        # delta² terms
        conv[2 * donor] += delta * delta
        conv[2 * receiver] += delta * delta
        conv[donor + receiver] -= 2.0 * delta * delta

        # ----- update the sequence -----
        a[donor] -= delta
        a[receiver] += delta

        # recompute the peak
        cur_max = conv.max()

        if cur_max < best_max:
            # improvement – keep it
            best_max = cur_max
            best_a = a.copy()
            best_conv = conv.copy()
        else:
            # revert – keep the best state seen so far
            a = best_a.copy()
            conv = best_conv.copy()
            # continue with next iteration
            it += 1
            continue

        # periodic re‑projection to cure numerical drift
        if it % 500 == 0:
            a = _project_to_simplex(a, target_sum)
            conv = _conv_self(a)

        it += 1

    # final projection (guard against tiny negative entries)
    a = _project_to_simplex(best_a, target_sum)
    final_max = _max_convolution_fft(a)
    return a, final_max

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser returning a list of non‑negative heights whose
    evaluation score (max autocorrelation after ℓ¹‑normalisation) is
    minimised.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size choice
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 700:
        n = 10000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # seed generation (diverse family of candidates)
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        seeds.append(prev)

    # power‑law family
    for p in np.linspace(0.2, 1.0, 5):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 5):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # equally spaced spikes (K ≈ √n spikes)
    K = max(4, int(np.sqrt(n)))
    spike_seq = np.zeros(n, dtype=np.float64)
    spike_idx = np.linspace(0, n - 1, K, dtype=int)
    spike_seq[spike_idx] = target_sum / K
    seeds.append(spike_seq)

    # jitter each seed a little (helps diversity)
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # ------------------------------------------------------------------
    # pick the best seed according to the true objective
    # ------------------------------------------------------------------
    best_seed = min(seeds, key=_max_convolution_fft)
    best_seq = best_seed.copy()
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] Initial best max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # smooth gradient‑descent phases with increasing α
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate roughly 60 % of remaining time to gradient descent
    gd_budget = remaining * 0.60
    gd_deadline = time.time() + gd_budget

    # schedule of α values (smoothness increases)
    alphas = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    cur_seq = best_seq
    cur_score = best_score
    for alpha in alphas:
        if time.time() >= gd_deadline:
            break
        # give each α a fraction of the remaining gd time
        sub_deadline = min(gd_deadline, time.time() + gd_budget * 0.12)
        seq_new, score_new = _smooth_gradient_descent(cur_seq,
                                                      sub_deadline,
                                                      target_sum,
                                                      alpha=alpha,
                                                      init_step_factor=0.6)
        if score_new < cur_score:
            cur_seq, cur_score = seq_new, score_new
            if cur_score < best_score:
                best_seq, best_score = cur_seq.copy(), cur_score
                print(f"[{time.time():.2f}] After α={alpha:.1f} → max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # deterministic mass‑transfer local search (hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        mt_deadline = deadline - 8.0
        seq_mt, score_mt = _mass_transfer_local(best_seq,
                                                target_sum,
                                                mt_deadline,
                                                rng,
                                                max_iters=20000)
        if score_mt < best_score:
            best_seq, best_score = seq_mt.copy(), score_mt
            print(f"[{time.time():.2f}] After mass‑transfer → max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # an optional short up‑sampling pass (if we still have time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0 and n < 18000:
        up_len = min(18000, n * 2)
        up_seq = _resize_sequence(best_seq, up_len)
        up_seq = _project_to_simplex(up_seq, np.sqrt(2.0 * up_len))
        # brief smoothing at higher resolution
        up_deadline = time.time() + 2.5
        up_seq, up_score = _smooth_gradient_descent(up_seq,
                                                    up_deadline,
                                                    np.sqrt(2.0 * up_len),
                                                    alpha=40.0,
                                                    init_step_factor=0.5)
        # keep the better of the two lengths
        if up_score < best_score:
            best_seq, best_score = up_seq, up_score
            print(f"[{time.time():.2f}] Upsampled improvement → max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # reordering attempts (they affect the peak because convolution is linear)
    # ------------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1]
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] After reordering → max‑conv = {best_score:.6f}")

    # trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
