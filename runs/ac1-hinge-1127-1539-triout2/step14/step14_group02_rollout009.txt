def smooth_max_descent_adam(a0, target_sum, deadline, beta_start=0.1, beta_max=30.0,
                            step0_factor=0.8, beta_increase_iter=2000, beta_mult=1.5):
    a = a0.copy()
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)
    beta = beta_start
    step0 = target_sum * step0_factor
    it = 1
    while time.time() < deadline:
        # compute self conv
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] adam β={beta:.2f} it={it} best={best_val:.6f}")

        # compute softmax weights
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)
        g = 2.0 * conv_wa[n-1:n-1+n]

        # Adam updates
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1 - 0.9**it)
        v_hat = v / (1 - 0.999**it)
        step = step0 / np.sqrt(it) * (target_sum / (np.linalg.norm(g) + 1e-12))
        # compute adaptive step
        update = step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - update
        a = _project_to_simplex(a, target_sum)

        # increase beta
        if it % beta_increase_iter == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mult)
        it += 1
    return best_a, best_val

"""
Hybrid optimizer for the step‑function benchmark.

The routine combines three stages:

1.  Adam‑accelerated projected gradient descent on a smooth‐max
    surrogate of max(a∗a).  The temperature β is ramped from a modest
    start (≈0.5) to a large final value (≈150) so the optimisation
    eventually targets the true L∞ norm.

2.  Aggressive simulated‑annealing “mass‑transfer” moves.  A donor
    index is chosen proportionally to its contribution to the current
    peak of the convolution; a receiver is favoured where the height
    is small.  The allowed transfer size grows early (high temperature)
    and shrinks later; occasional “big” jumps help escape shallow
    local minima.

3.  A short polishing pass of the smooth‑max Adam optimiser with a
    very high β and tiny steps to clean up any residual noise.

All updates are projected onto the simplex Σa = √(2n) (the normalisation
used by the benchmark).  The algorithm respects the supplied time
budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact benchmark score for a given self‑convolution b."""
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – Adam‑accelerated smooth‑max descent
# ----------------------------------------------------------------------


def smooth_max_descent_adam(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.5,
    beta_max: float = 150.0,
    step0_factor: float = 0.6,
    beta_increase_iter: int = 1500,
    beta_mult: float = 1.4,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a*a) using Adam and simplex
    projection.  The temperature β is increased gradually up to ``beta_max``.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    step0 = target_sum * step0_factor
    it = 1

    while time.time() < deadline:
        # self‑convolution & smooth‑max weights
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ADAM β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # soft‑max weights (stable)
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth objective
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)            # length 3n‑2
        g = 2.0 * conv_wa[n - 1 : n - 1 + n]      # central slice

        # Adam update (step size scaled by ‖g‖ and iteration count)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        norm_g = np.linalg.norm(g) + eps
        eta = step0 / np.sqrt(it) * (target_sum / norm_g)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

        # increase β periodically
        if it % beta_increase_iter == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – refined simulated annealing (mass transfer)
# ----------------------------------------------------------------------


def anneal_mass_transfer_refined(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    donor_frac: float = 0.08,
    delta_target_frac: float = 0.02,
    big_move_prob: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves mass from indices contributing heavily
    to the current peak toward low‑height indices.  The transfer size is
    governed by a temperature schedule; occasional big jumps improve
    exploration.
    """
    a = a0.copy()
    n = a.size
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # exponential cooling (fast early, gentle later)
        temperature = T0 * np.exp(-5.0 * frac)

        # locate current convolution peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – weighted by contribution a[i]·a[k‑i]
        contrib = a[idx] * a[k_max - idx]
        sum_contrib = contrib.sum()
        if sum_contrib > 0.0:
            donor = rng.choice(idx, p=contrib / sum_contrib)
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # decide transfer magnitude
        if rng.random() < big_move_prob:
            max_delta = min(a[donor] * 0.5, target_sum * 0.05)
        else:
            max_delta = min(a[donor] * donor_frac, target_sum * delta_target_frac)

        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of the convolution (exact)
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic self‑terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # occasional full recomputation to curb drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – final polishing (high‑β Adam)
# ----------------------------------------------------------------------


def polish_highbeta(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 80.0,
    beta_max: float = 300.0,
    step0_factor: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Very short Adam run with a large β and tiny steps to fine‑tune the
    final candidate.  Typically only a few hundred iterations fit
    before the overall budget expires.
    """
    # reuse the Adam implementation with tighter parameters
    return smooth_max_descent_adam(
        a0,
        target_sum,
        deadline,
        beta_start=beta_start,
        beta_max=beta_max,
        step0_factor=step0_factor,
        beta_increase_iter=500,
        beta_mult=1.1,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives finer control but costs more time.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)  # normalisation used by the benchmark

    # --------------------------------------------------------------
    # Build initial candidates (warm start + deterministic + random)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    # deterministic power‑law start (often a good baseline)
    power_exp = -0.78
    base = np.arange(1, n + 1, dtype=np.float64) ** power_exp
    base = base * (target_sum / base.sum())
    init_candidates.append(base)

    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run optimisation phases for each start, respecting the remaining budget.
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx + 1} -----")

        # ---- Phase 1 : Adam‑accelerated smooth‑max descent -----------------
        sub_deadline = time.time() + 0.30 * (deadline - time.time())
        a_smooth, val_smooth = smooth_max_descent_adam(
            init,
            target_sum,
            sub_deadline,
            beta_start=0.5,
            beta_max=150.0,
            step0_factor=0.6,
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ---- Phase 2 : refined annealing -----------------------------------
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer_refined(
                a_smooth, target_sum, deadline
            )
            if val_ann < best_score:
                best_score = val_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

        # ---- Phase 3 : polishing high‑β Adam (if time permits) -------------
        if time.time() < deadline:
            a_final, val_final = polish_highbeta(best_seq, target_sum, deadline)
            if val_final < best_score:
                best_score = val_final
                best_seq = a_final.copy()
                print(f"[{time.time():.2f}] New global best after polish: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – should never happen but guards against empty result.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Trim trailing near‑zeros for a compact representation
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Final evaluation (outside the optimisation loops)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm‑start in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
