def _simulated_annealing(a0, target_sum, deadline):
    a = a0.copy()
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score
    n = a.shape[0]
    rng = np.random.default_rng()
    T_start = 0.1 * best_score # maybe
    T_end = 1e-6
    total_iters = 0
    # compute schedule based on time left
    while time.time() < deadline:
        # fraction of time passed
        f = (time.time() - start) / (deadline - start)
        T = T_start * (1 - f) + T_end * f
        # propose move
        donor = rng.integers(n)
        while a[donor] <= 0:
            donor = rng.integers(n)
        receiver = rng.integers(n)
        if donor == receiver:
            continue
        # choose delta as fraction of donor mass
        max_delta = a[donor] * 0.2
        delta = rng.random() * max_delta
        if delta == 0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        # ensure sum consistent? Already,
        # no need to project because we maintain sum and positivity
        # evaluate
        new_score = _score(a)
        # use Metropolis
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # reject: revert
            a[receiver] -= delta
            a[donor] += delta
        total_iters++
    return best, best_score

"""
Step‑function optimizer for the max self‑convolution challenge.

The objective for a sequence a (with Σa = √(2n)) equals
    max_k (a * a)[k] .
We therefore minimise the maximal entry of the linear convolution
(a * a).  The algorithm combines several complementary heuristics:

1.  Warm‑start by scanning simple parametric families (power‑law,
    exponential‑tilted power‑law, Gaussian) on a fairly large problem
    size (n ≈ 15000–20000).
2.  Projected Adam gradient descent on a smooth soft‑max approximation
    of the max, with an exponential β‑schedule and occasional jitter.
3.  Exact deterministic mass‑transfer moves that strictly lower the
    current peak.
4.  Small‑LP (Frank‑Wolfe style) refinement on the active constraints.
5.  Simulated annealing with Metropolis acceptance to escape local
    minima.
6.  Final high‑β Adam polishing.

All phases respect a global deadline (default 1000 s).  The best
sequence found is trimmed of trailing near‑zero entries and returned
as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective value.  For Σa = √(2 n) the denominator cancels and the
    score equals the maximal convolution entry.
    """
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv_array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)   # soft‑max weights over convolution indices

    a_rev = a[::-1]           # reversed a for convolution with p
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # gradient for a (only the central n entries are relevant)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the soft‑max surrogate.
    Returns the best sequence seen and its (exact) score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform several Adam‑GD restarts, each starting from a jittered copy of
    the current best.  Returns the best sequence found.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy single‑donor / single‑receiver moves that strictly lower the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}

        # donors – indices where a[idx] > 0
        donor_mask = a[idx] > 0.0
        if not np.any(donor_mask):
            break
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receivers – smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)   # allow up to 90 % of donor mass
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based Frank‑Wolfe style refinement
# ----------------------------------------------------------------------


def _lp_iterative_refinement(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             epsilon_rel: float = 0.02,
                             step: float = 0.20,
                             max_iters: int = 120) -> tuple[np.ndarray, float]:
    """
    In each iteration solve a tiny LP:
        max Σ g_j  s.t.  f * g ≤ M   (only on active constraints)
    Scale g to the simplex and perform a Frank‑Wolfe update.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    eps = epsilon_rel
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        # indices where convolution is within eps·M of the maximum
        idx = np.where(conv >= M - eps * M)[0]
        if idx.size == 0:
            break
        # keep at most 30 most critical constraints
        if idx.size > 30:
            idx = idx[np.argsort(-conv[idx])[:30]]

        C = idx.size
        A = np.zeros((C, n), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            A[row, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

        b_vec = np.full(C, M, dtype=np.float64)

        try:
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_vec,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception as e:
            print(f"[{time.time():.2f}] LP exception: {e}")
            break

        if not res.success:
            # relax epsilon and continue
            eps = min(eps * 2.0, 0.1)
            it += 1
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # Scale g onto the simplex
        g = g * (target_sum / sum_g)

        a_new = (1.0 - step) * a + step * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            eps = max(eps * 0.8, 1e-5)
            print(f"[{time.time():.2f}] LP refinement it={it:3d} eps={eps:.2e} score={best_score:.6f}")
        else:
            step = max(step * 0.6, 0.01)
            eps = min(eps * 1.5, 0.05)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (Metropolis) – a stochastic global move search
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis annealing that moves small amounts of mass between
    indices.  The temperature decays exponentially from a fraction of the
    current score to a near‑zero value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time
    T0 = max(best_score * 0.02, 1e-3)   # initial temperature
    T_end = 1e-8

    # pre‑allocate a view for fast convolution
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        # exponential temperature schedule
        T = T0 * (T_end / T0) ** frac

        # pick donor (must have positive mass)
        donor = rng.integers(n)
        while a[donor] <= 0.0:
            donor = rng.integers(n)

        # pick receiver (different from donor)
        receiver = rng.integers(n)
        if receiver == donor:
            continue

        # propose a delta (up to 20 % of donor mass)
        max_delta = a[donor] * 0.2
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # apply move
        a[donor] -= delta
        a[receiver] += delta

        # evaluate
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal new best={best_score:.6f} T={T:.2e}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_score


# ----------------------------------------------------------------------
# Utility – trim trailing near‑zero entries

def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and final evaluation
    # ------------------------------------------------------------------
    final_seq = _trim_sequence(best_seq)

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
