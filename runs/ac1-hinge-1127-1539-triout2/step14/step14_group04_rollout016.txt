"""
Propose a low‑peak step function.

Idea:
*  The evaluation after ℓ1‑normalisation (Σa_i = √(2·n)) equals the
   maximum entry of the self‑convolution a*a.
*  If we restrict the support to k positions and keep the unordered
   sum‑representation of each index pair ≤ L, the ordered count per sum
   never exceeds 2·L (plus a possible diagonal contribution).
   Hence   max(a*a) ≤ (2·L)·(√(2·n)/k)².
*  Choosing L=2 (so each sum can appear at most twice as an unordered
   pair) gives max ordered count ≤ 4.  For n≈12000 we need k≈250 to get
   an objective below 1.5.
*  We build such a set by a greedy random construction that keeps the
   unordered‑pair count ≤ L.  Many restarts are performed within the
   time budget; the best construction is returned.
*  The final vector puts equal mass on the selected positions and is
   renormalised to respect the ℓ1‑norm exactly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core construction utilities
# ----------------------------------------------------------------------


def _build_sparse_set(n: int, L: int, rng: np.random.Generator):
    """
    Greedy random construction of a set P⊂{0,…,n‑1} such that each unordered
    sum i+j (i≤j) occurs at most L times.
    Returns the list of positions and the dictionary sum_counts.
    """
    order = rng.permutation(n)
    sum_counts = {}          # sum → unordered count
    pos = []                 # selected positions

    for x in order:
        # check self‑pair capacity
        s_self = x + x
        if sum_counts.get(s_self, 0) >= L:
            continue

        # check all pairs with already chosen positions
        ok = True
        for y in pos:
            s = x + y
            if sum_counts.get(s, 0) >= L:
                ok = False
                break
        if not ok:
            continue

        # add x to the set
        for y in pos:
            s = x + y
            sum_counts[s] = sum_counts.get(s, 0) + 1
        sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
        pos.append(x)

    return pos, sum_counts


def _max_ordered_pair_count(pos, n):
    """Compute the maximal ordered pair count of the set pos."""
    conv = np.zeros(2 * n - 1, dtype=int)
    arr = np.array(pos, dtype=int)
    for i in arr:
        conv[i + arr] += 1          # vectorised addition of all ordered pairs
    return int(conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low self‑convolution peak.
    The routine respects the supplied time budget (≈1000 s) and returns a list
    of heights.  Intermediate progress is printed to stdout.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    # Larger n gives a larger feasible set but costs a bit more time.
    # We use 12 k when the budget allows it.
    n = 12000 if budget_s >= 700 else 8000
    target_sum = math.sqrt(2.0 * n)     # Σa_i must equal this value
    L = 2                               # unordered sum multiplicity bound

    rng = np.random.default_rng()
    best_obj = float("inf")
    best_seq = None
    attempts = 0

    # ------------------------------------------------------------------
    # Repeated random greedy constructions
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        attempts += 1
        pos, _ = _build_sparse_set(n, L, rng)
        k = len(pos)
        if k < 10:       # trivial, skip
            continue

        Kmax = _max_ordered_pair_count(pos, n)   # ordered peak
        # objective = Kmax * (target_sum/k)²  (identical to evaluate_sequence)
        obj = Kmax * (target_sum / k) ** 2

        if obj < best_obj:
            best_obj = obj
            seq_arr = np.zeros(n, dtype=float)
            seq_arr[pos] = target_sum / k
            # renormalise to eliminate tiny rounding errors
            seq_arr *= target_sum / seq_arr.sum()
            best_seq = seq_arr
            print(
                f"[{time.time():.2f}] attempt {attempts:4d} "
                f"k={k:3d} Kmax={Kmax:2d} obj={obj:.6f}"
            )

        # early exit if we are already under the target
        if best_obj < 1.48:
            break

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        # simple uniform distribution – guaranteed to be valid
        n = 4000
        target_sum = math.sqrt(2.0 * n)
        seq_arr = np.full(n, target_sum / n, dtype=float)
        best_seq = seq_arr
        best_obj = 2 * n * np.max(np.convolve(seq_arr, seq_arr)) / (target_sum ** 2)

    # ------------------------------------------------------------------
    # Trim trailing zeros and return
    # ------------------------------------------------------------------
    nz = np.nonzero(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    final_seq = best_seq.tolist()
    globals()["height_sequence_1"] = final_seq

    # optional logging with the provided evaluate_sequence
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] evaluate_sequence → {val:.6f}")
    except Exception:
        pass

    print(
        f"[{time.time():.2f}] final best obj={best_obj:.6f}, "
        f"length={len(final_seq)}"
    )
    return final_seq
