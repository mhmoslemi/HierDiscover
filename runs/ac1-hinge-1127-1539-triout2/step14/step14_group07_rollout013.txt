def _smooth_max_and_grad(a, beta):
    size = 1 << ((2*a.size - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*a.size-1]
    max_b = b.max()
    # compute p
    exps = np.exp(beta * (b - max_b))
    sum_exps = exps.sum()
    p = exps / sum_exps
    # gradient w.r.t a: compute conv of reversed a with p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*a.size-1]
    grad_a = 2.0 * conv_rev[a.size-1:a.size-1+a.size]
    # smooth max value:
    f = max_b + np.log(sum_exps) / beta
    return f, grad_a

6️⃣  **L‑BFGS‑B refinement** – optimise a smoothed max‑objective in the
    softmax‑parameterised space (log‑variables) using a few L‑BFGS iterations.
7️⃣  **Pairwise optimal transfer** – another greedy sweep.
8️⃣  **Final high‑β Adam polishing** – finish with a high‑β Adam run.

All phases respect the global deadline and print intermediate scores.
The best sequence (trimmed of trailing ≈0 entries) is returned as a plain
Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog, fmin_l_bfgs_b

# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(b)/(sum(a)^2).  For Σa = √(2 n) the denominator equals 2·n,
    hence the score equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _semicircle_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1) - 1.0  # map to [-1,1]
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # cosine raised to a power ensures non‑negativity, we use plain cosine on half‑period
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / n)
    a = np.clip(a, 0.0, None)
    a *= target_sum / a.sum()
    return a


def _parabolic_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1) - 1.0
    a = np.clip(1.0 - x * x, 0.0, None)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
#  Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns gradient of the smooth‑max (log‑sum‑exp) of the convolution,
    along with the raw convolution vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    sum_exps = np.sum(exps)
    p = exps / sum_exps                       # softmax weights

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # ∂/∂a of the smooth‑max
    smooth_val = max_b + np.log(sum_exps) / beta
    return grad, smooth_val, b


# ----------------------------------------------------------------------
#  Adam‑style projected GD (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of max(b).
    Returns the best vector found and its true score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        grad, _, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple Adam restarts with jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
#  Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
#  Global peak‑mass transfer (refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second maximum (used as stop‑condition)
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]
            diff = donor_mass * shift_r - conv_donors
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} α={best_alpha:.4g} new_score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Pairwise optimal mass transfer
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]     # view of length 2n‑1

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
#  LP‑based Frank‑Wolfe refinement (new component)
# ----------------------------------------------------------------------


def _lp_iterative_refinement(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             epsilon_rel: float = 0.02,
                             step: float = 0.20,
                             max_iters: int = 150) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    eps = epsilon_rel
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        idx = np.where(conv >= M - eps * M)[0]
        if idx.size == 0:
            break
        if idx.size > 30:
            idx = idx[np.argsort(-conv[idx])[:30]]

        C = idx.size
        A = np.zeros((C, n), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            A[row, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

        b = np.full(C, M, dtype=np.float64)

        try:
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception as e:
            print(f"[{time.time():.2f}] LP exception: {e}")
            break

        if not res.success:
            eps = min(eps * 2.0, 0.1)
            it += 1
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        g = g * (target_sum / sum_g)

        a_new = (1.0 - step) * a + step * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            eps = max(eps * 0.8, 1e-5)
            print(f"[{time.time():.2f}] LP refinement it={it:3d} eps={eps:.2e} score={best_score:.6f}")
        else:
            step = max(step * 0.6, 0.01)
            eps = min(eps * 1.5, 0.05)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  L‑BFGS‑B refinement (smooth‑max on log‑variables)
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Compute the smooth‑max (log‑sum‑exp) of a∗a and its gradient w.r.t. a.
    Returns (value, grad_a, raw_convolution).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    sum_exps = np.sum(exps)
    p = exps / sum_exps                     # softmax weights

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad_a = 2.0 * conv_rev[n - 1: n - 1 + n]
    smooth_val = max_b + np.log(sum_exps) / beta
    return smooth_val, grad_a, b


def _lbfgs_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  beta: float = 5e4,
                  max_iter: int = 150) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the soft‑max objective using log‑variables.
    Returns the refined vector and its true score.
    """
    y0 = np.log(np.maximum(a0, 1e-12))

    def func_grad(y):
        a = np.exp(y)
        a = a * (target_sum / a.sum())
        val, grad_a, _ = _smooth_max_and_grad(a, beta)
        # chain rule to the log‑variables
        c = np.dot(a, grad_a) / target_sum
        grad_y = a * (grad_a - c)
        return val, grad_y

    # time‑budgeted wrapper: we will limit the number of function evaluations
    maxfun = max_iter
    try:
        y_opt, f_opt, _ = fmin_l_bfgs_b(func_grad, y0,
                                        maxfun=maxfun,
                                        iprint=-1)
    except Exception as e:
        print(f"[{time.time():.2f}] L‑BFGS exception: {e}")
        y_opt = y0

    a_opt = np.exp(y_opt)
    a_opt = a_opt * (target_sum / a_opt.sum())
    score = _score(a_opt)
    return a_opt, score


# ----------------------------------------------------------------------
#  Utility – trim trailing near‑zero entries
# ----------------------------------------------------------------------


def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]
    if nz.size == 0:
        return [0.0]
    return a[: nz[-1] + 1].tolist()


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising evaluate_sequence(...).
    Returns the best sequence found as a list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → more freedom, but higher cost)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) → score = max(b)

    # --------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (incl. new families)
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = _power_law_sequence(n, 0.5, target_sum)  # fallback
    best_score = _score(best_seq)

    # power‑law grid
    for p in np.linspace(0.30, 1.20, 41):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # exp‑tilted power‑law grid (extended ranges)
    for p in np.linspace(0.30, 1.20, 31):
        for q in np.linspace(0.0, 10.0, 31):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # Gaussian family (wider sigma range)
    for sigma in np.linspace(0.05 * n, 0.7 * n, 9):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # New analytic families
    for maker in (_semicircle_sequence,
                  _cosine_sequence,
                  _parabolic_sequence):
        seq = maker(n, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # If a previous construction exists, try to up‑sample it
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # Small Gaussian jitter around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam‑style projected GD with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.012, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam‑GD improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (core refinement)
    # --------------------------------------------------------------
    while time.time() < deadline - 20.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 20.0:
            break

    # --------------------------------------------------------------
    # Phase C – Global peak‑mass transfer (refinement)
    # --------------------------------------------------------------
    if time.time() < deadline - 12.0:
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.025,
                                                      receiver_frac=0.025)
        if val_peak < best_score:
            best_score, best_seq = val_peak, a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – LP‑based Frank‑Wolfe refinement (new)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_lp, val_lp = _lp_iterative_refinement(best_seq,
                                                target_sum,
                                                deadline,
                                                epsilon_rel=0.02,
                                                step=0.20,
                                                max_iters=120)
        if val_lp < best_score:
            best_score, best_seq = val_lp, a_lp.copy()
            print(f"[{time.time():.2f}] LP refinement improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – L‑BFGS‑B refinement (smooth‑max on log‑space)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_lbfgs, val_lbfgs = _lbfgs_refine(best_seq,
                                          target_sum,
                                          deadline,
                                          beta=5e4,
                                          max_iter=120)
        if val_lbfgs < best_score:
            best_score, best_seq = val_lbfgs, a_lbfgs.copy()
            print(f"[{time.time():.2f}] L‑BFGS refinement improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase F – Pairwise optimal transfer
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.015,
                                                      receiver_frac=0.015)
        if val_pair < best_score:
            best_score, best_seq = val_pair, a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase G – Final high‑β Adam polishing
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    final_seq = _trim_sequence(best_seq)

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
```

This implementation expands the parametric warm‑start with additional analytic families,
adds a dedicated L‑BFGS‑B refinement on a smooth approximation of the max‑norm,
and retains the proven mass‑transfer / LP / Adam polishing pipeline,
all while respecting the 1000 second time budget.
