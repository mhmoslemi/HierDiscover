def _subgradient_descent_custom(a0, deadline, target_sum):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # compute tol based on elapsed fraction
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total
        tol = 0.9 + 0.0999 * frac  # gradually increase from 0.9 to 0.9999
        I = (conv >= maxb * tol).astype(np.float64)

        # gradient via FFT
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        # update step factor schedule
        base_step = 0.8
        step_factor = base_step * (1 - 0.5 * frac)  # decays to half
        momentum = 0.995 * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if int(elapsed / 5) != int((elapsed - 0.001) / 5):
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:

"""
Step‑function optimizer.

The algorithm builds a good candidate quickly from a few parametric families,
refines it with a long sub‑gradient descent on the max‑convolution (the exact
objective), then up‑scales the solution and continues the descent.  Finally a
short simulated‑annealing mass‑transfer pass and a few smoothing steps are
performed.  All updates keep the vector on the ℓ¹‑simplex (∑a = √(2 n)) so
the benchmark score equals the maximal convolution entry.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Single‑hump sine."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Scatter large values by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‐convolution objective.
    The active set (tol) and step size decay gradually.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        # convolution and mask of (almost) maximal entries
        conv = _conv_fft(a)
        maxb = conv.max()
        elapsed = time.time() - start
        total = deadline - start
        frac = min(1.0, elapsed / total)

        # tolerance slowly tightens from 0.90 to 0.9999
        tol = 0.90 + 0.0999 * frac
        I = (conv >= maxb * tol).astype(np.float64)

        # gradient of the (soft) max
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        # step‑size schedule (starts larger, then shrinks)
        base_step = 0.8
        step_factor = base_step * (1.0 - 0.5 * frac)   # → 0.4 at the end
        momentum = 0.995 * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus (every ~5 s)
        if int(elapsed) % 5 == 0 and rng.random() < 0.5:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad time={elapsed:.1f}s best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_sa(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      start_temp: float = 0.3,
                      end_temp: float = 1e-5,
                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Local search swapping mass from highly contributing positions to weak ones."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # tiny random jitter to keep the search fluid
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light smoothing (low‑pass) – helps to squish isolated spikes
# ----------------------------------------------------------------------


def _smooth(seq: np.ndarray, target_sum: float, steps: int = 5) -> np.ndarray:
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    for _ in range(steps):
        seq = np.convolve(seq, kernel, mode='same')
        seq = _project_to_simplex(seq, target_sum)
    return seq

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The function returns the best sequence discovered within the time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # tiny safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1.   Build an initial pool (n ≈ 5 000)
    # ------------------------------------------------------------------
    n0 = 5000
    target0 = math.sqrt(2.0 * n0)

    init_pool = []

    # parametric families
    for p in np.linspace(0.1, 2.0, 9):
        init_pool.append(_power_law_sequence(n0, float(p), target0))
    for lam in np.linspace(0.001, 0.025, 9):
        init_pool.append(_exponential_sequence(n0, float(lam), target0))
    for _ in range(6):
        init_pool.append(_cosine_sequence(n0, target0))
    for _ in range(6):
        init_pool.append(_sin_half_sequence(n0, target0))

    # random Dirichlet‑like vectors
    for _ in range(12):
        a = rng.exponential(scale=1.0, size=n0)
        a = a / a.sum() * target0
        init_pool.append(a)

    # reuse a previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size > 0:
            init_pool.append(_resize_sequence(prev, n0))

    # keep the best candidate from the pool
    best_seq = None
    best_val = np.inf
    for seq in init_pool:
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Best init max={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2.   Sub‑gradient descent on the modest size (≈5 k)
    # ------------------------------------------------------------------
    sub_deadline = min(deadline - 300.0, time.time() + 250.0)  # keep at least 5 min for later stages
    best_seq, best_val = _subgradient_descent(best_seq, sub_deadline, target0)
    print(f"[{time.time():.2f}] After sub‑grad (n={n0}) best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.   Upscale to a larger support (≈20 k) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        remaining = deadline - time.time()
        if remaining > 250.0:
            n_big = 25000
        elif remaining > 180.0:
            n_big = 20000
        else:
            n_big = 15000

        target_big = math.sqrt(2.0 * n_big)
        seq_big = _resize_sequence(best_seq, n_big)
        seq_big = _project_to_simplex(seq_big, target_big)

        # scatter large entries more uniformly
        seq_big = _even_spread_order(seq_big)

        # ------------------------------------------------------------------
        # 4.   Heavy sub‑gradient descent on the large problem
        # ------------------------------------------------------------------
        sub_big_deadline = deadline - 80.0
        seq_big, best_big_val = _subgradient_descent(seq_big, sub_big_deadline, target_big)
        print(f"[{time.time():.2f}] Large‑scale sub‑grad n={n_big} best={best_big_val:.6f}")

        best_seq = seq_big
        best_val = best_big_val

    # ------------------------------------------------------------------
    # 5.   Short mass‑transfer simulated‑annealing pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        mt_deadline = deadline - 30.0
        best_seq, best_val = _mass_transfer_sa(best_seq, mt_deadline, target_big if 'target_big' in locals() else target0,
                                               start_temp=0.25, end_temp=1e-5, low_frac=0.06)
        print(f"[{time.time():.2f}] After MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.   Light smoothing to soften isolated spikes
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        target = target_big if 'target_big' in locals() else target0

    # ------------------------------------------------------------------
    # 7.   Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
