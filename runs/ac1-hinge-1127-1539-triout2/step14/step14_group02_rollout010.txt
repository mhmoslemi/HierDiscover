    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())

    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

"""
Propose a low‑score step‑function sequence.

The algorithm combines several ideas:
*  Build a diverse pool of initial candidates (warm‑start,
  deterministic shapes, random simples).
*  Run a projected smooth‑max gradient descent (soft‑max approx.
  of the self‑convolution peak) with an annealing schedule for the
  soft‑max temperature β.  The method uses FFT based convolutions,
  Euclidean projection onto the simplex Σa = S (S = √(2 n)) and an
  adaptive step size.
*  Refine the result with an aggressive O(n) simulated‑annealing
  mass‑transfer routine that moves tiny amounts of mass from indices
  that contribute heavily to the current peak towards low‑height
  indices.
*  Finally polish the best found vector with a short high‑β smooth‑max
  descent.

The whole search respects the supplied time budget (default 1000 s)
and stores the best sequence in the global variable ``height_sequence_1``
so that later runs can warm‑start from it.
"""

# Low‑level convolution utilities (FFT, power‑of‑2 sizes)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a power‑of‑2 FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact benchmark score 2·n·max(b) / (target_sum²)."""
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.1,
    beta_max: float = 30.0,
    step_factor: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a*a) via projected gradient descent.

    The smooth objective is (1/β)·log Σ exp(β·(a*a)_i).  After each step the
    vector is projected onto the simplex Σa = target_sum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    step0 = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        # self‑convolution and current exact score
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # soft‑max weights (log‑sum‑exp trick)
        shift = b - max_b
        w = np.exp(beta * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ a_rev) (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)           # length 3n‑2
        g = 2.0 * conv_wa[n - 1 : n - 1 + n]    # central n entries

        # adaptive step size proportional to target_sum / ‖g‖
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # slowly increase the temperature β
        if it % 1000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.25)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that transfers tiny mass from “bad’’ (large
    contribution to the current peak) to “good’’ (small height) indices.
    The self‑convolution is updated in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9   # initial temperature proportional to current energy

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling schedule
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # index of the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak (a[i]·a[k‑i])
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # transfer size – scaled by temperature
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # normalisation used by the reference

    # --------------------------------------------------------------
    # Build a pool of deterministic initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # warm start, if any
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    # uniform
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    init_candidates.append(uniform)

    # linear decreasing (big → small)
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin = lin / lin.sum() * target_sum
    init_candidates.append(lin)

    # exponential families
    for lam in (0.5, 1.0, 2.0):
        exp_seq = np.exp(-lam * np.arange(n, dtype=np.float64))
        exp_seq = exp_seq / exp_seq.sum() * target_sum
        init_candidates.append(exp_seq)

    # power‑law families
    for alpha in (0.5, 1.0, 1.5, 2.0):
        pow_seq = np.arange(1, n + 1, dtype=np.float64) ** (-alpha)
        pow_seq = pow_seq / pow_seq.sum() * target_sum
        init_candidates.append(pow_seq)

    # Hann (raised‑cosine) window
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    hann = hann / hann.sum() * target_sum
    init_candidates.append(hann)

    # Sine window
    sine = np.sin(np.pi * np.arange(n) / (n - 1))
    sine = sine / sine.sum() * target_sum
    init_candidates.append(sine)

    # Gaussian windows
    for sigma_factor in (0.15, 0.25):
        x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
        sigma = sigma_factor * n
        gauss = np.exp(-0.5 * (x / sigma) ** 2)
        gauss = gauss / gauss.sum() * target_sum
        init_candidates.append(gauss)

    # --------------------------------------------------------------
    # Evaluate deterministic pool and keep the best as a seed
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, cand in enumerate(init_candidates):
        scr = _score_from_selfconv(_conv_self_fft(cand), n, target_sum)
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
        print(f"[{time.time():.2f}] deterministic #{idx} score={scr:.6f}")

    # --------------------------------------------------------------
    # Prepare a list of optimisation seeds (best deterministic + random)
    # --------------------------------------------------------------
    seeds = [best_seq]
    rng = np.random.default_rng()
    # add a few random seeds, creating them on‑the‑fly if time permits
    for _ in range(5):
        if time.time() >= deadline:
            break
        rnd = rng.random(n, dtype=np.float64)
        rnd = _project_to_simplex(rnd, target_sum)
        seeds.append(rnd)

    # --------------------------------------------------------------
    # Optimise each seed, respecting the remaining time budget
    # --------------------------------------------------------------
    for s_idx, seed in enumerate(seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        seeds_left = len(seeds) - s_idx
        # allocate a modest share for this seed
        per_seed_budget = remaining / seeds_left
        # 60 % smooth descent, 40 % annealing
        phase1_deadline = time.time() + 0.60 * per_seed_budget

        print(f"[{time.time():.2f}] ----- optimisation seed #{s_idx + 1} -----")

        # Phase 1 – smooth‑max descent
        a_smooth, val_smooth = smooth_max_descent(
            seed,
            target_sum,
            phase1_deadline,
            beta_start=0.1,
            beta_max=30.0,
            step_factor=0.8,
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # Phase 2 – annealing (use whatever time is left)
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(a_smooth, target_sum, deadline)
            if val_ann < best_score:
                best_score = val_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing with a short high‑β smooth descent (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print(f"[{time.time():.2f}] Final polishing phase")
        a_final, val_final = smooth_max_descent(
            best_seq,
            target_sum,
            deadline,
            beta_start=15.0,
            beta_max=80.0,
            step_factor=0.5,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] New global best after final polish: {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up: trim trailing near‑zeros and return as plain list
    # --------------------------------------------------------------
    arr = best_seq.copy()
    arr[arr < 1e-12] = 0.0
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Final external evaluation (optional, for logging only)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
