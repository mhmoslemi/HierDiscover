    one receiver, probing several donor/receiver fractions.
5.  a final polishing Adam run with a very high β and a small step‑size.
6.  a short stochastic hill‑climbing pass that tries random donor→receiver
    moves respecting the exact max‑convolution bound.

All phases stop when the global time budget (default 1000 s) is exhausted.
The best sequence found is trimmed of trailing near‑zero entries and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Evaluation score = max self‑convolution (sum is normalised)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto {x ≥ 0, Σx = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_decay_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _double_exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n) + np.exp(-q * (n - 1 - i) / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, m_frac: float, r: float, target_sum: float) -> np.ndarray:
    """Two‑level step: first m entries height h, rest height l, h/l = r."""
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a


def _tri_piecewise_sequence(
    n: int,
    m1_frac: float,
    m2_frac: float,
    r1: float,
    r2: float,
    target_sum: float,
) -> np.ndarray:
    """Three‑level step: high, medium, low."""
    m1 = max(1, int(round(m1_frac * n)))
    m2 = max(1, int(round(m2_frac * n)))
    if m1 + m2 > n:
        m2 = n - m1
    # heights: h1 = r1 * h2, h2 = r2 * l   (so h1/h2=r1, h2/l=r2)
    # unknown: l (low)
    # total = m1*h1 + m2*h2 + (n-m1-m2)*l = l * (m1*r1*r2 + m2*r2 + (n-m1-m2))
    denom = m1 * r1 * r2 + m2 * r2 + (n - m1 - m2)
    l = target_sum / denom
    h2 = r2 * l
    h1 = r1 * h2
    a = np.full(n, l, dtype=np.float64)
    a[:m1] = h1
    a[m1 : m1 + m2] = h2
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (for Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return gradient of soft‑max approximation of max‑convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_projected_gd(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _adam_restarts(
    a_start: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Multiple Adam restarts (each with its own jitter)."""
    best_a = a_start.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a_opt, val_opt = _adam_projected_gd(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=5e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact, more aggressive)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    donor_keep_frac: float = 0.95,
    delta_safety: float = 0.99,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy mass move from a strong contributor to a weak one."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest entry
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]  # one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]  # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose receiver with minimal partner weight
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * donor_keep_frac, delta_needed * delta_safety)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (aggressive, larger donor set)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    donor_frac: float = 0.08,
    receiver_frac: float = 0.08,
    min_alpha: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """Move a fraction α of the total donor mass to a weak receiver."""
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second largest entry
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)

        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate(
            [np.zeros(pad_len, dtype=np.float64), a, np.zeros(pad_len, dtype=np.float64)]
        )

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r : r + 2 * n - 1]
            diff = donor_mass * shift_r - conv_donors
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] / (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] + donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Global peak transfer it={it} α={best_alpha:.4g} score={best_score:.6f}"
            )
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (exact convolution evaluation)
# ----------------------------------------------------------------------


def _hill_climb(
    a0: np.ndarray, deadline: float, target_sum: float, max_trials: int = 200000
) -> tuple[np.ndarray, float]:
    """Random donor→receiver moves with exact max check."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_trials:
        # pick donor weighted by a (more mass => more impact)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 0.0:
            it += 1
            continue
        # pick receiver inversely weighted (prefer low entries)
        inv = 1.0 / (a + 1e-12)
        receiver = rng.choice(n, p=inv / inv.sum())
        if donor == receiver:
            it += 1
            continue

        # propose a small delta (up to 20% of donor)
        delta = rng.random() * 0.2 * a[donor]
        if delta <= 0.0:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta
        # No need to re‑project: sum is preserved, non‑negativity holds by construction
        cand_score = _score(a_candidate)
        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            a = a_candidate
            # occasional print
            if it % 5000 == 0:
                print(f"[{time.time():.2f}] Hill‑climb it={it} new best={best_score:.6f}")
        it += 1

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the maximum self‑convolution.  The routine respects the given time
    budget (seconds) and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Choose problem size.  Larger n gives finer freedom but costs more time.
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)  # Σa = √(2 n) ⇒ score = max convolution

    # ------------------------------------------------------------------
    # Warm‑start: parametric families + optional previous construction
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    rng = np.random.default_rng()

    # 1) power‑law scan (coarse)
    for p in np.linspace(0.0, 2.0, 31):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # 2) exponential‑tilted power‑law (moderate grid)
    for p in np.linspace(0.0, 2.0, 21):
        for q in np.linspace(0.0, 20.0, 21):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()

    # 3) pure exponential decay
    for q in np.linspace(0.0, 20.0, 21):
        seq = _exp_decay_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # 4) double‑exponential (symmetric)
    for q in np.linspace(0.0, 15.0, 16):
        seq = _double_exp_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # 5) Gaussian widths
    for sigma in np.linspace(0.1 * n, 0.5 * n, 9):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # 6) two‑level piecewise
    for m_frac in np.linspace(0.01, 0.5, 8):
        for r in np.linspace(1.0, 30.0, 16):
            seq = _piecewise_sequence(n, float(m_frac), float(r), target_sum)
            val = _score(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()

    # 7) three‑level piecewise (random sampling to keep time low)
    for _ in range(300):
        m1_frac = rng.random() * 0.3
        m2_frac = rng.random() * (0.5 - m1_frac)
        r1 = 1.0 + rng.random() * 20.0
        r2 = 1.0 + rng.random() * 5.0
        seq = _tri_piecewise_sequence(
            n, m1_frac, m2_frac, r1, r2, target_sum
        )
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # 8) use previously stored construction if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_val:
            best_val, best_seq = val, resized.copy()

    # 9) random jitter around current best (helps escape tiny flat zones)
    for _ in range(5):
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] Warm‑start best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional symmetrisation (often reduces peak)
    # ------------------------------------------------------------------
    sym = (best_seq + best_seq[::-1]) * 0.5
    sym = _project_to_simplex(sym, target_sum)
    val_sym = _score(sym)
    if val_sym < best_val:
        best_val, best_seq = val_sym, sym.copy()
        print(f"[{time.time():.2f}] Symmetrisation improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A: Adam projected GD with restarts (30 % of remaining time)
    # ------------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.30
    a_gd, val_gd = _adam_restarts(best_seq, target_sum, phase_a_deadline)
    if val_gd < best_val:
        best_val, best_seq = val_gd, a_gd.copy()
        print(f"[{time.time():.2f}] Adam phase improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B: Deterministic mass‑transfer (more aggressive)
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_dmt, val_dmt = _deterministic_mass_transfer_exact2(
            best_seq,
            deadline,
            target_sum,
            donor_keep_frac=0.95,
            delta_safety=0.99,
        )
        if val_dmt < best_val:
            best_val, best_seq = val_dmt, a_dmt.copy()
            print(f"[{time.time():.2f}] DMT improved to {best_val:.6f}")

        # occasional random perturbation
        a_rand, val_rand = _global_peak_mass_transfer(
            best_seq,
            deadline,
            target_sum,
            donor_frac=0.08,
            receiver_frac=0.08,
        )
        if val_rand < best_val:
            best_val, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Global peak move improved to {best_val:.6f}")

        if time.time() > deadline - 10.0:
            break

    # ------------------------------------------------------------------
    # Phase C: Global peak‑mass transfer with several fraction settings
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        for df, rf in [(0.08, 0.08), (0.12, 0.12), (0.04, 0.04)]:
            a_peak, val_peak = _global_peak_mass_transfer(
                best_seq, deadline, target_sum, donor_frac=df, receiver_frac=rf
            )
            if val_peak < best_val:
                best_val, best_seq = val_peak, a_peak.copy()
                print(
                    f"[{time.time():.2f}] Global peak (df={df}) improved to {best_val:.6f}"
                )

    # ------------------------------------------------------------------
    # Phase D: Stochastic hill‑climbing (exact evaluation)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_hc, val_hc = _hill_climb(
            best_seq,
            deadline,
            target_sum,
            max_trials=300000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, a_hc.copy()
            print(f"[{time.time():.2f}] Hill‑climb improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase E: Final polishing Adam with very high β
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_projected_gd(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.994,
        )
        if val_fin < best_val:
            best_val, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polishing gave {best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional verification with the supplied evaluator
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
