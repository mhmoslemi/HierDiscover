    rng = np.random.default_rng()

    total = deadline - time.time()
    while time.time() < deadline:
        # compute current peak index
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # candidate donor selection: choose index in [lo, hi] with probability proportional to a[i] * a[k_max-i]
        idx_range = np.arange(lo, hi+1, dtype=np.int64)
        contrib = a[idx_range] * a[k_max - idx_range]
        if contrib.sum() > 0:
            donor = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            donor = rng.choice(idx_range)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

            continue

"""
Propose a low‑objective step‑function.

This implementation improves on the reference by:
* Using a projected Adam optimiser on a smooth “log‑sum‑exp” approximation
  of the max‑convolution entry (phase 1).  The Adam update adapts the step
  size automatically and converges faster.
* Enforcing symmetry (a ≈ rev(a)) after each projection – the optimal
  solution is known to be symmetric, which reduces the search space.
* Refining the best candidate with a fast greedy hill‑climbing that moves
  tiny amounts of mass from indices that contribute most to the current
  peak towards indices with the smallest value (phase 2).  The convolution
  is updated in O(n) per move, exactly as in the reference annealer.
* Keeping several random restarts and a warm‑start from the previous run.
The code respects a user‑provided time budget (default 1000 s) and
returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution and utility functions
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _objective_from_conv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact objective 2·n·max(b) / (target_sum²)."""
    return float(2.0 * n * b.max() / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max Adam descent
# ----------------------------------------------------------------------


def _smooth_max_adam(a0: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     beta_start: float = 0.05,
                     beta_max: float = 20.0,
                     lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(b) where b = a*a.
    The smooth approximation is (1/β)·log Σ exp(β·b_i).
    Adam is used for the projected gradient steps.
    Symmetry (a ≈ a[::-1]) is enforced after each projection.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    b = _conv_self_fft(a)
    best_val = _objective_from_conv(b, n, target_sum)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    beta = beta_start

    while time.time() < deadline:
        it += 1

        # self‑convolution and smooth‑max weights
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = _objective_from_conv(b, n, target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # softmax over all 2n‑1 entries (numerically stable)
        b_shift = b - max_b                     # max(b_shift) == 0
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max w.r.t. a:
        #   ∂/∂a ( w · (a*a) ) = 2 * (w * a_rev)   (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)           # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step

        # projection onto the simplex Σa = target_sum , a ≥ 0
        a = _project_to_simplex(a, target_sum)

        # enforce (approximate) symmetry – this does not change the sum
        a = (a + a[::-1]) * 0.5

        # slowly increase β to focus on the true max
        if it % 500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.3)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – deterministic greedy hill‑climbing
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                target_sum: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that only accept improvements.
    At each step the current convolution peak is located;
    mass is taken from a donor that contributes most to that peak
    and transferred to a receiver with the smallest height.
    The convolution is updated in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _objective_from_conv(b, n, target_sum)

    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak: a[i] · a[k_max-i]
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.choice(idx)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # scheduled step size – larger early, shrinking to 0
        elapsed = time.time() - start
        frac = elapsed / total
        max_delta = min(a[donor] * 0.25, target_sum * 0.006)   # a few % of available mass
        delta = max_delta * (1.0 - frac) * rng.random()
        if delta <= 0.0:
            continue

        # ---- O(n) exact update of the convolution -----------------
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        # -----------------------------------------------------------

        new_val = _objective_from_conv(b_new, n, target_sum)

        # accept only if the max‑convolution entry improves
        if new_val < cur_val:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] hill‑climb improved best={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose a problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 14000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)          # normalisation used in the benchmark

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + random + uniform)
    # --------------------------------------------------------------
    init_candidates = []

    # warm start from previous run if available
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    # uniform vector (known symmetric baseline)
    uniform = np.full(n, target_sum / n)
    init_candidates.append(uniform)

    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optimisation loop over all starts
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ***** start #{idx + 1} / {len(init_candidates)} *****")

        # ---- Phase 1 : Adam smooth‑max descent --------------------
        phase1_deadline = time.time() + 0.35 * (deadline - time.time())
        a_adam, val_adam = _smooth_max_adam(
            init, target_sum, phase1_deadline,
            beta_start=0.05, beta_max=20.0, lr=0.5
        )
        if val_adam < best_score:
            best_score = val_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.2f}] New global best after Adam: {best_score:.6f}")

        # ---- Phase 2 : greedy hill‑climbing refinement -------------
        if time.time() < deadline:
            a_hc, val_hc = _hill_climb(a_adam, target_sum, deadline)
            if val_hc < best_score:
                best_score = val_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.2f}] New global best after hill‑climb: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and evaluation
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – simple decreasing power‑law
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)

    # trim trailing zeros (slightly shorter representation)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (outside the optimisation loops)
    try:
        final_val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                      # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for a possible warm start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
