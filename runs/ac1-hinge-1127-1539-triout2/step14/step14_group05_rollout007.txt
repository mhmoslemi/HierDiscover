    start = time.time()
    deadline = start + budget_s - 0.5
    rng = np.random.default_rng()

    # Determine n0
    n0 = 5000
    target0 = math.sqrt(2*n0)

    # Build initial sequences:
    init_seqs = []

    # param families
    for p in np.linspace(0.2, 1.2, 9):  # include near 1
        init_seqs.append(_power_law_sequence(n0, p, target0))
    for lam in np.linspace(0.0005, 0.03, 8):
        init_seqs.append(_exponential_sequence(n0, lam, target0))
    # Add raised cosine, sin half, random, etc.
    # maybe also lognormal shape?

    if 'height_sequence_1' in globals():

    for seq in list(init_seqs):

        s = seq + jitter
        s = np.maximum(s, 0)
        s = _project_to_simplex(s, target0)
        init_seqs.append(s)

    best_seq = None

        a_opt, val_opt = _adam_softmax_gradient_descent(seed, run_deadline, target0)

    # LP refinement

def _adam_softmax_gradient_descent(a0, deadline, target_sum,
                                   max_beta=9e5, step_factor=0.9,
                                   gamma=0.98,
                                   beta1=0.9, beta2=0.999, eps=1e-8):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n - 1]
        max_b = b.max()
        # softmax
        exps = np.exp(beta * (b - max_b))
        p = exps / exps.sum()
        # gradient via convolution with reversed a
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # shape (n,)
        # Adam updates
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = step_base / ((beta + 1.0) * (np.sqrt(v_hat) + eps))
        a -= lr * m_hat
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it} β={beta:.1f} val={best_val:.6f}")

        # increase beta
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val

"""
Optimisation of non‑negative step functions for the benchmark.

The pipeline combines:
  * diverse parametric families (power‑law, exponential, cosine, sinusoid,
    sigmoid, Gaussian) and random jittered variants,
  * Adam‑accelerated soft‑max gradient descent on a smooth approximation of
    the maximal convolution value,
  * a small linear‑program refinement on the most active convolution constraints,
  * up‑scaling to a larger support (up to 25 000 entries) with several
    deterministic reorderings,
  * Adam‑based exact sub‑gradient descent on the true max‑convolution,
  * a brief permutation search + hill‑climb,
  * a final simulated‑annealing mass‑transfer phase.

All updates project onto the ℓ¹‑simplex with target sum √(2 n),
so the evaluation score equals the maximal entry of the self‑convolution.
The routine respects the 1000 s budget and stores the best sequence found
in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _sigmoid_sequence(n: int, steep: float, target_sum: float) -> np.ndarray:
    """Shifted sigmoid, decreasing from left to right."""
    x = np.arange(n, dtype=np.float64) / n
    a = 1.0 / (1.0 + np.exp((x - 0.5) * steep))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle of the window."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Deterministic re‑ordering helpers (to spread large entries)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _adam_softmax_gradient_descent(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 9e5,
                                   step_factor: float = 0.9,
                                   beta1: float = 0.9,
                                   beta2: float = 0.999,
                                   eps: float = 1e-8) -> tuple[np.ndarray, float]:

    a = _project_to_simplex(a0.copy(), target_sum)

    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_b = b.max()

        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = (target_sum * step_factor) / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad‑Adam it={it:5d} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Small LP refinement (active‑set)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               max_iters: int,
               deadline: float,
               K: int = 12,
               mix_factor: float = 0.15) -> tuple[np.ndarray, float]:
    """Linear‑program refinement: maximise ℓ¹‑norm of g under active convolution constraints."""
    try:
        from scipy.optimize import linprog
    except Exception:
        print("SciPy not available – skipping LP refinement.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        # active set – the K largest convolution entries
        if K >= conv.size:
            active = np.arange(conv.size, dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # Build constraint matrix A (|active| × n)
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            A[row, idx] = a[k - idx]

        # LP: maximise sum(g)  →  minimise -sum(g)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"LP failed at iter {it}: {res.message}")
            break

        g = res.x
        g *= target_sum / g.sum()   # project onto simplex

        # convex combination
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation search helpers
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to keep diversity
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    Returns a Python list of floats (the heights).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1.  Build a diverse initial pool (n≈5 000)
    # --------------------------------------------------------------
    n0 = 5000
    target0 = math.sqrt(2.0 * n0)

    init_pool = []

    # parametric families covering a wide shape spectrum
    for p in np.linspace(0.1, 1.4, 12):
        init_pool.append(_power_law_sequence(n0, float(p), target0))
    for lam in np.linspace(0.0005, 0.025, 10):
        init_pool.append(_exponential_sequence(n0, float(lam), target0))
    for _ in range(6):
        init_pool.append(_cosine_sequence(n0, target0))
        init_pool.append(_sin_half_sequence(n0, target0))
    for steep in (5.0, 10.0, 15.0):
        init_pool.append(_sigmoid_sequence(n0, steep, target0))
    for sigma in (0.07, 0.12, 0.2):
        init_pool.append(_gaussian_sequence(n0, sigma, target0))

    for _ in range(12):
        init_pool.append(_random_simplex(n0, target0, rng))

    # reuse a previous run if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n0))

    # jittered variants of everything above
    for seq in list(init_pool):
        jitter = rng.normal(scale=target0 * 0.02, size=n0)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target0)
        init_pool.append(seq_j)

    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------------
    # 2.  Adam‑soft‑max GD runs (short bursts) on the modest size
    # --------------------------------------------------------------
    while time.time() < deadline - 40.0:
        # choose a seed – occasionally perturb the current best
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target0 * 0.005, size=n0)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target0)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 13.0, deadline - 30.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _adam_softmax_gradient_descent(
            seed, run_deadline, target0,
            max_beta=9e5, step_factor=0.9)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_val:.6f}")

    # --------------------------------------------------------------
    # 3.  Linear‑program refinement (moderate size)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        lp_deadline = time.time() + min(18.0, deadline - time.time() - 12.0)
        best_seq, lp_score = _lp_refine(best_seq,
                                        target0,
                                        max_iters=30,
                                        deadline=lp_deadline,
                                        K=30,
                                        mix_factor=0.1)
        if lp_score < best_val:
            best_val = lp_score
            print(f"[{time.time():.2f}] After LP refinement best={best_val:.6f}")

    # --------------------------------------------------------------
    # 4.  Upscaling to a larger support
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 500.0:
        n_big = 25000
    elif remaining > 350.0:
        n_big = 20000
    elif remaining > 200.0:
        n_big = 15000
    else:
        n_big = n0

    target_big = math.sqrt(2.0 * n_big)

    if best_seq is not None:
        seq_big = _resize_sequence(best_seq, n_big)
    else:
        seq_big = _random_simplex(n0, target0, rng)
        seq_big = _resize_sequence(seq_big, n_big)

    seq_big = _project_to_simplex(seq_big, target_big)

    # try a few deterministic reorderings, keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        rng.permutation(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 5.  Heavy sub‑gradient descent (Adam) on the large scale
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 120.0
        seq_big, sub_val = _subgradient_descent_adam(seq_big,
                                                     sub_deadline,
                                                     target_big,
                                                     step_factor=0.55,
                                                     tol=0.995)
        if sub_val < best_val:
            best_val = sub_val
            best_seq = seq_big.copy()
            print(f"[{time.time():.2f}] Sub‑grad‑Adam best={best_val:.6f}")

    # --------------------------------------------------------------
    # 6.  Short Adam‑soft‑max GD on the large problem (fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline - 70.0:
        gd_deadline = deadline - 70.0
        seq_big, gd_val = _adam_softmax_gradient_descent(seq_big,
                                                         gd_deadline,
                                                         target_big,
                                                         max_beta=5e5,
                                                         step_factor=0.8)
        if gd_val < best_val:
            best_val = gd_val
            best_seq = seq_big.copy()
            print(f"[{time.time():.2f}] Fine‑tune GD best={best_val:.6f}")

    # --------------------------------------------------------------
    # 7.  Permutation search (random + hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline - 45.0:
        perm_deadline = time.time() + min(8.0, deadline - time.time() - 40.0)
        seq_big, _ = _permutation_search(seq_big, trials=120, deadline=perm_deadline)
        print(f"[{time.time():.2f}] After random perm max={_max_convolution_fft(seq_big):.6f}")

    if time.time() < deadline - 30.0:
        hill_deadline = time.time() + min(6.0, deadline - time.time() - 25.0)
        seq_big, _ = _local_permutation_hillclimb(seq_big, hill_deadline)
        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 8.  Final greedy mass‑transfer simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = time.time() + min(5.0, deadline - time.time() - 3.0)
        seq_big, mt_score = _mass_transfer_simulated_annealing(seq_big,
                                                               mt_deadline,
                                                               target_big,
                                                               start_temp=0.25,
                                                               end_temp=1e-5,
                                                               low_frac=0.05)
        if mt_score < best_val:
            best_val = mt_score
            best_seq = seq_big.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # --------------------------------------------------------------
    # 9.  Trim trailing near‑zero entries and return the result
    # --------------------------------------------------------------
    final_arr = np.array(best_seq if best_seq is not None else seq_big, dtype=float)
    nz = np.nonzero(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]

    final_seq = final_arr.tolist()
    globals()["height_sequence_1"] = final_seq  # for possible reuse

    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception:
        pass

    return final_seq
