def _lp_improvement(a, target_sum):
    n = len(a)
    M = _max_convolution_fft(a)
    # Build constraint matrix A (dense?), maybe using csr matrix.
    # Use loops to fill rows:
    data = []
    rows = []
    cols = []
    for i in range(n):
        start = i
        end = i + n
        rows.extend(range(start, end))
        cols.extend([i] * n)
        data.extend([a[i]] * n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Set bounds: 0 <= g_i <= None
    # Solve max sum(g) s.t. A @ g <= M
    c = -np.ones(n) # minimize negative sum
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if not res.success:
        return a # fallback
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a
    # normalize to target sum
    g_scaled = g * (target_sum / sum_g)
    # mix with small t
    t = 0.2
    new_a = (1 - t) * a + t * g_scaled
    # project onto simplex (should maintain sum)
    new_a = _project_to_simplex(new_a, target_sum)

def _mass_transfer_local(a, deadline, target_sum, start_temp=0.5, end_temp=1e-5):
    # Use simulated annealing moving mass from high w to low w
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    conv = _conv_fft(a)
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # find kmax (index of max conv)
        kmax = int(np.argmax(conv))

        # compute overlapping index range
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi+1)
        partner = kmax - idx
        # product contributions
        w = a[idx] * a[partner]
        # For i == partner case (when kmax even and i=partner), w = a_i^2

        # donor index: where w max
        donor_local = idx[np.argmax(w)]  # actual index
        # choose receiver among indices that are not in overlapping range (or low w)
        # possible receivers: complement set
        non_overlap = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        if non_overlap.size > 0:
            receiver = rng.choice(non_overlap)
        else:
            # fallback: choose min w index
            receiver = idx[np.argmin(w)]

        if donor_local == receiver:
            continue

        # delta: fraction of donor's value
        max_delta = a[donor_local] * 0.5
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # transfer
        a[donor_local] -= delta
        a[receiver] += delta

        # ensure nonnegative sum above - maybe small negative due to rounding: fix
        a = np.maximum(a, 0.0)

        # produce conv update
        conv = _conv_fft(a)
        new_score = _score(a)

        # accept if improvement or probabilistic acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT local new best={best_score:.6f}")
        else:
            # revert
            a[donor_local] += delta
            a[receiver] -= delta

        # keep constraint sum = target_sum? We moved delta from donor to receiver, sum unchanged.
        # So no need to project; but after rounding may need small adjustment.
        # Possibly we may deviate due to clipping; we can project occasionally.
        if time.time() % 5.0 < 0.1:

"""
Improved search for low‑peak step‑function.

Main idea:
* Explore a few simple parametric families (power‑law, exponential, random)
  using projected sub‑gradient descent on a moderate size.
* Upscale the best sequence to a larger dimension and apply an
  alternating ordering that spreads large masses to the ends.
* Perform a long simulated‑annealing mass‑transfer phase that repeatedly
  moves mass from indices heavily contributing to the current peak to
  indices that do not overlap that peak.  The loop runs until the time
  budget is exhausted.
* Trim trailing zeros and return the result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(sum a)^2 (the value returned by evaluate_sequence)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the values in descending order alternately from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution entry.
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the max‑convolution entry."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # gradient of the max entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        # double count the middle term when kmax even (i == partner)
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            # occasional jitter to escape plateaus
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer refinement (focused on the current peak)
# ----------------------------------------------------------------------


def _mass_transfer_local(a0: np.ndarray,
                        deadline: float,
                        target_sum: float,
                        start_temp: float = 0.5,
                        end_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """Local mass‑transfer moves that continuously try to lower the peak."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # current peak
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # overlapping index range for this shift
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        w = a[idx] * a[partner]                     # contribution product

        # donor = index inside [lo,hi] with largest contribution
        donor_local = idx[np.argmax(w)]

        # try to pick a receiver outside the overlapping range
        candidate_set = np.arange(n)
        mask_out = (candidate_set < lo) | (candidate_set > hi)
        if np.any(mask_out):
            receiver = rng.choice(candidate_set[mask_out])
        else:
            # fallback: pick the index with smallest contribution inside the range
            receiver = idx[np.argmin(w)]

        if donor_local == receiver:
            continue

        # amount to move – a fraction of the donor value, limited to keep positivity
        max_delta = a[donor_local] * 0.5
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor_local] -= delta
        a[receiver] += delta

        # numerical safety
        a = np.maximum(a, 0.0)

        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT local new best={best_score:.6f}")
        else:
            # revert move
            a[donor_local] += delta
            a[receiver] -= delta

        # Occasionally re‑project to the simplex to correct accumulated numerical drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

"""
Improved search for a low‑peak step function.

The algorithm proceeds in three stages:
1.  Exploration on a moderate dimension (≈ 4000) using projected
    sub‑gradient descent from several simple seeds (power‑law,
    exponential, random Dirichlet, linear).
2.  Upscaling the best candidate to a larger size (≈ 10000–12000) and
    reordering it with an alternating pattern that pushes large
    values to the ends – this reduces the overlap with the reversed
    sequence.
3.  A long simulated‑annealing mass‑transfer phase that repeatedly
    moves mass from indices heavily contributing to the current peak
    of the convolution to indices that do not overlap that peak.
    The loop runs until the time budget is exhausted, keeping the
    best vector found.

All vectors stay on the ℓ¹‑simplex (∑aᵢ = √(2·n)) and are non‑negative.
The final sequence is trimmed of trailing zeros before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(sum a)^2 (same as evaluate_sequence)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place values in descending order alternately from left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution entry.
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the max‑convolution entry."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # gradient of the max entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer refinement (focused on the current peak)
# ----------------------------------------------------------------------


def _mass_transfer_local(a0: np.ndarray,
                        deadline: float,
                        target_sum: float,
                        start_temp: float = 0.5,
                        end_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """Local mass‑transfer moves that try to lower the convolution peak."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # current peak of the convolution
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # indices that actually contribute to this peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        w = a[idx] * a[partner]                     # product contributions

        donor_local = idx[np.argmax(w)]

        # try to pick a receiver outside the overlapping range
        mask_out = (np.arange(n) < lo) | (np.arange(n) > hi)
        if np.any(mask_out):
            receiver = rng.choice(np.where(mask_out)[0])
        else:
            receiver = idx[np.argmin(w)]

        if donor_local == receiver:
            it += 1
            continue

        max_delta = a[donor_local] * 0.5
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta

        a[donor_local] -= delta
        a[receiver] += delta
        a = np.maximum(a, 0.0)

        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT local new best={best_score:.6f}")
        else:
            # revert move
            a[donor_local] += delta
            a[receiver] -= delta

        # occasional re‑projection to combat numeric drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the given time budget (≈1000 s) and returns a list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # short safety margin

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 0 – parameters for the exploratory stage
    # --------------------------------------------------------------
    n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    # --------------------------------------------------------------
    # Phase 1 – build a small pool of seed sequences
    # --------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.4, 0.8, 5):
        seeds.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.003, 0.02, 4):
        seeds.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like seeds
    for _ in range(4):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        seeds.append(a)

    # linear decreasing seed
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    seeds.append(lin)

    # reuse previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            seeds.append(_resize_sequence(prev, n_expl))

    # --------------------------------------------------------------
    # Phase 2 – exploration with projected sub‑gradient descent
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # reserve ~40 % of total time for this phase
    explore_deadline = deadline - 300.0
    while time.time() < explore_deadline:
        # pick a seed (occasionally jitter the current best)
        if best_seq is not None and rng.random() < 0.45:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(seeds)

        # run a short sub‑gradient descent run
        run_dead = min(time.time() + 12.0, explore_deadline)
        a_opt, val_opt = _subgradient_descent(
            seed,
            run_dead,
            target_expl,
            max_iter=120_000,
        )

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

        if time.time() >= explore_deadline:
            break

    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.6, target_expl)

    # --------------------------------------------------------------
    # Phase 3 – upscale to a larger dimension
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 400.0:
        n_final = 12000
    elif remaining > 200.0:
        n_final = 10000
    else:
        n_final = max(n_expl, int(remaining * 12))

    target_final = np.sqrt(2.0 * n_final)

    a_big = _resize_sequence(best_seq, n_final)
    a_big = _project_to_simplex(a_big, target_final)

    # spread large values to the ends
    a_big = _alternating_order(a_big)

    # --------------------------------------------------------------
    # Phase 4 – intensive local mass‑transfer refinement (dominant time)
    # --------------------------------------------------------------
    a_opt, final_score = _mass_transfer_local(
        a_big,
        deadline,
        target_final,
        start_temp=0.5,
        end_temp=1e-5,
    )

    # --------------------------------------------------------------
    # Phase 5 – trim trailing zeros and return the result
    # --------------------------------------------------------------
    arr = np.array(a_opt, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:

    final_seq = arr.tolist()

    # optional final evaluation for logging

        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
