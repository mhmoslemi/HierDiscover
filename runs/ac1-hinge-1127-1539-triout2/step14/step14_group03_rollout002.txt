def _conv_full_fft(x, y):
    n = x.size
    m = y.size
    size = 1 << ((n + m -1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n+m-1]
    return conv

    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2 = 0.999; epsilon = 1e-8; lr = 0.2 # adjust
    t = 0
    pow1 = 1.0; pow2 = 1.0
    beta = 0.5
    beta_max = 128.0
    beta_inc_iter = 2000  # increase every this many iterations
    
    while time.time() < deadline:
        t += 1
        pow1 *= beta1
        pow2 *= beta2
        # compute conv
        conv = _conv_full_fft(a, a)  # length 2n-1
        max_conv = conv.max()
        # compute softmax weights
        shifted = conv - max_conv  # shift to avoid overflow
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # compute gradient
        a_rev = a[::-1]
        grad_full = _conv_full_fft(a_rev, w)  # length 3n-2
        grad = 2.0 * grad_full[n-1:n-1+n]  # slice
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - pow1)
        v_hat = v / (1 - pow2)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        # non-neg clip
        np.maximum(a, 0.0, out=a)
        # project onto simplex (sum target_sum)
        a = _project_to_simplex(a, target_sum)
        # evaluate
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD iter={t} beta={beta:.3f} score={best_score:.6f}")
        # increase beta periodically
        if t % beta_inc_iter == 0 and beta < beta_max:
            beta = min(beta_max, beta * 2.0)
        # optional jitter
        if t % 1000 == 0:
            # add small random perturbation to avoid local minima
            jitter = np.random.default_rng().normal(scale=target_sum*1e-4, size=n)
            a += jitter
            np.maximum(a, 0.0, out=a)
            a = _project_to_simplex(a, target_sum)

    (power‑law, exponential, combined, beta) and a stored global
    sequence.  The best candidate (lowest true max‑convolution) is kept.

2️⃣  Smooth‑max Adam optimisation – minimise the true peak of the
    self‑convolution using a differentiable soft‑max surrogate.
    The temperature β is increased gradually (continuation) while
    Adam updates are projected onto the simplex by simple scaling
    (maintain Σa = √(2n)).  This gives a fast, gradient‑based refinement.

3️⃣  Aggressive simulated‑annealing mass transfer – a fast O(n) move
    that shifts a small amount of mass from “bad’’ (large contribution)
    to “good’’ indices.  The remaining time budget is spent here.

All helper functions are top‑level, no closures or lambdas are used.
The routine respects the given time budget and returns the best list
of non‑negative heights it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using an FFT of size
    next power‑of‑two.
    """
    n = x.size
    m = y.size
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Convenience wrapper for a*a."""
    return _conv_full_fft(a, a)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0 , Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n‑1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam·i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape on (0,1):
    a_i ∝ x^{α-1}(1‑x)^{α‑1},  x = (i+0.5)/n.
    """
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


# ----------------------------------------------------------------------
# Scoring utilities (exact, no clipping)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the benchmark value 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _conv_self_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max Adam optimisation
# ----------------------------------------------------------------------


def _adam_smooth_max(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    The temperature β starts small and is doubled periodically (continuation).
    After each step the vector is re‑scaled to satisfy Σa = target_sum.
    Returns the best vector seen and its true score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.25

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    pow1 = 1.0
    pow2 = 1.0

    best_a = a.copy()
    best_val = _score(a)

    # continuation parameters
    beta = 0.4          # initial soft‑max temperature
    beta_max = 128.0
    beta_inc_every = 2000  # iterations

    rng = np.random.default_rng()
    it = 0
    start = time.time()

    while time.time() < deadline:
        it += 1
        pow1 *= beta1
        pow2 *= beta2

        # ----- compute full self‑convolution -----
        conv = _conv_self_fft(a)               # length 2n‑1
        max_conv = conv.max()

        # ----- soft‑max weights -----
        shifted = conv - max_conv               # stabilise exp
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        a_rev = a[::-1]
        grad_full = _conv_full_fft(a_rev, w)    # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - pow1)
        v_hat = v / (1.0 - pow2)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        np.maximum(a, 0.0, out=a)                # enforce non‑negativity

        # ----- project onto simplex by simple scaling -----
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)

        # ----- record best -----
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} β={beta:.2f} score={best_val:.6f}")

        # ----- continuation step -----
        if it % beta_inc_every == 0 and beta < beta_max:
            beta = min(beta_max, beta * 2.0)

        # ----- occasional jitter to escape plateaus -----
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a += jitter
            np.maximum(a, 0.0, out=a)
            s = a.sum()
            if s > 0.0:
                a *= (target_sum / s)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution via FFT (power‑of‑2 size)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Fast O(n) simulated‑annealing that moves tiny mass from indices which
    contribute heavily to the current convolution peak to indices with low
    height.  The temperature follows a square‑root cooling schedule.
    Returns the best sequence found and its true score.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                       # exact self‑convolution
    cur_val = float(2.0 * n * b.max() / (target_sum * target_sum))
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    T0 = cur_val * 0.9                     # initial temperature

    it = 0
    while time.time() < deadline:
        # cooling
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate the current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – prefer small heights
        inv = 1.0 / (a + 1e-12)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0.0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- O(n) update of the convolution -----
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new = b.copy()
        b_new[donor: donor + n] -= 2.0 * delta * a
        b_new[receiver: receiver + n] += 2.0 * delta * a

        # quadratic part (Δ²) – careful handling of indices
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = float(2.0 * n * b.max() / (target_sum * target_sum))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – trade‑off between granularity and speed
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 13000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)      # normalisation used by the benchmark

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start: deterministic families + optional warm‑start
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # warm start from previous run (if any)
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        val = _score(warm)
        best_seq = warm.copy()
        best_score = val
        print(f"[{time.time():.2f}] Warm start score = {val:.6f}")

    # power‑law sweep (finer grid)
    for p in np.linspace(0.4, 0.8, 9):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} score={val:.6f}")

    # exponential sweep
    for lam in np.linspace(0.4, 9.0, 9):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exp λ={lam:.3f} score={val:.6f}")

    # combined power‑law + exponential (randomised)
    rng = np.random.default_rng()
    for _ in range(12):
        p = rng.random() * 0.8 + 0.1        # 0.1 … 0.9
        lam = rng.random() * 9.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] PowerExp p={p:.3f} λ={lam:.3f} score={val:.6f}")

    # symmetric Beta family
    for alpha in np.linspace(0.25, 0.9, 9):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.3f} score={val:.6f}")

    # a few random Dirichlet starts
    for _ in range(6):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _score(rnd)
        if val < best_score:
            best_score = val
            best_seq = rnd.copy()
            print(f"[{time.time():.2f}] Random start score={val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Smooth‑max Adam optimisation (continuation)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # allocate about 55 % of the remaining budget to this stage
        remaining = deadline - time.time()
        sub_deadline = time.time() + 0.55 * remaining
        a_adam, val_adam = _adam_smooth_max(best_seq, target_sum, sub_deadline)
        if val_adam < best_score:
            best_score = val_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.2f}] New global best after Adam {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Final annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = _anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] New global best after anneal {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, truncation of trailing zeros and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final check using the provided evaluate_sequence
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
