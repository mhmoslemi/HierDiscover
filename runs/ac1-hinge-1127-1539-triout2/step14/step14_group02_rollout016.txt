def make_geometric(alpha, n, target_sum):
    if alpha <= 0:
        # uniform
        a = np.ones(n, dtype=np.float64)
    else:
        r = 1.0 - alpha
        # log1p for precision
        log_r = np.log1p(-alpha)
        a = np.exp(np.arange(n) * log_r)
    # scaling
    sum_a = a.sum()
    if sum_a == 0:
        return a
    a = a * (target_sum / sum_a)
    return a

"""
Propose a low‑score step‑function sequence.

The search works in three stages:

1. **Parametric sweep** – generate simple families (geometric decay and
   power‑law) with the correct ℓ₁‑norm (√(2 n)).  The objective
   2·n·max(conv(a,a)) / (∑a)² is evaluated by an FFT‑based self‑convolution
   (much faster than the provided evaluate_sequence).  The best
   candidate from this cheap sweep is kept.

2. **Optional refinement** – if time permits, the best candidate is fed
   to a smooth‑max projected‑gradient descent (soft‑max approximation of
   the max) followed by a fast simulated‑annealing “mass‑transfer”
   stage (identical to the reference implementation).  Both phases honor
   the remaining time budget.

3. **Return** – the final vector is trimmed of trailing zeros,
   stored globally for warm‑starts, and its true score is printed using the
   supplied `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast and exact for real data)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a power‑of‑2 FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """
    Objective 2·n·max(b) / (target_sum²) – b must be a*a.
    target_sum is the desired ℓ₁‑norm (√(2 n)).
    """
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution.
    The smoothing parameter β is increased gradually.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    step0 = target_sum * 0.8
    it = 0

    while time.time() < deadline:
        b = _conv_self_fft(a)                     # length 2n‑1
        cur_val = _score_from_selfconv(b, n, target_sum)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # soft‑max weights
        max_b = b.max()
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ a_rev)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)                     # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]                # central slice

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Move tiny amounts of mass from indices that contribute heavily to the current
    convolution peak to indices with little mass.  O(n) updates keep the run fast.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # square‑root cooling

        # locate current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # exact O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = _score_from_selfconv(b_new, n, target_sum)

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------


def make_geometric(alpha: float, n: int, target_sum: float) -> np.ndarray:
    """
    a_i = r^i,   r = 1 - alpha   (alpha > 0)
    The vector is scaled to have ℓ₁‑norm = target_sum.
    """
    if alpha <= 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        # log1p gives accurate log(1‑alpha) even for tiny alpha
        log_r = np.log1p(-alpha)
        a = np.exp(np.arange(n, dtype=np.float64) * log_r)
    s = a.sum()
    if s == 0.0:
        return a
    a *= target_sum / s
    return a


def make_power(p: float, n: int, target_sum: float) -> np.ndarray:
    """
    a_i = (i+1)^{-p}, scaled to ℓ₁‑norm = target_sum.
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    s = a.sum()
    if s == 0.0:
        return a
    a *= target_sum / s
    return a


def fast_score(seq: np.ndarray, n: int, target_sum: float) -> float:
    """Rapid evaluation using the internal FFT‑based convolution."""
    b = _conv_self_fft(seq)
    return _score_from_selfconv(b, n, target_sum)

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # normalisation used in the reference

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Helper to update the global best
    # --------------------------------------------------------------
    def try_candidate(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = fast_score(seq, n, target_sum)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) geometric family (parameter α = 1 - r)
    # --------------------------------------------------------------
    # explore α in a wide range; focus around 1/n where the optimum is expected
    approx_one_over_n = 1.0 / n
    alphas = np.concatenate((
        np.logspace(-5, -3, 30),               # 1e‑5 … 1e‑3
        np.logspace(-3, -1, 30),               # 1e‑3 … 1e‑1
        approx_one_over_n * np.linspace(0.5, 5.0, 30)  # refined window
    ))
    alphas = np.unique(alphas)

    for alpha in alphas:
        if time.time() >= deadline:
            break
        seq = make_geometric(alpha, n, target_sum)
        try_candidate(seq)

    # --------------------------------------------------------------
    # 2) power‑law family a_i = (i+1)^{-p}
    # --------------------------------------------------------------
    ps = np.linspace(0.1, 2.0, 40)
    for p in ps:
        if time.time() >= deadline:
            break
        seq = make_power(p, n, target_sum)
        try_candidate(seq)

    # --------------------------------------------------------------
    # 3) warm‑start from the previous run (if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = _project_to_simplex(warm, target_sum)
        try_candidate(warm)

    # --------------------------------------------------------------
    # 4) optional refinement (smooth descent + annealing)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 30.0:
        # allocate half the remaining budget to each phase
        sub_deadline = time.time() + 0.5 * time_left
        a_smooth, val_smooth = smooth_max_descent(
            best_seq, target_sum, sub_deadline,
            beta_start=0.1, beta_max=50.0
        )
        if val_smooth < best_score:
            best_seq = a_smooth.copy()
            best_score = val_smooth
            print(f"[{time.time():.2f}] after smooth best = {best_score:.6f}")

        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
            if val_ann < best_score:
                best_seq = a_ann.copy()
                best_score = val_ann
                print(f"[{time.time():.2f}] after anneal best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) final clean‑up and evaluation
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: decreasing power‑law (very unlikely to be used)
        best_seq = make_power(0.55, n, target_sum)

    arr = best_seq
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation using the provided (secure) function
    try:
        final_val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {final_val:.6f}")
    except Exception as exc:                      # pragma: no cover
        print(f"[{time.time():.2f}] evaluation error: {exc}")

    # store for possible warm‑starts in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
