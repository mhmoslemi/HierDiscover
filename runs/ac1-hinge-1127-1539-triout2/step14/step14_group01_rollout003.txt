"""
Hybrid optimizer for the minimum‑peak auto‑convolution problem.

* Seed pool – power‑law, exponential, Dirichlet (various concentrations),
  symmetric versions, and the previous best (if any).  All seeds are
  projected onto the ℓ¹‑simplex Σa = √(2n).

* Differential Evolution – modest population, projected mutation/crossover.

* Smooth‑max (log‑sum‑exp) descent – a differentiable surrogate for the
  ‖a*a‖∞ objective.  The surrogate uses a temperature‑like α that is
  increased from a small value up to a large one, while an
  annealed Metropolis rule allows occasional uphill moves.

* Random‑permutation hill‑climbing – shuffles the final vector many times
  and keeps the best ordering (the convolution peak is highly ordering‑
  dependent).

* Simple reorderings – alternating and even‑spread patterns are tried as
  cheap post‑processes.

The routine adapts its time allocation to the supplied budget (default
1000 s) and stores the best sequence in ``globals()['height_sequence_1']``
for warm‑starting the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, conc: float, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(α,…,α) with concentration 'conc'."""
    a = rng.dirichlet([conc] * n)
    a *= target_sum
    return a


def _symmetric_sequence(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Make a sequence symmetric (a_i = a_{n-1-i}) and re‑project."""
    a = (seq + seq[::-1]) * 0.5
    return _project_to_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Return full linear convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution) using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """Objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Differential Evolution (projected to simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean projection onto the ℓ¹‑simplex."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind, target_sum) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # Choose three distinct mates
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # Binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial, target_sum)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        # early break to preserve time for later phases
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) descent on the peak convolution
# ----------------------------------------------------------------------


def _smooth_max_descent(a0: np.ndarray,
                        target_sum: float,
                        deadline: float,
                        rng: np.random.Generator,
                        step_factor: float = 0.6,
                        alpha_start: float = 2.0,
                        alpha_end: float = 200.0,
                        T_end: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed max (log‑sum‑exp) of a*a.
    Uses a temperature‑controlled Metropolis acceptance rule.
    """
    n = a0.shape[0]
    size = _next_pow2(3 * n - 2)            # enough for correlation
    a = a0.copy()
    # initial values
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    cur_max = float(conv.max())
    best_a = a.copy()
    best_max = cur_max

    start_time = time.time()
    T_start = max(1e-4, cur_max * 0.2)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0

        # annealing schedules
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        T = T_start * (1.0 - frac) + T_end * frac

        # recompute A and conv for current a (may be stale from previous iteration)
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
        cur_max = float(conv.max())

        # softmax weights (stable version)
        w = np.exp(alpha * (conv - cur_max))
        w /= w.sum()                               # now Σ w = 1

        # gradient of the smoothed max
        W = np.fft.rfft(w, n=size)
        grad_fft = A * np.conj(W)
        grad = 2.0 * np.fft.irfft(grad_fft, n=size)[:n]
        # numerical noise may introduce tiny imag part – ignore
        grad = np.real(grad)

        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        step = step_factor * target_sum / (norm + 1e-12)
        a_candidate = a - step * grad
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        # evaluate candidate
        Ac = np.fft.rfft(a_candidate, n=size)
        conv_c = np.fft.irfft(Ac * Ac, n=size)[: 2 * n - 1]
        new_max = float(conv_c.max())

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

        # optional tiny jitter to escape shallow plateaus (very rare)
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 5e-5, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

    return best_a, best_max

# Random permutation hill‑climbing
# ----------------------------------------------------------------------


def _permute_hillclimb(a: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       max_iter: int = 5000) -> np.ndarray:
    """
    Randomly permute the vector repeatedly and keep the best ordering.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _max_convolution_fft(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        perm = rng.permutation(n)
        cand = best_seq[perm]          # keep using the current best as base
        val = _max_convolution_fft(cand)
        if val < best_val:
            best_seq = cand
            best_val = val
        it += 1
    return best_seq

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest→leftmost, second‑largest→rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimisation for the minimum‑peak auto‑convolution problem.
    Returns a list of non‑negative heights (the step‑function coefficients).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives richer space but higher cost)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 500:
        n = 6000
    else:
        n = 4000
    target_sum = math.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation (diverse pool)
    # --------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.0, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 6):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # Dirichlet (various concentrations)
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        seeds.append(_dirichlet_sequence(n, conc, target_sum, rng))

    # pure random simplex
    for _ in range(8):
        seeds.append(_random_simplex(n, target_sum, rng))

    # symmetric version of a random seed
    for _ in range(4):
        base = _random_simplex(n, target_sum, rng)
        seeds.append(_symmetric_sequence(base, target_sum))

    # warm start from previous run
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            # interpolate to current length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev_resized = np.interp(new_x, old_x, prev)
            seeds.append(_project_to_simplex(prev_resized, target_sum))
        else:
            seeds.append(_project_to_simplex(prev, target_sum))

    # small jitter to increase diversity
    jitter_scale = target_sum * 0.02
    for seq in list(seeds):
        jitter = rng.normal(scale=jitter_scale, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # --------------------------------------------------------------
    # Initialise DE population
    # --------------------------------------------------------------
    pop_size = max(40, int(n / 200))   # ~40‑45 individuals for n≈8000
    population = []

    best_seed = min(seeds, key=lambda s: _score(s, target_sum))
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # --------------------------------------------------------------
    # Differential Evolution phase (~35 % of budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.35 * budget_s
    de_deadline = min(de_deadline, deadline - 6.0)   # reserve time for refinements
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Smooth‑max (log‑sum‑exp) gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        grad_deadline = deadline - 4.0
        seq_gd, max_gd = _smooth_max_descent(best_seq, target_sum, grad_deadline, rng)
        if max_gd < _max_convolution_fft(best_seq):
            best_seq = seq_gd
            print(f"[{time.time():.2f}] Smoothed‑max descent improved – new peak = {max_gd:.6f}")

    # --------------------------------------------------------------
    # Random permutation hill‑climbing
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_deadline = deadline - 1.0
        seq_perm = _permute_hillclimb(best_seq, perm_deadline, rng, max_iter=2000)
        if _max_convolution_fft(seq_perm) < _max_convolution_fft(best_seq):
            best_seq = seq_perm
            print(f"[{time.time():.2f}] Permutation hill‑climb improved – new peak = {_max_convolution_fft(best_seq):.6f}")

    # --------------------------------------------------------------
    # Simple deterministic reorderings (often help)
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq)
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq, target_sum)
    print(f"[{time.time():.2f}] After reordering – final peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store for warm‑starting the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator (optional)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
