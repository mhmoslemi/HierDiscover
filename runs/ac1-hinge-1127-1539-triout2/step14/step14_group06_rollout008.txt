def _infinity_gradient_descent(a, deadline, target_sum):
    while time left:
        conv = _conv_fft(a)
        max_val = conv.max()
        peak_indices = np.where(conv >= max_val - 1e-12)[0]
        # compute gradient sum over these peaks
        grad = np.zeros_like(a)
        n = len(a)
        for k in peak_indices:
            # compute B for k
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi+1)
            grad[idx] += a[k-idx]
            if k <= n-1:
                idx2 = np.arange(k, n)
                grad[idx2] += a[idx2 - k]
        # gradient of max: subgradient includes convex hull of these; we pick sum.
        # direction for descent
        d = -(grad - np.mean(grad))
        # compute feasible step
        neg = d < 0
        if not np.any(neg):
            break
        alpha_max = np.min(-a[neg] / d[neg])
        # choose step factor less than alpha_max
        alpha = alpha_max * 0.5  # maybe 0.5 or 0.9
        a_new = a + alpha * d
        a_new = np.clip(a_new, 0.0, None)  # ensure nonneg.
        # reproject onto simplex to adjust sum drift
        a_new = _project_to_simplex(a_new, target_sum)
        new_max = _max_convolution_fft(a_new)
        if new_max < max_val - 1e-9:
            a = a_new
        else:
            # reduce step factor
            alpha *= 0.5
            if alpha < 1e-12:
                break
        # update best if improves
        if new_max < best_max:
            best_a = a.copy()
            best_max = new_max
    return best_a, best_max

def _infinity_descent(a, deadline, target_sum, max_iter=10000):
    a = a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    iter_count = 0
    while time.time() < deadline and iter_count < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        # early stop if no improvement potential
        if best_max - max_val < 1e-12:
            # no improvement
            pass
        # find peak indices
        peaks = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros(n, dtype=np.float64)
        for k in peaks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx1 = np.arange(lo, hi+1)
            grad[idx1] += a[k - idx1]
            if k <= n-1:
                idx2 = np.arange(k, n)
                grad[idx2] += a[idx2 - k]
        # direction for decreasing max conv: negative of grad
        # adjust to zero sum: d = -(grad - np.mean(grad))
        d = -(grad - np.mean(grad))
        # compute max feasible step size to keep nonnegativity
        neg = d < -1e-15
        if not np.any(neg):
            break
        alpha_max = np.min(-a[neg] / d[neg])  # note d[neg] negative
        # choose step size as fraction of max feasible (like 0.8)
        # also limit step size to avoid overshoot
        alpha = alpha_max * 0.9
        # apply step
        a_new = a + alpha * d
        # ensure nonnegative due to rounding
        a_new = np.maximum(a_new, 0.0)
        # project onto simplex to maintain sum
        a_new = _project_to_simplex(a_new, target_sum)
        new_max = _max_convolution_fft(a_new)

"""
Propose a non‑negative step function (a list of heights) that minimises the
maximum entry of its self‑convolution.
The objective used by the competition is

    score = 2·n·max(conv) / (sum(a)²)

and the evaluation routine forces sum(a)=√(2·n), so the score is simply
max(conv).  The search proceeds in several stages, all respecting the
overall 1000 s time budget:

1.  Rich parametric warm‑start (power‑law, tilted power‑law,
    exponential, double‑exponential, Gaussian, two‑level piecewise,
    beta‑distribution, sine‑power).  Very fine grids are tried and the
    best sequence is kept.

2.  Optional symmetrisation (averaging with the reversal) – this often
    reduces the peak.

3.  Projected Adam optimisation on a smooth soft‑max approximation of
    the ∞‑norm.  Several random restarts are performed.

4.  Deterministic mass‑transfer moves (the “DMT2” step from the reference)
    – a greedy local move that lowers the current peak.

5.  Global peak‑mass transfer – move a large fraction of the total mass
    of a set of “donor’’ indices to a single weak “receiver’’; the
    feasible fraction α is computed analytically.

6.  **New** L∞‑gradient descent:  at each iteration the exact gradient
    of the current peak (or of all peak entries) is formed, a descent
    direction with zero‑sum is taken, and the maximal feasible step
    size (respecting non‑negativity) is applied.  A simple back‑track
    guarantees monotone decrease of the peak.

7.  Final polishing with Adam at a very large soft‑max temperature.

8.  Trailing near‑zero entries are trimmed and the final list is
    returned.

All phases stop early when the global deadline is reached.  The best
sequence found so far is kept throughout the run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective returned by the competition: 2·n·max(conv)/(sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_decay_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _double_exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n) + np.exp(-q * (n - 1 - i) / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, m_frac: float, r: float, target_sum: float) -> np.ndarray:
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)  # (m·h + (n-m)·l) = target_sum, h = r·l
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1) + 1e-30)
    a *= target_sum / a.sum()
    return a


def _sin_power_sequence(n: int, gamma: float, target_sum: float) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n)^γ  (non‑negative, symmetric)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(np.pi * x) ** gamma
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities for Adam‑GD
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple Adam restarts from jittered starting points."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

# Deterministic mass‑transfer (exact DMT2) – same as reference

def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest entry
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second largest entry
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]
            diff = donor_mass * shift_r - conv_donors
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] / (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} α={best_alpha:.4g} new_score={best_score:.6f}")

        it += 1

    return best_a, best_score

# New: Exact L∞‑gradient descent (monotone peak reduction)

def _infinity_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Greedy descent on the true max‑norm:
    * evaluate current peak(s);
    * build the exact sub‑gradient of the peak(s);
    * take the largest feasible step (respecting non‑negativity);
    * project onto the simplex;
    * accept only if the peak is lowered.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max >= best_max - 1e-14:
            # no improvement so far
            pass

        # indexes attaining the current maximum (within a tight tolerance)
        peaks = np.where(conv >= cur_max - 1e-12)[0]

        # build gradient of sum of the selected peak entries
        grad = np.zeros_like(a)
        for k in peaks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx1 = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx1] += a[k - idx1]               # term a_{k-i}
            if k <= n - 1:
                idx2 = np.arange(k, n, dtype=np.int64)
                grad[idx2] += a[idx2 - k]           # term a_{i-k}

        # descent direction: move opposite to the gradient, then make it zero‑sum
        direction = -(grad - np.mean(grad))

        # maximal step keeping all entries non‑negative
        neg = direction < -1e-15
        if not np.any(neg):
            break
        alpha_max = np.min(-a[neg] / direction[neg])

        # try a step; we start with a fairly aggressive fraction of the feasible size
        step_factors = (0.9, 0.6, 0.3, 0.1, 0.05)
        improved = False
        for sf in step_factors:
            alpha = alpha_max * sf
            a_new = a + alpha * direction
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_to_simplex(a_new, target_sum)
            new_max = _max_convolution_fft(a_new)
            if new_max < cur_max - 1e-13:
                a = a_new
                cur_max = new_max
                improved = True
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] ∞‑descent it={it:5d} max={best_max:.6f}")
                break
        if not improved:
            # cannot make progress – stop
            break
        it += 1

    return best_a, best_max

    sym = (best_seq + best_seq[::-1]) * 0.5
    sym = _project_to_simplex(sym, target_sum)
    val_sym = _score(sym)
    if val_sym < best_score:
        best_score = val_sym
        best_seq = sym.copy()
        print(f"[{time.time():.2f}] Symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Projected Adam gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # occasional random perturbation
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Global peak‑mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.025,
                                                      receiver_frac=0.025)
        if val_peak < best_score:
            best_score = val_peak
            best_seq = a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – New exact L∞‑gradient descent (aggressive peak reduction)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_inf, max_inf = _infinity_gradient_descent(best_seq,
                                                    deadline,
                                                    target_sum,
                                                    max_iter=20000)
        if max_inf < best_score:
            best_score = max_inf
            best_seq = a_inf.copy()
            print(f"[{time.time():.2f}] ∞‑gradient descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final polishing with Adam (very large β)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final verification (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
