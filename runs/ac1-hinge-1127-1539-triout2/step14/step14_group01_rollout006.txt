"""
Hybrid optimizer for the minimum peak auto‑convolution problem.

The goal is to minimise   2·n·max(a*a) / (∑a)²
under a_i ≥ 0.  By fixing the ℓ¹‑norm to √(2n) the objective becomes
exactly the peak of the linear convolution a*a.
The code proceeds in three stages:

1.  **Seeding** – generate a diverse set of feasible vectors
    (power‑law, exponential, random Dirichlet and the warm‑start from the
    previous run).

2.  **Smooth‑gradient descent** – minimise a smooth approximation of the
    max‑convolution:
        f_β(a) = (1/β)·log∑_k exp(β·(a*a)_k) .
    The gradient of f_β can be written as a correlation between the weight
    vector w_i ∝ exp(β·(a*a)_i) and the current a, and is computed with
    three FFTs per iteration.  β is increased from ≈0.5 to ≈80 so that the
    optimisation first smoothes the landscape and then focuses on the true
    peak.  After each step we Euclidean‑project onto the ℓ¹‑simplex.

3.  **Polishing** – a deterministic “mass‑transfer’’ greedy loop (as in the
    previous state‑of‑the‑art) and a final re‑ordering (alternating/even‑spread)
    to lower any remaining peak.

The routine respects the 1000 s time budget, stores the best sequence in the
global variable ``height_sequence_1`` for warm‑starting the next invocation,
and returns the final list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Sequence factories (different shapes)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (same as previous state‑of‑the‑art)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = float(conv.max())

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑gradient descent on a log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             max_iters: int = 5000,
                             beta_start: float = 0.5,
                             beta_end: float = 80.0,
                             step_scale: float = 0.5,
                             momentum_factor: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on   f_β(a) = (1/β)·log∑_k exp(β·conv_k) .
    β is increased exponentially during the run.
    The gradient is computed via a correlation using three FFTs per iteration.
    Returns the best vector (lowest peak of the exact convolution) found.
    """
    a = a0.copy()
    n = a.shape[0]
    # prepare FFT size sufficient for correlation (length 3n‑2)
    size = 1 << ((3 * n - 2).bit_length())
    momentum = np.zeros_like(a)

    # keep track of the best exact peak encountered
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # β schedule (exponential increase)
        frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** frac

        # ----- convolution of a with itself -----
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

        # ----- smooth weights (softmax) -----
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            break
        w /= w_sum

        # ----- gradient via correlation w ⋆ a -----
        fw = np.fft.rfft(w, n=size)
        corr = np.fft.irfft(fw * np.conj(fa), n=size)  # linear correlation, length = size
        # correlation indices 0 … n‑1 correspond to lags  –(n‑1)…0
        grad = 2.0 * corr[:n][::-1]                     # g_j = 2·∑_i w_i a_{i‑j}

        # ----- momentum & step -----
        momentum = momentum_factor * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            eta = step_scale * target_sum / (norm + 1e-12)
            a = a - eta * momentum
        else:
            # tiny jitter to escape possible stalling
            a += np.random.normal(scale=target_sum * 1e-4, size=n)

        # ----- projection onto the ℓ¹‑simplex -----
        a = _project_to_simplex(a, target_sum)

        # ----- keep the best exact solution -----
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser: diverse seeding → smooth‑gradient descent →
    deterministic mass‑transfer → final re‑ordering.
    The best sequence (as a plain Python list) is returned and stored in
    ``globals()['height_sequence_1']`` for warm‑starting the next call.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ----- problem size ------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 500:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ----- seed generation ---------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.0, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 6):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # pure random Dirichlet points
    for _ in range(8):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # ----- pick the best seed as starting point -------------------------
    best_seed = min(seeds, key=lambda s: _max_convolution_fft(s))
    best_seq = best_seed.copy()
    best_val = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] initial seed peak = {best_val:.6f}")

    # ----- smooth‑gradient descent – several random restarts -------------
    num_restarts = 5
    for r in range(num_restarts):
        if time.time() > deadline - 15.0:
            break

        # jitter the current best to obtain a fresh start
        start_seq = best_seq + rng.normal(scale=target_sum * 0.01, size=n)
        start_seq = np.maximum(start_seq, 0.0)
        start_seq = _project_to_simplex(start_seq, target_sum)

        seq, val = _smooth_gradient_descent(start_seq, target_sum, deadline)
        if val < best_val:
            best_seq, best_val = seq, val
            print(f"[{time.time():.2f}] restart {r}: improved peak = {best_val:.6f}")

    # ----- deterministic mass‑transfer polishing -------------------------
    if time.time() < deadline - 5.0:
        mt_dead = deadline - 5.0
        seq_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_sum,
                                                      start_temp=0.4,
                                                      end_temp=1e-6,
                                                      low_frac=0.07)
        if val_mt < best_val:
            best_seq, best_val = seq_mt, val_mt
            print(f"[{time.time():.2f}] mass‑transfer polished peak = {best_val:.6f}")

    # ----- try a few simple reorderings ---------------------------------
    candidates = [best_seq,
                  _alternating_order(best_seq),
                  _even_spread_order(best_seq)]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_val = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] after ordering final peak = {best_val:.6f}")

    # ----- trim trailing near‑zero entries -------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator (if it exists)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
