"""
Hybrid parametric + long sub‑gradient optimisation for the
minimum‑peak auto‑convolution problem.

Steps
-----
1. Choose a problem size `n` (larger n gives a richer search space).
2. Build a diverse set of *parametric* candidates:
   - power‑law a_i ∝ (i+1)^(-p)
   - exponential a_i ∝ exp(-λ·i)
   - “block’’ distributions (all mass on the first f·n entries)
   - a few random points on the ℓ¹‑simplex and the previous best.
   All candidates are projected onto the simplex Σa = √(2n).
3. Keep the best of those candidates (lowest max‑convolution).
4. Run a long *projected sub‑gradient descent* on the exact
   max‑convolution objective.  The step size decays linearly,
   a momentum term (γ≈0.95) is used, and a tiny jitter is added
   every few hundred iterations to escape plateaus.
5. If time permits, perform a cheap Monte‑Carlo “mass‑transfer’’
   hill‑climb with a simple Metropolis acceptance rule.
6. Finally test three simple reorderings (original, alternating,
   even‑spread) and keep the ordering with the smallest peak.
7. Trim trailing near‑zero entries, store the result for warm‑starts,
   and return it.

The routine respects the 1000 s budget, prints intermediate scores,
and works with any `evaluate_sequence` already available in the
environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s , x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto {x_i ≥ 0, Σx_i = s} (Euclidean norm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence factories (all return a vector with Σ = target_sum)
# ----------------------------------------------------------------------


def _power_law_seq(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_seq(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _block_seq(n: int, frac: float, target_sum: float) -> np.ndarray:
    """All mass on the first `frac` part of the vector (uniform within the block)."""
    block_len = max(1, int(np.round(frac * n)))
    a = np.zeros(n, dtype=np.float64)
    a[:block_len] = target_sum / block_len
    return a


def _random_simplex_seq(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (often lower the peak)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Long projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent_long(a0: np.ndarray,
                               target_sum: float,
                               start_time: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum and linear step decay."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    # parameters
    gamma = 0.95
    start_factor = 0.5   # step size when t≈0 (as a fraction of target_sum)
    end_factor = 0.02    # step size when t≈deadline

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = float(conv.max())

        # --- sub‑gradient of the max‑convolution --------------------
        # indices where the maximum is attained (within 1e-12 tolerance)
        active = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += a[k - idx]
        grad *= 2.0  # each unordered pair appears twice

        # momentum update
        momentum = gamma * momentum + grad

        # step‑size schedule (linear decay)
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 0.0
        step_factor = start_factor * (1 - frac) + end_factor * frac

        norm_m = np.linalg.norm(momentum)
        if norm_m > 0:
            step = step_factor * target_sum / (norm_m + 1e-12)
            a = a - step * momentum

        a = _project_to_simplex(a, target_sum)

        # keep the best seen so far
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny jitter to escape shallow plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple Metropolis mass‑transfer hill‑climb (very cheap per iteration)
# ----------------------------------------------------------------------


def _local_random_moves(a0: np.ndarray,
                        target_sum: float,
                        start_time: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Random donor‑receiver moves with a linear cooling schedule."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 0.0
        T = 0.5 * (1 - frac) + 1e-4 * frac   # start ≈0.5, cool to ≈1e‑4

        # pick donor (large mass) and receiver (small mass)
        donor = rng.integers(n)
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        # propose moving a fraction of donor's mass
        max_delta = a[donor] * 0.4          # at most 40 % of donor's mass
        delta = max_delta * rng.random() * (T / 0.5)
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = _project_to_simplex(a_new, target_sum)

        new_val = _max_convolution_fft(a_new)

        # Metropolis acceptance
        if new_val < best_val or rng.random() < np.exp(-(new_val - best_val) / max(T, 1e-12)):
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine builds a set of parametric seeds, picks the best one,
    refines it with a long projected sub‑gradient descent and a final
    cheap random hill‑climb, applies a few simple reorderings and
    returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    # --------------------------------------------------------------
    # Choose problem dimension (larger n ⇒ richer search space)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 500:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation – a rich collection of parametric families
    # --------------------------------------------------------------
    seeds = []

    # power‑law family (fine grid)
    for p in np.linspace(0.0, 4.0, 81):
        seeds.append(_power_law_seq(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0, 0.2, 81):
        seeds.append(_exponential_seq(n, float(lam), target_sum))

    # block family (uniform mass on a prefix)
    for frac in np.linspace(0.02, 1.0, 50):
        seeds.append(_block_seq(n, float(frac), target_sum))

    # a few random points on the simplex
    for _ in range(12):
        seeds.append(_random_simplex_seq(n, target_sum, rng))

    # warm‑start from the previous run, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # --------------------------------------------------------------
    # Evaluate all seeds and keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in seeds:
        cur = _max_convolution_fft(seq)  # because Σa = √(2n) ⇒ objective = max conv
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Best seed after parametric sweep – max conv = {best_score:.6f}")

    # --------------------------------------------------------------
    # Long sub‑gradient descent (majority of the remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        sub_dead = time.time() + max(5.0, remaining * 0.86)   # keep a margin for final polish
        seq_sg, score_sg = _subgradient_descent_long(best_seq, target_sum, start_time, sub_dead)
        if score_sg < best_score:
            best_score = score_sg
            best_seq = seq_sg.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved – max conv = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional cheap random local moves (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        move_dead = deadline
        seq_mt, score_mt = _local_random_moves(best_seq, target_sum, start_time, move_dead)
        if score_mt < best_score:
            best_score = score_mt
            best_seq = seq_mt.copy()
            print(f"[{time.time():.2f}] Random moves improved – max conv = {best_score:.6f}")

    # --------------------------------------------------------------
    # Try a few simple reorderings – they sometimes lower the peak
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq)
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] After reordering – final max conv = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for potential warm‑starts in a future run
    globals()["height_sequence_1"] = final_seq

    # ------------------------------------------------------------------
    # Sanity check with the provided evaluator (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
