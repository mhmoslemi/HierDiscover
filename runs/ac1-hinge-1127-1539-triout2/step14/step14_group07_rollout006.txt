"""
Step‑function optimizer for the “max self‑convolution” challenge.

Strategy
--------
1. Warm‑start from the best known construction (`height_sequence_1`) or a
   simple power‑law shape.
2. Sub‑gradient descent on the exact max‑norm (the gradient of each active
   convolution peak is computed analytically).  The iterate is projected onto
   the simplex Σa = √(2 n), a_i ≥ 0.
3. Stochastic hill‑climbing: repeatedly move a random fraction of mass from a
   heavy coefficient to a light one; accept only improving moves.  The full
   convolution is recomputed with an FFT after each accepted move.
4. Frank‑Wolfe style LP refinement (identical to the previous baseline but
   with tighter tolerances and more iterations).
5. Enforce symmetry after each major phase – the optimal step‑function is
   expected to be (approximately) symmetric, which reduces the peak.
6. Trim trailing zeros and return the final list.

All phases respect the global deadline (default 1000 s) and print progress.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(b)/(sum(a)^2).  For Σa = √(2 n) the denominator is 2·n,
    hence the score equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
#  Symmetrisation helper
# ----------------------------------------------------------------------


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce (approximate) symmetry a_i ≈ a_{n‑1‑i}."""
    b = (a + a[::-1]) * 0.5
    return b


# ----------------------------------------------------------------------
#  Sub‑gradient descent on the exact max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_k (a*a)_k.
    Uses the exact sub‑gradient (sum of 2·a shifted by each active peak).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        # active peak indices (within tiny tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # compute sub‑gradient
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            segment = a[lo:hi + 1][::-1]          # a_{k‑j} for j∈[lo,hi]
            g[lo:hi + 1] += 2.0 * segment

        # step size: diminishing 1/√t scaled to keep updates moderate
        norm_g = np.linalg.norm(g) + 1e-12
        step = target_sum / norm_g / np.sqrt(it + 1)

        a_new = a - step * g
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional symmetrisation (helps the problem structure)
        if it % 100 == 0:
            a_new = _symmetrize(a_new)
            a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val - 1e-12:
            best_val = val
            best = a_new.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        a = a_new
        it += 1

    return best, best_val


# ----------------------------------------------------------------------
#  Stochastic hill‑climbing (random mass transfers)
# ----------------------------------------------------------------------


def _random_hill_climbing(a0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from a heavy coefficient to a light one.
    Accept only improving moves (deterministic hill‑climb).  Full FFT is used
    after each accepted move to keep the evaluation exact.
    """
    n = a0.shape[0]
    a = a0.copy()
    best = a.copy()
    best_val = _score(best)

    # pre‑compute a probability distribution that favours heavy donors
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        # donors weighted by current mass (avoid zero‑mass donors)
        donor_probs = best / best.sum()
        donor = rng.choice(n, p=donor_probs)

        # receivers weighted towards low mass (inverse weighting)
        inv = 1.0 / (best + 1e-12)
        receiver_probs = inv / inv.sum()
        receiver = rng.choice(n, p=receiver_probs)

        if donor == receiver:
            it += 1
            continue

        # move up to 20 % of the donor's mass
        delta = rng.random() * 0.20 * best[donor]
        if delta <= 0.0:
            it += 1
            continue

        a_candidate = best.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        # optional symmetrisation every few steps
        if it % 250 == 0:
            a_candidate = _symmetrize(a_candidate)
            a_candidate = _project_to_simplex(a_candidate, target_sum)

        val = _score(a_candidate)
        if val < best_val - 1e-12:
            best_val = val
            best = a_candidate
            print(f"[{time.time():.2f}] Random climb it={it:6d} new best={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
#  LP‑based Frank‑Wolfe refinement (same idea as the baseline)
# ----------------------------------------------------------------------


def _lp_iterative_refinement(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             epsilon_rel: float = 0.02,
                             step: float = 0.20,
                             max_iters: int = 150) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style updates using a tiny linear programme.
    In each iteration we solve:
        max Σ g_j   s.t.   f * g ≤ M   (only on a few active indices)
    The solution g is scaled to Σg = target_sum and blended with f.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    eps = epsilon_rel
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        # indices whose value is close to the peak
        idx = np.where(conv >= M - eps * M)[0]
        if idx.size == 0:
            break
        # keep at most 30 most critical constraints
        if idx.size > 30:
            idx = idx[np.argsort(-conv[idx])[:30]]

        C = idx.size
        n = a.size
        A = np.zeros((C, n), dtype=np.float64)

        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            A[row, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

        b = np.full(C, M, dtype=np.float64)

        try:
            # maximise Σ g_j  <=>  minimise -Σ g_j
            res = linprog(c=-np.ones(n),
                          A_ub=A,
                          b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception as e:
            print(f"[{time.time():.2f}] LP exception: {e}")
            break

        if not res.success:
            # relax epsilon and retry
            eps = min(eps * 2.0, 0.1)
            it += 1
            continue

        g = res.x
        if g.sum() == 0.0:
            break
        # scale g to the required simplex
        g = g * (target_sum / g.sum())

        # Frank‑Wolfe update + projection
        a_new = (1.0 - step) * a + step * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_val = _score(a_new)
        if new_val < best_val - 1e-12:
            best_val = new_val
            best = a_new.copy()
            a = a_new
            eps = max(eps * 0.8, 1e-5)
            print(f"[{time.time():.2f}] LP it={it:3d} eps={eps:.2e} score={best_val:.6f}")
        else:
            step = max(step * 0.6, 0.01)
            eps = min(eps * 1.5, 0.05)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
#  Utility – trim trailing near‑zero entries
# ----------------------------------------------------------------------


def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]
    if nz.size == 0:
        return [0.0]
    return a[: nz[-1] + 1].tolist()


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising
    evaluate_sequence(...).  Returns the best sequence found as a list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives more freedom but costs more time.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) → score = max self‑convolution

    # ------------------------------------------------------------------
    # Warm‑start (use previous best if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if a0.size != n:
            # linear interpolation if the old length differs
            old_x = np.linspace(0.0, 1.0, a0.size)
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = _project_to_simplex(a0, target_sum)
    else:
        # fallback – a simple power‑law family (p≈0.6 works well)
        a0 = _power_law_sequence(n, 0.6, target_sum)

    best = a0.copy()
    best_val = _score(best)
    print(f"[{time.time():.2f}] Warm‑start score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Sub‑gradient descent on the exact max‑norm
    # ------------------------------------------------------------------
    sub_end = min(deadline, start_time + 0.25 * budget_s)   # ~250 s for a 1000 s budget
    best, best_val = _subgradient_descent(best, target_sum, sub_end)
    print(f"[{time.time():.2f}] After sub‑gradient best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Stochastic hill‑climbing (many random moves)
    # ------------------------------------------------------------------
    if time.time() < deadline - 50.0:   # reserve time for later refinements
        hill_end = deadline - 40.0
        best, best_val = _random_hill_climbing(best, target_sum, hill_end)
        print(f"[{time.time():.2f}] After random climbing best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP‑based Frank‑Wolfe refinement (tighten the peak)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        lp_end = deadline - 5.0
        best, best_val = _lp_iterative_refinement(best,
                                                  target_sum,
                                                  lp_end,
                                                  epsilon_rel=0.015,
                                                  step=0.15,
                                                  max_iters=200)
        print(f"[{time.time():.2f}] After LP refinement best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final symmetrisation & projection
    # ------------------------------------------------------------------
    best = _symmetrize(best)
    best = _project_to_simplex(best, target_sum)

    final_seq = _trim_sequence(best)

    # ------------------------------------------------------------------
    # Optional final verification (does not affect the returned list)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
