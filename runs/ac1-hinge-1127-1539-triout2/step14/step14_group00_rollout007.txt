def _adam_step(a, grad, m, v, t, lr, beta1=0.9,beta2=0.999, eps=1e-8):
    m = beta1*m + (1-beta1)*grad
    v = beta2*v + (1-beta2)*(grad**2)
    m_hat = m/(1-beta1**t)
    v_hat = v/(1-beta2**t)
    step = lr * m_hat/(np.sqrt(v_hat) + eps)
    a_new = a - step

"""
Hybrid optimisation for the step‑function max‑convolution problem.

The optimiser works in a clean staged fashion:

1. **Problem size** – the length `n` grows with the time budget (up to
   12 000).  The ℓ¹‑norm is fixed to `√(2 n)` so the objective becomes
   simply `max(a*a)`.

2. **Seed generation** – a diverse collection (power‑law, exponential,
   uniform, random Dirichlet, previous best, binary spikes) is created
   and projected onto the simplex.

3. **Projected Differential Evolution** – a modest population evolves
   while always staying on the feasible simplex.  This quickly finds a
   good region of the search space.

4. **Continuation of smooth gradient descent** – we minimise a
   log‑sum‑exp surrogate of the max‑convolution.  The smoothing
   parameter `α` is increased linearly from 2 to 30, while a simple
   simulated‑annealing acceptance rule allows occasional uphill moves.
   The step size adapts to the gradient norm.

5. **Deterministic mass transfer** – a greedy move of mass from the
   “donor” index that contributes most to the current peak towards a
   low‑contributing “receiver”, with a Metropolis schedule.

6. **Projected sub‑gradient descent** – a short exact‑max sub‑gradient
   phase refines the solution.

7. **Random re‑ordering** – a few random permutations are tried;
   re‑ordering can dramatically change the autocorrelation peak.

8. **Upsampling & polishing** – the best sequence is linearly
   interpolated to a larger dimension (up to 12 000) and a short
   continuation‑gradient run is performed on the longer vector.

9. **Trimming & return** – trailing near‑zero entries are removed,
   the result is stored in the global `height_sequence_1` and returned.

All stages respect the global time budget (default 1000 s).  The code
uses only NumPy and the provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective: 2·n·max(a*a)/(sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _uniform_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.full(n, target_sum / n, dtype=np.float64)
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _binary_spike_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """few large spikes, rest very small (still non‑negative)."""
    # number of spikes between 3% and 10% of n
    k = max(1, int(n * rng.uniform(0.03, 0.10)))
    # generate spike magnitude uniformly
    spike_val = target_sum * rng.uniform(0.5, 0.9) / k
    a = np.full(n, (target_sum - k * spike_val) / (n - k), dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = spike_val
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after each step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth surrogate gradient descent with continuation & simulated annealing
# ----------------------------------------------------------------------


def _smooth_continuation(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         start_alpha: float = 2.0,
                         end_alpha: float = 30.0,
                         start_temp: float = 0.2,
                         end_temp: float = 1e-5,
                         init_step_factor: float = 0.7,
                         decay: float = 0.97) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a*a) using projected
    gradient descent.  ``α`` (the smoothness) grows linearly from
    ``start_alpha`` to ``end_alpha``.  An annealed Metropolis rule
    permits occasional uphill moves.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    start_time = time.time()
    step_factor = init_step_factor

    while time.time() < deadline:
        t = (time.time() - start_time) / max(deadline - start_time, 1e-12)
        alpha = start_alpha + (end_alpha - start_alpha) * t
        temp = start_temp * (end_temp / start_temp) ** t

        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        # smooth weights
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        # gradient of the surrogate (see analysis)
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # slice of length n
        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)
        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            step_factor = min(step_factor * 1.02, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                target_sum: float,
                                deadline: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / max(deadline - start_time, 1e-12)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where convolution peak occurs
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # rollback
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            # small jitter to escape plateaus
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Random re‑ordering search
# ----------------------------------------------------------------------


def _random_reorder_search(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           trials: int = 30) -> tuple[np.ndarray, float]:
    """Try a few random permutations of the vector and keep the best."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline and trials > 0:
        perm = rng.permutation(n)
        cand = a[perm]
        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
        trials -= 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimises a non‑negative step function to minimise the max‑convolution
    bound.  The routine respects the time budget (default 1000 s) and
    stores the best sequence globally under ``height_sequence_1``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem dimension
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 600:
        n = 10000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.15, 1.2, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(5e-4, 2.5e-2, 7):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # uniform
    seeds.append(_uniform_sequence(n, target_sum))

    # random Dirichlet‑like points
    for _ in range(10):
        seeds.append(_random_simplex(n, target_sum, rng))

    # binary spikes
    for _ in range(6):
        seeds.append(_binary_spike_sequence(n, target_sum, rng))

    # previous best (warm‑start)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little to increase diversity
    jittered: list[np.ndarray] = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # --------------------------------------------------------------
    # Initialise DE population
    # --------------------------------------------------------------
    pop_size = max(35, n // 250)          # about 35‑50 individuals
    population: list[np.ndarray] = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())
    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # --------------------------------------------------------------
    # Stage 1 – Differential Evolution
    # --------------------------------------------------------------
    de_time = min(0.30 * budget_s, 260.0)
    de_deadline = min(time.time() + de_time, deadline - 250.0)
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Smooth continuation gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 200.0:
        grad_deadline = time.time() + min(220.0, deadline - time.time() - 150.0)
        seq_g, score_g = _smooth_continuation(best_seq, target_sum, grad_deadline)
        if score_g < best_score:
            best_score, best_seq = score_g, seq_g.copy()
            print(f"[{time.time():.2f}] After smooth‑cont. best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Deterministic mass transfer
    # --------------------------------------------------------------
    if time.time() < deadline - 140.0:
        mt_dead = time.time() + min(120.0, deadline - time.time() - 110.0)
        seq_mt, score_mt = _deterministic_mass_transfer(best_seq, target_sum, mt_dead)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – Sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sg_dead = time.time() + min(80.0, deadline - time.time() - 70.0)
        seq_sg, score_sg = _subgradient_descent(best_seq, target_sum, sg_dead)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – Random re‑ordering search
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:
        perm_dead = time.time() + min(50.0, deadline - time.time() - 45.0)
        seq_perm, score_perm = _random_reorder_search(best_seq, target_sum, perm_dead, trials=50)
        if score_perm < best_score:
            best_score, best_seq = score_perm, seq_perm.copy()
            print(f"[{time.time():.2f}] After random reorder best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 6 – Upsampling & polishing (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0 and n < 18000:
        up_len = min(18000, n * 2)
        up_seq = _resize_sequence(best_seq, up_len)
        up_target = np.sqrt(2.0 * up_len)
        up_seq = _project_to_simplex(up_seq, up_target)

        up_dead = time.time() + min(25.0, deadline - time.time() - 20.0)
        up_seq, up_score = _smooth_continuation(up_seq, up_target, up_dead,
                                                start_alpha=4.0,
                                                end_alpha=35.0,
                                                start_temp=0.2,
                                                end_temp=1e-4)
        if up_score < best_score:
            best_seq, best_score = up_seq, up_score
            n = up_len
            target_sum = up_target
            print(f"[{time.time():.2f}] Upsampled polishing improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final re‑ordering attempts (deterministic patterns)
    # --------------------------------------------------------------
    def _alternating_order(seq: np.ndarray) -> np.ndarray:
        sorted_vals = np.sort(seq)[::-1]
        n_ = seq.shape[0]
        out = np.empty_like(seq)
        left, right = 0, n_ - 1
        place_left = True
        for v in sorted_vals:
            if place_left:
                out[left] = v
                left += 1
            else:
                out[right] = v
                right -= 1
            place_left = not place_left
        return out

    def _even_spread_order(seq: np.ndarray) -> np.ndarray:
        n_ = seq.shape[0]
        sorted_vals = np.sort(seq)[::-1]
        step = max(1, int(round(np.sqrt(n_))))
        while np.gcd(step, n_) != 1:
            step = (step + 1) % n_ or 1
        out = np.empty_like(seq)
        for i, v in enumerate(sorted_vals):
            out[(i * step) % n_] = v
        return out

    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1],
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After deterministic reorder best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and finalise
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store globally for warm‑starts in future calls
    globals()["height_sequence_1"] = final_seq

    # optional evaluation print‑out
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
