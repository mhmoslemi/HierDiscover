"""
Hybrid evolutionary optimizer for the non‑negative step‑function benchmark.

The pipeline:
1.  Parameter sweep over a few analytic families (power‑law, exponential,
    Gaussian, raised‑cosine) to obtain a decent shape on a modest size
    (≈8 000).  All candidates are projected onto the ℓ¹‑simplex (sum=√(2n))
    because scaling does not affect the benchmark score.

2.  A longer annealed soft‑max gradient descent (projected GD) refines the
    best shape.

3.  A small linear‑program refinement (active‑set) improves the ℓ¹‑norm
    under the current maximal convolution constraints.

4.  Upscaling (linear interpolation) to a larger support (≈20 000) and
    a thorough “spread‑order’’ step‑permutation search (including the
    golden‑ratio step) to separate large values.

5.  Multi‑phase exact sub‑gradient descent with a tightening tolerance
    (0.95 → 0.99 → 0.999) to drive the maximal convolution down.

6.  Random permutation search, local Hill‑climb swaps and a final
    simulated‑annealing mass‑transfer pass.

All updates respect the non‑negativity and ℓ¹‑simplex constraints,
and after each stage the best sequence is kept.  The function returns
the best sequence found before the 1000 s budget expires.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# 1️⃣  Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# 2️⃣  ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 3️⃣  Analytic families of candidate sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _raised_cosine_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(0.5·(1‑cos(2πx)))^p"""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (0.5 * (1 - np.cos(2 * np.pi * x))) ** p
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# 4️⃣  Re‑ordering helpers (to scatter large entries)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _golden_ratio_spread(seq: np.ndarray) -> np.ndarray:
    """Spread using the golden‑ratio step (⌊ϕ·n⌋, ϕ≈0.618)."""
    n = seq.shape[0]
    step = int(round((math.sqrt(5) - 1) / 2 * n))
    if step <= 0:
        step = 1
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _best_spread_order(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Try many coprime step permutations and keep the best."""
    n = seq.shape[0]
    best_seq = seq.copy()
    best_val = _max_convolution_fft(seq)
    rng = np.random.default_rng()

    # a small deterministic set
    candidates = [_alternating_order(seq), _even_spread_order(seq), _golden_ratio_spread(seq)]
    for cand in candidates:
        val = _max_convolution_fft(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # random coprime steps (up to ~150 attempts)
    attempts = 0
    while time.time() < deadline and attempts < 200:
        step = rng.integers(1, n)
        if math.gcd(step, n) != 1:
            continue
        sorted_vals = np.sort(seq)[::-1]
        perm = np.empty_like(seq)
        for i, v in enumerate(sorted_vals):
            perm[(i * step) % n] = v
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
        attempts += 1

    print(f"[{time.time():.2f}] Best spread order max={best_val:.6f}")
    return best_seq

# 5️⃣  Gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 6️⃣  Linear‑program refinement (active‑set LP)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               max_iters: int,
               deadline: float,
               K: int = 12,
               mix_factor: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Small LP that, given the current a, finds a g with large ℓ¹‑norm
    such that a∗g ≤ M on the K most critical convolution positions.
    The new a is a convex combination (1‑t)·a + t·g.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        print("SciPy unavailable → skipping LP refinement.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        # active set: top‑K indices
        if K >= conv.size:
            active = np.arange(conv.size, dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # Build A_ub * g ≤ M
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            A[row, idx] = a[k - idx]

        # Solve LP: maximise sum(g)  ⇔ minimise -sum(g)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        g = g * (target_sum / g.sum())

        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# 7️⃣  Exact sub‑gradient descent (multi‑phase)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995,
                         tol: float = 0.99) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _subgradient_descent_adapt(a0: np.ndarray,
                               deadline: float,
                               target_sum: float) -> tuple[np.ndarray, float]:
    """
    Run three sub‑gradient phases with increasingly tight tolerance.
    """
    now = time.time()
    total = deadline - now
    # allocate time
    t1 = total * 0.30
    t2 = total * 0.40
    t3 = total - t1 - t2

    # Phase 1 – tol=0.95
    phase_end = now + t1
    a1, v1 = _subgradient_descent(a0, phase_end, target_sum,
                                  step_factor=0.6, gamma=0.995, tol=0.95)

    # Phase 2 – tol=0.99
    now = time.time()
    phase_end = now + t2
    a2, v2 = _subgradient_descent(a1, phase_end, target_sum,
                                  step_factor=0.6, gamma=0.995, tol=0.99)

    # Phase 3 – tol=0.999
    now = time.time()
    phase_end = now + t3
    a3, v3 = _subgradient_descent(a2, phase_end, target_sum,
                                  step_factor=0.6, gamma=0.995, tol=0.999)

    # pick the best
    best_seq, best_val = a1, v1
    if v2 < best_val:
        best_seq, best_val = a2, v2
    if v3 < best_val:
        best_seq, best_val = a3, v3
    return best_seq, best_val


# ----------------------------------------------------------------------
# 8️⃣  Permutation searches
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb it={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# 9️⃣  Greedy mass‑transfer with simulated‑annealing
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 10️⃣  Utility to compute the benchmark objective (for internal use)
# ----------------------------------------------------------------------


def _compute_objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv) / (sum(seq)²)."""
    n = seq.shape[0]
    maxb = _max_convolution_fft(seq)
    s = seq.sum()
    if s <= 1e-12:
        return np.inf
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# 11️⃣  Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The function returns the best sequence discovered within the time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1️⃣  Analytic pool on a moderate size (≈8 000)
    # ------------------------------------------------------------------
    n0 = 8000
    target0 = math.sqrt(2.0 * n0)

    pool: List[np.ndarray] = []

    # power‑law
    for p in np.linspace(0.2, 2.5, 20):
        pool.append(_power_law_sequence(n0, float(p), target0))
    # exponential
    for lam in np.linspace(0.001, 0.03, 20):
        pool.append(_exponential_sequence(n0, float(lam), target0))
    # Gaussian
    for sigma_frac in np.linspace(0.05, 0.5, 20):
        sigma = sigma_frac * n0
        pool.append(_gaussian_sequence(n0, sigma, target0))
    # raised‑cosine power
    for p in np.linspace(0.5, 3.0, 15):
        pool.append(_raised_cosine_power_sequence(n0, float(p), target0))
    # random simplexes
    for _ in range(20):
        pool.append(_random_simplex(n0, target0, rng))

    # reuse previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        pool.append(_resize_sequence(prev, n0))

    # jittered variants of existing pool members (10 extra)
    for seq in list(pool):
        if time.time() >= deadline - 120:
            break
        jitter = rng.normal(scale=target0 * 0.025, size=n0)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target0)
        pool.append(seq_j)

    best_seq = None
    best_obj = np.inf

    for seq in pool:
        if time.time() >= deadline - 120:
            break
        obj = _compute_objective(seq)
        if obj < best_obj:
            best_obj = obj
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] After analytic sweep best obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Annealed gradient descent on the best seed
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 45.0:
        gd_deadline = time.time() + min(30.0, remaining - 15.0)
        best_seq, gd_val = _annealed_gradient_descent(best_seq, gd_deadline, target0,
                                                      max_beta=9e5, step_factor=0.9, gamma=0.98)
        if gd_val < best_obj:
            best_obj = gd_val
            print(f"[{time.time():.2f}] GD improvement → obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  LP refinement on the modest size
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        lp_deadline = time.time() + min(15.0, deadline - time.time() - 20.0)
        best_seq, lp_val = _lp_refine(best_seq, target0, max_iters=25,
                                      deadline=lp_deadline, K=14, mix_factor=0.12)
        if lp_val < best_obj:
            best_obj = lp_val
            print(f"[{time.time():.2f}] LP refinement → obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Upscale to a larger problem size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120:
        n_big = 25000
    elif remaining > 80:
        n_big = 20000
    elif remaining > 50:
        n_big = 15000
    else:
        n_big = n0

    target_big = math.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_seq if best_seq is not None else _power_law_sequence(n0, 1.0, target0),
                               n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # Apply a thorough spread ordering
    seq_big = _best_spread_order(seq_big, deadline - 5.0)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems; start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Multi‑phase exact sub‑gradient descent (tightening tolerances)
    # ------------------------------------------------------------------
    if time.time() < deadline - 40.0:
        sub_deadline = deadline - 40.0
        seq_big, sub_val = _subgradient_descent_adapt(seq_big, sub_deadline, target_big)
        if sub_val < best_obj:
            best_obj = sub_val
            print(f"[{time.time():.2f}] Sub‑grad descent → obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Random permutation search
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        perm_deadline = time.time() + min(8.0, deadline - time.time() - 30.0)
        seq_big, perm_val = _permutation_search(seq_big, trials=120, deadline=perm_deadline)
        if perm_val < best_obj:
            best_obj = perm_val
            print(f"[{time.time():.2f}] Random perm best={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Local Hill‑climb on permutations
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        hill_deadline = time.time() + min(6.0, deadline - time.time() - 20.0)
        seq_big, hill_val = _local_permutation_hillclimb(seq_big, hill_deadline)
        if hill_val < best_obj:
            best_obj = hill_val
            print(f"[{time.time():.2f}] Hill‑climb best={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣  Final greedy mass‑transfer with simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = time.time() + min(5.0, deadline - time.time() - 5.0)
        seq_big, mt_val = _mass_transfer_simulated_annealing(seq_big,
                                                             mt_deadline,
                                                             target_big,
                                                             start_temp=0.3,
                                                             end_temp=1e-5,
                                                             low_frac=0.05)
        if mt_val < best_obj:
            best_obj = mt_val
            print(f"[{time.time():.2f}] Final MT best={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 9️⃣  Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
