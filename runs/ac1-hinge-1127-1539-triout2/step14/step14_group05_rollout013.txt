def _adam_subgradient(a0, deadline, target_sum, step_size=0.5, beta1=0.9, beta2=0.999, eps=1e-8, tol=0.99):
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb*tol).astype(np.float64)
        # gradient using I mask
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft*rev_fft, n=size)[:len(I)+len(a_rev)-1]
        grad = 2.0*grad_full[len(a_rev)-1:len(a_rev)-1+len(a)]

        it += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        # step = step_size * m_hat / (np.sqrt(v_hat) + eps)
        step = step_size * m_hat / (np.sqrt(v_hat) + eps)

        # optional line search: if new val not improvement reduce step factor
        # we directly apply step then project
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)
        cur_val = _max_convolution_fft(a_new)
        # Accept if improves else keep old maybe reduce step size gradually
        if cur_val <= best_val:
            best_val = cur_val
            best_a = a_new.copy()
            a = a_new
        else:

            a = np.maximum(a + jitter, 0.0)

            # recompute conv later next loop

    return best_a, best_val

import time, math, numpy as np
from typing import List

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return float(conv.max())

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

def _project_to_simplex(v, s):
    # Euclidean projection onto simplex >=0 sum = s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv/np.arange(1, len(v)+1) > 0)[0]
    if rho.size==0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

        # Build constraint matrix rows: each active conv index k -> coefficients 2*a_{k-i} for d_i.
        # Actually we need constraint: (a+δ)*(a+δ) <= M (target). For linearization: conv_k + grad_k⋅δ <= M.
        # So row coefficients are 2*a_{k-i}
        rows = []
        for k in active:
            row = np.zeros(n)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # indices i where a_i contributes
            # For i in [i_start,i_end], coefficient = 2*a_{k-i}
            idx = np.arange(i_start, i_end+1)
            row[idx] = 2.0 * a[k - idx]
            rows.append(row)
        A = np.vstack(rows)
        b = np.full(A.shape[0], M)

        # However linprog only handles inequality; we can incorporate by adding two opposite inequalities.
        # Use A_eq, b_eq.
        A_eq = np.ones((1, n))
        b_eq = np.array([0.0])
        # objective: minimize t (max value after step). Instead we directly minimize M'? Actually we can set variable t and add constraints conv_k + grad_k⋅δ <= t.
        # Simpler: we can try to solve for δ that reduces max by directly solving LP that maximizes improvement in M: minimize M_new = M + max_k grad_k⋅δ (approx). Hard.
        # For simplicity, simply solve feasibility of decreasing M by small epsilon via LP: find δ such that conv_k + grad_k⋅δ <= M - eps, sum δ =0, δ_i >= -a_i.

"""
Hybrid optimizer for the step‑function benchmark.

Key ideas
---------
1. Build a modest‑size seed from several parametric families and the
   best sequence found in previous runs (height_sequence_1).
2. Upscale the seed to a large support (≈20 k–40 k) and disperse large
   coefficients with a deterministic permutation.
3. Run a very long exact sub‑gradient descent on the max‑convolution
   (the true objective).  The mask for the sub‑gradient is gradually
   tightened so that the optimisation focuses on the true peak.
4. Finish with cheap random permutation searches, a local hill‑climb
   and a short simulated‑annealing mass‑transfer phase.
5. Trim trailing zeros and return the result.  The whole pipeline
   respects non‑negativity and the ℓ¹‑norm constraint (the objective
   equals max (a∗a) when the sum is √(2n)).

The algorithm spends the bulk of the budget (≈80 %) on sub‑gradient
descent at the final problem size – the part that yields the biggest
improvements on the benchmark.
"""

import time
import math
import numpy as np
from typing import List


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (several parametric families)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Sinusoid peaking in the centre."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple deterministic permutations that spread large values
# ----------------------------------------------------------------------
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995,
                         tol_start: float = 0.99,
                         tol_end: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent with momentum.
    The tolerance for the active set is increased linearly over time.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    t0 = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()

        # linearly increase the tolerance
        elapsed = time.time() - t0
        total = max(deadline - t0, 1e-12)
        cur_tol = tol_start + (tol_end - tol_start) * (elapsed / total)

        I = (conv >= maxb * cur_tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a = a - lr * momentum

        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} tol={cur_tol:.4f} score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------
def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.25,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random permutation search & local hill‑climb
# ----------------------------------------------------------------------
def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val

    if time.time() < deadline - 30:
        perm_deadline = time.time() + min(8.0, deadline - time.time() - 22.0)
        seq_big, _ = _permutation_search(seq_big, trials=100, deadline=perm_deadline)
        print(f"[{time.time():.2f}] After random perm max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 5. Local hill‑climb on permutations
    # --------------------------------------------------------------
    if time.time() < deadline - 20:
        hill_deadline = time.time() + min(6.0, deadline - time.time() - 12.0)
        seq_big, _ = _local_permutation_hillclimb(seq_big, hill_deadline)
        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 6. Final greedy mass‑transfer (simulated‑annealing)
    # --------------------------------------------------------------
    if time.time() < deadline - 10:
        mt_deadline = time.time() + min(10.0, deadline - time.time() - 5.0)
        seq_big, mt_score = _mass_transfer_simulated_annealing(seq_big,
                                                               mt_deadline,
                                                               target_big,
                                                               start_temp=0.2,
                                                               end_temp=1e-5,
                                                               low_frac=0.04)
        if mt_score < best_val:
            best_val = mt_score
            best_seq = seq_big.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Trim trailing near‑zero entries and return the result
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq   # for possible reuse

    # Log the final benchmark value (optional)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as e:
        print(f"Evaluation error: {e}")

    return final_seq
