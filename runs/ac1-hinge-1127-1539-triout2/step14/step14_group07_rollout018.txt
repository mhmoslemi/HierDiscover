"""
Propose a step‐function (non‑negative heights) that minimises the
`evaluate_sequence` score.

Algorithm
---------

0️⃣  Choose a problem size `n` (larger n → finer discretisation,
    but still fast enough for a 1000 s budget).

1️⃣  Warm‑start: a few parametric families (power‑law, tilted power‑law,
    Gaussian, triangular) and optionally re‑use the previous best
    construction (`height_sequence_1`).  Keep the best of these.

2️⃣  Adam‑style projected gradient descent on a smooth soft‑max
    approximation of the max‑convolution.  Several restarts are used.

3️⃣  **Precise peak‑mass transfer** – a deterministic greedy improvement.
    For the current peak `k = argmax(a*a)` we pick a small set of
    “donor’’ indices (large product a_i·a_{k‑i}) and a set of
    “receiver’’ indices (small partner a_{k‑j}).  For every donor/receiver
    pair we compute the largest admissible mass `δ` that does not raise any
    other convolution entry above the current peak.  The pair giving the
    smallest new peak is applied.  The convolution is updated with an
    O(n) vectorised formula, so thousands of such moves fit easily
    into the time budget.

4️⃣  (Optional) a tiny LP / Frank‑Wolfe step on the active constraints
    (as in the reference code).

5️⃣  Final high‑β Adam polishing.

All phases respect the overall deadline, print intermediate scores,
and finally return the trimmed sequence (removing trailing near‑zero
entries).  The code runs on a single CPU and uses only NumPy and SciPy.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(b)/(sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    # symmetric triangle, peak in the middle
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = inc[:-1][::-1]
    a = np.concatenate([inc, dec])
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Precise peak‑mass transfer (deterministic greedy refinement)
# ----------------------------------------------------------------------


def _precise_peak_mass_transfer(a0: np.ndarray,
                                target_sum: float,
                                deadline: float,
                                max_iters: int = 20000,
                                K: int = 10) -> tuple[np.ndarray, float]:
    """
    Greedy optimisation of the max‑convolution peak.
    For the current peak index k we examine at most K
    donors (largest a_i·partner_i) and K receivers (smallest partner_i)
    and move a feasible amount of mass from donor to receiver.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_score = _score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # locate peak and second‑largest value
        k = int(np.argmax(b))
        max_val = b[k]
        # second max (ignore all occurrences of the max)
        b_tmp = b.copy()
        b_tmp[b_tmp == max_val] = -np.inf
        second_max = b_tmp.max()
        if max_val - second_max < 1e-12:
            break

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k - idx_range]                     # a_{k‑i}

        # donor candidates – largest a_i·partner_i
        donor_scores = a[idx_range] * partner
        nd = min(K, donor_scores.size)
        donor_idx_local = np.argpartition(-donor_scores, nd - 1)[:nd]
        donor_idxs = idx_range[donor_idx_local]

        # receiver candidates – smallest partner values
        receiver_loc = np.argpartition(partner, K - 1)[:K]
        receiver_idxs = idx_range[receiver_loc]

        # keep the best move among all donor/receiver pairs
        best_new_max = max_val
        best_move = None
        a_old = a.copy()                     # keep a snapshot for diff computation

        # temporary buffer for the diff vector
        diff = np.zeros_like(b)

        for donor in donor_idxs:
            w_donor = a[k - donor]
            for receiver in receiver_idxs:
                if receiver == donor:
                    continue
                w_receiver = a[k - receiver]
                if w_donor <= w_receiver:
                    continue

                diff.fill(0.0)
                diff[receiver:receiver + n] = a_old
                diff[donor:donor + n] -= a_old

                mask_pos = diff > 0
                if np.any(mask_pos):
                    # bound to keep every entry ≤ current max_val
                    bound_vals = (max_val - b[mask_pos]) / (2.0 * diff[mask_pos])
                    bound_vals = bound_vals[bound_vals > 0.0]
                    if bound_vals.size == 0:
                        delta_bound = 0.0
                    else:
                        delta_bound = bound_vals.min()
                else:
                    delta_bound = np.inf

                # also limited by donor mass
                delta_bound = min(delta_bound, a_old[donor])

                # we would like to reduce the peak at least to second_max
                delta_needed = (max_val - second_max) / (2.0 * (w_donor - w_receiver))
                delta = min(delta_bound, 0.99 * delta_needed)

                if delta <= 1e-12:
                    continue

                # evaluate the resulting peak (exact via temporary array)
                b_cand = b.copy()
                b_cand[donor:donor + n] -= 2.0 * delta * a_old
                b_cand[receiver:receiver + n] += 2.0 * delta * a_old
                b_cand[2 * donor] += delta ** 2
                b_cand[2 * receiver] += delta ** 2
                b_cand[donor + receiver] -= 2.0 * delta ** 2
                cand_max = b_cand.max()

                if cand_max < best_new_max - 1e-12:
                    best_new_max = cand_max
                    best_move = (donor, receiver, delta, b_cand)

        if best_move is None:
            break

        donor, receiver, delta, b_new = best_move
        # apply the move (use a_old for the slice updates)
        a_old = a.copy()
        a[donor] -= delta
        a[receiver] += delta

        b[donor:donor + n] -= 2.0 * delta * a_old
        b[receiver:receiver + n] += 2.0 * delta * a_old
        b[2 * donor] += delta ** 2
        b[2 * receiver] += delta ** 2
        b[donor + receiver] -= 2.0 * delta ** 2

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PrecisePeak it={it:5d} new_max={best_new_max:.6f} score={best_score:.6f}")

        # occasional re‑synchronisation to avoid drift
        if it % 200 == 0:
            b = _conv_fft(a)

        it += 1

    return best_a, best_score

# Linear‑programming Frank‑Wolfe refinement (optional)
# ----------------------------------------------------------------------


def _lp_iterative_refinement(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             epsilon_rel: float = 0.02,
                             step: float = 0.20,
                             max_iters: int = 120) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    eps = epsilon_rel
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        idx = np.where(conv >= M - eps * M)[0]
        if idx.size == 0:
            break
        if idx.size > 30:
            idx = idx[np.argsort(-conv[idx])[:30]]
        C = idx.size

        A = np.zeros((C, n), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            A[row, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

        b_vec = np.full(C, M, dtype=np.float64)

        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_vec,
                      bounds=[(0, None)] * n,
                      method='highs', options={'disp': False})
        if not res.success:
            eps = min(eps * 2.0, 0.1)
            it += 1
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g = g * (target_sum / sum_g)

        a_new = (1.0 - step) * a + step * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            eps = max(eps * 0.8, 1e-5)
            print(f"[{time.time():.2f}] LP it={it:3d} eps={eps:.2e} score={best_score:.6f}")
        else:
            step = max(step * 0.6, 0.01)
            eps = min(eps * 1.5, 0.05)

        it += 1

    return best_a, best_score

    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 16000          # a power‑of‑two‑friendly size
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (parametric families + previous best if available)
    # --------------------------------------------------------------
    best_seq = _power_law_sequence(n, 0.5, target_sum)  # fallback
    best_score = _score(best_seq)

    # power‑law sweep
    for p in np.linspace(0.30, 0.85, 25):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # tilted power‑law sweep
    for p in np.linspace(0.30, 0.85, 17):
        for q in np.linspace(0.0, 6.0, 17):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.5 * n, 5):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # Triangular shape
    seq = _triangular_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # previous global best if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] Warm‑start best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam GD with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = np.random.default_rng().normal(scale=target_sum * 0.015,
                                                size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam‑GD improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Precise peak‑mass transfer (deterministic greedy)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        a_pt, val_pt = _precise_peak_mass_transfer(best_seq,
                                                   target_sum,
                                                   deadline - 20.0,
                                                   max_iters=20000,
                                                   K=12)
        if val_pt < best_score:
            best_score, best_seq = val_pt, a_pt.copy()
            print(f"[{time.time():.2f}] PrecisePeak improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Optional LP refinement (Frank‑Wolfe style)
    # --------------------------------------------------------------
    if time.time() < deadline - 12.0:
        a_lp, val_lp = _lp_iterative_refinement(best_seq,
                                                target_sum,
                                                deadline - 5.0,
                                                epsilon_rel=0.02,
                                                step=0.20,
                                                max_iters=120)
        if val_lp < best_score:
            best_score, best_seq = val_lp, a_lp.copy()
            print(f"[{time.time():.2f}] LP refinement improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Final Adam polishing (high‑β)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    final_seq = _trim_sequence(best_seq)

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
