def _smooth_gradient_descent(a0, deadline, target_sum, alpha=10.0, step_factor=0.5, gamma=0.99):
    a = _project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a) # full conv
        maxc = conv.max()
        # compute softmax weights
        # w = np.exp(alpha*(conv - maxc))
        # use scaling to avoid overflow: subtract maxc then multiply.
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # compute gradient via convolution of a and reversed w:
        # conv_rev = np.convolve(a, w[::-1])
        # gradient = 2 * conv_rev[n-1:2*n-1]
        conv_rev = _conv_generic_fft(a, w[::-1]) # need generic convolution
        grad = 2.0 * conv_rev[len(a)-1:len(a)-1+len(a)]  # slice length n
        
        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / (norm + 1e-12)
            a = a - lr * momentum
        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

def _conv_generic_fft(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n+m-1]
    return conv

def _optimize_ordering(seq, deadline):
    best_seq = seq.copy()
    best_val = _max_convolution_fft(seq)
    n = len(seq)
    rng = np.random.default_rng()

    # We'll perform random swaps with simulated annealing.
    start_temp = best_val * 0.1
    end_temp = 1e-6
    while time.time() < deadline:
        # decide temp schedule
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = start_temp * (1 - frac) + end_temp * frac
        # propose swap
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # swap i and j
        seq_swapped = best_seq.copy()
        seq_swapped[i], seq_swapped[j] = seq_swapped[j], seq_swapped[i]
        val_swapped = _max_convolution_fft(seq_swapped)
        if val_swapped < best_val or rng.random() < np.exp(-(val_swapped - best_val)/ max(T,1e-12)):
            best_seq = seq_swapped
            best_val = val_swapped
    return best_seq

    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)

   - Instead of the sub‑gradient on the exact max, minimise a smoothed
     version  f(a)=log‑sum‑exp(α·(a∗a))  by projected gradient descent.
   - The gradient is obtained efficiently via FFT‑based correlation.
   - An exponential schedule for α (from 5 to 30) sharpens the
     approximation and drives the solution toward a lower peak.

3. **Permutation search**
   - For a fixed multiset of values, the ordering heavily influences the
     auto‑convolution peak.  A fast random‑permutation / simulated‑annealing
     loop (≈ few × 10⁴ trials) is performed, keeping the best ordering.
   - Finally three deterministic orderings (alternating, even‑spread,
     original) are also tested.

All vectors are kept non‑negative and projected onto the simplex after
each update, guaranteeing that the evaluator’s ℓ¹‑norm constraint
(Σa = √(2n)) holds.  The best sequence found is stored in the global
variable ``height_sequence_1`` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _corr_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation r = a ⊗ b (full length n+m‑1) using FFT.
    r[k] = Σ_i a[i]·b[i+k] with zero‑padding outside the support.
    """
    n, m = a.shape[0], b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    corr = np.fft.irfft(fa * np.conj(fb), n=size)[: n + m - 1]
    return corr


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray, target_sum: float) -> float:
    """Objective 2·n·max(a*a) / (Σa)²; with Σa = target_sum the value = max conv."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (different shapes)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(−p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(−lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with projection onto the ℓ¹‑simplex."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind, target_sum) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # pick three distinct mates different from i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial, target_sum)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        # early exit guard
        if time.time() > deadline - 1.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp gradient descent
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha_start: float = 5.0,
                             alpha_end: float = 30.0,
                             step_factor: float = 0.4,
                             gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth objective
        f(a) = log‑sum‑exp( α·(a∗a) ),
    where α is increased from ``alpha_start`` to ``alpha_end``.
    The gradient is obtained via FFT‑based correlation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        # current convolution and its maximum
        conv = _conv_fft(a)
        maxc = conv.max()

        # schedule for α
        tau = (time.time() - start) / max(1e-12, (deadline - start))
        alpha = alpha_start * (alpha_end / alpha_start) ** tau

        # soft‑max weights over the convolution entries
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient = 2 * (a ⊗ w) restricted to the n lags
        corr = _corr_fft(a, w)                 # length n + (2n‑1) - 1 = 3n‑2
        offset = w.size - 1
        grad = 2.0 * corr[offset:offset + a.shape[0]]

        # momentum update and normalized step
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / (norm + 1e-12)
            a = a - lr * momentum

        # projection back onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to avoid plateaus
        if rng.random() < 0.03:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent (fallback)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = float(_conv_fft(a).max())
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = float(_conv_fft(a).max())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine builds a diverse seed pool, runs DE, refines with
    a smooth log‑sum‑exp gradient descent, and finishes with a permutation
    optimisation.  The best sequence is stored in ``globals()['height_sequence_1']``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 10000
    elif budget_s >= 600:
        n = 8000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.2, 11):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.015, 10):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # pure random points on the simplex
    for _ in range(10):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm‑start from previous run, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.025, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # --------------------------------------------------------------
    # Initialise DE population
    # --------------------------------------------------------------
    pop_size = max(30, int(n // 250))   # roughly 30‑40 individuals
    population = []

    best_seed = min(seeds, key=lambda s: _score(s, target_sum))
    population.append(best_seed.copy())
    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # --------------------------------------------------------------
    # Differential Evolution phase (≈ 35 % of the budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.35 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Smooth log‑sum‑exp gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 12.0:
        sgd_deadline = min(deadline - 5.0, de_deadline + 0.45 * budget_s)
        seq_sgd, peak_sgd = _smooth_gradient_descent(best_seq, sgd_deadline, target_sum,
                                                    alpha_start=5.0,
                                                    alpha_end=30.0,
                                                    step_factor=0.4,
                                                    gamma=0.99)
        if peak_sgd < best_score:
            best_seq, best_score = seq_sgd, peak_sgd
            print(f"[{time.time():.2f}] SGD improved – new peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent as a fallback (short budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 7.0:
        sg_deadline = time.time() + min(15.0, deadline - time.time() - 5.0)
        seq_sg, peak_sg = _subgradient_descent(best_seq, sg_deadline, target_sum,
                                               step_factor=0.35,
                                               gamma=0.995)
        if peak_sg < best_score:
            best_seq, best_score = seq_sg, peak_sg
            print(f"[{time.time():.2f}] Sub‑grad improved – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Permutation optimisation (random search + deterministic orders)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        # start from the current best ordering
        seq_best = best_seq.copy()
        val_best = _max_convolution_fft(seq_best)
        n = seq_best.shape[0]
        rng = np.random.default_rng()
        while time.time() < deadline:
            perm = rng.permutation(n)
            seq_perm = seq_best[perm]
            val = _max_convolution_fft(seq_perm)
            if val < val_best:
                val_best = val
                seq_best = seq_perm
        # deterministic candidates
        candidates = [seq_best,
                      _alternating_order(seq_best),
                      _even_spread_order(seq_best),
                      best_seq]  # original ordering as fallback
        best_seq = min(candidates, key=_max_convolution_fft)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.2f}] Permutation search – final peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for warm‑starting the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
