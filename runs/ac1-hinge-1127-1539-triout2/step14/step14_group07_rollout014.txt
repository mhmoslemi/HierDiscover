def _gradient_softmax(a, beta):
    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    # compute gradient: derivative of sum(p_k * b_k) w.r.t a
    # derivative of b_k w.r.t a = conv of a_rev with unit basis; they compute via FFT.
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, b
```

def _adam_optimize(a0, target_sum, deadline, max_beta=1e7, step_factor=0.6, gamma=0.995):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc = 150
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)
        # optional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it} beta={beta:.1e} score={best_val:.6f}")
        if it % beta_inc == 0 and beta < max_beta:
            beta = min(beta * 2, max_beta)
        it += 1
    return best_a, best_val

"""
Step‑function optimizer for the “max self‑convolution” challenge.

The algorithm mixes several complementary phases:
0️⃣  Warm‑start using several parametric families (power‑law, tilted‑power,
    Gaussian, sinusoidal, uniform).
1️⃣  Heavy‑weight projected Adam (soft‑max) with many restarts, an
    exponentially‑increasing soft‑max temperature β and adaptive
    step‑sizes.  This efficiently minimises a smooth approximation of the
    L∞‑norm of the self‑convolution.
2️⃣  Exact deterministic mass‑transfer (DMT) – a greedy single‑donor /
    single‑receiver move that provably lowers the current peak.
3️⃣  Global peak‑mass transfer – moves a bulk of mass from the strongest
    contributors to a weak receiver, solving the exact 1‑D line‑search.
4️⃣  LP‑based Frank‑Wolfe refinement – a tiny linear programme on the
    active peak constraints yields a descent direction.
5️⃣  Pairwise optimal transfer – exhaustive search over top donors /
    bottom receivers for a further local improvement.
6️⃣  Final Adam polishing with a very high β.

All phases respect the global deadline (default 1000 s).  The best
sequence found is trimmed of trailing ≈0 entries and returned as a plain
Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(b) / (sum(a)^2).  For Σa = √(2 n) the denominator equals 2·n,
    hence the score equals max(b).  For any other sum we return the exact formula.
    """
    n = a.shape[0]
    max_b = _max_conv_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    # based on the algorithm of Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A single bump shaped by a sine half‑wave."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _uniform_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.full(n, target_sum / n, dtype=np.float64)
    return a


# ----------------------------------------------------------------------
#  Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (gradient, raw convolution) of the soft‑max approximation of max(b),
    where b = a*a.  grad = ∂  Σ p_k b_k / ∂a,   p = softmax(β·(b−max)).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # avoid overflow – shift by max_b
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # only the n entries that correspond to a
    return grad, b


# ----------------------------------------------------------------------
#  Adam‑style projected GD (soft‑max)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_beta: float = 1e7,
                   step_factor: float = 0.6,
                   gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the soft‑max approximation.
    Returns the best vector seen before `deadline`.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 200  # increase temperature every this many steps

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)

        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:.1e} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _adam_multiple_restarts(init: np.ndarray,
                            target_sum: float,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Run several Adam‑GD restarts with jittered starts until `deadline`.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.015, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a_opt, val_opt = _adam_optimize(start,
                                         target_sum,
                                         time.time() + run_budget,
                                         max_beta=1e7,
                                         step_factor=0.6,
                                         gamma=0.995)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
#  Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy single‑donor / single‑receiver move that strictly lowers the peak.
    Terminates when the gap between the maximal and second maximal convolution
    value falls below `tol` or when the deadline is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]  # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Global peak‑mass transfer (refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Bulk move a fraction of total donor mass to the weakest receiver.
    The donor set consists of the top `donor_frac` entries (by a_i·partner_i);
    the receiver candidates are the lowest `receiver_frac` entries (by partner).
    """
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second maximum (used as stop‑condition)
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights for the current peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        # donors – indices with largest a_i * w_i
        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        # contributions of donors only
        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        # receivers – smallest partner values
        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        # padded view for quick shifting
        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if r in donor_idx:
                continue
            shift_r = base[r:r + 2 * n - 1]
            # we move a proportion α of donor_mass to position r
            diff = donor_mass * shift_r - conv_donors
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        # apply the chosen move
        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak it={it} α={best_alpha:.4g} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  LP‑based Frank‑Wolfe refinement
# ----------------------------------------------------------------------


def _lp_iterative_refinement(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             epsilon_rel: float = 0.02,
                             step: float = 0.20,
                             max_iters: int = 120) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style updates using a tiny linear programme.
    In each iteration we solve:
        max Σ g_j   s.t.   f * g ≤ M
    on a small active set of peak‑constraints (indices where the convolution
    is within `epsilon_rel * M` of the current maximum `M`).
    The solution g is then scaled to the simplex and blended with f.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    eps = epsilon_rel
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        idx = np.where(conv >= M - eps * M)[0]
        if idx.size == 0:
            break
        if idx.size > 40:
            idx = idx[np.argsort(-conv[idx])[:40]]   # keep most critical constraints

        C = idx.size
        # construct constraint matrix A (C × n)
        A = np.zeros((C, n), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            A[row, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

        b_vec = np.full(C, M, dtype=np.float64)

        # linear programme: maximise Σ g_j   (<=> minimise -Σ g_j)
        try:
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_vec,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception as e:
            print(f"[{time.time():.2f}] LP exception: {e}")
            break

        if not res.success:
            # relax tolerance and continue
            eps = min(eps * 2.0, 0.1)
            it += 1
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        g = g * (target_sum / sum_g)          # scale to the simplex
        a_new = (1.0 - step) * a + step * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            eps = max(eps * 0.8, 1e-5)
            print(f"[{time.time():.2f}] LP it={it:3d} eps={eps:.2e} score={best_score:.6f}")
        else:
            step = max(step * 0.6, 0.01)
            eps = min(eps * 1.5, 0.05)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Pairwise optimal mass transfer
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Exhaustive search over top donors and bottom receivers.
    For each pair we compute the maximal feasible δ that reduces the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)        # effect of moving mass from i → j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
#  Utility – trim trailing near‑zero entries
# ----------------------------------------------------------------------


def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]
    if nz.size == 0:
        return [0.0]
    return a[: nz[-1] + 1].tolist()


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence found as a Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but higher cost)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) → score = max(b)

    # --------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # --------------------------------------------------------------
    best_seq = _uniform_sequence(n, target_sum)   # fallback
    best_score = _score(best_seq)

    # power‑law grid
    for p in np.linspace(0.30, 0.85, 31):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # exp‑tilted power‑law grid
    for p in np.linspace(0.30, 0.85, 21):
        for q in np.linspace(0.0, 6.0, 21):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # Gaussian family
    for sigma in np.linspace(0.1 * n, 0.5 * n, 7):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # sinusoidal bump
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # incorporate previous global solution if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # a few jittered perturbations around the current best
    rng = np.random.default_rng()
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam‑style projected GD with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.35
    while time.time() < gd_deadline - 5.0:
        # each restart receives a modest time slice
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_multiple_restarts(start_vec,
                                                 target_sum,
                                                 gd_deadline)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam‑GD phase improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (core refinement)
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT2 improved best = {best_score:.6f}")

        # occasional random jump to avoid stagnation
        a_rand, val_rand = _deterministic_mass_transfer_exact2(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random DMT2 improved best = {best_score:.6f}")

        if time.time() > deadline - 30.0:
            break

    # --------------------------------------------------------------
    # Phase C – Global peak‑mass transfer (refinement, repeated)
    # --------------------------------------------------------------
    while time.time() < deadline - 15.0:
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.05,
                                                      receiver_frac=0.05)
        if val_peak < best_score:
            best_score, best_seq = val_peak, a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best = {best_score:.6f}")
        else:
            break      # no gain → exit early

    # --------------------------------------------------------------
    # Phase D – LP‑based Frank‑Wolfe refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        a_lp, val_lp = _lp_iterative_refinement(best_seq,
                                                target_sum,
                                                deadline,
                                                epsilon_rel=0.015,
                                                step=0.25,
                                                max_iters=200)
        if val_lp < best_score:
            best_score, best_seq = val_lp, a_lp.copy()
            print(f"[{time.time():.2f}] LP refinement improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Pairwise optimal transfer
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.02,
                                                      receiver_frac=0.02)
        if val_pair < best_score:
            best_score, best_seq = val_pair, a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase F – Final polishing (high‑β Adam)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_optimize(best_seq,
                                         target_sum,
                                         deadline,
                                         max_beta=5e7,
                                         step_factor=0.5,
                                         gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polishing best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    final_seq = _trim_sequence(best_seq)

    # Optional final verification (ignore errors if evaluate_sequence is not imported)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
