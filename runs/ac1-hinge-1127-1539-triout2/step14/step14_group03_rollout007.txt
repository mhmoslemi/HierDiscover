import numpy as np
import time

def _project_to_simplex(v, s):
    # Euclidean projection onto simplex {x >= 0, sum x = s}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    return conv

def _score(a):
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _shape_powerlaw(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    return idx ** (-p)

def _shape_exponential(n, lam):
    if n == 1:
        return np.array([1.0])
    x = np.arange(n, dtype=np.float64) / (n-1)
    return np.exp(-lam * x)

def _shape_powerexp(n, p, lam):
    idx = np.arange(1, n+1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n-1)
    return idx ** (-p) * np.exp(-lam * x)

def _shape_beta(n, alpha):
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a

def _gradient_softmax(a, size_fft, beta):
    """
    Compute gradient of (1/beta) log sum exp(beta * (a*a)) w.r.t a.
    Returns gradient vector of length n.
    """
    n = a.shape[0]
    A = np.fft.rfft(a, n=size_fft)
    b = np.fft.irfft(A * A, n=size_fft)[:2*n-1]
    max_b = b.max()
    # Softmax weights:
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # Correlation between a and w:
    W = np.fft.rfft(w, n=size_fft)
    corr_full = np.fft.irfft(A * np.conj(W), n=size_fft)
    offset = 2 * n - 2
    grad = 2.0 * corr_full[offset: offset + n]
    return grad, max_b

def _softmax_descent(a0, target_sum, deadline, beta_start=2.0, beta_end=30.0):
    """
    Projected gradient descent on the smooth approximation of max convolution.
    Returns best a seen and its max_convolution value.
    """
    n = a0.shape[0]
    a = a0.copy()
    # FFT size for full correlation (size >= 3n-2)
    size_fft = 1 << ((3 * n - 2).bit_length())
    best_a = a.copy()
    # compute initial max
    _, cur_max = _gradient_softmax(a, size_fft, beta_start)
    best_val = cur_max
    start = time.time()
    it = 0
    beta = beta_start
    # schedule to increase beta linearly to beta_end over allocated time
    while time.time() < deadline:
        # compute gradient and current max for this beta
        grad, cur_max = _gradient_softmax(a, size_fft, beta)
        # step size adaptation: diminishing ~ 1/sqrt(it+1)
        step = target_sum * 0.2 / (np.linalg.norm(grad) + 1e-12)  # ensures reasonable magnitude
        step = step / np.sqrt(it + 1)
        a_new = a - step * grad
        a_new = _project_to_simplex(a_new, target_sum)
        a = a_new
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Softmax descent it={it} beta={beta:.2f} best_max={best_val:.6f}")
        # update beta gradually
        elapsed = time.time() - start
        total = deadline - start
        # linear increase
        beta = beta_start + (beta_end - beta_start) * min(1.0, elapsed / total)
        it += 1
    return best_a, best_val

def _anneal_mass_transfer(a0, target_sum, deadline):
    """
    Aggressive simulated‑annealing mass‑transfer refinement (same as earlier version).
    Returns best a and its max_convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_max = b.max()
    best_val = cur_max
    best_a = a.copy()
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_max * 0.9 if cur_max > 0 else 0.1
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi+1)
        # receiver prefers low heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)
        if donor == receiver:
            continue
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0:
            continue
        # O(n) update of convolution
        b_new = b.copy()
        # removal from donor
        b_new[donor:donor + n] -= 2.0 * delta * a
        # addition to receiver
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic corrections
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        new_max = b_new.max()
        accept = False
        if new_max <= cur_max:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_max - cur_max) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")
        if it % 4000 == 0:
            # occasional full recompute
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_max = b.max()
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")
        it += 1
    return best_a, best_val

                                          beta_start=2.0, beta_end=30.0)

"""
search for low‑max‑convolution step sequences.

Approach
--------
1. Warm‑start from the previously‑found sequence (if any) and from a
   small collection of analytic families (power‑law, exponential,
   combined, symmetric Beta).  The best of those is kept.

2. Projected soft‑max gradient descent.
   We minimise a smooth approximation of the peak of the self‑convolution:
        Lβ(a) = (1/β)·log( Σ_k exp(β·(a∗a)_k) )
   Its gradient is a simple correlation that can be obtained with FFTs.
   After each gradient step the vector is projected onto the simplex
   {x≥0, Σx = √(2·n)}.  The smoothing parameter β is increased
   linearly from a modest value to a larger one, turning the smooth
   problem into the original max‑convolution problem.

3. Aggressive simulated‑annealing “mass‑transfer” refinement
   (identical to the one used previously) to escape any remaining
   local minima.

The routine respects a user‑supplied time budget and stores the final
sequence in the global variable ``height_sequence_1`` for warm‑starts in
future invocations.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Primitive utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Analytic families for warm‑starts
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i/(n‑1))."""
    if n == 1:
        return np.ones(1)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam·i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, size_fft: int, beta: float):
    """
    Gradient of Lβ(a) = (1/β)·log Σ exp(β·(a*a)).

    Returns
    -------
    grad : ndarray, shape (n,)
        Euclidean gradient of Lβ.
    max_b : float
        Exact max of the current convolution (not the smoothed value).
    """
    n = a.shape[0]
    A = np.fft.rfft(a, n=size_fft)
    # exact convolution a*a
    b = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
    max_b = b.max()

    # soft‑max weights (probabilities)
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w gives ∂L/∂a (up to factor 2)
    W = np.fft.rfft(w, n=size_fft)
    corr_full = np.fft.irfft(A * np.conj(W), n=size_fft)
    offset = 2 * n - 2                      # lag‑0 appears at index M‑1, M=2n‑1
    grad = 2.0 * corr_full[offset: offset + n]
    return grad, max_b


def _softmax_descent(a0: np.ndarray, target_sum: float,
                     deadline: float,
                     beta_start: float = 2.0,
                     beta_end: float = 30.0):
    """
    Projected gradient descent on the smooth approximation of the peak.
    Returns the best vector encountered and its exact max‑convolution.
    """
    n = a0.shape[0]
    a = a0.copy()
    # FFT size must accommodate a (n) and w (2n‑1) for a⋆w → length ≤ 3n‑2
    size_fft = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    _, cur_max = _gradient_softmax(a, size_fft, beta_start)
    best_val = cur_max

    start = time.time()
    it = 0
    while time.time() < deadline:
        # current smoothing parameter (linear schedule)
        elapsed = time.time() - start
        total = deadline - start
        beta = beta_start + (beta_end - beta_start) * min(1.0, elapsed / total)

        grad, cur_max = _gradient_softmax(a, size_fft, beta)

        # step size – diminishing ~1/√t and scaled to the simplex size
        step = target_sum * 0.2 / (np.linalg.norm(grad) + 1e-12)
        step /= np.sqrt(it + 1)

        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Soft‑max it={it:5d} β={beta:5.2f} best_max={best_val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from “bad” indices
    (large contribution to the current peak) to “good” ones.
    Returns the best vector seen and its max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_max = b.max()
    best_val = cur_max
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_max * 0.9 if cur_max > 0 else 0.1
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        accept = False
        if new_max <= cur_max:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_max - cur_max) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_max = b.max()
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low max‑convolution.
    The routine respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # Choose problem size (larger n → finer control but higher cost)
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # the normalisation used in evaluate_sequence

    # ------------------------------------------------------------------
    # 1. Warm‑start / guided initialisation
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    rng = np.random.default_rng()

    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        val = _score(warm)
        best_seq = warm.copy()
        best_val = val
        print(f"[{time.time():.2f}] Warm start score = {best_val:.6f}")

    # deterministic families
    for p in np.linspace(0.45, 0.75, 7):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} score={val:.6f}")

    for lam in np.linspace(0.5, 8.0, 7):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.3f} score={val:.6f}")

    for _ in range(8):
        p = rng.random() * 0.8 + 0.1
        lam = rng.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] PowerExp p={p:.3f} λ={lam:.3f} score={val:.6f}")

    for alpha in np.linspace(0.25, 0.9, 7):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.3f} score={val:.6f}")

    # a few random Dirichlet‑like starts
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _score(rnd)
        if val < best_val:
            best_val = val
            best_seq = rnd.copy()
            print(f"[{time.time():.2f}] Random start score={val:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected soft‑max gradient descent (major phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.45 * (deadline - time.time())
        a_opt, val_opt = _softmax_descent(best_seq, target_sum,
                                          phase_deadline,
                                          beta_start=2.0, beta_end=30.0)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] After soft‑max descent best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Aggressive annealing refinement (final phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, val_opt = _anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] After anneal best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # optional final check using the provided evaluate_sequence
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
