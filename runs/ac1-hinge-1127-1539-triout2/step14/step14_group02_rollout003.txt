def subgradient_max_descent(a0, target_sum, deadline, step0=0.5):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)
    it = 0
    while time.time() < deadline:
        b = _conv_self_fft(a)
        max_idx = int(np.argmax(b))
        max_val = b[max_idx]
        # gradient for max index
        g = np.zeros_like(a)
        # compute contributions only for valid indices
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        # indices i in [lo, hi]
        i = np.arange(lo, hi+1)
        g[i] = 2 * a[max_idx - i]
        # step size
        norm = np.linalg.norm(g)
        if norm == 0:
            break
        eta = step0 / np.sqrt(it + 1) * (target_sum / norm)
        a = a - eta * g
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # evaluate
        cur_val = max_val

            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] subgrad new best {best_val:.6f}")
        it += 1
    return best, best_val

Step‑function optimisation for the max‑self‑convolution objective.

The routine searches for a non‑negative sequence `a` with
    Σ a = √(2·n)
that minimises  max(a * a)  (the convolution peak).
The code combines three complementary optimisation phases:

1. **Smooth‑max projected gradient descent** – a differentiable
   approximation of the max (log‑sum‑exp) is minimised.  The temperature
   β is increased gradually up to a large value (default 200) so the
   iterates approach the true objective.

2. **Exact (sub‑)gradient descent on the true max** – after the smooth
   phase we perform a few iterations of sub‑gradient descent on the
   peak itself, projecting onto the simplex after every step.

3. **Mass‑transfer simulated annealing** – a fast O(n) update moves
   small amounts of mass from indices that contribute most to the
   current peak to indices that contribute least, with a Metropolis
   criterion that gradually cools.

Multiple random restarts (including a warm‑start from the previous run
and several deterministic initial shapes) are tried within the time
budget.  The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a power‑of‑2 FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact objective = max(b) when Σa = sqrt(2·n)."""
    return float(b.max())  # target_sum cancels out in the given normalisation


# ----------------------------------------------------------------------
# Simple geometric helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 200.0,
                       step0_factor: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Minimise a log‑sum‑exp approximation of max(a*a).
    The iterates are projected onto the L1‑simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    step0 = target_sum * step0_factor
    it = 0
    while time.time() < deadline:
        b = _conv_self_fft(a)                # length 2n‑1
        max_b = b.max()
        cur_val = max_b                       # exact objective
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.1f} it={it:5d} best={best_val:.6f}")

        # soft‑max weights (stable)
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w * a_rev)  (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)                     # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]                # central n entries

        # adaptive step size
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # increase β gradually
        if it % 1500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.4)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – exact sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def subgradient_max_descent(a0: np.ndarray,
                            target_sum: float,
                            deadline: float,
                            step0_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient steps on max(a*a) with projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    step0 = target_sum * step0_factor
    it = 0
    while time.time() < deadline:
        b = _conv_self_fft(a)
        k_max = int(np.argmax(b))
        # gradient of b[k_max] w.r.t. a
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k_max - idx]

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)
        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _conv_self_fft(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] subgrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that relocates tiny amounts of mass from indices
    that contribute heavily to the current convolution peak to indices
    that contribute little.  Updates of the full convolution are performed
    in O(n) using the slice‑wise formula from the reference implementation.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)                      # 2n‑1 entries
    cur_val = b.max()
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature as a fraction of the current energy
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # sqrt cooling

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour positions that *do not* increase the peak much
        # i.e. small a[k_max - r] (if k_max - r inside range)
        score = np.ones_like(a)
        partner = k_max - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        score[mask] = 1.0 / (a[partner[mask]] + 1e-12)   # inverse of partner height
        prob = score / score.sum()
        receiver = rng.choice(n, p=prob)

        if donor == receiver:
            continue

        # magnitude of transfer, scaled by temperature
        max_delta = min(a[donor] * 0.09, target_sum * 0.012)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- O(n) exact update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (removing from donor, adding to receiver)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = b_new.max()

        # Metropolis acceptance (energy = current max)
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob_acc = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob_acc:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal iter={it:6d} best={best_val:.6f}")

        # occasional full recompute to avoid drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute iter={it:6d} best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low max‑self‑convolution value.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives finer granularity but is slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 24000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)       # normalisation used by evaluate_sequence

    # --------------------------------------------------------------
    # Build a diverse set of initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # Warm start from previous run (if any)
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()

    # Power‑law families
    for p in (0.45, 0.55, 0.65):
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-p)
        a = _project_to_simplex(a, target_sum)
        init_candidates.append(a)

    # Hann window (cosine‑squared shape)
    x = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    init_candidates.append(_project_to_simplex(hann, target_sum))

    # Random Dirichlet draws
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optimisation loop over all starts (time‑aware)
    # --------------------------------------------------------------
    for start_idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ***** start #{start_idx + 1} (n={n}) *****")

        # ----- Phase 1 : smooth‑max gradient descent -----
        t1_dead = time.time() + 0.30 * (deadline - time.time())
        a_smooth, val_smooth = smooth_max_descent(
            init, target_sum, t1_dead,
            beta_start=0.1, beta_max=200.0, step0_factor=0.6
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ----- Phase 2 : exact sub‑gradient descent -----
        if time.time() < deadline:
            t2_dead = time.time() + 0.15 * (deadline - time.time())
            a_sub, val_sub = subgradient_max_descent(
                a_smooth, target_sum, t2_dead, step0_factor=0.5
            )
            if val_sub < best_score:
                best_score = val_sub
                best_seq = a_sub.copy()
                print(f"[{time.time():.2f}] New global best after subgrad: {best_score:.6f}")

        # ----- Phase 3 : annealing refinement -----
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(
                a_sub if time.time() < deadline else a_smooth,
                target_sum,
                deadline
            )
            if val_ann < best_score:
                best_score = val_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the unlikely case that nothing was produced
    # --------------------------------------------------------------
    if best_seq is None:
        fallback = (np.arange(1, n + 1, dtype=np.float64) ** (-0.57))
        best_seq = _project_to_simplex(fallback, target_sum)

    # Trim near‑zero trailing entries for a compact representation
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final evaluation (for logging)
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
