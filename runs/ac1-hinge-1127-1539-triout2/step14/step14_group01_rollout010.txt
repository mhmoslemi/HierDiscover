def _smooth_gradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    t0 = time.time()
    n = a.shape[0]
    # parameters
    alpha_start = 0.5
    alpha_end = 30.0
    # constant for step factor schedule:
    step_start = 1.0
    step_end = 0.01

    while time.time() < deadline:
        elapsed = time.time() - t0
        frac = elapsed / (deadline - t0)
        alpha = alpha_start + (alpha_end - alpha_start) * (frac ** 2)  # increase quadratically
        # compute b
        b = _conv_fft(a)
        max_b = b.max()
        # compute softmax weights
        # shift
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        # potential overflow: w may be underflow to zero if alpha huge; but max_b is max, b-max_b negative.
        if w_sum == 0:
            # fallback to one-hot on argmax
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # compute gradient: 2 * conv of a with reversed w
        w_rev = w[::-1]
        # convolution via FFT

        norm = np.linalg.norm(grad) + 1e-12

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
    # return best
    return best_a, best_val

"""
Hybrid smooth‑gradient optimizer for the minimum‑peak auto‑convolution problem.

The core idea is to minimise a smooth approximation of the
maximum entry of the auto‑convolution a*a while keeping a on the
ℓ¹‑simplex (non‑negative, fixed sum = √(2n)).  The smooth
approximation is the soft‑max (log‑sum‑exp) of the convolution,
whose gradient can be computed efficiently by a single FFT‑based
correlation.  A projected (Euclidean) gradient descent with a
time‑varying temperature (soft‑max “sharpness”) and a simple
momentum term is run for most of the time budget.  The best
vector is then polished by a short deterministic mass‑transfer
step and a few heuristic reorderings.

The routine adapts the problem size to the time budget (up to 12 k
entries) and stores the final sequence in the global variable
`height_sequence_1` for warm‑starting later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (equals the peak when Σa = target_sum)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean simplex projection (non‑negative, Σ = s)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (different shapes)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Smooth‑gradient descent on the soft‑max approximation of the peak
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth (soft‑max) maximum of a*a.
    The soft‑max parameter α is increased from α_start to α_end,
    while a diminishing step size and a momentum term drive the
    optimisation.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    # hyper‑parameters
    alpha_start = 0.5
    alpha_end = 30.0
    step_start = 1.0       # initial step factor
    step_end = 0.02        # final step factor
    beta = 0.9             # momentum decay
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    t0 = time.time()

    while time.time() < deadline:
        elapsed = time.time() - t0
        frac = elapsed / (deadline - t0)  # [0,1]

        # increase soft‑max sharpness (quadratically for a slower start)
        alpha = alpha_start + (alpha_end - alpha_start) * (frac ** 2)

        # self‑convolution and soft‑max weights
        b = _conv_fft(a)
        max_b = b.max()
        w = np.exp(alpha * (b - max_b))      # max entry gets weight 1, others ≤1
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical safety
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the soft‑max objective:
        #   ∇ = 2 * (a  *  reverse(w))  (restricted to the original length)
        w_rev = w[::-1]
        conv_full = _conv_fft_two(a, w_rev)   # length 3n‑2
        grad = 2.0 * conv_full[n - 1: 2 * n - 1]   # exactly n components

        # momentum update
        momentum = beta * momentum + grad

        # step size that scales with target_sum / ||momentum||
        step_factor = step_start + (step_end - step_start) * frac
        norm = np.linalg.norm(momentum) + 1e-12
        eta = step_factor * target_sum / norm

        # gradient step + projection onto the simplex
        a = a - eta * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape shallow plateaus
        if int(frac * 1000) % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (as in the previous code)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-6,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = float(conv.max())

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

# Main entry point required by the outer harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the minimum‑peak convolution problem.
    Uses a smooth‑gradient descent on a soft‑max surrogate,
    followed by a short deterministic polishing and a few
    heuristic reorderings.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ---- problem size -------------------------------------------------
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 600:
        n = 10000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ---- seed generation ------------------------------------------------
    seeds = []

    # power‑law families
    for p in np.linspace(0.2, 1.5, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential families
    for lam in np.linspace(0.001, 0.04, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random points on the simplex
    for _ in range(6):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm‑start from previous run (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # pick the best seed (lowest peak) as starting point
    best_seed = min(seeds, key=_max_convolution_fft)
    best_seq = best_seed.copy()
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] Initial best peak = {best_score:.6f}")

    # ---- smooth‑gradient descent (majority of the budget) -------------
    grad_deadline = deadline - 12.0    # keep a few seconds for polishing
    if grad_deadline > time.time():
        best_seq, best_score = _smooth_gradient_descent(best_seq, target_sum, grad_deadline)
        print(f"[{time.time():.2f}] After gradient descent peak = {best_score:.6f}")

    # ---- short deterministic mass‑transfer polishing -------------------
    if time.time() < deadline - 5.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 2.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.35, end_temp=1e-6, low_frac=0.06)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] MT polished peak = {best_score:.6f}")

    # ---- reordering heuristics -----------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq)
    ]

    # a few random reorderings (time permitting)
    reorder_dead = deadline - 0.5
    while time.time() < reorder_dead:
        perm = rng.permutation(n)
        cand = best_seq[perm]
        val = _max_convolution_fft(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
        # break early if negligible time left
        if time.time() + 0.1 > reorder_dead:
            break

    best_seq = min(candidates + [best_seq], key=_max_convolution_fft)
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] Final peak after reordering = {best_score:.6f}")

    # ---- trim trailing near‑zero entries --------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for possible warm‑starts in subsequent runs
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
