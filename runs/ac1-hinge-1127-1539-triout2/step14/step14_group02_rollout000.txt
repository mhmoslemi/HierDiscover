"""
Optimisation of a non‑negative step function for the evaluation metric.

The algorithm consists of three stages:

1. **Initialisation** – several families of candidate vectors are built
   (warm‑start, random power‑laws, uniform) and projected onto the
   simplex Σ a_i = sqrt(2·n).

2. **Smooth‑max projected gradient descent** – a differentiable surrogate
   for the max‑convolution entry is minimised.  The surrogate is
   (1/β)·log Σ exp(β·(a∗a)_k).  The gradient is obtained with a single FFT,
   the iterate is projected back onto the simplex and β is increased
   gradually (up to 100) to tighten the approximation.

3. **Aggressive annealing / greedy mass transfer** – starting from the best
   vector of stage 2 we run a long simulated‑annealing loop that repeatedly
   moves a tiny amount of mass from “bad’’ indices (large contribution to
   the current peak of a∗a) to “good’’ indices (small height).  The convolution
   is updated in O(n) per move, and moves are always accepted when they
   lower the true objective.  The temperature schedule is linear to keep
   exploration for a long time.

The whole search respects the time budget (default 1000 s) and returns the
best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    # choose next power of two for speed
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _objective_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact objective: 2·n·max(b) / (target_sum²)."""
    return float(2.0 * n * b.max() / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a vector to a different length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 100.0,
                       beta_mul: float = 1.3,
                       beta_update: int = 1000,
                       step_factor: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of max(a∗a) using projected gradient.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    b = _conv_self_fft(a)
    best_val = _objective_from_selfconv(b, n, target_sum)

    beta = beta_start
    it = 0
    # base step size (before normalisation)
    base_step = step_factor * target_sum

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Compute convolution and soft‑max weights
        # --------------------------------------------------------------
        b = _conv_self_fft(a)               # length 2n‑1
        max_b = b.max()
        cur_val = _objective_from_selfconv(b, n, target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:6d} best={best_val:.6f}")

        # soft‑max over all entries (log‑sum‑exp trick)
        b_shift = b - max_b                # now max(b_shift)=0
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # Gradient: 2·(w ∗ a_rev)  (central slice)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)      # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]   # central n entries

        # --------------------------------------------------------------
        # Adaptive step size (scaled by gradient norm)
        # --------------------------------------------------------------
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = base_step / np.sqrt(it + 1.0) * (target_sum / norm_g)

        # Take a trial step and project back onto the simplex
        a_trial = a - eta * g
        a_trial = _project_to_simplex(a_trial, target_sum)

        # Evaluate trial objective (exact max)
        b_trial = _conv_self_fft(a_trial)
        trial_val = _objective_from_selfconv(b_trial, n, target_sum)

        # Simple Armijo‑like acceptance: accept if not worse, else shrink step
        if trial_val <= cur_val or eta < 1e-12:
            a = a_trial
        else:
            # backtrack a few times
            for _ in range(4):
                eta *= 0.5
                if eta < 1e-12:
                    break
                a_trial = a - eta * g
                a_trial = _project_to_simplex(a_trial, target_sum)
                b_trial = _conv_self_fft(a_trial)
                trial_val = _objective_from_selfconv(b_trial, n, target_sum)
                if trial_val <= cur_val:
                    a = a_trial
                    break

        # --------------------------------------------------------------
        # Update β (more focus on true max)
        # --------------------------------------------------------------
        if it % beta_update == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        it += 1

    return best_a, best_val

# Phase 2 – aggressive annealing / greedy mass transfer
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,

    contribute heavily to the current peak of a∗a to indices with low height.
    Returns the best vector found and its exact objective.

    b = _conv_self_fft(a)
    cur_val = _objective_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()
    rng = np.random.default_rng()

    start = time.time()
    total = deadline - start
    # initial temperature proportional to current energy
    T0 = cur_val * temp_frac

    it = 0
    while time.time() < deadline:
        # linear cooling (keeps some exploration for long)
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate global max of convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to this peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # choose a receiver that is currently low
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # magnitude of the transfer
        max_delta = min(a[donor] * max_delta_frac, target_sum * 0.005)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            continue

        # ---- exact O(n) update of the convolution -----------------
        # The convolution a*a is a triangular sum. Updating a[donor] and a[receiver]
        # by ±delta can be handled with O(n) corrections (see reference code).
        b_new = b.copy()

        # linear contribution: 2*a*Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a

        # quadratic contribution Δ²
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = _objective_from_selfconv(b_new, n, target_sum)

        # Acceptance test (Metropolis)
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # occasional full recompute to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _objective_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # slight safety margin

    # --------------------------------------------------------------
    # Problem size – longer sequences give a finer search but cost more.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 24000          # larger budget → longer vector
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)   # standard normalisation (scale‑invariant)

    # --------------------------------------------------------------
    # Build a collection of initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # warm‑start if a previous solution exists
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()

    # power‑law families – a_i ∝ (i+α)^{-p}
    for _ in range(3):
        p = rng.uniform(0.45, 0.70)
        shift = rng.uniform(0.0, 5.0)
        idx = np.arange(n, dtype=np.float64) + shift
        cand = idx ** (-p)
        cand = _project_to_simplex(cand, target_sum)
        init_candidates.append(cand)

    # pure uniform random vectors
    for _ in range(2):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run optimisation for each start, respecting the global deadline
    # --------------------------------------------------------------
    for start_idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}] ----- start #{start_idx + 1} (n={n}) -----")

        # Phase 1: smooth‑max projected gradient descent
        remaining = deadline - time.time()
        # allocate roughly 25 % of remaining time to this phase
        phase1_deadline = time.time() + 0.25 * remaining
        a_smooth, val_smooth = smooth_max_descent(
            init,
            target_sum,
            phase1_deadline,
            beta_start=0.1,
            beta_max=100.0,
            beta_mul=1.3,
            beta_update=800,
            step_factor=0.6,
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # Phase 2: aggressive annealing (uses whatever time is left)
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(
                a_smooth,
                target_sum,
                deadline,
                max_delta_frac=0.04,
                temp_frac=0.9,
            )
            if val_ann < best_score:
                best_score = val_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the unlikely event that nothing was created
    # --------------------------------------------------------------
    if best_seq is None:
        # simple decreasing power‑law as a guaranteed feasible point
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = _project_to_simplex(seq, target_sum)

    # Trim trailing near‑zeros for a cleaner representation
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final evaluation (outside of the optimisation loop)
    try:
        final_val = evaluate_sequence(final_seq)    # evaluate_sequence is provided by the runner
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store the result for warm‑starting future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
