"""
Improved step‑function search (≈ 1000 s budget).

The algorithm consists of three stages:

1.  Several diversified warm‑starts (previous best, random, power‑law,
    “U‑shaped”, etc.) are projected onto the simplex Σa = √(2 n).

2.  For every start we run a *smooth‑max* projected gradient descent.
    The true max‑convolution entry is approximated by a soft‑max
    (log‑sum‑exp) with slowly increasing temperature β.  A momentum term
    (Adam‑style) stabilises the updates.  The best vector from this phase
    is kept.

3.  The best vector is refined by a long *mass‑transfer simulated annealing*
    that moves tiny amounts of mass from indices heavily contributing to
    the current peak of the convolution to indices with small height.
    The convolution is updated in O(n) per move (exact update).  The
    temperature schedule is linear and a small deterministic “greedy”
    move is applied every few thousand iterations.

Finally the result is trimmed, evaluated with the provided
`evaluate_sequence`, stored in the global variable `height_sequence_1`
(for warm‑starts in later runs) and returned.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, power‑of‑2 size)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_selfconv(b: np.ndarray, n: int) -> float:
    """Exact evaluation score = 2·n·max(b)/(Σa)²  (Σa = √(2n) → score = max(b))."""
    return float(b.max())


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent with momentum
# ----------------------------------------------------------------------


def smooth_max_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.1,
    beta_max: float = 80.0,
    step_factor: float = 0.55,
    momentum_coeff: float = 0.9,
) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution entry.

    Parameters
    ----------
    a0 : np.ndarray
        Initial non‑negative vector (already on the simplex Σa = target_sum).
    target_sum : float
        Desired ℓ₁‑norm (√(2·n)).
    deadline : float
        Absolute timestamp when the routine must stop.
    beta_start, beta_max : float
        Temperature schedule for the soft‑max (log‑sum‑exp).
    step_factor : float
        Base step size as a fraction of target_sum.
    momentum_coeff : float
        Exponential moving‑average coefficient for gradient momentum.

    Returns
    -------
    best_a, best_score
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # initial exact score (used for bookkeeping)
    best_score = _score_from_selfconv(_conv_self_fft(a), n)

    beta = beta_start
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        # ----- self‑convolution -------------------------------------------------
        b = _conv_self_fft(a)               # length 2n‑1
        max_b = b.max()
        cur_score = max_b                    # because Σa = √(2n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:5d} best={best_score:.6f}")

        # ----- soft‑max weights -------------------------------------------------
        b_shift = b - max_b                  # shift for numeric stability
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (sub‑gradient of the smooth objective) -----------------
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)        # length 3n‑2
        g = 2.0 * conv_wa[n - 1 : n - 1 + n]  # central n entries

        # ----- momentum update -------------------------------------------------
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * g

        # ----- step size (scaled by gradient norm) ----------------------------
        norm_m = np.linalg.norm(momentum)
        if norm_m == 0.0:
            break
        eta = step_factor * target_sum / np.sqrt(it + 1.0) / norm_m

        # ----- gradient step + projection --------------------------------------
        a = a - eta * momentum
        a = _project_to_simplex(a, target_sum)

        # ----- β schedule -------------------------------------------------------
        if it % 1500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.4)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing with deterministic greedy moves
# ----------------------------------------------------------------------


def anneal_mass_transfer(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves tiny amounts of mass from indices that
    heavily contribute to the current peak of the self‑convolution to
    indices with small height.  The convolution is updated in O(n) per
    move (exact update).  A deterministic greedy move is applied every
    few thousand iterations.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_score = _score_from_selfconv(b, n)
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature as a fraction of the current score
    T0 = cur_score * 0.9
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # linear cooling (fast at the beginning, slower later)
        temperature = max(0.0, T0 * (1.0 - frac))

        # ---- locate the current peak ---------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)]

        # choose a receiver that is currently small (inverse‑proportional weights)
        inv = 1.0 / (a + 1e-14)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        # avoid self‑transfer
        if donor == receiver:
            continue

        # ---- size of the transfer -------------------------------------------
        # max possible delta (limited by donor mass and a global fraction)
        max_delta = min(a[donor] * 0.07, target_sum * 0.008)
        if max_delta <= 0.0:
            continue
        # scale with temperature (hot → larger steps)
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 1e-12:
            continue

        # ----- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²) and cross term
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_score = b_new.max()   # because Σa stays √(2n)

        # ----- Metropolis acceptance -----------------------------------------
        accept = False
        if new_score <= cur_score:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_score:.6f}")

        # ----- occasional deterministic greedy move -------------------------
        if it % 5000 == 0:
            # recompute a little to keep things stable
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_score = b.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] recompute after {it} it, best={best_score:.6f}")

        it += 1

    return best_a, best_score

def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} , projected onto the simplex."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return _project_to_simplex(seq, math.sqrt(2.0 * n))


def _u_shape(n: int) -> np.ndarray:
    """U‑shaped: larger values near the ends, smaller in the middle."""
    x = np.linspace(0.0, 1.0, n)
    seq = np.sqrt(x * (1.0 - x) + 1e-12)   # zero in middle, peaks at edges
    return _project_to_simplex(seq, math.sqrt(2.0 * n))


def _ramp(n: int) -> np.ndarray:
    """Linear ramp up then down (triangular window)."""
    half = n // 2
    ramp = np.concatenate([np.linspace(1, 2, half, endpoint=False),
                           np.linspace(2, 1, n - half)])
    return _project_to_simplex(ramp, math.sqrt(2.0 * n))

    # store for warm starts in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
