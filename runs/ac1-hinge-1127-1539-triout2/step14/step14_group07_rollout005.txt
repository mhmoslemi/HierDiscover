def _lp_reduce_max_peak(a0, target_sum, deadline, epsilon_ratio=0.02, max_active=30, min_delta=1e-8):
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # active indices
        eps = epsilon_ratio * M
        idx = np.where(conv >= M - eps)[0]
        if idx.size == 0:
            break
        if idx.size > max_active:
            idx = idx[np.argsort(-conv[idx])[:max_active]]
        C = idx.size

        # Build A_ub matrix and b_ub
        # dimensions (C, n+1)
        A_ub = np.zeros((C, n+1), dtype=np.float64)
        # Fill rows
        for r, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j_range = np.arange(lo, hi+1)
            A_ub[r, lo:hi+1] = 2.0 * a[k - j_range]  # 2*a_{k-j}
        # delta coefficient
        A_ub[:, n] = 1.0
        b_ub = M - conv[idx]

        # equality constraint sum(d) = 0
        A_eq = np.zeros((1, n+1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        bounds = [(-a_i, a_i) for a_i in a] + [(0.0, None)]

        # objective: maximize delta -> minimize -delta
        c = np.zeros(n+1)
        c[n] = -1.0

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP reduce failed: {res.message}")
            break

        delta_opt = res.x[n]
        if delta_opt < min_delta:
            # no significant improvement
            # reduce epsilon perhaps?
            epsilon_ratio *= 0.5
            if epsilon_ratio < 1e-4:

"""
Hybrid optimizer for the “max self‑convolution” challenge.

The algorithm proceeds in three stages:

1️⃣  Warm‑start – load the best known construction (global
    ``height_sequence_1``) or fall back to a simple power‑law and
    project it onto the simplex Σa = √(2 n).

2️⃣  LP‑based peak‑reduction – repeatedly solve a tiny linear programme
    that finds a feasible direction d (with Σd = 0, a+d ≥ 0) which,
    under the linearised convolution model, would lower the current
    maximal convolution value M.  After each LP we perform a short
    back‑tracking line‑search to guarantee an actual decrease of M.
    The active set of convolution indices is limited to the top
    few peaks (≤30), so each LP has only O(n) variables and a few
    constraints and solves in milliseconds.

3️⃣  Final polishing – a short Adam‑style projected gradient descent
    on a soft‑max smooth approximation of the max‑norm, using a very
    large β (≈10⁶) to finish the optimisation.

All phases respect the global time budget (default 1000 s).  The
resulting sequence is trimmed of trailing near‑zero entries before
being returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(b)/(sum(a)^2).  For Σa = √(2 n) this equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
#  Adam‑style projected GD (soft‑max) – optional final polish
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the log‑sum‑exp smooth max of the self‑convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 1e6,
                           step_factor: float = 0.6,
                           gamma: float = 0.995) -> np.ndarray:
    """Project‑Adam on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 200

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a


# ----------------------------------------------------------------------
#  LP‑based peak reduction (new component)
# ----------------------------------------------------------------------


def _lp_reduce_max_peak(a0: np.ndarray,
                        target_sum: float,
                        deadline: float,
                        epsilon_ratio: float = 0.02,
                        max_active: int = 30,
                        min_delta: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Iteratively solve a tiny linear programme to obtain a direction d
    that (linearly) lowers the current maximal convolution value.
    After each LP we back‑track with a simple line search and accept
    the move only if the true max actually decreases.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # select the most critical indices (within eps·M of the peak)
        eps = epsilon_ratio * M
        idx = np.where(conv >= M - eps)[0]
        if idx.size == 0:
            break
        if idx.size > max_active:
            idx = idx[np.argsort(-conv[idx])[:max_active]]
        C = idx.size

        # Build A_ub and b_ub for the LP
        # variables: d_0..d_{n-1}, delta (last)
        A_ub = np.zeros((C, n + 1), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j_range = np.arange(lo, hi + 1, dtype=np.int64)
            A_ub[row, lo:hi + 1] = 2.0 * a[k - j_range]   # coefficient for d_j
        A_ub[:, n] = 1.0                                        # coefficient for delta
        b_ub = M - conv[idx]                                    # RHS: M - current value

        # Equality constraint Σ d_i = 0
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # Bounds:       -a_i ≤ d_i ≤ a_i   (keep a_i non‑negative after update)
        bounds = [(-a_i, a_i) for a_i in a] + [(0.0, None)]      # last var is delta ≥ 0

        # Objective: maximise delta  → minimise -delta
        c = np.zeros(n + 1)
        c[-1] = -1.0

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      A_eq=A_eq,
                      b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # LP failed – tighten epsilon and retry
            epsilon_ratio *= 0.5
            if epsilon_ratio < 1e-5:
                break
            continue

        delta_opt = res.x[-1]
        if delta_opt < min_delta:
            # No useful reduction can be found – tighten epsilon
            epsilon_ratio *= 0.5
            if epsilon_ratio < 1e-5:
                break
            continue

        d = res.x[:n]

        # ----- line search to respect the true nonlinear convolution -----
        s = 1.0
        improved = False
        while s > 1e-6 and time.time() < deadline:
            a_candidate = a + s * d
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_to_simplex(a_candidate, target_sum)

            new_M = _conv_fft(a_candidate).max()
            if new_M < M - 1e-12:
                a = a_candidate
                best_score = _score(a)
                print(f"[{time.time():.2f}] LP‑reduce it={it:3d} Δ={delta_opt:.3e} step={s:.3f} score={best_score:.6f}")
                improved = True
                break
            s *= 0.5

        if not improved:
            # Could not apply the direction – shrink the active set
            epsilon_ratio *= 0.5
            if epsilon_ratio < 1e-5:
                break

        it += 1
        if it >= 200:                     # safety guard
            break

    return a, best_score


# ----------------------------------------------------------------------
#  Utility – trim trailing near‑zero entries
# ----------------------------------------------------------------------


def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]
    if nz.size == 0:
        return [0.0]
    return a[: nz[-1] + 1].tolist()


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising
    evaluate_sequence(...).  Returns the best sequence found as a list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)                    # Σa = √(2 n)

    # --------------------------------------------------------------
    # Warm‑start
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.size != n:
            # resize by linear interpolation
            old_x = np.linspace(0.0, 1.0, seq0.size)
            new_x = np.linspace(0.0, 1.0, n)
            seq0 = np.interp(new_x, old_x, seq0)
        best_seq = _project_to_simplex(seq0, target_sum)
    else:
        best_seq = _power_law_sequence(n, 0.5, target_sum)

    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Warm‑start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Small random jitter passes (quick exploration)
    # --------------------------------------------------------------
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_seq, best_score = cand, val
            print(f"[{time.time():.2f}] Jitter improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # LP‑based peak reduction (core improvement loop)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        best_seq, best_score = _lp_reduce_max_peak(best_seq,
                                                   target_sum,
                                                   deadline,
                                                   epsilon_ratio=0.02,
                                                   max_active=30)
        print(f"[{time.time():.2f}] After LP‑reduce score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final Adam polishing (high‑β soft‑max)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin = _adam_gradient_descent(best_seq,
                                        deadline,
                                        target_sum,
                                        max_beta=1e6,
                                        step_factor=0.6,
                                        gamma=0.995)
        fin_score = _score(a_fin)
        if fin_score < best_score:
            best_seq, best_score = a_fin, fin_score
            print(f"[{time.time():.2f}] Final Adam polish score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    final_seq = _trim_sequence(best_seq)

    try:
        val = evaluate_sequence(final_seq)   # the harness will provide this
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
