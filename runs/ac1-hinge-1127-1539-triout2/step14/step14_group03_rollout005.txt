"""
Advanced search for a low‑score non‑negative step sequence.

The algorithm consists of three stages:

1.  Warm‑start – generate several deterministic families (power‑law,
    exponential, combined, Beta) and a few random Dirichlet vectors.
    The best of those is kept.

2.  Smoothed projected gradient descent – minimise a smooth approximation
    of the max‑convolution (soft‑max with a temperature that grows from
    0.5 to 30).  The gradient is obtained by a single FFT‑based correlation,
    after which the iterate is projected onto the simplex
    {x≥0, Σx = √(2·n)}.  This phase typically yields a sizeable drop in the
    objective.

3.  Aggressive simulated‑annealing refinement – the same mass‑transfer
    annealer used previously, now starting from the gradient‑descent
    result.

All steps respect the global time budget (default 1000 s).  The final
sequence is returned as a plain list of floats and stored in the global
variable ``height_sequence_1`` for warm‑starts in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_full_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using an FFT of power‑of‑2 size."""
    size = 1 << ((u.size + v.size - 1).bit_length())
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fu * fv, n=size)[: u.size + v.size - 1]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    return _conv_full_fft(a, a)


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    b = _conv_self_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = s, x ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n‑1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-lam * x)
    return a


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam·i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = idx ** (-p) * np.exp(-lam * x)
    return a


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape, shifted to avoid zeros."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (alpha - 1.0)
    )
    return a


# ----------------------------------------------------------------------
# Smoothed projected gradient descent (soft‑max of the convolution)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Minimise a soft‑max approximation of max(a*a) with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    # schedule parameters
    alpha_start, alpha_end = 0.5, 30.0          # smoothing parameter
    eta_start, eta_end = 0.5, 0.001           # step‑size (linear decay)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total

        # ---- compute convolution and soft‑max weights ----
        b = _conv_self_fft(a)
        max_b = b.max()
        # soft‑max (stable)
        exp_terms = np.exp(alpha_start + (alpha_end - alpha_start) * frac *
                           (b - max_b))
        p = exp_terms / exp_terms.sum()  # shape (2n‑1,)

        # ---- gradient of the smooth function ----
        a_rev = a[::-1]
        conv_grad = _conv_full_fft(p, a_rev)                # length 3n‑2
        grad = 2.0 * conv_grad[n - 1 : n - 1 + n]           # slice length n

        # ---- step size (linear decay) ----
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # ---- gradient step + projection ----
        a = a - eta * grad
        # occasional jitter to shake out of plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a += jitter
        a = _project_to_simplex(a, target_sum)

        # enforce hard upper bound (evaluate_sequence caps at 1000)
        if a.max() > 1000.0:
            a = np.minimum(a, 1000.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (max‑convolution only)
# ----------------------------------------------------------------------


def _subgradient_projected_descent(a0: np.ndarray, target_sum: float,
                                 deadline: float):
    """Classic sub‑gradient descent on the true max entry of a*a."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_self_fft(a)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        # linear step‑size schedule
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑proj it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Simulated‑annealing with rank‑one mass transfers."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ---- exact O(n) update of the convolution ----
        b_new = b.copy()
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # problem size – larger n gives finer granularity at higher cost
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)  # normalisation used in the benchmark

    # ------------------------------------------------------------------
    # 1. Guided initialisation (deterministic families + random starts)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # warm start from previous run if available
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        # resize to current n (linear interpolation)
        if warm.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, warm.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            warm = np.interp(new_x, old_x, warm)
        warm = warm * (target_sum / warm.sum())
        val = _score(warm)
        best_seq = warm.copy()
        best_score = val
        print(f"[{time.time():.2f}] Warm start score = {best_score:.6f}")

    # deterministic families
    for p in np.linspace(0.45, 0.75, 7):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        if a.max() > 1000.0:
            continue
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} score={val:.6f}")

    for lam in np.linspace(0.5, 8.0, 7):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        if a.max() > 1000.0:
            continue
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.3f} score={val:.6f}")

    for _ in range(8):
        p = np.random.random() * 0.8 + 0.1   # 0.1 … 0.9
        lam = np.random.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        if a.max() > 1000.0:
            continue
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Power‑exp p={p:.3f} λ={lam:.3f} score={val:.6f}")

    for alpha in np.linspace(0.25, 0.9, 7):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        if a.max() > 1000.0:
            continue
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.3f} score={val:.6f}")

    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _score(rnd)
        if val < best_score:
            best_score = val
            best_seq = rnd.copy()
            print(f"[{time.time():.2f}] Random Dirichlet start score={val:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed projected gradient descent (soft‑max of convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.35 * (deadline - time.time())
        a_gd, val_gd = _smooth_gradient_descent(best_seq, target_sum, sub_deadline)
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] GD improved global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Classic sub‑gradient descent (max‑only) – short refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.25 * (deadline - time.time())
        a_sub, val_sub = _subgradient_projected_descent(
            best_seq, target_sum, sub_deadline
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑proj improved global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Aggressive simulated‑annealing mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = _anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] Anneal improved global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # trim trailing zeros for a cleaner output
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # sanity check with the provided evaluate_sequence
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
