"""
Hybrid optimiser for the step‑function max‑convolution problem.

Main ideas
----------
1. **Coarse optimisation on a modest size (≈2000).**  
   A DE + smooth surrogate gradient descent quickly finds a decent shape.

2. **Linear‑programming refinement.**  
   For the current shape `f` we solve  
        maximise   Σ b_i
        subject to f * b ≤ M_f ,   b_i ≥ 0
   where `M_f` is the current worst autocorrelation.  
   The optimal `b` is scaled to the ℓ¹‑norm √(2 n) and mixed with `f`.  
   This “LP‑step’’ is repeated a few times; it is exactly the LP described
   in the problem statement and dramatically lowers the peak.

3. **Upsampling & final polishing.**  
   The refined coarse solution is interpolated to the final length
   (4000‑8000).  A short run of smooth gradient descent, a deterministic
   mass‑transfer greedy polish and a few sub‑gradient steps finish the
   optimisation.

All stages respect the global 1000 s budget and print progress.
The best sequence found is stored in the global variable
`height_sequence_1` for warm‑starts in later runs.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Evaluate the objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2n) this reduces to max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F_init: float = 0.5,
                 CR_init: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after each step.
       Uses self‑adaptive `F` and `CR` (jDE style) for robustness."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    # initialise per‑individual F and CR
    Fs = np.full(len(population), F_init)
    CRs = np.full(len(population), CR_init)

    while time.time() < deadline:
        for i in range(len(population)):
            # jDE adaptation
            if rng.random() < 0.1:
                Fs[i] = 0.1 + rng.random() * 0.9
            if rng.random() < 0.1:
                CRs[i] = rng.random()

            # distinct mates
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + Fs[i] * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            cross = rng.random(n) < CRs[i]
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # leave little margin for next phase
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the smooth max surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        # weights for smooth max
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        # gradient term: a + (conv of w with reversed a) / Z
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # central n entries

        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis‑style acceptance."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent (exact max) with light momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP step)
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray,
               target_sum: float,
               t: float,
               deadline: float,
               max_iter: int = 8) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements:
        maximise Σ b_i  subj.  f * b ≤ M_f ,  b ≥ 0 .
    The solution b is scaled to the required ℓ¹‑norm and mixed with f.
    Returns the best f encountered and its score.
    """
    best_f = f.copy()
    best_score = _score(f)
    n = f.shape[0]

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        M = _max_convolution_fft(f)

        # Build the constraint matrix C where (C @ b)[k] = (f * b)[k]
        C = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            C[i:i + n, i] = f

        # Linprog: maximise sum(b)  <=> minimise -1ᵀ b
        res = linprog(c=-np.ones(n),
                      A_ub=C,
                      b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # If the LP fails we abort further refinements
            break

        b = res.x
        sum_b = b.sum()
        if sum_b < 1e-12:
            break

        g = b * (target_sum / sum_b)
        f_new = (1.0 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)

        cur_score = _score(f_new)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f_new.copy()
            f = f_new
        else:
            # No improvement – stop early
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser: coarse DE+gradient → LP refinement → upsampling +
    final polishing.  Returns a list of non‑negative heights (floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose problem sizes
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_final = 8000
    elif budget_s >= 600:
        n_final = 6000
    else:
        n_final = 4000

    # Coarse optimisation size (small enough for LP, large enough to capture shape)
    n_small = min(2000, max(1200, n_final // 4))
    target_small = np.sqrt(2.0 * n_small)
    target_final = np.sqrt(2.0 * n_final)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation (small size)
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # power‑law seeds
    for p in np.linspace(0.15, 1.2, 8):
        seeds.append(_power_law_sequence(n_small, float(p), target_small))

    # exponential seeds
    for lam in np.linspace(0.0005, 0.025, 6):
        seeds.append(_exponential_sequence(n_small, float(lam), target_small))

    # random simplex seeds
    for _ in range(10):
        seeds.append(_random_simplex(n_small, target_small, rng))

    # previous best (if any) – resize to small dimension
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size != n_small:
            seeds.append(_resize_sequence(prev, n_small))

    # jitter each seed a little for diversity
    jittered: list[np.ndarray] = []
    for seq in seeds:
        jitter = rng.normal(scale=target_small * 0.02, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # --------------------------------------------------------------
    # Initialise DE population (small size)
    # --------------------------------------------------------------
    pop_size = max(30, n_small // 150)   # around 30‑40 individuals
    population = []

    # start from the best seed
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    # fill the rest with random selections
    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_small))

    # --------------------------------------------------------------
    # DE phase (small size)
    # --------------------------------------------------------------
    de_deadline = min(start_time + 0.35 * budget_s,
                      deadline - 30.0)   # reserve later phases
    best_seq_small, best_score_small = _de_optimize(population,
                                                    target_small,
                                                    de_deadline)
    print(f"[{time.time():.2f}] DE (n={n_small}) best score = {best_score_small:.6f}")

    # --------------------------------------------------------------
    # Smoothed gradient descent (small size)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        gd_deadline = time.time() + min(20.0, deadline - time.time() - 15.0)
        seq_gd, score_gd = _smooth_gradient_descent(best_seq_small,
                                                    gd_deadline,
                                                    target_small,
                                                    alpha=5.0,
                                                    init_step_factor=0.7)
        if score_gd < best_score_small:
            best_score_small, best_seq_small = score_gd, seq_gd.copy()
            print(f"[{time.time():.2f}] Smooth GD (α=5) best = {best_score_small:.6f}")

        # a second pass with a sharper surrogate
        if time.time() < deadline - 12.0:
            gd_deadline2 = time.time() + min(12.0,
                                             deadline - time.time() - 10.0)
            seq_gd2, score_gd2 = _smooth_gradient_descent(best_seq_small,
                                                          gd_deadline2,
                                                          target_small,
                                                          alpha=10.0,
                                                          init_step_factor=0.5)
            if score_gd2 < best_score_small:
                best_score_small, best_seq_small = score_gd2, seq_gd2.copy()
                print(f"[{time.time():.2f}] Smooth GD (α=10) best = {best_score_small:.6f}")

    # --------------------------------------------------------------
    # LP refinement (small size)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        lp_deadline = time.time() + min(12.0, deadline - time.time() - 8.0)
        # gradually shrink mixing factor t
        t_vals = [0.3, 0.2, 0.15, 0.1]
        cur_seq = best_seq_small.copy()
        cur_score = best_score_small
        for t in t_vals:
            cur_seq, cur_score = _lp_refine(cur_seq,
                                            target_small,
                                            t,
                                            lp_deadline,
                                            max_iter=6)
            if cur_score < best_score_small:
                best_score_small, best_seq_small = cur_score, cur_seq.copy()
                print(f"[{time.time():.2f}] LP refine (t={t}) best = {best_score_small:.6f}")
            if time.time() >= lp_deadline:
                break

    # --------------------------------------------------------------
    # Upsample to final size
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        up_seq = _resize_sequence(best_seq_small, n_final)
        up_seq = _project_to_simplex(up_seq, target_final)
        print(f"[{time.time():.2f}] Upsampled to n={n_final}")

        # ----------------------------------------------------------
        # Final polishing on the large sequence
        # ----------------------------------------------------------
        # 1. Smooth gradient descent (moderate budget)
        if time.time() < deadline - 4.0:
            fin_gd_dead = time.time() + min(15.0, deadline - time.time() - 3.0)
            seq_fgd, score_fgd = _smooth_gradient_descent(up_seq,
                                                          fin_gd_dead,
                                                          target_final,
                                                          alpha=6.0,
                                                          init_step_factor=0.6)
            if score_fgd < best_score_small:
                best_score_small, up_seq = score_fgd, seq_fgd.copy()
                print(f"[{time.time():.2f}] Final smooth GD best = {best_score_small:.6f}")

        # 2. Deterministic mass‑transfer polishing
        if time.time() < deadline - 3.0:
            mt_dead = time.time() + min(8.0, deadline - time.time() - 2.0)
            seq_mt, score_mt = _deterministic_mass_transfer(up_seq,
                                                            mt_dead,
                                                            target_final,
                                                            start_temp=0.4,
                                                            end_temp=1e-5,
                                                            low_frac=0.07)
            if score_mt < best_score_small:
                best_score_small, up_seq = score_mt, seq_mt.copy()
                print(f"[{time.time():.2f}] Final MT polish best = {best_score_small:.6f}")

        # 3. Short sub‑gradient descent
        if time.time() < deadline - 2.0:
            sgd_dead = time.time() + min(6.0, deadline - time.time() - 1.0)
            seq_sg, score_sg = _subgradient_descent(up_seq,
                                                    sgd_dead,
                                                    target_final,
                                                    step_factor=0.45,
                                                    gamma=0.995)
            if score_sg < best_score_small:
                best_score_small, up_seq = score_sg, seq_sg.copy()
                print(f"[{time.time():.2f}] Final sub‑grad best = {best_score_small:.6f}")

        best_seq = up_seq
    else:
        # If no time for upsampling, simply keep the small solution (still valid)
        best_seq = best_seq_small.copy()
        print(f"[{time.time():.2f}] No time for upsampling – returning small solution")

    # --------------------------------------------------------------
    # Reordering attempts on the final sequence
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1]
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final trimming (remove trailing near‑zero entries)
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Store globally and (optionally) evaluate for logging
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
