"""
Hybrid LP + local refinement for the minimum‐peak convolution problem.

1.  Initialise a random non‑negative sequence on the ℓ¹‑simplex
    (∑a = √(2 n)).
2.  Iteratively solve a linear programme:
       maximise   ∑b_j
       subject to  b_j ≥ 0
                   (f * b)_k ≤ M   for all k
    where M is the current maximal auto‑convolution of f.
    The solution b is rescaled to the same ℓ¹‑norm and blended with f.
    This provably never increases the peak M and usually reduces it
    rapidly.
3.  After a few LP steps a small deterministic “mass‑transfer”
    refinement (the same routine used in the previous baseline) is
    applied.
4.  Finally a few simple re‑orderings are tested and the best is kept.

The whole routine respects the 1000 s budget, works for any n (the
default is n≈1500 – small enough for the LP) and returns a list of
non‑negative heights ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple factories for initial seeds
# ----------------------------------------------------------------------


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Reordering helpers (simple but often useful)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, …."""
    vals = np.sort(seq)[::-1]
    n = vals.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (same as previous baseline)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> np.ndarray:
    a = a0.copy()
    conv = _conv_fft(a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_val = cur_max
    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_max = float(conv.max())

        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a


# ----------------------------------------------------------------------
# Linear‑programming step (core of the new method)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that
        (f * b)_k = Σ_j A[k, j] * b_j .
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # fill the Toeplitz‑like structure: for each i we set the diagonal
    # A[i + j, j] = f[i]   (j = 0..n‑1)
    j = np.arange(n)
    for i in range(n):
        A[i:i + n, j] = f[i]
    return A


def _lp_improve(f: np.ndarray,
                target_sum: float,
                deadline: float,
                blend_factor: float = 0.5,
                max_iters: int = 15) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described in the docstring, blend the result
    with the current sequence, and stop when time runs out or no further
    improvement is observed.
    Returns the best sequence and its peak value.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_M = _max_conv_fft(f)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # Build constraints for the current f
        A = _build_conv_matrix(best_f)
        M = best_M
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise Σ b_j   ⇔   minimise −Σ b_j
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c,
                      A_ub=A,
                      b_ub=ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            # something went wrong; abort the LP stage
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # Rescale b to the required ℓ¹‑norm
        scale = target_sum / sum_b
        g = b * scale

        # Blend
        new_f = (1.0 - blend_factor) * best_f + blend_factor * g
        new_f = _project_to_simplex(new_f, target_sum)

        new_M = _max_conv_fft(new_f)

        if new_M < best_M * (1 - 1e-6):          # tangible improvement
            best_f = new_f
            best_M = new_M
            # continue with the improved sequence
        else:
            # No progress – stop early
            break

    return best_f, best_M

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimisation based on a small LP‑driven reduction of the peak
    convolution followed by a deterministic mass‑transfer polish.
    The routine adapts the problem size to the time budget, stores the
    best sequence for warm‑starting the next run, and prints progress.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a tiny safety margin

    # --------------------------------------------------------------
    # Choose a modest problem size that keeps the LP tractable.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 1800
    elif budget_s >= 500:
        n = 1400
    else:
        n = 1000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation – a few diverse shapes.
    # --------------------------------------------------------------
    seeds = [
        _random_simplex(n, target_sum, rng),
        _power_law_sequence(n, 0.5, target_sum),
        _power_law_sequence(n, 0.9, target_sum),
        _exponential_sequence(n, 0.005, target_sum),
        _exponential_sequence(n, 0.015, target_sum),
    ]

    # Warm‑start if a previous run exists (resize if needed)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        seeds.append(_project_to_simplex(prev, target_sum))

    # --------------------------------------------------------------
    # Pick the best seed as the starting point for the LP stage.
    # --------------------------------------------------------------
    best_seed = min(seeds, key=_max_conv_fft)
    f = _project_to_simplex(best_seed, target_sum)

    # --------------------------------------------------------------
    # LP‑driven improvement (≈ 45 % of the budget).
    # --------------------------------------------------------------
    lp_deadline = start_time + 0.45 * budget_s
    if lp_deadline > deadline - 5.0:
        lp_deadline = deadline - 5.0

    f, M = _lp_improve(f, target_sum, lp_deadline,
                       blend_factor=0.5,
                       max_iters=12)

    print(f"[{time.time():.2f}] LP stage – peak = {M:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer polishing (the remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_deadline = deadline
        f_mt = _deterministic_mass_transfer(f, mt_deadline, target_sum,
                                            start_temp=0.4,
                                            end_temp=1e-6,
                                            low_frac=0.07)
        M_mt = _max_conv_fft(f_mt)
        if M_mt < M:
            f, M = f_mt, M_mt
            print(f"[{time.time():.2f}] MT polish – peak = {M:.6f}")

    # --------------------------------------------------------------
    # Simple re‑ordering tricks (often give a last tiny boost).
    # --------------------------------------------------------------
    candidates = [
        f,
        _alternating_order(f),
        _even_spread_order(f)
    ]
    f = min(candidates, key=_max_conv_fft)
    M = _max_conv_fft(f)
    print(f"[{time.time():.2f}] After reordering – final peak = {M:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness.
    # --------------------------------------------------------------
    arr = np.array(f, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store for the next run
    globals()["height_sequence_1"] = final_seq

    # Sanity check with the provided evaluator (optional)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
