"""
Hybrid optimiser with simulated‑annealing polishing.
The pipeline:

1. Load the previous best construction (global ``height_sequence_1``) or
   generate a few simple seeds (power‑law, exponential, random simplex).
2. Project the chosen seed onto the ℓ¹‑simplex Σa = √(2 n).
3. Smooth‑gradient descent on a log‑sum‑exp surrogate, executed with a
   continuation of the smoothing parameter α (4 → 10 → 20) to get a
   good local minimum.
4. Heavy simulated‑annealing (SA) polishing:
   – At each step the current convolution peak kₘₐₓ is identified.
   – The index i that contributes most to this peak is taken as a donor.
   – A receiver index is sampled among the low‑contributing indices.
   – A small amount of mass δ is moved from donor to receiver.
   – The new sequence is accepted if the peak decreases, otherwise with
     probability exp(-(Δpeak)/T) where the temperature T decays from
     30 % of the current peak to 10⁻⁴ % by the end of the SA phase.
5. A final re‑ordering pass (original, alternating, even‑spread, reversal)
   chooses the ordering with the smallest peak.
6. Trailing near‑zero entries are trimmed, the result is stored in the
   global ``height_sequence_1`` and returned as a plain list of floats.

All steps respect the global time budget (default 1000 s).  The
objective evaluated by ``evaluate_sequence`` equals the maximal entry
of the autocorrelation when the ℓ¹‑norm is √(2 n).  The aim is to push
this maximal value below the known bound 1.506815.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2n) the denominator cancels and the score equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto
    {x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise `log_sum_exp(alpha·(a*a)) / alpha` using projected gradient descent.
    The surrogate approaches the true max as `alpha → ∞`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        # smooth weights (stable because conv-M ≤ 0)
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        # compute gradient of the surrogate
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # slice length n
        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑Annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer SA that attempts to lower the max autocorrelation.
    Returns the best sequence found and its maximal convolution value.
    """
    n = a0.shape[0]
    a = _project_to_simplex(a0, target_sum).copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_seq = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = max(deadline - start, 1e-12)
    rng = np.random.default_rng()

    # temperature schedule: start at 30% of cur_max, end at 0.01% of cur_max
    T0 = cur_max * 0.30
    T_final = cur_max * 0.0001

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total_time
        # exponential decay of temperature
        T = T0 * (T_final / T0) ** progress

        # 1) locate the current peak
        kmax = int(np.argmax(conv))

        # 2) donor = index strongly contributing to this peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[kmax - idx]
        donor = int(idx[int(np.argmax(prod))])

        # 3) pick a receiver among low‑contributing indices
        low_cnt = max(1, int(len(idx) * 0.08))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        # 4) propose a transfer of a small amount of mass
        max_delta = min(a[donor], a[receiver]) * 0.5
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new[donor] = max(a_new[donor], 0.0)

        # 5) evaluate new peak
        conv_new = _conv_self(a_new)
        new_max = conv_new.max()

        # 6) Metropolis acceptance
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_seq = a.copy()
        # occasional re‑projection to keep the sum numerically exact
        if rng.random() < 0.001:
            a = _project_to_simplex(a, target_sum)
            conv = _conv_self(a)
            cur_max = conv.max()
    return best_seq, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser: initial seeds → smooth gradient descent → simulated‑annealing.
    Returns a list of non‑negative heights whose evaluation score is as low
    as possible within the allotted time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # problem dimension (larger n gives more flexibility)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation (use previous best if available)
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.15, 1.2, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        seeds.append(prev)

    # jitter each seed for diversity
    jittered: list[np.ndarray] = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # --------------------------------------------------------------
    # pick the best seed as a starting point
    # --------------------------------------------------------------
    best_seq = min(seeds, key=_score)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Smooth gradient descent (continuation of α)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate up to 30 s for each α‑stage, respecting the global deadline
        for alpha, stage_time in [(4.0, 15.0), (10.0, 15.0), (20.0, 15.0)]:
            if time.time() >= deadline - 5.0:
                break
            stage_dead = time.time() + min(stage_time, deadline - time.time() - 5.0)
            seq_gd, sc_gd = _smooth_gradient_descent(best_seq,
                                                     stage_dead,
                                                     target_sum,
                                                     alpha=alpha,
                                                     init_step_factor=0.7)
            if sc_gd < best_score:
                best_score = sc_gd
                best_seq = seq_gd.copy()
                print(f"[{time.time():.2f}] After smooth α={alpha:.0f} best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Simulated‑Annealing polishing (core improvement phase)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        sa_seq, sa_max = _simulated_annealing(best_seq, deadline, target_sum)
        sa_score = float(sa_max)                      # when Σa = √(2n), score = max_b
        if sa_score < best_score:
            best_score = sa_score
            best_seq = sa_seq.copy()
            print(f"[{time.time():.2f}] After SA best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Reordering attempts (may lower the peak further)
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1],
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Final trimming and storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
