def _adam_descent(a0, deadline, target_sum, beta_initial=0.5, beta_max=1e6, beta_mult=2.0, lr=0.1, epsilon=1e-8):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    it = 0
    beta = beta_initial
    step_base = target_sum*0.5  # base step factor

    size = 1 << ((2*a.shape[0] - 1).bit_length())
    rng = np.random.default_rng()

    while time.time() < deadline:
        n = a.shape[0]
        # compute FFT and convolution
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        max_c = conv.max()
        # softmax with beta
        exps = np.exp(beta * (conv - max_c))
        p = exps / np.sum(exps)

        # gradient
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        grad_rev = np.fft.irfft(rev_fft * p_fft, n=size)[:2*n-1]
        grad = 2.0 * grad_rev[n-1:n-1+n]

        # Adam updates
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step = step_base / (np.sqrt(v_hat) + epsilon) * m_hat

        a -= step
        a = _project_to_simplex(a, target_sum)

        # evaluate
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(...)

        # schedule beta
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta*beta_mult, beta_max)

        # jitter occasionally
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

"""
Hybrid optimizer for the step‑function benchmark (improved version).

Key ideas
--------
* Use a much larger support (≈20 k–30 k) – more “room’’ to decorrelate large
  entries.
* Start from strong parametric families (exponential, power‑law, random)
  and immediately apply a permutation that pairs the largest values with the
  smallest ones in the reversed index (the “large‑small’’ ordering).  This
  dramatically reduces the centre convolution value.
* Perform a soft‑max (annealed) gradient descent with momentum, followed by
  an exact sub‑gradient descent on the large problem.
* Finish with an aggressive simulated‑annealing mass‑transfer phase and a
  permutation hill‑climb.
* All updates are projected onto the ℓ¹‑simplex (sum = √(2 n)) after every
  step, guaranteeing a valid candidate at any time.

The routine respects the 1000 s budget and always returns the best found
sequence.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (various parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Permutation helpers
# ----------------------------------------------------------------------


def _pair_large_small_order(seq: np.ndarray) -> np.ndarray:
    """
    Permute the entries so that the largest values are paired with the
    smallest values in the reversed index (i ↔ n‑1‑i).  This minimises
    the centre convolution term.
    """
    vals = np.sort(seq)[::-1]               # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    i, j = 0, n - 1
    lo, hi = 0, len(vals) - 1
    while i < j:
        out[i] = vals[lo]
        out[j] = vals[hi]
        lo += 1
        hi -= 1
        i += 1
        j -= 1
    if i == j:
        out[i] = vals[lo]   # the remaining median element
    return out


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Randomly shuffle and keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """Swap two random indices, keep improvement."""
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent – same as before
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected GD on the soft‑max approximation of the max‑convolution.
    Slightly more conservative step factor / momentum than the original.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        # FFT of current a and its reverse
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_c = conv.max()
        exps = np.exp(beta * (conv - max_c))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        grad_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * grad_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        cur_val = max_c  # exact max‑convolution value
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        # schedule β
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.55,
                         gamma: float = 0.995,
                         tol: float = 0.995) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        conv = np.fft.irfft(np.fft.rfft(a, n=size) *
                            np.fft.rfft(a, n=size), n=size)[: 2 * n - 1]
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        cur_val = maxb
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake the system
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

    else:

        best_seq = _random_simplex(n_big, target_big, rng)
    else:
        # upscale via linear interpolation and re‑project
        best_seq = _resize_sequence(best_seq, n_big)
        best_seq = _project_to_simplex(best_seq, target_big)

    # Apply the large‑small permutation right after upscaling.
    best_seq = _pair_large_small_order(best_seq)

    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(best_seq):.6f}")

    # --------------------------------------------------------------

                                                   tol=0.995)
        if sub_score < best_score:

        print(f"[{time.time():.2f}] After random permutations max={_max_convolution_fft(best_seq):.6f}")

    # --------------------------------------------------------------
    # 6. Local permutation hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline - 50.0:
        hill_deadline = time.time() + min(10.0, deadline - time.time() - 60.0)
        best_seq, _ = _local_permutation_hillclimb(best_seq, hill_deadline)
        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(best_seq):.6f}")

    # --------------------------------------------------------------
    # 7. Final greedy mass‑transfer (simulated‑annealing)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        mt_deadline = time.time() + min(15.0, deadline - time.time() - 30.0)
        best_seq, mt_score = _mass_transfer_simulated_annealing(best_seq,
                                                               mt_deadline,
                                                               target_big,
                                                               start_temp=0.35,
                                                               end_temp=1e-5,
                                                               low_frac=0.07)
        if mt_score < best_score:
            best_score = mt_score
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # 8. Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    # expose for reuse in later runs
    globals()["height_sequence_1"] = final_seq

    # Optional: evaluate for logging (ignore if not available)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
