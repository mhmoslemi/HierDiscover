def _conv_self(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

def _smooth_gradient_descent(a0, deadline, target_sum, alpha):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum) # ensure sum correct.
    best_a = a.copy()
    best_score = _score(a)
    # compute step factor adaptively
    step_factor = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        M = conv.max()
        # compute weights
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        # compute Tg
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)  # we need to implement _fft_convolve
        Tg = conv_w_rev[len(w)-1:len(w)-1 + len(a)] # Wait conv length? Actually using same as earlier
        grad = (2.0 * Tg) / Z
        # A direct gradient that points toward increase; we move opposite.
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)
        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)
        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.05, 2.0)
        else:
            step_factor = max(step_factor * 0.7, 1e-4)
        a = a_new
    return best_a, best_score

def _simulated_annealing(a0, deadline, target_sum):
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    # parameters
    start_temp = 0.5
    end_temp = 1e-4
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau
        # compute conv and kmax
        conv = _conv_self(a)
        kmax = int(conv.argmax())
        # compute contributions
        idx = np.arange(len(a))
        other = kmax - idx
        valid = (other >= 0) & (other < len(a))
        contributions = np.zeros_like(a)
        contributions[valid] = a[idx] * a[other[valid]]
        donor = int(np.argmax(contributions))
        # choose receiver from low contributions
        low_idx = np.where(contributions == contributions.min())[0]
        # maybe pick randomly among low
        receiver = int(rng.choice(low_idx))
        if donor == receiver:
            continue
        max_delta = min(a[donor] * 0.5, target_sum * 0.5)  # ensure not too large
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # no need to project because sum unchanged; but enforce nonnegative (ensured)
        new_score = _score(a_new)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score

"""
Hybrid random/gradient/annealing optimiser for the step‑function
max‑convolution problem.

The algorithm proceeds in three phases:

1. **Diverse seeding** – a handful of deterministic families
   (power‑law, exponential) together with many random Dirichlet‑like
   points on the ℓ¹‑simplex Σa = √(2 n).  The best seed becomes the
   starting point.

2. **Smooth‑max gradient descent** – minimise
   `log‑sum‑exp(α·(a*a)) / α` with a correct analytic gradient.
   The surrogate approaches the true max as the temperature α grows.
   A schedule α = 4, 8, 16, 32, 64 is applied, each stage using a
   fraction of the remaining time.

3. **Simulated‑annealing polishing** – a simple mass‑transfer move
   (donor → receiver) that respects the ℓ¹‑norm and non‑negativity.
   The move is accepted if it lowers the exact max‑convolution or,
   with a temperature‑dependent probability, otherwise.  The temperature
   decays logarithmically.

Finally a few deterministic reorderings are tried and the best
configuration is returned.  All helper routines are top‑level,
use only NumPy and the provided `evaluate_sequence`, and respect the
global time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective & projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Exact objective  max_k (a*a)_k .
    The vector is assumed to satisfy Σa = √(2·n)   (the optimiser
    keeps the sum exact, so the evaluation reduces to the max‑convolution).
    """
    max_b = _conv_self(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(max_b)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple seed factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_dirichlet(n: int, target_sum: float, rng: np.random.Generator,
                      shape: float = 1.0) -> np.ndarray:
    """
    Uniform (shape=1) or sparse (shape<1) Dirichlet point on the ℓ¹‑simplex.
    """
    a = rng.gamma(shape, 1.0, size=n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (exact gradient)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 8.0,
                             init_step: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate  log‑sum‑exp(α·(a*a))/α.
    The gradient is derived analytically:
        ∇ = 2 * (w * a_rev)[n‑1:n‑1+n] / Z ,   Z = Σ w .
    The iterate is projected onto the ℓ¹‑simplex after each step.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step
    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2·n‑1
        M = conv.max()
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3·n‑2
        Tg = conv_w_rev[len(w) - 1:len(w) - 1 + len(a)]   # slice of length n
        grad = (2.0 * Tg) / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing mass‑transfer
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from a “donor” index (large contribution
    to the current peak) to a “receiver” index (small contribution).
    Accept downhill moves always; uphill moves with a Metropolis probability.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_self(a)
        kmax = int(conv.argmax())

        idx = np.arange(n)
        other = kmax - idx
        valid = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[idx[valid]] * a[other[valid]]

        donor = int(np.argmax(contrib))
        # pick a receiver among the low‑contributing indices
        low_candidates = np.where(contrib == contrib.min())[0]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        # transfer a modest amount of mass
        max_delta = min(a[donor] * 0.5, target_sum * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # sum is unchanged, all entries stay non‑negative

        new_score = _score(a_new)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score
