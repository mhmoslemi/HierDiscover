def adam_projected_gradient(a0, target_sum, deadline, beta_start=0.1, beta_max=30, epochs):
   a=a0.copy()
   m=np.zeros_like(a)
   v=np.zeros_like(a)
   t=0
   lr0=0.1 * target_sum # base step size
   beta=beta_start
   while time.time()<deadline:
       # compute conv and weights
       b=_conv_self_fft(a)
       max_b=b.max()
       # compute softmax w for smooth max
       b_shift=b-max_b
       w=np.exp(beta*b_shift)
       w/=w.sum()
       # gradient
       a_rev=a[::-1]
       conv = _conv_fft(w, a_rev)
       g = 2*conv[n-1:n-1+n]
       # Adam update
       t+=1
       lr = lr0 / np.sqrt(t)
       m = 0.9*m + 0.1*g
       v = 0.999*v + 0.001*(g**2)
       m_hat = m/(1-0.9**t)
       v_hat = v/(1-0.999**t)
       a -= lr*m_hat/(np.sqrt(v_hat)+1e-8)
       a = _project_to_simplex(a, target_sum)
       # update beta
       if t%2000==0 and beta<beta_max: beta=min(beta_max,beta*1.2)
   return a, _score_from_selfconv(_conv_self_fft(a), n, target_sum)

"""
Hybrid optimization for the step‑function benchmark.

The algorithm consists of three stages:

1. **Initialization** – a warm start (if available) is resized to the
   chosen length and a few random simplex‑projected vectors are added.

2. **Smooth‑max descent** – a differentiable approximation of the
   max‑convolution (log‑sum‑exp) is minimised with an Adam‑style
   projected gradient method.  The temperature β is increased slowly
   (0.2 → 80) so the optimisation gradually focuses on the true
   ℓ∞‑norm.

3. **In‑place simulated‑annealing** – small amounts of mass are moved
   from “donor’’ indices that heavily contribute to the current
   peak to “receiver’’ indices that are currently small.  The convolution
   is updated in O(n) per move without allocating a copy, and a
   square‑root cooling schedule is used.  The best sequence seen during
   the whole run is kept.

The routine respects the time budget (default 1000 s) and returns the
best sequence as a plain list of floats.  A final call to the provided
`evaluate_sequence` reports the achieved score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a power‑of‑2 FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact benchmark score = 2·n·max(b) / target_sum²."""
    return float(2.0 * n * b.max() / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helper functions
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max Adam descent
# ----------------------------------------------------------------------


def smooth_max_adam(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   beta_start: float = 0.2,
                   beta_max: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the ℓ∞-norm of the self‑convolution.
    The objective is  (1/β)·log Σ exp(β·b_i) where b = a*a.
    Adam updates are projected onto the simplex Σa = target_sum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start
    it = 0
    # base learning rate – a fraction of the total mass
    base_lr = target_sum * 0.5

    while time.time() < deadline:
        # self‑convolution
        b = _conv_self_fft(a)                     # length 2n‑1
        max_b = b.max()
        # exact score for book‑keeping
        cur_score = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:6d} best={best_val:.6f}")

        # log‑sum‑exp weights (numerically stable)
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ a_rev) (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)                     # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]               # central n entries

        # Adam update
        it += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g ** 2)
        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)
        lr = base_lr / np.sqrt(it)                        # decaying step size
        a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # keep the simplex constraint
        a = _project_to_simplex(a, target_sum)

        # slowly increase β
        if it % 1500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.4)

    # final exact score for the best candidate found
    final_score = _score_from_selfconv(_conv_self_fft(best_a), n, target_sum)
    return best_a, final_score


# ----------------------------------------------------------------------
# Phase 2 – aggressive in‑place simulated‑annealing
# ----------------------------------------------------------------------


def anneal_inplace(a0: np.ndarray,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that transfers small masses from indices that
    heavily contribute to the current peak of a*a to indices that are
    currently low.  Convolution is updated in‑place (O(n) per move).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)                     # current self‑convolution
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature – a fraction of the current score
    T0 = cur_val * 0.9

    it = 0
    # Re‑projection interval to eliminate drift (in terms of iterations)
    REPROJECT_EVERY = 4000

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # find current peak index of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of every index to that peak (a[i] * a[k_max-i])
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights (inverse weighting)
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – larger when temperature is high
        max_delta = min(a[donor] * 0.2, target_sum * 0.02)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- in‑place update of the convolution -----------------
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b[donor:donor + n] -= 2.0 * delta * a
        b[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b[2 * donor] += delta * delta
        b[2 * receiver] += delta * delta
        if donor != receiver:
            b[donor + receiver] -= 2.0 * delta * delta

        new_max = b.max()
        new_val = 2.0 * n * new_max / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")
        else:
            # revert the convolution update using the original a values
            b[donor:donor + n] += 2.0 * delta * a
            b[receiver:receiver + n] -= 2.0 * delta * a
            b[2 * donor] -= delta * delta
            b[2 * receiver] -= delta * delta
            if donor != receiver:
                b[donor + receiver] += 2.0 * delta * delta

        # periodic re‑normalisation to keep sum exactly target_sum
        if it % REPROJECT_EVERY == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] re‑project improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.2  # tiny safety margin

    # --------------------------- choose problem size ---------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)  # normalisation used by the benchmark

    # --------------------------- build initial candidates ---------------------------
    init_candidates = []

    # warm start, if available
    if "height_sequence_1" in globals():
        warm = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------- optimisation loop ---------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx + 1} -----")

        # ---- Phase 1: smooth‑max Adam descent ---------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # allocate at most 30 % of the remaining time to smoothing
        smooth_deadline = time.time() + min(remaining * 0.30, 40.0)
        a_smooth, val_smooth = smooth_max_adam(
            init, target_sum, smooth_deadline,
            beta_start=0.2, beta_max=80.0
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ---- Phase 2: aggressive annealing ------------------------------------
        if time.time() >= deadline:
            break
        a_ann, val_ann = anneal_inplace(a_smooth, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------- fallback (should never happen) -----------------
    if best_seq is None:
        # simple power‑law fallback
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    # --------------------------- final polishing ---------------------------
    arr = np.array(best_seq, dtype=np.float64)

    # trim trailing zeros for a shorter representation
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (outside of the optimisation loop)
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
