"""
Hybrid optimizer for the step‑function benchmark (v2).

Main ideas
----------
1.  Broad parametric search on a modest size (≈4 000) that now also
    includes *truncated* exponential / power‑law families.  More
    aggressive grids (very small decay rates, many support lengths)
    give a richer set of seeds.

2.  The best seed is refined by an Adam‑style sub‑gradient descent on
    the exact max‑convolution (no soft‑max approximation).  Momentum
    and adaptive step sizes make the descent far more stable than the
    plain momentum GD used before.

3.  The refined sequence is interpolated to a larger support
    (≈20 000).  Because the shape is already monotone decreasing we
    keep the ordering – no random shuffling – which is crucial for
    keeping the inner product ‑ a·a_rev – small.

4.  A second, longer Adam sub‑gradient phase finishes the optimisation.
    Afterwards a short simulated‑annealing mass‑transfer pass removes
    any remaining local spikes.

5.  Zero entries at the tail are trimmed before returning the result.

The whole pipeline respects the non‑negativity and ℓ¹‑simplex
constraints and always projects back onto the simplex after every
update.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (including truncated families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _power_law_truncated_sequence(n: int, p: float, m: int,
                                 target_sum: float) -> np.ndarray:
    """Power‑law on the first *m* entries, zeros afterwards."""
    a = np.zeros(n, dtype=np.float64)
    a[:m] = (np.arange(1, m + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _exponential_truncated_sequence(n: int, lam: float, m: int,
                                   target_sum: float) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[:m] = np.exp(-lam * np.arange(m, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _linear_sequence(n: int, m: int, target_sum: float) -> np.ndarray:
    """
    Decreasing linear ramp on the first *m* entries,
    zero afterwards.
    """
    a = np.zeros(n, dtype=np.float64)
    if m > 1:
        ramp = np.linspace(1.0, 0.0, m, dtype=np.float64)
        a[:m] = ramp
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient descent for the exact max‑convolution
# ----------------------------------------------------------------------


def _adam_subgradient_descent(a0: np.ndarray,
                              deadline: float,
                              target_sum: float,
                              base_lr: float = 0.6,
                              beta1: float = 0.9,
                              beta2: float = 0.999,
                              eps: float = 1e-8,
                              tol: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the exact max‑convolution (sub‑gradient).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        conv = _conv_fft(a)
        maxb = conv.max()
        # mask of near‑max entries (helps with non‑uniqueness)
        I = (conv >= maxb * tol).astype(np.float64)

        # sub‑gradient of the max‑operator
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)
        grad_full = grad_full[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = base_lr * target_sum / (np.sqrt(v_hat) + eps)  # adaptive per‑coordinate
        a -= lr * m_hat

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.3,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # tiny random jitter occasionally
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    Uses a richer parametric seed set, Adam sub‑gradient descent, and a
    final simulated‑annealing mass‑transfer pass.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1. Broad parametric search on a modest size (≈4 000)
    # --------------------------------------------------------------
    n_expl = 4000
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool = []

    # 1a. Full‑support families
    for lam in np.logspace(-5, -2, 12):               # exponential
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for p in np.linspace(0.05, 2.5, 20):              # power law
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # 1b. Truncated families – varying support length
    for frac in [0.30, 0.45, 0.60, 0.75, 0.90]:
        m = max(2, int(frac * n_expl))
        for lam in np.logspace(-5, -2, 8):
            init_pool.append(_exponential_truncated_sequence(n_expl,
                                                             float(lam), m,
                                                             target_expl))
        for p in np.linspace(0.05, 2.0, 12):
            init_pool.append(_power_law_truncated_sequence(n_expl,
                                                            float(p), m,
                                                            target_expl))
        init_pool.append(_linear_sequence(n_expl, m, target_expl))

    # 1c. Random simplex seeds ( Useful for escaping local minima )
    for _ in range(15):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # 1d. Jittered variants of everything gathered so far
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Pick the best sequence from the pool (cheap evaluation)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in init_pool:
        cur = _max_convolution_fft(seq)
        if cur < best_val:
            best_val = cur
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Seed pool best max={best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Adam sub‑gradient descent on the modest size
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        adam_deadline = time.time() + min(120.0,
                                          deadline - time.time() - 120.0)
        best_seq, best_val = _adam_subgradient_descent(best_seq,
                                                       adam_deadline,
                                                       target_expl,
                                                       base_lr=0.8)
        print(f"[{time.time():.2f}] After Adam on small size best={best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Upscale to a larger problem size (≈20 k)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200.0:
        n_big = 25000
    elif remaining > 120.0:
        n_big = 20000
    elif remaining > 80.0:
        n_big = 15000
    else:
        n_big = n_expl

    target_big = math.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_seq if best_seq is not None
                               else _exponential_sequence(n_expl, 0.001, target_expl),
                               n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # Keep monotone decreasing order (the shape that was optimised)
    seq_big = np.sort(seq_big)[::-1]

    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 4. Second, longer Adam sub‑gradient descent on the large scale
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        big_adam_deadline = deadline - 80.0
        seq_big, big_val = _adam_subgradient_descent(seq_big,
                                                     big_adam_deadline,
                                                     target_big,
                                                     base_lr=0.6)
        if big_val < best_val:
            best_val = big_val
            best_seq = seq_big.copy()
            print(f"[{time.time():.2f}] Large‑scale Adam best={best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final greedy mass‑transfer (simulated‑annealing)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        mt_deadline = deadline - 30.0
        seq_big, mt_score = _mass_transfer_simulated_annealing(seq_big,
                                                               mt_deadline,
                                                               target_big,
                                                               start_temp=0.25,
                                                               end_temp=1e-5,
                                                               low_frac=0.05)
        if mt_score < best_val:
            best_val = mt_score
            best_seq = seq_big.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq  # make it available for the next run

    # sanity evaluation (optional – may be omitted in final run)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
