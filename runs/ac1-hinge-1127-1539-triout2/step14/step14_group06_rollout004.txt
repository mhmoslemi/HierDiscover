"""
Propose a non‑negative step function minimizing the max self‑convolution.

Algorithm
---------
0.  Choose a modest size (n≈2000) for fast linear‑programming.
1.  Warm‑start: scan a few simple parametric families (power‐law,
    exponential, two‑level piecewise) and keep the best candidate.
2.  LP refinement (the “exchange” step from the paper):
    * Let M = max(f∗f).  Solve the LP
          maximise 1·g                      (i.e. sum(g))
            s.t.   f∗g ≤ M   (elementwise)
                   g ≥ 0
      The LP is linear because (f∗g)_k = Σ_i f_i g_{k‑i}.
    * Scale g to the required simplex sum √(2n) and replace
      f ← (1‑t)·f + t·g (t≈0.5).  Iterate while the score improves.
3.  Upsample the refined vector to a large n (≈12000) by linear
    interpolation and re‑project onto the simplex.
4.  Final polishing: projected Adam gradient descent on a soft‑max
    approximation of the max‑convolution, using the same simplex
    projection as before.
5.  Trim trailing near‑zero entries and return the list.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a∗a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective used by the competition: 2·n·max(b)/(sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the cheap warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_decay_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, m_frac: float, r: float, target_sum: float) -> np.ndarray:
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)  # (m*h + (n-m)*l) = target_sum with h = r*l
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a


# ----------------------------------------------------------------------
# Build the sparse matrix for the linear constraints  f∗g ≤ M
# ----------------------------------------------------------------------


def _build_conv_ub_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Return a sparse matrix A of shape (2n‑1, n) such that (A @ g)[k]
    equals (f∗g)[k] for linear convolution.
    """
    n = f.shape[0]
    # Each offset i contributes a diagonal with value f[i]
    data = []
    offsets = []
    for i, val in enumerate(f):
        if val == 0.0:
            continue
        # a constant column of length n with value `val`
        data.append(np.full(n, val, dtype=np.float64))
        offsets.append(i)          # row = col + i
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ----------------------------------------------------------------------
# LP refinement step (exchange algorithm)
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the linear‑programming exchange step.
    Returns the refined vector and its score.
    """
    best = f.copy()
    best_score = _score(best)
    n = f.shape[0]

    while time.time() < deadline - 1.0:
        # 1) current bound M
        M = _max_convolution_fft(best)

        # 2) build sparse constraint matrix A (f∗g ≤ M)
        A = _build_conv_ub_matrix(best)

        # 3) set up LP
        # maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # give linprog a soft time limit (highs respects `time_limit`)
        remaining = deadline - time.time()
        linprog_opts = {"presolve": True}
        if remaining < 300:
            linprog_opts["time_limit"] = remaining

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options=linprog_opts,
        )

        if not res.success:
            # nothing more can be done
            print(f"[{time.time():.2f}] LP failed ({res.message}), stopping refinement.")
            break

        g = res.x
        if g.sum() == 0.0:
            break
        # 4) scale g to the required simplex sum
        g = g * (target_sum / g.sum())

        # 5) convex combination (t = 0.5 works well in practice)
        t = 0.5
        new_vec = (1.0 - t) * best + t * g
        new_vec = np.maximum(new_vec, 0.0)
        new_vec = _project_to_simplex(new_vec, target_sum)

        new_score = _score(new_vec)
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best = new_vec
            print(f"[{time.time():.2f}] LP iter improved score = {best_score:.6f}")
        else:
            # no improvement – stop early
            break

    return best, best_score


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns the gradient of the soft‑max approximation of max(conv(a,a))
    w.r.t. a, together with the raw convolution values.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) ∗ p) restricted to original length
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex, minimizing the soft‑max approximation.
    Returns the best vector seen and its exact score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Run a few Adam restarts from jittered starts."""
    best = init.copy()
    best_val = _score(best)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best.shape)
        start = best + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximum
    self‑convolution.  The routine respects the budget (default 1000 s)
    and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size for the cheap LP stage (small enough for a dense LP)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_small = 2000
        n_large = 12000
    elif budget_s >= 400:
        n_small = 1500
        n_large = 8000
    else:
        n_small = 1000
        n_large = 5000

    target_small = np.sqrt(2.0 * n_small)
    target_large = np.sqrt(2.0 * n_large)

    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0) Warm‑start via parametric families (few grids – fast)
    # ------------------------------------------------------------------
    for p in np.linspace(0.0, 2.0, 11):
        seq = _power_law_sequence(n_small, float(p), target_small)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    for q in np.linspace(0.0, 20.0, 11):
        seq = _exp_decay_sequence(n_small, float(q), target_small)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    for m_frac in (0.05, 0.10, 0.20):
        for r in (5, 15, 30):
            seq = _piecewise_sequence(n_small, float(m_frac), float(r), target_small)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # a few random jitters around the best so far
    for _ in range(5):
        jitter = rng.normal(scale=target_small * 0.005, size=n_small)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_small)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] Warm‑start best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) LP exchange refinement (core of the method from the statement)
    # ------------------------------------------------------------------
    lp_deadline = time.time() + (deadline - time.time()) * 0.30
    refined_seq, refined_score = _lp_refine(best_seq, target_small, lp_deadline)
    if refined_score < best_score:
        best_score = refined_score
        best_seq = refined_seq
        print(f"[{time.time():.2f}] After LP refinement score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Upsample to a large n for final polishing
    # ------------------------------------------------------------------
    old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
    new_x = np.linspace(0.0, 1.0, n_large)
    up_seq = np.interp(new_x, old_x, best_seq)
    up_seq = _project_to_simplex(up_seq, target_large)

    # keep the up‑sampled version as the current best
    best_seq = up_seq
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After up‑sampling score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing with Adam‑GD (soft‑max) and a few restarts
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        gd_deadline = time.time() + (deadline - time.time()) * 0.40
        polished_seq, polished_score = _gradient_descent_with_restarts(
            best_seq, target_large, gd_deadline
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq
            print(f"[{time.time():.2f}] Final Adam polishing improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Very short final Adam run with larger beta (aggressive)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        final_seq, final_score = _adam_gradient_descent(
            best_seq,
            deadline,
            target_large,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq
            print(f"[{time.time():.2f}] Final aggressive Adam improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_list = arr.tolist()

    # optional verification (harmless if evaluate_sequence not present)
    try:
        val = evaluate_sequence(final_list)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_list
