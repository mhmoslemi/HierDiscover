"""
Optimisation for minimizing the maximal self‑convolution of a non‑negative
step‑function under a fixed ℓ₁‐norm.

The algorithm proceeds in several stages, all respecting the overall time
budget:

1. **Parametric warm‑start** – a fast scan of several one‑parameter families
   (power‑law, tilted power‑law, exponential, double‑exponential,
   Gaussian and two‑level piecewise) to obtain a decent initial sequence.

2. **Optional symmetrisation** – averaging with the reversed sequence often
   lowers the peak of the convolution.

3. **Projected Adam (soft‑max) gradient descent** – minimises a smooth
   approximation of the max‑norm while staying on the simplex
   Σa=√(2 n). Multiple random restarts are performed.

4. **Deterministic mass‑transfer (DMT2)** – a greedy move that reduces the
   current convolution peak by moving mass from a donor index to a receiver
   index using the exact feasible step size (no artificial 0.7‑factor).

5. **Global peak‑mass transfer** – moves a fraction of the total donor mass
   to a single weak receiver; the fraction is chosen analytically to keep
   every convolution entry below the previous maximum.

6. **Random hill‑climbing** – many small random donor‑receiver transfers,
   accepting only improvements. This often escapes shallow local minima.

7. **Sub‑gradient descent on the true max‑norm** – a simple sub‑gradient
   method that uses the exact max‑convolution (no soft‑max) and projects
   back onto the simplex.

8. **Final polishing** – a last run of Adam with an even larger soft‑max
   temperature.

All phases stop early when the global deadline is reached.  The resulting
sequence is trimmed of trailing near‑zero entries and returned as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective used by the competition: 2·n·max(b)/(sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_decay_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _double_exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n) + np.exp(-q * (n - 1 - i) / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, m_frac: float, r: float, target_sum: float) -> np.ndarray:
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)          # (m*h + (n-m)*l) = target_sum with h = r*l
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

# Deterministic mass‑transfer (exact DMT2 – larger steps)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick the first maximal index
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # values that pair with a[idx]

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        # use the exact feasible amount (no artificial 0.7 factor)
        max_delta = min(a[donor], delta_needed * 0.998)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Global peak‑mass transfer (aggressive refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.04,
                               receiver_frac: float = 0.04,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second largest entry
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)

        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]
            diff = donor_mass * shift_r - conv_donors
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] / (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move

        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} α={best_alpha:.4g} new_score={best_score:.6f}")

        it += 1

    return best_a, best_score

def _random_hill_climbing(a0: np.ndarray,
                          deadline: float,
                          target_sum: float,
                          max_moves: int = 8000,
                          move_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Try many small random donor→receiver moves, keep improvements."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    # pre‑compute cumulative distribution for weighted donor sampling
    while time.time() < deadline and max_moves > 0:
        # donor weighted by current mass
        prob = a / a.sum()
        donor = rng.choice(n, p=prob)
        # receiver weighted towards small values (inverse mass)
        inv = 1.0 / (a + 1e-12)
        prob_r = inv / inv.sum()
        receiver = rng.choice(n, p=prob_r)
        if donor == receiver:
            continue

        max_delta = a[donor] * move_frac
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta
        # projection onto the simplex (preserves sum & non‑negativity)
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        val = _score(a_candidate)
        if val < best_score:
            best_score = val
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] Random hill move improved to {best_score:.6f}")

        max_moves -= 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 4000,
                         eta0: float = 0.2) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient method on max (a*a) with projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        # set a tiny tolerance to collect all active constraints
        active = np.where(conv >= max_val - 1e-12)[0]

        if active.size == 0:
            break

        # compute sub‑gradient = 2 * sum_i w^{(k)} over active k
        subgrad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            subgrad[idx] += a[k - idx]

        subgrad *= 2.0

        # diminishing step size
        step = eta0 / np.sqrt(it + 1)
        a = a - step * subgrad
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑gradient it={it} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximum
    self‑convolution value.  The routine respects the given time budget
    and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a smoother landscape)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000          # a bit larger than the previous 12000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) ⇒ score = max convolution

    # --------------------------------------------------------------
    # Phase 0 – parametric warm‑start (fast, cheap)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # 1) pure power‑law
    for p in np.linspace(0.0, 2.0, 31):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 2) power‑law with exponential tilt
    p_grid = np.linspace(0.0, 2.0, 21)
    q_grid = np.linspace(0.0, 20.0, 21)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # 3) pure exponential decay
    for q in np.linspace(0.0, 20.0, 15):
        seq = _exp_decay_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 4) symmetric double‑exponential
    for q in np.linspace(0.0, 15.0, 11):
        seq = _double_exp_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 5) Gaussian (various widths)
    for sigma in np.linspace(0.05 * n, 0.3 * n, 7):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 6) two‑level piecewise constant
    for m_frac in np.linspace(0.02, 0.4, 9):
        for r in np.linspace(1.0, 30.0, 15):
            seq = _piecewise_sequence(n, float(m_frac), float(r), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # 7) use any previously stored construction if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # 8) random jitter around the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] Warm‑start best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional symmetrisation
    # --------------------------------------------------------------
    sym = (best_seq + best_seq[::-1]) * 0.5
    sym = _project_to_simplex(sym, target_sum)
    val_sym = _score(sym)
    if val_sym < best_score:
        best_score = val_sym
        best_seq = sym.copy()
        print(f"[{time.time():.2f}] Symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam gradient descent (larger time share)
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.35
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (core refinement)
    # --------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # occasional random perturbation (helps escape plateaus)
        a_rand, val_rand = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.04,
                                                      receiver_frac=0.04)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best = {best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # --------------------------------------------------------------
    # Phase C – Random hill‑climbing
    # --------------------------------------------------------------
    if time.time() < deadline - 7.0:
        a_hill, val_hill = _random_hill_climbing(best_seq,
                                                 deadline,
                                                 target_sum,
                                                 max_moves=8000,
                                                 move_frac=0.04)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = a_hill.copy()
            print(f"[{time.time():.2f}] Random hill‑climbing improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Sub‑gradient descent on the exact max‑norm
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_sub, val_sub = _subgradient_descent(best_seq,
                                              deadline,
                                              target_sum,
                                              max_iter=4000,
                                              eta0=0.15)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient descent improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing with very large β
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and final verification
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final check (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
