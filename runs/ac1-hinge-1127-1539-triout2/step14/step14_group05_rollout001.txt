"""
Hybrid optimizer for the step‑function benchmark (≈1000 s).

The optimiser proceeds through several deliberately timed stages:

1. **Initial pool** – a mixture of parametric families (power‑law,
   exponential, cosine, sinusoid) plus random simplex vectors and the
   previous best sequence if available.

2. **Annealed soft‑max gradient descent** – cheap but effective
   optimisation on a modest size (n≈5 000).  Multiple short runs are
   performed and the best result is retained.

3. **Linear‑program refinement** – a tiny LP (≤30 active convolution
   constraints) that finds a new direction `g` with larger ℓ¹‑norm while
   respecting the current max‑convolution bound.  The current sequence
   is mixed with `g` repeatedly (≥30 iterations).

4. **Upscaling** – the refined sequence is interpolated to a larger
   support (≈20 000 – 30 000) and reordered with a low‑collision
   permutation that spreads large values.

5. **Long sub‑gradient descent** – exact sub‑gradient of the max‑convolution
   (active‑set tolerance 0.9999) with momentum (γ≈0.999) and an
   adaptive step‑size.  This stage runs for the bulk of the remaining
   budget.

6. **Intensive permutation search** – random shuffles followed by a
   cheap hill‑climb, consuming the remaining time before the final
   refinement.

7. **Simulated‑annealing mass‑transfer** – a greedy pairwise mass move
   that reduces the max‑convolution further.

All updates are projected onto the ℓ¹‑simplex `{x≥0, Σx = √(2 n)}`.
The function returns the best sequence discovered before the deadline.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (full linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (to scatter large values)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest values alternating from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread values with a step size coprime to the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program refinement (active‑set LP)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               max_iters: int,
               deadline: float,
               K: int = 30,
               mix_factor: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Small LP that, given the current a, finds a g with large ℓ¹‑norm s.t.
    a * g ≤ M on the K most critical convolution positions.
    The new a is a convex combination (1‑t)·a + t·g.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        print("SciPy not available, skipping LP refinement.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        # active set: top‑K indices
        if K >= conv.size:
            active = np.arange(conv.size, dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # Build linear constraints A @ g ≤ M
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            A[row, idx] = a[k - idx]

        # Solve LP: maximise sum(g) (i.e. minimise -sum(g))
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        g = g * (target_sum / g.sum())  # scale to exact ℓ¹‑norm

        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.999,
                         tol: float = 0.9999) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 3000 == 0:
            # occasional jitter to escape plateaus
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        if it % 5000 == 0:
            cur_val = _max_convolution_fft(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SubGrad it={it:6d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Permutation search – random shuffles & hill‑climb
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Randomly shuffle repeatedly until time runs out, keeping the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    while time.time() < deadline:
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """Simple hill‑climb by swapping random pairs."""
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")

    return a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.25,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index that currently carries the maximal convolution value
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns the best sequence discovered within the time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin
    rng = np.random.default_rng()

    # ------------------------------
    # 1. Build initial pool (n≈5 000)
    # ------------------------------
    n_expl = 5000
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool: List[np.ndarray] = []

    # parametric families (denser sampling)
    for p in np.linspace(0.1, 2.5, 16):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0003, 0.04, 12):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for _ in range(10):
        init_pool.append(_cosine_sequence(n_expl, target_expl))
    for _ in range(10):
        init_pool.append(_sin_half_sequence(n_expl, target_expl))
    for _ in range(20):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # include previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered variants of every pool element (light jitter)
    jitter_scale = target_expl * 0.02
    for seq in list(init_pool):
        seq_j = seq + rng.normal(scale=jitter_scale, size=n_expl)
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 2. Annealed gradient descent (many short runs, total ~30 s)
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # pick a seed – either a random pool element or a jittered version of the current best
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 12.0, deadline - 18.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_expl,
            max_beta=9e5, step_factor=0.9, gamma=0.98)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. LP refinement on modest size
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        lp_deadline = time.time() + min(22.0, deadline - time.time() - 12.0)
        best_seq, lp_score = _lp_refine(best_seq,
                                        target_expl,
                                        max_iters=45,
                                        deadline=lp_deadline,
                                        K=30,
                                        mix_factor=0.15)
        if lp_score < best_score:
            best_score = lp_score
            print(f"[{time.time():.2f}] After LP refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Upscale to larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600.0:
        n_big = 30000
    elif remaining > 400.0:
        n_big = 25000
    elif remaining > 250.0:
        n_big = 20000
    else:
        n_big = n_expl

    target_big = math.sqrt(2.0 * n_big)

    # interpolate and project
    seq_big = _resize_sequence(best_seq if best_seq is not None else _power_law_sequence(n_expl, 0.5, target_expl),
                               n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # try three deterministic reorderings and pick the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 5. Heavy sub‑gradient descent on the large scale
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        seq_big, sub_score = _subgradient_descent(seq_big,
                                                  sub_deadline,
                                                  target_big,
                                                  step_factor=0.6,
                                                  gamma=0.999,
                                                  tol=0.9999)
        if sub_score < best_score:
            best_score = sub_score
            print(f"[{time.time():.2f}] Sub‑grad descent best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Extensive permutation search (random + hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        perm_deadline = time.time() + min(10.0, deadline - time.time() - 12.0)
        seq_big, _ = _permutation_search(seq_big, perm_deadline)
        print(f"[{time.time():.2f}] After random perm max={_max_convolution_fft(seq_big):.6f}")

    if time.time() < deadline - 15.0:
        hill_deadline = time.time() + min(7.0, deadline - time.time() - 9.0)
        seq_big, _ = _local_permutation_hillclimb(seq_big, hill_deadline)
        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 7. Final greedy mass‑transfer (simulated‑annealing)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = time.time() + min(6.0, deadline - time.time() - 3.0)
        seq_big, mt_score = _mass_transfer_simulated_annealing(seq_big,
                                                               mt_deadline,
                                                               target_big,
                                                               start_temp=0.25,
                                                               end_temp=1e-5,
                                                               low_frac=0.05)
        if mt_score < best_score:
            best_score = mt_score
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # 8. Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq   # expose for possible next run

    # optional sanity evaluation (won’t affect the answer if evaluate_sequence is undefined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
