def _permutation_hillclimb(seq, deadline):
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    best_seq = a.copy()
    n = len(a)
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        attempts += 1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        else:
            a[i], a[j] = a[j], a[i] # revert
        if attempts % 5000 == 0:
            print(f"[{time.time():.2f}] Perm hill attempts={attempts}, best={best_val:.6f}")
    return best_seq, best_val

    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

def _subgradient_descent(a0, deadline, target_sum, step_factor=0.5, gamma=0.99, tol=0.999):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute convolution
        conv = _conv_fft(a)
        maxb = conv.max()
        # mask for near-maximum entries
        I = (conv >= maxb * tol).astype(np.float64)
        # compute gradient: 2 * (I * a_rev) convolve
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # optional jitter to avoid stagnation
        if it % 1000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad iter={it} best={best_val:.6f}")
        it += 1
    # after loop, return best found
    return best_a, best_val
```

def _mass_transfer_simulated_annealing(a0, deadline, target_sum, start_temp=0.2, end_temp=1e-6, low_frac=0.07):
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # find index of max convolution entry
        kmax = int(np.argmax(conv))

        # positions contributing to this entry: i in [max(0, kmax-(n-1)), min(kmax, n-1)]
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # compute product contributions a[i] * a[kmax - i]
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

"""
A fast multi‑phase optimizer for the step‑function benchmark.

The pipeline:
1.  Initialise from the best known global sequence (if available) or a
    simple power‑law seed; resize to a large problem size (≈20‑30k).
2.  Heavy sub‑gradient descent on the exact max‑convolution objective.
    The method uses an active‑set mask (entries ≳ max·tol) to obtain the
    sub‑gradient, momentum and Euclidean projection onto the ℓ¹‑simplex.
3.  A local permutation hill‑climb (random swaps) to break any remaining
    correlation peaks.
4.  A short mass‑transfer simulated‑annealing pass for fine‑grained
    improvement.
All stages respect non‑negativity and the ℓ¹‑norm ∑a = √(2 n); the final
sequence is trimmed of trailing zeros and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used only if no global seed exists)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Reordering helpers (dispersion of large entries)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest …"""
    sorted_vals = np.sort(seq)[::-1]
    n = len(seq)
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Place values at roughly equal gaps (step coprime with n)."""
    n = len(seq)
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Linear interpolation to a new length
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent on max‑convolution (exact mask)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.99,
                         tol: float = 0.999) -> tuple[np.ndarray, float]:
    """Minimise max(a*a) subject to ℓ¹‑simplex using sub‑gradient descent."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # current convolution and mask of near‑max entries
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)

        # gradient = 2 * (I * a_rev) convolved
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        # projection back onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 1000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # evaluate improvement
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:6d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation hill‑climb (local swaps)
# ----------------------------------------------------------------------


def _permutation_hillclimb(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Random pairwise swaps improving the max‑convolution."""
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    best_seq = a.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        attempts += 1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        else:
            a[i], a[j] = a[j], a[i]  # revert
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Perm hill attempts={attempts} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mass‑transfer simulated‑annealing (fine‑grained local moves)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.2,
                                      end_temp: float = 1e-6,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake the system
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function with a low benchmark score.

    The routine follows the multi‑phase strategy described in the module
    docstring.  It returns the best sequence discovered before the time budget
    expires.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1. Determine problem size based on remaining budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n_big = 30000
    elif remaining > 500:
        n_big = 25000
    elif remaining > 300:
        n_big = 20000
    elif remaining > 150:
        n_big = 15000
    else:
        n_big = 10000

    target_big = math.sqrt(2.0 * n_big)

    # --------------------------------------------------------------
    # 2. Initialise a seed sequence (reuse global if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.array(globals()["height_sequence_1"], dtype=float)
    else:
        # fallback: a simple power‑law seed
        base_seq = _power_law_sequence(5000, 0.6, math.sqrt(2.0 * 5000))

    seq_big = _resize_sequence(base_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # try a few deterministic re‑orderings and keep the best start point
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Initial n={n_big}, max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 3. Sub‑gradient descent on the large problem (main workhorse)
    # --------------------------------------------------------------
    if deadline - time.time() > 30.0:
        sub_deadline = deadline - 10.0  # leave some time for later phases
        seq_big, sub_best = _subgradient_descent(seq_big,
                                                 sub_deadline,
                                                 target_big,
                                                 step_factor=0.6,
                                                 gamma=0.99,
                                                 tol=0.999)
        print(f"[{time.time():.2f}] After sub‑grad best={sub_best:.6f}")

    # --------------------------------------------------------------
    # 4. Local permutation hill‑climb
    # --------------------------------------------------------------
    if deadline - time.time() > 15.0:
        perm_deadline = deadline - 5.0
        seq_big, perm_best = _permutation_hillclimb(seq_big, perm_deadline)
        print(f"[{time.time():.2f}] After perm climb best={perm_best:.6f}")

    # --------------------------------------------------------------
    # 5. Final short mass‑transfer simulated annealing
    # --------------------------------------------------------------
    if deadline - time.time() > 5.0:
        mt_deadline = deadline
        seq_big, mt_best = _mass_transfer_simulated_annealing(seq_big,
                                                             mt_deadline,
                                                             target_big,
                                                             start_temp=0.15,
                                                             end_temp=1e-6,
                                                             low_frac=0.07)
        print(f"[{time.time():.2f}] After mass‑transfer best={mt_best:.6f}")

    # --------------------------------------------------------------
    # 6. Trim trailing near‑zero entries and expose for reuse
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq  # keep for possible next run

    # --------------------------------------------------------------
    # 7. Final evaluation (optional, only for logging)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
