# ----------------------------------------------------------------------
# FFT utilities – convolution & max‑convolution
# ----------------------------------------------------------------------


def _fft_size_for_len(n: int) -> int:
    """Smallest power‑of‑two >= 2·n‑1 (used for linear convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using the FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _fft_size_for_len(n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the autocorrelation a*a."""
    return float(_conv_fft(a).max())

# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial sequences (all ℓ¹‑normalised to the target sum)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Permutation helpers – deterministic low‑collision patterns
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, second largest → right, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _cyclic_step_order(seq: np.ndarray, step: int) -> np.ndarray:
    """Place the sorted values at positions (i*step) % n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _best_cyclic_permutation(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Search a handful of co‑prime step values and return the permutation
    with the smallest max‑convolution.
    """
    n = seq.shape[0]
    best = seq.copy()
    best_val = _max_convolution_fft(best)

    # deterministic candidates – golden‑ratio step and a few random co‑primes
    phi = (1 + math.sqrt(5)) / 2
    cand_steps = set()
    cand_steps.add(int(round(n / phi)))          # approx golden‑ratio
    cand_steps.add(int(round(n / (phi * phi)))) # another irrational
    cand_steps.add(n // 2)                       # simple half‑shift (if coprime)
    # add a few random co‑prime steps
    rng = np.random.default_rng()
    while len(cand_steps) < 12 and time.time() < deadline:
        step = rng.integers(1, n)
        if math.gcd(step, n) == 1:
            cand_steps.add(step)

    for step in cand_steps:
        if time.time() >= deadline:
            break
        if math.gcd(step, n) != 1:
            continue
        perm = _cyclic_step_order(seq, step)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best = perm.copy()
    return best


# ----------------------------------------------------------------------
# Annealed soft‑max gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.8,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max proxy of
    max‑convolution.  Uses an Adam‑style adaptive learning‑rate.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        n = a.shape[0]
        size = _fft_size_for_len(n)

        # FFT of a
        a_fft = np.fft.rfft(a, n=size)
        # convolution (full)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        maxc = conv.max()
        # soft‑max weights
        exps = np.exp(beta * (conv - maxc))
        p = exps / np.sum(exps)

        # gradient = 2 * (a_rev * p) (central n entries)
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        grad_full = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # Adam update
        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        # increase β (slowly) to sharpen the soft‑max
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.8, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max (with momentum)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995,
                         tol: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # active set – entries close to the max (within tol)
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        size = _fft_size_for_len(len(I))
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple mass‑transfer hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.3,
                                      end_temp: float = 5e-6,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    # keep a current convolution to avoid recomputing from scratch each move
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # pick the index of the current global maximum in the convolution
        kmax = int(np.argmax(conv))

        # indices that contribute to this peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution = a[i] * a[kmax-i]; choose a donor (largest contribution)
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        # pick a receiver among the weakest contributors
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level orchestration (the entry point)
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function with a low max‑autocorrelation.
    The function respects the 1000‑second time budget and returns the best
    sequence it found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Broad exploration on a modest size (≈8 000)
    # ------------------------------------------------------------------
    n_expl = 8000
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool = []

    # diverse parametric families
    for p in np.linspace(0.1, 2.2, 14):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0004, 0.035, 12):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for _ in range(10):
        init_pool.append(_cosine_sequence(n_expl, target_expl))
    for _ in range(10):
        init_pool.append(_sin_half_sequence(n_expl, target_expl))
    for _ in range(18):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # include the previously found best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered variants of every seed
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.03, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # run many short annealed GD runs until ~60 s remain
    while time.time() < deadline - 60.0:
        # bias towards the current best
        if best_seq is not None and rng.random() < 0.65:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.006, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 16.0, deadline - 70.0)
        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_deadline,
            target_expl,
            max_beta=5e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Sub‑gradient descent on the modest size (polish)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 45.0:
        sub_deadline = time.time() + min(30.0, deadline - time.time() - 40.0)
        best_seq, sub_val = _subgradient_descent(
            best_seq,
            sub_deadline,
            target_expl,
            step_factor=0.55,
            gamma=0.996,
            tol=0.99,
        )
        if sub_val < best_score:
            best_score = sub_val
            print(f"[{time.time():.2f}] Sub‑grad after exploration best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upscale to a larger problem size (≈30 000 – 40 000)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 300.0:
        n_big = 40000
    elif remaining > 200.0:
        n_big = 30000
    elif remaining > 140.0:
        n_big = 25000
    else:
        n_big = n_expl

    target_big = math.sqrt(2.0 * n_big)

    # interpolate the current best (or fall back to a fresh power‑law)
    if best_seq is not None:
        seq_big = _resize_sequence(best_seq, n_big)
    else:
        seq_big = _power_law_sequence(n_expl, 0.6, target_expl)
        seq_big = _resize_sequence(seq_big, n_big)

    seq_big = _project_to_simplex(seq_big, target_big)

    # deterministic re‑ordering (alternating + golden‑ratio step)
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _best_cyclic_permutation(seq_big, deadline - 5.0),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy annealed GD on the large scale (the main fine‑tuner)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        gd_deadline = time.time() + min(90.0, deadline - time.time() - 130.0)
        seq_big, gd_val = _annealed_gradient_descent(
            seq_big,
            gd_deadline,
            target_big,
            max_beta=8e5,
            step_factor=0.7,
            gamma=0.985,
        )
        if gd_val < best_score:
            best_score = gd_val
            print(f"[{time.time():.2f}] Large‑scale GD best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient polish on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        sub_big_deadline = time.time() + min(45.0, deadline - time.time() - 80.0)
        seq_big, sub_big_val = _subgradient_descent(
            seq_big,
            sub_big_deadline,
            target_big,
            step_factor=0.45,
            gamma=0.998,
            tol=0.995,
        )
        if sub_big_val < best_score:
            best_score = sub_big_val
            print(f"[{time.time():.2f}] Large‑scale sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final permutation search (systematic cyclic steps + random)
    # ------------------------------------------------------------------
    if time.time() < deadline - 45.0:
        perm_deadline = time.time() + min(20.0, deadline - time.time() - 55.0)
        seq_big = _best_cyclic_permutation(seq_big, perm_deadline)
        print(f"[{time.time():.2f}] After systematic perm max={_max_convolution_fft(seq_big):.6f}")

        # Additional random shuffles (quick)
        rng = np.random.default_rng()
        best_val = _max_convolution_fft(seq_big)
        while time.time() < perm_deadline:
            perm = rng.permutation(seq_big)
            val = _max_convolution_fft(perm)
            if val < best_val:
                best_val = val
                seq_big = perm.copy()
                print(f"[{time.time():.2f}] Random perm improved to {best_val:.6f}")

        if best_val < best_score:
            best_score = best_val

    # ------------------------------------------------------------------
    # 7. Final tiny mass‑transfer / jitter pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        mt_deadline = time.time() + min(12.0, deadline - time.time() - 25.0)
        seq_big, mt_val = _mass_transfer_simulated_annealing(
            seq_big,
            mt_deadline,
            target_big,
            start_temp=0.22,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if mt_val < best_score:
            best_score = mt_val
            print(f"[{time.time():.2f}] Final mass‑transfer best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq  # store for possible reuse

    # final evaluation (optional – only for logging)
    try:
        val = evaluate_sequence(final_seq)  # assume evaluate_sequence is in globals
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
