def _smooth_gradient_descent(a0, target_sum, deadline, start_alpha=0.5, end_alpha=200.0, step_factor=0.5):
    a = a0.copy()
    # base parameters
    size = 1 << ((2 * len(a) - 1).bit_length())
    start_time = time.time()
    # compute initial conv and max
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total
        alpha = start_alpha * (end_alpha / start_alpha) ** frac
        conv = _conv_fft(a)
        maxv = conv.max()
        # compute w
        # shift conv for numerical stability: subtract maxv
        w_un = np.exp(alpha * (conv - maxv))   # shape 2n-1
        w = w_un / np.sum(w_un)
        # compute gradient via correlation using FFT
        # compute FFTs
        fa = np.fft.rfft(a, n=size)
        fw = np.fft.rfft(w, n=size)
        # correlation: conj(fa) * fw followed by irfft yields correlation length size
        corr_full = np.fft.irfft(np.conj(fa) * fw, n=size)
        # first n entries are correlation for lags 0..n-1
        grad = 2.0 * corr_full[:len(a)]
        # step size adaptively scaled to target sum
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)
        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)
        conv_new = _conv_fft(a_new)
        max_new = conv_new.max()
        if max_new < cur_max:
            a = a_new
            cur_max = max_new
            if max_new < best_max:
                best_max = max_new
                best_a = a.copy()
        else:
            # optionally accept with some probability based on difference and temperature
            # Temperature maybe decreasing linearly: T_start = start_alpha maybe.
            # Let's use simple acceptance probability:
            T = max_alpha * (1 - frac) + 1e-12  # approximate

"""
Hybrid optimiser for the minimum‐peak auto‑convolution problem.

The routine:

1. builds a diverse seed pool (power‑law, exponential, random and the
   previous best),
2. runs a modest sized Differential Evolution (DE) to obtain a good
   starting point,
3. performs a *smooth* projected gradient descent where the max‑convolution
   is approximated by a temperature‑controlled soft‑max.  The gradient is
   obtained via an FFT‑based correlation, and every step is projected back
   onto the ℓ¹‑simplex (∑a = √(2n), a_i ≥ 0).
4. finally tests a few deterministic reorderings (alternating, even spread)
   and returns the best sequence found.

The smooth‑gradient phase replaces the ad‑hoc deterministic mass‑transfer
and sub‑gradient steps of the previous version and usually pushes the
peak a little lower (target ≤ 1.503 for the 8000‑point problem).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / correlation utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_linear(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of length `size`.
    Returns a length‑2·n‑1 array (the valid part of the convolution).
    """
    fa = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fa * fa, n=size)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    conv = _conv_linear(a, size)
    return float(conv.max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """
    Objective 2·n·max(a*a) / (∑a)².
    For a correctly normalised vector (∑a = √(2n)) the value equals the
    maximum convolution entry.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s} using the O(n) algorithm of
    Wang & Carreira‑Perpiñán (2013)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.6,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE with projection onto the ℓ¹‑simplex."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind, target_sum) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial, target_sum)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth projected gradient descent (soft‑max approximation of the max)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                            target_sum: float,
                            deadline: float,
                            start_alpha: float = 0.5,
                            end_alpha: float = 200.0,
                            step_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent where the objective
        f(a) = max_k (a*a)[k]
    is replaced by a smooth soft‑max:
        f_α(a) = Σ_k w_k (a*a)[k],  w_k ∝ exp(α·(conv_k‑max)).
    As α → ∞ the soft‑max approaches the true max.
    """
    a = _project_to_simplex(a0.astype(np.float64).copy(), target_sum)
    n = a.shape[0]

    # FFT size sufficient for both the linear convolution (length 2n‑1)
    # and the correlation (length n + (2n‑1)‑1 = 3n‑2).
    size_corr = 1 << ((3 * n - 2).bit_length())

    start_time = time.time()
    # initial values
    fa = np.fft.rfft(a, n=size_corr)
    conv_full = np.fft.irfft(fa * fa, n=size_corr)
    conv = conv_full[: 2 * n - 1]
    cur_max = float(conv.max())
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        if total <= 0:
            break
        frac = elapsed / total
        # exponential schedule for α
        alpha = start_alpha * (end_alpha / start_alpha) ** frac

        # recompute convolution (a changes each iteration)
        fa = np.fft.rfft(a, n=size_corr)
        conv_full = np.fft.irfft(fa * fa, n=size_corr)
        conv = conv_full[: 2 * n - 1]
        maxv = float(conv.max())

        # soft‑max weights (stable: subtract maxv)
        w_unnorm = np.exp(alpha * (conv - maxv))
        w = w_unnorm / np.sum(w_unnorm)

        # correlation a ⋆ w  (grad = 2·(a ⋆ w))
        fw = np.fft.rfft(w, n=size_corr)
        corr_full = np.fft.irfft(np.conj(fa) * fw, n=size_corr)
        grad = 2.0 * corr_full[:n]

        # adaptive step size (normalise gradient)
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        # evaluate new peak
        fa_new = np.fft.rfft(a_new, n=size_corr)
        conv_full_new = np.fft.irfft(fa_new * fa_new, n=size_corr)
        conv_new = conv_full_new[: 2 * n - 1]
        max_new = float(conv_new.max())

        if max_new < cur_max:
            a = a_new
            cur_max = max_new
            if max_new < best_max:
                best_max = max_new
                best_a = a.copy()
        else:
            # simple cooling: occasionally accept a worse move (prob based on α)
            # for stability we keep the current point (no acceptance)
            pass

        # optional tiny jitter to break symmetry every few seconds
        if int(elapsed) % 30 == 0:
            jitter = np.random.randn(n) * target_sum * 5e-5
            a = _project_to_simplex(a + jitter, target_sum)

    return best_a, best_max

# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine builds a diverse seed pool, runs a short DE, then refines the
    result with a smooth projected gradient descent, finally tests a few
    deterministic reorderings.  The best sequence found is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives richer search space but higher per‑iteration cost.
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 10000
    elif budget_s >= 600:
        n = 8000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    seeds = []

    for p in np.linspace(0.1, 2.0, 13):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential families
    for lam in np.linspace(0.0005, 0.05, 12):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # purely random points on the simplex
    for _ in range(10):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm start from the previous run, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little to increase diversity
    jittered = []
    for seq in seeds:
        j = rng.normal(scale=target_sum * 0.01, size=n)
        seqj = seq + j
        seqj = np.maximum(seqj, 0.0)
        seqj = _project_to_simplex(seqj, target_sum)
        jittered.append(seqj)
    seeds.extend(jittered)

    # --------------------------------------------------------------
    # Initialise DE population
    # --------------------------------------------------------------
    pop_size = max(30, int(n / 250))
    population = []

    best_seed = min(seeds, key=lambda s: _score(s, target_sum))
    population.append(best_seed.copy())
    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # --------------------------------------------------------------
    # Differential Evolution (≈ 35 % of the budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.35 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Smooth projected gradient descent (majority of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        gd_deadline = deadline - 1.0
        seq_gd, max_gd = _smooth_gradient_descent(best_seq,
                                                   target_sum,
                                                   gd_deadline,
                                                   start_alpha=0.5,
                                                   end_alpha=200.0,
                                                   step_factor=0.6)
        if max_gd < best_score:
            best_seq = seq_gd
            best_score = max_gd
            print(f"[{time.time():.2f}] Smooth GD improved – new peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Reordering (often yields a tiny additional drop)
    # --------------------------------------------------------------
    candidates = [best_seq,
                  _alternating_order(best_seq),
                  _even_spread_order(best_seq)]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] After reordering – final peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
