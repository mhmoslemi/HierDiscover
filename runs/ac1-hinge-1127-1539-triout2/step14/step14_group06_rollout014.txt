def _subgradient_descent(a0, target_sum, deadline, eps=1e-12):
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_score = _score(a)
    it = 0
    step0 = target_sum * 0.1 # initial step magnitude relative
    # Or step scaling as fraction of sum.

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        second_max = np.max(conv[conv < max_val]) if np.any(conv < max_val) else 0.0
        gap = max_val - second_max
        # early exit if gap small
        if gap < 1e-12:
            break

        # compute active mask of indices achieving max
        active = np.zeros_like(conv)
        # choose tolerance? we want strict maxima.
        mask = conv >= max_val - 1e-12
        active[mask] = 1.0

        # compute gradient using FFT convolution: gradient = 2 * (a * rev(active))
        # rev(active) = active[::-1]
        # convolution a (len n) with rev_active (len 2n-1) => length 3n-2
        rev_active = active[::-1]
        # Use FFT convolution with proper size
        size = 1 << ((n + (2*n - 1) - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        fr = np.fft.rfft(rev_active, n=size)
        conv_grad_full = np.fft.irfft(fa * fr, n=size)
        # length size; but we need actual full convolution length: len(conv_grad_full) = size, but relevant part is up to n+2n-2 -1 = 3n-2.
        # We can slice conv_grad_full[ (2*n - 2) : (2*n - 2 + n) ]
        start = (2*n - 2)
        grad = 2.0 * conv_grad_full[start:start+n]

        # might get slight negative rounding errors -> clip
        grad = np.maximum(grad, 0.0)

        # step size: use diminishing or based on gap and gradient norm.
        norm_g = np.linalg.norm(grad) + 1e-16
        # choose step = gap / norm_g**2 * some factor
        step = gap / (norm_g**2) * 0.5  # factor 0.5
        # Ensure step not too large
        max_step = target_sum * 0.5
        if step > max_step:
            step = max_step

        # Then try line search decreasing step if needed
        improved = False
        for _ in range(10):
            a_candidate = a - step * grad
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            new_score = _score(a_candidate)
            if new_score < best_score - 1e-12:
                best = a_candidate.copy()
                best_score = new_score
                a = a_candidate
                improved = True
                # maybe break out of line search
                break
            else:
                step *= 0.5
        if not improved:
            # if no improvement, maybe jitter random small?
            # jitter to escape local minima
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            # continue
        it += 1
        # Log
        if it % 10 == 0:
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_score:.6f} max={max_val:.6f}")

    return best, best_score

"""
Subgradient descent plus deterministic mass‐transfer for minimizing
the peak of the self‑convolution of a non‑negative step function.
"""

import time
import numpy as np

# ---------- convolution utilities ----------
def _conv_fft(a):
    """Full linear convolution a*a."""
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

def _conv_fft_general(a, b):
    """Full linear convolution of a and b using FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa*fb, n=size)[:n+m-1]
    return conv

def _max_convolution_fft(a):
    return float(_conv_fft(a).max())

def _score(a):
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# ---------- simplex projection ----------
def _project_to_simplex(v, s):
    """Euclidean projection of v onto {x >=0, sum x = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ---------- parametric families ----------
def _power_law_sequence(n, p, target_sum):
    i = np.arange(1, n+1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a

def _exp_power_law_sequence(n, p, q, target_sum):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a

def _exp_decay_sequence(n, q, target_sum):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a

def _double_exp_sequence(n, q, target_sum):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n) + np.exp(-q * (n-1-i) / n)
    a *= target_sum / a.sum()
    return a

def _gaussian_sequence(n, sigma, target_sum):
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a

def _piecewise_sequence(n, m_frac, r, target_sum):
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a

# ---------- subgradient descent ----------
def _subgradient_descent(a0, target_sum, deadline, eps=1e-12):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # update best if improved
        cur_score = max_val  # because sum fixed
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} new best={best_score:.6f}")

        # second largest value
        mask_less = conv < max_val - eps
        if np.any(mask_less):
            second_max = np.max(conv[mask_less])
        else:
            second_max = max_val
        gap = max_val - second_max

        # active mask (indices tied for max)
        active = (conv >= max_val - eps).astype(np.float64)

        # gradient via convolution with reversed active
        rev_active = active[::-1]
        grad_full = _conv_fft_general(a, rev_active)
        # slice the portion that corresponds to gradient entries
        start = 2 * n - 2
        grad = 2.0 * grad_full[start:start + n]

        # if gradient is all zeros (unlikely) break
        if np.all(grad == 0):
            break

        # step size: classic subgradient rule
        norm2 = np.dot(grad, grad) + 1e-16
        # use gap if positive else small fallback
        if gap > 0:
            step = gap / norm2
        else:
            step = target_sum * 0.01 / np.sqrt(norm2)
        # limit step magnitude
        max_step = target_sum * 0.5
        if step > max_step:
            step = max_step

        # try move, with backtracking if it does not improve
        improved = False
        for _ in range(5):
            a_candidate = a - step * grad
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            new_max = _conv_fft(a_candidate).max()
            if new_max < max_val - 1e-12:
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # random jitter to escape possible stall
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
        # occasional logging
        if it % 100 == 0:
            print(f"[{time.time():.2f}] Subgrad it={it} max={max_val:.6f}")

    return best_a, best_score

# ---------- deterministic mass transfer (existing) ----------
def _deterministic_mass_transfer_exact2(a0, deadline, target_sum, tol=1e-12):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

    else:

"""
Sub‑gradient descent + deterministic mass‑transfer for minimizing the peak
value of the self‑convolution of a non‑negative step function.
The routine respects a time budget (default 1000 s) and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b using FFT."""
    n, m = a.size, b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective used by the competition (2·n·max(b)/sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) * exp(-q*i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_decay_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    """Pure exponential decay a_i = exp(-q*i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _double_exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    """Symmetric double‑exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n) + np.exp(-q * (n - 1 - i) / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Gaussian centred in the middle of the interval."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, m_frac: float, r: float, target_sum: float) -> np.ndarray:
    """
    Two‑level step: first m = round(m_frac*n) entries have height h,
    the remaining n‑m entries have height l, with ratio h/l = r ≥ 1.
    """
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the L‑∞ norm of the self‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method that directly minimises max(a*a).
    The gradient of the current maximum entry is obtained via a single
    FFT convolution with the reversed indicator of the active indices.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()

        # keep the best seen so far
        cur_score = max_val  # sum is fixed → score = max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} new best={best_score:.6f}")

        # second largest entry (used for a step‑size heuristic)
        mask_less = conv < max_val - eps
        if np.any(mask_less):
            second_max = np.max(conv[mask_less])
        else:
            second_max = max_val
        gap = max_val - second_max

        # active set (indices attaining the current maximum)
        active = (conv >= max_val - eps).astype(np.float64)

        # gradient = 2 * (a * rev(active)) restricted to the original length
        rev_active = active[::-1]
        grad_full = _conv_fft_general(a, rev_active)
        start = 2 * n - 2
        grad = 2.0 * grad_full[start:start + n]

        if np.all(grad == 0):
            break

        # classic sub‑gradient step size: gap / ||grad||² (if gap>0)
        norm2 = np.dot(grad, grad) + 1e-16
        if gap > 0:
            step = gap / norm2
        else:
            step = target_sum * 0.01 / np.sqrt(norm2)

        # keep the step bounded
        max_step = target_sum * 0.5
        if step > max_step:
            step = max_step

        # back‑tracking line search – ensure the maximal convolution value drops
        improved = False
        for _ in range(5):
            a_candidate = a - step * grad
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            new_max = _conv_fft(a_candidate).max()
            if new_max < max_val - 1e-12:
                a = a_candidate
                improved = True
                break
            step *= 0.5

        if not improved:
            # occasional jitter to escape possible stagnation
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
        if it % 200 == 0:
            print(f"[{time.time():.2f}] Subgrad it={it} max={max_val:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact DMT2) – fine‑grained local refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (used as a target)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose a receiver with minimal partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the peak of its
    self‑convolution.  The routine runs a warm‑start over several
    parametric families, then applies a dedicated sub‑gradient optimiser,
    and finally cleans up with a deterministic mass‑transfer pass.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives a smoother landscape)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) ⇒ score = max convolution

    # --------------------------------------------------------------
    # Phase 0 – dense parametric warm‑start
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # 1) pure power‑law
    for p in np.linspace(0.0, 2.0, 41):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # 2) power‑law with exponential tilt
    p_grid = np.linspace(0.0, 2.0, 31)
    q_grid = np.linspace(0.0, 20.0, 31)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # 3) pure exponential decay
    for q in np.linspace(0.0, 20.0, 21):
        seq = _exp_decay_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # 4) symmetric double‑exponential
    for q in np.linspace(0.0, 15.0, 16):
        seq = _double_exp_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # 5) Gaussian
    for sigma in np.linspace(0.1 * n, 0.5 * n, 9):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # 6) two‑level piecewise constant
    for m_frac in np.linspace(0.01, 0.5, 10):
        for r in np.linspace(1.0, 30.0, 20):
            seq = _piecewise_sequence(n, float(m_frac), float(r), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # 7) reuse a previously stored construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # 8) small random jitter round (helps break ties)
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] Warm‑start best = {best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
