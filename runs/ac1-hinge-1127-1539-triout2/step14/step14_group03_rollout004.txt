def smooth_max_conv(a, alpha):
    conv = _conv_fft(a)
    max_val = np.max(conv)
    # smooth max via logsumexp

   known to give decent scores.

2. **Smooth‑max gradient descent** – minimise a smoothed version of the
   objective 2·n·max(a∗a)/‖a‖₁².  The smooth max is
   `log‑sum‑exp(α·conv)`; its gradient can be written as a simple
   correlation which we compute with an FFT (O(n log n)).
   The parameter `α` is increased from a small value to a large one,
   turning the smooth objective into the true max‑convolution while the
   optimiser follows a well‑behaved path.  After each step the vector is
   projected onto the simplex {x≥0, Σx = √(2·n)}.

3. **Refinement** – a short projected sub‑gradient descent on the *exact*
   max‑convolution (as in the previous state‑of‑the‑art) followed by the
   aggressive “mass‑transfer” simulated‑annealing used before.

The best sequence seen throughout all phases is returned as a plain list
of floats.  The routine respects the 1000‑second budget; 0.5 s are kept
as a safety margin.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a)/(sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for the warm start
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i/(n‑1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam·i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape:
    a_i ∝ x^{α‑1}·(1‑x)^{α‑1},   x = (i+0.5)/n ∈ (0,1).
    """
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (Phase 1)
# ----------------------------------------------------------------------


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max, i.e.

        g = 2 * Σ_k w_k * a_{k‑j}

    for all j.  The correlation is computed via FFT (O(n log n)).
    """
    n = a.shape[0]
    # reverse the weight vector (length 2n‑1)
    rev = w[::-1]
    size = 1 << ((3 * n - 2).bit_length())
    fw = np.fft.rfft(rev, n=size)
    fa = np.fft.rfft(a, n=size)
    conv_rev = np.fft.irfft(fw * fa, n=size)[: 3 * n - 2]
    # gradient = 2 * conv_rev[n‑1 : 2n‑1] reversed
    grad = 2.0 * conv_rev[n - 1 : 2 * n - 1][::-1]
    return grad


def smooth_max_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Minimise a smoothed version of the max‑convolution.
    Returns the best vector and its **exact** score (using the true max).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    # momentum parameters (simple Nesterov‑like)
    v = np.zeros_like(a)
    beta = 0.9

    start = time.time()
    it = 0

    while time.time() < deadline:
        # ----- current convolution -----
        conv = _conv_fft(a)                     # length 2n‑1

        # ----- smooth max (log‑sum‑exp) -----
        frac = (time.time() - start) / (deadline - start)
        # α grows from 0.5 to ~50 as the run proceeds
        alpha = 0.5 + 49.5 * frac

        m = np.max(conv)                        # for numeric stability
        exp_term = np.exp(alpha * (conv - m))
        w = exp_term / exp_term.sum()           # Σ w = 1

        # ----- gradient of the smooth max -----
        grad = _gradient_smooth_max(a, w)

        # ----- step size (decaying) -----
        eta = (0.8 * (1.0 - frac) + 0.005)      # between ~0.8 and ~0.005

        # momentum update
        v = beta * v + eta * grad
        a = a - v

        # projection onto the simplex (keeps Σa = target_sum)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 5e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # periodic exact evaluation
        if it % 1500 == 0:
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] smooth‑grad it={it:5d} exact score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution (Phase 2)
# ----------------------------------------------------------------------


def subgradient_projected_descent(a0: np.ndarray, target_sum: float,
                                 deadline: float):
    """Classic sub‑gradient descent that follows only the active peak."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer (Phase 3)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Mass‑transfer annealing – same code as in the previous run."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step sequence.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a list of floats.
    """
    start = time.time()
    # safety margin – we need a few hundred ms to exit gracefully
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # standard normalisation

    # --------------------------------------------------------------
    # 0️⃣ Warm start from previous run or simple families
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        val = _score(warm)
        best_seq = warm.copy()
        best_score = val
        print(f"[{time.time():.2f}] warm start score = {best_score:.6f}")

    # a few deterministic families (cheap, give solid baselines)
    # power‑law
    for p in np.linspace(0.45, 0.75, 7):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] power‑law p={p:.3f} score={val:.6f}")

    # exponential
    for lam in np.linspace(0.5, 8.0, 7):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] exponential λ={lam:.3f} score={val:.6f}")

    # combined power‑law + exponential
    rng = np.random.default_rng()
    for _ in range(8):
        p = rng.random() * 0.8 + 0.1      # 0.1 … 0.9
        lam = rng.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] power‑exp p={p:.3f} λ={lam:.3f} score={val:.6f}")

    # symmetric beta (U‑shaped when α<1)
    for alpha in np.linspace(0.25, 0.9, 7):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] beta α={alpha:.3f} score={val:.6f}")

    # a few random Dirichlet starts
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _score(rnd)
        if val < best_score:
            best_score = val
            best_seq = rnd.copy()
            print(f"[{time.time():.2f}] random start score={val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (≈20 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase1_deadline = time.time() + 0.20 * remaining
        a_smooth, val_smooth = smooth_max_descent(best_seq, target_sum, phase1_deadline)
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] after smooth‑grad best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max (≈35 %)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + 0.35 * remaining
        a_sub, val_sub = subgradient_projected_descent(best_seq, target_sum, phase2_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] after sub‑grad best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – aggressive annealing mass‑transfer (rest of the time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] after anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up, evaluation and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # trim trailing zeros (not strictly needed but tidy)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final check using the provided evaluate_sequence (if present)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] evaluation error: {exc}")

    # store for a warm‑start in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
