def _conv_fft_full(a, b):
    size = 1 << ((a.shape[0] + b.shape[0] - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fb = np.fft.rfft(b, n=size)
    res = np.fft.irfft(Fa * Fb, n=size)
    return res[: a.shape[0] + b.shape[0] - 1]

def _grad_logmax(a, alpha):
    n = len(a)
    conv = _conv_fft(a)  # length 2n-1
    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))  # shift
    w = exp_vals / exp_vals.sum()
    # compute correlation a ⋆ w for shifts 0..n-1
    # compute convolution a with w_rev
    w_rev = w[::-1]
    size = 1 << ((n + len(w_rev) - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    full = np.fft.irfft(fa * fw, n=size)[: n + len(w_rev) - 1]
    offset = len(w_rev) - 1
    # full length = n + (2n-1) - 1 = 3n-2; offset = (2n-1)-1 = 2n-2
    # So g = 2 * full[offset - np.arange(n)]
    g = 2.0 * full[offset - np.arange(n)]
    return g

"""
Hybrid optimizer for the minimum peak convolution problem.

The algorithm combines three phases:

1. **Differential Evolution (DE)** – a quick global search to obtain a
   decent starting point.  A small population is used because the
   sequences are long (n≈10 000) and each evaluation is cheap (FFT).

2. **Smoothed log‑sum‑exp descent** – the exact peak (max) is approximated
   by the smooth function  

        F_α(a) = (1/α)·log Σ_k exp(α·(a∗a)[k]) .

   Its gradient can be written analytically and is evaluated by a pair of
   FFTs.  By increasing α we gradually turn the smooth surrogate into
   the true max while performing projected gradient steps on the
   ℓ¹‑simplex (∑a = √(2n), a_i ≥ 0).  This phase yields a monotone
   reduction of the peak.

3. **Deterministic mass‑transfer polishing** – a fast greedy refinement
   that moves a small amount of mass from the largest contributing
   product pair at the current peak to a weak pair.  The procedure is
   identical to the one used in the reference implementation.

Finally a few simple reorderings are tried and trailing near‑zero
elements are trimmed.  The best sequence found is stored in the global
``height_sequence_1`` for warm‑starting subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear auto‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear auto‑convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    sz = 1 << ((len(a) + len(b) - 1).bit_length())
    fa = np.fft.rfft(a, n=sz)
    fb = np.fft.rfft(b, n=sz)
    conv = np.fft.irfft(fa * fb, n=sz)
    return conv[: len(a) + len(b) - 1]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto simplex)
# ----------------------------------------------------------------------


def _de_optimize(population, target_sum, deadline, F=0.5, CR=0.9):
    """Classic DE loop with projection onto the ℓ¹‑simplex."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_max_convolution_fft(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _max_convolution_fft(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp gradient descent
# ----------------------------------------------------------------------


def _gradient_logmax(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of   (1/α)·log Σ_k exp(α·(a∗a)[k])   w.r.t. a.
    Returns a real vector of the same length as a.
    """
    n = a.shape[0]
    conv = _conv_fft(a)                     # length 2·n‑1
    max_c = conv.max()
    # soft‑max weights (shifted for numerical stability)
    w = np.exp(alpha * (conv - max_c))
    w /= w.sum()

    # correlation a ⋆ w  (indices 0 … n‑1)
    w_rev = w[::-1]
    full = _convolve_fft(a, w_rev)         # length 3·n‑2
    offset = len(w) - 1                     # = 2·n‑2
    grad = 2.0 * full[offset - np.arange(n)]
    return grad


def _smooth_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    alphas,
                    step_factor: float = 0.5,
                    min_lr_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max‑convolution.
    ``alphas`` is a list of smoothing parameters; each gets an equal
    share of the remaining time.
    """
    a = a0.copy()
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_max
    start = time.time()
    total = deadline - start
    n_al = len(alphas)

    for idx_alpha, alpha in enumerate(alphas):
        sub_deadline = start + total * (idx_alpha + 1) / n_al
        while time.time() < sub_deadline:
            grad = _gradient_logmax(a, alpha)
            norm = np.linalg.norm(grad)
            if norm == 0.0:
                break
            # initial learning rate (scaled to the simplex size)
            lr = step_factor * target_sum / (norm + 1e-12)
            # simple backtracking line‑search to guarantee non‑increase
            while lr > target_sum * min_lr_factor:
                a_new = _project_to_simplex(a - lr * grad, target_sum)
                new_max = _max_convolution_fft(a_new)
                if new_max <= cur_max:
                    a = a_new
                    cur_max = new_max
                    if new_max < best_val:
                        best_val = new_max
                        best_a = a.copy()
                    break
                lr *= 0.5
            # if lr became too small we abort this iteration
            if lr <= target_sum * min_lr_factor:
                break
    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-6,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = float(conv.max())

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
            else:
                a[donor] += delta
                a[receiver] -= delta
                a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (projected) on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = float(_conv_fft(a).max())
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = float(_conv_fft(a).max())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation
    score.  The routine builds a seed pool, runs a short DE phase,
    follows with a smoothed log‑sum‑exp gradient descent, then applies
    deterministic mass‑transfer and sub‑gradient polishing.  Finally a
    couple of simple reorderings are tested.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives more flexibility.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.0, 7):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 5):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random points on the simplex
    for _ in range(10):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm‑start from the previous run (if any) – resize if needed
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little for diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # --------------------------------------------------------------
    # Initialise DE population (modest size)
    # --------------------------------------------------------------
    pop_size = max(25, int(n / 300))   # roughly 25‑30 individuals
    population = []

    best_seed = min(seeds, key=lambda s: _max_convolution_fft(s))
    population.append(best_seed.copy())

    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # --------------------------------------------------------------
    # Differential Evolution phase (≈ 25 % of budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.25 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Smoothed log‑sum‑exp gradient descent (≈ 45 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 15.0:
        gd_deadline = start_time + 0.45 * budget_s
        if gd_deadline > deadline - 5.0:
            gd_deadline = deadline - 5.0
        alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
        seq_gd, score_gd = _smooth_descent(best_seq, target_sum,
                                           gd_deadline,
                                           alphas,
                                           step_factor=0.6)
        if score_gd < best_score:
            best_seq, best_score = seq_gd, score_gd
            print(f"[{time.time():.2f}] Smoothed GD improved – new peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer polishing (≈ 15 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        mt_dead = time.time() + min(30.0,
                                    deadline - time.time() - 5.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.4, end_temp=1e-6, low_frac=0.07)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] MT polished – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent refinement (≈ 10 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sg_dead = time.time() + min(20.0,
                                    deadline - time.time() - 3.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_dead, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_seq, best_score = seq_sg, score_sg
            print(f"[{time.time():.2f}] Sub‑grad refined – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Reordering – try a few simple permutations
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq)
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] After reordering – final peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator (optional)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
