def smooth_max_gradient(a, beta):
    n = a.shape[0]
    # compute convolution via FFT
    L = 1 << ((2*n-1).bit_length())
    # But conv length < L, we'll compute conv via FFT optimum size.
    # Use separate larger size for correlation.
    conv = np.fft.ifft(np.fft.rfft(a, L) * np.fft.rfft(a, L)).real[:2*n-1] -- same as earlier _conv_fft
    # compute softmax weights
    # avoid overflow: subtract max before exponentiate
    max_c = conv.max()
    exp_vec = np.exp(beta * (conv - max_c))
    w = exp_vec / exp_vec.sum()
    # compute correlation: using FFT of length L2 >= n + len(w) -1 = n + (2n-1) -1 = 3n-2
    L2 = 1 << ((3*n-2).bit_length())
    # compute FFT of a and w zero-padded to L2 (convert to real)
    A = np.fft.rfft(a, n=L2)
    W = np.fft.rfft(w, n=L2)
    corr = np.fft.irfft(A * np.conj(W), n=L2)[:3*n-2]  # linear correlation
    grad = 2.0 * corr[:n]  # first n elements correspond to shift i >=0?
    # Actually correlation yields indices from 0..L2-1; The part we need is index shift from 0..n-1.
    # Our correlation form sum_t a[t] * w[t+i] corresponds to lag i, so we need corr[i] for i in range(n). That matches first n entries.
    return grad

def smooth_gradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    # pre compute L2
    L2 = 1 << ((3*n - 2).bit_length())
    # schedule: we allocate total_time = deadline - time.time()
    # We'll use phases: [(0.2, 0.5), (0.4, 2.0), (0.4, 8.0)] (proportion of remaining_time)
    # Or we just adjust beta based on iteration count.

    start = time.time()
    it = 0
    # set initial step
    step0 = target_sum / (10 * n)   # small
    # for dynamic beta: set beta = 0.5 * (2 ** (it // 1000)) but limited max 20
    while time.time() < deadline:
        # compute FFT of a
        A = np.fft.rfft(a, n=L2)
        # convolution via A*A
        conv_full = np.fft.irfft(A * A, n=L2)[:2*n-1]
        # compute beta schedule
        # based on iteration number: after 5000 it -> higher beta.
        if it < 2000:
            beta = 0.5
        elif it < 5000:
            beta = 2.0
        else:
            beta = 8.0
        # compute softmax weights
        max_c = conv_full.max()
        # compute w = exp(beta*(conv - max_c)) for stability
        exp_vec = np.exp(beta * (conv_full - max_c))
        w = exp_vec / exp_vec.sum()

        # gradient via correlation: need FFT of w as well
        W = np.fft.rfft(w, n=L2)
        corr = np.fft.irfft(A * np.conj(W), n=L2)[:n]
        grad = 2.0 * corr

        # step size schedule
        eta = step0 / np.sqrt(it + 1) * (1.0 if it < 5000 else 0.5)  # maybe smaller later
        a_new = a - eta * grad
        # project to simplex (nonnegative sum)
        a_new = _project_to_simplex(a_new, target_sum)

        # evaluate actual score (with real max)
        cur_val = _score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] GD it={it} beta={beta} score={cur_val:.6f}")

        a = a_new
        it += 1
    return best_a, best_val

"""
Propose a low‑score non‑negative step sequence.

The algorithm consists of three phases:

1. **Guided warm‑start** – a few deterministic families (power‑law,
   exponential, mixed, beta) are evaluated and the best is kept.
2. **Smooth‑max projected gradient descent** – we minimise a
   temperature‑smoothed version of the max‑convolution using an
   Euclidean projection onto the simplex Σa = √(2n).  The temperature
   (soft‑max β) is increased during the run, turning the smoothed loss
   into the true max‑convolution.  All heavy operations (FFT‑based
   convolutions and correlations) are implemented with NumPy’s fast
   FFTs.
3. **Fine‑grained simulated‑annealing mass‑transfer** – a fast O(n)
   rank‑one update moves a tiny amount of mass from indices that
   contribute most to the current peak towards low‑height indices,
   further reducing the peak.

The whole routine respects the allotted time budget (≈1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray, target_sum: float) -> float:
    """
    Objective 2·n·max(a*a) / (target_sum²).
    When Σa == target_sum the denominator cancels and the score equals
    the peak of the self‑convolution.
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    # target_sum is √(2·n) in our code, so the formula reduces to max_b.
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families (used for deterministic warm‑starts)
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (alpha - 1.0)
    )
    return a


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def _smooth_gradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed max‑convolution using projected gradient descent.
    The temperature β is increased during the run (continuation method).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a, target_sum)

    # pre‑compute FFT length for the correlation (must be ≥ 3·n‑2)
    L2 = 1 << ((3 * n - 2).bit_length())

    start = time.time()
    it = 0
    # base step – roughly 5 % of the average entry
    step_base = target_sum / (20.0 * n)

    while time.time() < deadline:
        # ----- FFT of current a -------------------------------------------------
        A = np.fft.rfft(a, n=L2)

        # ----- self‑convolution (full) -----------------------------------------
        conv_full = np.fft.irfft(A * A, n=L2)[: 2 * n - 1]

        # ----- continuation schedule (β grows linearly with time) ---------------
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total
        beta = 0.5 + 15.0 * frac  # final β ≈ 15.5

        # ----- smooth‑max weights ------------------------------------------------
        m = conv_full.max()
        w = np.exp(beta * (conv_full - m))
        w /= w.sum()

        # ----- gradient = 2 * (a ⋆ w) -----------------------------------------
        W = np.fft.rfft(w, n=L2)
        corr = np.fft.irfft(A * np.conj(W), n=L2)[:n]
        grad = 2.0 * corr.real

        # ----- step size (decaying) --------------------------------------------
        eta = step_base * (1.0 - frac) / math.sqrt(it + 1)
        a_new = a - eta * grad
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- evaluate true objective (peak of convolution) --------------------
        cur_val = _score(a_new, target_sum)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(
                f"[{time.time():.2f}] GD it={it:5d} β={beta:5.2f} score={cur_val:.6f}"
            )

        # occasional jitter to escape shallow plateaus
        if it % 2000 == 0 and it > 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# (re‑used from the previous solution, slightly tweaked)
# ----------------------------------------------------------------------


def _anneal_mass_transfer(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)  # exact convolution a*a
    cur_val = _score(a, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – proportional to contribution to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- O(n) exact update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Pick a problem size – larger n gives a finer discretisation.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 25000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = math.sqrt(2.0 * n)  # Σa = √(2·n)

    # --------------------------------------------------------------
    # 1. Guided deterministic families (warm‑starts)
    # --------------------------------------------------------------
    candidates = []

    # Warm start from previous run, if present
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        candidates.append(warm)

    # Power‑law sweep
    for p in np.linspace(0.45, 0.75, 9):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    # Exponential sweep
    for lam in np.linspace(0.5, 9.0, 9):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    # Mixed power‑law × exponential
    rng = np.random.default_rng()
    for _ in range(12):
        p = rng.random() * 0.8 + 0.1  # 0.1 … 0.9
        lam = rng.random() * 9.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    # Symmetric Beta family
    for alpha in np.linspace(0.25, 0.9, 9):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    # A few random Dirichlet‑like starts
    for _ in range(5):
        r = rng.random(n)
        r = _project_to_simplex(r, target_sum)
        candidates.append(r)

    # -----------------------------------------------------------------
    # Evaluate all candidates with the cheap _score routine and keep the best
    # -----------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for a in candidates:
        val = _score(a, target_sum)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init candidate score={val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth‑max projected gradient descent (main optimisation phase)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.55 * (deadline - time.time())
        a_opt, val_opt = _smooth_gradient_descent(best_seq, target_sum, gd_deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] After GD best score={best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Aggressive annealing refinement (optional, uses remaining time)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = _anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] After anneal best score={best_val:.6f}")

    # -----------------------------------------------------------------
    # Final clean‑up and evaluation
    # -----------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final check using the provided evaluate_sequence (if available)
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
