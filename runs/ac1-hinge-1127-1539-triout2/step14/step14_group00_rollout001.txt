def _adam_gradient_descent(a0, deadline, target_sum, start_alpha=2.0, end_alpha=20.0, lr_init=1.0, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step = 0
    while time.time() < deadline:
        # compute alpha based on progress: linearly increasing
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = start_alpha + progress * (end_alpha - start_alpha)
        conv = _conv_self(a)
        M = conv.max()
        exp_vals = np.exp(alpha * (conv - M))
        Z = exp_vals.sum()
        if Z == 0:
            w = np.ones_like(conv) / conv.size
            Z = 1.0
        else:
            w = exp_vals / Z
        # compute Tg as convolution of w with reversed a (similar as before)
        a_rev = a[::-1]
        Tg = _fft_convolve(w, a_rev)  # length 3n-2
        Tg = Tg[n-1:n-1+n]  # length n slice
        grad = a + Tg / Z  # same as before
        # Adam update:
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        # step size: try lr = lr_init * target_sum / (norm + 1e-12)
        #but incorporate sqrt(v_hat)
        step_size = lr_init * target_sum / (np.linalg.norm(grad) + 1e-12)

"""
Hybrid multi‑scale optimiser for the max‑convolution bound.

Key ideas
---------
1. **Two‑scale search** – first optimise a modest‑size problem
   (n≈2000) where many iterations are cheap, then up‑sample the
   best small solution to the target size (up to 8000) and polish it.
2. **Adam‑based smoothed gradient descent** – minimise a
   log‑sum‑exp surrogate of the max‑convolution.  The surrogate
   becomes tighter as the temperature α grows, so α is increased
   gradually.  Adam’s adaptive steps together with Euclidean projection
   onto the ℓ¹‑simplex give stable, fast convergence.
3. **Diverse seeds** – power‑law, exponential and uniform‑simplex
   initialisations, plus the previous best construction (if any).
4. **Deterministic mass‑transfer polishing** – a greedy move of mass
   from a “donor’’ index that contributes most to the current peak to
   a “receiver’’ that contributes least, with a Metropolis acceptance
   schedule.
5. **Reordering & trimming** – try a few permutations (alternating,
   even‑spread, reversal) and drop trailing near‑zero entries.

The function respects the global time budget (default 1000 s) and
returns a list of non‑negative heights ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (real‑to‑real)."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective & projection utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum a)^2.
    When Σa = √(2n) this is simply max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray], target_sum: float,
                 deadline: float, F: float = 0.5, CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after every step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # select three distinct mates a,b,c ≠ i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # safety margin
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent using Adam (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           start_alpha: float = 2.0,
                           end_alpha: float = 20.0,
                           lr_factor: float = 0.7,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate ``log_sum_exp(alpha·(a*a))/alpha``
    using Adam with a linearly increasing temperature α.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    start_time = time.time()
    it = 0
    while time.time() < deadline:
        # linearly increase α from start_alpha to end_alpha
        prog = (time.time() - start_time) / (deadline - start_time)
        alpha = start_alpha + prog * (end_alpha - start_alpha)

        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        # compute softmax weights  w_i ∝ exp(alpha·(conv_i-M))
        # the subtraction of M avoids overflow
        w_raw = np.exp(alpha * (conv - M))
        Z = w_raw.sum()
        if Z == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
            Z = 1.0
        else:
            w = w_raw / Z

        # gradient term from the smooth max:
        #   grad = a + (w * a_rev) convolved (as in the original code)
        a_rev = a[::-1]
        Tg_full = _fft_convolve(w, a_rev)               # length 3n‑2
        # slice the central n entries (indices n‑1 … n‑1+n‑1)
        Tg = Tg_full[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
        grad = a + Tg / Z

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step size proportional to the current gradient norm,
        # tuned by lr_factor.
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break
        step = lr_factor * target_sum / (grad_norm + 1e-12)

        # Adam update (element‑wise)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new, target_sum)

        cur_score = _score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new
        it += 1

        # occasional early‑exit if we are out of time
        if time.time() > deadline - 0.01:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing (same as prior version)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # rollback
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score
