def _eg_subgradient(a, target_sum, deadline):
    best = a.copy(); best_score = _score(a)
    eta0 = 0.5; t = 0
    while time.time() < deadline:
        # compute conv
        conv = _conv_fft(a)
        max_val = conv.max()
        idxs = np.where(conv >= max_val - 1e-12)[0] # all peaks
        # compute subgradient g
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (len(a)-1))
            hi = min(k, len(a)-1)
            # partner slice = a[ k - i ] for i in lo..hi
            i_range = np.arange(lo, hi+1)
            partner = a[ k - i_range ]
            g[i_range] += 2 * partner
        g /= len(idxs)
        # step size
        eta = eta0 / np.sqrt(t+1)
        # exponentiated update
        a = a * np.exp(-eta * g)
        # renormalize
        sum_a = a.sum()
        if sum_a <= 0:
            break
        a = a * (target_sum / sum_a)
        # keep best
        sc = _score(a)
        if sc < best_score:
            best_score = sc; best = a.copy()

def _eg_subgradient(a0, target_sum, time_deadline):
   a = a0.copy()
   best_a = a.copy()
   best_val = _score(a)
   it = 0
   eta0 = 0.5   # or maybe 1.0
   start = time.time()
   n = a.shape[0]
   while time.time() < time_deadline:
       # compute conv
       conv = _conv_fft(a)
       max_val = conv.max()
       idxs = np.where(conv >= max_val - 1e-12)[0]  # tolerance
       # Compute subgradient g
       g = np.zeros_like(a)
       # average across all kp (if many)
       for k in idxs:
           lo = max(0, k - (n - 1))
           hi = min(k, n - 1)
           i_range = np.arange(lo, hi+1)
           partner = a[k - i_range]
           g[i_range] += 2.0 * partner
       # average if multiple idxs
       if idxs.size > 0:
           g /= idxs.size
       # step size
       eta = eta0 / np.sqrt(it+1)
       # exponentiated update: a <- a * exp(-eta * g)
       a = a * np.exp(-eta * g)
       # renormalize onto simplex
       sum_a = a.sum()
       if sum_a == 0:   # should not happen
           a = np.full_like(a, target_sum / n)
       else:
           a = a * (target_sum / sum_a)
       # compute current score
       val = _score(a)
       if val < best_val - 1e-12:
           best_val = val
           best_a = a.copy()
           print(f"[{time.time():.2f}] EG iter {it} score {best_val:.6f}")
       it += 1
   return best_a, best_val

        p = rng.uniform(0.2,1.2)
        a = _power_law_sequence(n, p, target_sum)

"""
Step‑function optimizer for the “max self‑convolution” challenge.

Main ideas
----------
1️⃣  Initialise with a cheap parametric sweep (power‑law, tilted power‑law,
    Gaussian) and keep the best shape.
2️⃣  Run many *exponentiated‑gradient* (EG) sub‑gradient iterations.
    The sub‑gradient of the max‑convolution is simply 2·a shifted by the
    current peak index; EG naturally respects non‑negativity and, after a
    renormalisation step, stays on the simplex Σa = √(2 n).
3️⃣  Apply a deterministic “mass‑transfer” refinement (the exact DMT2
    routine) to lower the peak further.
4️⃣  (Optional) A tiny LP‑based Frank‑Wolfe step and a short Adam polishing
    pass are performed if time permits.
The whole search respects a 1000 s budget; intermediate improvements are
printed and the best sequence (trailing near‑zeros trimmed) is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(b)/(sum(a)^2).  For Σa = √(2 n) the denominator equals 2·n,
    hence the score equals max(b).  The routine is safe for internal use.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
#  Random initialiser (used for EG restarts)
# ----------------------------------------------------------------------


def _random_init(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Return a random non‑negative sequence on the simplex."""
    choice = rng.integers(5)
    if choice == 0:
        p = rng.uniform(0.2, 1.2)
        a = _power_law_sequence(n, p, target_sum)
    elif choice == 1:
        p = rng.uniform(0.2, 1.2)
        q = rng.uniform(0.0, 8.0)
        a = _exp_power_law_sequence(n, p, q, target_sum)
    elif choice == 2:
        sigma = rng.uniform(0.05 * n, 0.5 * n)
        a = _gaussian_sequence(n, sigma, target_sum)
    elif choice == 3:
        a = np.full(n, target_sum / n)
    else:
        a = rng.exponential(scale=1.0, size=n)
        a = a / a.sum() * target_sum

    # small jitter to avoid exact ties
    jitter = rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a + jitter, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
#  Exponentiated‑gradient sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _eg_subgradient(a0: np.ndarray,
                    target_sum: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with the exponentiated‑gradient update.
    Sub‑gradient of max(a*a) at the current peak(s) is 2·a shifted.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    n = a.shape[0]
    it = 0
    eta0 = 0.5  # initial step size

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # collect up to five indices attaining (or nearly attaining) the max
        idxs = np.where(conv >= max_val - 1e-12)[0]
        if idxs.size > 5:
            idxs = idxs[:5]

        # sub‑gradient (average over selected peak indices)
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            # partner = a_{k-i}
            partner = a[k - i_range]
            g[i_range] += 2.0 * partner
        g /= max(1, idxs.size)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (guarantees positivity)
        a = a * np.exp(-eta * g)

        # re‑project onto the simplex (simple scaling works because all entries stay ≥0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, target_sum / n)
        else:
            a = a * (target_sum / s)

        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] EG it={it:6d}  score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
#  Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # locate the second‑largest value (used as target)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # choose one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner values a_{k‑i}
        partner = a[k - idx]

        # donor = index with largest a_i * partner_i
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index with smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:6d}  score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  LP‑based Frank‑Wolfe refinement (optional)
# ----------------------------------------------------------------------


def _lp_iterative_refinement(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             epsilon_rel: float = 0.02,
                             step: float = 0.20,
                             max_iters: int = 120) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style updates using a tiny linear programme.
    In each iteration we solve:
        max Σ g_j   s.t.   a * g ≤ M   (only on the currently active peak indices)
    The solution g is scaled to Σg = target_sum and blended with a.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    eps = epsilon_rel
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        idx = np.where(conv >= M - eps * M)[0]
        if idx.size == 0:
            break
        if idx.size > 30:
            idx = idx[np.argsort(-conv[idx])[:30]]

        C = idx.size
        A = np.zeros((C, n), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            A[row, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

        b = np.full(C, M, dtype=np.float64)

        try:
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception as e:
            print(f"[{time.time():.2f}] LP exception: {e}")
            break
        if not res.success:
            eps = min(eps * 2.0, 0.1)
            it += 1
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g = g * (target_sum / sum_g)

        a_new = (1.0 - step) * a + step * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            eps = max(eps * 0.8, 1e-5)
            print(f"[{time.time():.2f}] LP it={it:3d} eps={eps:.2e} score={best_score:.6f}")
        else:
            step = max(step * 0.6, 0.01)
            eps = min(eps * 1.5, 0.05)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Utility – trim trailing near‑zero entries
# ----------------------------------------------------------------------


def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]
    if nz.size == 0:
        return [0.0]
    return a[: nz[-1] + 1].tolist()

#  Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing
    evaluate_sequence(...).  Returns the best sequence found as a Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but higher cost)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)  →  score = max convolution

    rng = np.random.default_rng()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    best_seq = _power_law_sequence(n, 0.5, target_sum)   # fallback
    best_score = _score(best_seq)

    # power‑law sweep
    for p in np.linspace(0.30, 0.85, 31):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # exp‑tilted power‑law sweep (broader ranges)
    for p in np.linspace(0.30, 0.85, 21):
        for q in np.linspace(0.0, 8.0, 21):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.5 * n, 7):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # if a previously stored solution exists, reuse it (resize if needed)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)

            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)

    print(f"[{time.time():.2f}] Initial best after parametric sweep = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – EG sub‑gradient descent with many random restarts
    # --------------------------------------------------------------
    remaining = deadline - time.time()

    eg_deadline = time.time() + eg_time

        per_run_deadline = time.time() + min(12.0, eg_deadline - time.time())
        seq, val = _eg_subgradient(init, target_sum, per_run_deadline)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] EG restart improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exact deterministic mass‑transfer refinement (core)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT2 improved best = {best_score:.6f}")

                                                step=0.20,

            print(f"[{time.time():.2f}] LP refinement improved best = {best_score:.6f}")

        def _gradient_softmax(a: np.ndarray, beta: float):

            a_fft = np.fft.rfft(a, n=size)
            b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

                val = _score(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

                if it % beta_inc_iter == 0 and beta < max_beta:
                    beta = min(beta * 2.0, max_beta)

                it += 1

            return best_a, best_val

        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,

                                                step_factor=0.6,
                                                gamma=0.995)

            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing zeros and evaluate final score

    final_seq = _trim_sequence(best_seq)

        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
