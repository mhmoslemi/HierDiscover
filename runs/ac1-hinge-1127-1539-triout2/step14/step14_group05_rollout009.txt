"""
Hybrid multi‑scale optimizer for the step‑function benchmark.

The pipeline:
1. Build a modest‑size candidate pool from several parametric families.
2. Run many short soft‑max (softmax‑Adam) gradient descent runs on the pool.
3. Refine the best small solution with a *min‑t* linear program (active‑set)
   that directly reduces the maximal convolution value.
4. Upscale the refined solution to a larger support (≈15‑25k) and reorder
   the entries to spread the large masses.
5. Perform a long exact sub‑gradient descent (Adam) on the large problem.
6. Apply a second LP refinement with a larger active set.
7. Finish with random permutation + hill‑climb and a short mass‑transfer
   simulated‑annealing pass.
All steps respect the non‑negativity and ℓ¹‑simplex constraints and keep
projecting back onto the simplex after each update.
"""

import time
import math
import random
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Sinusoid peaking in the centre."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (to scatter large values)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the sorted values alternately at the left and right ends."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Scatter values using a step that is coprime with the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

# Soft‑max gradient descent with Adam (smooth approximation of max)
# ----------------------------------------------------------------------


def _softmax_adam_descent(a0: np.ndarray,
                          deadline: float,
                          target_sum: float,
                          max_beta: float = 9e5,
                          step_factor: float = 0.9,
                          gamma1: float = 0.9,
                          gamma2: float = 0.999,
                          eps: float = 1e-12) -> Tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of the max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t_iter = 0

    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())

        # FFT of current a
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = b.max()
        # soft‑max probabilities
        exps = np.exp(beta * (b - max_b))
        p = exps / (exps.sum() + eps)

        # gradient of soft‑max objective w.r.t. a
        a_rev = a[::-1]
        Pr_fft = np.fft.rfft(p, n=size)
        Ar_fft = np.fft.rfft(a_rev, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = gamma1 * m + (1 - gamma1) * grad
        v = gamma2 * v + (1 - gamma2) * (grad ** 2)
        m_hat = m / (1 - gamma1 ** (t_iter + 1))
        v_hat = v / (1 - gamma2 ** (t_iter + 1))
        step = step_base / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t_iter % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Soft‑max Adam it={t_iter:5d} β={beta:8.1f} score={best_val:.6f}")

        # increase β progressively
        if t_iter % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        t_iter += 1

    return best_a, best_val

def _exact_subgrad_adam(a0: np.ndarray,
                        deadline: float,
                        target_sum: float,
                        step_factor: float = 0.55,
                        gamma1: float = 0.9,
                        gamma2: float = 0.999,
                        tol: float = 0.99,
                        eps: float = 1e-12) -> Tuple[np.ndarray, float]:
    """
    Adam on the exact sub‑gradient of max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    rng = np.random.default_rng()
    t_iter = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # mask of indices that are within tol of the max
        I = (conv >= maxb * tol).astype(np.float64)

        # gradient = 2 * a_rev * I (convolution)
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        # Adam update
        m = gamma1 * m + (1 - gamma1) * grad
        v = gamma2 * v + (1 - gamma2) * (grad ** 2)
        m_hat = m / (1 - gamma1 ** (t_iter + 1))
        v_hat = v / (1 - gamma2 ** (t_iter + 1))
        step = (target_sum * step_factor) / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if t_iter % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact sub‑grad Adam it={t_iter:5d} score={best_val:.6f}")

        t_iter += 1

    return best_a, best_val

def _lp_min_t_refine(a0: np.ndarray,
                     target_sum: float,
                     max_iters: int,
                     deadline: float,
                     K: int = 20,
                     mix_factor: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Small LP that, given the current a, finds a vector g and a bound t
    such that a * g ≤ t on the K most critical convolution positions.
    The new a is a convex mixture (1‑t)·a + t·g.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        print("[LP] SciPy not available – skipping LP refinement.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        # active set: K indices with largest convolution values
        if K >= conv.size:
            active = np.arange(conv.size, dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # Build matrix A (len(active) × n)
        rows = len(active)
        A = np.zeros((rows, n), dtype=np.float64)
        for r, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            A[r, idx] = a[k - idx]

        # LP: min t  s.t. A g - t ≤ 0,  Σg = target_sum, g ≥ 0, t ≥ 0
        c = np.concatenate([np.zeros(n), [1.0]])          # objective: minimize t
        A_ub = np.hstack([A, -np.ones((rows, 1))])        # A g - t ≤ 0
        b_ub = np.zeros(rows)
        A_eq = np.concatenate([np.ones((1, n)), np.zeros((1, 1))], axis=1)
        b_eq = np.array([target_sum])
        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x[:n]
        t_val = res.x[-1]

        # Mix the new vector into the current one
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LP] iter={it} new best={best_val:.6f} (t≈{t_val:.6f})")

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation search – random shuffles & hill‑climb
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> Tuple[np.ndarray, float]:
    """Try a number of random permutations and keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> Tuple[np.ndarray, float]:
    """Swap random pairs while they improve the max‑convolution."""
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]          # revert
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[Hill‑climb] attempts={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_sa(a0: np.ndarray,
                     deadline: float,
                     target_sum: float,
                     start_temp: float = 0.25,
                     end_temp: float = 1e-5,
                     low_frac: float = 0.05) -> Tuple[np.ndarray, float]:
    """
    Transfer a random amount of mass from a “donor’’ position (large
    product contribution) to a “receiver’’ (small contribution) using
    simulated‑annealing acceptance.
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[MT] new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to avoid stagnation
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns the best sequence found within the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1. Small‑scale exploration (≈4 k elements)
    # --------------------------------------------------------------
    n_small = 4200
    target_small = math.sqrt(2.0 * n_small)

    init_pool = []

    # parametric families
    for p in np.linspace(0.1, 2.0, 14):
        init_pool.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.0003, 0.025, 12):
        init_pool.append(_exponential_sequence(n_small, float(lam), target_small))
    for _ in range(6):
        init_pool.append(_cosine_sequence(n_small, target_small))
    for _ in range(6):
        init_pool.append(_sin_half_sequence(n_small, target_small))
    for _ in range(14):
        init_pool.append(_random_simplex(n_small, target_small, rng))

    # reuse previously discovered sequence if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_small))

    # jittered copies of the pool (adds diversity)
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_small * 0.03, size=n_small)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # Run multiple soft‑max Adam runs on random seeds until ~40 s left
    while time.time() < deadline - 40.0:
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_small * 0.004, size=n_small)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 13.0, deadline - 20.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _softmax_adam_descent(seed,
                                                run_deadline,
                                                target_small,
                                                max_beta=9e5,
                                                step_factor=0.9,
                                                gamma1=0.9,
                                                gamma2=0.999)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[Exploration] new best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. LP refinement on the modest size
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 25.0:
        lp_deadline = time.time() + min(18.0, deadline - time.time() - 12.0)
        best_seq, lp_score = _lp_min_t_refine(best_seq,
                                              target_small,
                                              max_iters=30,
                                              deadline=lp_deadline,
                                              K=20,
                                              mix_factor=0.18)
        if lp_score < best_score:
            best_score = lp_score
            print(f"[LP‑refine] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Upscaling to a larger support
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600.0:
        n_big = 26000
    elif remaining > 400.0:
        n_big = 22000
    elif remaining > 250.0:
        n_big = 18000
    elif remaining > 150.0:
        n_big = 15000
    else:
        n_big = n_small

    target_big = math.sqrt(2.0 * n_big)

    if best_seq is None:
        best_seq = _power_law_sequence(n_small, 0.6, target_small)

    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # try a few deterministic re‑ordering strategies and keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[Upscale] n={n_big}, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 4. Heavy exact sub‑gradient descent (Adam)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        seq_big, sub_score = _exact_subgrad_adam(seq_big,
                                                 sub_deadline,
                                                 target_big,
                                                 step_factor=0.55,
                                                 gamma1=0.9,
                                                 gamma2=0.999,
                                                 tol=0.99)
        if sub_score < best_score:
            best_score = sub_score
            print(f"[Sub‑grad] new best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Second LP refinement on the large problem
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        lp2_deadline = time.time() + min(12.0, deadline - time.time() - 8.0)
        seq_big, lp2_score = _lp_min_t_refine(seq_big,
                                              target_big,
                                              max_iters=25,
                                              deadline=lp2_deadline,
                                              K=30,
                                              mix_factor=0.15)
        if lp2_score < best_score:
            best_score = lp2_score
            print(f"[LP‑refine‑2] new best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Permutation search (random shuffles + local hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline - 15.0:
        perm_deadline = time.time() + min(9.0, deadline - time.time() - 6.0)
        seq_big, _ = _permutation_search(seq_big, trials=120, deadline=perm_deadline)
        print(f"[Perm] after random shuffles max={_max_convolution_fft(seq_big):.6f}")

    if time.time() < deadline - 12.0:
        hill_deadline = time.time() + min(6.0, deadline - time.time() - 4.0)
        seq_big, _ = _local_permutation_hillclimb(seq_big, hill_deadline)
        print(f"[Hill‑climb] max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 7. Final mass‑transfer simulated‑annealing pass
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = time.time() + min(5.0, deadline - time.time() - 2.0)
        seq_big, mt_score = _mass_transfer_sa(seq_big,
                                              mt_deadline,
                                              target_big,
                                              start_temp=0.2,
                                              end_temp=1e-5,
                                              low_frac=0.06)
        if mt_score < best_score:
            best_score = mt_score
            print(f"[Final MT] best={best_score:.6f}")

    # --------------------------------------------------------------
    # 8. Trim trailing near‑zero entries and return the result
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    # expose for possible reuse in later runs
    globals()["height_sequence_1"] = final_seq

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[Result] final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
