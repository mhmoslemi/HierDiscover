"""
Hybrid search for low‐maximimum convolution step functions.

The algorithm follows the same multi‑stage strategy as the reference
implementation but with a larger final problem size, a longer
sub‑gradient phase and an aggressive permutation / hill‑climbing
stage.  In addition to the original ordering heuristics we add a
bit‑reversal order and a random coprime‑step order, and we devote more
time to LP refinement (more active constraints).  The search is fully
self‑contained; `evaluate_sequence` is assumed to be available in the
environment and is called only for the final reporting.

The objective for a sequence with ℓ¹‑norm √(2 n) equals the maximum
entry of its self‑convolution, therefore we try to minimise this value.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Permutation helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _bitrev_order(seq: np.ndarray) -> np.ndarray:
    """Order by bit‑reversed index (mod n)."""
    n = len(seq)
    bits = (n - 1).bit_length()
    # compute revbits for all positions
    rev = np.arange(n, dtype=np.int64)
    rev = ((rev & 0x55555555) << 1) | ((rev & 0xAAAAAAAA) >> 1)
    rev = ((rev & 0x33333333) << 2) | ((rev & 0xCCCCCCCC) >> 2)
    rev = ((rev & 0x0F0F0F0F) << 4) | ((rev & 0xF0F0F0F) >> 4)
    rev = ((rev & 0x00FF00FF) << 8) | ((rev & 0xFF00FF00) >> 8)
    rev = ((rev & 0x0000FFFF) << 16) | ((rev & 0xFFFF0000) >> 16)
    rev = rev >> (32 - bits)
    order = np.argsort(rev % n)
    return seq[order]


def _random_step_order(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random coprime step permutation (like even_spread but random)."""
    n = len(seq)
    step = rng.integers(1, n)
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max operator)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995,
                         tol: float = 0.99) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on max‑convolution with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            # occasional jitter to escape shallow plateaus
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program refinement (active‑set)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               max_iters: int,
               deadline: float,
               K: int = 12,
               mix_factor: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Solve a small LP that maximises ℓ¹‑norm of a new sequence g under the
    active convolution constraints a·g ≤ M on the K most critical indices.
    The updated sequence is (1‑t)·a + t·g.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        print("SciPy not available – skipping LP refinement.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        # choose top‑K active constraints
        if K >= conv.size:
            active = np.arange(conv.size, dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # Build constraint matrix A (size |active| × n)
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            A[row, idx] = a[k - idx]   # coefficient of g_j is a_{k-j}

        # Solve: maximise Σ g_j   <=> minimise -Σ g_j
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        # rescale to exact ℓ¹‑norm
        g = g * (target_sum / g.sum())

        # convex combination
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} best={best_val:.6f}")

    return best_a, best_val

def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Try many random shuffles and keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """Swap‑based hill climbing (accept only improving moves)."""
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # try swap
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]      # revert
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Move mass between two positions (preserving ℓ¹) with SA acceptance."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate index of current maximum in convolution
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to keep the walk alive
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low maximum
    self‑convolution.  The method runs several stages (exploration,
    LP refinement, up‑scaling, long sub‑gradient descent, heavy
    permutation search, hill‑climb and a final mass‑transfer SA) and
    returns the best sequence found within the time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1. Exploration on a modest size (≈8 000)
    # --------------------------------------------------------------
    n_expl = 8000
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool = []

    # parametric families
    for p in np.linspace(0.2, 2.0, 14):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0005, 0.025, 12):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for _ in range(8):
        init_pool.append(_cosine_sequence(n_expl, target_expl))
    for _ in range(8):
        init_pool.append(_sin_half_sequence(n_expl, target_expl))
    for _ in range(14):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # reuse previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))
        # jittered version of the previous
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seqj = _resize_sequence(prev, n_expl) + jitter
        seqj = np.maximum(seqj, 0.0)
        seqj = _project_to_simplex(seqj, target_expl)
        init_pool.append(seqj)

    best_seq = None
    best_score = np.inf

    # allocate ~120 s for this stage
    exploration_deadline = min(deadline - 200.0, start_time + 120.0)

    while time.time() < exploration_deadline:
        # pick a seed (occasionally bias towards current best)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # run a short sub‑gradient phase
        run_deadline = min(time.time() + 9.0, exploration_deadline)
        a_opt, val_opt = _subgradient_descent(seed,
                                               run_deadline,
                                               target_expl,
                                               step_factor=0.65,
                                               gamma=0.995,
                                               tol=0.995)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. LP refinement on the modest size
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        lp_deadline = time.time() + min(35.0, deadline - time.time() - 115.0)
        best_seq, lp_score = _lp_refine(best_seq,
                                        target_expl,
                                        max_iters=50,
                                        deadline=lp_deadline,
                                        K=30,
                                        mix_factor=0.2)
        if lp_score < best_score:
            best_score = lp_score
            print(f"[{time.time():.2f}] After LP refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Upscale to a larger problem size (≈20‑30 k)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 400:
        n_big = 30000
    elif remaining > 250:
        n_big = 25000
    elif remaining > 150:
        n_big = 20000
    else:
        n_big = n_expl

    target_big = math.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_seq if best_seq is not None else _power_law_sequence(n_expl, 0.6, target_expl),
                               n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # try several deterministic reorderings and keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        _bitrev_order(seq_big),
        _random_step_order(seq_big, rng)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 4. Heavy sub‑gradient descent on the large scale
    # --------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sub_deadline = deadline - 100.0
        seq_big, sub_score = _subgradient_descent(seq_big,
                                                  sub_deadline,
                                                  target_big,
                                                  step_factor=0.55,
                                                  gamma=0.995,
                                                  tol=0.99)
        if sub_score < best_score:
            best_score = sub_score
            print(f"[{time.time():.2f}] Sub‑grad large best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Aggressive permutation search
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        perm_deadline = time.time() + min(25.0, deadline - time.time() - 60.0)
        seq_big, perm_score = _permutation_search(seq_big,
                                                  trials=4000,
                                                  deadline=perm_deadline)
        if perm_score < best_score:
            best_score = perm_score
            print(f"[{time.time():.2f}] Permutation best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Local hill‑climb (swap‑based)
    # --------------------------------------------------------------
    if time.time() < deadline - 55.0:
        hill_deadline = time.time() + min(20.0, deadline - time.time() - 35.0)
        seq_big, hill_score = _local_permutation_hillclimb(seq_big, hill_deadline)
        if hill_score < best_score:
            best_score = hill_score
            print(f"[{time.time():.2f}] Hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Final mass‑transfer simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        mt_deadline = time.time() + min(15.0, deadline - time.time() - 20.0)
        seq_big, mt_score = _mass_transfer_simulated_annealing(seq_big,
                                                               mt_deadline,
                                                               target_big,
                                                               start_temp=0.25,
                                                               end_temp=1e-5,
                                                               low_frac=0.05)
        if mt_score < best_score:
            best_score = mt_score
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # 8. Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # final report (uses the provided security‑checked evaluator)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
