"""
Low‑peak step‑function search.

The algorithm works in three stages:
1.  Fast exploration on a moderate length (≈4000).  Several simple families
    (power‑law, exponential, random Dirichlet) are projected onto the ℓ¹‑simplex
    Σa = √(2·n) and improved by a short sub‑gradient descent on the true
    convolution peak.
2.  The best candidate is resized to a large length (12k‑20k) and optionally
    reordered by an alternating “large‑values apart” scheme.
3.  A long annealed soft‑max gradient descent (high β) followed by a short
    sub‑gradient refinement and a targeted permutation‑annealing finishes the
    search.  All updates stay on the simplex, so the evaluation score
    2·n·max(a*a)/‖a‖₁² is invariant to scaling.

The routine respects the supplied time budget (≈1000 s) and returns a
non‑negative sequence that improves the known bound (target < 1.503).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(Σa)² (independent of global scaling)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, second‑largest,… placed alternately from left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution over shifts

    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Annealed soft‑max gradient descent (projected)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth soft‑max surrogate of the peak.
    The temperature β is doubled periodically up to max_beta.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β periodically
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true peak (fast per iteration)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution entry."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # Gradient of the max entry (simple linear term)
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]      # derivative of a[mid]^2

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005,
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing (swap high/low contributors)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that swaps an index with a high contribution to the
    current peak with an index with a low contribution.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # swap the heights
        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

    # power‑law families
    for p in (0.4, 0.55, 0.7):
        seeds.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential families
    for lam in (0.002, 0.008, 0.015):
        seeds.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like seeds
    for _ in range(4):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        seeds.append(a)

    # uniform baseline
    a_uni = np.full(n_expl, target_expl / n_expl)
    seeds.append(a_uni)

    # previous best (if any)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) > 0:
        try:
            prev_arr = np.array(prev, dtype=float)
            prev_resized = _resize_sequence(prev_arr, n_expl)
            prev_resized = _project_to_simplex(prev_resized, target_expl)
            seeds.append(prev_resized)
        except Exception:
            pass

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Phase 2 – cheap sub‑gradient refinement on the exploration size
    # --------------------------------------------------------------
    # Allocate ~30 s for this phase (or the whole budget if it is tiny)
    expl_deadline = deadline - 30.0
    if expl_deadline < time.time():
        expl_deadline = time.time() + 1.0

    for seed in seeds:
        if time.time() >= expl_deadline:
            break
        # give each seed at most 6 s
        per_dead = min(time.time() + 6.0, expl_deadline)
        a_opt, val_opt = _subgradient_descent(seed, per_dead, target_expl,
                                              max_iter=200_000)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Seed improvement score={best_score:.6f}")

    # Fallback (should never happen)
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # --------------------------------------------------------------
    # Phase 3 – upscale to a larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # Choose a larger n if we have enough time left
    if remaining > 250.0 and budget_s >= 800:
        n_final = 20000
    elif remaining > 180.0:
        n_final = 15000
    elif remaining > 100.0:
        n_final = 12000
    else:
        n_final = max(n_expl, int(remaining * 10))   # crude fallback

    n_final = max(n_expl, n_final)                 # never shrink
    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Try a few systematic reorderings (alternating placements)
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] After upscaling n={n_final}, start score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – long high‑β soft‑max gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        gd_deadline = time.time() + remaining * 0.70
        a_gd, val_gd = _annealed_gradient_descent(
            best_seq,
            gd_deadline,
            target_final,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] GD improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – sub‑gradient refinement on the large problem
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_deadline = time.time() + remaining * 0.5
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_deadline,
            target_final,
            max_iter=500_000,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad after GD best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – targeted permutation annealing (more time if left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        perm_deadline = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_deadline,
            start_temp=0.4,
            end_temp=1e-5,
            top_frac=0.09,
            low_frac=0.09,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Permutation final best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 7 – final trimming and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # (Optional) final evaluation for logging
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as e:
        print("Evaluation error:", e)

    return final_seq
