def subgradient_descent(a0, target_sum, deadline):
    # a0 shape (n)
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    
    rng = np.random.default_rng()
    start = time.time()
    # parameters
    mu = 0.9
    v = np.zeros_like(a)
    while time.time() < deadline:
        # compute convolution and max index
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]
        # step size: decreasing over time
        elapsed = time.time() - start
        total = deadline - start
        eta = 4.0 * (1.0 - elapsed/total) + 0.01  # larger
        v = mu * v - eta * grad
        a = a + v
        a = _project_to_simplex(a, target_sum)
        # occasionally jitter
        if np.random.rand() < 0.01:
            jitter = rng.normal(scale=target_sum*1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
        # maybe break early if no improvement for many loops? not needed
    return best_a, best_val

"""
Propose a low‑score non‑negative step sequence.

The algorithm works in three stages:

1. **Extensive analytic sweep** – we evaluate many power‑law (a_i ∝
   (i+1)^(-p)) and power‑law × exponential families on a moderate
   length (n≈12000).  The sweep uses a very fine grid on the exponent
   (step 0.005) and a modest grid on the exponential cutoff.  The best
   shape from this sweep is taken as a warm start.

2. **Projected sub‑gradient descent** – we minimise the true
   max‑convolution entry by taking a sub‑gradient of the active peak
   and projecting after every step onto the simplex
   {x ≥ 0, Σx = √(2·n)}.  Momentum is added and the step‑size decays
   linearly.  This phase consumes roughly 50 % of the remaining time.

3. **Aggressive simulated‑annealing** – we repeatedly move a small
   amount of mass from indices that contribute heavily to the current
   peak towards indices with little height.  The convolution is updated
   in O(n) per move, and a square‑root cooling schedule is used.
   This phase uses the rest of the time budget.

All helper functions are defined at top level; no closures or
lambdas are used.  The final sequence is returned as a plain list of
floats and stored in ``height_sequence_1`` for possible warm‑starts in
future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of a power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    ``a`` is assumed to be non‑negative; the function does not clip or
    rescale the vector – it is used only internally during the search.
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:                     # sanity check – should never happen
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s} (the simplex)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n‑1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) * exp(-lam * i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape:
    a_i ∝ x^{α‑1}(1‑x)^{α‑1},  x = (i+0.5)/n .
    """
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (alpha - 1.0)
    )
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected sub‑gradient descent on the active max‑convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    v = np.zeros_like(a)               # momentum buffer
    mu = 0.9                           # momentum coefficient

    while time.time() < deadline:
        # --- current peak -------------------------------------------------
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]       # sub‑gradient of the active peak

        # --- step‑size (linear decay) ------------------------------------
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01   # >0 throughout

        # momentum update
        v = mu * v - eta * grad
        a = a + v
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally to prevent stagnation
        if int(elapsed * 100) % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad iter best={best_val:.6f}")

    return best_a, best_val


def _anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Aggressive simulated‑annealing with O(n) convolution updates."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                     # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # peak of the current convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak (a_i * a_{k‑i})
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with low height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # transfer magnitude (scaled by temperature)
        max_delta = min(a[donor] * 0.20, target_sum * 0.015)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- O(n) convolution update ------------------------------------
        b_new = b.copy()

        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a

        # quadratic corrections (Δ² terms)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal new best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute best={best_val:.6f}")

        it += 1

    return best_a, best_val

    return final_seq
