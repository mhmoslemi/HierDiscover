"""
Propose a step‑function (non‑negative vector) with a low
evaluate_sequence score.

Algorithm
---------
* Choose a problem size n (larger n → more degrees of freedom).
* Build several diverse initial vectors:
    - warm start (if a previous solution is present)
    - a simple power‑law shape a_i ∝ (i+1)^(-α)
    - a uniform vector
    - a few pure random vectors projected onto the simplex.
* For each start run two optimisation phases
    1. **Smooth‑max projected gradient descent**  
       – minimise a soft‑max approximation of the max‑convolution,  
       – use a gradually increasing temperature β (up to 150),  
       – simple momentum is added to stabilise the steps.  
    2. **Aggressive annealing with mass‑transfer**  
       – repeatedly move a tiny amount of mass from an index that
         contributes most to the current peak to an index that is
         very small, updating the convolution in O(n).  
       – temperature is cooled linearly (slow cooling gives more
         exploration).  
* Keep the best exact objective (using the true max‑convolution) and
  return the corresponding sequence.
* The final vector is trimmed of trailing near‑zero entries, stored in
  a global variable for warm‑starts in future runs and returned as a
  plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, power‑of‑2 size for speed)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b."""
    n, m = a.size, b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _objective_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact objective: 2·n·max(b) / (target_sum²)."""
    return float(2.0 * n * b.max() / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    # Duchi et al. (2008) – O(n) after sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_seq(alpha: float, n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-α) , projected onto the simplex."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-alpha)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max gradient descent (with simple momentum)
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 150.0,
                       momentum: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a*a) using projected gradient
    descent with a soft‑max (log‑sum‑exp) surrogate.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    it = 0
    # simple momentum buffer
    m_buf = np.zeros_like(a)

    # Base step size – a fraction of the total mass; adjusted per iteration.
    base_step = target_sum * 0.6

    while time.time() < deadline:
        # ---- current convolution and exact objective (for bookkeeping) ----
        b = _conv_self_fft(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max / (target_sum * target_sum)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it} best={best_val:.6f}")

        # ---- soft‑max weights (log‑sum‑exp) ----
        shift = b - cur_max               # max(b_shift) == 0
        w = np.exp(beta * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of smooth max (central slice) ----
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)            # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]      # central n entries

        # ---- momentum update ----
        m_buf = momentum * m_buf + (1.0 - momentum) * g

        # ---- step size (scaled by gradient norm) ----
        norm = np.linalg.norm(m_buf)
        if norm == 0.0:
            break
        eta = base_step / np.sqrt(it + 1.0) * (target_sum / norm)

        # ---- gradient step + simplex projection ----
        a = a - eta * m_buf
        a = _project_to_simplex(a, target_sum)

        # ---- increase β slowly (focus increasingly on true max) ----
        if it % 1500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.6)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – mass‑transfer annealing (linear cooling)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves tiny mass amounts from indices that
    heavily contribute to the current convolution peak to indices with
    very small height.  The full convolution is kept up‑to‑date in O(n)
    per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _objective_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # Initial temperature: 80 % of the current energy.
    T0 = cur_val * 0.8

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - frac))   # linear cooling

        # ---- locate the current peak of the convolution ----
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # ---- donor: weighted by contribution to the peak ----
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # ---- receiver: favour tiny heights ----
        inv = 1.0 / (a + 1e-14)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # ---- size of the transfer (temperature‑scaled) ----
        max_delta = min(a[donor] * 0.15, target_sum * 0.025)
        if max_delta <= 0.0:
            continue
        # scale Δ linearly with temperature (more aggressive early on)
        delta = rng.random() * max_delta * (temperature / T0)

        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----
        b_new = b.copy()
        # linear part: 2·a·Δ
        b_new[donor: donor + n] -= 2.0 * delta * a
        b_new[receiver: receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = _objective_from_selfconv(b_new, n, target_sum)

        # ---- Metropolis acceptance ----
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # ---- periodic exact recompute to avoid drift ----
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _objective_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val
