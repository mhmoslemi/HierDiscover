"""
Hybrid optimisation for the “minimum peak convolution” problem.

The code builds a diverse pool of seeds, runs a relatively large
Differential‑Evolution (DE) phase, and then spends the remaining
budget on aggressive local refinements:

* projected gradient descent on the exact max‑convolution (sub‑gradient
  with an adaptive step size);
* deterministic mass–transfer moves that always reduce the current
  peak; the move size is scaled with a slowly decaying temperature;
* a long simulated‑annealing loop that transfers mass from the most
  contributing indices of the peak to almost any other index (even
  outside the current support).

The refinements are performed repeatedly while time permits and the
best sequence found is finally reordered by three simple permutations.
The sequence is trimmed of trailing near‑zero entries, stored in the
global ``height_sequence_1`` (for warm‑starting the next run) and
returned as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Exact evaluation function used by the competition.
    It is invariant under scaling of ``a``.
    """
    n = a.shape[0]
    mx = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * mx / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed factories
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int, target_sum: float) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    out = np.interp(new_x, old_x, seq)
    out *= target_sum / out.sum()
    return out.astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (often lower the peak)
# ----------------------------------------------------------------------
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

# Differential Evolution
# ----------------------------------------------------------------------
def _de_optimize(population, target_sum, deadline,
                 F: float = 0.7, CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE with Euclidean simplex projection."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        # early exit to keep a safety margin
        if time.time() > deadline - 2.0:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-6,
                                low_frac: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer with a decaying temperature.
    Always transfers mass from an index that contributes heavily to the
    current peak to any other index (often far away) – this gives a
    stronger reduction than the original version which only chose a
    low‑contributing partner inside the active window.
    """
    a = a0.copy()
    conv = _conv_fft(a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start
        # linear temperature schedule
        tau = elapsed / (deadline - start)
        temp = start_temp * (1 - tau) + end_temp * tau

        # location of the current peak
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        # maybe the partner itself is a better donor
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        # choose a receiver *anywhere* (including far outside the active window)
        # but with a bias towards positions that hardly contribute to the peak
        if rng.random() < 0.7:
            # weighted towards low‑product indices inside the window
            low_cnt = max(1, int(len(idx) * low_frac))
            low_candidates = idx[np.argsort(prod)[:low_cnt]]
            receiver = int(rng.choice(low_candidates))
        else:
            # completely random receiver – allows the mass to disappear
            # from the active support
            receiver = int(rng.integers(n))

        if donor == receiver:
            continue

        # fraction of donor's mass, scaled by temperature
        max_delta = min(a[donor] * 0.5, a[donor] * temp)
        delta = max_delta * rng.random()
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_max = float(conv.max())

        # accept if it improves, otherwise Metropolis decided
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_max


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         init_step: float = 0.9,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with a simple back‑tracking line search."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        kmax = int(np.argmax(conv))

        # build sub‑gradient (only from the maximal index)
        n = a.shape[0]
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        grad = np.zeros_like(a)
        # each pair contributes twice except the central one when i==partner
        grad[idx] += 2.0 * a[partner]

        # simple line search on the scalar step size
        step = init_step * target_sum / (np.linalg.norm(grad) + 1e-12)
        while True:
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new, target_sum)
            new_val = _max_convolution_fft(a_new)
            if new_val <= maxb or step < 1e-12:
                break
            step *= 0.5

        a = a_new
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (global mass transfer)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """One‑temperature SA loop that moves mass from the peak contributors."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_max = cur_max
    start = time.time()

    # initial temperature as a fraction of the current peak height
    T0 = max(1e-4, cur_max * 0.25)
    T_end = 1e-9

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * (1 - frac) + T_end * frac

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        # sometimes pick the symmetric partner instead
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        # receiver anywhere (including far away) – this gives the strongest
        # possible reduction of the peak
        if rng.random() < 0.6:
            # try to pick a low‑product index inside the window
            low_cnt = max(1, int(len(idx) * 0.1))
            low_candidates = idx[np.argsort(prod)[:low_cnt]]
            receiver = int(rng.choice(low_candidates))
        else:
            receiver = int(rng.integers(n))

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        delta = max_delta * rng.random() * (T / T0)
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = _project_to_simplex(a_new, target_sum)

        conv_new = _conv_fft(a_new)
        new_max = float(conv_new.max())

        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine builds a rich seed pool, performs a sizable DE phase,
    then applies several aggressive local refinements (sub‑gradient,
    deterministic mass‑transfer and simulated annealing), finally tests a few
    reorderings and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # a tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size – larger n gives a richer design space.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 10000          # a bit larger than the previous 8000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000
    target_sum = 1.0       # we enforce an L1‑norm of 1 (any scale works)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.2, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random Dirichlet‑like points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm start from previous run if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n, target_sum))

    # jitter each seed a little to increase diversity
    jitter_scale = target_sum * 0.03
    for s in list(seeds):
        s_j = s + rng.normal(scale=jitter_scale, size=n)
        s_j = np.maximum(s_j, 0.0)
        s_j = _project_to_simplex(s_j, target_sum)
        seeds.append(s_j)

    # ------------------------------------------------------------------
    # Initialise DE population
    # ------------------------------------------------------------------
    pop_size = max(40, int(n / 220))   # around 45‑55 individuals
    population = []
    # insert the best seed right away
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())
    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # ------------------------------------------------------------------
    # Differential Evolution phase (≈ 45 % of the total budget)
    # ------------------------------------------------------------------
    de_deadline = start_time + 0.45 * budget_s
    if de_deadline > deadline - 30.0:
        de_deadline = deadline - 30.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE done – best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Local refinement loop – we repeat the three stages as long as
    # time permits.  The loops are deliberately generous to squeeze out
    # the last few decimals.
    # ------------------------------------------------------------------
    while time.time() < deadline - 15.0:
        remaining = deadline - time.time()

        # 1) Sub‑gradient descent (up to ~30 % of remaining budget)
        sub_dead = time.time() + min(remaining * 0.30, 30.0)
        seq_sg, val_sg = _subgradient_descent(best_seq, sub_dead, target_sum,
                                             init_step=0.8, gamma=0.997)
        if val_sg < _max_convolution_fft(best_seq):
            best_seq = seq_sg
            print(f"[{time.time():.2f}] sub‑grad improved – max conv = {val_sg:.7f}")

        # 2) Deterministic mass‑transfer (≈ 25 % of remaining)
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        mt_dead = time.time() + min(remaining * 0.25, 20.0)
        seq_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_sum,
                                                      start_temp=0.6,
                                                      end_temp=1e-7,
                                                      low_frac=0.20)
        if val_mt < _max_convolution_fft(best_seq):
            best_seq = seq_mt
            print(f"[{time.time():.2f}] deterministic MT improved – max conv = {val_mt:.7f}")

        # 3) Simulated annealing (use whatever time is left, but keep a
        #    small safety margin for final reordering)
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        sa_dead = time.time() + remaining - 5.0
        seq_sa, val_sa = _simulated_annealing(best_seq, target_sum,
                                              sa_dead, rng)
        if val_sa < _max_convolution_fft(best_seq):
            best_seq = seq_sa
            print(f"[{time.time():.2f}] SA improved – max conv = {val_sa:.7f}")

        # loop again – the next iteration will see a fresh landscape

    # ------------------------------------------------------------------
    # Final reordering pass (try a few simple permutations)
    # ------------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq)
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    final_score = _score(best_seq)
    print(f"[{time.time():.2f}] final score after reordering = {final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-13)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for warm‑starting the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator (optional)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] evaluated with official function = {val:.6f}")
    except Exception:
        pass

    return final_seq
