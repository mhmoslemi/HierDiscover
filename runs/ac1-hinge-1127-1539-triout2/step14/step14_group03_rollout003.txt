    best = a.copy()

def _conv_full_fft(x, y):
    n = x.size + y.size - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv

    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    t = 0
    start = time.time()
    while time.time() < deadline:
        # compute self-convolution
        b = _conv_fft(a)
        max_b = b.max()
        # epsilon factor
        eps = max_b * 0.07  # fraction; adjust
        if eps < 1e-12:
            eps = 1e-12
        # compute softmax weights (w normalized)
        w = np.exp((b - max_b) / eps)  # stable
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_full = _conv_full_fft(w, a_rev)  # length 3n-2

def anneal_mass_transfer(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()
    rng = np.random.default_rng()
    start = time.time()
    # compute initial temperature as cur_val
    T0 = cur_val * 0.8 + 1e-6
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = max(0.0, T0 * (1.0 - frac))
        # pick donor weighted by contribution to current peak
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.choice(idx)
        # receiver weighted by small a values
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)
        if donor == receiver:
            continue
        # delta size
        max_delta = min(a[donor] * 0.3, target_sum * 0.05)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # ensure non-negative
        if a_new[donor] < -1e-12:  # shouldn't happen
            continue
        # evaluate
        new_val = _score(a_new)
        accept = False
        if new_val <= cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True
        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved to {best_val:.6f}")
        # optional full recompute occasionally
        if t % 5000 == 0:
            cur_val = _score(a)
    return best_a, best_val

"""
Propose a low‑score non‑negative step sequence.

The algorithm consists of three stages:

1️⃣  **Parameterized warm‑start** – generate many deterministic families
    (power‑law, shifted power‑law, exponential, mixed power‑exp,
    symmetric Beta) together with a few random Dirichlet vectors.
    The best of these sequences is kept as a warm‑start.

2️⃣  **Smooth‑max projected gradient descent** – minimise a smooth
    approximation of the max‑convolution
    ( ε·log∑exp(b/ε) ) using the exact gradient.
    The gradient is obtained by a single FFT‑based convolution,
    the iterate is projected on the simplex {x≥0, Σx = √(2·n)}.
    A linearly decaying step‑size together with occasional jitter
    yields a high‑precision refinement.

3️⃣  **Aggressive mass‑transfer simulated annealing** – a fast O(n) update
    moves a small amount of mass from “bad’’ indices (large contribution
    to the current peak) to “good’’ indices (small height).  Temperature
    follows a linear cooling schedule, and every improvement is recorded.

All helper functions are defined at module level (no closures or lambdas)
and the routine respects the given time budget.  The final sequence is
trimmed of trailing zeros and stored globally for warm‑starts in later
invocations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    n = x.size + y.size - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²) (no normalisation)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_shifted_powerlaw(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift+1)^(-p).  shift ≥ 0 allows a softer start."""
    idx = np.arange(1, n + 1, dtype=np.float64) + shift
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n-1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam·i/(n-1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape on a uniform mesh."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (alpha - 1.0)
    )
    return a


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected gradient descent on ε·log‑sum‑exp(b/ε)."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # self‑convolution
        b = _conv_fft(a)
        max_b = b.max()
        # ε as fraction of the current peak (keeps the softmax sharp)
        eps = max(max_b * 0.07, 1e-12)

        # soft‑max weights – shifted to avoid overflow
        w = np.exp((b - max_b) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: g = 2·(w * a_rev)_{valid}
        a_rev = a[::-1]
        grad_full = _conv_full_fft(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[len(a) - 1 : len(a) - 1 + len(a)]

        # linearly decreasing step scale (absolute magnitude)
        elapsed = time.time() - start
        total = deadline - start
        decay = 0.5 * (1.0 - elapsed / total) + 0.02   # in [0.02,0.5]
        step = decay * target_sum / (np.linalg.norm(grad) + 1e-12)

        # gradient step + projection
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth_descent it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass transfer
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Fast O(n) mass‑transfer SA aimed directly at the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    T0 = cur_val * 0.9 + 1e-12
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = max(0.0, T0 * (1.0 - frac))          # linear cooling

        # current peak and its active window
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor chosen proportionally to its contribution to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.choice(idx)

        # receiver favours small heights
        inv = 1.0 / (a + 1e-12)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # amount to move (scaled by temperature)
        max_delta = min(a[donor] * 0.25, target_sum * 0.05)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        new_val = _score(a_new)

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # occasional full recompute to avoid drift
        if it % 5000 == 0:
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem dimension (larger n gives finer control)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)              # normalising constant

    # ------------------------------------------------------------------
    # Warm‑start / deterministic families
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # Warm start from a previously stored solution, if available
    if "height_sequence_1" in globals():
        wk = np.array(globals()["height_sequence_1"], dtype=np.float64)
        wk = _resize_sequence(wk, n)
        wk = wk * (target_sum / wk.sum())
        val = _score(wk)
        best_seq = wk.copy()
        best_val = val
        print(f"[{time.time():.2f}] Warm start score = {best_val:.6f}")

    # Sweep simple families (deterministic)
    for p in np.linspace(0.44, 0.62, 9):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} score={val:.6f}")

    # Shifted power‑law (adds a small offset to the denominator)
    for p in (0.45, 0.5, 0.55):
        for shift in (0.0, 2.0, 5.0, 10.0):
            a = _shape_shifted_powerlaw(n, p, shift)
            a = a * (target_sum / a.sum())
            val = _score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[{time.time():.2f}] ShiftPower p={p:.2f} shift={shift:.1f} score={val:.6f}")

    # Exponential decay
    for lam in np.linspace(0.5, 8.0, 7):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.3f} score={val:.6f}")

    # Mixed power‑exp (randomly sampled for diversity)
    rng = np.random.default_rng()
    for _ in range(12):
        p = rng.random() * 0.6 + 0.2          # 0.2 … 0.8
        lam = rng.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] PowerExp p={p:.3f} λ={lam:.3f} score={val:.6f}")

    # Symmetric Beta family (U‑shaped when α<1)
    for alpha in np.linspace(0.25, 0.9, 7):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.3f} score={val:.6f}")

    # Random Dirichlet starts (few)
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _score(rnd)
        if val < best_val:
            best_val, best_seq = val, rnd.copy()
            print(f"[{time.time():.2f}] Random start score={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max projected gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        grad_deadline = time.time() + 0.55 * remaining
        a_grad, val_grad = smooth_descent(best_seq, target_sum, grad_deadline)
        if val_grad < best_val:
            best_val, best_seq = val_grad, a_grad.copy()
            print(f"[{time.time():.2f}] After smooth descent best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – aggressive simulated‑annealing mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[{time.time():.2f}] After anneal best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing & output
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # Trim trailing zeros (tiny values are irrelevant for the evaluator)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final safety check using the provided evaluator
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm‑starts in the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
