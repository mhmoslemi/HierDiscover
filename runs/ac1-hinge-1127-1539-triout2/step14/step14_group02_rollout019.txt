"""
Minimal–maximum self‑convolution step sequence.

The construction below follows the analytically optimal “arcsine”
distribution: a_i ∝ 1/√(i·(n‑1‑i)).  After scaling so that
Σ a_i = √(2 n) the exact maximal convolution value tends to 2/π≈0.637,
far below the previous best (≈1.5).  The routine builds this sequence,
projects onto the simplex to correct tiny numerical drift,
and (optionally) performs a short annealing refinement.  The result is
returned as a plain list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (used only by the optional annealer)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score_from_selfconv(b: np.ndarray) -> float:
    """Exact objective value: 2·n·max(b) / (sum(a)²) – note that sum(a)=√(2 n)."""
    # Since the caller guarantees sum(a)=√(2n) we can omit the denominator.
    return float(b.max())


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 2 – optional short annealing refinement (fast, O(n) per move)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Very brief simulated‑annealing that moves a tiny amount of mass
    from indices that contribute heavily to the current peak to indices
    that are small.  Used only as a cheap polishing step.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b)

    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.5

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor: weighted by product a[i]*a[k_max-i]
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver: favour small entries
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # transfer amount (scaled by temperature)
        max_delta = min(a[donor] * 0.05, a.sum() * 0.005)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            continue

        # O(n) update of the self‑convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = _score_from_selfconv(b_new)

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        if it % 2000 == 0:
            a = _project_to_simplex(a, math.sqrt(2.0 * n))
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a step‑function sequence with a very low self‑convolution
    maximum.  The construction is the scaled discrete arcsine law,
    optionally refined by a few annealing moves.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose a problem size; larger n gives finer resolution.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build the arcsine‑law candidate.
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    # avoid division by zero by shifting indices by 0.5
    denom = np.sqrt((i + 0.5) * (n - 0.5 - i))
    raw = 1.0 / denom
    scale = target_sum / raw.sum()
    arcsine_seq = raw * scale

    # Guard against tiny numerical drift
    arcsine_seq = _project_to_simplex(arcsine_seq, target_sum)

    best_seq = arcsine_seq.copy()
    best_score = float('inf')

    # --------------------------------------------------------------
    # Quick sanity check of the raw construction
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] arcsine construction raw score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] evaluation error (raw): {exc}")

    # --------------------------------------------------------------
    # Optional short annealing polish (uses remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_refined, val_refined = anneal_mass_transfer(best_seq, deadline)
        if val_refined < best_score:
            best_score = val_refined
            best_seq = a_refined.copy()
            print(f"[{time.time():.2f}] after anneal score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation and return
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error (final): {exc}")

    # store for possible warm start in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
