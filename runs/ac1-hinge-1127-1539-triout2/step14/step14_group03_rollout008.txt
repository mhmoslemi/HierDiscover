def softmax_gradient(a, beta):
    conv = _conv_fft(a) # full conv.
    max_c = conv.max()
    # compute stable exponent
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # compute gradient using convolution of a and w_rev
    w_rev = w[::-1]
    # compute conv_rev = full convolution a * w_rev via FFT.
    L = a.shape[0]; M = w_rev.shape[0]
    size = 1 << ((L + M - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    conv_rev = np.fft.irfft(fa * fw, n=size)[:L + M - 1]  # length 3n-2
    # extract gradient slice: indices L-1 : 2*L-1 (since M=2L-1)
    # conv_rev shape is 3n -2.
    start = L - 1
    end = 2 * L - 1
    grad = 2.0 * conv_rev[start:end][::-1]
    return grad, conv.max()

Note that M = 2*L-1; correct.

    max_c = conv.max()
    exp_vals = np.exp(beta * (conv - max_c))

def smooth_descent(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    start = time.time()
    it = 0
    # precompute size for FFT
    total_len = 3*n - 2
    size_fft = 1 << ((total_len).bit_length())  # next power of 2
    # schedule for beta: linearly increase from beta0=2 to beta_max=40
    beta0 = 2.0
    beta_max = 40.0
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        # compute beta schedule: increase linearly
        beta = beta0 + (beta_max - beta0) * (elapsed / total)
        # compute conv
        conv = _conv_fft(a)  # using size 65536 maybe
        max_c = conv.max()
        # compute softmax weights
        # subtract max to avoid overflow
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        # w_rev:
        w_rev = w[::-1]
        # FFT conv of a and w_rev
        fa = np.fft.rfft(a, n=size_fft)
        fw = np.fft.rfft(w_rev, n=size_fft)
        conv_rev = np.fft.irfft(fa * fw, n=size_fft)[: total_len]
        # extract gradient slice and reverse
        grad = 2.0 * conv_rev[n-1:2*n-1][::-1]
        # step size
        # compute scale based on target sum and iteration
        step_factor = 0.02 * (1 - elapsed / total) + 0.001
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = step_factor * target_sum * grad / grad_norm
        a = a - step
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

            a = np.maximum(a + perturb, 0.0)
            a = _project_to_simplex(a, target_sum)
        it += 1
    return best_a, best_val

"""
Propose a low‑score non‑negative step sequence.

The algorithm works in three stages:

1. **Deterministic warm‑start** – a short sweep of analytically known
   families (power‑law, exponential, beta, mixed) plus a possible warm‑start
   from the previous run.  The best of these is taken as the initial vector.

2. **Smooth soft‑max gradient descent** – we minimise a smoothed version of
   the max‑convolution   f(a)=max_k (a*a)[k]  by replacing the max with a
   soft‑max (log‑sum‑exp).  The gradient of this smooth objective can be
   obtained by a single FFT‑based convolution, after which the iterate is
   projected onto the simplex {a≥0, Σa=√(2n)}.  The “temperature’’ β of the
   soft‑max is increased from a low value (global moves) to a high value
   (focus on the true peak).

3. **Aggressive mass‑transfer annealing** – a fast O(n) rank‑one update
   moves a tiny amount of mass from indices that contribute heavily to the
   current peak to “good’’ indices of low height.  The temperature follows a
   square‑root cooling schedule.  The best vector seen throughout the run is
   returned.

All heavy operations are performed with FFTs (size is a power‑of‑two),
so even for n≈18000 each iteration costs only a few milliseconds, allowing
hundreds of thousands of iterations within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied.
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n‑1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam*i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape: a_i ∝ x^{α‑1}·(1‑x)^{α‑1},
    where x = (i+0.5)/n.
    """
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * \
        (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _smooth_gradient_step(a: np.ndarray, beta: float, target_sum: float,
                         size_fft: int) -> tuple[np.ndarray, float]:
    """
    One optimisation step: compute the soft‑max gradient of max(a*a)
    with temperature beta, take a small step, and project onto the simplex.
    Returns the new vector and the exact max‑convolution value of the new vector.
    """
    n = a.size

    # ----- self‑convolution (needed for the soft‑max weights) -----
    fa = np.fft.rfft(a, n=size_fft)
    conv_full = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
    max_c = conv_full.max()

    # ----- soft‑max weights -----
    # shift by max_c for numerical stability
    exp_vals = np.exp(beta * (conv_full - max_c))
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # ----- gradient of the smooth objective -----
    # gradient = 2 * (a ⋆ w)  where ⋆ denotes cross‑correlation.
    # We obtain it via FFT: reverse w, convolve with a, then flip.
    w_rev = w[::-1]                         # also length 2n‑1
    fw = np.fft.rfft(w_rev, n=size_fft)
    conv_rev = np.fft.irfft(fa * fw, n=size_fft)[: (3 * n - 2)]
    # slice corresponding to indices n‑1 … 2n‑2 (inclusive) and reverse
    grad = 2.0 * conv_rev[n - 1:2 * n - 1][::-1]

    # ----- step size -----
    # we move a fraction of the total mass in the direction of -grad.
    # The fraction decays linearly with time (handled outside).
    grad_norm = np.linalg.norm(grad) + 1e-12
    # scale so that |step| ≈ step_factor * target_sum
    step_factor = 0.02                     # will be multiplied later outside
    # Here we only compute the direction; the caller decides the actual scale.
    direction = -grad / grad_norm

    # return direction and current exact max‑convolution value (max_c)
    return direction, max_c


def _gradient_descent_softmax(a0: np.ndarray, target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Smooth gradient descent with a temperature schedule.
    Returns the best vector seen and its exact (unsmoothed) score.
    """
    a = a0.copy()
    n = a.shape[0]

    # pre‑compute an FFT length that can hold the longest convolution (3n‑2)
    size_fft = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start

        # temperature schedule for the soft‑max (beta)
        beta0, beta_max = 2.0, 45.0
        beta = beta0 + (beta_max - beta0) * (elapsed / total)

        # obtain descent direction and current max‑convolution
        direction, cur_max = _smooth_gradient_step(a, beta, target_sum, size_fft)

        # step magnitude – linearly decreasing, plus a small floor
        step_fraction = 0.02 * (1.0 - elapsed / total) + 0.001
        step = step_fraction * target_sum * direction

        # gradient step + projection onto the simplex
        a = a + step
        a = _project_to_simplex(a, target_sum)

        # evaluate exact objective
        cur_val = _score(a)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:6d} β={beta:5.2f} score={best_val:.6f}")

        # occasional jitter to shake out of flat regions
        if it % 2500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from “bad’’ indices
    (large contribution to the current peak) to “good’’ ones (small height).
    An O(n) rank‑one update keeps the convolution up‑to‑date.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact self‑convolution
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                    # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))  # square‑root cooling

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height (inverse proportional)
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part (2·a·Δ) – donor removal, receiver addition
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        # compute new objective quickly (no full convolution needed)
        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:6d} best={best_val:.6f}")

        # full recomputation every few thousand steps to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute it={it} best={best_val:.6f}")

        it += 1

    return best_a, best_val

    return final_seq
