"""
Hybrid optimisation for the minimum peak convolution problem.

The routine builds a diverse seed pool, runs a projected Differential
Evolution (DE) phase, refines the best individual with a long projected
sub‑gradient descent, adds a short deterministic mass‑transfer polish,
and finally performs a simulated‑annealing search over permutations of
the multiset of heights (the ordering of values has a strong influence on
the convolution peak).  The whole process respects the ℓ¹‑simplex
constraint Σ a_i = √(2 n) (which makes the evaluation function equal to
the peak of the auto‑convolution).  The algorithm adapts its phase
durations to the remaining time budget and stores the best sequence in
the global ``height_sequence_1`` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (different shapes)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector and project onto the simplex."""
    # Concentration parameters drawn uniformly in a moderate range
    alpha = rng.random(n) * 4.0 + 0.2
    a = rng.dirichlet(alpha) * target_sum
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (often lower the peak)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto simplex)
# ----------------------------------------------------------------------


def _de_optimize(population, target_sum, deadline, F=0.5, CR=0.9):
    """Classic DE loop with projection onto the ℓ¹‑simplex."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_max_convolution_fft(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # select three distinct mates different from i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _max_convolution_fft(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        # reserve a couple of seconds for later phases
        if time.time() > deadline - 2.0:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0, deadline, target_sum,
                                start_temp=0.5, end_temp=1e-5, low_frac=0.07):
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        # consider the symmetric partner as an alternative donor
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        # pick a low‑contribution receiver
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = float(conv.max())

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0, deadline, target_sum,
                         step_factor=0.5, gamma=0.99):
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = float(_conv_fft(a).max())
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # tolerance to capture all max entries
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = float(_conv_fft(a).max())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation simulated annealing (reordering only)
# ----------------------------------------------------------------------


def _permute_sa(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over permutations of `seq`.
    Only the order of the values changes; the multiset of heights stays fixed.
    """
    rng = np.random.default_rng()
    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_val = _max_convolution_fft(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    start = time.time()
    T_start = max(1e-6, cur_val * 0.15)
    T_end = 1e-8

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # swap in‑place
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _max_convolution_fft(cur_seq)

        # linear cooling schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T_start * (1 - frac) + T_end * frac

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine consists of:

    1. Seed generation (power‑law, exponential, Dirichlet, random, warm‑start).
    2. Projected Differential Evolution.
    3. Long projected sub‑gradient descent.
    4. Brief deterministic mass‑transfer polishing.
    5. Permutation simulated annealing (reordering only).
    6. Trim trailing zeros and store the result in ``height_sequence_1``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a larger n gives a richer search space.
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 10000
    elif budget_s >= 600:
        n = 8000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    seeds = []

    # power‑law shapes
    for p in np.linspace(0.25, 1.2, 11):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential shapes
    for lam in np.linspace(0.0005, 0.03, 9):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # Dirichlet‑like random shapes
    for _ in range(6):
        seeds.append(_dirichlet_sequence(n, target_sum, rng))

    # pure uniform random points on the simplex
    for _ in range(8):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm start from previous run (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            seeds.append(_resize_sequence(prev, n))
        else:
            seeds.append(prev.copy())

    # jitter each seed slightly to increase diversity
    jitter_scale = target_sum * 0.015
    for seq in list(seeds):
        jitter = rng.normal(scale=jitter_scale, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # --------------------------------------------------------------
    # Choose an initial best seed (based on peak height)
    # --------------------------------------------------------------
    best_seq = min(seeds, key=lambda s: _max_convolution_fft(s))
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] Initial best peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Initialise DE population (modest size, diverse)
    # --------------------------------------------------------------
    pop_size = max(30, int(n / 260))   # ~30‑40 individuals
    population = [best_seq.copy()]
    while len(population) < pop_size:
        population.append(_project_to_simplex(_random_simplex(n, target_sum, rng), target_sum))

    # --------------------------------------------------------------
    # Differential Evolution phase (≈ 35 % of the budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.35 * budget_s
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent – long polishing phase (≈ 45 % of the budget)
    # --------------------------------------------------------------
    sg_deadline = start_time + 0.35 * budget_s + 0.45 * budget_s
    if sg_deadline > deadline - 5.0:
        sg_deadline = deadline - 5.0
    best_seq, sg_val = _subgradient_descent(best_seq, sg_deadline, target_sum,
                                            step_factor=0.55, gamma=0.995)
    if sg_val < best_score:
        best_score = sg_val
        print(f"[{time.time():.2f}] Sub‑grad descent improved – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer polishing (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        mt_deadline = time.time() + min(15.0, deadline - time.time() - 5.0)
        seq_mt, mt_score = _deterministic_mass_transfer(best_seq, mt_deadline,
                                                        target_sum,
                                                        start_temp=0.45,
                                                        end_temp=1e-6,
                                                        low_frac=0.07)
        if mt_score < best_score:
            best_seq, best_score = seq_mt, mt_score
            print(f"[{time.time():.2f}] Deterministic MT improved – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Permutation simulated annealing (reordering only)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        # start from the best of a few simple reorderings
        candidates = [
            best_seq,
            _alternating_order(best_seq),
            _even_spread_order(best_seq)
        ]
        best_seq = min(candidates, key=_max_convolution_fft)
        best_score = _max_convolution_fft(best_seq)

        # allocate remaining time to permutation SA
        perm_deadline = deadline
        seq_perm, perm_score = _permute_sa(best_seq, perm_deadline)
        if perm_score < best_score:
            best_seq, best_score = seq_perm, perm_score
            print(f"[{time.time():.2f}] Permutation SA improved – final peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final trimming of trailing near‑zero entries
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for warm‑starting the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
