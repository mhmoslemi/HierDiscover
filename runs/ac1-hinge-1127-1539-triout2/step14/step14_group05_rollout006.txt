"""
Hybrid optimizer for the step‑function benchmark (target < 1.503).

The algorithm is a streamlined version of the previous “Hybrid optimizer’’
with a few crucial changes that allow deeper exploitation of a larger
support size (≈20 000) while staying well inside the 1000 s time budget:

1. **Init (n≈5 000).**  A small pool of parametric families
   (exponential, double‑exponential, power‑law, random, cosine…) is
   generated, projected onto the ℓ¹‑simplex (sum = √(2 n)),
   and the best member (lowest max‑convolution) is kept.

2. **Soft‑max (annealed) GD** on the modest size – this quickly obtains a
   solid shape.  The temperature β is doubled every few hundred steps,
   momentum (γ≈0.98) and an ℓ²‑normalised step size keep the iterates
   stable.

3. **Upscale** to a larger support (≈20 000).  Linear interpolation is
   used, then a deterministic “low‑collision’’ permutation
   (alternating order) is applied to spread the large entries.

4. **Exact sub‑gradient descent** on the large problem.  The full
   convolution is obtained via FFT, the active set is taken as all
   indices whose value is ≥ tol·M (tol≈0.999), and a momentum update
   (γ≈0.995) with an adaptive step size drives the maximum down.

5. **Permutation annealing** – random swaps are proposed; a simple
   temperature schedule (geometric decay) accepts uphill moves with a
   Metropolis probability.  The convolution is recomputed with FFT after
   each accepted swap.

6. **Mass‑transfer simulated annealing** – the classic “donor/receiver’’
   move that reduces the product contributing to the current peak is
   applied many times with a slowly cooling temperature.

7. **Final clean‑up** – truncate trailing ≈0 entries, store the result
   for possible reuse, and return the list.

All stages respect non‑negativity and the ℓ¹‑simplex constraint, and
the routine reports intermediate max‑convolution values (which equal
the benchmark score because the sum is √(2 n)).
"""

import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (various parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _double_exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a *= target_sum / a.sum()
    return a


def _sym_exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Peak at centre, symmetric decay."""
    i = np.arange(n, dtype=np.float64)
    dist = np.abs(i - (n - 1) / 2.0)
    a = np.exp(-lam * dist)
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Reordering helpers (to scatter large values)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Place values on a step that is (approximately) coprime with n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_c = conv.max()
        # soft‑max probabilities
        exps = np.exp(beta * (conv - max_c))
        p = exps / exps.sum()

        # gradient of soft‑max loss w.r.t a
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            # small jitter to escape plateaus
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 800 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.55,
                         gamma: float = 0.995,
                         tol: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # active set (indices close to the max)
        active_mask = (conv >= M * tol).astype(np.float64)

        # gradient = 2 * (a_rev * active_mask) convolution
        a_rev = a[::-1]
        size = 1 << ((len(active_mask) + len(a_rev) - 1).bit_length())
        act_fft = np.fft.rfft(active_mask, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(act_fft * rev_fft, n=size)[: len(active_mask) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            # jitter to avoid stagnation
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Permutation annealing (random swaps with Metropolis acceptance)
# ----------------------------------------------------------------------


def _perm_annealing(a0: np.ndarray,
                    deadline: float,
                    start_temp: float = 0.3,
                    end_temp: float = 1e-5,
                    max_trials: int = 0) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on permutations."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_seq = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)  # [0,1]
        temp = start_temp * (end_temp / start_temp) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]  # propose swap

        new_score = _max_convolution_fft(a)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] PermAnneal new best={best_score:.6f}")
        else:
            # reject – swap back
            a[i], a[j] = a[j], a[i]

        if max_trials and it >= max_trials:
            break
        it = it + 1 if 'it' in locals() else 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Mass‑transfer simulated annealing (donor/receiver move)
# ----------------------------------------------------------------------


def _mass_transfer_sa(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      start_temp: float = 0.25,
                      end_temp: float = 1e-5,
                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """Classic donor‑receiver mass‑transfer with cooling."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MassTransfer new best={best_score:.6f}")
        else:
            # undo
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to keep dynamics
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns the best sequence discovered within the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1. Broad exploration on a modest size (≈5 000)
    # --------------------------------------------------------------
    n_expl = 5000
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool = []

    # parametric families
    for p in np.linspace(0.2, 2.0, 14):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0002, 0.015, 12):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
        init_pool.append(_double_exponential_sequence(n_expl, float(lam), target_expl))
        init_pool.append(_sym_exponential_sequence(n_expl, float(lam), target_expl))
    for _ in range(6):
        init_pool.append(_cosine_sequence(n_expl, target_expl))
    for _ in range(6):
        init_pool.append(_sin_half_sequence(n_expl, target_expl))
    for _ in range(12):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # if a previous run exists, reuse it as a seed
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] >= 10:
            init_pool.append(_resize_sequence(prev, n_expl))

    # add jittered copies of everything (small perturbations)
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # pick the best start from the pool
    best_seq = None
    best_score = np.inf
    for seq in init_pool:
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best (n={n_expl}) score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Annealed GD on modest size (≈5 000) – run a few short bursts
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # occasionally re‑seed from current best with a tiny jitter
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 13.0, deadline - 18.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_expl,
            max_beta=9e5, step_factor=0.7, gamma=0.98)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD‑burst new best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Upscale to a larger problem size (≈20 000) if time permits
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 250.0:
        n_big = 20000
    elif remaining > 150.0:
        n_big = 15000
    else:
        n_big = n_expl

    target_big = math.sqrt(2.0 * n_big)

    if best_seq is None:
        # fallback to a simple exponential if nothing was found
        best_seq = _exponential_sequence(n_expl, 0.001, target_expl)

    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # try a few deterministic reorderings and keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start score={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 4. Sub‑gradient descent on the large scale
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 115.0
        seq_big, sub_score = _subgradient_descent(
            seq_big, sub_deadline, target_big,
            step_factor=0.55, gamma=0.995, tol=0.999)
        if sub_score < best_score:
            best_score = sub_score
            print(f"[{time.time():.2f}] Sub‑grad descent improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Permutation annealing (random swaps)
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:
        perm_deadline = deadline - 55.0
        seq_big, perm_score = _perm_annealing(seq_big, perm_deadline,
                                              start_temp=0.3, end_temp=1e-5)
        if perm_score < best_score:
            best_score = perm_score
            print(f"[{time.time():.2f}] Permutation anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final mass‑transfer (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        mt_deadline = deadline - 25.0
        seq_big, mt_score = _mass_transfer_sa(seq_big, mt_deadline,
                                              target_big,
                                              start_temp=0.22,
                                              end_temp=1e-5,
                                              low_frac=0.05)
        if mt_score < best_score:
            best_score = mt_score
            print(f"[{time.time():.2f}] Mass‑transfer best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    # expose for possible reuse in later runs
    globals()["height_sequence_1"] = final_seq

    # Final evaluation (for logging only)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
