"""
Propose a non‑negative step‑function that minimises the maximum of its
self‑convolution.
The algorithm consists of the following stages (all respecting the global
time budget, default 1000 s):

1. **Rich parametric warm‑start** – randomised scans over many
   families (power‑law, power‑law with exponential tilt, pure exponential,
   double‑exponential, Gaussian, simple two‑level, three‑level,
   beta‑distribution, polynomial decay).  The best candidate is kept.

2. **Optional symmetrisation** – averaging with the reversed sequence.

3. **Projected Adam with soft‑max** – several restarts, β (softmax temperature)
   is increased exponentially up to 2·10⁵, then a final polish with β≈10⁸.

4. **Deterministic mass‑transfer (exact DMT2)** – greedy moves that
   reduce the current peak of the convolution.

5. **Global peak‑mass transfer** – moves a fraction of the total donor mass
   to a single receiver, analytically computed α.

6. **Pairwise optimal transfer** – a finer greedy move between a donor and a
   receiver identified by their local contribution to the peak.

7. **Final polishing** – a last Adam run with a very large β and a smaller
   step size to fine‑tune the solution.

All phases stop early when the deadline is reached.  The returned sequence
is trimmed of trailing near‑zero entries and is a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective used by the competition: 2·n·max(b)/(sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return an array of length n, already normalised)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_decay_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _double_exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n) + np.exp(-q * (n - 1 - i) / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, m_frac: float, r: float, target_sum: float) -> np.ndarray:
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)          # (m·h + (n‑m)·l) = target_sum,  h = r·l
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a


def _threelevel_sequence(n: int,
                         m1_frac: float,
                         r1: float,
                         m2_frac: float,
                         r2: float,
                         target_sum: float) -> np.ndarray:
    """Two‑step high‑medium‑low piecewise constant shape."""
    m1 = max(1, int(round(m1_frac * n)))
    m2 = max(1, int(round(m2_frac * n)))
    if m1 + m2 >= n:
        m2 = max(1, n - m1 - 1)
    denom = m1 * r1 + m2 * r2 + (n - m1 - m2)
    l = target_sum / denom
    h1 = r1 * l
    h2 = r2 * l
    a = np.full(n, l, dtype=np.float64)
    a[:m1] = h1
    a[m1:m1 + m2] = h2
    return a


def _beta_sequence(n: int,
                   alpha: float,
                   beta_param: float,
                   target_sum: float) -> np.ndarray:
    """Beta‑distribution shape on {0,…,n‑1}: a_i ∝ x^α (1‑x)^β."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.power(x, alpha) * np.power(1.0 - x, beta_param)
    # Guard against NaNs at the boundaries for α=0 or β=0
    a = np.nan_to_num(a, nan=0.0, copy=False)
    # Avoid the trivial zero‑vector
    if a.sum() == 0.0:
        a = np.ones_like(a) * 1e-12
    a *= target_sum / a.sum()
    return a


def _poly_decay_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Polynomial decay: a_i = (1‑i/(n‑1))^p."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(1.0 - i / max(1, n - 1), p)
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b) w.r.t a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient of sum_i p_i * (a * a)_i w.r.t a
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected Adam optimizer (soft‑max surrogate)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short Adam runs from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (Exact DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass transfer
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = base[i:i + 2 * n - 1]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = base[j:j + 2 * n - 1]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Global random mass transfer (exploratory)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (aggressive refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Reduce the current convolution peak by moving a fraction of the total
    donor mass to a single weak receiver.  The fraction α is chosen as the
    largest feasible value respecting all convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second largest entry – stop condition
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)   # contribution of donors alone

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]            # shifted view for receiver r
            diff = donor_mass * shift_r - conv_donors
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] / (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move

        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} α={best_alpha:.4g} new_score={best_score:.6f}")
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the peak of its
    self‑convolution.  The routine respects the provided time budget and
    returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # Problem size – a larger n gives a richer search space.
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) ⇒ score = max convolution

    # ------------------------------------------------------------------
    # Phase 0 – enlarged randomised parametric warm‑start
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # Helper to evaluate and possibly replace the current best
    def _maybe_update(seq: np.ndarray, label: str):
        nonlocal best_seq, best_score
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] {label} improved best = {best_score:.6f}")

    # 0.1) include any previously stored construction
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        _maybe_update(resized, "Prev‑known")

    # 0.2) systematic grids (coarse) – keep them – they are cheap
    for p in np.linspace(0.0, 2.0, 21):
        _maybe_update(_power_law_sequence(n, float(p), target_sum), f"PowerLaw p={p:.2f}")

    for q in np.linspace(0.0, 20.0, 11):
        _maybe_update(_exp_decay_sequence(n, float(q), target_sum), f"ExpDec q={q:.2f}")

    for q in np.linspace(0.0, 15.0, 8):
        _maybe_update(_double_exp_sequence(n, float(q), target_sum), f"DoubleExp q={q:.2f}")

    for sigma in np.linspace(0.1 * n, 0.4 * n, 6):
        _maybe_update(_gaussian_sequence(n, float(sigma), target_sum), f"Gauss σ={sigma:.0f}")

    for m_frac in np.linspace(0.01, 0.4, 5):
        for r in np.linspace(1.0, 30.0, 12):
            _maybe_update(_piecewise_sequence(n, float(m_frac), float(r), target_sum),
                          f"Piecewise m={m_frac:.2f} r={r:.1f}")

    # 0.3) random sampling of richer families
    N_samples = 500

    # a) power‑law with exponential tilt
    for _ in range(N_samples):
        p = rng.uniform(0.0, 2.5)
        q = rng.uniform(0.0, 25.0)
        seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
        _maybe_update(seq, "ExpPower")

    # b) beta‑distribution shapes
    for _ in range(N_samples):
        alpha = rng.uniform(0.0, 5.0)
        beta_p = rng.uniform(0.0, 5.0)
        seq = _beta_sequence(n, float(alpha), float(beta_p), target_sum)
        _maybe_update(seq, "BetaDist")

    # c) three‑level piecewise constant
    for _ in range(N_samples):
        m1 = rng.uniform(0.01, 0.3)
        r1 = rng.uniform(2.0, 30.0)
        m2 = rng.uniform(0.01, 0.3)
        r2 = rng.uniform(1.0, 10.0)
        seq = _threelevel_sequence(n, float(m1), float(r1), float(m2), float(r2), target_sum)
        _maybe_update(seq, "ThreeLevel")

    # d) polynomial decay
    for _ in range(N_samples):
        p = rng.uniform(0.0, 12.0)
        seq = _poly_decay_sequence(n, float(p), target_sum)
        _maybe_update(seq, "PolyDecay")

    print(f"[{time.time():.2f}] Initial best after warm‑start = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional symmetrisation
    # ------------------------------------------------------------------
    sym = (best_seq + best_seq[::-1]) * 0.5
    sym = _project_to_simplex(sym, target_sum)
    _maybe_update(sym, "Symmetrisation")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (Exact DMT2)
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # occasional random perturbation
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Global peak‑mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.025,
                                                      receiver_frac=0.025)
        if val_peak < best_score:
            best_score = val_peak
            best_seq = a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Pairwise optimal transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.015,
                                                      receiver_frac=0.015)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final polishing with a very large β (soft‑max ≈ max)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e8,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final verification (harmless if evaluate_sequence is absent)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
