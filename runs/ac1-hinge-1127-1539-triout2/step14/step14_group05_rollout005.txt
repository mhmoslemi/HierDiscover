"""
Hybrid optimizer for the step‑function benchmark (target ≤ 1.503).

The pipeline:
1. Sample a few parametric families (exponential, power‑law, random simplex)
   on a moderately large support (n≈5·10⁴).  Keep the best candidate.
2. Optional re‑ordering (alternating / even‑spread) to disperse large values.
3. Long annealed “soft‑max” gradient descent on the max‑convolution,
   with momentum, jitter and an aggressive β‑schedule.
4. Exact sub‑gradient descent (max‑convolution active set) for fine‑tuning.
5. A short simulated‑annealing mass‑transfer phase to polish the result.
6. Trim trailing zeros and return the sequence.

All updates respect the ℓ¹‑simplex constraint Σa = √(2 n) and a≥0.
The algorithm runs under the given time budget (≈1000 s) and logs progress.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a * a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (various parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) and then scaled to L1‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _exponential_ends_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """
    Symmetric exponential decreasing from both ends toward the centre:
    a_i ∝ exp(-lam * min(i, n-1-i)).
    """
    idx = np.arange(n, dtype=np.float64)
    d = np.minimum(idx, n - 1 - idx)
    a = np.exp(-lam * d)
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (to spread large values)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the sorted values alternately at the left and right ends."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Scatter the sorted values with a coprime stride."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e6,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth soft‑max approximation
    of max (a*a).  Momentum is used, β is doubled periodically.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        # FFT of a
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_c = conv.max()
        # soft‑max weights
        exps = np.exp(beta * (conv - max_c))
        p = exps / np.sum(exps)

        # gradient w.r.t. a (using reversed a for convolution)
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        # increase β gradually (more accurate soft‑max)
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.999,
                         tol: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max (a*a) using the exact active set.
    The tolerance tol selects all entries within (1‑tol) of the current maximum.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # mask of near‑max entries (soft active set)
        I = (conv >= maxb * tol).astype(np.float64)

        # gradient = 2 * (a_rev * I)  (convolution of reversed a with I)
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.25,
                                      end_temp: float = 1e-6,
                                      low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Transfer mass from a 'donor' pair to a 'receiver' pair, accepting
    uphill moves with a temperature schedule."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate the current worst convolution index
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to keep the search alive
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The function returns the best sequence discovered within the time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Choose a working size.
    #    Larger n gives more “room” for spreading mass.
    # ------------------------------------------------------------------
    # allocate most of the budget to a single large problem
    n = 40000                                           # ≈ 4·10⁴ elements
    target_sum = math.sqrt(2.0 * n)                     # L1‑norm to make score = max_conv

    print(f"[{time.time():.2f}] Using n={n}, target L1={target_sum:.4f}")

    # ------------------------------------------------------------------
    # 2. Build a modest pool of candidate families.
    # ------------------------------------------------------------------
    candidates = []

    # exponential families (fast decay)
    for lam in np.linspace(0.0004, 0.025, 18):
        candidates.append(_exponential_sequence(n, lam, target_sum))
        candidates.append(_exponential_ends_sequence(n, lam, target_sum))

    # power‑law families
    for p in np.linspace(0.3, 2.0, 14):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # sinusoidal windows
    candidates.append(_cosine_sequence(n, target_sum))
    candidates.append(_sin_half_sequence(n, target_sum))

    # random simplex seeds
    for _ in range(12):
        candidates.append(_random_simplex(n, target_sum, rng))

    # If a previous run exists, reuse it (after resizing)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        candidates.append(_resize_sequence(prev, n))

    # ------------------------------------------------------------------
    # 3. Quick evaluation of the pool; keep the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best max‑conv = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Apply a couple of deterministic reorderings to spread large values.
    # ------------------------------------------------------------------
    for reorder in (_alternating_order, _even_spread_order):
        seq2 = reorder(best_seq)
        val = _max_convolution_fft(seq2)
        if val < best_val:
            best_val = val
            best_seq = seq2.copy()
            print(f"[{time.time():.2f}] Reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Annealed soft‑max gradient descent (long phase).
    # ------------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 60.0:
        gd_deadline = time.time() + min(remain - 30.0, 400.0)
        print(f"[{time.time():.2f}] Starting annealed GD (deadline {gd_deadline:.2f})")
        a_opt, val_opt = _annealed_gradient_descent(best_seq.copy(),
                                                    gd_deadline,
                                                    target_sum,
                                                    max_beta=5e6,
                                                    step_factor=0.7,
                                                    gamma=0.99)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] After GD best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Exact sub‑gradient descent (fine‑tuning).
    # ------------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 30.0:
        sg_deadline = time.time() + min(remain - 10.0, 250.0)
        print(f"[{time.time():.2f}] Starting sub‑gradient descent (deadline {sg_deadline:.2f})")
        a_opt, val_opt = _subgradient_descent(best_seq.copy(),
                                              sg_deadline,
                                              target_sum,
                                              step_factor=0.6,
                                              gamma=0.999,
                                              tol=0.9995)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] After SubGrad best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final short mass‑transfer simulated annealing polishing.
    # ------------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 8.0:
        mt_deadline = time.time() + min(remain - 2.0, 15.0)
        print(f"[{time.time():.2f}] Starting mass‑transfer SA (deadline {mt_deadline:.2f})")
        a_opt, val_opt = _mass_transfer_simulated_annealing(best_seq.copy(),
                                                            mt_deadline,
                                                            target_sum,
                                                            start_temp=0.2,
                                                            end_temp=1e-6,
                                                            low_frac=0.04)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Trim trailing near‑zero entries.
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq   # for possible reuse in later runs

    # ------------------------------------------------------------------
    # 9. Final evaluation (just for logging – not required for the return)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {final_score:.6f}")
    except Exception:
        pass

    return final_seq
