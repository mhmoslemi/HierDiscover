"""
Improved step‐function search.

The algorithm consists of four phases:

1. **Warm‑start / deterministic families** – a few analytic shapes
   (power‑law, exponential, blended) are evaluated and the best one
   is kept as a starting point.

2. **Linear‑programming “mass‑inflation” step** – given the current
   sequence `a` we compute `M = max(a*a)`.  Solving the LP

        maximise   1ᵀ g
        subject to  a * g  ≤  M   (convolution, element‑wise)
                    g ≥ 0

   yields a vector `g` that can be added without raising the peak.
   After rescaling `g` to the required ℓ₁‑norm we mix it with `a`
   (`a ← (1‑t) a + t g`).  Repeating a few times provably never
   increases the objective and usually drives it down appreciably.

3. **Projected sub‑gradient descent** – we directly minimise the
   peak of the convolution using a sub‑gradient step and a Euclidean
   projection onto the simplex {x ≥ 0, Σx = √(2·n)}.

4. **Aggressive simulated‑annealing mass transfer** – tiny amounts of
   probability mass are moved from “bad’’ indices (large contribution
   to the current peak) to “good’’ ones (small height).  A square‑root
   cooling schedule is employed.

The whole routine respects the supplied time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution & scoring utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape factories for the warm‑start phase
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p) ."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam * i/(n‑1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)·exp(‑lam*i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


# ----------------------------------------------------------------------
# Phase 1 – LP “inflation” step
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A such that A @ g = conv(f, g)
    (full linear convolution, length 2·n‑1).
    Row k contains f[i] for all i with 0 ≤ k‑i < n.
    """
    n = f.shape[0]
    # rows = i + j  (i ranges over f, j over g)
    i = np.arange(n, dtype=np.int32)[:, None]          # (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]          # (1,n)
    rows = (i + j).ravel()                             # length n*n
    cols = np.tile(np.arange(n, dtype=np.int32), n)    # length n*n
    data = np.repeat(f, n)                             # each f[i] repeated n times
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_inflation_step(a: np.ndarray, target_sum: float,
                       t: float, time_limit: float):
    """
    Perform one LP‑based inflation + mixing step.
    Returns the new sequence (or the unchanged one if LP fails or the
    mixing would not improve the peak).
    """
    n = a.shape[0]
    M = _conv_fft(a).max()
    # build convolution matrix for the *current* a
    A = _build_conv_matrix(a)

    # linear program: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # solve with HiGHS, respecting the remaining time budget
    opts = {"presolve": True, "time_limit": time_limit}
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options=opts)

    if not res.success:
        # LP failed – simply return the untouched vector
        return a, M

    g = res.x
    g_sum = g.sum()
    if g_sum <= 0.0:
        return a, M

    # rescale g to the required ℓ₁‑norm
    g = g * (target_sum / g_sum)

    # convex combination
    a_new = (1.0 - t) * a + t * g
    a_new = _project_to_simplex(a_new, target_sum)     # keep sum exact

    new_M = _conv_fft(a_new).max()
    # accept only if the peak has not increased
    if new_M <= M:
        return a_new, new_M
    else:
        # fall back to the original vector
        return a, M


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_projected_descent(a0: np.ndarray, target_sum: float,
                                 deadline: float):
    """Sub‑gradient descent on the active convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Simulated‑annealing that relocates tiny mass chunks."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]

        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

        val = _score(warm)
        best_seq = warm.copy()
        best_score = val
        print(f"[{time.time():.2f}] Warm start score = {best_score:.6f}")

    # sweep a few analytic shapes (cheap and quite strong)
    for p in np.linspace(0.45, 0.75, 7):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} score={val:.6f}")

    for lam in np.linspace(0.5, 8.0, 7):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.3f} score={val:.6f}")

    for _ in range(8):
        p = np.random.random() * 0.8 + 0.1
        lam = np.random.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Power‑exp p={p:.3f} λ={lam:.3f} score={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP inflation & mixing (time‑boxed)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + 0.30 * (deadline - time.time())
        a_lp = best_seq.copy()
        # schedule for the mixing weight – start with 0.2 and decay
        t = 0.20
        it_lp = 0
        while time.time() < lp_deadline and it_lp < 10:
            # allocate a small slice of the remaining LP budget to the solver
            lp_time = max(0.1, (deadline - time.time()) * 0.20)
            a_lp, cur_M = _lp_inflation_step(a_lp, target_sum, t, lp_time)
            cur_val = _score(a_lp)
            if cur_val < best_score:
                best_score, best_seq = cur_val, a_lp.copy()
                print(f"[{time.time():.2f}] LP it={it_lp:2d} score={cur_val:.6f}")
            # slowly reduce the mixing factor to stabilise
            t = max(0.05, t * 0.85)
            it_lp += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (remaining budget ≈ 45 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.45 * (deadline - time.time())
        a_sub, val_sub = subgradient_projected_descent(best_seq, target_sum,
                                                       sub_deadline)
        if val_sub < best_score:
            best_score, best_seq = val_sub, a_sub.copy()
            print(f"[{time.time():.2f}] After sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive annealing (use up what is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score, best_seq = val_ann, a_ann.copy()
            print(f"[{time.time():.2f}] After anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # drop trailing zeros (they do not affect the score)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final check with the provided benchmark function
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start on the next call
    globals()["height_sequence_1"] = final_seq
    return final_seq
