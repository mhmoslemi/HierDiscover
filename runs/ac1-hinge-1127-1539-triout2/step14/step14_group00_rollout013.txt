def _local_random_search(a0, target_sum, deadline):
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    n = len(a)

import time

import scipy.linalg
import scipy.optimize

def _project_to_simplex(v, s):
    # Euclidean projection onto simplex {x >=0, sum = s}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv

def _max_convolution_fft(a):
    return float(_conv_self(a).max())

def _score(a):
    n = a.shape[0]
    maxb = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * maxb / (s * s))

    new_x = np.linspace(0.0, 1.0, new_len)

    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def _smooth_gradient_descent(a0, target_sum, deadline):
    """
    Continuously increase the smoothness parameter alpha.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    # schedule of alphas – start low, finish high
    alphas = [2,5,10,20,40,80,160,320]
    step_factor = 0.6
    for alpha in alphas:
        # run a block of iterations for this alpha
        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()
            w = np.exp(alpha * (conv - M))
            Z = w.sum()
            if Z == 0.0:
                w[:] = 1.0
                Z = w.size
            a_rev = a[::-1]
            conv_w_rev = _fft_convolve(w, a_rev)
            Tg = conv_w_rev[len(a)-1:len(a)-1+len(a)]
            grad = a + Tg / Z
            norm = np.linalg.norm(grad)
            if norm == 0.0:
                break
            lr = step_factor * target_sum / (norm + 1e-12)
            a_new = a - lr * grad
            a_new = _project_to_simplex(a_new, target_sum)
            new_score = _score(a_new)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                step_factor = min(step_factor * 1.03, 2.0)
                a = a_new
            else:
                step_factor = max(step_factor * 0.7, 1e-4)
                a = a_new
            # early‑exit the block if progress stalls
            if time.time() > deadline - 0.5:
                break
        if time.time() > deadline - 0.5:
            break
    return best_a, best_score

def _deterministic_mass_transfer(a0, target_sum, deadline):
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = 0.4 * (1e-5 / 0.4) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0]-1))
        hi = min(kmax, a.shape[0]-1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * 0.07))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def _lp_refine(f, target_sum, deadline, t=0.1, max_iter=200):
    """
    Iterative LP refinement (small dimension).
    """
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iter:
        M = _max_convolution_fft(f)
        # Build Toeplitz matrix representing convolution f * g
        col = np.concatenate([f, np.zeros(n-1)])
        row = np.concatenate([f[:1], np.zeros(n-1)])
        A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
        b = np.full(2*n-1, M)
        c = -np.ones(n)  # maximize sum(g)
        bounds = [(0, None)] * n
        # linprog (highs) expects dense or sparse; using dense is fine for modest n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 * (target_sum / sum_g0)
        f_new = (1 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)
        new_score = _score(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        f = f_new
        iter_cnt += 1
    return best_f, best_score

"""
Hybrid optimizer for the max‑convolution step‑function problem.

The pipeline consists of

1. **Small‑scale LP refinement** – work on a modest length (≈200‑300)
   and repeatedly solve
   `max Σ g_i  s.t.  g_i ≥ 0 ,  f * g ≤ M`,
   where `M = max(f*f)`.  The LP solution is mixed with the current
   vector (`f←(1‑t)f+t·g`) and projected back onto the ℓ¹‑simplex.
   Even a few dozen LP steps dramatically reshape the profile.

2. **Upsampling** – the refined low‑dimensional shape is interpolated to
   the final length (4000‑8000) and projected again so that the ℓ¹‑norm
   equals √(2 n).

3. **High‑α smoothed gradient descent** – minimise the smooth surrogate
   `log‑sum‑exp(α·(a*a))` via projected gradient descent while
   progressively raising `α` (2 → 320).  This converges to a true
   max‑convolution minimiser.

4. **Deterministic mass‑transfer polishing** – a greedy “donor‑receiver”
   move that directly attacks the current peak of the convolution.

5. **Reordering trials** – try a few simple permutations
   (alternating, even‑spread, reversal) and keep the best.

All stages respect a total time budget (default 1000 s).  The final
sequence is stored in the global variable `height_sequence_1` for possible
warm‑starts in later runs.
"""

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_self(a).max())


def _score(a: np.ndarray) -> float:
    """Objective `2·n·max(a*a) / (Σa)²`.  `Σa` is expected to be √(2n)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

            a_rev = a[::-1]

            Tg = conv_w_rev[len(a) - 1 : len(a) - 1 + len(a)]

        if time.time() > deadline - 0.5:

def _lp_refine(f: np.ndarray,
               target_sum: float,
               deadline: float,
               t: float = 0.12,
               max_iter: int = 200) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve:
        maximise Σ g_i
        subject to g_i ≥ 0 , (f * g)_k ≤ M  ∀k,
    where M = max(f*f).  The solution g is mixed with f via
    `f ← (1‑t)·f + t·g` and projected back onto the ℓ¹‑simplex.
    """
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    it = 0

    while time.time() < deadline and it < max_iter:
        M = _max_convolution_fft(f)

        # Build the Toeplitz matrix that implements convolution f * g.
        col = np.concatenate([f, np.zeros(n - 1)])
        row = np.concatenate([f[:1], np.zeros(n - 1)])
        A = scipy.linalg.toeplitz(col, row)          # shape (2n‑1, n)
        b = np.full(2 * n - 1, M)

        # linprog (highs) - maximise Σg ⇒ minimise -Σg
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                    bounds=bounds, method='highs')
        if not res.success:
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # renormalise to the required ℓ¹‑norm
        g = g0 * (target_sum / sum_g0)

        # convex combination and projection
        f_new = (1 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)

        new_score = _score(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()

        f = f_new
        it += 1

    return best_f, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low max‑convolution ratio.

    Stages (respecting the time budget):
    1. LP refinement on a modest size (≈200‑300).
    2. Upsample to the final dimension (≈4000‑8000) and project.
    3. High‑α smoothed gradient descent.
    4. Deterministic mass‑transfer polishing.
    5. Simple reorderings + trimming.
    """
    start = time.time()
    deadline = start + budget_s - 0.5               # safety margin

    # ------------------------------------------------------------------
    # 1. Small‑scale LP refinement (dimension m)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        m = 300
    elif budget_s >= 500:
        m = 250
    else:
        m = 200
    target_sum_m = np.sqrt(2.0 * m)
    rng = np.random.default_rng()

    # a simple power‑law start, jittered a bit
    p0 = rng.uniform(0.2, 1.2)
    f = _power_law_sequence(m, p0, target_sum_m)
    jitter = rng.normal(scale=target_sum_m * 0.01, size=m)
    f = np.maximum(f + jitter, 0.0)
    f = _project_to_simplex(f, target_sum_m)

    lp_deadline = min(deadline - 40.0,
                      time.time() + max(10.0, 0.15 * (deadline - time.time())))

    f, score_lp = _lp_refine(f, target_sum_m, lp_deadline,
                             t=0.12, max_iter=180)
    print(f"[{time.time():.2f}] LP refinement score = {score_lp:.6f}")

    # ------------------------------------------------------------------
    # 2. Upsample to final length n
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum_n = np.sqrt(2.0 * n)

    a = _resize_sequence(f, n)
    a = _project_to_simplex(a, target_sum_n)

    # ------------------------------------------------------------------
    # 3. High‑α smoothed gradient descent
    # ------------------------------------------------------------------
    gd_deadline = min(deadline - 20.0,
                      time.time() + max(5.0, 0.2 * (deadline - time.time())))
    a, score_gd = _smooth_gradient_descent(a, target_sum_n, gd_deadline)
    print(f"[{time.time():.2f}] Smooth GD score = {score_gd:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = deadline - 5.0
        a, score_mt = _deterministic_mass_transfer(a,
                                                   target_sum_n,
                                                   mt_deadline)
        print(f"[{time.time():.2f}] Mass‑transfer score = {score_mt:.6f}")

    # ------------------------------------------------------------------
    # 5. Reordering attempts
    # ------------------------------------------------------------------
    candidates = [
        a,
        _alternating_order(a),
        _even_spread_order(a),
        a[::-1],
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final trimming
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store for possible warm‑starts
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
