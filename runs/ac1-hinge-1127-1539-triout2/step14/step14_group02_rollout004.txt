def adam_descent(a0, target_sum, deadline, beta_start=0.1, beta_max=200.0):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = beta_start
    step_size = 0.1 * target_sum # raw step

    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    while time.time() < deadline:
        b = _conv_self_fft(a)
        # compute smooth max weight
        max_b = b.max()
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w /= w.sum()
        # gradient
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)
        g = 2.0 * conv_wa[n-1:n-1+n]  # shape n

        # Adam update
        t += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        # step size epsilon decays with sqrt iteration maybe
        eta = step_size / (np.sqrt(v_hat) + 1e-8)

        a = a - eta * m_hat
        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        # track best
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max / (target_sum**2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam β={beta:.1f} iter={t} best={best_val:.6f}")

        # schedule beta: increase gradually
        if t % 2000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)
    return best_a, best_val

"""
Improved search for a low‑peak self‑convolution step function.

The algorithm consists of three stages:
1.  Build a few diverse initial vectors on the simplex Σa = √(2n).
2.  Refine each vector with a *smooth‑max* projected gradient descent.
    The smooth max is a log‑sum‑exp of the convolution entries; the
    gradient is obtained by a single extra FFT.  We use a back‑tracking
    line search and a moderately aggressive learning‑rate schedule.
    The temperature β is increased from a small value up to a large
    value (≈200) so the optimisation focuses on the true maximum.
3.  Finish with an O(n) simulated‑annealing “mass‑transfer’’ phase
    similar to the reference implementation but with a slightly
    different cooling schedule.

All helpers are defined at module level; the entry point
`propose_candidate` respects the time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact evaluation of the benchmark: 2·n·max(b) / (target_sum²)."""
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent with line‑search
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 200.0,
                       step_factor: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a*a) by projected gradient descent.
    Uses a back‑tracking line search and slowly increases the softmax
    temperature β.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    it = 0
    # Initial raw step size (a fraction of the total mass)
    raw_step = target_sum * 0.4

    while time.time() < deadline:
        # ----- 1) current self‑convolution and smooth‑max weights -----
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.1f} it={it:5d} best={best_val:.6f}")

        # softmax over convolution entries
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- 2) gradient of the smooth max -----
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)                 # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]           # central slice, shape n

        # normalise gradient direction
        norm_g = np.linalg.norm(g) + 1e-12
        direction = g / norm_g

        # ----- 3) back‑tracking line search -----
        # step size decays like O(1/√it) and is also scaled by raw_step
        step = raw_step / np.sqrt(it + 1.0)

        # try a few reductions if the step does not improve the max
        for _ in range(12):
            a_candidate = a - step * direction
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            b_candidate = _conv_self_fft(a_candidate)
            max_candidate = b_candidate.max()
            if max_candidate <= max_b:
                break
            step *= 0.5

        a = a_candidate

        # ----- 4) increase β gradually -----
        if it % 2500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.6)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing (O(n) per move).  Moves tiny
    amounts of mass from indices that contribute heavily to the current
    peak towards indices with low magnitude.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                     # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # square‑root cooling

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donors – proportional to contribution a[i]·a[k_max‑i]
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receivers – prefer small entries
        inv = 1.0 / (a + 1e-12)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.07, target_sum * 0.009)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance (based on current energy)
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to generate a power‑law candidate
# ----------------------------------------------------------------------


def _power_law_candidate(n: int, target_sum: float, exponent: float) -> np.ndarray:
    """Return a vector ∝ (i+1)^{-exponent} projected onto the simplex."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-exponent)
    seq = _project_to_simplex(seq, target_sum)
    return seq

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)              # normalisation used by the evaluator

    # --------------------------------------------------------------
    # Build initial candidates (warm start + random + power‑law)
    # --------------------------------------------------------------
    init_candidates = []

    # Warm start, if available
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()

    # Random simplex points
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    # Power‑law seeds with random exponent
    for _ in range(2):
        exp = rng.uniform(0.3, 1.2)
        seq = _power_law_candidate(n, target_sum, exp)
        init_candidates.append(seq)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optimisation loop over candidates
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start candidate #{idx + 1} -----")

        # ---- Phase 1 : smooth‑max projected gradient descent --------------
        remaining = deadline - time.time()
        phase1_deadline = time.time() + 0.38 * remaining
        a_smooth, val_smooth = smooth_max_descent(
            init, target_sum, phase1_deadline,
            beta_start=0.05, beta_max=250.0, step_factor=0.55
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ---- Phase 2 : aggressive annealing -------------------------------
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(
                a_smooth, target_sum, deadline
            )
            if val_ann < best_score:
                best_score = val_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and evaluation
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – decreasing power‑law
        best_seq = _power_law_candidate(n, target_sum, 0.7)

    arr = np.array(best_seq, dtype=np.float64)

    # Trim trailing near‑zero entries for a slightly shorter representation
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final evaluation (outside of optimisation loops)
    try:
        final_val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                         # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm‑start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
