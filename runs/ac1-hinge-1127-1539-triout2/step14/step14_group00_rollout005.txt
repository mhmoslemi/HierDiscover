"""
Hybrid evolutionary + momentum‐smooth optimisation for the step‑function
max‑convolution problem.

The optimiser works in several stages within the given time budget:

1. **Seed generation** – power‑law, exponential, random simplex points and,
   if available, the previously best construction.  All seeds are projected
   onto the ℓ¹‑simplex Σa = √(2·n).

2. **Projected Differential Evolution (DE)** – a modest population evolves
   with classic DE mutation / binomial crossover while staying feasible.
   The best individual from this phase is used as the starting point for
   the next stage.

3. **Momentum‑smooth descent** – minimises a log‑sum‑exp surrogate of the
   max‑convolution using a heavy‑ball (momentum) update.  The smoothing
   parameter α is increased gradually (2 → 4 → 8 → 12) so that the method
   first sees a smooth landscape and then focuses on the true maximum.

4. **Deterministic mass‑transfer polishing** – a greedy move of mass from
   a “donor” index (contributing most to the current peak) to a “receiver”
   index, with a Metropolis acceptance schedule.

5. **Short sub‑gradient descent** – a final fine‑tuning using the exact
   max‑convolution.

6. **Reordering** – a few simple permutations (original, alternating,
   even‑spread, reverse) are tried and the best ordering is kept.

7. **Optional up‑sampling** – if time permits the best sequence is
   interpolated to a larger length and a brief momentum‑smooth run is
   performed.

All stages respect the overall time budget (default 1000 s).  The final
sequence is stored in the global name ``height_sequence_1`` and returned
as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective `2·n·max(a*a) / (Σa)²`.  When Σa = √(2n) this equals `max(a*a)`.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after every step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # distinct mates a,b,c ≠ i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # safety margin
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Momentum‑smooth descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _momentum_smooth_optimize(a0: np.ndarray,
                              deadline: float,
                              target_sum: float,
                              alphas: list[float],
                              init_step_factor: float = 0.7,
                              beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate of max‑convolution using a
    heavy‑ball (momentum) update.  The smoothing parameter α is increased
    over time (list ``alphas``).  The step factor adapts based on whether
    an improvement was observed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)                 # momentum buffer
    step_factor = init_step_factor
    n = a.shape[0]

    start = time.time()
    # divide remaining time equally among the alphas
    total_alpha_time = deadline - start
    if total_alpha_time <= 0:
        return best_a, best_score
    alpha_times = [total_alpha_time / len(alphas)] * len(alphas)
    alpha_deadlines = np.cumsum(alpha_times) + start
    alpha_idx = 0
    current_alpha = alphas[0]

    while time.time() < deadline:
        # switch α when we cross its sub‑deadline
        if alpha_idx + 1 < len(alphas) and time.time() >= alpha_deadlines[alpha_idx]:
            alpha_idx += 1
            current_alpha = alphas[alpha_idx]

        # compute full self‑convolution
        conv = _conv_self(a)
        M = conv.max()

        # log‑sum‑exp weights
        w = np.exp(current_alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        # gradient of the smooth max
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)
        Tg = conv_w_rev[n - 1:n - 1 + n] / Z

        grad = a + Tg                     # same surrogate gradient as before

        # momentum update
        m = beta * m + (1.0 - beta) * grad
        norm = np.linalg.norm(m)
        if norm == 0.0:
            break

        lr = step_factor * target_sum / (norm + 1e-12)
        a_new = a - lr * m
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            # become bolder after improvement
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            # become more cautious
            step_factor = max(step_factor * 0.70, 1e-5)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (short phase)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function max‑convolution problem.
    Returns a list of non‑negative heights whose evaluation score is as low
    as possible (subject to the 1000 s time budget).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.4          # small safety margin

    # --------------------- problem size ---------------------
    # Use a larger dimension when we have enough time.
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------- seed generation -------------------
    seeds: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.15, 1.2, 12):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 10):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random Dirichlet‑like points
    for _ in range(15):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little to increase diversity
    jittered: list[np.ndarray] = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # --------------------- initialise DE population ----------
    pop_size = max(35, n // 200)          # ~35‑45 individuals
    population: list[np.ndarray] = []

    # start with the best seed we have
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # --------------------- DE phase -------------------------
    de_deadline = min(start_time + 0.55 * budget_s,
                      deadline - 35.0)          # leave time for polishing
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished; best score = {best_score:.6f}")

    # --------------------- Momentum‑smooth descent -------
    if time.time() < deadline - 30.0:
        smooth_deadline = time.time() + min(30.0, deadline - time.time() - 25.0)
        alphas = [2.0, 4.0, 8.0, 12.0]   # increasingly sharp surrogate
        seq_s, score_s = _momentum_smooth_optimize(best_seq,
                                                   smooth_deadline,
                                                   target_sum,
                                                   alphas,
                                                   init_step_factor=0.7,
                                                   beta=0.9)
        if score_s < best_score:
            best_score, best_seq = score_s, seq_s.copy()
            print(f"[{time.time():.2f}] After momentum‑smooth best = {best_score:.6f}")

    # --------------------- deterministic mass‑transfer ------
    if time.time() < deadline - 15.0:
        mt_dead = time.time() + min(12.0, deadline - time.time() - 13.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.45, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # --------------------- short sub‑gradient descent -------
    if time.time() < deadline - 8.0:
        sg_dead = time.time() + min(8.0, deadline - time.time() - 5.0)
        seq_sg, score_sg = _subgradient_descent(best_seq,
                                                sg_dead,
                                                target_sum,
                                                step_factor=0.48,
                                                gamma=0.995)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # --------------------- reordering attempts ---------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1]
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # --------------------- optional up‑sampling & polish -----
    if time.time() < deadline - 4.0 and n < 12000:
        up_len = min(12000, n * 2)
        up_seq = _resize_sequence(np.array(best_seq), up_len)
        up_seq = _project_to_simplex(up_seq, np.sqrt(2.0 * up_len))
        up_dead = time.time() + 3.5
        # a brief smooth run on the larger vector (moderate α)
        up_seq, up_score = _momentum_smooth_optimize(up_seq,
                                                     up_dead,
                                                     np.sqrt(2.0 * up_len),
                                                     alphas=[4.0, 8.0],
                                                     init_step_factor=0.6,
                                                     beta=0.9)
        if up_score < best_score:
            best_seq = up_seq
            best_score = up_score
            print(f"[{time.time():.2f}] Upsampled polishing improved to {best_score:.6f}")

    # --------------------- final trimming --------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------- store & return --------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
