import time

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    return conv

def _project_to_simplex(v, s):
    # Euclidean projection onto simplex of sum s
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n, p, target_sum):
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n, lam, target_sum):
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _cosine_sequence(n, target_sum):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a

def _sin_half_sequence(n, target_sum):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a

def _random_simplex(n, target_sum, rng):
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq):
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out

def _even_spread_order(seq):
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _annealed_gradient_descent(a0, deadline, target_sum,
                               max_beta=5e5, step_factor=0.8, gamma=0.98):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_b = np.max(b)
        # soft-max approximation of max
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it} β={beta:.1f} best={best_val:.6f}")
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1
    return best_a, best_val

def _subgradient_descent(a0, deadline, target_sum,
                         step_factor=0.55, gamma=0.998, tol=0.995):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)  # indicator for near-max entries
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        # occasional jitter to revive search
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} best={best_val:.6f}")
        it += 1
    return best_a, best_val

def _lp_refine(a0, target_sum, max_iters, deadline,
               K=12, mix_factor=0.12):
    try:
        from scipy.optimize import linprog
    except Exception:
        print("Scipy not available; skipping LP.")
        return a0, _max_convolution_fft(a0)
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a)
        M = conv.max()
        if K >= len(conv):
            active = np.arange(len(conv), dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]
        # Build matrix A (len(active) x n)
        A = np.zeros((len(active), n), dtype=np.float64)
        for row_idx, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end+1, dtype=int)
            # coefficient of g_j is a_{k-j}
            A[row_idx, idx] = a[k - idx]
        # Solve linear program: maximise sum(g) subject to A g <= M, g>=0
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n, method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed: {res.message}")
            break
        g = res.x
        # Scale to exact target sum
        scaling = target_sum / g.sum() if g.sum() > 0 else 0.0
        if scaling != 0.0:
            g *= scaling
        # Convex combination
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} best={best_val:.6f}")
    return best_a, best_val

def _permutation_search(base_seq, trials, deadline):
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    n = len(base_seq)
    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val

def _local_permutation_hillclimb(seq, deadline):
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hillclimb attempts={attempts} best={best_val:.6f}")
    return a, best_val

def _mass_transfer_simulated_annealing(a0, deadline, target_sum,
                                      start_temp=0.3, end_temp=1e-5, low_frac=0.07):
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau
        # locate maximal convolution location
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # product contributes for each index i in idx: a[i] * a[kmax-i]
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)
        if donor == receiver:
            continue
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)
        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)
    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Hybrid search for a step function minimizing the evaluation metric.
    Uses a multi‑stage pipeline: small‑scale random+gradient search,
    LP refinement, up‑scaling, long sub‑gradient descent, permutation
    optimization and a final mass‑transfer simulated‑annealing pass.
    The sum of coefficients is kept equal to sqrt(2·n) so the score
    equals the maximum autocorrelation value.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin
    rng = np.random.default_rng()
    # --------------------------------------------------------------
    # Phase 1: modest‑size exploratory search
    # --------------------------------------------------------------
    n_small = 3000
    target_small = math.sqrt(2.0 * n_small)
    init_candidates = []
    # parametric families
    for p in np.linspace(0.2, 2.0, 10):
        init_candidates.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.0005, 0.02, 8):
        init_candidates.append(_exponential_sequence(n_small, float(lam), target_small))
    for _ in range(6):
        init_candidates.append(_cosine_sequence(n_small, target_small))
    for _ in range(6):
        init_candidates.append(_sin_half_sequence(n_small, target_small))
    for _ in range(10):
        init_candidates.append(_random_simplex(n_small, target_small, rng))
    # optionally include prior global solution
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            init_candidates.append(_resize_sequence(prev, n_small))
    # jittered variants to diversify
    for seq in list(init_candidates):
        jitter = rng.normal(scale=target_small * 0.03, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        init_candidates.append(seq_j)
    best_seq = None
    best_score = np.inf
    # run short annealed GD runs until ~30 s left
    while time.time() < deadline - 35.0:
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_small * 0.004, size=n_small)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(init_candidates)
        run_deadline = min(time.time() + 12.0, deadline - 20.0)

"""
Hybrid multi‑stage optimizer for the step‑function benchmark.

The pipeline:
1) Small‑scale random+annealed‑gradient search to obtain a decent
   shape (n≈3 k, sum=√(2n) so the score equals the max autocorrelation).
2) Short linear‑program refinement on the modest size.
3) Upscaling (linear interpolation) to a larger support (n up to ~5·10⁴)
   and a deterministic re‑ordering that spreads large entries.
4) Long sub‑gradient descent on the large problem (exact max‑convolution
   sub‑gradient, momentum, adaptive step size).
5) A few additional LP refinement passes on the active constraints.
6) Random permutation search + local hill‑climb.
7) Final greedy mass‑transfer simulated‑annealing polishing.
All updates keep the vector on the ℓ¹‑simplex (sum = √(2·n)) and
respect non‑negativity.  The routine returns the best sequence found
within the allotted time budget.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (various parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (to scatter large values)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent (used for the quick phase)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.85,
                               gamma: float = 0.97) -> tuple[np.ndarray, float]:
    """Momentum projected GD on a soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_b = np.max(b)

        # soft‑max approximation of the max entry
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]

        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it} β={beta:.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient descent (heavy stage)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.998,
                         tol: float = 0.996) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program refinement (active‑set LP)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               max_iters: int,
               deadline: float,
               K: int = 12,
               mix_factor: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Small LP that, given the current a, finds a g with large ℓ¹‑norm
    such that a * g ≤ M on the K most critical convolution positions.
    New a = (1‑t)·a + t·g.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        print("SciPy not available; skipping LP refinement.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        # active set of K largest entries
        if K >= conv.size:
            active = np.arange(conv.size, dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # build A_ub * g ≤ M
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            A[row, idx] = a[k - idx]          # coefficient a_{k‑j}

        # maximise sum(g)  <=> minimise -sum(g)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.2f}] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        if g.sum() > 0:
            g = g * (target_sum / g.sum())

        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation search
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer (simulated‑annealing) polishing
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.25,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to keep the dynamics alive
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns the best sequence discovered within the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 1 – modest‑size exploration (≈3 k entries)
    # --------------------------------------------------------------
    n_small = 3000
    target_small = math.sqrt(2.0 * n_small)

    init_pool = []

    # parametric families
    for p in np.linspace(0.2, 2.0, 10):
        init_pool.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.0005, 0.02, 8):
        init_pool.append(_exponential_sequence(n_small, float(lam), target_small))
    for _ in range(6):
        init_pool.append(_cosine_sequence(n_small, target_small))
    for _ in range(6):
        init_pool.append(_sin_half_sequence(n_small, target_small))
    for _ in range(10):
        init_pool.append(_random_simplex(n_small, target_small, rng))

    # reuse previous global solution if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            init_pool.append(_resize_sequence(prev, n_small))

    # jittered variants to enrich the pool
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_small * 0.03, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # limited budget for this phase (max 150 s or until 350 s are left)
    phase1_end = min(deadline - 350.0, start_time + 150.0)
    while time.time() < phase1_end:
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_small * 0.004, size=n_small)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 12.0, deadline - 340.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_small,
            max_beta=8e5, step_factor=0.85, gamma=0.97)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Phase 1 new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – LP refinement on the modest size
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 340.0:
        lp_dead = time.time() + min(15.0, deadline - time.time() - 320.0)
        best_seq, lp_val = _lp_refine(best_seq, target_small,
                                      max_iters=25,
                                      deadline=lp_dead,
                                      K=12,
                                      mix_factor=0.12)
        if lp_val < best_score:
            best_score = lp_val
            print(f"[{time.time():.2f}] After LP best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Upscale to a larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 650.0:
        n_big = 50000
    elif remaining > 500.0:
        n_big = 40000
    elif remaining > 350.0:
        n_big = 30000
    elif remaining > 250.0:
        n_big = 20000
    else:
        n_big = n_small

    target_big = math.sqrt(2.0 * n_big)

    if best_seq is None:
        # fallback generic start if Phase 1 produced nothing
        best_seq = _power_law_sequence(n_small, 0.7, target_small)

    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # deterministic spread ordering to avoid clustering of large values
    candidates_big = [seq_big,
                      _alternating_order(seq_big),
                      _even_spread_order(seq_big)]
    seq_big = min(candidates_big, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Long sub‑gradient descent on the large scale
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # keep about 80 s for the final polishing stages
    sub_time = max(30.0, remaining - 80.0)
    sub_dead = time.time() + sub_time

    seq_big, sub_val = _subgradient_descent(seq_big, sub_dead,
                                            target_big,
                                            step_factor=0.6,
                                            gamma=0.998,
                                            tol=0.996)
    if sub_val < best_score:
        best_score = sub_val
        print(f"[{time.time():.2f}] Sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Short LP refinement passes on the big problem
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        lp2_dead = time.time() + min(15.0, remaining - 15.0)
        seq_big, lp2_val = _lp_refine(seq_big, target_big,
                                      max_iters=15,
                                      deadline=lp2_dead,
                                      K=16,
                                      mix_factor=0.10)
        if lp2_val < best_score:
            best_score = lp2_val
            print(f"[{time.time():.2f}] LP2 best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – Random permutation search
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        perm_dead = time.time() + min(8.0, remaining - 12.0)
        seq_big, _ = _permutation_search(seq_big, trials=150, deadline=perm_dead)
        print(f"[{time.time():.2f}] After random perm max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 7 – Local permutation hill‑climb
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        hill_dead = time.time() + min(7.0, remaining - 5.0)
        seq_big, _ = _local_permutation_hillclimb(seq_big, hill_dead)
        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 8 – Final mass‑transfer simulated‑annealing polishing
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        mt_dead = time.time() + min(6.0, remaining - 2.0)
        seq_big, mt_val = _mass_transfer_simulated_annealing(seq_big,
                                                             mt_dead,
                                                             target_big,
                                                             start_temp=0.25,
                                                             end_temp=1e-5,
                                                             low_frac=0.06)
        if mt_val < best_score:
            best_score = mt_val
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 9 – Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # optional final evaluation for logging
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
