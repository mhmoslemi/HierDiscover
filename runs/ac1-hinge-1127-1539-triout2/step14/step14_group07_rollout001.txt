"""
Propose a step‑function minimizing the self‑convolution max‑norm.

Algorithm
---------
* Choose a problem size n (≈12‑15k) – larger n gives finer granularity.
* Initialise the sequence by sweeping several simple parametric families
  (power‑law, exponential‑tilted, Gaussian, arcsine) and by importing the
  previous best construction if available.
* Perform a few rounds of projected Adam optimisation with a soft‑max
  approximation of the max‑norm (large β gradually).
* Run a simulated‑annealing style random‑local‑search that repeatedly moves
  a small amount of mass from a “donor’’ (large contribution to the current
  peak of the convolution) to a “receiver’’ (small coefficient).  The move
  is accepted either if it improves the score or with a temperature‑dependent
  probability – this allows escaping shallow local minima.
* Finish with a high‑β Adam polishing pass.
* Trim trailing near‑zero entries and return the list.

The code respects the 1000 s time budget, prints intermediate scores and
does not rely on any external files or network access.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(b) / (sum(a)^2)  (exactly what evaluate_sequence returns
    after the clipping steps).  It is cheap – we compute the convolution via FFT.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    max_b = _conv_fft(a).max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Euclidean projection on the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projects v onto {x ≥ 0, Σx = s} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Simple parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _arcsine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Discrete analogue of the arcsine density on [0,1]."""
    i = np.arange(n, dtype=np.float64)
    # avoid zero denominator – shift by 0.5
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
#  Soft‑max gradient used by Adam (the same as the previous code)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smoothed max (soft‑max) of the self‑convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # compute gradient = 2 * (a_rev * p) restricted to the original length
    a_rev = a[::-1]
    Ar_fft = rfft(a_rev, n=size)
    Pr_fft = rfft(p, n=size)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
#  Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run a few Adam‑GD restarts starting from jittered versions of *init*."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
#  Simulated annealing – mass‑transfer moves
# ----------------------------------------------------------------------


def _annealing_move(a: np.ndarray,
                    conv: np.ndarray,
                    target_sum: float,
                    rng: np.random.Generator) -> np.ndarray:
    """
    Propose a new sequence by moving a small amount of mass from a donor
    (large contribution to the current peak) to a receiver (small coefficient).
    """
    n = a.shape[0]
    k_max = int(np.argmax(conv))
    max_val = conv[k_max]

    # indices that participate in the peak at k_max
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx_range = np.arange(lo, hi + 1, dtype=np.int64)

    # partner values a_{k_max-i}
    partner = a[k_max - idx_range]

    # contribution of each index to the peak
    contrib = a[idx_range] * partner
    # donor = index with maximal contribution (break ties randomly)
    donor_candidates = idx_range[contrib == contrib.max()]
    donor = rng.choice(donor_candidates)

    # receiver = an index with relatively small coefficient (among the 2 % smallest)
    low_frac = max(1, int(0.02 * n))
    low_idxs = np.argsort(a)[:low_frac]
    # avoid picking the donor
    low_idxs = low_idxs[low_idxs != donor]
    if low_idxs.size == 0:
        # fallback to a random different index
        receiver = rng.integers(n - 1)
        while receiver == donor:
            receiver = rng.integers(n - 1)
    else:
        receiver = rng.choice(low_idxs)

    # amount to move – a modest fraction of the donor mass
    max_delta = a[donor] * 0.5
    if max_delta <= 0.0:
        return a.copy()
    delta = rng.random() * max_delta

    a_new = a.copy()
    a_new[donor] -= delta
    a_new[receiver] += delta
    # the sum is unchanged, but projection removes possible tiny negatives
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new


def _simulated_annealing(start_seq: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple SA loop that repeatedly proposes mass‑transfer moves.
    Temperature decays linearly to zero.
    """
    rng = np.random.default_rng()
    cur_a = start_seq.copy()
    cur_score = _score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature – a fraction of the current excess over 1.0
    T0 = max(1e-3, (cur_score - 1.0) * 0.4)

    start_time = time.time()
    while time.time() < deadline:
        # linear cooling
        t = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1.0 - t)

        # compute current convolution (needed for donor selection)
        conv = _conv_fft(cur_a)

        # propose move
        a_prop = _annealing_move(cur_a, conv, target_sum, rng)
        prop_score = _score(a_prop)

        # acceptance test
        if prop_score < cur_score or rng.random() < np.exp(-(prop_score - cur_score) / max(T, 1e-12)):
            cur_a, cur_score = a_prop, prop_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                print(f"[{time.time():.2f}] SA new best {best_score:.6f}")

        # occasional pure jitter move to inject diversity
        if rng.random() < 0.003:
            jitter = rng.normal(scale=target_sum * 0.001, size=cur_a.shape)
            a_jit = cur_a + jitter
            a_jit = np.maximum(a_jit, 0.0)
            a_jit = _project_to_simplex(a_jit, target_sum)
            jitter_score = _score(a_jit)
            if jitter_score < cur_score or rng.random() < np.exp(-(jitter_score - cur_score) / max(T, 1e-12)):
                cur_a, cur_score = a_jit, jitter_score
                if cur_score < best_score:
                    best_score, best_a = cur_score, cur_a.copy()
                    print(f"[{time.time():.2f}] SA jitter new best {best_score:.6f}")

    return best_a, best_score

def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]
    if nz.size == 0:
        return [0.0]
    return a[: nz[-1] + 1].tolist()
