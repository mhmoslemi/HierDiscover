def smooth_max_adam_descent(a0, target_sum, deadline, beta_start=0.1, beta_max=80.0, beta_mult=1.5, iter_per_beta=2000):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = target_sum * 0.8
    t = 0
    while time.time() < deadline:
        # compute self conv b
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] adam β={beta:.2f} it={t} best={best_val:.6f}")

        # compute softmax weights
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: 2*(w * a_rev) central slice
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)
        g = 2.0 * conv_wa[n-1 : n-1 + n]

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = step0 / np.sqrt(t) * (target_sum)  # maybe adjust factor because after dividing by sqrt(t)
        # Actually we can set lr = step0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # scale step to keep magnitude relative to target_sum; maybe compute norm
        # Ensuring not too large
        # simple: a = a - step
        a = a - step
        # projection onto simplex to enforce sum & nonnegativity
        a = _project_to_simplex(a, target_sum)

        # increase beta
        if t % iter_per_beta == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

    return best_a, best_val

def anneal_mass_transfer_v2(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()
    rng = np.random.default_rng()

    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.8  # smaller initial temperature?

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # Use linear cooling slower: fraction^0.3?
        frac = elapsed / total
        temperature = max(0.0, T0 * (1 - frac**0.5))

        # Determine peak index
        k_max = int(np.argmax(b))
        # donor selection:
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver selection: favor low heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.1, target_sum * 0.02)
        # scale with temperature fraction
        delta = rng.random() * max_delta * (temperature / T0 if T0>0 else 0)

        if delta <= 0:
            continue

        # Efficient O(n) update similar to previous
        b_new = b.copy()
        # Linear update: subtract 2*delta*a from donor's convolution region, add to receiver region
        b_new[donor:donor+n] -= 2.0 * delta * a
        b_new[receiver:receiver+n] += 2.0 * delta * a
        # Quadratic update
        b_new[2*donor] += delta*delta
        b_new[2*receiver] += delta*delta
        if donor != receiver:
            b_new[donor+receiver] -= 2.0 * delta*delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        accept = False
        if new_val <= cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val)/temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # occasional full recompute (e.g., every 5000 iterations)
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        # occasional global shuffle: move random 1% of total mass between random indices
        if it % 20000 == 0:
            # pick random subset of indices for redistribution
            total_mass = a.sum()
            # choose random target distribution from Dirichlet and project
            rand_vec = rng.random(n)
            rand_vec = _project_to_simplex(rand_vec, total_mass)
            a = rand_vec
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] random restart improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

"""
Improved search for a low‑value step sequence.

The pipeline consists of three phases:

1.  **Adam‑based smooth‑max descent** – a differentiable surrogate for the
    `max` in the objective is minimised by projected Adam.  The temperature
    β of the soft‑max is increased gradually, steering the solution towards the
    true non‑smooth problem.

2.  **Sub‑gradient descent** – after the smooth phase we directly minimise the
    true peak of the self‑convolution using a sub‑gradient (which is known in
    closed form).  This phase works completely in the time domain and needs
    only O(n) operations per iteration.

3.  **Aggressive mass‑transfer annealing** – a simulated‑annealing scheme that
    moves tiny amounts of mass from indices that contribute most to the current
    peak to indices that are small.  The temperature schedule is slower than
    before and a occasional random restart is performed to escape deep
    local minima.

All helper functions are defined at the module level; the entry point is
`propose_candidate`.  The routine respects the supplied time budget and returns
the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact benchmark value 2·n·max(b) / target_sum²."""
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – Adam‑based smooth‑max projected descent
# ----------------------------------------------------------------------


def smooth_max_adam_descent(a0: np.ndarray,
                            target_sum: float,
                            deadline: float,
                            beta_start: float = 0.1,
                            beta_max: float = 80.0,
                            beta_mult: float = 1.5,
                            it_per_beta: int = 1500) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution entry using Adam.
    The temperature β of the soft‑max is increased gradually.
    Returns the best sequence encountered and its exact benchmark value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step0 = target_sum * 0.7          # raw scaling for the Adam step
    it = 0

    while time.time() < deadline:
        # ----- evaluate current exact objective -----
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # ----- soft‑max weights for the smooth max -----
        b_shift = b - max_b                     # max shift = 0
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max (central slice) -----
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)           # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]      # central n entries

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = step0 / np.sqrt(it) * (target_sum / (np.linalg.norm(g) + eps))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_to_simplex(a, target_sum)

        # ----- increase temperature -----
        if it % it_per_beta == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

    return best_a, best_val

def subgradient_descent(a0: np.ndarray,
                        target_sum: float,
                        max_iters: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Directly minimise the true peak of the self‑convolution.
    Each iteration computes the current peak, forms the exact sub‑gradient
    (2·a_{k_max‑i}) and takes a decreasing step, projecting back onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    step0 = target_sum * 0.4      # base size of a step

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        # locate peak index
        k_max = int(np.argmax(b))

        # build sub‑gradient: g[i] = 2·a[k_max‑i] for valid i
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        g = np.zeros_like(a)
        idx = np.arange(low, high + 1, dtype=np.int64)
        if idx.size:
            g[idx] = 2.0 * a[k_max - idx]

        norm_g = np.linalg.norm(g) + 1e-12
        lr = step0 / np.sqrt(it) * (target_sum / norm_g)

        a = a - lr * g
        a = _project_to_simplex(a, target_sum)

    return best_a, best_val

def anneal_mass_transfer_v2(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves tiny amounts of mass from “bad’’ indices
    (large contribution to the current peak) to “good’’ indices (small height).
    The temperature cools slowly (square‑root law) and a periodic random
    restart is performed to explore distant basins.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9         # initial temperature as a fraction of the energy

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # square‑root cooling

        # ----- locate the current peak of the convolution -----
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – weighted by its contribution to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            it += 1
            continue

        # size of the transfer – scaled by current temperature
        max_delta = min(a[donor] * 0.09, target_sum * 0.012)
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            it += 1
            continue

        # ----- O(n) update of the convolution -----
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # ----- Metropolis acceptance -----
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # ----- periodic full recomputation to avoid drift -----
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        # ----- occasional random restart (global perturbation) -----
        if it % 25000 == 0 and it > 0:
            # generate a fresh random point on the simplex
            rnd = rng.random(n)
            rnd = _project_to_simplex(rnd, target_sum)
            a = rnd
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] random restart improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget and returns the best sequence
    found (as a list of floats).  The global variable ``height_sequence_1``
    is updated with the final result for warm‑starting future runs.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)      # the normalisation used in the reference code

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + diverse guesses)
    # --------------------------------------------------------------
    init_candidates = []

    # warm start from previous run, if it exists
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()

    # deterministic “tent’’ shape (increase then decrease linearly)
    tent = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                           np.linspace(1, 0, n - n // 2, endpoint=True)])
    tent = _project_to_simplex(tent, target_sum)
    init_candidates.append(tent)

    # a few power‑law shapes with random exponents
    for _ in range(2):
        exponent = rng.uniform(0.45, 0.80)
        pl = (np.arange(1, n + 1, dtype=np.float64) ** (-exponent))
        pl = _project_to_simplex(pl, target_sum)
        init_candidates.append(pl)

    # a completely random start
    rnd = rng.random(n)
    rnd = _project_to_simplex(rnd, target_sum)
    init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run the optimisation phases for every start, respecting time
    # --------------------------------------------------------------
    remaining_candidates = len(init_candidates)
    for idx, init in enumerate(init_candidates):
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now
        # Allocate a fraction of the remaining time to this candidate
        # (roughly equal for each still‑to‑process start)
        alloc = remaining / (remaining_candidates - idx)

        print(f"[{now:.2f}] ----- start #{idx + 1} (allocated {alloc:.1f}s) -----")

        # ----- Phase 1 : Adam‑based smooth‑max descent -----------------
        sub_deadline = time.time() + min(alloc * 0.35, 40.0)  # cap to keep room for later phases
        a_smooth, val_smooth = smooth_max_adam_descent(
            init, target_sum, sub_deadline,
            beta_start=0.1, beta_max=80.0, beta_mult=1.5, it_per_beta=1500
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ----- Phase 2 : deterministic sub‑gradient descent -----------
        # give it a modest share of the remaining budget
        now = time.time()
        sub_deadline2 = min(deadline, now + alloc * 0.20)
        a_sub, val_sub = subgradient_descent(a_smooth, target_sum,
                                             max_iters=2500,
                                             deadline=sub_deadline2)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] New global best after subgrad: {best_score:.6f}")

        # ----- Phase 3 : aggressive annealing -------------------------
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer_v2(a_sub, target_sum, deadline)
            if val_ann < best_score:
                best_score = val_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never be needed) – simple decreasing power law
    # --------------------------------------------------------------
    if best_seq is None:
        fallback = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = _project_to_simplex(fallback, target_sum)

    # --------------------------------------------------------------
    # Trim trailing zeros, final evaluation and storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                      # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starting later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
