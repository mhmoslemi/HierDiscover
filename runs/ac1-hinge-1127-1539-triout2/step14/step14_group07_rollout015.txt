    n = a.shape[0]
    best = a.copy()
    best_score = _score(a)
    it = 0
    eps = 0.02
    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        idx = np.where(conv >= M - eps * M)[0]
        if idx.size == 0:
            break

"""
Optimisation of a non‑negative step function for the “max self‑convolution”
challenge.

The algorithm combines several complementary strategies:

1️⃣  **Parametric warm‑start** – a dense scan of simple families
    (power‑law, exponential‑tilted power‑law, Gaussian, piecewise power‑law)
    to obtain a good initial shape.

2️⃣  **Projected Adam on a soft‑max surrogate** – minimises a smooth
    approximation of the max‑convolution while keeping the vector on the
    simplex (fixed ℓ1‑norm).

3️⃣  **Exact deterministic mass‑transfer (DMT)** – greedy donor/receiver
    moves that provably lower the convolution peak.

4️⃣  **Active‑set linear‑programming reduction** – solves a small LP that
    finds a displacement vector `d` (with `sum(d)=0` and `a+d≥0`)
    reducing the current maximal convolution value while respecting the
    active constraints.  This often yields a larger jump than the pairwise
    moves.

5️⃣  **Final high‑β Adam polishing** – a last Adam run with a very high
    soft‑max temperature to finish on a local optimum.

All phases respect the global deadline (≈1000 s) and the best sequence found
is trimmed of trailing near‑zero entries before being returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(b)/(sum(a)^2).  For Σa = √(2 n) the denominator equals 2·n,
    hence the score equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_power_law(
    n: int, split_frac: float, p1: float, p2: float, target_sum: float
) -> np.ndarray:
    """Two‑segment power law: exponent p1 on the left part, p2 on the right."""
    split = int(split_frac * n)
    a = np.empty(n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    a[:split] = (idx[:split] + 1) ** (-p1)
    a[split:] = (idx[split:] + 1) ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
#  Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
#  Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Multiple Adam‑GD restarts with jitter."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
#  Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray, deadline: float, target_sum: float, tol: float = 1e-12
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Active‑set linear‑programming reduction
# ----------------------------------------------------------------------


def _lp_active_set_reduction(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 12,
    init_eps: float = 0.02,
    max_constraints: int = 60,
) -> tuple[np.ndarray, float]:
    """
    Solve a small LP that finds a displacement `d` (sum(d)=0, a+d≥0)
    decreasing the maximal convolution value.
    Only constraints close to the current peak are kept (active set).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    eps = init_eps
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        # indices where the value is within eps of the max
        idx = np.where(conv >= M - eps * M)[0]
        if idx.size == 0:
            break
        if idx.size > max_constraints:
            idx = idx[np.argsort(-conv[idx])[:max_constraints]]

        C = idx.size
        A = np.zeros((C, n + 1), dtype=np.float64)
        b = np.empty(C, dtype=np.float64)

        for row_i, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            # row for d variables (2 * a_{k‑i})
            A[row_i, i_range] = 2.0 * a[k - i_range]
            # coefficient for the new peak variable M_new
            A[row_i, -1] = -1.0
            # RHS
            b[row_i] = -conv[k]

        # bounds for d_i : -a_i ≤ d_i  (no upper bound)
        bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]  # M_new ≥ 0

        # equality: sum_i d_i = 0
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0], dtype=np.float64)

        # objective: minimise the new peak M_new
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception as e:
            print(f"[{time.time():.2f}] LP exception: {e}")
            break

        if not res.success:
            # make the active set looser and retry
            eps = min(eps * 1.5, 0.1)
            it += 1
            continue

        M_new = res.x[-1]
        if M_new >= M - 1e-9:  # no genuine improvement
            eps = min(eps * 1.5, 0.1)
            it += 1
            continue

        d = res.x[:n]
        a = a + d
        a = _project_to_simplex(a, target_sum)  # clean up numeric drift
        cur_score = _score(a)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Active‑set LP it={it} eps={eps:.2e} new_score={best_score:.6f}"
            )
        else:
            # revert to previous a if no improvement (should not happen)
            a = best_a.copy()

        # tighten epsilon for a more focused next iteration
        eps = max(eps * 0.5, 1e-5)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Utility – trim trailing near‑zero entries
# ----------------------------------------------------------------------


def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]
    if nz.size == 0:
        return [0.0]
    return a[: nz[-1] + 1].tolist()


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising
    evaluate_sequence(...).  Returns the best sequence found as a list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) → score = max convolution

    # --------------------------------------------------------------
    # Phase 0 – parametric warm‑start (several families)
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = _power_law_sequence(n, 0.5, target_sum)  # fallback
    best_score = _score(best_seq)

    # power‑law grid
    for p in np.linspace(0.30, 0.85, 31):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # exp‑tilted power‑law grid
    for p in np.linspace(0.30, 0.85, 21):
        for q in np.linspace(0.0, 6.0, 21):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # Gaussian family
    for sigma in np.linspace(0.1 * n, 0.5 * n, 7):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # Piecewise power‑law family
    for split_frac in (0.3, 0.5, 0.7):
        for p1 in np.linspace(0.30, 0.60, 7):
            for p2 in np.linspace(0.60, 0.90, 7):
                seq = _piecewise_power_law(n, split_frac, float(p1), float(p2), target_sum)
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # reuse a previously stored construction if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] Initial best after families = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam‑style projected GD with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.35
    best_seq, best_score = _gradient_descent_with_restarts(best_seq, target_sum, gd_deadline)

    # --------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer
    # --------------------------------------------------------------
    dmt_deadline = time.time() + (deadline - time.time()) * 0.25
    a_dmt, val_dmt = _deterministic_mass_transfer_exact2(
        best_seq, dmt_deadline, target_sum
    )
    if val_dmt < best_score:
        best_seq, best_score = a_dmt, val_dmt
        print(f"[{time.time():.2f}] DMT improvement -> best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Active‑set LP reduction (new component)
    # --------------------------------------------------------------
    lp_deadline = time.time() + (deadline - time.time()) * 0.18
    a_lp, val_lp = _lp_active_set_reduction(best_seq, target_sum, lp_deadline)
    if val_lp < best_score:
        best_seq, best_score = a_lp, val_lp
        print(f"[{time.time():.2f}] Active‑set LP improvement -> best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Final high‑β Adam polishing
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_seq, best_score = a_fin, val_fin
            print(f"[{time.time():.2f}] Final Adam polish -> best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    final_seq = _trim_sequence(best_seq)

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
