        # for each column j (g_j), row k = j ... j+n-1
        # entry value = a[k - j]
        lo = j
        hi = j + n - 1
        # clip to rows range
        lo_clip = max(lo, 0)
        hi_clip = min(hi, rows - 1)
        # for k from lo_clip to hi_clip:
        # a_idx = k - j
        ks = np.arange(lo_clip, hi_clip + 1, dtype=np.int64)
        a_idx = ks - j
        A[ks, j] = a[a_idx]
    # Convert to csr for passing to linprog

    g = res.x
    return g

```python
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _linprog_opt_g(a, M):
    n = a.shape[0]
    rows = 2*n - 1
    data = []
    rows_ind = []
    cols_ind = []
    for j in range(n):
        # rows = j + i for i in 0..n-1
        i = np.arange(n, dtype=np.int64)
        row = j + i
        mask = row < rows
        row = row[mask]
        i = i[mask]
        rows_ind.append(row)
        cols_ind.append(np.full_like(row, j))
        data.append(a[i])
    # Concatenate
    rows_ind = np.concatenate(rows_ind)
    cols_ind = np.concatenate(cols_ind)
    data = np.concatenate(data)
    A = coo_matrix((data, (rows_ind, cols_ind)), shape=(rows, n))
    # Convert to csr for linprog
    A_csr = A.tocsr()
    c = -np.ones(n)  # maximize sum(g)
    b = np.full(rows, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_csr, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
```

```python
def _lp_refinement(a0, target_sum, deadline, max_iter=8):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        M = conv.max()
        g_raw = _linprog_opt_g(a, M)
        if g_raw is None:
            print(f"[{time.time()}] LP failed at iter {it}")
            break
        sum_g = g_raw.sum()
        if sum_g <= 1e-12:
            print(f"[{time.time()}] LP gave near-zero sum at iter {it}")
            break
        # Normalize g to target_sum
        g = g_raw * (target_sum / sum_g)
        # Determine best mixing t via simple line search
        best_local_score = best_score
        best_mix = None
        # Candidate t values
        for t in [0.2, 0.4, 0.6, 0.8]:
            a_candidate = (1 - t) * a + t * g
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            val = _score(a_candidate)
            if val < best_local_score:
                best_local_score = val
                best_mix = (t, a_candidate)
        if best_mix is not None:
            t_opt, a = best_mix
            if best_local_score < best_score - 1e-12:
                best_score = best_local_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] LP refinement iter {it} t={t_opt} improved score={best_score:.6f}")
        else:
            # no improvement, break
            print(f"[{time.time():.2f}] LP refinement iter {it} no improvement, stop")
            break
        it += 1
    return best_a, best_score
```

"""
Hybrid optimisation for the maximum self‑convolution problem.

The routine follows several stages:

  1.  A fast exploration of a few parametric families on a moderate
      support size (≈2000–2500).  The best candidate is kept.
  2.  A Frank–Wolfe style linear‑programming (LP) refinement:
      for the current sequence `a` we solve
          max   Σ g_i
          s.t.  (a * g)_k ≤ M   for all k,   g_i ≥ 0,
      where `M` is the present peak of `a*a`.  The optimiser `g` is
      normalised to the required ℓ₁‑norm and mixed with `a`.
  3.  Projected Adam gradient descent (soft‑max approximation of the
      max) with several random restarts.
  4.  Deterministic mass‑transfer (the “DMT2” move) and a global
      pairwise transfer for fine‑grained polishing.
  5.  Upscaling to the final support length (5 k–12 k depending on the
      time budget) by linear interpolation, projection onto the
      simplex and a short final Adam/transfer polishing phase.

All phases respect the global time budget (default 1000 s) and the
function returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective used by the competition: 2·n·max(b)/(sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_decay_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _double_exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n) + np.exp(-q * (n - 1 - i) / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, m_frac: float, r: float, target_sum: float) -> np.ndarray:
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)          # (m*h + (n-m)*l) = target_sum with h = r*l
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------
def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (the previous DMT2)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass transfer
# ----------------------------------------------------------------------
def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]     # view of length 2n‑1

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional re‑projection
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Linear‑programming Frank‑Wolfe oracle
# ----------------------------------------------------------------------
def _linprog_opt_g(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_i   s.t.   (a * g)_k ≤ M  (for all k),
    g_i ≥ 0.  Returns the raw optimiser g (not normalised) or None.
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build sparse matrix T where (T @ g)[k] = Σ_i a_i g_{k-i}
    data = []
    row_ind = []
    col_ind = []

    for j in range(n):
        # rows where column j participates: k = j + i, i = 0..n-1
        i = np.arange(n, dtype=np.int64)
        k = j + i
        mask = k < rows
        k = k[mask]
        i = i[mask]
        row_ind.append(k)
        col_ind.append(np.full_like(k, j))
        data.append(a[i])

    row_ind = np.concatenate(row_ind)
    col_ind = np.concatenate(col_ind)
    data = np.concatenate(data)

    A = coo_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    A_csr = A.tocsr()

    c = -np.ones(n)                # maximise Σg_i  ⇒ minimise -Σg_i
    b = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_csr, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


def _lp_refinement(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_iter: int = 8) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style refinement using the LP oracle.
    The current peak M = max(a*a) is kept as a moving constraint.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        M = _max_convolution_fft(a)
        g_raw = _linprog_opt_g(a, M)
        if g_raw is None:
            print(f"[{time.time():.2f}] LP failed at iter {it}")
            break
        s = g_raw.sum()
        if s <= 1e-12:
            print(f"[{time.time():.2f}] LP gave near‑zero sum at iter {it}")
            break

        # normalise to the required ℓ₁‑norm
        g = g_raw * (target_sum / s)

        # simple line‑search over a few mixing coefficients
        best_local = best_score
        best_mix = None
        for t in (0.2, 0.4, 0.6, 0.8):
            cand = (1 - t) * a + t * g
            cand = _project_to_simplex(cand, target_sum)
            val = _score(cand)
            if val < best_local:
                best_local = val
                best_mix = cand

        if best_mix is not None:
            a = best_mix
            best_score = best_local
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP step {it} improved score={best_score:.6f}")
        else:
            # no improvement ⇒ stop early
            print(f"[{time.time():.2f}] LP step {it} no improvement")
            break

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Resizing helper
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from the old grid to a new grid of length new_n."""
    old_n = seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    maximum self‑convolution (the competition objective).
    The routine respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose support sizes (small for the LP stage, large for final polish)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_large = 12000
    elif budget_s >= 400:
        n_large = 8000
    else:
        n_large = 5000

    # small size: enough to keep the LP tractable but still informative
    if budget_s >= 700:
        n_small = 2500
    elif budget_s >= 500:
        n_small = 2000
    else:
        n_small = 1500

    target_small = np.sqrt(2.0 * n_small)
    target_large = np.sqrt(2.0 * n_large)

    # ------------------------------------------------------------------
    # Phase 0 – quick parametric warm‑start on the small support
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # power law
    for p in np.linspace(0.0, 2.0, 21):
        seq = _power_law_sequence(n_small, float(p), target_small)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # power law with exponential tilt
    for p in np.linspace(0.0, 2.0, 11):
        for q in np.linspace(0.0, 15.0, 11):
            seq = _exp_power_law_sequence(n_small, float(p), float(q), target_small)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # pure exponential decay
    for q in np.linspace(0.0, 15.0, 11):
        seq = _exp_decay_sequence(n_small, float(q), target_small)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # symmetric double‑exponential
    for q in np.linspace(0.0, 12.0, 7):
        seq = _double_exp_sequence(n_small, float(q), target_small)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # two‑level piecewise constant
    for m_frac in np.linspace(0.01, 0.4, 5):
        for r in np.linspace(1.0, 30.0, 10):
            seq = _piecewise_sequence(n_small, float(m_frac), float(r), target_small)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # incorporate any previously stored construction
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n_small:
            prev = _resize_sequence(prev, n_small)
        prev = _project_to_simplex(prev, target_small)
        val = _score(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()

    # small random jitter around the best
    for _ in range(6):
        jitter = rng.normal(scale=target_small * 0.005, size=n_small)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_small)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] Warm‑start (n={n_small}) best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Frank‑Wolfe LP refinement (small support)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        lp_deadline = deadline - 30.0
        best_seq, best_score = _lp_refinement(best_seq,
                                              target_small,
                                              lp_deadline,
                                              max_iter=8)
        print(f"[{time.time():.2f}] After LP refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam GD with restarts (small support)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        gd_deadline = deadline - 20.0
        best_seq, best_score = _gradient_descent_with_restarts(best_seq,
                                                               target_small,
                                                               gd_deadline)
        print(f"[{time.time():.2f}] After GD (small) best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic mass‑transfer (small support)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_small)
        if val_ref < best_score:
            best_seq = a_ref.copy()
            best_score = val_ref
            print(f"[{time.time():.2f}] DMT improvement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Upscale to the final (large) support size
    # ------------------------------------------------------------------
    seq_large = _resize_sequence(best_seq, n_large)
    seq_large = _project_to_simplex(seq_large, target_large)
    best_seq = seq_large.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled to n={n_large} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final polishing on the large support
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        # a short high‑β Adam run
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_large,
                                                max_beta=5e5,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_seq = a_fin.copy()
            best_score = val_fin
            print(f"[{time.time():.2f}] Final Adam polish score={best_score:.6f}")

    # optional fine‑grained pairwise transfer
    if time.time() + 2.0 < deadline:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_large,
                                                      donor_frac=0.015,
                                                      receiver_frac=0.015)
        if val_pair < best_score:
            best_seq = a_pair.copy()
            best_score = val_pair
            print(f"[{time.time():.2f}] Pairwise refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # verification (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
