    a=a0.copy()
    n=a.size
    best_a=a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    it=1
    while time.time() < deadline:
        b=_conv_self_fft(a)
        max_b = b.max()
        cur_val = 2*n*max_b/(target_sum**2)
        if cur_val < best_val:
            best_val=cur_val
            best_a=a.copy()
            print(f"[{time.time():.2f}] Adam β={beta:.2f} it={it} best={best_val:.6f}")

        # softmax weights
        b_shift = b - max_b
        w=np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w = w / w_sum

        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)
        g = 2.0 * conv_wa[n-1: n-1+n]

        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        eta = 0.2 * target_sum  # base learning rate maybe

        a = _project_to_simplex(a, target_sum)

        # increase beta schedule
        if it % 500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.2)

        it+=1

    return best_a, best_val

        # compute max index
        k_max = int(np.argmax(b))
        # find contributions
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idx = np.arange(lo, hi+1)
        contrib = a[idx] * a[k_max - idx]
        # donor: index with largest contribution
        donor = idx[np.argmax(contrib)]
        # receiver: index with smallest a (excluding donor)
        receiver_candidates = np.where(np.arange(n)!= donor)[0]
        receiver = receiver_candidates[np.argmin(a[receiver_candidates])]

def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    if budget_s >= 800:
        n = 20000  # maybe larger
    elif budget_s >= 500:
        n = 15000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    init_candidates = generate_initial_candidates(n, target_sum)

    best_seq = None
    best_score = np.inf

    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ***** start candidate #{idx+1} *****")

        # allocate time for smooth descent: proportion 0.3 of remaining
        remain = deadline - time.time()
        if remain < 5.0:
            break
        smooth_budget = max(5.0, remain * 0.3)
        smooth_deadline = time.time() + smooth_budget

            init, target_sum, smooth_deadline,
            beta_start=0.1, beta_max=200.0,
            step_factor=0.5, beta_mult=1.3, beta_interval=500
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()

        # remaining time for anneal
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(
                a_smooth, target_sum, deadline

    n = a.shape[0]

    cur_val = _score_from_selfconv(b, n, target_sum)

    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        temperature = max(0.0, T0 * (1.0 - frac))  # linear cooling

        # Identify peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)] if contrib.sum() > 0 else rng.choice(idx)

        # deterministic receiver: index with minimal a
        receiver = int(np.argmin(a))

        if donor == receiver:
            # choose random alternative if same
            alt = rng.integers(0, n)
            if alt != donor:
                receiver = alt

        # delta scaled by temperature
        max_delta = min(a[donor] * 0.1, target_sum * 0.02)
        if max_delta <= 0.0:
            # nothing to move, break?
            if temperature == 0.0:
                break
            else:
                it += 1
                continue
        # delta proportion
        delta = max_delta * rng.random() * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            it += 1
            continue

        # O(n) update of convolution exactly
        b_new = b.copy()
        # subtract contribution from donor
        b_new[donor:donor + n] -= 2.0 * delta * a
        # add contribution from receiver
        b_new[receiver:receiver + n] += 2.0 * delta * a

        b_new[2 * receiver] += delta * delta

"""
Optimisation of a non‑negative step function for the benchmark
`evaluate_sequence`.  The method consists of three stages:

1. **Initialisation** – a diverse collection of candidate vectors is built:
   * warm‑start from the previous global best,
   * decreasing power‑law (difference of a power),
   * a symmetric “U‑shaped’’ variant of the power‑law,
   * geometric sequences,
   * simple inverse‑power sequences,
   * a few random Dirichlet draws.
   All candidates are projected onto the simplex Σa = √(2·n).

2. **Smooth‑max projected gradient descent** – the true objective
   `max(a*a)` is approximated by a soft‑max
   `log‑sum‑exp(β·(a*a)) / β`.  For a schedule of the temperature β
   (starting at 0.1, ending at 200) we perform gradient steps with an
   adaptive step size and project back onto the simplex.
   The best vector encountered during the run is kept.

3. **Focused simulated‑annealing (mass transfer)** – the convolution peak
   `k = argmax(a*a)` is located, the index that contributes most to this
   peak is chosen as a donor, and the index with the smallest height as a
   receiver.  A tiny amount of mass Δ is transferred; the convolution is
   updated in *O(n)* using the linear‑time formulas from the reference
   implementation.  The temperature is decreased linearly to 0, and
   Metropolis acceptance is applied.

A short final refinement with the smooth‑max descent is performed if time
remains.  The best sequence found is trimmed of trailing zeros, stored in
the global variable `height_sequence_1` (to warm‑start future runs) and
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact benchmark score: 2·n·max(b) / target_sum²."""
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Seed generators (parametric families)
# ----------------------------------------------------------------------


def _gen_power_diff(n: int, gamma: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^γ − i^γ, scaled to sum = target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** gamma - i ** gamma
    scale = target_sum / (n ** gamma)   # telescoping sum = n^γ
    return a * scale


def _gen_symmetric_diff(n: int, gamma: float, target_sum: float) -> np.ndarray:
    """U‑shaped version of the power‑diff sequence."""
    half = n // 2
    length = half + (n % 2)                       # centre element belongs to left half
    diff = ((np.arange(length, dtype=np.float64) + 1) ** gamma -
            np.arange(length, dtype=np.float64) ** gamma)
    a = np.zeros(n, dtype=np.float64)
    a[:length] = diff
    a[-length:] = diff[::-1]
    a = _project_to_simplex(a, target_sum)
    return a


def _gen_geometric(n: int, q: float, target_sum: float) -> np.ndarray:
    """Geometric decreasing sequence a_i = q^i, scaled to sum = target_sum."""
    a = q ** np.arange(n, dtype=np.float64)
    a = _project_to_simplex(a, target_sum)
    return a


def _gen_powerlaw(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = 1/(i+1)^p, scaled to sum = target_sum."""
    a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
    a = _project_to_simplex(a, target_sum)
    return a


def _generate_initial_candidates(n: int, target_sum: float) -> list[np.ndarray]:
    """Collect a list of diverse starting vectors."""
    candidates: list[np.ndarray] = []

    # warm‑start, if available
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = _project_to_simplex(warm, target_sum)
        candidates.append(warm)

    # parametric families
    for gamma in (0.4, 0.5, 0.6):
        candidates.append(_gen_power_diff(n, gamma, target_sum))
        candidates.append(_gen_symmetric_diff(n, gamma, target_sum))

    for q in (0.9999, 0.9995, 0.999):
        candidates.append(_gen_geometric(n, q, target_sum))

    for p in (0.5, 0.6, 0.7):
        candidates.append(_gen_powerlaw(n, p, target_sum))

    # a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        candidates.append(_project_to_simplex(rnd, target_sum))

    return candidates


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.1,
    beta_max: float = 200.0,
    step_factor: float = 0.5,
    beta_mult: float = 1.3,
    beta_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed surrogate of max(a*a) using projected gradient descent.
    The temperature β is slowly increased up to `beta_max`.  The vector is
    projected onto the simplex Σa = target_sum after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    step0 = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        # current self‑convolution and soft‑max weights
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:d} best={best_val:.6f}")

        # soft‑max over the entries of b
        b_shift = b - max_b                # max(b_shift) == 0
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ a_rev)   (central slice of length n)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)
        g = 2.0 * conv_wa[n - 1 : n - 1 + n]

        # step size – decays like 1/√it
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        # gradient step + projection
        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # temperature schedule
        if it % beta_interval == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – focused simulated‑annealing (mass transfer)
# ----------------------------------------------------------------------


def anneal_mass_transfer(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from the index that contributes most to the
    current convolution peak to the index with the smallest height.
    The convolution is updated in O(n) per move.  Temperature is linearly
    cooled from 0.9·cur_val to 0.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature proportional to the energy

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        temperature = max(0.0, T0 * (1.0 - frac))  # linear cooling

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)] if contrib.sum() > 0 else rng.choice(idx)

        # deterministic receiver: the currently smallest entry
        receiver = int(np.argmin(a))

        # avoid trivial move
        if donor == receiver:
            receiver = (receiver + 1) % n
            if donor == receiver:
                it += 1
                continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.1, target_sum * 0.02)
        if max_delta <= 0.0:
            it += 1
            continue
        delta = max_delta * rng.random() * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            it += 1
            continue

        # ---- O(n) update of the convolution (exact) --------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises the benchmark
    score.  The routine respects the given time budget and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives a finer search but costs more time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # normalisation used in the reference

    # --------------------------------------------------------------
    # Build the pool of initial candidates
    # --------------------------------------------------------------
    init_candidates = _generate_initial_candidates(n, target_sum)

    best_seq: np.ndarray | None = None
    best_score = np.inf

    cand_idx = 0
    while time.time() < deadline and cand_idx < len(init_candidates):
        init = init_candidates[cand_idx]
        cand_idx += 1

        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        print(f"[{time.time():.2f}] ----- start candidate #{cand_idx} -----")

        # ---- Phase 1 : smooth‑max descent (short budget) -----------------
        smooth_budget = min(remaining * 0.25, 30.0)   # at most 30 s per seed
        smooth_deadline = time.time() + smooth_budget

        a_smooth, val_smooth = smooth_max_descent(
            init,
            target_sum,
            smooth_deadline,
            beta_start=0.1,
            beta_max=200.0,
            step_factor=0.5,
            beta_mult=1.3,
            beta_interval=500,
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ---- Phase 2 : focussed annealing --------------------------------
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
            if val_ann < best_score:
                best_score = val_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final short refinement if time permits
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_ref, val_ref = smooth_max_descent(
            best_seq,
            target_sum,
            deadline,
            beta_start=0.1,
            beta_max=300.0,
            step_factor=0.3,
            beta_mult=1.2,
            beta_interval=200,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Refined global best: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – a simple decreasing power‑law if everything else failed
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _gen_power_diff(n, 0.5, target_sum)

    # Trim trailing near‑zeros for a compact representation
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final evaluation (outside of the optimisation loop)
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm‑starting future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
