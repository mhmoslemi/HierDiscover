    final_seq = arr.tolist()
    # store into globals

    except Exception:

    return final_seq

"""
Propose a low‑peak step‑function.

The algorithm proceeds in several stages using only NumPy (no external solvers).

1. **Diverse seeds** – power‑law, exponential, sinusoidal, triangular,
   random Dirichlet, sparse “spike” patterns and the previous best.
2. **Short local optimisation** – a mixture of soft‑max (smooth) gradient
   descent and true sub‑gradient descent on a moderate length (≈4k–6k).
3. **Upscaling** – linear interpolation to a larger length (≈12k–15k) and
   a few deterministic reorderings (alternating left/right).
4. **Long high‑β gradient polishing** – projected gradient descent on the
   soft‑max approximation with exponentially increasing temperature.
5. **Permutation annealing** – random swaps accepted by a simulated‑annealing
   schedule to explore the space of orderings.
6. **Final sub‑gradient refinement** – cheap but exact descent on the true
   max‑convolution.
7. **Trim trailing zeros** and return the sequence.

All vectors are kept on the ℓ¹‑simplex (Σ a_i = √(2·n)) so that the
objective value `2·n·max(conv)/Σa²` equals the maximum convolution entry.
The routine respects the supplied time budget (≈1000 s) and never returns
an invalid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective: 2·n·max(a*a) / (Σa)² ( = max convolution after ℓ¹‑normalisation)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _sinusoid_sequence(n: int, target_sum: float) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (t + 0.5) / n)  # positive half‑wave
    a *= target_sum / a.sum()
    return a


def _triangle_sequence(n: int, target_sum: float) -> np.ndarray:
    # peak at centre, linear to the sides
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = np.linspace(1, 0, n - n // 2)
    a = np.concatenate([left, right])
    a *= target_sum / a.sum()
    return a


def _spike_sequence(n: int, k: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.random(k)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, second‑largest, … placed alternately from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Apply alternating order, then reverse (useful as a second candidate)."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # distribution over shift indices

    # gradient = 2 * (a reversed) * p   (convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # central window of length n
    return grad, b


# ----------------------------------------------------------------------
# Momentum‑based annealed soft‑max gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of the peak.
    The temperature β is increased exponentially up to max_beta.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (fast per iteration)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution entry."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # gradient of the max entry (linear in a)
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Random‑swap simulated annealing (permutation search)
# ----------------------------------------------------------------------


def _permutation_random_swaps(a0: np.ndarray,
                             deadline: float,
                             start_temp: float = 0.5,
                             end_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """Simple annealing that swaps two indices at a time."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform swap
        a[i], a[j] = a[j], a[i]
        new_score = _score(a)

        # temperature schedule
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Swap‑anneal new best={best_score:.6f}")
        else:
            # revert
            a[i], a[j] = a[j], a[i]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the supplied time budget (≈1000 s) and returns a
    list of heights.  Intermediate progress is printed to stdout.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0 – basic parameters (size of the intermediate problem)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 500:
        n_expl = 5000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)       # ℓ¹‑target Σa_i = √(2n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1 – build a diverse pool of seeds
    # ------------------------------------------------------------------
    seed_pool: list[np.ndarray] = []

    # Uniform
    seed_pool.append(np.full(n_expl, target_expl / n_expl))

    # Power‑law
    for p in np.linspace(0.3, 1.0, 8):
        seed_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # Exponential
    for lam in np.linspace(0.002, 0.03, 7):
        seed_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # Sinusoid and triangular shapes
    seed_pool.append(_sinusoid_sequence(n_expl, target_expl))
    seed_pool.append(_triangle_sequence(n_expl, target_expl))

    # Random Dirichlet‑like draws (different concentration parameters)
    for alpha in [0.3, 0.7, 2.0]:
        a = rng.gamma(alpha, 1.0, n_expl)
        a = a * target_expl / a.sum()
        seed_pool.append(a)

    # Sparse spike patterns
    for k in [5, 10, 20, 50]:
        seed_pool.append(_spike_sequence(n_expl, k, target_expl, rng))

    # Previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size > 0:
            seed_pool.append(_resize_sequence(prev, n_expl))

    # Jittered copies for extra diversity
    for seq in list(seed_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        seed_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 2 – exploration: many short optimisation runs on the intermediate size
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate roughly 20 % of the total budget to this phase
    exploration_dead = start_time + (budget_s * 0.2)

    while time.time() < exploration_dead:
        # pick a seed (occasionally bias toward the current best)
        if best_seq is not None and rng.random() < 0.4:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(seed_pool)

        # run either soft‑max GD or true sub‑gradient descent
        if rng.random() < 0.6:
            run_dead = min(time.time() + 9.0, exploration_dead)
            a_opt, val_opt = _annealed_gradient_descent(
                seed,
                run_dead,
                target_expl,
                max_beta=5e5,
                step_factor=0.6,
                gamma=0.98,
            )
        else:
            run_dead = min(time.time() + 7.0, exploration_dead)
            a_opt, val_opt = _subgradient_descent(
                seed,
                run_dead,
                target_expl,
                max_iter=120_000,
            )

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # fallback
    if best_seq is None:
        best_seq = np.full(n_expl, target_expl / n_expl)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 3 – up‑scale to a larger problem size
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 300:
        n_final = 15000
    elif time_left > 150:
        n_final = 12000
    else:
        n_final = max(n_expl, int(time_left * 15))

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a handful of deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
    ]
    best_seq = candidates[0].copy()
    best_score = _score(best_seq)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Upscaling reorder improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4 – long high‑β gradient descent on the large problem
    # ------------------------------------------------------------------
    gd_dead = min(deadline - 12.0, time.time() + (deadline - time.time()) * 0.33)
    if time.time() < gd_dead:
        a_opt, val_opt = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] High‑β GD improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5 – permutation annealing (random swaps) on the remaining time
    # ------------------------------------------------------------------
    perm_dead = deadline - 5.0
    if time.time() < perm_dead:
        a_opt, val_opt = _permutation_random_swaps(best_seq, perm_dead)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Swap‑anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6 – final short sub‑gradient polishing
    # ------------------------------------------------------------------
    sub_dead = min(deadline - 2.0, time.time() + 4.0)
    if time.time() < sub_dead:
        a_opt, val_opt = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
            max_iter=200_000,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Final sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7 – trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # store for potential reuse by the next run
    globals()["height_sequence_1"] = final_seq

    # optional final logging (uses the provided evaluate_sequence)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
