def _adam_gradient_descent(a0, deadline, target_sum, max_beta=1e6, step_lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):

    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)  # shape (n,)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = step_lr / (np.sqrt(v_hat) + eps)  # elementwise

        a = a - lr * m_hat

        a = _project_to_simplex(a, target_sum)

        if t % 100 == 0:
            # increase beta slower
            beta = min(beta * 1.5, max_beta)

        cur = _score(a)
        if cur < best_val:

"""
Propose a low‑peak step‑function.

The algorithm builds a diverse pool of seed vectors (power‑law,
exponential, Dirichlet‑like).  Each seed is projected onto the ℓ¹‑simplex
with Σ a_i = √(2·n).  A fast “Adam” optimizer on a soft‑max
approximation of the peak of the self‑convolution is run for a long
time, with an exponential temperature schedule that drives the smooth
approximation to the true max.  Periodically the sequence is jittered
and re‑projected to escape plateaus.

After a first optimisation at a moderate length we upscale the best
candidate, try several deterministic reorderings (alternating,
reversed‑alternating, interleaved‑even‑odd and a bit‑reversal ordering)
and keep the ordering with the lowest score.  Finally a longer Adam run
polishes the enlarged sequence.

All updates keep the vector non‑negative and respect the ℓ¹‑simplex,
so the final sequence is always valid for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    rng = np.random.default_rng()
    a = rng.random(n)
    a += 1e-12                     # avoid exact zero for numerical stability
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    out = np.interp(new_x, old_x, seq).astype(np.float64)
    return out


# ----------------------------------------------------------------------
# Deterministic reorderings
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2‑nd largest → right, 3‑rd → next‑left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


def _interleaved_even_odd(seq: np.ndarray) -> np.ndarray:
    """Place the largest half on even positions, the rest on odd positions."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    half = (n + 1) // 2
    out[::2] = sorted_vals[:half]
    out[1::2] = sorted_vals[half:]
    return out


def _bit_reversal_order(seq: np.ndarray) -> np.ndarray:
    """Bit‑reversal permutation (useful for spreading large values)."""
    n = seq.shape[0]
    bits = (n - 1).bit_length()
    idx = np.arange(n, dtype=np.uint64)
    rev = np.zeros_like(idx)
    for _ in range(bits):
        rev = (rev << 1) | (idx & 1)
        idx >>= 1
    order = np.argsort(rev)
    return seq[order]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has shape n.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max probabilities

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # d/da of Σ p_k·c_k
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   max_beta: float = 5e5,
                   base_lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of the peak.
    β (temperature) starts at 0.5 and is multiplied by 1.5 every 400
    iterations until max_beta is reached.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0:
            beta = min(beta * 1.5, max_beta)

    return best_a, best_val
