    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_score = _score(a)
    best_score = cur_score
    best_a = a.copy()
    rng = np.random.default_rng()
    start = time.time()
    # Temperature schedule: T0 = cur_score * 1.0
    while time.time() < deadline:
        # fraction of done
        frac = (time.time() - start) / (deadline - start)
        T = max(0, cur_score * (1 - np.sqrt(frac))) # or use target_sum?
        # Choose donor index weighted by contribution to max.
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi+1)

        # Choose receiver index weighted inversely by value (prefer low)
        weight = 1.0/(a+1e-12)
        weight = weight / weight.sum()
        receiver = rng.choice(n, p=weight)

        if donor == receiver: continue
        # amount to move
        max_delta = min(a[donor], target_sum * 0.005) # small fraction
        if max_delta <= 0: continue
        # sampled delta
        delta = rng.random() * max_delta * (T/cur_score if cur_score>0 else 0)
        if delta <= 0: continue

        # propose new a
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        # recompute convolution through FFT (Full)
        b_new = _conv_fft(a_new)
        new_score = _score(a_new)
        # acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (T+1e-12)):
            a = a_new
            b = b_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal best={best_score:.6f}")
        # occasional full recompute to reduce drift if needed
        if rng.random() < 0.001:
            b = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score

"""
Propose a low‑score non‑negative step sequence.

Algorithm
---------

1. **Problem size** – choose n based on the time budget (larger n gives
   finer granularity).  The usual normalisation Σa = √(2·n) is used because
   the evaluation function is scale‑invariant.

2. **Guided families & fine grid search** – a few analytic families are
   examined (power‑law, exponential, product of both).  For each family a
   dense grid over the shape parameters is scanned; the best candidate
   (lowest evaluation score) is kept as a warm‑start.

3. **Projected sub‑gradient descent** – the true max‑convolution index
   is identified, a sub‑gradient is formed and the vector is updated.
   After every step the vector is projected onto the simplex
   {x≥0, Σx = √(2·n)}.  A decreasing step‑size (inverse‑square‑root)
   together with a small momentum term is used.  This phase consumes the
   major part of the remaining time.

4. **Aggressive annealing (mass transfer)** – a simulated‑annealing loop
   moves a tiny amount of mass from indices that contribute strongly to
   the current peak towards indices that are small.  The temperature is
   lowered with a square‑root schedule.  The convolution is recomputed
   with an FFT after each trial (fast enough for the modest number of
   annealing steps).

5. **Finalisation** – trailing near‑zero entries are stripped, the
   sequence is converted to a plain ``list[float]`` and stored in the
   global ``height_sequence_1`` for possible warm‑starts in future runs.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied.
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (i = 0,…,n‑1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i / (n‑1))."""
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-lam * x)
    return a


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) * exp(-lam * i/(n‑1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = idx ** (-p) * np.exp(-lam * x)
    return a


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape: a_i ∝ (x)^{α‑1} (1‑x)^{α‑1},
    x = (i+0.5)/n ∈ (0,1).
    """
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


# ----------------------------------------------------------------------
# Phase 1 – projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_projected_descent(a0: np.ndarray, target_sum: float,
                                 deadline: float):
    """
    Sub‑gradient descent on the entry achieving the max convolution.
    After each step the vector is projected onto the simplex to keep the
    sum fixed (√(2·n)).  Momentum is added to accelerate convergence.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    # momentum buffer
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        # convolution and the index of the max entry
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        # sub‑gradient of the active peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        # step size – classic 1/√t schedule
        eta = 1.0 / math.sqrt(it + 1.0)

        # simple momentum (β≈0.9)
        momentum = 0.9 * momentum + eta * grad
        a = a - momentum

        # projection onto the simplex Σa = target_sum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:6d}  score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from “bad”
    indices (large contribution to the current peak) to “good” indices
    (small height).  The convolution is recomputed via FFT after each
    trial – fast enough because the number of trials is modest.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature – a sizable fraction of the current energy
    T0 = cur_val * 0.8

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling
        temperature = max(0.0, T0 * (1.0 - math.sqrt(frac)))

        # locate current peak
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]

        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        rec_prob = 1.0 / (a + 1e-12)
        rec_prob = rec_prob / rec_prob.sum()
        receiver = rng.choice(n, p=rec_prob)

        if donor == receiver:
            continue

        # size of the transfer – a small fraction of the donor mass
        max_delta = min(a[donor] * 0.07, target_sum * 0.008)
        if max_delta <= 0.0:
            continue
        # scale with temperature (high T → larger moves)
        delta = rng.random() * max_delta * (temperature / (T0 + 1e-12))
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        new_val = _score(a_new)

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:6d}  best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 3000 == 0:
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute it={it} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the supplied time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the available budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 25000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = math.sqrt(2.0 * n)            # the usual normalisation

    # ------------------------------------------------------------------
    # Guided initialisation (shape families + fine grid)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # warm start from previous run, if available
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        val = _score(warm)
        best_seq = warm.copy()
        best_score = val
        print(f"[{time.time():.2f}] Warm‑start score = {best_score:.6f}")

    # ------- power‑law -------------------------------------------------
    for p in np.linspace(0.45, 0.75, 31):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.4f}  score={val:.6f}")

    # ------- exponential ------------------------------------------------
    for lam in np.linspace(0.2, 8.0, 31):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.4f}  score={val:.6f}")

    # ------- product of power‑law & exponential ------------------------
    # A modest random sampling is combined with a few deterministic points.
    rng = np.random.default_rng()
    for _ in range(2000):
        p = rng.uniform(0.4, 0.8)
        lam = rng.uniform(0.0, 8.0)
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] PowerExp p={p:.4f} λ={lam:.4f}  score={val:.6f}")

    # ------- symmetric Beta family --------------------------------------
    for alpha in np.linspace(0.25, 0.9, 21):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.4f}  score={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 : projected sub‑gradient descent (majority of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        sub_deadline = now + 0.60 * (deadline - now)
        a_sub, val_sub = subgradient_projected_descent(best_seq, target_sum,
                                                       sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] New global best after sub‑grad {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 : aggressive annealing mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] New global best after anneal {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)

    # strip trailing near‑zero entries for a tidy output
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final sanity check using the provided evaluate_sequence
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
