def _smooth_gradient_descent(a, deadline, target_sum, alphas, init_step_factor=0.7, decay=0.7):
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    for alpha in alphas:
        while time.time() < deadline and some_time_remain_for_this_alpha:
            conv = _conv_self(a)
            M = conv.max()
            w = np.exp(alpha * (conv - M))
            Z = w.sum()
            if Z == 0:
                w[:] = 1.0
                Z = w.size
            a_rev = a[::-1]
            conv_w_rev = _fft_convolve(w, a_rev)
            Tg = conv_w_rev[len(a)-1:len(a)-1+len(a)]
            grad = 2 * Tg / Z  # correct gradient
            norm = np.linalg.norm(grad)
            if norm == 0:
                break
            lr = step_factor * target_sum / (norm + 1e-12)
            a_new = a - lr * grad
            a_new = _project_to_simplex(a_new, target_sum)
            score_new = _score(a_new)

                best_a = a_new.copy()

                step_factor = min(step_factor * 1.1, 2.0)
            else:
                step_factor = max(step_factor * decay, 1e-5)
            a = a_new
            # optional break after some iterations if time
    return best_a, best_score

def _simulated_annealing(a, deadline, target_sum):
    best = a.copy()
    best_score = _score(a)
    cur = a.copy()
    cur_score = best_score
    n = a.shape[0]
    rng = np.random.default_rng()
    # schedule: start_temp = initial temperature relative to best_score?
    # We'll choose start_temp = 0.1 * best_score (maybe)
    start_temp = 0.1 * best_score
    end_temp = 1e-5
    # compute total iterations approx: determine number of loops = maybe 200k or less fitting time.
    # We'll compute while time < deadline:
    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** frac
        # choose donor with a_i > small epsilon
        donor = rng.choice(np.where(cur > 1e-12)[0])
        # choose receiver random any index
        receiver = rng.integers(n)
        if donor == receiver:
            continue
        max_delta = cur[donor]  # we cannot exceed donor's amount; we will transfer some fraction
        # choose delta as random fraction of donor mass
        delta = rng.random() * max_delta * 0.5  # up to half
        # ensure receiver not exceed max 1000? Not needed.
        cur2 = cur.copy()
        cur2[donor] -= delta
        cur2[receiver] += delta
        # no need to project onto simplex as sum unchanged, but ensure nonnegativity (donor >= delta)
        # compute score
        score2 = _score(cur2)
        if score2 < cur_score:
            accept = True
        else:
            prob = np.exp(-(score2 - cur_score) / (T + 1e-12))
            accept = rng.random() < prob
        if accept:
            cur = cur2
            cur_score = score2
            if score2 < best_score:
                best_score = score2
                best = cur.copy()
        # else reject
    return best, best_score

"""
Hybrid optimiser for the step‑function max‑convolution problem.

Improvements over the previous version
--------------------------------------
* The smooth‑max gradient is now mathematically correct (the spurious
  `+ a` term and missing factor 2 are removed).
* A multi‑stage smoothing schedule (α = 4 → 8 → 15 → 30) drives the
  surrogate towards the true max.
* Adaptive step‑size and simple momentum‑free updates.
* A final simulated‑annealing polishing phase that moves mass
  between two coordinates while keeping the ℓ¹‑norm exactly constant.
* Extra seed families (sqrt‑law and sinusoidal) and a richer jitter
  stage to diversify the initial population.
* The budget (≈ 1000 s) is split into DE (≈ 20 %), gradient descent
  (≈ 60 %) and SA (≈ 20 %) – each stage respects the global deadline.

The function `propose_candidate` returns a list of non‑negative heights
whose score (as reported by `evaluate_sequence`) is as low as possible.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then normalised to `target_sum`."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then normalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _sqrt_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-0.5) (a special power‑law)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-0.5))
    a *= target_sum / a.sum()
    return a


def _sinusoidal_sequence(n: int, target_sum: float, amp: float = 0.8) -> np.ndarray:
    """Positive sinusoid shifted up, then normalised."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * x / n)
    a = np.maximum(a, 1e-12)          # keep positivity
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    leftmost = True
    for v in sorted_vals:
        if leftmost:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        leftmost = not leftmost
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Classic DE loop with Euclidean simplex projection after every step.
    """
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # three distinct mates a, b, c ≠ i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # safety margin
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on the log‑sum‑exp surrogate (corrected)
# ----------------------------------------------------------------------


def _smooth_gradient_descent_corrected(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      alphas: list[float] = (4, 8, 15, 30),
                                      init_step_factor: float = 0.7,
                                      decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise `log_sum_exp(alpha·(a*a)) / alpha` by projected gradient descent.
    The surrogate approaches the true max as `alpha` grows.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    for alpha in alphas:
        # run for as long as time permits with the current α
        while time.time() < deadline:
            conv = _conv_self(a)                 # full convolution, length 2n‑1
            M = conv.max()
            # smooth‑max weights (numerically stable)
            w = np.exp(alpha * (conv - M))
            Z = w.sum()
            if Z == 0.0:
                w[:] = 1.0
                Z = w.size

            a_rev = a[::-1]
            conv_w_rev = _fft_convolve(w, a_rev)   # length 3n‑2
            Tg = conv_w_rev[n - 1:n - 1 + n]       # slice of length n

            # correct gradient of the smooth max
            grad = 2.0 * Tg / Z

            norm = np.linalg.norm(grad)
            if norm == 0.0:
                break
            # step size proportional to `target_sum` (≈ √(2n))
            lr = step_factor * target_sum / (norm + 1e-12)

            a_new = a - lr * grad
            a_new = _project_to_simplex(a_new, target_sum)

            new_score = _score(a_new)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                # become a bit bolder after a success
                step_factor = min(step_factor * 1.05, 5.0)
            else:
                # slowly shrink the step size
                step_factor = max(step_factor * decay, 1e-5)

            a = a_new
        # after exhausting the time for this α, restart the step factor
        step_factor = init_step_factor
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (mass transfer)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑style SA that repeatedly moves a random amount of
    mass from a donor index to a receiver index while keeping the ℓ¹‑norm
    constant.  The temperature is exponentially cooled.
    """
    cur = a0.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]
    rng = np.random.default_rng()
    start = time.time()

    # temperature schedule (from 20 % of current best down to ~1e‑5)
    start_temp = best_score * 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / max(deadline - start, 1e-12)
        temp = start_temp * (end_temp / start_temp) ** tau

        # choose a donor with some mass
        nonzero = np.where(cur > 1e-12)[0]
        if nonzero.size == 0:
            break
        donor = rng.choice(nonzero)
        receiver = rng.integers(n)
        if donor == receiver:
            continue

        # transfer a modest fraction (up to 30 % of donor's mass)
        delta = rng.random() * cur[donor] * 0.3
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        cand = _project_to_simplex(cand, target_sum)   # keep feasibility

        cand_score = _score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function problem.
    Returns a list of non‑negative heights whose evaluation score
    (max convolution after ℓ¹‑normalisation) is as low as possible.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------- problem size ---------------------
    if budget_s >= 850:
        n = 12000
    elif budget_s >= 600:
        n = 8000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------- seed generation -------------------
    seeds: list[np.ndarray] = []

    # power‑law family (broad range of exponents)
    for p in np.linspace(0.2, 1.5, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # sqrt‑law (p = 0.5) – a useful intermediate shape
    seeds.append(_sqrt_sequence(n, target_sum))

    # sinusoidal shape
    seeds.append(_sinusoidal_sequence(n, target_sum, amp=0.8))

    # random Dirichlet‑like points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best, if any (warm‑start)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed to increase diversity
    jittered: list[np.ndarray] = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.03, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # --------------------- initial best --------------------
    best_seq = min(seeds, key=_score)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] initial best score = {best_score:.6f}")

    # --------------------- DE phase -------------------------
    # allocate roughly 20 % of the total budget to DE
    de_deadline = min(start_time + 0.20 * budget_s,
                      deadline - 0.30 * budget_s)  # leave space for later phases
    pop_size = max(30, n // 250)          # ~30‑40 individuals (depends on n)
    population: list[np.ndarray] = [best_seq.copy()]
    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(),
                                              target_sum))

    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] after DE best score = {best_score:.6f}")

    # --------------------- smoothed gradient descent -------
    # allocate the majority of the remaining budget
    gd_deadline = deadline - 0.20 * budget_s
    best_seq, best_score = _smooth_gradient_descent_corrected(
        best_seq,
        gd_deadline,
        target_sum,
        alphas=[4, 8, 15, 30],
        init_step_factor=0.7,
        decay=0.7)
    print(f"[{time.time():.2f}] after smooth GD best score = {best_score:.6f}")

    # --------------------- simulated annealing -------------
    sa_deadline = deadline
    best_seq, best_score = _simulated_annealing(best_seq, sa_deadline, target_sum)
    print(f"[{time.time():.2f}] after SA best score = {best_score:.6f}")

    # --------------------- reordering attempts ---------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1]
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] after reordering best score = {best_score:.6f}")

    # --------------------- final trimming --------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------- store & return --------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
