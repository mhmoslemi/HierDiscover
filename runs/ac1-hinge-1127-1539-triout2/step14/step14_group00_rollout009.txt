"""
Hybrid optimiser with extended seed search, aggressive smooth‑gradient,
deterministic polishing and a final simulated‑annealing phase.
Designed to beat the previous bound (≈1.5068) on the max‑convolution
objective for the step‑function problem.

Main ideas
-----------
1. Produce a richer set of seed sequences (dense power‑law/exponential
   grids, mixed power‑law·exponential, random simplex points and the
   previous best).
2. Run a short Differential Evolution (DE) phase for global
   exploration.
3. Apply several passes of smooth gradient descent on the
   log‑sum‑exp surrogate, increasing the sharpness (α) up to 60.
4. Perform the deterministic mass‑transfer polishing that was used
   before.
5. Finish with a fast simulated‑annealing (SA) local search that moves
   mass from large entries to small ones, accepting uphill moves with
   a temperature schedule.
6. Try a few simple reorderings and trim trailing zeros.

The algorithm respects the 1000 s time budget, printing intermediate
scores and finally returning the best sequence found (as a plain list
of floats).  The global variable ``height_sequence_1`` is updated for
warm‑starts in future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective & projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    For Σa = √(2n) this equals max(a*a), i.e. the peak of the
    self‑convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _mixed_sequence(n: int, p: float, lam: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p)·exp(-lam·i) shape."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p)) * np.exp(-lam * np.arange(n))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected on the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise log‑sum‑exp surrogate of the max using projected GD.
    Larger ``alpha`` yields a tighter approximation.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2·n‑1
        M = conv.max()
        w = np.exp(alpha * (conv - M))           # weights for smooth max
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3·n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # slice of length n

        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves with a decaying Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing local search
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly transfers a small amount of mass from a large
    entry to a small entry, accepting uphill moves with a decaying temperature.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(np.array(a0, dtype=np.float64), target_sum)
    cur_score = _max_convolution_fft(a)          # same as _score when normalised
    best_seq = a.copy()
    best_score = cur_score

    start = time.time()
    T0 = cur_score * 0.05          # initial temperature (5 % of current peak)
    T_end = cur_score * 1e-5       # final temperature
    n = a.shape[0]

    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = T0 * ((T_end / T0) ** fraction)

        # pick donor (one of the largest values) and receiver (one of the smallest)
        donor = int(np.argmax(a))
        receiver = int(np.argmin(a))
        if donor == receiver:
            continue

        # move a random amount (up to half of donor's mass, but not too big)
        max_delta = min(a[donor] * 0.5, target_sum * 0.02)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # tiny numerical drift might appear – re‑project for safety
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _max_convolution_fft(a_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()

        # occasional random jitter to escape tiny plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function max‑convolution problem.
    Returns a list of non‑negative heights whose evaluation score is
    as low as possible.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ----- problem size -------------------------------------------------
    # 8000 works well for the full budget; smaller sizes for very short runs.
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ----- seed generation ------------------------------------------------
    seeds: list[np.ndarray] = []

    # power‑law family (dense grid)
    for p in np.linspace(0.15, 1.2, 15):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family (dense grid)
    for lam in np.linspace(0.0005, 0.04, 15):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # mixed power‑law ⋅ exponential
    for p in np.linspace(0.2, 1.0, 5):
        for lam in np.linspace(0.001, 0.03, 5):
            seeds.append(_mixed_sequence(n, p, lam, target_sum))

    # uniform random points on the simplex
    for _ in range(30):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previously stored best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little to increase diversity
    jittered: list[np.ndarray] = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # ----- evaluate seeds & pick initial best ----------------------------
    scored = [(seq, _score(seq)) for seq in seeds]
    scored.sort(key=lambda x: x[1])
    best_seq, best_score = scored[0]
    print(f"[{time.time():.2f}] Initial best (seed) score = {best_score:.6f}")

    # ----- DE phase (short, for global exploration) --------------------
    de_deadline = min(start_time + 0.35 * budget_s, deadline - 30.0)
    if time.time() < de_deadline:
        pop_size = max(30, n // 250)          # ~30‑40 individuals
        population = [best_seq.copy()]
        while len(population) < pop_size:
            population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))
        best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
        print(f"[{time.time():.2f}] DE finished; best score = {best_score:.6f}")

    # ----- smooth gradient descent with increasing α --------------------
    # Allocate a generous chunk for this phase.
    while time.time() < deadline - 120.0:
        remaining = deadline - time.time()
        # each α‑run gets ~20 s (or less if we are close to the end)
        run_time = min(20.0, remaining - 100.0)
        if run_time <= 10.0:
            break
        sub_dead = time.time() + run_time
        # start with moderate α and finish with a very sharp one
        for alpha in (4.0, 10.0, 30.0, 60.0):
            if time.time() >= sub_dead:
                break
            seq_s, score_s = _smooth_gradient_descent(best_seq,
                                                      sub_dead,
                                                      target_sum,
                                                      alpha=alpha,
                                                      init_step_factor=0.7)
            if score_s < best_score:
                best_score, best_seq = score_s, seq_s.copy()
                print(f"[{time.time():.2f}] Smooth (α={alpha}) improved: {best_score:.6f}")

    # ----- deterministic mass‑transfer polishing ------------------------
    if time.time() < deadline - 60.0:
        mt_dead = time.time() + 30.0
        seq_mt, score_mt = _deterministic_mass_transfer(best_seq,
                                                        mt_dead,
                                                        target_sum,
                                                        start_temp=0.4,
                                                        end_temp=1e-5,
                                                        low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved: {best_score:.6f}")

    # ----- simulated annealing local search ------------------------------
    if time.time() < deadline - 20.0:
        sa_dead = time.time() + min(25.0, deadline - time.time() - 5.0)
        seq_sa, score_sa = _simulated_annealing(best_seq, sa_dead, target_sum)
        if score_sa < best_score:
            best_score, best_seq = score_sa, seq_sa.copy()
            print(f"[{time.time():.2f}] Simulated annealing improved: {best_score:.6f}")

    # ----- reordering attempts --------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1]
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After ordering best score = {best_score:.6f}")

    # ----- optional up‑sampling & quick polish (if time permits) ----------
    if time.time() < deadline - 10.0 and n < 12000:
        up_len = min(12000, n * 2)
        up_seq = _resize_sequence(np.array(best_seq), up_len)
        up_seq = _project_to_simplex(up_seq, np.sqrt(2.0 * up_len))
        up_dead = time.time() + 5.0
        up_seq, up_score = _smooth_gradient_descent(up_seq,
                                                    up_dead,
                                                    np.sqrt(2.0 * up_len),
                                                    alpha=20.0,
                                                    init_step_factor=0.5)
        if up_score < best_score:
            best_seq = up_seq
            best_score = up_score
            print(f"[{time.time():.2f}] Upsampled polish improved: {best_score:.6f}")

    # ----- final trimming -------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ----- store & return -------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
