"""

Algorithm
---------

     ``height_sequence_1``) if available, otherwise a random simplex point.
   * The candidate is resized to the problem size n and projected onto the
     simplex Σa = sqrt(2n).

2. Phase 1 – smooth‑max descent:
   * Approximate max(b) with a temperature‑scaled soft‑max
     (log‑sum‑exp) and run projected gradient descent.
   * The gradient is obtained by a single FFT (O(n log n)).
   * β (temperature) is increased slowly (up to 80) to tighten the
     approximation.
   * Simple step‑size scaling  η≈step0·target/‖g‖·1/√t is used.

3. Phase 2 – sub‑gradient descent on the true max:
   * Locate the current peak k = argmax(b) and compute the exact
     sub‑gradient g_i = 2·a_{k‑i} (zero outside the support).
   * Perform a projected sub‑gradient step with a diminishing step size
     and keep the best point seen.

4. Phase 3 – aggressive mass‑transfer simulated annealing:
   * Repeatedly move a tiny amount of mass from a “donor’’ (large contribution
     to the current peak) to a “receiver’’ (small height) and accept moves
     with the Metropolis rule.
   * Convolution updates are performed in O(n) using the exact formula
     (identical to the reference implementation).

All phases respect a global time budget (default 1000 s).  The best sequence
found is returned as a plain list of floats, and stored in the global variable
``height_sequence_1`` for warm‑starting the next run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a power‑of‑2 FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact objective 2·n·max(b)/(target_sum²)."""
    return float(2.0 * n * b.max() / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.05,
    beta_max: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a*a) via projected gradient descent.
    The approximation is (1/β)·log Σ exp(β·b_i) where b = a*a.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start

        b = _conv_self_fft(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max / (target_sum * target_sum)
        if cur_val < best_val:

            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:6d} best={best_val:.6f}")

def anneal_mass_transfer(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that transfers small amounts of mass from indices
    that heavily contribute to the current peak to indices with a low height.
    Updates of the convolution are performed in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))  # square‑root cooling

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of transfer – scaled by temperature
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem dimension (larger n → finer granularity)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)   # normalisation used in the benchmark

    # --------------------------------------------------------------
    # Build initial candidates (warm‑start + a couple of random seeds)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    # a deterministic power‑law start (often a good baseline)
    power_law = np.arange(1, n + 1, dtype=np.float64) ** (-0.55)
    power_law = power_law * (target_sum / power_law.sum())
    init_candidates.append(power_law)

    # two random initial points
    for _ in range(2):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run optimisation phases for each candidate until time runs out
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx + 1} (n={n}) -----")

        # ---- Phase 1 : smooth‑max projected gradient descent ------------
        phase1_deadline = time.time() + 0.30 * (deadline - time.time())
        a_smooth, val_smooth = smooth_max_descent(
            init, target_sum, phase1_deadline, beta_start=0.05, beta_max=80.0
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        if time.time() >= deadline:
            break

        # ---- Phase 2 : sub‑gradient descent on the exact max ----------
        phase2_deadline = time.time() + 0.25 * (deadline - time.time())
        a_sub, val_sub = subgradient_descent(
            a_smooth, target_sum, phase2_deadline
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] New global best after subgrad: {best_score:.6f}")

        if time.time() >= deadline:
            break

        # ---- Phase 3 : aggressive annealing ---------------------------
        a_ann, val_ann = anneal_mass_transfer(
            a_sub, target_sum, deadline
        )
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never be needed) and final clean‑up
    # --------------------------------------------------------------
    if best_seq is None:
        # simple decreasing power‑law as a safe default
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)

    # Trim insignificant trailing zeros for a compact representation
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final evaluation (only for logging)
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for possible warm‑start in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
