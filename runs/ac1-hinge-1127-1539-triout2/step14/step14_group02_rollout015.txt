"""
Propose a low‑score step‑function sequence.

The search combines three stages within the allotted time:

1. **Warm‑start / random initialisation** – reuse the previously found
   `height_sequence_1` (rescaled) and a few random Dirichlet vectors.

2. **Smooth‑max projected gradient descent** – minimise a log‑sum‑exp
   surrogate of the convolution peak.  This rapidly produces a good
   structured shape.

3. **Exact sub‑gradient descent on the true max** – repeatedly locate the
   current peak of the self‑convolution, take a projected gradient step
   (the gradient of the peak is simply `2*a_rev` restricted to the
   support of the peak), and perform a simple back‑tracking line search.
   This refines the solution to the true objective.

All updates respect the simplex constraint `Σ a_i = sqrt(2n)` and
non‑negativity.  The best sequence found is returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Exact benchmark score from a self‑convolution array."""
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    beta_start: float = 0.1,
    beta_max: float = 60.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise a soft‑max approximation of the max‑convolution.  The
    approximation is (1/β)·log Σ exp(β·(a*a)_i).  After each step the
    vector is projected onto the simplex Σa = target_sum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    step0 = target_sum * 0.8
    it = 0
    while time.time() < deadline:
        # convolution and soft‑max weights
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # log‑sum‑exp weights (stable)
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ a_rev)  (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)          # length 3n‑2
        g = 2.0 * conv_wa[n - 1 : n - 1 + n]   # central n entries

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – exact sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    step0_factor: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Iteratively minimise the genuine objective 2·n·max(a*a)/sum(a)² by
    taking a projected sub‑gradient step at the current peak of the
    convolution.  A simple back‑tracking line search is used to guarantee
    a monotone decrease.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_a = a.copy()
    best_val = cur_val

    it = 0
    while time.time() < deadline:
        k = int(np.argmax(b))                     # index of the peak
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        # gradient of the peak (ordered pairs) – zero outside [lo,hi]
        g = np.zeros_like(a)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break

        # base step size – decays like 1/√it
        step0 = step0_factor * target_sum
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        # try a few scaling factors (line search)
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            a_new = a - eta * factor * g
            a_new = _project_to_simplex(a_new, target_sum)
            b_new = _conv_self_fft(a_new)
            new_val = _score_from_selfconv(b_new, n, target_sum)
            if new_val < cur_val:
                a = a_new
                b = b_new
                cur_val = new_val
                improved = True
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()
                    print(
                        f"[{time.time():.2f}] subgrad it={it:5d} best={best_val:.6f}"
                    )
                break
        if not improved:
            # No improvement with the tried factors – reduce step size a
            # little and continue; this often helps escape tiny plateaus.
            eta *= 0.5
        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a length – larger n gives more degrees of freedom.
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 24000
    elif budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build initial candidates (warm start + random)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optimise each start, respecting remaining time.
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx + 1} (n={n}) -----")

        # ---- Phase 1 : smooth‑max descent ---------------------------------
        phase1_deadline = time.time() + 0.25 * (deadline - time.time())
        a_smooth, val_smooth = smooth_max_descent(
            init, target_sum, phase1_deadline, beta_start=0.1, beta_max=60.0
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] Global best after smooth: {best_score:.6f}")

        # ---- Phase 2 : exact sub‑gradient refinement -----------------------
        if time.time() < deadline:
            a_refined, val_refined = subgradient_optimize(
                a_smooth, target_sum, deadline, step0_factor=0.6
            )
            if val_refined < best_score:
                best_score = val_refined
                best_seq = a_refined.copy()
                print(f"[{time.time():.2f}] Global best after sub‑grad: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and evaluation
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: decreasing power‑law (always feasible)
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)

    # trim trailing negligible entries for a compact representation
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
