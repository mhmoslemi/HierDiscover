"""
Improved step‑function search.

The algorithm proceeds in three stages:

1️⃣  A cheap *grid search* over two‑parameter families
    a_i = (i + α)^(-p) (i = 1 … n) projected onto the simplex
    Σa = √(2 n).  The grid (p,α) is scanned with FFT‑based exact
    evaluation of the objective 2·n·max(a*a)/Σa² (which is simply the
    peak of the self‑convolution when the sum is normalised).

2️⃣  The best vector from the grid is refined by the
    *smooth‑max projected gradient descent* from the reference solution.
    A softmax approximation of the max is minimised and the iterate is
    projected back onto the simplex after each step.

3️⃣  A final *mass‑transfer simulated annealing* stage removes any
    remaining peak by moving tiny amounts of mass from indices that
    heavily contribute to the current maximum to indices with low height.
    This stage is identical to the reference implementation.

All helper routines are top‑level; no closures or lambda expressions are
used.  The routine respects the supplied time budget (default 1000 s) and
returns a plain list of non‑negative floats suitable for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())       # smallest power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray, n: int, target_sum: float) -> float:
    """Exact value of the benchmark score for a normalised vector a."""
    b = _conv_self_fft(a)
    max_b = b.max()
    # because target_sum == sqrt(2*n) the factor simplifies to max_b,
    # but we keep the full expression for safety.
    return float(2.0 * n * max_b / (target_sum * target_sum))


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a*a) by projected gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_of_seq(a, n, target_sum)

    beta = beta_start
    step0 = target_sum * 0.8
    it = 0
    while time.time() < deadline:
        # self‑convolution
        b = _conv_self_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}]  smooth β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # soft‑max weights (stable log‑sum‑exp)
        b_shift = b - max_b
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ a_rev)  (central slice)
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)                # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]          # the central n entries

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that transfers tiny mass from ‘bad’ indices
    (large contribution to the current peak) to ‘good’ ones.
    The convolution is updated in O(n) per move (exactly as in the
    reference implementation).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_of_seq(a, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9            # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # index of the current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver favours small entries
        inv = 1.0 / (a + 1e-12)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}]  anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_of_seq(a, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}]  full recompute best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives finer granularity at higher cost.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)              # normalisation used by the benchmark

    # --------------------------------------------------------------
    # Warm‑start(s) – keep the previous best if it exists.
    # --------------------------------------------------------------
    init_candidates = []
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    # --------------------------------------------------------------
    # 1️⃣  Grid search over the two‑parameter family a_i = (i+α)^(-p)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_p = None
    best_alpha = None

    p_vals = np.linspace(0.50, 0.60, 101)        # step ≈0.001
    alpha_vals = np.linspace(0.0, 2.0, 41)       # step ≈0.05

    print(f"[{time.time():.2f}]  start grid search (n={n})")
    for p in p_vals:
        for alpha in alpha_vals:
            if time.time() >= deadline - 5.0:
                break
            # generate the raw (unnormalised) shape
            idx = np.arange(1, n + 1, dtype=np.float64) + alpha
            a = idx ** (-p)
            scale = target_sum / a.sum()
            a *= scale
            # exact evaluation via convolution
            max_b = _conv_self_fft(a).max()
            score = 2.0 * n * max_b / (target_sum * target_sum)   # = max_b
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                best_p = p
                best_alpha = alpha
                print(f"[{time.time():.2f}]  grid new best {best_score:.6f} (p={p:.4f}, α={alpha:.2f})")
        if time.time() >= deadline - 5.0:
            break

    # --------------------------------------------------------------
    # 2️⃣  Small random restarts (quick sanity checks)
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(3):
        if time.time() >= deadline - 5.0:
            break
        a = rng.random(n)
        a = _project_to_simplex(a, target_sum)
        score = _score_of_seq(a, n, target_sum)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}]  random new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Refine the best candidate with smooth‑max descent.
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        sub_deadline = time.time() + 0.35 * (deadline - time.time())
        a_smooth, val_smooth = smooth_max_descent(best_seq, target_sum,
                                                  sub_deadline,
                                                  beta_start=0.1, beta_max=30.0)
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}]  after smooth best {best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Aggressive annealing mass‑transfer.
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}]  after anneal best {best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and return
    # --------------------------------------------------------------
    if best_seq is None:
        # Fallback – simple decreasing power‑law
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)

    # remove trailing zeros for a slightly shorter representation
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final benchmark evaluation (outside the optimisation loop)
    try:
        final_val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}]  Final evaluated score = {final_val:.6f}")
    except Exception as exc:                      # pragma: no cover
        print(f"[{time.time():.2f}]  Evaluation error: {exc}")

    # store for possible warm‑start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
