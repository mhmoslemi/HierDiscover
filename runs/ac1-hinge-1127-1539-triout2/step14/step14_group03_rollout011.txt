def subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    iteration = 0
    rng = np.random.default_rng()
    n = len(a)
    start = time.time()
    while time.time() < deadline:
        # compute convolution
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]
        # correction for diagonal
        if k_max % 2 == 0:
            i_diag = k_max // 2
            if lo <= i_diag <= hi:
                grad[i_diag] -= a[i_diag]  # because we added 2*a_i but true grad is a_i
        # step size: choose diminishing
        elapsed = time.time() - start
        total = deadline - start
        # step size ~ step0 * (1 - elapsed/total) + epsilon
        step0 = target_sum * 0.005  # small fraction of total mass
        step = step0 * (1.0 - elapsed / total) + 1e-6
        a = a - step * grad
        # clip to nonnegative
        a = np.maximum(a, 0.0)
        # project to simplex to enforce sum
        a = _project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={iteration} score={val:.6f}")
        # occasional jitter
        if iteration % 1000 == 0:
            jitter = rng.normal(scale=target_sum*1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        iteration += 1
    return best_a, best_val

"""
Propose a low‑score non‑negative step sequence.

The search consists of three phases:

1. **Deterministic warm‑start** – a collection of analytic families
   (power‑law, exponential, combined power‑law·exp, beta, linear,
   quadratic, sinusoidal, symmetric power‑law) are generated, scaled to the
   usual simplex  Σa = sqrt(2·n) and the best one is kept.

2. **Projected sub‑gradient descent on the true max‑convolution** – using the
   exact active index kₘₐₓ we compute the correct sub‑gradient
   (2·a_{k‑i} except for the diagonal term) and perform a simple
   scaled‑projection onto the simplex after each step.  The step size
   decays linearly.  This phase receives ≈ 25 % of the time budget.

3. **Greedy‑annealing mass‑transfer** – repeatedly move a tiny amount of mass
   from the index contributing most to the current peak to a low‑height
   index.  The convolution is updated in O(n) using rank‑one updates and a
   Metropolis acceptance rule with a square‑root cooling schedule.
   This phase consumes the remaining time.

All helpers are top‑level; no closures or lambdas are used.
The final sequence is trimmed of trailing zeros and stored globally for a warm
start in the next invocation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score_full_conv(b: np.ndarray, target_sum: float) -> float:
    """Score from a pre‑computed convolution b (length 2n‑1)."""
    n = (b.size + 1) // 2
    return float(2.0 * n * b.max() / (target_sum * target_sum))


def _score(a: np.ndarray, target_sum: float) -> float:
    """Convenient wrapper that computes the score from a."""
    return _score_full_conv(_conv_fft(a), target_sum)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-lam * x)


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return idx ** (-p) * np.exp(-lam * x)


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


def _shape_linear(n: int) -> np.ndarray:
    """a_i ∝ (n - i)."""
    i = np.arange(n, dtype=np.float64)
    return (n - i)


def _shape_quadratic(n: int) -> np.ndarray:
    """a_i ∝ (n - i)^2."""
    i = np.arange(n, dtype=np.float64)
    return (n - i) ** 2


def _shape_sinusoid(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n) (positive half‑wave)."""
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 0.5) / n)


def _shape_symmetric_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric shape built from a decreasing power‑law on each side."""
    half = (n + 1) // 2
    left = _shape_powerlaw(half, p)
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = left[:-1][::-1]
    return np.concatenate([left, right])

# Phase 1 – projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Sub‑gradient descent on the index attaining the maximal convolution.
    After each step the vector is scaled to keep the sum fixed (√(2·n)).
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    # initial score
    b = _conv_fft(a)
    cur_score = 2.0 * n * b.max() / (target_sum * target_sum)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    # step‑size parameters
    step0 = target_sum * 0.0045  # ≈ 0.8 % of total mass per step at the start

    while time.time() < deadline:
        # recompute convolution and peak index
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient of the (non‑smooth) max entry
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        # diagonal correction when k_max is even and i = k_max/2 belongs to idx
        if k_max % 2 == 0:
            i_diag = k_max // 2
            if lo <= i_diag <= hi:
                grad[i_diag] -= a[i_diag]  # replace 2·a_i by a_i

        # linearly decreasing step size
        elapsed = time.time() - start
        total = deadline - start
        eta = step0 * (1.0 - elapsed / total) + 1e-8

        a = a - eta * grad
        # enforce non‑negativity and the simplex constraint by simple scaling
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # evaluate
        cur_score = 2.0 * n * b.max() / (target_sum * target_sum)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:6d} score={best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 1500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s > 0:
                a = a * (target_sum / s)

        it += 1

    return best_a, best_score

# Phase 2 – greedy simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Greedy mass‑transfer annealing.
    Moves a small amount of mass from the index that contributes most to the
    current peak to a low‑height index.  The convolution is updated in O(n)
    for each move.  Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_score = 2.0 * n * b.max() / (target_sum * target_sum)
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start

    # initial temperature (a fraction of the current “energy”)
    T0 = cur_score * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling; stays hot longer than linear cooling
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to that peak (a[i]·a[k‑i])
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)]  # deterministic “worst” donor

        # choose a receiver among the low‑height indices (bottom 10 %)
        # we use a cheap approximation: random draw and keep it if low enough
        receiver = rng.integers(n)
        if a[receiver] > a[donor]:
            # try again a few times; if still not better, keep the random one
            for _ in range(3):
                r2 = rng.integers(n)
                if a[r2] < a[receiver]:
                    receiver = r2

        if donor == receiver:
            continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.07, target_sum * 0.012)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_score = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance
        accept = False
        if new_score <= cur_score:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:7d} best={best_score:.6f}")

        # occasional full recomputation to avoid drift
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a * (target_sum / s)
            b = _conv_fft(a)
            cur_score = 2.0 * n * b.max() / (target_sum * target_sum)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute it={it} best={best_score:.6f}")

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    """
    start = time.time()
    deadline = start + budget_s - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)          # usual normalisation Σa = √(2·n)

    # ------------------------------------------------------------------
    # Guided initialisation (shape families)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Warm start from previous run, if any
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm_sum = warm.sum()
        if warm_sum > 0:
            warm = warm * (target_sum / warm_sum)
        else:
            warm = np.full_like(warm, target_sum / n)
        val = _score(warm, target_sum)
        best_seq = warm.copy()
        best_score = val
        print(f"[{time.time():.2f}] Warm start score = {best_score:.6f}")

    # deterministic shape sweep
    # power‑law
    for p in np.linspace(0.45, 0.75, 9):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _score(a, target_sum)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} score={val:.6f}")

    # exponential
    for lam in np.linspace(0.5, 8.0, 9):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        val = _score(a, target_sum)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.3f} score={val:.6f}")

    # combined power‑law × exponential
    for _ in range(6):
        p = np.random.random() * 0.8 + 0.1   # 0.1 … 0.9
        lam = np.random.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        val = _score(a, target_sum)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Power‑exp p={p:.3f} λ={lam:.3f} score={val:.6f}")

    # beta
    for alpha in np.linspace(0.25, 0.9, 7):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        val = _score(a, target_sum)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.3f} score={val:.6f}")

    # linear
    a = _shape_linear(n)
    a = a * (target_sum / a.sum())
    val = _score(a, target_sum)
    if val < best_score:
        best_score, best_seq = val, a.copy()
        print(f"[{time.time():.2f}] Linear shape score={val:.6f}")

    # quadratic
    a = _shape_quadratic(n)
    a = a * (target_sum / a.sum())
    val = _score(a, target_sum)
    if val < best_score:
        best_score, best_seq = val, a.copy()
        print(f"[{time.time():.2f}] Quadratic shape score={val:.6f}")

    # sinusoidal
    a = _shape_sinusoid(n)
    a = a * (target_sum / a.sum())
    val = _score(a, target_sum)
    if val < best_score:
        best_score, best_seq = val, a.copy()
        print(f"[{time.time():.2f}] Sinusoid shape score={val:.6f}")

    # symmetric power‑law
    for p in np.linspace(0.45, 0.70, 5):
        a = _shape_symmetric_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        val = _score(a, target_sum)
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] SymPowerLaw p={p:.3f} score={val:.6f}")

    # a few random Dirichlet starts
    rng = np.random.default_rng()
    for _ in range(3):
        rnd = rng.random(n)
        rnd = rnd / rnd.sum() * target_sum
        val = _score(rnd, target_sum)
        if val < best_score:
            best_score, best_seq = val, rnd.copy()
            print(f"[{time.time():.2f}] Random start score={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        sub_deadline = time.time() + 0.25 * remaining
        a_sub, val_sub = subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_score, best_seq = val_sub, a_sub.copy()
            print(f"[{time.time():.2f}] New global best after subgrad {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – aggressive annealing mass‑transfer refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        # split the remaining time into a few independent runs
        runs = 3
        per_run = remaining / runs
        for run in range(runs):
            a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum,
                                                  time.time() + per_run)
            if val_ann < best_score:
                best_score, best_seq = val_ann, a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal {best_score:.6f}")

            # small random perturbation before the next run
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            best_seq = np.maximum(best_seq + jitter, 0.0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq * (target_sum / s)

    # ------------------------------------------------------------------
    # Final cleaning and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # sanity check with the provided evaluator
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for a warm start in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
