def _golden_spread_order(seq):

    if step == 0: step = 1

        pos = (pos + step) % n

"""
Hybrid optimiser for the step‑function max‑convolution problem.

The pipeline:
1️⃣  Generate diverse seeds (power‑law, exponential, random simplex, previous best).
2️⃣  Differential Evolution on the simplex (projected) – the bulk of the budget.
3️⃣  Smoothed surrogate gradient descent (log‑sum‑exp) to quickly lower the peak.
4️⃣  Fast incremental mass‑transfer simulated‑annealing local search – the
    new component that can shave the last few hundredths off the bound.
5️⃣  Several deterministic re‑orderings (alternating, even‑spread, golden‑ratio).
6️⃣  Trim trailing zeros and store the result globally.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _golden_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread by a golden‑ratio step (ensuring a full cycle)."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    phi = (1 + 5 ** 0.5) / 2.0
    step = int(np.floor(phi * n)) % n
    if step == 0 or np.gcd(step, n) != 1:
        step = 1
        while np.gcd(step, n) != 1:
            step += 1
    out = np.empty_like(seq)
    pos = 0
    for v in sorted_vals:
        out[pos] = v
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after every step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # select three distinct mates
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # guard against overshooting deadline
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth max (log‑sum‑exp) surrogate of the autocorrelation.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                 # length 2n‑1
        M = conv.max()
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)      # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]           # slice length n

        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental mass‑transfer local search (simulated annealing)
# ----------------------------------------------------------------------


def _local_search(a0: np.ndarray,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer with a Metropolis acceptance rule.
    Performs O(n) updates using the exact convolution, thus extremely fast.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)                 # exact convolution
    best_a = a.copy()
    best_score = conv.max()
    cur_score = best_score

    rng = np.random.default_rng()
    start = time.time()
    move_cnt = 0

    while time.time() < deadline:
        # -------------------------------------------------
        # identify the current peak shift and a donor index
        # -------------------------------------------------
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k - idx]

        donor = idx[np.argmax(prod)]

        # -------------------------------------------------
        # pick a receiver that currently carries little mass
        # -------------------------------------------------
        low_thresh = np.percentile(a, 20)
        low_candidates = np.where(a <= low_thresh)[0]
        low_candidates = low_candidates[low_candidates != donor]
        if low_candidates.size == 0:
            receiver = rng.integers(n)
            while receiver == donor:
                receiver = rng.integers(n)
        else:
            receiver = rng.choice(low_candidates)

        # -------------------------------------------------
        # mass to transfer (up to half of donor's current mass)
        # -------------------------------------------------
        max_delta = a[donor] * 0.5
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        if delta < 1e-12:
            continue

        # -------------------------------------------------
        # incremental update of the convolution (exact!)
        # -------------------------------------------------
        # linear part
        conv[donor:donor + n] -= 2.0 * delta * a
        conv[receiver:receiver + n] += 2.0 * delta * a
        # quadratic corrections
        conv[2 * donor] += delta ** 2
        conv[2 * receiver] += delta ** 2
        conv[donor + receiver] -= 2.0 * delta ** 2

        # -------------------------------------------------
        # apply the mass move
        # -------------------------------------------------
        a[donor] -= delta
        a[receiver] += delta

        new_score = conv.max()

        # temperature schedule (linear decay)
        elapsed = time.time() - start
        total = deadline - start
        T = max(best_score * 0.005 * (1.0 - elapsed / total), 1e-7)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # reject – revert
            a[donor] += delta
            a[receiver] -= delta
            # recompute full convolution to restore consistency
            conv = _conv_self(a)
            cur_score = conv.max()

        move_cnt += 1
        # periodic full recomputation to avoid drift
        if move_cnt % 800 == 0:
            conv = _conv_self(a)
            cur_score = conv.max()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser that combines DE, smooth gradient descent and a fast
    incremental local search.  Returns a list of non‑negative heights.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # -------------------------------------------------
    # problem size (a bit larger when we have enough time)
    # -------------------------------------------------
    if budget_s >= 800:
        n = 10000
    elif budget_s >= 600:
        n = 8000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # -------------------------------------------------
    # seed generation
    # -------------------------------------------------
    seeds = []

    # power‑law family (wide range of exponents)
    for p in np.linspace(0.05, 2.0, 12):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family (including very flat and very steep)
    for lam in np.linspace(0.0002, 0.05, 10):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    for _ in range(14):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best (warm‑start) if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed to increase diversity
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq2 = seq + jitter
        seq2 = np.maximum(seq2, 0.0)
        seq2 = _project_to_simplex(seq2, target_sum)
        jittered.append(seq2)
    seeds.extend(jittered)

    # -------------------------------------------------
    # initialise DE population
    # -------------------------------------------------
    pop_size = max(40, n // 180)          # ~40‑60 individuals
    population = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # -------------------------------------------------
    # Differential Evolution phase (≈55 % of the budget)
    # -------------------------------------------------
    de_deadline = min(start_time + 0.55 * budget_s,
                      deadline - 30.0)          # keep room for polishing
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE done – best score = {best_score:.6f}")

    # -------------------------------------------------
    # Smoothed surrogate gradient descent (two passes)
    # -------------------------------------------------
    if time.time() < deadline - 20.0:
        # first pass – moderate smoothing
        sg_deadline = time.time() + min(20.0, deadline - time.time() - 15.0)
        seq_s, sc_s = _smooth_gradient_descent(best_seq,
                                                sg_deadline,
                                                target_sum,
                                                alpha=4.0,
                                                init_step_factor=0.7)
        if sc_s < best_score:
            best_score, best_seq = sc_s, seq_s.copy()
            print(f"[{time.time():.2f}] post‑smooth α=4 → {best_score:.6f}")

        # second pass – sharper smoothing
        if time.time() < deadline - 12.0:
            sg_deadline2 = time.time() + min(12.0,
                                             deadline - time.time() - 10.0)
            seq_s2, sc_s2 = _smooth_gradient_descent(best_seq,
                                                      sg_deadline2,
                                                      target_sum,
                                                      alpha=12.0,
                                                      init_step_factor=0.5)
            if sc_s2 < best_score:
                best_score, best_seq = sc_s2, seq_s2.copy()
                print(f"[{time.time():.2f}] post‑smooth α=12 → {best_score:.6f}")

    # -------------------------------------------------
    # Local search – fast incremental mass‑transfer (≈30 % of budget)
    # -------------------------------------------------
    if time.time() < deadline - 8.0:
        ls_deadline = deadline - 4.0  # leave a few seconds for reordering & final trim
        seq_ls, sc_ls = _local_search(best_seq, ls_deadline)
        if sc_ls < best_score:
            best_score, best_seq = sc_ls, seq_ls.copy()
            print(f"[{time.time():.2f}] local search improved → {best_score:.6f}")

    # -------------------------------------------------
    # Reordering attempts
    # -------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        _golden_spread_order(best_seq),
        best_seq[::-1]
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] after reordering best score = {best_score:.6f}")

    # -------------------------------------------------
    # Optional up‑sampling for the very last seconds
    # -------------------------------------------------
    if time.time() < deadline - 2.0 and n < 12000:
        up_len = min(12000, n * 2)
        up_seq = _resize_sequence(np.array(best_seq), up_len)
        up_seq = _project_to_simplex(up_seq, np.sqrt(2.0 * up_len))
        up_dead = time.time() + 2.0
        up_seq, up_sc = _smooth_gradient_descent(up_seq,
                                                 up_dead,
                                                 np.sqrt(2.0 * up_len),
                                                 alpha=8.0,
                                                 init_step_factor=0.5)
        if up_sc < best_score:
            best_seq = up_seq
            best_score = up_sc
            print(f"[{time.time():.2f}] up‑sample + polish → {best_score:.6f}")

    # -------------------------------------------------
    # Trim trailing near‑zero entries
    # -------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # -------------------------------------------------
    # store globally and report final evaluation
    # -------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
