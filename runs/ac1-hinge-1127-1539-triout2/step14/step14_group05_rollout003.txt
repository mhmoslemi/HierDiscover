"""
Hybrid optimizer for the step‑function benchmark.

The algorithm works in several phases within the allotted time budget.

1. **Small‑scale soft‑max GD** – on a modest size (≈5 000) we run a few
   short annealed gradient‑descent trials starting from simple parametric
   families (exponential, power‑law, random simplex).  This yields a decent
   shape for the coefficients.

2. **Upscaling** – the best small‑scale shape is interpolated to a larger
   support (≈10‑20 k).  The ℓ¹‑norm is restored by projection onto the simplex
   {x≥0, Σx = √(2·n)}.

3. **Permutation annealing** – keeping the multiset of values fixed,
   we perform a long simulated‑annealing search on the ordering
   (random swaps) while the objective (the maximum entry of the auto‑convolution)
   is evaluated via FFT.  The temperature is lowered exponentially.

4. **Exact sub‑gradient descent** – a few seconds of momentum sub‑gradient
   descent on the large problem refines the values a little further.

5. **Optional LP refinement** – a tiny linear‑program (if SciPy is present)
   is solved on the most critical convolution constraints.

6. **Trimming** – trailing near‑zero entries are removed and the final
   sequence is returned.  The global variable ``height_sequence_1`` is
   updated for possible reuse in later runs.

The whole pipeline respects non‑negativity and the ℓ¹‑simplex constraint.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Annealed soft‑max gradient descent (identical to the previous version)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (with momentum)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995,
                         tol: float = 0.99) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small linear‑program refinement (active‑set LP)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               max_iters: int,
               deadline: float,
               K: int = 12,
               mix_factor: float = 0.15) -> tuple[np.ndarray, float]:
    """Find a partner sequence g satisfying active convolution constraints and mix it."""
    try:
        from scipy.optimize import linprog
    except Exception:
        # SciPy missing – just return the input unchanged
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        # active set: top‑K indices (or all if fewer)
        if K >= len(conv):
            active = np.arange(len(conv), dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # Build A_ub * g ≤ M
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            # coefficient of g_j is a_{k-j}
            A[row, idx] = a[k - idx]

        # Maximise sum(g)  <=> minimise -sum(g)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        # Rescale to exact ℓ¹‑norm
        g = g * (target_sum / g.sum())

        # Convex combination
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation annealing (simulated‑annealing on the ordering)
# ----------------------------------------------------------------------


def _perm_anneal(seq: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 start_temp: float = 0.5,
                 end_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the permutation of a fixed multiset of coefficients.
    The multiset of values (seq) stays unchanged – only positions are swapped.
    """
    a = seq.copy()
    cur_val = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_val
    n = a.shape[0]
    rng = np.random.default_rng()
    base_time = time.time()
    total = deadline - base_time

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - base_time
        tau = elapsed / total
        temp = start_temp * ((end_temp / start_temp) ** tau)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        a[i], a[j] = a[j], a[i]
        new_val = _max_convolution_fft(a)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermAnneal it={it:6d} best={best_val:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        it += 1

    return best_a, best_val

    deadline = start_time + budget_s - 0.5
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Small‑scale exploration (soft‑max GD)
    # ------------------------------------------------------------------
    n_small = 5000
    target_small = math.sqrt(2.0 * n_small)

    # build an initial pool of candidates
    init_pool = []
    for p in np.linspace(0.2, 1.5, 6):
        init_pool.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in [0.001, 0.002, 0.004, 0.008]:
        init_pool.append(_exponential_sequence(n_small, float(lam), target_small))
    for _ in range(8):
        init_pool.append(_random_simplex(n_small, target_small, rng))

    # keep a reference to the best found so far
    best_seq_small = None
    best_val_small = np.inf

    # reserve about 200 s for the remaining phases
    while time.time() < deadline - 200.0:
        # pick a seed (occasionally jitter the current best)
        if best_seq_small is not None and rng.random() < 0.6:
            seed = best_seq_small.copy()
            jitter = rng.normal(scale=target_small * 0.004, size=n_small)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 9.0, deadline - 190.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_small,
            max_beta=5e5, step_factor=0.9, gamma=0.98)

        if val_opt < best_val_small:
            best_val_small = val_opt
            best_seq_small = a_opt.copy()
            print(f"[{time.time():.2f}] Small‑GD new best={best_val_small:.6f}")

    if best_seq_small is None:
        # fallback – a random simplex point
        best_seq_small = _random_simplex(n_small, target_small, rng)

    # ------------------------------------------------------------------
    # 2. Upscale to a larger problem size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 450:
        n_big = 25000
    elif remaining > 300:
        n_big = 20000
    elif remaining > 180:
        n_big = 15000
    elif remaining > 120:
        n_big = 10000
    else:
        n_big = n_small

    target_big = math.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_seq_small, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # ------------------------------------------------------------------
    # 3. Permutation annealing (the most time‑consuming phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        perm_deadline = deadline - 80.0
        seq_big, best_val_perm = _perm_anneal(seq_big, target_big, perm_deadline,
                                              start_temp=0.5, end_temp=1e-6)
        print(f"[{time.time():.2f}] After perm‑anneal max={best_val_perm:.6f}")
    else:
        best_val_perm = _max_convolution_fft(seq_big)

    # ------------------------------------------------------------------
    # 4. Sub‑gradient descent (fine‑grained refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        seq_big, best_val_sub = _subgradient_descent(seq_big, sub_deadline,
                                                     target_big,
                                                     step_factor=0.55,
                                                     gamma=0.995,
                                                     tol=0.995)
        print(f"[{time.time():.2f}] After sub‑grad max={best_val_sub:.6f}")
    else:
        best_val_sub = _max_convolution_fft(seq_big)

    # ------------------------------------------------------------------
    # 5. Optional short LP refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        lp_deadline = deadline - 5.0
        seq_big, best_val_lp = _lp_refine(seq_big, target_big,
                                          max_iters=12,
                                          deadline=lp_deadline,
                                          K=12,
                                          mix_factor=0.12)
        print(f"[{time.time():.2f}] After LP max={best_val_lp:.6f}")

    # ------------------------------------------------------------------
    # 6. Trim trailing near‑zeros and return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    # expose the result for possible reuse in later calls
    globals()["height_sequence_1"] = final_seq

    # final sanity check (may be omitted in the final run)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
