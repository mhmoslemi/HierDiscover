def _subgradient_max(a: np.ndarray, tol=1e-12):
    n = a.size
    conv = _conv_fft(a)
    max_b = conv.max()
    idx = np.where(conv >= max_b - tol)[0]

    g = np.zeros_like(a)
    for k in idx:
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        i = np.arange(lo, hi+1)
        g[i] += a[k - i]
    # average across argmax count
    if len(idx) > 0:
        g = g / len(idx)
    return g, max_b
```

def _adam_subgradient(a0, target_sum, deadline, max_iter=None):
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999
    step_factor = 0.8
    it = 0
    while time.time() < deadline:
        g, max_b = _subgradient_max(a)
        # Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** (it+1))
        v_hat = v / (1 - beta2 ** (it+1))
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")
        # jitter maybe every certain iterations
        if it % 200 == 0:
            # small random perturbation to avoid stagnation
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
        it += 1
        if max_iter and it >= max_iter:
            break
    return best, best_val

"""
Propose a non‑negative step function with a small maximal self‑convolution.

The search consists of the following stages (all respect the overall time
budget supplied to `propose_candidate`).

1. **Parametric warm‑start** – a quick scan over several analytic families
   (power‑law, tilted power‑law, exponential, double‑exponential,
   Gaussian, two‑level piecewise).  The best candidate among them is kept.
   A random Dirichlet start and the previous global best (if available)
   are also considered.

2. **Symmetrisation** – the current best vector is averaged with its
   reversal and projected back onto the simplex; this often lowers the
   convolution peak.

3. **Projected sub‑gradient descent on the true max‑convolution**.
   For a vector `a` the sub‑gradient of `max(a*a)` is
   `g_i = a_{k*-i}` where `k*` is an index attaining the current maximum.
   An ADAM‑style update is performed on the simplex with a decaying step
   size.  The method works directly on the non‑smooth objective (no
   soft‑max approximation).

4. **Aggressive global peak mass‑transfer** – a fraction of the total mass
   of a set of “donor” indices (chosen by large contribution to the peak)
   is moved to a single “receiver” index.  The maximal feasible
   transfer coefficient `α` is computed analytically, guaranteeing that the
   peak never increases.  The procedure is repeated while time permits.

5. **Deterministic mass‑transfer (exact DMT2)** – a greedy move that moves
   a modest amount of mass from a donor to a receiver while keeping all
   other convolution entries below the previous maximum.  This phase
   further shrinks the peak.

6. **Pairwise optimal transfer** – a finer greedy search that transfers
   mass between a single donor and a single receiver, exploiting the exact
   effect of the move on every convolution entry.

7. **Final polishing** – projected Adam gradient descent on a soft‑max
   approximation of the maximum (large β) to clean up any remaining
   imperfections.

After the optimisation the trailing near‑zero entries are trimmed and the
sequence is returned as a plain Python list.  All intermediate steps print
status information useful for debugging and for monitoring the score.
"""

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring (fast version – avoids recomputing the convolution)
# ----------------------------------------------------------------------


def _score_fast(a: np.ndarray, max_b: float) -> float:
    """Return 2·n·max_b / (sum(a)^2).  Assumes max_b is already the max
    entry of the full convolution of a with itself."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _score(a: np.ndarray) -> float:
    """Convenient wrapper that recomputes the convolution."""
    max_b = _max_convolution_fft(a)
    return _score_fast(a, max_b)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_decay_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _double_exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / n) + np.exp(-q * (n - 1 - i) / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, m_frac: float, r: float, target_sum: float) -> np.ndarray:
    m = max(1, int(round(m_frac * n)))
    denom = m * r + (n - m)       # (m*h + (n-m)*l) = target_sum with h = r*l
    l = target_sum / denom
    h = r * l
    a = np.full(n, l, dtype=np.float64)
    a[:m] = h
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (exact)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, tol: float = 1e-12):
    """
    Returns a sub‑gradient of max_k (a*a)[k] and the current maximal
    convolution value.
    """
    n = a.size
    conv = _conv_fft(a)
    max_b = conv.max()
    # all indices whose value is within tolerance of the maximum
    idx = np.where(conv >= max_b - tol)[0]

    g = np.zeros_like(a)
    for k in idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        g[i] += a[k - i]                     # contribution of a_i to (a*a)[k]
    if idx.size:
        g /= float(idx.size)                 # average if several maximisers
    return g, max_b

# ADAM‑style projected sub‑gradient descent (direct max)
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      step_factor: float = 0.8,
                      beta1: float = 0.9,
                      beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Minimise the true maximum of the self‑convolution using a
    sub‑gradient method with ADAM momentum, always staying on the simplex.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    # compute initial score once
    max_b = _max_convolution_fft(a)
    best_val = _score_fast(a, max_b)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, max_b = _subgradient_max(a)
        # ADAM updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        cur_val = _score_fast(a, max_b)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad ADAM it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Global peak‑mass transfer (aggressive variant)

def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.10,
                               receiver_frac: float = 0.10,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Reduce the current convolution peak by moving a fraction α of the total
    donor mass to a single weak receiver.  The fraction α is chosen as the
    largest feasible value that respects all convolution entries.
    """

    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    max_b = _max_convolution_fft(a)
    best_score = _score_fast(a, max_b)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second largest entry – stop condition

        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)   # contribution of donors alone

        # candidate receivers = smallest w_i (weak partners)
        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_r = None
        best_alpha = 0.0

            # α must satisfy max(conv + α·diff) ≤ max_val

            if np.any(mask_pos):

                # Since we want to reduce the peak, we look at the peak entry k_max:
                # new_peak = max_val + α * (donor_mass*shift_r[k_max] - conv_donors[k_max])
                # For other indices we must not exceed max_val:

                α_candidate = min(α_candidate, 1.0)
            else:
                α_candidate = 1.0
            if α_candidate < min_alpha:

            new_max = max_val + delta_peak

                best_r = r
                best_alpha = α_candidate

        if best_r is None:
            break

        # apply the move

        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_max = _max_convolution_fft(a)
        cur_score = _score_fast(a, cur_max)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} α={best_alpha:.4g} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        # Pre‑compute shifted views of a for every considered index
        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]     # view of length 2n‑1

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k                 # entries that would increase
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = feasible.min()
                    delta_max = min(delta_max, a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional re‑projection (numerical drift)
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} score={best_score:.6f}")

    return a, best_score

def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

    # 2) power‑law with exponential tilt (coarser grid for speed)
    for p in np.linspace(0.0, 2.0, 21):
        for q in np.linspace(0.0, 20.0, 21):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # 3) pure exponential decay
    for q in np.linspace(0.0, 20.0, 21):
        seq = _exp_decay_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 4) double exponential
    for q in np.linspace(0.0, 15.0, 16):
        seq = _double_exp_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 5) Gaussian (several widths)
    for sigma in np.linspace(0.1 * n, 0.5 * n, 7):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 6) two‑level piecewise constant
    for m_frac in np.linspace(0.01, 0.5, 8):
        for r in np.linspace(1.0, 30.0, 16):
            seq = _piecewise_sequence(n, float(m_frac), float(r), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # 7) random Dirichlet start (few samples)
    for _ in range(3):
        seq = rng.dirichlet(np.ones(n)) * target_sum
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 8) previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # 9) small jitter around the current best (helps leave flat regions)
    for _ in range(5):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] Warm‑start best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional symmetrisation
    # --------------------------------------------------------------
    sym = (best_seq + best_seq[::-1]) * 0.5
    sym = _project_to_simplex(sym, target_sum)
    sym_score = _score(sym)
    if sym_score < best_score:
        best_score = sym_score
        best_seq = sym.copy()
        print(f"[{time.time():.2f}] Symmetrisation improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Projected sub‑gradient descent on true max
    # --------------------------------------------------------------
    sub_deadline = time.time() + (deadline - time.time()) * 0.35
    best_seq, best_score = _adam_subgradient(best_seq,
                                             target_sum,
                                             sub_deadline,
                                             step_factor=0.9)
    print(f"[{time.time():.2f}] After sub‑gradient descent score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Aggressive global peak mass‑transfer (repeated)
    # --------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.10,
                                                      receiver_frac=0.10)
        if val_peak < best_score:
            best_score = val_peak
            best_seq = a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved score = {best_score:.6f}")
        else:
            # stop early if no further improvement is observed
            break

    # --------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer (exact DMT2)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Pairwise optimal transfer (fine‑grained)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing with soft‑max Adam
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam‑softmax polish improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and final verification
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final check (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
