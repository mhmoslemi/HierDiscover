"""
Hybrid optimiser for the step‑function max‑convolution problem (extended version).

Key ideas
---------
* Larger problem size (up to 12 000 entries) – the lower bound on the
  objective is ~1, so a bigger n gives more degrees of freedom.
* A richer seed pool (power‑law, exponential, Gaussian, bimodal and
  random Dirichlet‑like points) plus jitter for diversity.
* Longer DE phase with a bigger population and adaptive mutation factor.
* Adaptive smoothed gradient descent – several passes with increasing
  log‑sum‑exp temperature α (2, 5, 10, 20, 40) to tighten the max.
* Aggressive deterministic mass‑transfer (greedy + occasional random
  swaps) for many iterations.
* Short sub‑gradient descent with momentum as a final polish.
* Several cheap re‑orderings (alternating, even‑spread, random shuffle)
  and an optional up‑sampling + short smoothing if time permits.
* All stages respect the global time budget (default 1000 s) and the
  programme prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective `2·n·max(a*a) / (Σa)²`.  If Σa = √(2n) this equals `max(a*a)`.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle of the array."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _bimodal_sequence(n: int, peak_ratio: float, target_sum: float) -> np.ndarray:
    """
    Two separated bumps: left 0..n/4 and right 3n/4..n‑1,
    each following a decreasing power‑law.
    """
    a = np.zeros(n, dtype=np.float64)
    quarter = n // 4
    left_idxs = np.arange(quarter, dtype=np.int64)
    right_idxs = np.arange(3 * quarter, n, dtype=np.int64)

    left = (left_idxs + 1) ** (-peak_ratio)
    right = (n - right_idxs) ** (-peak_ratio)

    a[left_idxs] = left
    a[right_idxs] = right
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _random_shuffle_order(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """A simple random permutation (kept for a few milliseconds)."""
    perm = rng.permutation(seq.shape[0])
    return seq[perm]


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.7,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after every step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # distinct mates a,b,c ≠ i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # safety margin
        if time.time() > deadline - 1.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise `log_sum_exp(alpha·(a*a)) / alpha` by projected gradient descent.
    The surrogate approaches the true max as `alpha→∞`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        # smooth max weights (stay bounded)
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size
        # gradient term: a + (a_rev * w) / Z   (see analysis)
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # slice of length n
        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.04, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-5)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.07,
                                 random_swaps: bool = True) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of current peak
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        # pick a receiver among low contributors
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional random swap to escape plateaus
        if random_swaps and rng.random() < 0.001:
            i, j = rng.integers(0, a.shape[0], size=2)
            a[i], a[j] = a[j], a[i]

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)

        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # derivative of a_i a_{k-i} wrt a_i is a_{k-i} + a_{i} (both sides)
            grad[idx] += a[k - idx]  # first term
            grad[idx] += a[k - idx]  # second term (identical for convolution)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            # small jitter to avoid stagnation
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    seeds: list[np.ndarray] = []

    except Exception:
