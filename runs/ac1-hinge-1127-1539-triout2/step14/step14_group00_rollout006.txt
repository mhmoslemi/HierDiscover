"""
Hybrid optimizer with a linear‑programming (LP) refinement.

The algorithm proceeds in stages:

1. **Low‑dimensional seed** – generate a few diverse sequences
   (power‑law, exponential, random simplex) of size `n0` (≈2000‑3000)
   and keep the one with smallest max‑autocorrelation.

2. **Smooth surrogate descent** – a few iterations of gradient descent
   on the smooth `log‑sum‑exp` approximation of the max‑convolution.

3. **LP refinement** – given the current vector `f`,
   solve the LP  
        maximize   Σ b_i  
        s.t.      b_i ≥ 0 ,  (f * b)_k ≤ M   ∀k ,
   where `M = max (f*f)`.  
   The solution `b` is scaled back to the ℓ¹‑norm √(2n) and mixed
   with `f`.  Repeating this step strictly reduces `M`.

4. **Polishing** – a short run of the smooth descent, a deterministic
   mass‑transfer move (greedy Metropolis), and a few sub‑gradient steps.

5. **Reordering** – try several permutations (alternating, even‑spread,
   reversal) and keep the best ordering.

6. **Upsampling** – linearly interpolate the result to a larger length
   (default 8000) and run a final quick smooth‑descent pass.

All stages respect the global time budget (default 1000 s) and
periodically print progress.  The final sequence is stored in the global
variable `height_sequence_1` and returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Projection onto ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise `log_sum_exp(alpha·(a*a)) / alpha` by projected gradient descent.
    The surrogate approaches the true max as `alpha→∞`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        # smooth weights for the max (avoid overflow)
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        # a_rev is needed because convolution in Python uses flipped kernel
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # central n entries

        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _max_convolution_fft(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / max(deadline - start_time, 1e-12)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Convolution matrix for LP (linear constraints f * b ≤ M)
# ----------------------------------------------------------------------


def _convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return the (2n‑1)×n matrix A such that (A @ b) = f * b (linear conv)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = sp.lil_matrix((rows, n), dtype=np.float64)
    # each column i contributes the vector f shifted down by i rows
    for i in range(n):
        A[i:i + n, i] = f
    return A.tocsr()


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve  max Σ b_i   s.t.   b_i ≥ 0 ,   f * b ≤ M .
    Return the scaled vector g (ℓ¹‑norm = target_sum) or None on failure.
    """
    n = f.shape[0]
    A = _convolution_matrix(f)
    b = cp.Variable(n)

    # Inequality constraints
    constraints = [b >= 0,
                   A @ b <= M * np.ones(2 * n - 1)]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=3000, warm_start=True, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except Exception:
            return None

    if b.value is None:
        return None
    b_val = np.maximum(b.value, 0.0)
    s = b_val.sum()
    if s <= 0:
        return None
    g = b_val * (target_sum / s)
    return g


def _lp_iterative_improve(f: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          t: float = 0.2,
                          max_iter: int = 20) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP step and mix the result with the current vector.
    Returns the improved vector and its max‑convolution value.
    """
    best = f.copy()
    best_M = _max_convolution_fft(best)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        g = _lp_step(best, best_M, target_sum)
        if g is None:
            break

        new_f = (1.0 - t) * best + t * g
        new_f = _project_to_simplex(new_f, target_sum)
        new_M = _max_convolution_fft(new_f)

        if new_M < best_M - 1e-7:
            best = new_f
            best_M = new_M
            t = min(0.5, t * 1.05)   # become a bit bolder after improvement
        else:
            t = t * 0.7
            if t < 1e-3:
                break

    return best, best_M


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function max‑convolution problem.
    Returns a list of non‑negative heights whose evaluation score
    (max convolution after ℓ¹‑normalisation) is as low as possible.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a manageable dimension for the LP stage
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n0 = 3000
    else:
        n0 = 2000
    target_sum = np.sqrt(2.0 * n0)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.2, 1.0, 8):
        seeds.append(_power_law_sequence(n0, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 6):
        seeds.append(_exponential_sequence(n0, float(lam), target_sum))

    # random simplex points
    for _ in range(10):
        seeds.append(_random_simplex(n0, target_sum, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n0:
            seeds.append(_resize_sequence(prev, n0))
        else:
            seeds.append(prev.copy())

    # evaluate seeds (max‑convolution) and keep the best one
    seed_scores = [_max_convolution_fft(s) for s in seeds]
    best_idx = int(np.argmin(seed_scores))
    f = seeds[best_idx].copy()
    best_score = seed_scores[best_idx]
    print(f"[{time.time():.2f}] Initial best max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Short smooth descent to clean up the seed
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        f, sc = _smooth_gradient_descent(f,
                                         deadline - 10,
                                         target_sum,
                                         alpha=5.0,
                                         init_step_factor=0.7)
        if sc < best_score:
            best_score = sc
            print(f"[{time.time():.2f}] After smooth init best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # LP refinement (core improvement step)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        lp_deadline = deadline - 15
        f, sc = _lp_iterative_improve(f,
                                      target_sum,
                                      lp_deadline,
                                      t=0.2,
                                      max_iter=15)
        if sc < best_score:
            best_score = sc
            print(f"[{time.time():.2f}] After LP improve best max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Further polishing: smooth descent, mass‑transfer, sub‑gradient
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        # high‑alpha smooth descent
        f, sc = _smooth_gradient_descent(f,
                                         time.time() + 8,
                                         target_sum,
                                         alpha=8.0,
                                         init_step_factor=0.6)
        if sc < best_score:
            best_score = sc
            f = f.copy()
            print(f"[{time.time():.2f}] After smooth polish best = {best_score:.6f}")

    if time.time() < deadline - 8:
        f, sc = _deterministic_mass_transfer(f,
                                             time.time() + 6,
                                             target_sum,
                                             start_temp=0.4,
                                             end_temp=1e-5,
                                             low_frac=0.07)
        if sc < best_score:
            best_score = sc
            f = f.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best = {best_score:.6f}")

    if time.time() < deadline - 6:
        f, sc = _subgradient_descent(f,
                                     time.time() + 5,
                                     target_sum,
                                     step_factor=0.45,
                                     gamma=0.995)
        if sc < best_score:
            best_score = sc
            f = f.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Reordering attempts
    # ------------------------------------------------------------------
    candidates = [
        f,
        _alternating_order(f),
        _even_spread_order(f),
        f[::-1]
    ]
    f = min(candidates, key=_max_convolution_fft)
    best_score = _max_convolution_fft(f)
    print(f"[{time.time():.2f}] After reorder best max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Upsample to a larger length (8000) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 4:
        n_big = 8000
        if n_big > n0:
            up_seq = _resize_sequence(np.array(f), n_big)
            target_big = np.sqrt(2.0 * n_big)
            up_seq = _project_to_simplex(up_seq, target_big)

            # a quick smooth‑descent polishing on the larger vector
            up_seq, up_sc = _smooth_gradient_descent(up_seq,
                                                     time.time() + 3,
                                                     target_big,
                                                     alpha=6.0,
                                                     init_step_factor=0.5)
            if up_sc < best_score:
                f = up_seq
                best_score = up_sc
                print(f"[{time.time():.2f}] Upsample improved max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final trimming of trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(f, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store globally for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    # Optional final evaluation (for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
