"""
Multi‑stage optimizer for minimizing max self‑convolution.
Stage 0: diverse random seeds (power‑law, exponential, Dirichlet, previous best).
Stage 1: short sub‑gradient runs on modest size (n≈6000) to find a good seed.
Stage 2: upscale to a large size (≈15000‑20000) and apply a long sub‑gradient descent
          that runs for most of the remaining budget.
Stage 3: targeted permutation annealing and deterministic mass‑transfer refinement.
Final: trim trailing zeros and return the best sequence.
"""

import time
import numpy as np

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score(a):
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v, s):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n, p, target_sum):
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n, lam, target_sum):
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq):
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def _reversed_alternating(seq):
    rev = _alternating_order(seq[::-1])
    return rev[::-1]

def _subgradient_descent(a0, deadline, target_sum, max_iter=10**9):
    """Projected sub‑gradient descent on the true max‑convolution entry."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_b = conv.max()
        # locate all indices achieving the max (within epsilon)
        kmax = int(np.argmax(conv))

        # gradient of the max entry (sub‑gradient)
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        # double count off‑diagonal contributions (the convolution counts ordered pairs)
        # The expression above already includes both ordered contributions because we iterate all i
        # where a_i*a_{kmax-i} appears. No need for extra factor.

        # for even kmax add derivative of the square term (i=j)
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]   # derivative of a[mid]^2

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 3000 == 0:
            # small random perturbation to escape plateaus
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:6d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _mass_transfer_refine(a0, deadline, target_sum, low_frac=0.05):
    """Simple mass‑transfer heuristic (similar to earlier version)."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        w = a[idx] + a[kmax - idx]               # contribution of each ordered pair (i,k-i)

        donor = idx[np.argmax(w)]

        # select a low contribution index from the same window
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")

    return best_a, best_score

def _permutation_annealing(a0, deadline, start_temp=0.5, end_temp=1e-5, top_frac=0.07, low_frac=0.07):
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax_val = conv.max()
        # which shift(s) are maximal?
        ks = np.flatnonzero(conv == kmax_val)
        k = rng.choice(ks)

        # compute contribution of each index to this shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        # note that ordered pairs appear twice except diagonal;
        # we keep this linear contribution as proxy for importance.

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))
        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Perm new best={best_score:.6f}")
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs):
    """
    Multi‑stage optimizer that first explores many short sub‑gradient runs,
    then up‑scales the best seed and performs a long sub‑gradient descent on a
    large problem size, followed by permutation annealing and a final
    mass‑transfer polish.  Returns a list of non‑negative heights.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – basic parameters (exploration size)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 500:
        n_expl = 5000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – build a diverse seed pool
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.30, 0.90, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like seeds
    for _ in range(12):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing seed
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size:
            init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    copies = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        copies.append(seq_j)
    init_pool.extend(copies)

    # ------------------------------------------------------------------
    # Phase 2 – short exploration runs (sub‑gradient)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate ~30 % of budget to exploration
    explore_deadline = deadline - (budget_s * 0.3)

    while time.time() < explore_deadline:
        # pick a seed (biased towards current best)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # run a short sub‑gradient descent (≈ 8 s)
        run_dead = min(time.time() + 8.0, explore_deadline)
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_expl)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

        if time.time() >= explore_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – upscale to a larger size
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback
        best_seq = _power_law_sequence(n_expl, 0.6, target_expl)

    remaining = deadline - time.time()
    # decide final length based on remaining budget and total budget
    if remaining > 250.0 and budget_s >= 800:
        n_final = 18000
    elif remaining > 180.0:
        n_final = 15000
    elif remaining > 100.0:
        n_final = 12000
    else:
        n_final = max(n_expl, int(remaining * 12))   # rough fallback
    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Try a few deterministic re‑orderings and keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
    ]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled n={n_final}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – long sub‑gradient descent on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_dead = deadline - 20.0   # leave time for later phases
        a_big, val_big = _subgradient_descent(best_seq, sub_dead, target_final)
        if val_big < best_score:
            best_score = val_big
            best_seq = a_big.copy()
            print(f"[{time.time():.2f}] Long sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – permutation annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        perm_dead = deadline - 5.0
        a_perm, val_perm = _permutation_annealing(best_seq, perm_dead,
                                                  start_temp=0.5,
                                                  end_temp=1e-5,
                                                  top_frac=0.07,
                                                  low_frac=0.07)
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Perm anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = deadline - 0.5
        a_final, val_final = _mass_transfer_refine(best_seq, mt_dead, target_final)
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation for logging
    try:
        val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Multi‑stage optimizer for the step‑function problem.

The algorithm works as follows:

1. Build a diverse pool of seeds (power‑law, exponential, random Dirichlet,
   a linear decreasing profile and, if available, the previous best solution).
2. Run many *short* sub‑gradient optimisations on a modest length (≈ 5–6 k)
   to locate a promising seed.
3. Upscale the best seed to a large length (≈ 12–18 k) and apply a long
   sub‑gradient descent that consumes most of the remaining time.
4. Follow up with a permutation‑annealing that swaps indices contributing
   heavily to the current peak with low‑contributing ones.
5. End with a deterministic mass‑transfer refinement and trim trailing zeros.

All updates keep the vector on the ℓ¹‑simplex
{ x ≥ 0 , Σ x = √(2·n) } so that the evaluation function reduces to the
maximum entry of the self‑convolution.
The routine respects the supplied time budget (≈ 1000 s) and never returns
an invalid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² ( = max convolution after normalization)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, second‑largest, … placed alternately from the left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 10**9) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the max entry of a*a."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of max entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]                     # each ordered pair contributes once
        if kmax % 2 == 0:                          # diagonal term a[mid]^2
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]                # adds the missing second copy

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 3000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:6d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (local moves)
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Local mass‑transfer moves within the active window of the current peak."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        # proxy for contribution: sum of the two masses that meet at this shift
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray,
                           deadline: float,
                           start_temp: float = 0.5,
                           end_temp: float = 1e-5,
                           top_frac: float = 0.07,
                           low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Swap an index with a large contributor to the peak with a low contributor."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)
        k = rng.choice(ks)

        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))
        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)
        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Perm new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (~1000 s) and returns a
    list of heights.  Intermediate progress is printed to stdout.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – parameters for the exploration stage
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 500:
        n_expl = 5000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – build a diverse initial pool
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.30, 0.90, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like seeds
    for _ in range(12):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing seed
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size:
            init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Phase 2 – short exploration runs (sub‑gradient)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    explore_deadline = deadline - (budget_s * 0.30)   # ~30 % of the total budget
    while time.time() < explore_deadline:
        # pick a seed (bias towards current best)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # short sub‑gradient descent (≈8 s)
        run_dead = min(time.time() + 8.0, explore_deadline)
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_expl)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

        if time.time() >= explore_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – upscale to a large problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.6, target_expl)

    remaining = deadline - time.time()
    if remaining > 250.0 and budget_s >= 800:
        n_final = 18000
    elif remaining > 180.0:
        n_final = 15000
    elif remaining > 100.0:
        n_final = 12000
    else:
        n_final = max(n_expl, int(remaining * 12))   # very rough fallback

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few systematic reorderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled n={n_final}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – long sub‑gradient descent on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_dead = deadline - 20.0   # keep ~20 s for later phases
        a_big, val_big = _subgradient_descent(best_seq, sub_dead, target_final)
        if val_big < best_score:
            best_score = val_big
            best_seq = a_big.copy()
            print(f"[{time.time():.2f}] Long sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – permutation annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        perm_dead = deadline - 5.0
        a_perm, val_perm = _permutation_annealing(best_seq, perm_dead,
                                                  start_temp=0.5,
                                                  end_temp=1e-5,
                                                  top_frac=0.07,
                                                  low_frac=0.07)
        if val_perm < best_score:

            print(f"[{time.time():.2f}] Perm anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)

    except Exception:

    globals()["height_sequence_1"] = final_seq
    return final_seq
