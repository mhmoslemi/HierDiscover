"""
Hybrid hill‑climbing & simulated‑annealing optimizer for the
minimum‑peak convolution problem.

The routine builds an initial simplex‑projected candidate (interpolated
from the previous best if available, otherwise a random power‑law or
exponential seed), then spends the majority of the time budget performing
mass‑transfer moves that lower the current peak of the auto‑convolution.
A linear cooling schedule (simulated‑annealing) is used to accept
occasionally worsening moves, allowing escape from local minima.
Every few thousand iterations a cheap reordering (alternating,
even‑spread or a random permutation) is tried; any improvement is kept.
Finally the sequence is trimmed and stored in ``height_sequence_1`` for
warm‑starting subsequent runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negativity and fixed ℓ¹‑norm
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories – diverse seeds
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers – often lower the peak
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Objective wrapper (uses the true evaluator's formula)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective used in the competition (no scaling needed)."""
    n = a.shape[0]
    maxb = _max_conv_fft(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * maxb / (s * s))

# ----------------------------------------------------------------------

                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] new best = {best_val:.6f}")
        else:
            # revert move
            cur_seq[donor] += delta
            cur_seq[receiver] -= delta

        iter_cnt += 1

        # --------------------------------------------------------------
        # Periodic reordering attempts (alternating, even‑spread, random)
        # --------------------------------------------------------------
        if iter_cnt % reord_interval == 0:
            candidates = [
                _alternating_order(best_seq),
                _even_spread_order(best_seq),
                np.random.permutation(best_seq)
            ]
            for cand in candidates:
                cand_val = _objective(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    cur_seq = cand.copy()
                    cur_max = _max_conv_fft(cur_seq)
                    print(f"[{time.time():.2f}] reorder improve = {best_val:.6f}")

        # --------------------------------------------------------------
        # Every few seconds re‑project to the simplex to avoid drift.
        # --------------------------------------------------------------
        if iter_cnt % 5000 == 0:
            cur_seq = _project_to_simplex(cur_seq, target_sum)

    # --------------------------------------------------------------
    # Final deterministic polishing phase (greedy moves only)
    # --------------------------------------------------------------
    polish_deadline = deadline - 1.0
    while time.time() < polish_deadline:
        conv = _conv_fft(cur_seq)
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = kmax - idx
        prod = cur_seq[idx] * cur_seq[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        if cur_seq[partner[donor_local]] > cur_seq[donor]:
            donor = int(partner[donor_local])

        # pick a receiver with a low contribution
        low_cnt = max(1, int(len(idx) * low_frac))
        low_loc = np.argpartition(prod, low_cnt - 1)[:low_cnt]
        receiver = int(rng.choice(idx[low_loc]))
        if donor == receiver:
            continue

        delta = cur_seq[donor] * 0.3   # aggressive greedy step
        cur_seq[donor] -= delta
        cur_seq[receiver] += delta

        new_max = _max_conv_fft(cur_seq)
        if new_max < cur_max:
            cur_max = new_max
            cur_val = 2.0 * n * cur_max / (target_sum ** 2)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] polishing improved = {best_val:.6f}")
        else:
            # revert
            cur_seq[donor] += delta
            cur_seq[receiver] -= delta

    # --------------------------------------------------------------
    # Trim trailing zeros for a compact representation.
    # --------------------------------------------------------------
    arr = best_seq
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store for warm‑starting the next run.
    globals()["height_sequence_1"] = final_seq

    # Sanity check with the provided evaluator (optional).
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
