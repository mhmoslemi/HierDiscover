"""
Hybrid optimizer for the step‑function benchmark (≈1000 s budget).

Main ideas
----------
1.  A rich pool of parametric seeds (power‑law, exponential,
    cosine, half‑sine, Dirichlet‑draws) is explored with a fast
    annealed‑soft‑max gradient descent on a modest size (≈6 k).
2.  The best seed is refined by a small linear‑program (active‑set
    on the K largest convolution entries) – this pushes the ℓ∞
    bound down.
3.  The refined shape is up‑sampled to a large support (≈30‑40 k) and
    a deterministic “low‑collision’’ permutation is applied.
4.  A long exact‑subgradient descent (with momentum) is run on the
    large problem.  Most of the remaining time is spent here.
5.  Finally a short permutation‑search, hill‑climb, mass‑transfer
    simulated‑annealing and a last soft‑max GD pass are performed to
    polish the solution.
6.  Trailing near‑zero entries are trimmed; the final sequence is
    returned.  The whole pipeline obeys non‑negativity and the
    ℓ¹‑simplex (sum = √(2 n)) constraints at every step.

The evaluation function `evaluate_sequence` is assumed to be present
in the global scope and is used only for the final printed score.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _raised_cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _half_sine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, alpha: float, target_sum: float,
                        rng: np.random.Generator) -> np.ndarray:
    # Gamma‑Dirichlet construction
    a = rng.gamma(shape=alpha, scale=1.0, size=n)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float,
                    rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Deterministic low‑collision permutations
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest, then smallest, then second‑largest, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the sorted values with a step that is coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max (annealed) gradient descent on the smooth max‑convex surrogate
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} max={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑conv) with momentum
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.55,
                         gamma: float = 0.995,
                         tol: float = 0.995) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * tol).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} max={best_val:.6f}")

        it += 1

    return best_a, best_val

# Random permutation search + local hill‑climb
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        max_trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb it={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer (pairwise donor/receiver moves)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.25,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of current max
        kmax = int(np.argmax(conv))

        # indices that actually contribute to that entry
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributions a[i] * a[kmax-i]
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:        # occasional jitter
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small LP refinement on the modest‑size problem (active‑set)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               max_iters: int,
               deadline: float,
               K: int = 20,
               mix_factor: float = 0.12) -> tuple[np.ndarray, float]:
    try:
        from scipy.optimize import linprog
    except Exception:
        print("SciPy not available – skipping LP stage.")
        return a0, _max_convolution_fft(a0)

    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        M = conv.max()

        # select the K largest convolution entries (or all if less)
        if K >= conv.size:
            active = np.arange(conv.size, dtype=int)
        else:
            active = np.argpartition(conv, -K)[-K:]

        # build the inequality matrix: for each selected k,
        # sum_j a_{k-j} * g_j ≤ M
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            A[row, idx] = a[k - idx]

        # maximise ∑ g_j  <=> minimise -∑ g_j
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True})
        if not res.success:
            print(f"[{time.time():.2f}] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        g = g * (target_sum / g.sum())   # exact ℓ¹‑norm

        # convex combination with a small mixing factor
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter={it} max={best_val:.6f}")

    return best_a, best_val

    # 1. Broad exploration on a modest size (≈6 k)
    # --------------------------------------------------------------
    n_mod = 6000
    target_mod = math.sqrt(2.0 * n_mod)

    init_pool: List[np.ndarray] = []

    # parametric families – richer grid
    for p in np.linspace(0.05, 3.0, 18):
        init_pool.append(_power_law_sequence(n_mod, float(p), target_mod))
    for lam in np.linspace(0.0002, 0.07, 22):
        init_pool.append(_exponential_sequence(n_mod, float(lam), target_mod))
    for _ in range(10):
        init_pool.append(_raised_cosine_sequence(n_mod, target_mod))
    for _ in range(10):
        init_pool.append(_half_sine_sequence(n_mod, target_mod))
    for alpha in [0.2, 0.4, 0.6, 0.8, 1.2]:
        init_pool.append(_dirichlet_sequence(n_mod, alpha, target_mod, rng))
    for _ in range(15):
        init_pool.append(_random_simplex(n_mod, target_mod, rng))

    # reuse previous successful construction if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_mod))

    # jittered variants of each
    jitter_scale = target_mod * 0.03
    for seq in list(init_pool):
        seq_j = seq + rng.normal(scale=jitter_scale, size=n_mod)
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_mod)
        init_pool.append(seq_j)

    best_seq = None
    best_val = np.inf

    # annealed GD runs – consume most of the first ~200 s
    while time.time() < deadline - 180.0:
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_mod * 0.008, size=n_mod)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_mod)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 15.0, deadline - 165.0)
        a_opt, val_opt = _annealed_gradient_descent(seed, run_deadline,
                                                    target_mod,
                                                    max_beta=9e5,
                                                    step_factor=0.9,
                                                    gamma=0.98)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_val:.6f}")

    # --------------------------------------------------------------
    # 2. LP refinement on the modest size
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        lp_deadline = time.time() + min(30.0, deadline - time.time() - 120.0)
        best_seq, lp_val = _lp_refine(best_seq, target_mod,
                                      max_iters=70,
                                      deadline=lp_deadline,
                                      K=25,
                                      mix_factor=0.12)
        if lp_val < best_val:
            best_val = lp_val
            print(f"[{time.time():.2f}] After LP refinement best={best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Upscale to a larger problem size (≈30‑40 k)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 500.0:
        n_big = 40000
    elif remaining > 350.0:
        n_big = 30000
    elif remaining > 250.0:
        n_big = 25000
    else:
        n_big = max(8000, n_mod)   # fall back to a medium size

    target_big = math.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_seq if best_seq is not None
                               else _power_law_sequence(n_mod, 0.7, target_mod),
                               n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # deterministic spread of large values
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # 4. Heavy sub‑gradient descent (majority of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sub_deadline = deadline - 100.0
        seq_big, sub_val = _subgradient_descent(seq_big, sub_deadline,
                                                target_big,
                                                step_factor=0.58,
                                                gamma=0.997,
                                                tol=0.997)
        if sub_val < best_val:
            best_val = sub_val
            print(f"[{time.time():.2f}] Sub‑grad descent best={best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Random permutation search
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        perm_deadline = time.time() + min(12.0, deadline - time.time() - 70.0)
        seq_big, perm_val = _permutation_search(seq_big,
                                                max_trials=300,
                                                deadline=perm_deadline)
        if perm_val < best_val:
            best_val = perm_val
            print(f"[{time.time():.2f}] Random perm best={best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Local permutation hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline - 70.0:
        hill_deadline = time.time() + min(8.0, deadline - time.time() - 60.0)
        seq_big, hill_val = _local_permutation_hillclimb(seq_big,
                                                         hill_deadline)
        if hill_val < best_val:
            best_val = hill_val
            print(f"[{time.time():.2f}] Hill‑climb best={best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Mass‑transfer simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 50.0:
        mt_deadline = time.time() + min(20.0, deadline - time.time() - 45.0)
        seq_big, mt_val = _mass_transfer_simulated_annealing(seq_big,
                                                             mt_deadline,
                                                             target_big,
                                                             start_temp=0.30,
                                                             end_temp=1e-5,
                                                             low_frac=0.06)
        if mt_val < best_val:
            best_val = mt_val
            print(f"[{time.time():.2f}] MT best={best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Final short annealed‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        final_deadline = deadline - 25.0
        seq_big, final_val = _annealed_gradient_descent(seq_big,
                                                        final_deadline,
                                                        target_big,
                                                        max_beta=2e6,
                                                        step_factor=0.85,
                                                        gamma=0.97)
        if final_val < best_val:
            best_val = final_val
            print(f"[{time.time():.2f}] Final GD polish best={best_val:.6f}")

    # --------------------------------------------------------------
    # 9. Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-13)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq   # for possible next run

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
```

This implementation follows the described multi‑stage pipeline, uses only standard scientific‑Python libraries, respects all constraints, and aggressively spends the allocated ~1000 seconds on the most promising optimisation components.
