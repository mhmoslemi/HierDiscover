"""
Improved step‑function optimizer.

The algorithm explores several diverse initializations (warm start,
random Dirichlet, exponential decay, semi‑circular shape) and for
each start performs two optimisation phases:

1. Smooth‑max projected gradient descent.  The true max in the
   objective is approximated by a soft‑max (log‑sum‑exp) with a
   gradually increasing temperature β.  A larger β (up to 150) gives
   a tighter approximation, while a simple Euclidean projection
   onto the simplex Σa = √(2n) enforces the normalisation.

2. Aggressive simulated‑annealing mass‑transfer.  Moves shift a
   small amount of mass from indices that contribute heavily to the
   current peak of the convolution to indices with very little mass.
   The temperature schedule is square‑root cooling and the move size
   is scaled with the temperature, allowing large exploratory steps
   early and fine adjustments later.

The two phases are repeated (smooth → anneal → smooth) within the
time budget, and the best sequence (according to the exact max‑
convolution value) is returned.  All numbers are non‑negative, and
the final sequence is trimmed of trailing zeros.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a via a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b via a power‑of‑2 FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int) -> float:
    """Exact objective: max(b) (since sum = √(2n) ⇒ denominator = 2n)."""
    return float(b.max())


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 150.0,
                       step0_factor: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution entry.
    After each step the vector is projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_from_selfconv(_conv_self_fft(a), n)

    beta = beta_start
    step0 = target_sum * step0_factor
    it = 0

    while time.time() < deadline:
        # 1) self‑convolution and soft‑max weights
        b = _conv_self_fft(a)                # length 2n‑1
        max_b = b.max()
        cur_val = max_b                      # exact objective
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # log‑sum‑exp (numerically stable)
        b_shift = b - max_b                 # max = 0
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 2) gradient = 2·(w ∗ a_rev) on the central slice
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)       # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]  # central n entries

        # 3) step size (decaying with √it)
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        # gradient step + projection
        a -= eta * g
        a = _project_to_simplex(a, target_sum)

        # 4) increase β slowly
        if it % 1500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.3)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that transfers tiny masses from “heavy”
    contributors to “light” indices.  The convolution is updated
    in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)
    cur_val = _score_from_selfconv(b, n)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling (fast early, slow later)
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour low heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – larger when temperature is high
        max_delta = min(a[donor] * 0.15, target_sum * 0.03)   # a bit larger than before
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ for donor removal / receiver addition
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ² terms)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = b_new.max()

        # Metropolis acceptance (always accept improving moves)
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # periodic re‑projection & full recomputation to avoid drift
        if it % 3000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def _gen_random_simplex(n: int, target_sum: float) -> np.ndarray:
    """Uniform Dirichlet‑like random point on the simplex."""
    rng = np.random.default_rng()
    rnd = rng.random(n)
    return _project_to_simplex(rnd, target_sum)


def _gen_geometric_seq(n: int, target_sum: float) -> np.ndarray:
    """Exponentially decaying sequence a_i ∝ r^i with random r ∈ (0.95,0.999)."""
    rng = np.random.default_rng()
    r = rng.uniform(0.95, 0.9995)
    seq = np.power(r, np.arange(n, dtype=np.float64))
    return _project_to_simplex(seq, target_sum)


def _gen_semicircle_seq(n: int, target_sum: float) -> np.ndarray:
    """Discrete semi‑circular shape: sqrt(1 - (2i/(n‑1)‑1)^2)."""
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1) - 1.0          # map to [-1, 1]
    seq = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return _project_to_simplex(seq, target_sum)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score
    within the given time budget (seconds).  Returns the best sequence
    found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size – larger n gives more freedom at the cost
    # of slower iterations.  The thresholds are tuned for the 1000 s
    # budget but can be adjusted.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)       # normalisation used by evaluate_sequence

    # --------------------------------------------------------------
    # Build a diverse set of initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # warm start from previous run, if available
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    # random Dirichlet points
    for _ in range(3):
        init_candidates.append(_gen_random_simplex(n, target_sum))

    # geometric decay shapes
    for _ in range(2):
        init_candidates.append(_gen_geometric_seq(n, target_sum))

    # semi‑circular shape (often a good deterministic start)
    init_candidates.append(_gen_semicircle_seq(n, target_sum))

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Main optimisation loop – one pass per candidate, respecting time
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx + 1} (n={n}) -----")

        # allocate a fraction of the remaining time for this start.
        remaining = deadline - time.time()
        # we keep at most 30 % for the first smooth phase,
        # the rest for annealing, and a final short smooth refinement.
        smooth_deadline = time.time() + 0.30 * remaining
        a_smooth, val_smooth = smooth_max_descent(
            init,
            target_sum,
            smooth_deadline,
            beta_start=0.1,
            beta_max=150.0,
            step0_factor=0.5,
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # annealing phase – consumes the rest of the budget for this start
        if time.time() < deadline:
            a_anneal, val_anneal = anneal_mass_transfer(a_smooth, target_sum, deadline)
            if val_anneal < best_score:
                best_score = val_anneal
                best_seq = a_anneal.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

        # optional second smooth refinement with a higher β (tightening)
        if time.time() < deadline:
            ref_deadline = time.time() + 0.10 * (deadline - time.time())
            a_refine, val_refine = smooth_max_descent(
                best_seq,
                target_sum,
                ref_deadline,
                beta_start=0.5,
                beta_max=200.0,
                step0_factor=0.3,
            )
            if val_refine < best_score:
                best_score = val_refine
                best_seq = a_refine.copy()
                print(f"[{time.time():.2f}] New global best after refinement: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        print(f"[{time.time():.2f}] No candidate found – falling back to simple power‑law.")
        best_seq = np.arange(1, n + 1, dtype=np.float64) ** (-0.55)
        best_seq = _project_to_simplex(best_seq, target_sum)

    # --------------------------------------------------------------
    # Final clean‑up, trimming trailing zeros and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)

    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (outside of the optimisation loop)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                      # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
