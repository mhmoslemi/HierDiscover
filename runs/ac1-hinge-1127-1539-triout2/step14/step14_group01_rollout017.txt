def _smooth_max_descent(a0, target_sum, deadline):
    a = a0.copy()
    n = len(a)
    # parameters
    max_iters = 0  # we'll use while time < deadline
    t = 0
    # hyperparameters
    alpha_start = 0.5
    alpha_end = 30.0
    lr_start = target_sum * 0.5   # step size relative to sum
    lr_end = target_sum * 0.001
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        alpha = alpha_start * (1 - frac) + alpha_end * frac # linear increase
        lr = lr_start * (1 - frac) + lr_end * frac

        # compute conv
        conv = _conv_fft(a)    # full conv vector length 2n-1
        maxc = conv.max()
        # compute softmax weights
        # to avoid overflow, subtract maxc.
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / len(conv)
        else:
            w = w / w_sum

        # gradient = 2 * cross_corr(w, a)
        grad = 2.0 * _cross_corr(w, a)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # project to nonnegative and simplex
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a > 0:
            a = a * (target_sum / sum_a)
        else:
            # reinitialize to uniform
            a = np.full_like(a, target_sum / n)

        # optionally print some info every few seconds
        if int(time.time() - start_time) % 30 == 0:
            cur_peak = conv.max()
            print(f"[{time.time():.2f}] smooth descent: alpha={alpha:.2f}, lr={lr:.5f}, peak={cur_peak:.6f}")

        # check for improvement; track best
        # Actually we can keep best seen
        # compute actual max using _max_convolution_fft (fast)
        cur_peak = _max_convolution_fft(a)
        if cur_peak < best_peak:
            best_peak = cur_peak
            best_a = a.copy()
    return best_a, best_peak

"""
Smooth-maximum gradient descent optimizer for the minimum peak convolution problem.
Starts from the previous best sequence (if available), then repeatedly minimizes
a smooth approximation of the max‑convolution using an Adam‑style update,
projecting onto the ℓ¹‑simplex after each step.  The temperature (soft‑max
sharpness) and learning rate are annealed.  The final sequence is optionally
polished by a short deterministic mass‑transfer pass and reordered for a final
peak reduction.
"""

import time
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out

def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = float(conv.max())

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def _gradient_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute 2 * sum_i w_i * a_{i - k} for all k (gradient of
    weighted sum of convolution entries).  The result is length len(a).
    """
    n = a.shape[0]
    # convolution of w (len 2n-1) with reversed a
    size = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a[::-1], n=size)
    conv_full = np.fft.irfft(fw * fa_rev, n=size)[: len(w) + n - 1]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad

def _smooth_max_descent(a0: np.ndarray,
                        target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style gradient descent on a smooth approximation of max{a*a}.
    The smooth max is defined via a soft‑max with temperature parameter α.
    α is annealed from a small to a larger value; the learning rate is also
    annealed.  The iterate is projected onto the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_peak = _max_convolution_fft(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # schedules
    alpha_start = 0.2
    alpha_end = 30.0
    lr_start = target_sum * 0.2
    lr_end = target_sum * 0.001

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)

        # exponential increase of soft‑max sharpness
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)
        # linear annealing of learning rate
        lr = lr_start * (1 - frac) + lr_end * frac

        # compute convolution and smooth weights
        conv = _conv_fft(a)          # full conv, length 2n-1
        maxc = conv.max()
        # stable soft‑max: max weight = 1, others <=1
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum

        # gradient of the smooth objective
        grad = _gradient_from_weights(w, a)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # occasional jitter to escape shallow plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter

        # keep non‑negative and renormalise onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # evaluate true peak (not the smooth surrogate)
        cur_peak = _max_convolution_fft(a)
        if cur_peak < best_peak:
            best_peak = cur_peak
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth descent improvement: peak={best_peak:.6f}")

    return best_a, best_peak

    else:

        pass

"""
Smooth‑max Adam descent optimiser for the “minimum peak convolution’’ problem.

The routine starts from the best sequence stored in the global variable
`height_sequence_1` (or a random simplex point if none exists).  It then
minimises a smooth approximation of the maximum entry of the auto‑convolution
`a*a` using an Adam‑style update.  The smoothness parameter (soft‑max sharpness)
and the learning rate are annealed, and after each step the iterate is projected
onto the non‑negative ℓ¹‑simplex with the required ℓ¹‑norm `√(2n)`.  A short
deterministic mass‑transfer polishing phase and a final reordering (alternating
and even‑spread) are applied, and the resulting sequence is returned.

The algorithm runs for the whole time budget (default 1000 s) and keeps track
of the best peak encountered.  It aims to improve the previous bound
≈ 1.5068 down towards the target ≈ 1.503.

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

    """Maximum entry of a*a using FFT."""
    n = a.shape[0]

    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]

    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:

    else:

        theta = cssv[rho] / (rho + 1)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]

            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis rule."""
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score

    """
    n = a.shape[0]
    size = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a[::-1], n=size)
    conv_full = np.fft.irfft(fw * fa_rev, n=size)[: len(w) + n - 1]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max Adam descent
# ----------------------------------------------------------------------


def _smooth_max_descent(a0: np.ndarray,
                        target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style descent on a smooth approximation of max{a*a}.
    The soft‑max sharpness α and the learning rate are annealed.
    The iterate is projected onto the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()

    best_a = a.copy()
    best_peak = _max_convolution_fft(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # schedules
    alpha_start = 0.2
    alpha_end = 30.0
    lr_start = target_sum * 0.2
    lr_end = target_sum * 0.001

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)

        # soft‑max sharpness (exponential increase)
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)
        # learning rate (linear decay)
        lr = lr_start * (1 - frac) + lr_end * frac

        # full auto‑convolution
        conv = _conv_fft(a)
        maxc = conv.max()
        # stable soft‑max weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum

        # gradient of the smooth surrogate
        grad = _gradient_from_weights(w, a)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter

        # projection onto the non‑negative simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # true peak (unsmoothed) for bookkeeping
        cur_peak = _max_convolution_fft(a)
        if cur_peak < best_peak:
            best_peak = cur_peak
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth descent improvement – peak={best_peak:.6f}")

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sequence reshaping helpers
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function to minimise the evaluation score.
    The optimiser uses a smooth‑max Adam descent, a brief deterministic
    mass‑transfer polish and a final reordering.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size – larger n gives a richer search space
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 600:
        n = 10000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # initialise from previous run (warm start) or random simplex point
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        a0 = _resize_sequence(prev, n)
        a0 = _project_to_simplex(a0, target_sum)
    else:
        a0 = rng.random(n)
        a0 = _project_to_simplex(a0, target_sum)

    # ------------------------------------------------------------------
    # Smooth‑max Adam descent (core optimisation)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 10.0   # keep a little time for polishing
    best_a, best_peak = _smooth_max_descent(a0, target_sum, smooth_deadline)
    print(f"[{time.time():.2f}] after smooth descent – peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Brief deterministic mass‑transfer polishing (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 2.0)
        seq_mt, peak_mt = _deterministic_mass_transfer(
            best_a, mt_dead, target_sum,
            start_temp=0.3, end_temp=1e-6, low_frac=0.07)
        if peak_mt < best_peak:
            best_a, best_peak = seq_mt, peak_mt
            print(f"[{time.time():.2f}] MT polishing – peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Final reordering (alternating / even‑spread)
    # ------------------------------------------------------------------
    candidates = [
        best_a,
        _alternating_order(best_a),
        _even_spread_order(best_a)
    ]
    best_a = min(candidates, key=_max_convolution_fft)
    best_peak = _max_convolution_fft(best_a)
    print(f"[{time.time():.2f}] after reordering – final peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=np.float64)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # make the sequence available for the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check via the provided evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
