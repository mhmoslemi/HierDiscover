def _hann_window(n):
    i = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n-1)))
    return w

def _hamming_window(n):
    i = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n-1))
    return w

def _blackman_window(n):
    i = np.arange(n)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = a0 - a1 * np.cos(2 * np.pi * i / (n-1)) + a2 * np.cos(4 * np.pi * i / (n-1))
    return w

def _sine_window(n):
    i = np.arange(n)
    w = np.sin(np.pi * (i + 0.5) / n)
    return w

def _triangular_window(n):
    i = np.arange(n)
    mid = (n-1)/2
    w = 1 - np.abs(i - mid) / (mid + 1)
    w = np.maximum(w, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) with a low max self‑convolution.

Algorithm (≈ 1000 s):
0️⃣ Choose the problem size (n≈12 k) and target L1 norm Σa = √(2 n).
1️⃣ Scan several parametric families (power‑law, tilted‑power‑law,
   Gaussian, Hann/Hamming/Blackman windows, sine, triangular) and keep the
   best.
2️⃣ Adam‑style projected gradient descent (soft‑max approximation of the
   max‑norm) with many random restarts.
3️⃣ Deterministic mass‑transfer (DMT2) – a greedy single‑donor / single‑receiver
   move that strictly lowers the convolution peak (time‑capped).
4️⃣ Random local search / simulated‑annealing style hill climbing that
   moves mass between random indices (time‑capped).
5️⃣ Final high‑β Adam polishing.
6️⃣ Trim trailing near‑zero entries and return the list.

All phases respect the global deadline (default 1000 s).  The code uses only
numpy, scipy (for convex refinement if desired) and the provided
evaluate_sequence() for the final check.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(b)/(sum(a)^2).  If Σa = √(2 n) the denominator = 2·n,
    therefore the score equals max(b).  We nevertheless compute the full
    expression to keep the function robust.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _hann_window(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    w *= target_sum / w.sum()
    return w


def _hamming_window(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    w *= target_sum / w.sum()
    return w


def _blackman_window(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1))
    w *= target_sum / w.sum()
    return w


def _sine_window(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n)
    w *= target_sum / w.sum()
    return w


def _triangular_window(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = 1.0 - np.abs(i - mid) / (mid + 1.0)
    w = np.maximum(w, 0.0)
    w *= target_sum / w.sum()
    return w


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of the max‑norm."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple Adam‑GD restarts with jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy single‑donor / single‑receiver moves that lower the peak."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner weights a_{k-i}
        partner = a[k - idx]

        # donor: index with largest a_i * partner_i
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver: index with minimal partner weight (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random local hill‑climbing search
# ----------------------------------------------------------------------


def _random_local_search(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_move_frac: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill climbing:
    repeatedly move a random fraction of mass from a random donor to a
    random receiver and keep the move if it improves the score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        donor = rng.integers(n)
        if a[donor] <= 0.0:
            continue
        receiver = rng.integers(n)
        if receiver == donor:
            continue
        # fraction of donor's mass to move (uniform in [0, max_move_frac])
        frac = rng.random() * max_move_frac
        delta = frac * a[donor]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # keep sum exact (project back to simplex to avoid tiny drift)
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_score:
            best_score = val
            best_a = a_new.copy()
            a = a_new
            print(f"[{time.time():.2f}] Random hillclimb improved score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Utility – trim trailing near‑zero entries
# ----------------------------------------------------------------------


def _trim_sequence(a: np.ndarray, tol: float = 1e-12) -> list[float]:
    nz = np.where(a > tol)[0]
    if nz.size == 0:
        return [0.0]
    return a[: nz[-1] + 1].tolist()

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------- problem size -------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) → score = max convolution

    # ------------------- initial warm‑start -------------------
    rng = np.random.default_rng()
    best_seq = _power_law_sequence(n, 0.5, target_sum)  # fallback
    best_score = _score(best_seq)

    # ----- scan simple parametric families (including windows) -----
    # power‑law grid
    for p in np.linspace(0.20, 1.2, 41):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # exp‑tilted power‑law grid
    for p in np.linspace(0.20, 1.0, 21):
        for q in np.linspace(0.0, 10.0, 21):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # Gaussian family
    for sigma in np.linspace(0.05 * n, 0.6 * n, 9):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # Window families
    for win_func in (_hann_window, _hamming_window, _blackman_window,
                     _sine_window, _triangular_window):
        seq = win_func(n, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # Use a previously stored construction if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # Small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] After parametric search best score = {best_score:.6f}")

    # ------------------- Phase A: Adam GD with restarts -------------------
    # Allocate ~30% of remaining time to this phase
    remaining = deadline - time.time()
    if remaining > 20.0:
        gd_deadline = time.time() + 0.30 * remaining
        while time.time() < gd_deadline - 5.0:
            jitter = rng.normal(scale=target_sum * 0.015, size=n)
            start_vec = best_seq + jitter
            start_vec = np.maximum(start_vec, 0.0)
            start_vec = _project_to_simplex(start_vec, target_sum)

            a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                             target_sum,
                                                             gd_deadline)
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] Adam‑GD improved best = {best_score:.6f}")

    # ------------------- Phase B: Deterministic Mass Transfer (time‑capped) -------------------
    # Give it at most ~20% of the total budget
    remaining = deadline - time.time()
    if remaining > 30.0:
        dmt_deadline = time.time() + min(180.0, 0.20 * remaining)
        a_dmt, val_dmt = _deterministic_mass_transfer_exact2(best_seq,
                                                             dmt_deadline,
                                                             target_sum)
        if val_dmt < best_score:
            best_score, best_seq = val_dmt, a_dmt.copy()
            print(f"[{time.time():.2f}] DMT2 improved best = {best_score:.6f}")

    # ------------------- Phase C: Random Local Hill‑Climbing -------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        rl_deadline = time.time() + min(200.0, 0.25 * remaining)
        a_rl, val_rl = _random_local_search(best_seq,

        if val_rl < best_score:
            best_score, best_seq = val_rl, a_rl.copy()
            print(f"[{time.time():.2f}] Random hillclimb improved best = {best_score:.6f}")

    # ------------------- Phase D: Final high‑β Adam polishing -------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,

        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polishing best = {best_score:.6f}")

    # ------------------- Post‑processing -------------------
    final_seq = _trim_sequence(best_seq)

    # Final sanity check (optional, does not affect the returned value)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
