"""
Propose a sequence of non‑negative heights that minimises
`evaluate_sequence`.  The search consists of two stages:

1.  A broad parametric sweep over several families of probability
    distributions (power‑law, geometric, symmetric product‑power and
    Beta‑U‑shaped / arcsine).  For each family we try a modest grid of
    parameters and keep the best candidate found (the vectors are
    normalised to sum to 1 – the objective is scale‑invariant).

2.  A smooth gradient‑descent refinement.  The hard ℓ∞‑norm of the
    auto‑convolution is replaced by a soft‑max with a temperature
    parameter that is slowly increased.  The gradient of the soft‑max
    is obtained by a correlation (`np.correlate`).  After each step the
    iterate is projected back onto the probability simplex.  Random
    jitter is injected occasionally to escape plateaus.

The routine respects a 1000 s time budget and returns the best
sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i >= 0, sum x_i = s }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # If already on the simplex just return a copy
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper that feeds a list to the externally provided function."""
    return evaluate_sequence(seq.tolist())

def generate_powerlaw(N: int, p: float, sigma: float) -> np.ndarray:
    """
    a_i = (i + sigma)^{-p}, i = 0,…,N‑1  (no truncation – length fixed).
    Returned vector is normalised to sum 1.
    """
    i = np.arange(N, dtype=np.float64)
    a = (i + sigma) ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(N, 1.0 / N)
    else:
        a = a / a_sum
    return a


def generate_geometric(N: int, r: float) -> np.ndarray:
    """
    a_i = r^{i}, i = 0,…,N‑1, normalised to sum 1.
    """
    i = np.arange(N, dtype=np.float64)
    a = r ** i
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(N, 1.0 / N)
    else:
        a = a / a_sum
    return a


def generate_symmetric_product(N: int, gamma: float, delta: float = 0.5) -> np.ndarray:
    """
    Symmetric U‑shaped family:
        a_i = 1 / ((i+delta)*(N-1-i+delta))^{gamma}
    Normalised to sum 1.
    """
    i = np.arange(N, dtype=np.float64) + delta
    j = (N - 1 - np.arange(N, dtype=np.float64)) + delta
    # avoid overflow – clip the exponentiated denominator
    denom = (i * j) ** gamma
    # set tiny denominator to a large number to avoid infinities
    denom = np.where(denom == 0, np.inf, denom)
    a = 1.0 / denom
    a_sum = a.sum()
    if a_sum == 0 or not np.isfinite(a_sum):
        a = np.full(N, 1.0 / N)
    else:
        a = a / a_sum
    return a


def generate_beta_u(N: int, alpha: float, delta: float = 0.5) -> np.ndarray:
    """
    Discretisation of the Beta(α,α) density on the interval (0,1).
    For α = 0.5 this is the arcsine law (U‑shaped).
    Parameters:
        N      – length of the vector
        alpha  – shape parameter (0 < α < 1 gives U‑shape)
        delta  – tiny shift to avoid exact 0/1.
    The vector is normalised to sum 1.
    """
    i = np.arange(N, dtype=np.float64) + delta
    x = i / (N + 2 * delta)               # x ∈ (0,1)
    # density ∝ x^{α-1} (1-x)^{α-1}
    # Guard against overflow when α is small → exponent very negative
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # Replace NaNs/infs with 0 (they arise only at the extreme ends)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(N, 1.0 / N)
    else:
        a = a / a_sum
    return a

def softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Return a probability vector w approximating the ℓ∞‑max of `conv`
    via a soft‑max with temperature parameter.
    Higher `temperature` makes the distribution concentrate on the
    maximum entry.
    """
    # shift for numerical stability (conv.max() becomes 0)
    shift = conv - conv.max()
    # exponentiated and normalised
    exp_term = np.exp(temperature * shift)
    total = exp_term.sum()
    if total == 0 or not np.isfinite(total):
        # fall back to uniform distribution
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / total


def gradient_step(x: np.ndarray, lr: float, temperature: float) -> np.ndarray:
    """
    One iteration of gradient descent on the smooth ℓ∞‑approximation.
    `x` is a probability vector (sum = 1, non‑negative).
    Returns the projected next iterate.
    """
    # auto‑convolution
    conv = np.convolve(x, x)
    # soft‑max weights approximating the max
    w = softmax_weights(conv, temperature)

    # gradient of the soft‑max value:  g = 2 * correlation(w, x)
    # Correlation at lag `p` = sum_i w_i * x_{i-p}
    corr = np.correlate(w, x, mode='full')
    n = x.shape[0]
    grad = 2.0 * corr[n - 1: n - 1 + n]

    # gradient descent step + simplex projection
    y = x - lr * grad
    # project back onto the simplex (sum = 1, >=0)
    y = project_onto_simplex(y, s=1.0)
    return y

def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_score = float('inf')
    best_seq = None

    # ---------- Phase 1 : parametric sweep ----------
    print("=== Phase 1 : parametric families ===")
    # length candidates (moderate, up to about 4000)
    length_candidates = [256, 512, 1024, 2048, 3000, 4096]

    # power‑law grid
    p_vals = np.linspace(0.35, 0.65, 7)          # exponent around 0.5
    sigma_vals = np.logspace(-3, 0.5, 9)        # sigma ∈ [1e‑3, 3.16]

    # geometric grid
    r_vals = np.linspace(0.55, 0.95, 9)         # ratio 0.55 … 0.95

    # symmetric product‑power grid
    gamma_vals = np.linspace(0.2, 0.9, 8)       # power of the inverse product
    delta_sp = 0.5                               # offset to avoid zero

    # Beta‑U‑shaped (arcsine) grid
    alpha_vals = np.linspace(0.15, 0.9, 8)       # α = 0.5 gives the classic arcsine
    delta_beta = 0.5

    # Helper to evaluate and possibly update best
    def try_candidate(seq: np.ndarray, descr: str):
        nonlocal best_score, best_seq
        if time.time() >= deadline:
            return
        try:
            sc = evaluate(seq)
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = seq.tolist()
            print(f"new best | {descr} | len={len(seq)} | score={best_score:.6f}")

    # Iterate over families
    for N in length_candidates:
        # ---- Power law ----
        for p in p_vals:
            for sigma in sigma_vals:
                if time.time() >= deadline:
                    break
                seq = generate_powerlaw(N, p, sigma)
                try_candidate(seq, f"powerlaw(p={p:.3f},σ={sigma:.3e})")
        if time.time() >= deadline:
            break

        # ---- Geometric ----
        for r in r_vals:
            if time.time() >= deadline:
                break
            seq = generate_geometric(N, r)
            try_candidate(seq, f"geom(r={r:.3f})")

        # ---- Symmetric product power ----
        for gamma in gamma_vals:
            if time.time() >= deadline:
                break
            seq = generate_symmetric_product(N, gamma, delta=delta_sp)
            try_candidate(seq, f"sprod(gamma={gamma:.3f})")

        # ---- Beta‑U‑shaped / arcsine ----
        for alpha in alpha_vals:
            if time.time() >= deadline:
                break
            seq = generate_beta_u(N, alpha, delta=delta_beta)
            try_candidate(seq, f"beta(α={alpha:.3f})")

        if best_score < 1.45:      # early exit if we are already very good
            break

    # If nothing succeeded (unlikely) fall back to a uniform vector
    if best_seq is None:
        N0 = 1024
        best_seq = [1.0 / N0] * N0
        best_score = evaluate(np.array(best_seq))
        print("fallback: uniform sequence, score =", best_score)

    # ---------- Phase 2 : smooth gradient descent ----------
    print("\n=== Phase 2 : smooth gradient descent refinement ===")
    x = np.array(best_seq, dtype=np.float64)
    # Ensure we are on the simplex (the objective is scale‑invariant)
    x = x / x.sum()
    it = 0
    last_improve_it = 0
    while time.time() < deadline:
        it += 1
        # Step‑size schedule (decrease slowly)
        lr = 0.5 / (1.0 + it ** 0.6)

        # Temperature schedule: start smooth, end sharp
        temperature = min(200.0, 5.0 + it * 0.015)

        # Gradient step
        x = gradient_step(x, lr, temperature)

        # Occasional random jitter to escape plateaus
        if np.random.rand() < 0.02:
            idx = np.random.randint(len(x))
            delta = (np.random.rand() - 0.5) * 0.04 * x[idx]
            x[idx] = max(0.0, x[idx] + delta)
            x = project_onto_simplex(x, s=1.0)

        # Evaluate
        try:
            cur_score = evaluate(x)
        except Exception:
            continue

        if cur_score < best_score:
            best_score = float(cur_score)
            best_seq = x.tolist()
            last_improve_it = it
            print(f"iter {it:5d} -> improved score = {best_score:.6f}")
            if best_score < 1.503:
                print("Target reached – stopping early.")
                break

        # If we have gone many iterations without improvement, inject a bigger
        # random restart around the current point (helps with very flat landscapes)
        if it - last_improve_it > 20000:
            # perturb a handful of coordinates
            for _ in range(5):
                i = np.random.randint(len(x))
                x[i] = max(0.0, x[i] + np.random.randn() * 0.01)
            x = project_onto_simplex(x, s=1.0)
            last_improve_it = it

    print("final best score:", best_score, "length:", len(best_seq))
    return best_seq
