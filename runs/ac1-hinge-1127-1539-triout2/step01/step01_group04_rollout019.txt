            v_hat = v / (1 - beta2 ** (iter_count + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step

            # projection: non-negativity, then scale to target sum
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, target_sum / n)
            else:
                a *= target_sum / s

            # increase gamma gradually until max
            gamma = min(gamma * gamma_inc, max_gamma)

            # jitter occasionally
            if iter_count % 200 == 0 and iter_count > 0:
                sigma = target_sum * 1e-3 / math.sqrt(n)
                a += np.random.randn(n) * sigma
                a = np.maximum(a, 0.0)
                a *= target_sum / a.sum()

            iter_count += 1

        # end while per n

    # after loops, ensure best_seq as list
    return best_seq.tolist()

"""
Linear‑programming (Frank‑Wolfe) optimisation of a non‑negative step
function.

The routine starts from the classic √‑difference sequence (a_i = √(i+1) –
√i) normalised so that Σ a_i = √(2·n).  For the current sequence `f`
the value M = max(f∗f) is an upper bound on the evaluation score
(because the score = max_b when the ℓ₁‑norm is fixed).

A linear programme is solved:

    maximise   Σ g_j
    subject to g_j ≥ 0  and  (f ∗ g)_k ≤ M   for all k = 0 … 2n‑2.

The optimum `g` carries as much ℓ₁‑mass as possible while respecting the
same convolution bound.  After rescaling `g` to the required ℓ₁‑norm it
is blended with `f` (f ← (1‑t)·f + t·g).  Repeating this step monotonically
decreases M.  When the time budget is almost exhausted a short projected‑
sub‑gradient descent (the same scheme used in the reference code) is run
as a polishing phase.

All heavy work (convolutions, gradient) uses FFTs, and the linear
programme is solved with `scipy.optimize.linprog` (method='highs') which
handles the dense constraint matrix efficiently for n up to a few
thousand.  The best sequence found before the time limit is returned.
"""

import time
import math
import numpy as np

# Optional imports – if SciPy is unavailable we fall back to a pure
# gradient method (the previous approach).
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution of a real vector using an FFT length L.
    Returns the first (2·len(a)‑1) entries.
    """
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)
    return conv_full[: 2 * a.size - 1]


def _conv_fft_full(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Convolution of two real vectors using FFT of length L (full mode)."""
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv_full = np.fft.irfft(fx * fy, n=L)
    return conv_full[: x.size + y.size - 1]


def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """Add the sub‑gradient contribution of shift k to `grad`."""
    n = a.size
    if k < n:
        # indices 0 … k
        grad[: k + 1] += 2.0 * a[k :: -1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]


def _max_convolution(a: np.ndarray, L_fft: int) -> float:
    """Return the maximum entry of the self‑convolution of a."""
    conv = _conv_fft(a, L_fft)
    return float(conv.max())


def _lp_improvement_step(f: np.ndarray, M: float, target_sum: float,
                         rows: np.ndarray, cols: np.ndarray,
                         L_self: int, t: float = 0.2) -> np.ndarray:
    """
    Perform one Frank‑Wolfe style LP step:
    - maximise ℓ₁‑norm of g under (f ∗ g) ≤ M,
    - rescale g to the required ℓ₁‑norm,
    - blend with f using step size t.
    Returns the updated f (or the original f if the LP fails).
    """
    n = f.size
    # build the data vector for the convolution matrix (same sparsity pattern each time)
    data = np.tile(f, n)                     # shape (n*n,)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # linear programme: maximise sum(g)  ⇔ minimise –sum(g)
    c = -np.ones(n, dtype=np.float64)        # objective vector
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # bounds g_j ≥ 0
    bounds = [(0, None)] * n

    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        # LP failed – fall back to the original f
        return f

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    # scale g to the required ℓ₁‑norm
    g *= (target_sum / sum_g)

    # convex blend
    f_new = (1.0 - t) * f + t * g

    # numerical projection onto the feasible set (non‑negative & correct ℓ₁‑norm)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s == 0.0:
        f_new = np.full_like(f_new, target_sum / n)
    else:
        f_new *= (target_sum / s)
    return f_new
