def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # Determine length
    try:
        n = len(height_sequence_1)
        # Use that
    except NameError:
        n = 4000

    # Setup parameters
    S = math.sqrt(2.0 * n)
    L = 1 << ((3*n - 2).bit_length()) # FFT length for gradient

    # Pre-alloc arrays for Adam
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Initialize a
    # Option: if height_sequence_1 defined, use it scaled to sum S; else sqrt-diff.
    try:
        a0 = np.array(height_sequence_1, dtype=np.float64)
        # clip large numbers? also all >0? We'll ensure sum = S by scaling.
        a0 = np.maximum(a0, 0.0)
        a0 = a0 * (S / a0.sum())
    except NameError:
        idx = np.arange(n, dtype=np.float64)
        a0 = np.sqrt(idx + 1.0) - np.sqrt(idx)
        a0 = a0 * (S / a0.sum())
    a = a0.copy()

    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Tracking training iteration.
    iteration = 0
    # schedule lambda
    lam = 1.0
    lam_max = 60.0
    lam_inc_iter = 2000

    while time.time() < deadline:
        # Compute FFT of a once
        FA = np.fft.rfft(a, n=L)

        # Compute self-convolution
        conv = np.fft.irfft(FA * FA, n=L)[: 2*n - 1]

        # Softmax over conv
        # update lambda gradually:
        if iteration % lam_inc_iter == 0 and lam < lam_max:
            lam = min(lam_max, lam * 1.5)

        # compute weights w
        # stable softmax
        maxc = conv.max()
        exps = np.exp(lam * (conv - maxc))
        sumexps = exps.sum()
        w = exps / sumexps  # sum to 1

        # Compute weighted gradient via correlation
        w_rev = w[::-1]
        FW_rev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(FA * FW_rev, n=L)
        # gradient vector:
        grad = 2.0 * corr[n-1:2*n-1][::-1]

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)

        lr = 0.1 / (np.sqrt(iteration) + 1)   # decaying factor maybe not needed due to Adam.

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project onto feasible: nonnegativity and sum fixed.
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, S / n)
        else:
            a *= (S / s)

        # Evaluate candidate periodically
        if iteration % 50 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Iter {iteration:6d} – new best score {best_score:.6f}")

        # occasional jitter
        if iteration % 200 == 0:
            sigma = S * 1e-3 / np.sqrt(n)
            a += np.random.randn(n) * sigma
            a = np.maximum(a, 0.0)
            a *= (S / a.sum())

        # possible restart after stagnation
        if iteration % 5000 == 0:
            # restart from best with small random perturb
            a = best_seq.copy()
            a += np.random.randn(n) * sigma
            a = np.maximum(a, 0.0)
            a *= (S / a.sum())
            m[:] = 0.0
            v[:] = 0.0

    # final eval
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Finished; best score {best_score:.6f}, final {final_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑function sequence that minimizes the evaluation function.

The algorithm is a projected Adam optimiser on a smooth approximation of the
∞‑norm of the self‑convolution.  The loss is the smooth‑max
   L(a) = log‑sum‑exp( λ·(a*a) ) / λ ,
which converges to the true maximum when λ grows.  The gradient of L is a
weighted sum of the sub‑gradients for each shift and can be computed by a
single FFT‑based correlation, avoiding the O(n²) cost of a naïve
convolution.  After each gradient step we project onto the feasible set
(non‑negative entries and a fixed L¹‑norm √(2n)), add small random jitter
occasionally, and increase λ gradually (annealing).  The best sequence seen
throughout the run is returned.

The implementation stays well within the 1000 s budget even for a few
ten‑thousand coefficients.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution of a real vector `a` using FFT length `L`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * a.size - 1]


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_k ∝ exp(λ·conv_k) that form the
    gradient of the smooth‑max loss.
    """
    if lam <= 0.0:
        return np.full_like(conv, 1.0 / conv.size, dtype=np.float64)

    # Stable formulation: subtract the maximum before exponentiation.
    cmax = conv.max()
    exps = np.exp(lam * (conv - cmax))
    s = exps.sum()
    if s == 0.0:                     # numerical safeguard
        return np.full_like(conv, 1.0 / conv.size, dtype=np.float64)
    return exps / s


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max loss with respect to `a`.
    It equals 2·(a ⋆ w) where ⋆ denotes correlation.
    Implemented via FFT convolution with the reversed weight vector.
    """
    n = a.size
    # reverse w (correlation ↔ convolution)
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    corr_full = np.fft.irfft(fa * fw, n=L)

    # Extract the slice that corresponds to lags 0 … n‑1 and reverse it.
    # slice = corr_full[n-1 : 2*n-1]  (length n)
    grad = 2.0 * corr_full[n - 1 : 2 * n - 1][::-1]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs for up to `budget_s` seconds
    (leaving a small safety margin) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ---- Choose a length ------------------------------------------------
    try:
        # Use the previously discovered construction if available.
        n = len(height_sequence_1)          # type: ignore[name-defined]
        init = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        # No earlier result – pick a moderately large size.
        n = 8000
        init = None

    # ---- Normalisation --------------------------------------------------
    target_sum = math.sqrt(2.0 * n)           # L¹‑norm for which the objective == max(conv)

    # ---- Initial point ---------------------------------------------------
    if init is not None:
        a = np.maximum(init, 0.0)
        a = a * (target_sum / a.sum())
    else:
        idx = np.arange(n, dtype=np.float64)
        a = np.sqrt(idx + 1.0) - np.sqrt(idx)      # sqrt‑difference shape
        a = a * (target_sum / a.sum())

    # ---- Adam state ------------------------------------------------------
    m = np.zeros(n, dtype=np.float64)   # first moment
    v = np.zeros(n, dtype=np.float64)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # ---- FFT size (needs to accommodate correlation length 3n‑2) -----
    L = _next_pow_two(3 * n - 2)

    # ---- Tracking best solution -----------------------------------------
    best_seq = a.copy()
    # After normalisation the objective equals the max of the self‑convolution.
    conv = _self_convolution_fft(a, L)
    best_score = float(conv.max())
    print(f"Initial max‑conv (objective) = {best_score:.6f}")

    # ---- Hyper‑parameters ------------------------------------------------
    lam = 0.5                 # start with a gentle smooth‑max
    lam_max = 60.0
    lam_step = 500            # increase λ every `lam_step` iterations
    lam_factor = 1.18

    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        # ---- Full self‑convolution (needed for loss & max) ---------------
        conv = _self_convolution_fft(a, L)

        # ---- Smooth‑max weights -----------------------------------------
        w = _softmax_weights(conv, lam)

        # ---- Gradient of smooth‑max --------------------------------------
        grad = _gradient_from_weights(a, w, L)

        # ---- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # A modest base learning‑rate; Adam already adapts per‑coordinate.
        lr = 0.15 / (math.sqrt(iteration) + 1.0)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- Projection onto the feasible set -----------------------------
        a = np.maximum(a, 0.0)                     # non‑negative
        s = a.sum()
        if s == 0.0:
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a *= (target_sum / s)                  # fix L¹‑norm

        # ---- Anneal λ (smooth‑max becomes sharper) -----------------------
        if iteration % lam_step == 0 and lam < lam_max:
            lam = min(lam_max, lam * lam_factor)

        # ---- Occasionally add tiny random jitter -------------------------
        if iteration % 200 == 0:
            sigma = target_sum * 1e-3 / math.sqrt(n)
            a += np.random.randn(n) * sigma
            a = np.maximum(a, 0.0)
            a *= (target_sum / a.sum())

        # ---- Track the best sequence discovered ---------------------------
        cur_score = float(conv.max())   # equals objective for our normalisation
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = iteration
            print(f"Iter {iteration:6d} – new best = {best_score:.6f}")

        # ---- Simple stagnation handling: restart from the best ------------
        if iteration % 5000 == 0 and iteration - last_improve_iter > 2000:
            # Re‑initialise moments and perturb the best known solution.
            a = best_seq.copy()
            sigma = target_sum * 1e-3 / math.sqrt(n)
            a += np.random.randn(n) * sigma
            a = np.maximum(a, 0.0)
            a *= (target_sum / a.sum())
            m.fill(0.0)
            v.fill(0.0)
            lam = 0.5
            last_improve_iter = iteration

    # ----------------------------------------------------------------------
    # Final reporting
    # ----------------------------------------------------------------------
    final_conv = _self_convolution_fft(best_seq, L)
    final_score = float(final_conv.max())
    print(f"Finished after {iteration:,} iterations – best max‑conv = {best_score:.6f}")
    print(f"Final (re‑computed) max‑conv = {final_score:.6f}")

    # Return as a plain Python list (the surrounding harness will call
    # `evaluate_sequence` on it).
    return best_seq.tolist()
