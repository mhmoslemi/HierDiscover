"""
Linear‐programming based optimizer for the step‑function bound.

The evaluation function (after normalising the ℓ1‑norm to √(2·n)) equals
the peak of the self‑convolution.  The algorithm iteratively reduces this
peak:

1. Start from a power‑law profile a_i ∝ (i+1)^{-p} (p≈0.5) and normalise
   Σa = √(2·n).

2. Let M = max(a ⋆ a).  Solve the linear programme

        maximise   Σ b_j
        subject to  (a ⋆ b)_k ≤ M   for every convolution index k
                    b_j ≥ 0

   The matrix of the constraints is the Toeplitz matrix of a.  The LP
   yields a vector b with a larger ℓ1‑norm than a while keeping the mixed
   convolution a⋆b bounded by the current peak M.

3. Scale b to the required norm (α = √(2·n)/Σb) producing g = α·b.
   Because a⋆g ≤ α·M, the peak is strictly reduced unless b already has
   the maximal possible norm.

4. Replace the current profile by a convex combination
        a ← (1‑t)·a + t·g            (t≈0.5)
   and repeat the process until the time budget is exhausted or no
   further progress is observed.

The LP is solved with SciPy’s HiGHS solver using a sparse Toeplitz matrix.
The whole search tries a few sequence lengths and exponents, keeping the
best sequence found.  The final vector is returned as a plain Python list.
"""

import time
import math

# ----------------------------------------------------------------------

    SCIPY_AVAILABLE = False

def _max_convolution(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution a⋆a.
    The convolution is performed with an FFT of length power‑of‑two.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _initial_sequence(p: float, n: int) -> np.ndarray:
    """
    Power‑law start: a_i ∝ (i+1)^{-p}, normalised to Σa = √(2·n).
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    total = np.sqrt(2.0 * n)
    a *= total / a.sum()
    return a


def _build_toeplitz_matrix(f: np.ndarray) -> coo_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix A with
        A_{k,j} = f_{k‑j}   if 0 ≤ k‑j < n else 0 .
    The matrix encodes the linear constraints (f ⋆ b)_k.
    """
    n = f.shape[0]
    i = np.arange(n, dtype=np.int32)[:, None]          # shape (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]          # shape (1,n)

    rows = (i + j).ravel()             # 0 … 2n‑2
    cols = np.broadcast_to(j, (n, n)).ravel()
    data = np.broadcast_to(f[:, None], (n, n)).ravel()

    return coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)


def _lp_step(f: np.ndarray, M: float, total_norm: float):
    """
    Solve the LP described in the module docstring.
    Returns (g, sum_b) where g is the scaled vector (already normalised)
    and sum_b = Σb before scaling.  If the LP fails, (None, None) is returned.
    """
    n = f.size
    # Build the Toeplitz matrix for the constraints f⋆b ≤ M.
    A = _build_toeplitz_matrix(f).tocsr()

    # Upper bound for every row
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    # Objective: maximise Σb  → minimise -Σb
    c = -np.ones(n, dtype=np.float64)

    # Solve with HiGHS (fast sparse LP solver)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None, None

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:                     # safeguard – should not happen
        return None, None

    alpha = total_norm / sum_b
    g = b * alpha
    return g, sum_b


def _optimize_for_length(n: int, deadline: float):
    """
    Run the LP‑based optimisation for a fixed length n.
    Several power‑law exponents are tried; the best sequence (smallest
    max convolution) encountered before the deadline is returned.
    """
    total_norm = math.sqrt(2.0 * n)
    best_seq = None
    best_peak = np.inf

    # A handful of exponents around the known ½‑law.
    exponents = [0.45, 0.48, 0.5, 0.52, 0.55]

    for p in exponents:
        if time.time() >= deadline:
            break

        # ----- initialise -------------------------------------------------
        a = _initial_sequence(p, n)
        M = _max_convolution(a)

        if M < best_peak:
            best_peak = M
            best_seq = a.copy()
            print(f"[{time.time():.1f}] start n={n} p={p:.3f}  peak={M:.6f}")

        # ----- iterate ----------------------------------------------------
        # The number of LP iterations is kept modest to stay well within the
        # global time budget.
        max_iter = 12
        t = 0.5                     # convex‑combination weight

        for it in range(max_iter):
            if time.time() >= deadline:
                break

            g, sum_b = _lp_step(a, M, total_norm)
            if g is None:
                # LP failed – abort further iterations for this start.
                break

            # Convex update
            a_new = (1.0 - t) * a + t * g
            M_new = _max_convolution(a_new)

            # Record improvement
            if M_new < best_peak:
                best_peak = M_new
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] n={n} p={p:.3f} it={it}  peak={M_new:.6f}")

            # Stop if progress stalls
            if M_new >= M * 0.9999:
                break

            a = a_new
            M = M_new

    return best_seq, best_peak

            if time.time() >= deadline:
                break
            for p in [0.45, 0.48, 0.5, 0.52, 0.55]:
                if time.time() >= deadline:
                    break
                a = _initial_sequence(p, n)
                cur_score = _score(a)
                if cur_score < best_peak_overall:
                    best_peak_overall = cur_score
                    best_overall = a.copy()
                sub_deadline = min(deadline, time.time() + 8.0)
                a_opt, opt_score = _local_search(a, sub_deadline)
                if opt_score < best_peak_overall:
                    best_peak_overall = opt_score
                    best_overall = a_opt.copy()

    # ------------------------------------------------------------------
    # Safety net – should never happen.
    # ------------------------------------------------------------------
    if best_overall is None:
        n = 1000
        best_overall = np.ones(n, dtype=np.float64)
        best_overall *= math.sqrt(2.0 * n) / best_overall.sum()

    # Trim trailing (near‑zero) entries – evaluate_sequence ignores them.
    nz = np.where(best_overall > 1e-12)[0]
    if nz.size:
        best_overall = best_overall[: nz[-1] + 1]

    # Return a plain Python list as required by the harness.
    return best_overall.tolist()
