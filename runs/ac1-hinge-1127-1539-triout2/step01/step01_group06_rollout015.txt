"""
Advanced search for a step‑function that minimizes `evaluate_sequence`.

The algorithm mixes three ideas:

1. **Parametric seeds** – a handful of analytic families
   (power law, shifted power law, inverse‑sqrt, √‑differences and exponential)
   with randomly sampled parameters.  Each seed is normalised so that
   `∑a_i = sqrt(2·n)`, which makes the evaluation equal to the maximum of the
   auto‑convolution.

2. **Linear‑programming refinement** – given a current vector `f`,
   we solve a small LP that maximises the ℓ₁‑norm of a new vector `g`
   under the linear constraints `(f * g)_k ≤ M` where `M = max(f*f)`.
   The solution is rescaled and mixed with `f`.  The LP is tiny
   (≤ 1800 variables) and solved with SciPy/HiGHS.

3. **Stochastic annealed hill‑climbing** – we repeatedly move a random
   fraction of mass from one index to another while preserving the total
   sum.  Moves that improve the objective are always kept; occasionally a
   worsening move is accepted with a temperature that decays linearly.

The search runs until the allotted time expires (≈ 1000 s) and returns the
best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the LP refinement.
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    import scipy.sparse as sp

    _HAS_SCIPY = True
except Exception:  # pragma: no cover – SciPy may be unavailable.
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Evaluation of a normalised vector – equal to max(a*a)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    return _max_convolution(a)


def _normalize(a: np.ndarray) -> np.ndarray:
    """Scale a to have ℓ₁‑norm sqrt(2·n) (n = len(a))."""
    n = a.size
    target = math.sqrt(2.0 * n)
    s = a.sum()
    if s == 0:
        return a
    a = a * (target / s)
    return a


# ----------------------------------------------------------------------
# Analytic families of seed sequences
# ----------------------------------------------------------------------


def _shape_power(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return _normalize(a)


def _shape_power_shift(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64) + shift
    a = i ** (-p)
    return _normalize(a)


def _shape_inv_sqrt(n: int) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-0.5))
    return _normalize(a)


def _shape_sqrt_diff(n: int) -> np.ndarray:
    i = np.arange(0, n + 1, dtype=np.float64)
    diffs = np.sqrt(i[1:]) - np.sqrt(i[:-1])
    return _normalize(diffs)


def _shape_exp(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    return _normalize(a)


# ----------------------------------------------------------------------
# Linear‑programming refinement (only when SciPy is available)
# ----------------------------------------------------------------------


def _build_toeplitz_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that A @ g == f * g (full convolution).
    The matrix has constant diagonals equal to the entries of f.
    """
    n = f.size
    # each diagonal offset d (0 ≤ d < n) contains the constant f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)               # 0,1,…,n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def lp_update(f: np.ndarray, M: float, t: float = 0.5) -> np.ndarray:
    """
    One LP‑based refinement step.

    * maximise   sum(g)
    * subject to   (f * g)_k ≤ M   for every k
                  g ≥ 0

    The optimiser (HiGHS) works on the sparse Toeplitz matrix.
    The resulting vector is scaled to the required ℓ₁‑norm and mixed
    with the old vector using the weight `t`.
    """
    if not _HAS_SCIPY:
        return f

    n = f.size
    A = _build_toeplitz_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    # linprog minimises c·x, so we use -1 to maximise sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success:
        # fallback – return the unmodified vector
        return f

    g = res.x
    # Rescale to the mandated ℓ₁‑norm
    target = math.sqrt(2.0 * n)
    s = g.sum()
    if s > 0:
        g = g * (target / s)

    # Convex combination with the previous iterate
    new_f = (1.0 - t) * f + t * g
    new_f = np.maximum(new_f, 0.0)
    new_f = _normalize(new_f)
    return new_f


# ----------------------------------------------------------------------
# Stochastic annealed local search
# ----------------------------------------------------------------------


def local_search_anneal(a: np.ndarray,
                        deadline: float,
                        max_frac: float = 0.12,
                        temp_start: float = 1.0,
                        temp_end: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill climbing.
    Randomly moves a fraction of mass from i → j while preserving the total sum.
    A move is always accepted if it improves the score; otherwise it is accepted
    with probability `exp((old‑new)/T)`, where the temperature `T` decreases
    linearly from `temp_start` to `temp_end` during the run.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        # linear cooling schedule
        T = max(temp_end, temp_start * (1.0 - elapsed / total))

        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * max_frac
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] anneal improved to {best_val:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the challenge harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.

    The routine repeatedly draws a random analytic seed (power law,
    shifted power law, √‑differences, inverse‑sqrt or exponential), optionally
    refines it with a tiny LP, and then performs a short annealed hill‑climb.
    The best vector found before the time budget expires is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 2.0                 # small safety margin

    best_seq = None
    best_val = np.inf

    # admissible lengths – longer vectors give a smaller factor `2·n` in the
    # original formula, but also cost more per iteration.
    lengths = [800, 1200, 1800, 2500, 3500]

    # ------------------------------------------------------------------
    # Main optimisation loop – keep spawning fresh candidates while time permits.
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:               # leave a few seconds for finalisation
        n = int(rng.choice(lengths))

        # Choose a random shape family and random parameters.
        shape_id = rng.integers(0, 5)

        if shape_id == 0:                     # pure power law
            p = rng.uniform(0.35, 0.65)
            a = _shape_power(n, p)

        elif shape_id == 1:                   # shifted power law
            p = rng.uniform(0.35, 0.65)
            shift = rng.uniform(0.0, 5.0)
            a = _shape_power_shift(n, p, shift)

        elif shape_id == 2:                   # √‑differences
            a = _shape_sqrt_diff(n)

        elif shape_id == 3:                   # inverse‑sqrt
            a = _shape_inv_sqrt(n)

        else:                                 # exponential decay
            lam = rng.uniform(5e-4, 2e-2)
            a = _shape_exp(n, lam)

        # Occasionally add a tiny random perturbation to diversify the search.
        if rng.random() < 0.25:
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _normalize(a)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[seed] n={n} shape={shape_id} score={cur_val:.6f}")

        # --------------------------------------------------------------
        # Optional LP refinement (cheap for n ≤ 1800).  Performed with 50 %
        # probability to keep the overall budget under control.
        # --------------------------------------------------------------
        if _HAS_SCIPY and n <= 1800 and rng.random() < 0.5:
            M = _max_convolution(a)
            # One or two LP steps – enough to move noticeably towards a better region.
            for _ in range(rng.integers(1, 3)):
                a = lp_update(a, M, t=0.3)
                M = _max_convolution(a)

            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[LP]   n={n} shape={shape_id} improved to {cur_val:.6f}")

        # --------------------------------------------------------------
        # Annealed hill‑climbing – give each candidate a small time slice.
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + 4.0)
        a_opt, opt_val = local_search_anneal(a, sub_deadline,
                                             max_frac=0.09,
                                             temp_start=0.8,
                                             temp_end=0.02)
        if opt_val < best_val:
            best_val = opt_val
            best_seq = a_opt.copy()
            print(f"[anneal] n={n} shape={shape_id} best={opt_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely event that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _shape_power(1000, 0.5)

    # Trim trailing zeros that may have appeared due to rounding.
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
