"""
Improved search for a low‑score step‑function using an exponentiated‑gradient
descent on a *smoothed* max‑self‑convolution.  The surrogate objective is the
log‑sum‑exp (soft‑max) of the convolution entries; its gradient is obtained by
convolving the softmax weights with the reversed coefficient vector.  The
algorithm runs multiple short restarts, keeps the best sequence seen and only
calls the expensive ``evaluate_sequence`` when a surrogate improvement occurs.
It fits comfortably inside the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·n‑1) using an FFT of the next power of two.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _smooth_gradient_and_max(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the soft‑max surrogate
        (1/lam)·log Σ_k exp(lam·(a*a)[k])
    and also return the current maximal entry of the exact convolution.
    The gradient w.r.t. a is
        g = (w * a_rev)[n‑1:n‑1+n] + a
    where w are the soft‑max weights of the convolution and a_rev is a reversed.
    """
    n = a.shape[0]

    # ----- exact self‑convolution (needed for the surrogate weights) -----
    conv = _full_convolution(a)                # length 2·n‑1
    cur_max = np.max(conv)

    # ----- soft‑max weights w_k = exp(lam·conv_k) / Σ exp(...) -----
    shift = np.max(conv)                       # for numerical stability
    exps = np.exp(lam * (conv - shift))
    w = exps / np.sum(exps)                    # shape (2·n‑1,)

    # ----- convolution w * a_rev  (to obtain Σ_k w_k a_{k‑j}) -----
    a_rev = a[::-1]                            # reversed a
    conv_len = 3 * n - 2                       # (2·n‑1) + n‑1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv2 = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:conv_len]

    # slice that corresponds to Σ_k w_k a_{k‑j}
    slice_start = n - 1
    slice_end = slice_start + n
    weighted_sum = conv2[slice_start:slice_end]

    # gradient of the smooth objective
    grad = weighted_sum + a                     # shape (n,)

    return grad, cur_max


# ----------------------------------------------------------------------
# Exponentiated‑gradient step
# ----------------------------------------------------------------------


def _exp_gradient_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative update a ← a·exp(‑η·grad) followed by L1‑normalisation to
    Σ a = √(2·n).  All entries stay non‑negative.
    """
    a_new = a * np.exp(-eta * grad)

    # re‑normalise to the required L1‑norm (√(2·n))
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:                       # extremely unlikely
        a_new = np.full(n, target_sum / n)
    else:
        a_new *= (target_sum / total)
    return a_new


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative sequence of length n, summed to √(2·n).
    With 50 % probability the existing ``height_sequence_1`` (if present) is used,
    otherwise a random Dirichlet draw is taken and mildly perturbed.
    """
    S = np.sqrt(2.0 * n)

    if 'height_sequence_1' in globals() and rng.random() < 0.5:
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            a = np.maximum(0.0, a)
            if a.sum() == 0.0:
                raise ValueError
            a = a / a.sum() * S
            # small random perturbation to avoid staying on a flat plateau
            a += rng.normal(scale=0.01 * S / n, size=n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * S
            return a
        except Exception:
            pass

    # fall back to a uniform Dirichlet draw
    a = rng.random(n)
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _run_optimisation(n: int, seed: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform the exponentiated‑gradient search for a single run.  Returns the best
    sequence found and its external score.
    """
    rng = np.random.default_rng(seed)
    a = _initial_sequence(n, rng)

    # external evaluation of the starting point
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[run init] length={n}, score={best_score:.6f}")

    # hyper‑parameters
    eta = 0.8                     # initial step size
    eta_decay = 0.9996
    lam = 5.0                     # smoothing factor (soft‑max temperature)
    lam_max = 30.0
    lam_growth = 1.0015           # multiplicative growth per iteration
    jitter_prob = 0.015
    restart_wait = 30.0           # seconds without improvement → restart
    last_improve = time.time()

    while time.time() < deadline:
        # ---- compute smoothed gradient and the exact max entry ----
        grad, cur_max = _smooth_gradient_and_max(a, lam)

        # ---- multiplicative update ----
        a = _exp_gradient_step(a, grad, eta)

        # ---- schedule updates ----
        eta *= eta_decay
        if lam < lam_max:
            lam = min(lam_max, lam * lam_growth)

        # ---- external evaluation only on surrogate improvement ----
        if cur_max < best_score:
            ext = evaluate_sequence(a.tolist())
            if ext < best_score:
                best_score = ext
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[improve] time={time.time():.1f}s  score={best_score:.6f}")

        # ---- occasional random jitter to escape tiny plateaus ----
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=0.01 * np.sqrt(2.0 * n) / n, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()

        # ---- restart if we have been idle for a while ----
        if time.time() - last_improve > restart_wait:
            a = _initial_sequence(n, rng)
            eta = 0.8
            lam = 5.0
            last_improve = time.time()
            print(f"[restart] time={time.time():.1f}s")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the current bound of 1.503.

    The method runs for at most ``budget_s`` seconds, performing several
    independent optimisation runs (different seeds, occasional restarts) and
    keeps the best sequence encountered.
    """
    # safety margin – leave a couple of seconds for the caller to finish
    deadline = time.time() + budget_s - 5.0

    # default length is the size of the known construction (5641);
    # if it is missing we fall back to a reasonable size.
    n = 5641
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)
        except Exception:
            pass

    # we will launch a few (up to 5) independent runs, each with its own seed.
    best_overall_seq = None
    best_overall_score = np.inf

    run_seeds = [seed + i for i in range(5)]
    for run_seed in run_seeds:
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget for this run
        remaining = deadline - time.time()
        run_deadline = time.time() + remaining * 0.85   # keep a small margin
        seq, sc = _run_optimisation(n, run_seed, run_deadline)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_seq = seq
            print(f"[global improve] new best {best_overall_score:.6f}")

    # fallback – uniform distribution (should never be needed)
    if best_overall_seq is None:
        best_overall_seq = np.full(n, np.sqrt(2.0 * n) / n)

    print(f"[final] best score = {best_overall_score:.6f}  length = {len(best_overall_seq)}")
    return best_overall_seq.tolist()
