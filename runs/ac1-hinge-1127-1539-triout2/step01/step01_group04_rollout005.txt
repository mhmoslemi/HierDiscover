def compute_score(a):
    n=len(a)
    # compute convolution max via FFT (real)
    L=1<<((2*n-1).bit_length())
    conv=np.fft.irfft(np.fft.rfft(a, n=L)**2, n=L)[:2*n-1]
    maxb=conv.max()
    sum_a = a.sum()
    score = 2.0*n*maxb/(sum_a*sum_a)
    return score

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Return the linear (self) convolution of a with itself,
    using an FFT of length L (a power of two ≥ 2·len(a)‑1).
    The result is a 1‑D array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]


def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """
    Accumulate the sub‑gradient contribution for a single active shift k.
    The contribution is 2·a_{k‑i} for every valid i.
    """
    n = a.size
    if k < n:
        # indices 0 … k  (the reversed slice a[k::-1] has length k+1)
        grad[: k + 1] += 2.0 * a[k :: -1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]


def _score_from_convolution(max_b: float, n: int, sum_a: float) -> float:
    """
    Exact score according to the supplied evaluate_sequence routine:
        2·n·max_b / (sum_a)²
    """
    return 2.0 * n * max_b / (sum_a * sum_a)


def _fast_score(a: np.ndarray, L: int) -> float:
    """
    Compute the score for a vector a (sum is not assumed to be 1)
    using an FFT convolution.  Returns the same value that
    evaluate_sequence would return (modulo the clipping that never
    triggers for the magnitudes we use).
    """
    conv = _conv_fft(a, L)
    max_b = conv.max()
    return _score_from_convolution(max_b, a.size, a.sum())

    # --------------------------------------------------------------
    # 1) Choose the length of the sequence.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
    except Exception:
        n = 8000                               # default length

    # Normalisation target – we keep the L1‑norm equal to 1.0.
    # The objective is scale‑invariant, so this is convenient.
    target_sum = 1.0

    # Pre‑compute the FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 2) Coarse grid search over symmetric Beta‑type distributions.
    #    a_i(α,δ) ∝ (i+δ)^{α‑1}·(n‑1‑i+δ)^{α‑1}
    # --------------------------------------------------------------
    alphas = np.array([0.30, 0.40, 0.50, 0.60, 0.70, 0.80, 0.90,
                       1.00, 1.10, 1.20, 1.30, 1.40, 1.50, 1.60,
                       1.80, 2.00])
    deltas = np.array([0.01, 0.05, 0.10, 0.25, 0.50, 1.0])

    idx = np.arange(n, dtype=np.float64)

    for α in alphas:
        # break early if we run out of time
        if time.time() >= deadline:
            break
        for δ in deltas:
            # Avoid division by zero for the extreme endpoints.
            # The formula works for any δ>0.
            a = (idx + δ) ** (α - 1.0) * (n - 1 - idx + δ) ** (α - 1.0)
            s = a.sum()
            if s == 0.0:
                continue
            a *= (target_sum / s)                   # normalise to L1 = 1
            # compute score quickly
            score = _fast_score(a, L)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(
                    f"grid α={α:.2f} δ={δ:.3f} → score {best_score:.6f}"
                )

    # --------------------------------------------------------------
    # 3) A few classic shapes (sqrt‑difference, uniform, triangular)
    # --------------------------------------------------------------
    # sqrt‑difference (the shape used in the reference implementation)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    a_sqrt *= target_sum / a_sqrt.sum()
    score = _fast_score(a_sqrt, L)
    if score < best_score:
        best_score, best_seq = score, a_sqrt.copy()
        print(f"sqrt‑diff shape → score {best_score:.6f}")

    # Uniform
    a_uniform = np.full(n, target_sum / n, dtype=np.float64)
    score = _fast_score(a_uniform, L)
    if score < best_score:
        best_score, best_seq = score, a_uniform.copy()
        print(f"uniform shape → score {best_score:.6f}")

    # Simple triangular (linear rise then fall)
    a_tri = np.concatenate([np.arange(1, n // 2 + 1),
                            np.arange(n // 2, 0, -1)])
    a_tri = a_tri[:n].astype(np.float64)
    a_tri *= target_sum / a_tri.sum()
    score = _fast_score(a_tri, L)
    if score < best_score:
        best_score, best_seq = score, a_tri.copy()
        print(f"triangular shape → score {best_score:.6f}")

    # If for any reason we did not obtain a candidate, fall back to uniform.
    if best_seq is None:
        best_seq = a_uniform.copy()
        best_score = _fast_score(best_seq, L)

    # --------------------------------------------------------------
    # 4) Refine the best candidate with projected sub‑gradient descent.
    # --------------------------------------------------------------
    a = best_seq.copy()
    iteration = 0
    lr0 = 0.80                # initial step‑size factor

    while time.time() < deadline:
        # ----- convolution & current score -----
        conv = _conv_fft(a, L)
        max_b = conv.max()
        cur_score = _score_from_convolution(max_b, n, a.sum())

        # ----- keep the best ever observed -----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:6d} – new best {best_score:.6f}")

        # ----- identify active shifts (within a tiny tolerance) -----
        eps = 1e-12 * max_b
        active_ks = np.where(conv >= max_b - eps)[0]

        # ----- accumulate sub‑gradient -----
        grad = np.zeros_like(a)
        for k in active_ks:
            _add_gradient_for_k(grad, a, int(k))

        # ----- step size (decays like 1/√t) -----
        lr = lr0 / math.sqrt(iteration + 1.0)

        # ----- gradient step -----
        a -= lr * grad

        # ----- projection onto feasible set (non‑negative, fixed L1) -----
        np.maximum(a, 0.0, out=a)
        s = a.sum()
        if s == 0.0:
            # degenerate case – reset to uniform
            a.fill(target_sum / n)
        else:
            a *= (target_sum / s)

        # ----- occasional jitter to avoid stagnation -----
        if iteration % 200 == 0 and iteration > 0:
            sigma = target_sum * 5e-4 / math.sqrt(n)
            noise = np.random.randn(n) * sigma
            a += noise
            np.maximum(a, 0.0, out=a)
            a *= (target_sum / a.sum())

        iteration += 1

    # --------------------------------------------------------------
    # 5) Return the best sequence (as a Python list).
    # --------------------------------------------------------------
    best_list = best_seq.tolist()
    # One final safety evaluation (so the surrounding harness can see the score)
    try:
        final_score = evaluate_sequence(best_list)
        print(f"Final evaluated score: {final_score:.6f}")
    except Exception:
        print("Final evaluation raised an exception.")
    return best_list
