"""
Propose a step‑function sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a hybrid of
* a soft‑max (log‑sum‑exp) gradient that smooths the maximum of the
  self‑convolution,
* a hard sub‑gradient that directly targets the current peak,
* a simple momentum scheme,
* a slowly increasing “temperature’’ (γ) and a decreasing step‑size (η).

The sequence is always projected onto the non‑negative orthant and
renormalised so that sum(a)=sqrt(2·n).  The best sequence seen within the
time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official evaluation (no security checks).
    Returns max(conv(a, a)) because the denominator is fixed to sqrt(2n).
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # the official formula simplifies to max_c when sum = sqrt(2n)
    return float(2.0 * n * max_c / (s * s))

# Main entry point
# ----------------------------------------------------------------------

        # 3.10 Update hyper‑parameters (cooling / focusing)
        # ------------------------------------------------------------------
        gamma = min(gamma * 1.005, gamma_max)
        eta = max(eta * 0.998, eta_min)

        # optional very occasional “restart‑near‑best’’ to diversify search
        if iteration % 5000 == 0:
            a = best_a + 0.01 * np.random.rand(n)
            a = np.maximum(a, 0.0)
            a *= target_sum / a.sum()
            velocity.fill(0.0)

    # ----------------------------------------------------------------------
    # Return the best sequence discovered
    # ----------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_a.tolist()
