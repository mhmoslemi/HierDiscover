def lp_optimize(f, M):
    n = len(f)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # indices j from lo to hi inclusive
        idx = np.arange(lo, hi+1, dtype=int)
        A[k, idx] = f[k - idx]
    b = np.full(2*n-1, M, dtype=np.float64)

    # objective: maximize sum(g) => minimize -1^T g
    c = -np.ones(n, dtype=np.float64)

    # call linprog
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return f
        return f.copy()
    g = res.x
    if g.sum() == 0:
        return f.copy()
    return g / g.sum()

def lp_optimize(f, M, time_limit=None):
    # Return optimized g vector on simplex maximizing sum under f*g <= M.
    # If time limit shortage, fallback.
    n = len(f)
    # Build A
    # For speed, we can use Toeplitz: conv matrix.
    # Using loops is fine.
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for k in range(2*n - 1):
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        idx = np.arange(lo, hi+1)
        A[k, idx] = f[k - idx]
    b = np.full(2*n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'time_limit': time_limit} if time_limit else None)
    if res.success:
        g = res.x
        sum_g = g.sum()
        if sum_g > 0:
            g = g / sum_g
        return g
    else:
        # fallback: return current f.
        return f.copy()

"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.

Algorithm
---------
1. **Initial pool** – generate a modest number of candidate vectors of
   different lengths (100…500) using:
   * uniform distribution,
   * power‑law families a_i = (i+σ)^(-p) (p≈0.5, σ>0),
   * random Dirichlet draws,
   * the optional global `height_sequence_1` if it exists.

2. **Linear‑programming refinement** – for a candidate `f` (ℓ₁‑normalised)
   compute M = max(conv(f,f)).  Solve the LP

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k
                    g_j ≥ 0

   (`f * g` is a linear function of `g`).  The LP pushes mass into positions
   where `f` is small, therefore reducing the maximum convolution value.
   Mix the solution back:   f ← (1‑t)·f + t·g   (t = 0.5) and re‑normalise.
   Repeat a few iterations (≈15) per candidate.

3. **Mirror‑descent polishing** – take the best vector found so far and
   perform a classic entropy‑mirror‑descent step that directly reduces the
   ℓ∞‑norm of the self‑convolution:

        x_i ← x_i·exp(‑η·g_i)   (g_i = x_{k‑i} where k is the index of the
                                 current maximal convolution entry)

   The step size η decays as 0.2/√iteration.  Occasionally a tiny random
   jitter is added to escape plateaus.

4. **Time management** – the whole routine stops a few seconds before the
   supplied budget (default 1000 s).  The best sequence observed is returned.

All vectors are kept on the probability simplex (sum = 1); the objective is
scale‑invariant, so this normalisation does not affect the score.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray) -> np.ndarray:
    """Return v scaled to sum 1 (leaves zero vector unchanged)."""
    s = v.sum()
    if s == 0.0:
        return v
    return v / s


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g)_k = (f * g)[k].
    All entries are non‑negative because f ≥ 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        A[k, idx] = f[k - idx]
    return A


def _lp_optimize(f: np.ndarray, M: float, time_limit: float = None) -> np.ndarray:
    """
    Solve the LP that maximises Σ g_j subject to (f * g)_k ≤ M and g ≥ 0.
    Returns a normalised feasible g (sum = 1).  If the solver fails,
    the original f is returned unchanged.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g → minimise -Σ g

    linprog_opts = {"presolve": True}
    if time_limit is not None:
        # recent SciPy versions accept a time limit for the HiGHS solver
        linprog_opts["time_limit"] = max(0.0, time_limit)

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                bounds=(0, None),
                                method='highs',
                                options=linprog_opts)

    if res.success:
        g = res.x
        return _normalise(g)
    # fallback – return the input unchanged
    return f.copy()


def _mirror_descent_step(x: np.ndarray, eta: float) -> np.ndarray:
    """
    One iteration of entropy‑mirror‑descent aimed at reducing
    max(conv(x,x)).  Returns a new simplex‑normalised vector.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)
    k = int(conv.argmax())                     # index of the maximal entry

    # sub‑gradient g_i = x_{k‑i} (zero when out of range)
    g = np.zeros_like(x)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    jdx = k - idx
    g[idx] = x[jdx]

    y = x * np.exp(-eta * g)
    if y.sum() == 0.0:
        y = np.abs(np.random.randn(n))
    y = y / y.sum()
    return y
