def optimizer(a_init, time_budget):
    a = normalize(a_init.copy())
    n = a.shape[0]
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    lr_min = 0.001
    gamma = 2.0
    gamma_max = 200.0

    best_a = a.copy()
    best_score = _compute_score(a)

    iteration = 0
    while time.time() < deadline - 0.5: # maybe need inner param
        iteration += 1

        # conv via FFT
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[:2 * n - 1]
        max_conv = conv.max()

        # softmax weights
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum

        # gradient via correlation
        wf = np.fft.rfft(w, n=nfft_corr)
        ar_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * ar_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        # Evaluate score
        score = _compute_score(a_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            elapsed = time.time() - start
            print(f"Iter {iteration:6d}, new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

        # Accept update
        a = a_new

        # Update hyperparameters
        gamma = min(gamma * 1.004, gamma_max)
        lr = max(lr * 0.998, lr_min)

        # occasional jitter for global search
        if iteration % 50 == 0:
            jitter = 1e-6 * np.random.randn(n)
            a = np.maximum(a + jitter, 0.0)
            a = normalize(a)

        # stop early if time near deadline
        if time.time() > deadline - 0.5:
            break

    return best_a, best_score

def optimizer(a_init, deadline, start_time):
    a = a_init.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    def normalize(arr):
        s = arr.sum()
        if s == 0:
            return np.full_like(arr, target_sum / n)
        return arr * (target_sum / s)

    # Precompute FFT sizes
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12
    lr_min = 0.001
    gamma = 5.0
    gamma_max = 200.0

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score(a)

    iter_cnt = 0
    # parameter to determine jitter interval
    jitter_interval = max(50, n // 100)  # at least every 50 steps

    while time.time() < deadline - 0.5:
        iter_cnt += 1

        # convolution a*a using FFT
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # compute softmax weights
        # note max_conv may be large, but exponent will be <=1
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum

        # gradient via cross-correlation
        wf = np.fft.rfft(w, n=nfft_corr)
        ar_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * ar_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # projection onto non-negative orthant
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        # evaluate
        score = _compute_score(a_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.1f}s] iter {iter_cnt:6d}: new best {best_score:.6f}")

        # Accept this step
        a = a_new

        # Hyperparameter updates
        gamma = min(gamma * 1.003, gamma_max)
        lr = max(lr * 0.998, lr_min)

        if iter_cnt % jitter_interval == 0:
            jitter = 1e-4 * np.random.randn(n)
            a = np.maximum(a + jitter, 0.0)
            a = normalize(a)

    return best_a, best_score

"""
Propose a sequence that (hopefully) beats the previous best bound.

Algorithm
---------
1. Choose several generic initial shapes (inverse‑sqrt, exponential decay,
   linear decline, uniform, random).  All are rescaled so that
   Σa = √(2 n) (the scaling does not affect the objective).

2. For each initial shape run a fast gradient‑descent on the smoothed
   maximum of the auto‑convolution a*a.
   • The smoothed maximum is obtained by a soft‑max with a temperature
     parameter γ that is slowly increased.
   • The gradient of the soft‑max is  g = 2·(w ⊛ a_rev)  and is computed
     efficiently with FFT‑based cross‑correlation.
   • An Adam optimiser (first‑ and second‑moment adaptation) is used for
     the updates, together with a projection onto the non‑negative orthant
     and a renormalisation to keep the L1‑norm fixed.
   • Occasionally a tiny random jitter is added to escape plateaus.

3. The best sequence seen over all initialisations is kept.
   If time permits a final refinement is performed starting from this best
   sequence with a smaller learning‑rate and a larger γ.

All heavy work (FFT, gradient) is O(n log n) and runs until the allotted
time (default 1000 s) expires.  The function returns the best list of
coefficients found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of the evaluation function (without security checks).
    Returns 2 * n * max(conv(a, a)) / (sum(a) ** 2).
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _normalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a to have Σa = target_sum, keep non‑negative entries."""
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, target_sum / a.size)
    return a * (target_sum / s)

# Initial shape generators
# ----------------------------------------------------------------------


def _init_inverse_sqrt(n: int) -> np.ndarray:
    """a_i = 1 / sqrt(i + 0.5)  (classic starting shape)."""
    a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
    return _normalise(a, np.sqrt(2.0 * n))


def _init_exponential(n: int) -> np.ndarray:
    """Exponential decay a_i = exp(-i / τ)."""
    tau = n / 5.0
    a = np.exp(-np.arange(n, dtype=np.float64) / tau)
    return _normalise(a, np.sqrt(2.0 * n))


def _init_linear(n: int) -> np.ndarray:
    """Linear decline from 1 to 0."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    return _normalise(a, np.sqrt(2.0 * n))


def _init_uniform(n: int) -> np.ndarray:
    """Uniform vector."""
    a = np.ones(n, dtype=np.float64)
    return _normalise(a, np.sqrt(2.0 * n))


def _init_random(n: int) -> np.ndarray:
    """Random positive entries."""
    a = np.random.rand(n)
    return _normalise(a, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Optimiser – Adam + soft‑max of the convolution
# ----------------------------------------------------------------------


def _optimise(a_start: np.ndarray, deadline: float, start_time: float):
    """
    Perform Adam‑style optimisation of the smoothed max of a*a.
    Returns (best_sequence, best_score) found before `deadline`.
    """
    a = a_start.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # FFT sizes (pre‑computed)
    nfft_conv = _next_pow2(2 * n - 1)      # for a*a
    nfft_corr = _next_pow2(3 * n - 2)      # for w ⊛ a_rev

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12
    lr_min = 0.001
    gamma = 5.0
    gamma_max = 200.0

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score(a)

    it = 0
    jitter_every = max(50, n // 100)  # occasional perturbation
