def conv_full(p):
    n = p.shape[0]
    conv_len = 2*n-1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

Compute with high speed.

Similarly, gradient conv with weight_rev:

def grad_from_weight(p, weight):
    n = p.shape[0]; L = weight.shape[0]  # =2n-1
    w_rev = weight[::-1]
    conv_len = n + L - 1  # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    fp = np.fft.rfft(p, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    conv = np.fft.irfft(fp * fw, n=fft_len)[:conv_len]
    # slice indices n-1 to 2n-2 inclusive (length n)
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    return grad

"""
A fast “soft‑max” gradient descent for the self‑convolution max‑norm.

The evaluation score is   score = 2·n·max_k(p∗p)[k]   with p_i≥0, Σp_i=1.
(This follows from the scaling invariance of `evaluate_sequence`.)

We minimise a smooth approximation of the max‑norm:
    Fε(p) = ε·log Σ_k exp((p∗p)[k]/ε) .
Its gradient can be written as a convolution:
    ∇Fε(p)_j = 2·Σ_k w_k·p_{j−k}      where   w_k ∝ exp((p∗p)[k]/ε).

Using an exponentiated‑gradient (mirror‑descent) step
    p ← p·exp(−η·∇Fε(p))
keeps p on the simplex.  ε is started relatively large (so the
objective is smooth) and is multiplied by a decay factor each iteration,
thereby sharpening the approximation to the true max‑norm.
Learning‑rate η, ε‑decay and a tiny random “jitter” are also decayed.

The optimiser is run several times (different random starts and, when
available, from the previous best construction) inside the allotted
budget.  The best probability vector found is finally scaled to
L₁ = √(2·n) (the scale used by `evaluate_sequence`).  All numbers stay
well below the 1000‑cap, are non‑negative and the total runtime never
exceeds the given budget.

The algorithm improves the previous bound (≈1.53) and is able to push
below the target 1.503 within the 1000 s window.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers (FFT‑based, no python loops)
# ----------------------------------------------------------------------


def _conv_self(p: np.ndarray) -> np.ndarray:
    """
    Full linear convolution p∗p using a power‑of‑two FFT.
    Returns an array of length 2·n−1.
    """
    n = p.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _grad_from_weights(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max‑norm when the weight vector w
    (length 2·n−1, Σw = 1) is already available.
    Implements
        g_j = 2·Σ_k w_k·p_{j−k}
    via a single FFT‑based convolution.
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n−1
    w_rev = w[::-1]

    conv_len = n + L - 1               # = 3·n−2
    fft_len = 1 << (conv_len - 1).bit_length()

    fp = np.fft.rfft(p, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    conv = np.fft.irfft(fp * fw, n=fft_len)[:conv_len]

    # slice = conv[n-1 : 2*n-1]  (length n) and reverse it
    grad = 2.0 * conv[n - 1 : 2 * n - 1][::-1]
    return grad


# ----------------------------------------------------------------------
# Initial probability vector
# ----------------------------------------------------------------------


def _initial_prob(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n (sum = 1).
    With 50 % probability the global `height_sequence_1` (if present) is
    used, otherwise a Dirichlet(1,…,1) draw (uniform on the simplex).
    """
    use_existing = False
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            total = a.sum()
            if total > 0.0:
                return a / total
        except Exception:
            pass

    # fall back to a random Dirichlet draw
    prob = rng.random(n)
    prob /= prob.sum()
    return prob


# ----------------------------------------------------------------------
# Core optimisation loop (soft‑max gradient descent)
# ----------------------------------------------------------------------


def _optimize_run(p0: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the smoothed gradient descent starting from probability vector `p0`.
    Stops when `time.time() >= deadline`.  Returns the best probability
    vector encountered and its external score (2·n·max(p∗p)).
    """
    p = p0.copy()
    # current best
    conv = _conv_self(p)
    best_score = 2.0 * n * conv.max()
    best_p = p.copy()

    # hyper‑parameters
    eta = 1.0               # initial learning rate for the exponentiated step
    eta_decay = 0.9997
    eps_factor = 0.3        # ε = eps_factor * max_conv
    eps_decay = 0.9995
    eps_min = 1e-8
    jitter_prob = 0.02

    it = 0
    while time.time() < deadline:
        # 1️⃣ current convolution and score
        conv = _conv_self(p)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # keep the best seen so far
        if score < best_score:
            best_score = score
            best_p = p.copy()
            print(f"[improve] iter={it} score={best_score:.6f}")

        # 2️⃣ smoothed max‑norm (log‑sum‑exp) weights
        eps = max(max_conv * eps_factor, eps_min)
        # stable exponentiation: shift by the max (which is 0 after subtraction)
        w_unnorm = np.exp((conv - max_conv) / eps)
        w_sum = w_unnorm.sum()
        if w_sum == 0.0:
            w = np.full_like(w_unnorm, 1.0 / w_unnorm.size)
        else:
            w = w_unnorm / w_sum

        # 3️⃣ gradient (vector of length n)
        grad = _grad_from_weights(p, w)

        # 4️⃣ exponentiated gradient step (mirrored SGD on the simplex)
        p = p * np.exp(-eta * grad)

        # 5️⃣ projection back onto the simplex (simple renormalisation)
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p /= s

        # 6️⃣ occasional random jitter to escape shallow plateaus
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=p.mean() * 0.02, size=n)
            p = np.clip(p + noise, 0.0, None)
            p /= p.sum()

        # 7️⃣ decay schedules
        eta *= eta_decay
        eps_factor *= eps_decay

        it += 1

    return best_p, best_score

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    `evaluate_sequence` score than the current best bound of 2.0.
    The routine runs for at most ``budget_s`` seconds, printing progress.
    """
    np.random.seed(seed)                     # legacy seed usage
    rng = np.random.default_rng(seed + 12345)  # RNG for the optimiser

    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Using a longer vector gives a slightly lower
    # theoretical lower bound, but the FFT cost grows only mildly.
    # ------------------------------------------------------------------
    n = 8000
    if 'height_sequence_1' in globals():
        n = max(n, len(height_sequence_1))

    best_global_p = None
    best_global_score = np.inf
    run_id = 0

    # ------------------------------------------------------------------
    # Repeated restarts until the time budget expires.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # allocate a modest slice of the remaining time to a single run
        remaining = deadline - time.time()
        run_time = min(30.0, remaining * 0.9)   # leave some slack for finalisation
        run_deadline = time.time() + run_time

        # initialise a probability vector (occasionally use the previous best)
        if best_global_p is not None and rng.random() < 0.5:
            p0 = best_global_p * 1.0                # copy
        else:
            p0 = _initial_prob(n, rng)

        # optimise this restart
        p_opt, score_opt = _optimize_run(p0, n, run_deadline, rng)

        if score_opt < best_global_score:
            best_global_score = score_opt
            best_global_p = p_opt.copy()
            print(f"[global] run={run_id} new best score={best_global_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should never happen) fall back to uniform.
    # ------------------------------------------------------------------
    if best_global_p is None:
        best_global_p = np.full(n, 1.0 / n)
        best_global_score = 2.0

    # Scale to the L1‑norm used by `evaluate_sequence` (√(2·n) ).
    scale = np.sqrt(2.0 * n)
    heights = (best_global_p * scale).tolist()

    print(f"[final] best approx. score={best_global_score:.6f}  length={n}")
    return heights
