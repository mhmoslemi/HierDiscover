"""
Propose a coefficient sequence that (hopefully) improves the current best
score ≈ 1.572.  

The algorithm works in two phases:

1. **Broad parametric sweep** – we generate many simple families
   (power‑law, exponential) for a variety of lengths, exponents and offsets.
   Every candidate is evaluated with `evaluate_sequence` and the best
   candidate is kept.

2. **Frank‑Wolfe style refinement** – starting from the best candidate we
   repeatedly solve a linear programme
        maximise   Σ b_i
        subject to b_i ≥ 0   and   (f * b)_k ≤ M  (∀k)
   where `f` is the current sequence and `M` is its current maximal
   convolution value.  The solution `b` is normalised to a probability vector
   and mixed back into `f` (convex combination).  Each iteration strictly
   reduces the objective or leaves it unchanged.  The LP is solved with
   CVXPY/ECOS; the size of the problem (n ≤ 800) makes a solve cheap.

Both phases respect the 1000 s wall‑clock budget and the best sequence
found is returned.
"""

import time
import numpy as np

# cvxpy is part of the allowed scientific stack.  Import lazily in case it
# is not available on the execution platform – the code will still run
# without the refinement phase.
try:
    import cvxpy as cp
except Exception:                # pragma: no cover
    cp = None


def _powerlaw_seq(p: float, sigma: float, n: int) -> np.ndarray:
    """Return a raw power‑law vector a_i = (i+σ)^{-p} of length *n*."""
    i = np.arange(n, dtype=np.float64)
    # avoid a division by zero when σ==0 and i==0
    if sigma == 0.0:
        sigma = 1e-12
    a = (i + sigma) ** (-p)
    return a


def _exp_seq(lam: float, n: int) -> np.ndarray:
    """Exponential decay a_i = exp(-λ·i) of length *n*."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a


def _conv_max(seq: np.ndarray) -> float:
    """Return the maximal entry of the aperiodic autocorrelation (full conv)."""
    return float(np.max(np.convolve(seq, seq)))


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    One Frank‑Wolfe step.

    Solve   maximise  Σ b_i
    s.t.    b ≥ 0,
            (f * b) ≤ M   (elementwise)
    and return the normalised solution b̂ = b / Σ b .
    """
    n = f.shape[0]
    # Build the constraint matrix A such that (A @ b)[k] = (f * b)[k].
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        # column i contributes f shifted by i
        A[i:i + n, i] = f

    # cvxpy variables
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS is the default free solver in the environment.
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)   # type: ignore

    if b.value is None:
        # In the unlikely event that the solver fails, fall back to the
        # current vector (no progress).
        return f.copy()
    b_vec = np.asarray(b.value).ravel()
    s = b_vec.sum()
    if s <= 0.0:
        # safeguard against division by zero
        return f.copy()
    return b_vec / s


def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the 1000 s time limit
    and returns the best sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a tiny safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep
    # ------------------------------------------------------------------
    print("=== Phase 1 : parametric sweep ===")
    # try a handful of lengths; short lengths are cheap and give a coarse
    # search, longer lengths give a finer approximation of the continuous
    # optimum.
    lengths = [200, 500, 800, 1200, 2000]
    p_vals = np.linspace(0.3, 0.9, 13)          # include the classical 0.5
    sigma_vals = np.concatenate([
        np.logspace(-4, -1, 5),              # very small offsets
        np.logspace(-1, 0.5, 7),             # moderate offsets
        np.logspace(0.5, 1.2, 5)             # large offsets (up to ~15)
    ])
    lam_vals = np.linspace(0.01, 4.0, 15)       # for exponential families

    for n in lengths:
        if time.time() >= deadline:
            break
        # --- power law family ------------------------------------------------
        for p in p_vals:
            for sigma in sigma_vals:
                if time.time() >= deadline:
                    break
                seq = _powerlaw_seq(p, sigma, n)
                # normalisation is irrelevant for the objective, but we keep a
                # moderate scale to avoid extreme values that trigger the
                # built‑in clipping.
                seq = seq.astype(np.float64)
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                if np.isfinite(score) and score < best_score:
                    best_score = float(score)
                    best_seq = seq.tolist()
                    print(f'new best (pow) | n={n} p={p:.3f} σ={sigma:.5f} '
                          f'score={best_score:.6f}')
        # --- exponential family ----------------------------------------------
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            seq = _exp_seq(lam, n)
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f'new best (exp) | n={n} λ={lam:.3f} score={best_score:.6f}')

    # If nothing succeeded (unlikely) fall back to a uniform vector.
    if best_seq is None:
        n_fallback = 300
        best_seq = np.ones(n_fallback, dtype=np.float64).tolist()
        best_score = evaluate_sequence(best_seq)
        print('fallback to uniform sequence, score =', best_score)

    # ------------------------------------------------------------------
    # Phase 2 – Frank‑Wolfe / LP refinement (requires cvxpy)
    # ------------------------------------------------------------------
    if cp is None:
        print("cvxpy not available – skipping LP refinement.")
        print('final best score:', best_score, 'length:', len(best_seq))
        return best_seq

    print("\n=== Phase 2 : Frank‑Wolfe (LP) refinement ===")
    f = np.array(best_seq, dtype=np.float64)
    # Normalise to a probability simplex – the objective is scale‑invariant.
    f = f / f.sum()

    iteration = 0
    eta0 = 0.5                     # initial mixing weight
    while time.time() < deadline:
        iteration += 1
        # current maximal convolution value
        M = _conv_max(f)

        # one LP step
        g = _lp_step(f, M)

        # mixing weight slowly decays – similar to a diminishing step size.
        eta = eta0 / np.sqrt(iteration)
        f = (1.0 - eta) * f + eta * g
        # re‑normalise to avoid drift due to numerical errors
        f = np.maximum(f, 0.0)
        f /= f.sum()

        # evaluate current candidate
        try:
            cur_score = evaluate_sequence(f.tolist())
        except Exception:
            continue

        if cur_score < best_score:
            best_score = float(cur_score)
            best_seq = f.tolist()
            print(f'iteration {iteration:5d} -> improved score = {best_score:.6f}')
            # early stop if we already beat a reasonable target
            if best_score < 1.55:
                print('Good enough – stopping early.')
                break

        # occasional random perturbation to escape tiny plateaus
        if np.random.rand() < 0.02:
            idx = np.random.randint(f.shape[0])
            delta = (np.random.rand() - 0.5) * 0.02 * f[idx]
            f[idx] = max(0.0, f[idx] + delta)
            f /= f.sum()

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
