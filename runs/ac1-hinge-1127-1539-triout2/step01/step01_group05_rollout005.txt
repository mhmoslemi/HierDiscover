"""
Propose a sequence that (hopefully) improves the current best bound.

Algorithm
---------
1. **Parameter sweep** – explore a simple parametric family
       a_i = (i + β)^(-p)   (i = 0 … n‑1)
   for several values of n, exponent p and shift β.  Each candidate is
   normalised to have Σa = sqrt(2·n) (the evaluation function is scale‑invariant)
   and its score is evaluated with a fast FFT based routine.  The best
   candidate from the sweep (or the previously known construction
   ``height_sequence_1`` if it is better) is taken as a starting point.

2. **Sub‑gradient descent on the true sup‑norm** –  with the current vector
   a we compute its autocorrelation c = a * a.
   The maximal value max(c) is the quantity we want to minimise.
   A sub‑gradient of max(c) is obtained by giving equal weight to every
   index where the maximum is attained and correlating this weight vector
   with the reversed a:
        g = 2·( w ⊛ a_rev )
   (implemented efficiently by FFT).  A projected gradient step
   a ← a – η·g, followed by clipping to the non‑negative orthant and
   renormalisation (Σa stays sqrt(2·n)), is performed repeatedly.
   The step size η is adapted on‑the‑fly: it is increased when a step
   improves the score and decreased otherwise.  Occasionally a tiny random
   jitter is added to escape plateaus.

3. **Book‑keeping** – the best sequence and its score seen so far are
   stored; any improvement is printed.  The whole optimisation runs for
   the allotted time budget (≈1000 s) and finally returns the best
   sequence as a plain Python list.
"""

import time
import math

import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) minimise the
    evaluation function.

    The function follows the two‑stage strategy described in the module
    docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  If a previous construction is available, evaluate it first.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n_prev = prev.shape[0]
        target_prev = math.sqrt(2.0 * n_prev)
        prev = prev * (target_prev / prev.sum())
        score_prev = _compute_score(prev)
        best_seq = prev.copy()
        best_score = score_prev
        print(f"Initial candidate from height_sequence_1: score {best_score:.6f}")
    except Exception:
        # No previous sequence – continue without it.
        pass

    # ------------------------------------------------------------------
    # 1.  Coarse grid search over a simple parametric family.
    # ------------------------------------------------------------------
    # Candidate lengths (feel free to adjust).  The runtime is tiny.
    n_candidates = [800, 1500, 3000, 5000, 8000]
    p_vals = np.linspace(0.45, 1.30, 17)          # exponent range
    beta_vals = [0.0, 0.5, 1.0, 2.0, 5.0]        # shift range

    for n in n_candidates:
        target_sum = math.sqrt(2.0 * n)
        # Pre‑compute the FFT length for convolution (used by _compute_score).
        # The function already accounts for it, so we just call it.
        for p in p_vals:
            for beta in beta_vals:
                if time.time() >= deadline:
                    break
                # a_i = (i + β)^(-p)
                idx = np.arange(n, dtype=np.float64)
                a = (idx + beta) ** (-p)
                s = a.sum()
                if s == 0.0:
                    continue
                a *= target_sum / s
                score = _compute_score(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    elapsed = time.time() - start_time
                    print(
                        f"Grid n={n:5d} p={p:0.3f} β={beta:0.1f} → "
                        f"score {score:.6f} (elapsed {elapsed:.1f}s)"
                    )
        if time.time() >= deadline:
            break

    # If we somehow have not found any candidate yet (unlikely), fall back
    # to a tiny uniform vector.
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _compute_score(best_seq)

    # ------------------------------------------------------------------
    # 2.  Sub‑gradient descent on the true maximum of the convolution.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)
    # FFT sizes needed each iteration.
    nfft_conv = _next_pow2(2 * n - 1)          # for a*a
    nfft_corr = _next_pow2(3 * n - 2)          # for w ⊛ a_rev

    # Initial step size – a modest value that works well for the scales we use.
    eta = 0.08
    eta_min = 1e-6
    eta_max = 2.0

    cur_score = _compute_score(a)
    iteration = 0

    # Main optimisation loop.
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 2.1 Compute autocorrelation and locate all indices attaining the max.
        # --------------------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_c = conv.max()
        # Indices where the maximum is attained (within a tiny absolute tolerance).
        # Using a relative tolerance would be problematic when max_c is small.
        eps = 1e-12
        active = np.where(conv >= max_c - eps)[0]
        # Build the weight vector w (uniform over the active set).
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        # --------------------------------------------------------------
        # 2.2 Sub‑gradient of the sup‑norm via correlation.
        # --------------------------------------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        # The sub‑gradient lives in the centre part of the correlation.
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # 2.3 Gradient step with projection onto the simplex (non‑negative,
        #      fixed L1‑norm).
        # --------------------------------------------------------------
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)                     # non‑negativity
        s_new = a_new.sum()
        if s_new == 0.0:
            # Extremely unlikely, but fall back to a uniform vector.
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / s_new                # renormalise to fixed sum

        # --------------------------------------------------------------
        # 2.4 Evaluate and decide whether to accept the step.
        # --------------------------------------------------------------
        new_score = _compute_score(a_new)

        if new_score < cur_score:
            # Accept the improvement.
            a = a_new
            cur_score = new_score
            # Increase step size slightly (up to a cap) to accelerate when we are
            # moving downhill.
            eta = min(eta * 1.02, eta_max)

            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(
                    f"Iter {iteration:6d}: new best score {best_score:.6f} "
                    f"(elapsed {elapsed:.1f}s)"
                )
        else:
            # Reject – decrease step size to become more cautious.
            eta = max(eta * 0.5, eta_min)

        # --------------------------------------------------------------
        # 2.5 Occasional random jitter to shake out of flat regions.
        # --------------------------------------------------------------
        if iteration % 137 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()
            cur_score = _compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(
                    f"Jitter at iter {iteration:6d}: new best {best_score:.6f} "
                    f"(elapsed {elapsed:.1f}s)"
                )

        # Safety break (if anything goes terribly wrong).
        if iteration % 10000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a Python list of floats).
    # ------------------------------------------------------------------
    return best_seq.tolist()
