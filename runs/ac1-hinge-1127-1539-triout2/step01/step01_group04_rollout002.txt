def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Determine n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 15000

    target_sum = math.sqrt(2.0 * n)

    # Initialize a
    if 'height_sequence_1' in globals() and len(height_sequence_1) > 0:
        a = np.array(height_sequence_1, dtype=np.float64)
        if a.shape[0] != n:
            # adjust
            if a.shape[0] > n:
                a = a[:n]
            else:
                # pad zeros
                pad = np.zeros(n - a.shape[0], dtype=np.float64)
                a = np.concatenate([a, pad])
    else:
        idx = np.arange(n, dtype=np.float64)
        a = np.sqrt(idx + 1.0) - np.sqrt(idx)
        a *= target_sum / a.sum()

    # Scale to target sum
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a *= (target_sum / a_sum)

    # Setup for gradient descent
    L = 1 << ((2 * n - 1).bit_length())
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial score: {best_score:.6f}")

    v = np.zeros_like(a) # momentum
    momentum = 0.9
    lr_initial = 0.8
    iteration=0
    last_improve_iter=0

    jitter_interval=200
    swap_interval=500
    stagnation_limit = 2000

    while time.time() < deadline:
        # compute conv
        conv = _conv_fft(a, L)
        max_b = conv.max()
        cur_score = float(max_b)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = iteration
            print(f"Iter {iteration:6d} – new best {best_score:.6f}")
        # gradient
        eps = max_b * 1e-12
        ks = np.where(conv >= max_b - eps)[0]
        grad = np.zeros_like(a)
        for k in ks:
            _add_gradient_for_k(grad, a, k)

        lr = lr_initial / np.sqrt(iteration + 1.0)
        v = momentum * v - lr * grad
        a_candidate = a + v

        # clip + project
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0:
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a = a_candidate * (target_sum / s)

        # jitter
        if iteration % jitter_interval == 0 and iteration > 0:
            sigma = target_sum * 2e-3 / math.sqrt(n)  # somewhat higher sigma
            noise = np.random.randn(n) * sigma
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, target_sum / n, dtype=np.float64)
            else:
                a *= (target_sum / a_sum)

        # occasional swap local move
        if iteration % swap_interval == 0 and iteration > 0:
            # try a few random transfers
            for _ in range(5):
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                # Transfer up to 10% of a_i
                max_transfer = min(a[i], a[i] * 0.1)
                if max_transfer <= 0:
                    continue
                delta = np.random.rand() * max_transfer
                a_new = a.copy()
                a_new[i] -= delta

"""
Propose a candidate step‑function sequence.

The algorithm is a projected momentum gradient descent on the infinity‑norm
of the self‑convolution, which (for a fixed L¹‑norm) equals the score
returned by ``evaluate_sequence``.  It consists of:

* initialization with the square‑root‑difference sequence (a known good
  start);
* an FFT‑based computation of the self‑convolution and its maximal shift;
* a sub‑gradient step for all shifts that are (almost) maximal;
* Euclidean projection onto the simplex {a ≥ 0, Σa = √(2 n)} using scaling
  after clipping (fast and sufficient);
* occasional small Gaussian jitter and cheap local “mass‑transfer” moves
  to escape plateaus;
* a simple momentum term and a decaying learning‑rate  η₀ / √t.

The loop runs until the allotted time budget (≈ 1000 s) is exhausted and
returns the best sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are deliberately defined at top‑level.
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution of a real vector using an FFT of length L.
    Returns the convolution of length (2·len(a)‑1).
    """
    fa = np.fft.rfft(a, n=L)
    # element‑wise product in frequency domain gives linear convolution
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * a.size - 1]


def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """
    Add the sub‑gradient contributed by shift ``k`` to ``grad``.
    For the self‑convolution b_k = Σ a_i·a_{k‑i}, the sub‑gradient
    ∂b_k/∂a_j = a_{k‑j} + a_{j‑k} (when the index is valid).
    The implementation adds both contributions at once.
    """
    n = a.size
    if k < n:
        # indices 0 … k  (the second term is symmetric)
        grad[: k + 1] += 2.0 * a[k :: -1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises the supplied
    ``evaluate_sequence`` score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function stops slightly before this).

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    np.random.seed(seed)
    # a small safety margin prevents being killed exactly at the deadline
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size.
    # Longer sequences give the optimizer more freedom; 15000 is a
    # comfortable default that still fits easily into the time budget.
    # ------------------------------------------------------------------
    n = 15000

    # ------------------------------------------------------------------
    # 1) Initialise with the square‑root‑difference sequence.
    #    After scaling its L¹‑norm equals √(2 n), which makes the
    #    evaluation score equal to the maximal convolution entry.
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1.0) - np.sqrt(idx)               # a_i = √(i+1) – √i
    target_sum = math.sqrt(2.0 * n)                     # required Σa
    a *= target_sum / a.sum()                           # normalise to √(2 n)

    # ------------------------------------------------------------------
    # Prepare bookkeeping.
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())                # power‑of‑two FFT length
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))  # score = max conv
    print(f"Initial score: {best_score:.6f}")

    # Momentum vector (starts from zero)
    velocity = np.zeros_like(a)
    momentum = 0.9
    lr_initial = 0.8                                     # starting step size
    iteration = 0
    last_improve = 0

    # Frequencies of auxiliary operations
    jitter_interval = 200
    swap_interval = 500
    stagnation_limit = 4000

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Compute self‑convolution and identify the maximal shift(s).
        # --------------------------------------------------------------
        conv = _conv_fft(a, L)
        max_b = conv.max()
        cur_score = float(max_b)                         # equals evaluation score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d} – new best {best_score:.6f}")

        # Sub‑gradient of the max‑convolution.
        # --------------------------------------------------------------
        # Use a slightly relaxed tolerance so that several nearly‑maximal
        # shifts contribute to the direction – this yields a smoother descent.
        eps = max_b * 1e-6
        ks = np.where(conv >= max_b - eps)[0]
        grad = np.zeros_like(a)
        for k in ks:
            _add_gradient_for_k(grad, a, k)

        # --------------------------------------------------------------
        # Momentum‑augmented gradient step and projection onto the simplex.
        # --------------------------------------------------------------
        lr = lr_initial / math.sqrt(iteration + 1.0)
        velocity = momentum * velocity - lr * grad
        a_candidate = a + velocity

        # Simple (fast) projection: clip negatives and renormalise the sum.
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a = a_candidate * (target_sum / s)

        # --------------------------------------------------------------
        # Small Gaussian jitter every few hundred iterations.
        # --------------------------------------------------------------
        if iteration % jitter_interval == 0 and iteration > 0:
            sigma = target_sum * 1e-3 / math.sqrt(n)
            noise = np.random.randn(n) * sigma
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            a = a * (target_sum / a_sum)

        # --------------------------------------------------------------
        # Cheap local “mass‑transfer” move to escape flat plateaus.
        # --------------------------------------------------------------
        if iteration % swap_interval == 0 and iteration > 0:
            # Try a handful of random transfers; keep the first improvement.
            for _ in range(5):
                i, j = np.random.randint(0, n, size=2)
                if i == j or a[i] == 0.0:
                    continue
                # Transfer at most 10 % of the mass at i.
                max_transfer = min(a[i], a[i] * 0.1)
                delta = np.random.rand() * max_transfer
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                conv_new = _conv_fft(a_new, L)
                max_new = conv_new.max()
                if max_new < max_b:
                    a = a_new
                    max_b = max_new
                    if max_new < best_score:
                        best_score = max_new
                        best_seq = a.copy()
                        last_improve = iteration
                        print(f"Iter {iteration:6d} – swap improvement {best_score:.6f}")
                    break  # accept first improving swap

        # --------------------------------------------------------------
        # Restart if we have not improved for a long while.
        # --------------------------------------------------------------
        if iteration - last_improve > stagnation_limit:
            print(f"Iter {iteration:6d} – stagnation, re‑initialising.")
            # Blend the current best with a fresh random Dirichlet draw.
            alpha = np.full(n, 0.5)
            random_dir = np.random.gamma(alpha, 1.0)
            random_dir = random_dir / random_dir.sum() * target_sum
            a = 0.5 * best_seq + 0.5 * random_dir
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            velocity.fill(0.0)
            last_improve = iteration

        iteration += 1

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
