"""
Propose a low‑score step‑function.

The algorithm works in three phases:

1. **Length selection** – we try a few candidate lengths (larger
   sequences get closer to the asymptotic bound π/2).  The longest length
   that fits into the time budget is kept.

2. **Coarse grid search** – for the chosen length we scan a fine grid of
   the two parameters of the classic 1/√(i) construction:
        a_i = 1 / (i + 1 + c)^β .
   The best pair (β,c) is kept.

3. **Projected sub‑gradient descent** – the objective
        2·n·max(conv(a,a)) / (sum(a)^2)
   is scale‑invariant, therefore we normalise the sum to 1 and minimise
   `max(conv(a,a))`.  At each step the index `k_max` attaining the
   current maximum is identified; the sub‑gradient of that entry is

        g_i = a_i + a_{k_max-i}      (zero if the second index is out of range)

   The vector is moved a little in the negative direction, all negative
   components are clipped to zero and the vector is re‑normalised to the
   probability simplex.  A diminishing step size together with occasional
   random mass‑exchange perturbations helps to escape plateaus.
   The loop runs until the time budget is exhausted or no further
   improvement is observed.

The whole search is bounded by the supplied `budget_s` (default 1000 s)
and returns the best sequence discovered as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Fast evaluation of the score 2·n·max(conv(seq,seq)) / (sum(seq)^2)
    using a real FFT.  ``seq`` may be any 1‑D array of non‑negative numbers.
    """
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # FFT length – next power of two for speed
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of ``seq`` with itself."""
    n = len(seq)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return conv


def _power_law_sequence(n: int, beta: float, offset: float) -> np.ndarray:
    """
    Produce the classic candidate a_i = 1 / (i + 1 + offset)^beta,
    then normalise the sum to 1 (the value of the sum does not affect the
    final score because the evaluation is scale‑invariant).
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    seq = 1.0 / (idx ** beta)
    seq /= np.sum(seq)                     # normalise to 1
    return seq


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}.
    Implementation follows the algorithm of
    Duchi et al. (2008) – O(n log n) due to a single sorting step.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution as a safe fallback
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget (seconds).  The function stops a few seconds
        before the limit to guarantee a return value.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, no NaNs/Infs).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length.  We try a few candidates, starting from the
    #    largest that comfortably fits into the time budget.
    # ------------------------------------------------------------------
    # Rough estimate (empirical): ~0.08 s per iteration for n=20 000.
    # We therefore start with n=15000; if time permits we fall back to
    # smaller lengths.
    candidate_lengths = [15000, 10000, 5000]
    best_global_seq = None
    best_global_score = np.inf

    for n in candidate_lengths:
        if time.time() > deadline:
            break
        print(f"[+] Searching with length n = {n}")

        # ------------------------------------------------------------------
        # 2. Grid search over (beta, offset).  The classic optimum is at
        #    beta ≃ 0.5; we explore a fine neighbourhood.
        # ------------------------------------------------------------------
        best_seq = None
        best_score = np.inf

        betas = np.linspace(0.48, 0.52, 41)          # step ≈ 0.001
        offsets = np.linspace(0.0, 5.0, 51)          # step ≈ 0.1

        for beta in betas:
            for offset in offsets:
                cand = _power_law_sequence(n, beta, offset)
                score = _objective(cand)
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    print(f"[grid] n={n} beta={beta:.3f} off={offset:.2f} → {score:.6f}")

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

        if best_seq is None:          # very unlikely, but stay safe
            best_seq = np.ones(n) / n
            best_score = _objective(best_seq)

        # ------------------------------------------------------------------
        # 3. Refine with projected sub‑gradient descent.
        #    The sum is kept equal to 1 – this does not change the score.
        # ------------------------------------------------------------------
        a = best_seq.copy()
        target_sum = 1.0
        step_factor = 0.5
        no_improve = 0
        it = 0
        best_local_seq = a.copy()
        best_local_score = best_score

        # Pre‑compute FFT size for the current n (used repeatedly)
        fft_len = 1 << ((2 * n - 1).bit_length())

        while time.time() < deadline and it < 5000:
            # Full convolution of the current iterate
            fft_a = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * n - 1]
            max_conv = np.max(conv)

            # Indices attaining (or almost attaining) the maximum
            # (allow a tiny tolerance to capture flat plateaus)
            tol = 1e-12
            max_idx = np.flatnonzero(np.isclose(conv, max_conv, atol=tol))
            # Aggregate sub‑gradients of all active indices
            grad = np.zeros_like(a)
            for k in max_idx:
                shift_idx = k - np.arange(n)
                mask = (shift_idx >= 0) & (shift_idx < n)
                shifted = np.zeros_like(a)
                shifted[mask] = a[shift_idx[mask]]
                grad += a + shifted
            grad /= len(max_idx)                     # average over the active set

            # Step size – diminishes slowly with the iteration counter
            step = step_factor * max_conv / (it + 1.0)

            new_a = a - step * grad
            new_a = np.maximum(new_a, 0.0)           # keep non‑negative
            new_a = _project_onto_simplex(new_a)     # enforce Σ a_i = 1

            new_score = _objective(new_a)

            # ----------------------------------------------------------------
            # Accept improvement, otherwise apply a small random reshuffle.
            # ----------------------------------------------------------------
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_seq = new_a.copy()
                if new_score < best_global_score:
                    best_global_score = new_score
                    best_global_seq = new_a.copy()
                    print(f"[best] n={n} it={it} score={new_score:.6f}")

                a = new_a
                no_improve = 0
                step_factor = min(step_factor * 1.02, 2.0)   # gently enlarge steps
            else:
                no_improve += 1
                if no_improve % 20 == 0:
                    # random local perturbation to escape a flat region
                    i, j = np.random.randint(n, size=2)
                    delta = np.random.rand() * a[i] * 0.04
                    a[i] = max(0.0, a[i] - delta)
                    a[j] += delta
                    a = _project_onto_simplex(a)
                    step_factor = max(step_factor * 0.9, 0.02)
                else:
                    a = new_a

            it += 1

            # Optional early stopping if we are already very close to the
            # theoretical bound π/2 ≈ 1.570796.
            if best_local_score < 1.5710:
                # we are within a few 1e‑4 of the limit – keep iterating a bit
                # but stop early if the gain stalls
                if no_improve > 100:
                    break

        # keep the best sequence found for this length
        if best_local_score < best_global_score:
            best_global_score = best_local_score
            best_global_seq = best_local_seq.copy()

    # ----------------------------------------------------------------------
    # Fallback – if for any reason nothing was discovered, return a uniform
    # distribution of moderate length.
    # ----------------------------------------------------------------------
    if best_global_seq is None:
        n_fallback = 5000
        best_global_seq = np.ones(n_fallback, dtype=np.float64) / n_fallback

    # Convert to plain Python list of floats as required by the evaluator.
    final_seq = [float(v) for v in best_global_seq]
    # One final sanity check (expensive but cheap for the final size)
    final_score = _objective(np.array(final_seq))
    print(f"[+] Search finished. Best score = {final_score:.6f}")
    return final_seq
