def simplex_projection(v, z):
    """Project vector v onto simplex {x >= 0, sum x = z}."""
    # uses algorithm of Duchi et al. (2008)
    if z < 0:
        raise ValueError
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

But note that this projection algorithm expects v real; it will produce sum exactly z.

def simplex_projection(v, z):
    """Project vector v onto the simplex {x >= 0, sum x = z}."""
    n = v.shape[0]
    # shift to ensure sum positive:
    # Sort in decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho_arr = np.arange(1, n+1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # all values zero? return uniform
        return np.full(n, z / n)
    rho = rho_arr[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Projected Adam optimiser for the self‑convolution max problem.

The algorithm proceeds in three phases:

1. **Parametric warm‑start** –  a small grid search over the
   family  a_i = (i+α+1)^p – (i+α)^p  (α ≥ 0, 0 < p ≤ 1) is performed.
   The best member (after normalisation to the required L¹‑norm
   √(2·n)) is taken as the initial point.  This family includes the
   classic “√‑difference’’ sequence (α=0, p=½) and allows the optimiser
   to start closer to the true optimum.

2. **Projected Adam descent** –  the objective is the infinity norm of
   the self‑convolution, i.e.  max_k (a * a)[k].  At the current active
   shift(s) k* the sub‑gradient is  ∂/∂a_i max = 2·a_{k*–i}.
   Adam’s adaptive moments are used to choose a direction, after which
   the iterate is projected onto the simplex
   {x ≥ 0, Σx = √(2·n)}.  This maintains feasibility at every step.

3. **Exploratory jitter** –  every few hundred iterations a tiny
   zero‑mean Gaussian perturbation is added and the vector is re‑projected.
   The jitter helps to escape flat plateaus.

The loop stops a few seconds before the allocated budget and the best
sequence seen is returned as a plain list.  The final score is the
maximum of the self‑convolution, which is exactly the value that
`evaluate_sequence` returns (because the L¹‑norm is kept fixed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the linear self‑convolution of a real vector `a` using FFTs.
    `L` must be a power of two ≥ 2·len(a)–1.
    """
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * a.size - 1]


def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """
    Add the sub‑gradient contribution for a given active shift `k`
    to the array `grad`.  The contribution is 2·a_{k‑i} for all valid i.
    """
    n = a.size
    if k < n:
        # indices 0 … k  (i runs from 0 to k)
        grad[: k + 1] += 2.0 * a[k :: -1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]


def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of a vector `v` onto the simplex
    { x ≥ 0, Σ x = z } (Duchi et al., 2008).

    Parameters
    ----------
    v : np.ndarray
        Input vector (will not be modified).
    z : float
        Desired L¹‑norm (positive).

    Returns
    -------
    np.ndarray
        Projected vector of the same shape as `v`.
    """
    if z <= 0:
        raise ValueError("Simplex radius must be positive")

    u = np.sort(v)[::-1]               # sorted descending
    cssv = np.cumsum(u) - z
    rho_arr = np.arange(1, v.size + 1)
    cond = u - cssv / rho_arr > 0

    if not np.any(cond):               # degenerate case – fall back to uniform
        return np.full_like(v, z / v.size, dtype=np.float64)

    rho = rho_arr[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _evaluate_max_conv(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] for a given vector a."""
    conv = _conv_fft(a, L)
    return float(conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation function.  The procedure respects the 1000 s time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Choose a length.
    # Prefer the length of a previously found construction if available.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 12000                     # a comfortable size (≈10 k)

    target_sum = math.sqrt(2.0 * n)    # required L¹‑norm (scale‑invariant)

    # --------------------------------------------------------------
    # Pre‑compute FFT length (next power of two ≥ 2·n‑1)
    # --------------------------------------------------------------
    L = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1)  Parametric warm‑start: scan a small grid of (α, p) pairs.
    # --------------------------------------------------------------
    index = np.arange(n, dtype=np.float64)

    best_a = None
    best_score = float("inf")

    # Grid (α ∈ [0, 5] step 0.5, p ∈ [0.35, 0.80] step 0.05)
    alphas = np.linspace(0.0, 5.0, 11)          # 0.0 … 5.0
    ps = np.linspace(0.35, 0.80, 10)           # 0.35 … 0.80

    print("=== Warm‑start grid search ===")
    for alpha in alphas:
        for p in ps:
            # a_i = (i+α+1)^p – (i+α)^p  (always ≥ 0)
            a = (np.power(index + alpha + 1.0, p) -
                 np.power(index + alpha, p))

            # Normalise to the required L¹‑norm
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a = a * (target_sum / a_sum)

            # Evaluate (max‑convolution)
            cur_score = _evaluate_max_conv(a, L)

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"α={alpha:4.2f}, p={p:5.2f} → new best {best_score:.6f}")

    # Fallback: classic sqrt‑difference shape if the grid failed for any reason
    if best_a is None:
        a = np.sqrt(index + 1.0) - np.sqrt(index)
        a = a * (target_sum / a.sum())
        best_a = a.copy()
        best_score = _evaluate_max_conv(best_a, L)
        print("Fallback to sqrt‑difference shape, score:", best_score)

    # Preserve the best ever seen throughout the whole run
    best_seq = best_a.copy()
    overall_best = best_score

    # --------------------------------------------------------------
    # 2) Projected Adam optimisation
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.25                      # initial (raw) learning rate
    m = np.zeros_like(best_a)       # first moment
    v = np.zeros_like(best_a)       # second moment

    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        # ----- compute convolution and current score ----------------
        conv = _conv_fft(best_a, L)
        max_b = float(conv.max())
        cur_score = max_b                      # because Σa = √(2n)

        # ----- keep global best ------------------------------------
        if cur_score < overall_best:
            overall_best = cur_score
            best_seq = best_a.copy()
            last_improve_iter = iteration
            print(f"Iter {iteration:6d} – new overall best {overall_best:.6f}")

        # ----- identify active shift(s) ----------------------------
        tol = 1e-12 * max_b
        active_ks = np.where(conv >= max_b - tol)[0]

        # ----- compute sub‑gradient (negative direction) ----------
        grad = np.zeros_like(best_a)
        for k in active_ks:
            _add_gradient_for_k(grad, best_a, k)
        grad = -grad                            # we want to *reduce* the max

        # ----- Adam moment updates --------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # adaptive step
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- tentative update followed by projection ------------
        a_candidate = best_a + step
        a_candidate = _simplex_projection(a_candidate, target_sum)

        # ----- accept the candidate (always, because projection keeps feasibility)
        best_a = a_candidate

        # ----- jitter every few hundred steps ----------------------
        if iteration % 300 == 0 and iteration > 0:
            sigma = target_sum * 8e-4 / math.sqrt(n)
            noise = np.random.randn(n) * sigma
            best_a = _simplex_projection(np.maximum(best_a + noise, 0.0), target_sum)

        iteration += 1

        # ----- early exit if we have not improved for a long time ---
        if iteration - last_improve_iter > 20000:
            # No improvement for many iterations – stop to save time
            print("No improvement for a long while, terminating early.")
            break

    print(f"Finished after {iteration} iterations – best score {overall_best:.6f}")
    return best_seq.tolist()

* **A parametric warm‑start** that searches a small family of analytically‑motivated shapes
  (shifted power‑law differences).  The best shape from this grid is used as the
  initial point for the optimiser.
* **Projected Adam** – an adaptive learning‑rate scheme that generally converges
  faster than a simple 1/√t schedule.
* **Simplex projection** guaranteeing non‑negativity and the exact L¹‑norm at every step.
* **Periodic jitter** to escape plateaus and a simple early‑stop criterion.

All heavy work (convolution, gradient) is performed with FFTs, keeping the
runtime well within the 1000 s budget even for a few‑ten‑thousand‑element
sequence. The function returns the best sequence discovered, ready to be
