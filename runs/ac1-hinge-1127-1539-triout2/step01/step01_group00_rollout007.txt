    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _softmax_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the soft‑max approximation of max(a∗a).

    Parameters
    ----------
    a : np.ndarray
        Current non‑negative vector (length n, L1‑norm already fixed).
    lam : float
        Soft‑max temperature (larger → closer to the hard max).

    Returns
    -------
    g : np.ndarray
        Gradient w.r.t. a (all entries non‑negative).
    cur_max : float
        The current hard maximum of the convolution (for external logging).
    """
    n = a.shape[0]

    # ----- convolution a*a ------------------------------------------------
    conv = _full_convolution_fft(a)                # length 2n‑1
    cur_max = float(conv.max())

    # ----- soft‑max weights ------------------------------------------------
    # Shift by the max to avoid overflow.
    shifted = lam * (conv - cur_max)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()                 # length 2n‑1, sum = 1

    # ----- gradient = Σ_k w_k ∂(a*a)[k]/∂a -------------------------------
    a_rev = a[::-1]

    # convolution w * a_rev  (length 3n‑2)
    conv_len2 = w.shape[0] + n - 1                 # = (2n‑1)+(n‑1) = 3n‑2
    fft_len2 = 1 << (conv_len2 - 1).bit_length()
    Fw = np.fft.rfft(w, n=fft_len2)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len2)
    conv_w_rev = np.fft.irfft(Fw * Fa_rev, n=fft_len2)[:conv_len2]

    # the off‑diagonal part: 2·(w∗a_rev)[n‑1 … n‑1+n‑1]
    start = n - 1
    g = 2.0 * conv_w_rev[start:start + n]

    # diagonal correction (k = 2·i contributes only a_i, not 2·a_i)
    diag_idx = 2 * np.arange(n)                    # 0,2,4,…,2n‑2  (always valid)
    g -= w[diag_idx] * a

    # all entries are non‑negative; clip tiny numerical noise
    g = np.maximum(g, 0.0)
    return g, cur_max


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative start vector of length n whose L1‑norm equals the
    target √(2·n).  With 50 % probability the supplied global ``height_sequence_1``
    is used (if it exists), otherwise a Dirichlet(1,…,1) sample is drawn.
    """
    S = _target_l1_norm(n)

    use_existing = False
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            use_existing = True
        except Exception:
            use_existing = False

    if use_existing and rng.random() < 0.5:
        a = a.copy()
    else:
        # Dirichlet(1,…,1) = uniform on the simplex
        a = rng.random(n)

    # normalise to the required L1‑norm
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def _optimize_softmax(n: int, seed: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic exponential descent on the smoothed ‑∞‑norm of a∗a.
    Returns the best vector found (as a NumPy array) and its external score.
    """
    rng = np.random.default_rng(seed)
    S = _target_l1_norm(n)

    a = _initial_sequence(n, rng)

    # external evaluation of the start point
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, score={best_score:.6f}")

    # hyper‑parameters
    eta = 0.7                # initial step size (exponential update)
    eta_decay = 0.99996
    lam = 1.0                # start soft‑max temperature
    lam_factor = 1.0005
    lam_max = 20.0
    jitter_prob = 0.015      # occasional Gaussian perturbation

    while time.time() < deadline:
        # gradient of the smoothed max‑norm and the current hard max
        g, cur_max = _softmax_gradient(a, lam)

        # exponential (multiplicative) update – guarantees non‑negativity
        # to avoid numerical under‑flow we clip the exponent magnitude
        step = -eta * np.clip(g, 0.0, 10.0)          # 10 is an arbitrary safe cap
        a = a * np.exp(step)

        # renormalise to the required L1‑norm
        total = a.sum()
        if total == 0.0:                     # pathological – re‑initialise uniformly
            a = np.full(n, S / n)
        else:
            a *= (S / total)

        # tiny random jitter helps to leave tiny plateaux
        if np.random.random() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.008
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # schedule updates
        eta *= eta_decay
        lam = min(lam_max, lam * lam_factor)

        # log improvement (compare with external score)
        if cur_max < best_score - 1e-9:
            ext_score = evaluate_sequence(a.tolist())
            if ext_score < best_score:
                best_score = ext_score
                best_seq = a.copy()
                print(f"[improve] time={time.time():.1f}s  score={best_score:.6f}")

    return best_seq, best_score

# Public entry point required by the challenge

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best bound.  The routine runs for at most ``budget_s`` seconds.
    """
    # safety margin for housekeeping
    start = time.time()
    deadline = start + budget_s - 0.3

    # Prefer the length of the known construction if it exists.
    n = 5641
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)
        except Exception:
            pass

    best_seq, best_score = _optimize_softmax(n, seed, deadline)

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
