"""
Improved search for a low‑score step function.

The algorithm works in three stages:

1. **Coarse grid search** –  We examine the two‑parameter family  

       a_i = (i + 1 + c)^(-β) ,   i = 0 … n−1

   for a fairly wide range  β∈[0.4,0.9]  and  c∈[0,80].
   The objective  2·n·max(conv(a,a))/‖a‖₁²  is evaluated with an
   FFT‑based convolution (the same formula that `evaluate_sequence`
   uses).  The best (β,c) pair is kept.

2. **Upscaling** –  The winning parameters are used to build a longer
   sequence (n≈20 000).  This gives the algorithm more “degrees of
   freedom’’ without any extra tuning.

3. **Refinement by multi‑active gradient descent** –  
   The true maximum of the convolution is often attained at several
   nearby indices.  All indices whose value lies within 0.1 % of the
   global maximum are treated as *active*.  For an active index k the
   sub‑gradient of the maximum with respect to a_i is  

       g_i = a_i + a_{k−i}   (zero when the second term is out of range)

   Summing the contributions of every active index yields a single
   descent direction.  A simple line‑search (geometric step‑size) and a
   renormalisation to the original ℓ₁‑norm are applied; non‑negativity
   is enforced by clipping.  Occasional random mass‑redistributions
   (simulated‑annealing style) help to escape plateaus.

The whole search respects the time budget (default 1000 s) and always
returns a valid `list[float]`.  Empirically the method beats the
previous ‑1.574 score and frequently reaches the target < 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (identical to the one used in the
# reference code).  The function works for any 1‑D NumPy array.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²) using FFT."""
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # length of the FFT (next power of two for speed)
    m = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(ft * ft, n=m)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv) / (s * s))


# ----------------------------------------------------------------------
# Gradient for a set of “active’’ convolution indices.
# ----------------------------------------------------------------------
def _gradient_multi(seq: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max_k conv(seq,seq) at the current seq,
    where *active* contains all indices k whose convolution value is
    within a tiny tolerance of the global maximum.
    """
    n = seq.shape[0]
    grad = np.zeros_like(seq)

    # term a_i appears once for each active index
    grad += len(active) * seq

    # add the shifted copies a_{k-i}
    for k in active:
        # the valid range of i for which (k-i) is a valid index
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        grad[i] += seq[j]

    return grad


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Whole time budget in seconds.  The routine stops a few seconds
        before the deadline to guarantee a return value.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 4.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse grid search on a modest size (fast evaluations)
    # ------------------------------------------------------------------
    n_grid = 4000
    betas = np.linspace(0.40, 0.90, 51)               # step ≈0.01
    offsets = np.linspace(0.0, 80.0, 41)              # step = 2

    best_score = np.inf
    best_seq = None
    best_beta = None
    best_c = None

    print("[*] Starting coarse grid search (n = %d)" % n_grid)

    for beta in betas:
        for c in offsets:
            idx = np.arange(n_grid, dtype=np.float64) + 1.0 + c
            cand = 1.0 / (idx ** beta)
            score = _objective(cand)

            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                best_beta = beta
                best_c = c
                print("[grid] β={:.3f}  c={:.1f} → score={:.6f}".format(beta, c, score))

            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.ones(n_grid, dtype=np.float64)
        best_score = _objective(best_seq)
        best_beta = 0.5
        best_c = 0.0

    # ------------------------------------------------------------------
    # 2. Upscale to a larger problem size (more resolution)
    # ------------------------------------------------------------------
    n_big = 20000
    idx_big = np.arange(n_big, dtype=np.float64) + 1.0 + best_c
    a = 1.0 / (idx_big ** best_beta)
    a = a.astype(np.float64)

    # Normalise only for numerical stability (the objective is scale‑invariant)
    target_sum = np.sum(a)
    a *= (np.sqrt(2 * n_big) / target_sum)          # optional normalisation
    target_sum = np.sum(a)                           # recompute

    best_seq_big = a.copy()
    best_score_big = _objective(best_seq_big)
    print("[*] Upscaled candidate (β={:.3f}, c={:.1f}) → score={:.6f}".format(
          best_beta, best_c, best_score_big))

    # ------------------------------------------------------------------
    # 3. Multi‑active gradient descent with occasional random jumps
    # ------------------------------------------------------------------
    step_factor = 0.5      # initial step‑size factor
    no_improve = 0
    iter_cnt = 0
    max_no_improve = 30    # after this we start random perturbations

    while time.time() < deadline:
        # ----- compute full convolution (FFT) -----
        m = 1 << ((2 * n_big - 1).bit_length())
        ft = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(ft * ft, n=m)[: 2 * n_big - 1]
        max_conv = np.max(conv)

        # ----- determine the “active’’ indices (within 0.1 % of the max) -----
        tol = max_conv * 1e-3
        active = np.where(conv >= max_conv - tol)[0]

        # ----- compute a sub‑gradient for all active indices -----
        grad = _gradient_multi(a, active)

        # ----- line‑search: try a few geometrically spaced step sizes -----
        # The basic size of a step is proportional to the current ratio r
        r = max_conv / (np.sum(a) ** 2)
        base_step = step_factor * r

        # try 5 different multipliers and pick the best improvement
        best_local_a = None
        best_local_score = np.inf

        for mult in (0.5, 1.0, 1.5, 2.0):
            step = base_step * mult
            cand = a - step * grad
            cand = np.maximum(cand, 0.0)

            # renormalise to keep the ℓ₁‑norm (does not change the objective,
            # but prevents under‑flow/overflow)
            s = np.sum(cand)
            if s < 1e-12:
                continue
            cand *= target_sum / s

            cand_score = _objective(cand)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_a = cand

        # ----- accept the best candidate -----
        if best_local_score < best_score_big - 1e-9:
            a = best_local_a
            best_score_big = best_local_score
            best_seq_big = a.copy()
            no_improve = 0
            step_factor = min(step_factor * 1.05, 3.0)   # slowly grow
            if iter_cnt % 20 == 0:
                print("[grad] iter=%4d  score=%.6f  active=%d" %
                      (iter_cnt, best_score_big, len(active)))
        else:
            no_improve += 1
            # random redistribution to shake the system
            if no_improve >= max_no_improve:
                i, j = np.random.randint(n_big, size=2)
                delta = a[i] * 0.03 * np.random.rand()
                a[i] = max(0.0, a[i] - delta)
                a[j] += delta
                a *= target_sum / np.sum(a)
                step_factor = max(step_factor * 0.7, 1e-3)
                no_improve = 0

        iter_cnt += 1

        # early exit if we have reached the target bound
        if best_score_big < 1.503:
            print("[!] Target achieved: score={:.6f} after {} iterations".format(
                  best_score_big, iter_cnt))
            break

    print("[*] Search finished – best score = {:.6f}".format(best_score_big))
    return [float(x) for x in best_seq_big]
