def _lp_mix(a, sub_deadline, t=0.2, max_iter=5):
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = a.copy()
    while time.time() < sub_deadline and max_iter > 0:
        M = _max_convolution(a)
        # Build A_ub
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            col_idx = np.arange(j_start, j_end + 1)
            A[k, col_idx] = a[k - col_idx]
        # constraints
        b = np.full(rows, M, dtype=np.float64)
        # Solve LP
        # cvxopt / scipy linprog
        from scipy.optimize import linprog
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 1.0}) # time limit per solve=1 sec
        if not res.success:
            # cannot find improvement
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        # scale g to same L1
        g_scaled = g * (target_sum / sum_g)
        # mix
        a = (1 - t) * a + t * g_scaled
        max_iter -= 1
        # optional progress
        new_score = _score(a)
        print(f"[{time.time():.1f}] LP mix step score = {new_score:.6f}")
    return a

"""
Propose a step‑function (non‑negative list) that minimises

        2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm combines several simple optimisation ideas:

1. **Power‑law initialisation** – a_i ∝ (i+1)^(-p) with several exponents.
2. **Fast hill‑climbing** – random mass‑transfer moves that keep the ℓ₁‑norm
   fixed (the sum is kept at √(2·n)).
3. **Linear‑programming mixing** – given the current vector `f`, solve the LP
   “maximise Σg_j  s.t.  g≥0  and  f∗g ≤ M”, where `M = max(f∗f)`.  Scaling the
   LP solution and mixing it back guarantees a non‑increase of the objective.
4. **Projected sub‑gradient descent** – the sub‑gradient of `max(conv)` is
   analytically known; a step opposite to it is taken and the result is
   projected onto the simplex (non‑negative, fixed sum).

All phases respect a global time budget (default = 1000 s).  The best
sequence seen during the run is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a (FFT implementation)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Exact score used by the competition (sum is fixed → just max‑conv)."""
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _initial_sequence(p: float, n: int) -> np.ndarray:
    """
    Power‑law start a_i ∝ (i+1)^{-p} and normalise the ℓ₁‑norm to √(2·n).
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project v onto the simplex {x ≥ 0, Σx = target_sum}.
    Implementation of the classic O(n log n) algorithm.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - target_sum) / (np.arange(len(v)) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - target_sum) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (keeps the ℓ₁‑norm fixed)
# ----------------------------------------------------------------------


def _local_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that transfers a random fraction of mass from i → j.
    Only strictly improving moves are kept.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.18                # up to 18 % of the source entry
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  hill‑climb improved = {best_val:.6f}")
        else:
            # reject
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# LP‑mixing step (the “linear‑programming improvement” described in the
# statement).  A handful of iterations are usually enough to push the
# bound noticeably lower.
# ----------------------------------------------------------------------


def _build_A_ub(a: np.ndarray) -> np.ndarray:
    """
    Build matrix A (size (2n‑1)×n) where
        (A @ g)_k = Σ_i a_i * g_{k-i}   (linear convolution a∗g)_k .
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    for k in range(rows):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        cols = np.arange(j_start, j_end + 1)
        A[k, cols] = a[k - cols]                     # a_{k‑j}
    return A


def _lp_mix(a: np.ndarray, sub_deadline: float,
            mix_factor: float = 0.25, max_iters: int = 4) -> np.ndarray:
    """
    Perform a few LP‑mixing iterations:
        maximise Σ g_j   s.t.   g ≥ 0  and  a∗g ≤ M,
    where M = max(a∗a).  The solution is scaled to the required ℓ₁‑norm
    and mixed back with the current vector.
    """
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = a.copy()

    while time.time() < sub_deadline and max_iters > 0:
        M = _max_convolution(a)
        A = _build_A_ub(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # linprog solves   min cᵀx   →   we minimise –Σg_j.
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # Scale to the same ℓ₁‑norm and mix.
        g_scaled = g * (target_sum / sum_g)
        a = (1.0 - mix_factor) * a + mix_factor * g_scaled
        max_iters -= 1

        cur_score = _score(a)
        print(f"[{time.time():.1f}]  LP‑mix step score = {cur_score:.6f}")

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------


def _subgradient_descent(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the function max(conv(a,a)).
    Each iteration moves opposite to the average sub‑gradient of all
    active (maximal) convolution entries and then projects onto the simplex.
    """
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    it = 0

    while time.time() < deadline:
        conv = np.convolve(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # average sub‑gradient over active indices
        grad = np.zeros_like(a)
        for k in active:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # a[i_min … i_max] reversed gives a_{k‑i}
            seg = a[i_min:i_max + 1][::-1] * 2.0
            grad[i_min:i_max + 1] += seg
        grad /= max(1, len(active))

        # diminishing step size
        it += 1
        step = 0.4 * target_sum / (it ** 0.5)

        a_new = _project_to_simplex(a - step * grad, target_sum)

        # accept if we actually improve the objective
        if _score(a_new) < _score(a):
            a = a_new
            print(f"[{time.time():.1f}]  sub‑grad improved = {_score(a):.6f}")

    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a sequence of non‑negative heights that minimises the

    described above.
    """

    deadline = time.time() + budget_s - 1.0        # small safety margin

    best_score = np.inf

    # ------------------------------------------------------------------

            print(f"Could not use height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # 2) Systematic exploration over a modest set of lengths / exponents.
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400]          # stay below ~2 000 000 entries in A_ub
    exponents = [0.44, 0.46, 0.48, 0.50, 0.52, 0.54]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        for p in exponents:
            if time.time() >= deadline:
                break

            # ---- initialise -------------------------------------------------
            a = _initial_sequence(p, n)

            # small random perturbation to increase diversity
            for _ in range(2):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                frac = rng.random() * 0.1
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta

            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Init (n={n}, p={p}) score={cur_score:.6f}")

            # ---- local stochastic hill‑climb --------------------------------
            local_deadline = min(deadline, time.time() + 4.0)
            a_hc, hc_score = _local_search(a, local_deadline)
            if hc_score < best_score:
                best_score = hc_score
                best_seq = a_hc.copy()
                print(f"Hill‑climb (n={n}, p={p}) best={hc_score:.6f}")

            # ---- LP mixing --------------------------------------------------
            mix_deadline = min(deadline, time.time() + 3.0)
            a_mix = _lp_mix(a_hc, mix_deadline, mix_factor=0.25, max_iters=4)
            mix_score = _score(a_mix)
            if mix_score < best_score:
                best_score = mix_score
                best_seq = a_mix.copy()
                print(f"LP‑mix (n={n}, p={p}) score={mix_score:.6f}")

            # ---- sub‑gradient refinement ------------------------------------
            sub_deadline = min(deadline, time.time() + 2.0)
            a_sub = _subgradient_descent(best_seq, sub_deadline)
            sub_score = _score(a_sub)
            if sub_score < best_score:
                best_score = sub_score
                best_seq = a_sub.copy()
                print(f"Sub‑grad (n={n}) score={sub_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)
        best_score = _score(best_seq)
        print(f"Fallback used, score={best_score:.6f}")

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"Final best score ≈ {best_score:.6f} with length {best_seq.size}")
    return best_seq.tolist()
