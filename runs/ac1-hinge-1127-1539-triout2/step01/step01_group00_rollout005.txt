"""
Propose a step‑function with a very low self‑convolution ∞‑norm.

Algorithm
---------
1. Choose a moderate length `n` (here 1500) – large enough to allow a
   non‑trivial distribution but small enough that a dense linear program
   (variables = n, constraints ≈ 2 n) fits comfortably in memory and time.
2. Initialise a random non‑negative vector `f` with ℓ¹‑norm
   `S = sqrt(2·n)`.  With this normalisation the external score returned by
   ``evaluate_sequence`` equals the maximal entry of the linear convolution
   `f*f`.
3. Iterate the following *LP‑step* until the time budget expires:

   * Compute `M = max(f*f)` (FFT based, O(n log n)).
   * Solve the linear program

         maximise   sum(g)
         subject to  (f * g)[k] ≤ M   for all k = 0…2n‑2
                     g ≥ 0

     The convolution constraints are linear in the unknown vector `g`.
     The LP is built with a dense matrix `A` such that `A @ g = f*g`.
   * Scale the LP solution to the required ℓ¹‑norm:
         g ← (S / sum(g)) · g
   * If the new self‑convolution `max(g*g)` is strictly smaller than `M`,
     replace `f` by `g`.  Every accepted replacement reduces the external
     score, so the procedure monotonically improves the bound.

4. When the LP iterations stop (either because no further improvement is
   possible or the time budget is close), run a short projected
   multiplicative‑gradient descent (the same scheme used in the reference
   implementation) to fine‑tune the current best vector.
5. Return the best vector found.  All entries are non‑negative, the sum is
   exactly `sqrt(2·n)`, and the function respects the evaluation‑routine
   limits.

The combination of a global LP move (which quickly spreads the mass while
respecting the current convolution ceiling) and a local gradient descent
yields a markedly lower objective than the baseline 2.0 and improves on the
previous 1.5308 → target < 1.503.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Core numerical helpers
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the linear convolution a*a using FFT, returning the full
    convolution array, its maximal value and the index where the maximum occurs.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A with shape (2n‑1, n) such that
    A @ g == np.convolve(f, g, mode='full').
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j holds a copy of `f` shifted down by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_step(f: np.ndarray, M: float, S: float, warm_start_g=None):
    """
    Solve the linear program described in the docstring.
    Returns a new vector g (scaled to ℓ¹‑norm = S) or None on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    # one inequality per convolution entry
    constraints = [A @ g <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # warm start improves speed when the same problem size is solved repeatedly
    if warm_start_g is not None:
        g.value = warm_start_g

    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

    if g.value is None:
        return None, None

    g0 = np.array(g.value, dtype=np.float64)
    total = g0.sum()
    if total <= 0.0:
        return None, None
    g_scaled = g0 * (S / total)
    return g_scaled, g0  # also return the raw solution for warm‑starting


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One multiplicative gradient step that reduces the current maximal
    convolution entry.  The step size `eta` is a positive scalar.
    """
    n = a.shape[0]
    _, _, k = _conv_max(a)

    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]

    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]   # diagonal term

    a_new = a * np.exp(-eta * g)

    # re‑normalise to the required ℓ¹‑norm sqrt(2·n)
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)

    return a_new


def _gradient_descent_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Short multiplicative‑gradient descent starting from `a`.  Runs until `deadline`.
    Returns the best vector encountered together with its external score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[refine init] score={best_score:.6f}")

    eta = 0.5
    decay = 0.9999
    jitter_prob = 0.01

    while time.time() < deadline:
        a = _gradient_step(a, eta)

        # fast surrogate: the current max of the convolution
        _, cur_max, _ = _conv_max(a)

        if cur_max < best_score:
            ext = evaluate_sequence(a.tolist())
            if ext < best_score:
                best_score = ext
                best_seq = a.copy()
                print(f"[refine improve] time={time.time():.1f}s  score={best_score:.6f}")

        eta *= decay

        # occasional jitter to escape tiny plateaus
        if np.random.random() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()

    return best_seq, best_score

        # ------------------------------------------------------------------
        max_lp_iters = 40
        warm_g = None
        for it in range(max_lp_iters):
            if time.time() >= deadline:
                break

            # current bound M = max self‑convolution
            _, M, _ = _conv_max(a)

            # solve LP: maximise ℓ¹‑norm of g under the current bound M
            g, warm_g = _lp_step(a, M, S, warm_start_g=warm_g)
            if g is None:
                print(f"[restart {restart_id}] LP failed at iter {it}")
                break

            # compute new bound after rescaling
            _, M_new, _ = _conv_max(g)

            # accept only if we made progress
            if M_new < M - 1e-12:
                a = g
                if M_new < best_score:
                    best_score = M_new
                    best_seq = a.copy()
                    print(f"[LP improve r{restart_id} it{it}] score={best_score:.6f}")
                # keep the LP solution as warm start for the next iteration
                warm_g = g
            else:
                # no further reduction possible for this restart
                break

        # ------------------------------------------------------------------
        # Fine‑grained gradient descent (uses far fewer evaluations)
        # ------------------------------------------------------------------
        remaining_time = deadline - time.time()
        if remaining_time > 5.0:          # leave a couple of seconds for final I/O
            a_refined, refined_score = _gradient_descent_refine(a, deadline)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = a_refined.copy()
                print(f"[final refine] new best score={best_score:.6f}")

        # If we already beat the target, we can stop early
        if best_score < 1.503:
            print("[target reached] exiting early")
            break

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    if best_seq is None:
        # Very unlikely – fall back to a uniform vector
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
