The objective is  max(conv(a,a))  with the ℓ₁–norm fixed to √(2·n).  

We minimise the smooth surrogate 
    f(a)=∑_k w_k·(a*a)_k ,   w_k∝exp(λ·(a*a)_k) 
which approaches the true max when λ grows.  The gradient of f is obtained
by a single FFT‑based correlation.  After each gradient step a = a–η∇f
the vector is projected onto the non‑negative simplex {a≥0,∑a=√(2·n)}.
The schedule slowly raises λ (from 5 to ≈80) while the step size η is
adaptively reduced when a trial does not improve the true bound.
Several random restarts with power‑law initialisations are performed
until the time budget (≈1000 s) expires.  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _initial_power_law(p: float, n: int) -> np.ndarray:
    """Generate a_i ∝ (i+1)^{-p} and renormalise to sum √(2·n)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    target = np.sqrt(2.0 * n)
    a *= target / a.sum()
    return a


def _max_convolution_fft(a: np.ndarray, M: int) -> float:
    """Maximum value of the full convolution a*a using FFT size M."""
    A = np.fft.rfft(a, n=M)
    conv = np.fft.irfft(A * A, n=M)[: 2 * a.size - 1]
    return float(conv.max())


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x ≥ 0, ∑x = target_sum}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    # shift to ensure non‑negativity
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, target_sum / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
