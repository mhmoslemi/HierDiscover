"""
Search for a low‑evaluation step function using a two‑parameter family
a_i = ((i + γ + 1)^p - (i + γ)^p) and a fine grid on the parameters.
The algorithm proceeds in several stages:
1. A coarse grid on a modest length (n≈2500) exploring p∈(0.1,0.55) and
   the offset ratio α = γ/n in [0,2].
2. The best few candidates are refined on a medium length (n≈6000) with
   a fine mesh around the coarse optimum.
3. The final sequence is built on a large length (≈20000) by scaling
   γ proportionally to the length.  A short stochastic “mass‑transfer”
   local search is performed until the time budget expires.
The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a non‑negative L1‑normalised sequence of length ``n``.
    The coefficients are the differences of the power function
    (i + gamma + 1)^p - (i + gamma)^p.
    """
    i = np.arange(n, dtype=np.float64)
    # power‑law differences
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    # normalising constant so that sum = 1
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # avoid division by zero (p>0 guarantees total>0 for n>0)
    a /= total
    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score.

    The search uses a coarse–fine grid over the two parameters of the
    ``_generate_sequence`` family and a short stochastic refinement.
    ``budget_s`` is the maximum running time in seconds.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0    # keep a small safety margin

    # ----------------------------------------------------------------------
    # 1️⃣  Coarse grid on a modest length
    # ----------------------------------------------------------------------
    n_small = 2500
    best_score = float('inf')
    best_params = (0.2, 0.0)      # (p, gamma)
    top_k = 12
    top = []                     # list of (score, p, gamma)

Search for a low‑evaluation step function using a two‑parameter family.

The sequence is defined by the differences of a power function

    a_i = (i + γ + 1)^p – (i + γ)^p ,   i = 0 … n‑1

and is L1‑normalised (∑ a_i = 1).  The algorithm works in three phases:

1. **Coarse grid** (n≈2500) – exhaustive search over p∈[0.07,0.55] and the
   offset ratio α = γ / n ∈ [0,2].  The K best candidates are kept.
2. **Fine refinement** (n≈6000) – for each of the K candidates a fine
   mesh around (p,γ) is explored; the best parameters are retained.
3. **Final construction** (n≈18000) – the best (p,γ) are scaled to the
   larger length, a sequence is built and a short stochastic “mass‑transfer”
   local search is performed until the time budget expires.

The returned list can be fed directly to the provided `evaluate_sequence`
function.
"""

import time
import random
import numpy as np

def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a non‑negative L1‑normalised sequence of length ``n``.
    The coefficients are the differences of the power function
    (i + gamma + 1)^p – (i + gamma)^p.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    total = np.power(n + gamma, p) - np.power(gamma, p)

    return a
