        seq = _generate_sequence(n, p, gamma)
        return evaluate_sequence(seq.tolist())
    # Phase1 grid coarse

"""
Propose a low‑evaluation step‑function.

The algorithm builds on the analytical family
    a_i = ((i + γ + 1) ** p - (i + γ) ** p)   (i = 0 … n‑1)

which, after normalising to L1 = 1, gives the asymptotic value
    2·p²·B(p,p)   (B is the Beta function).
For small p this constant can be far below the target 1.503.

The search proceeds in three stages inside the allotted time
(≈1000 s) :

1️⃣ Coarse grid search on a modest length (n≈2500) over a wide
   range of p∈[0.05,0.5] and γ∈[0,300] – the evaluation is cheap
   because convolution is O(n²) with n small.

2️⃣ Local stochastic refinement of the best (p,γ) pair found
   (±0.01 in p, ±2 in γ).  The length stays the same, allowing many
   cheap trials.

3️⃣ Construction of a long final sequence (n≈8000) using the best
   parameters, followed by a very light “mass‑transfer’’ hill‑climb
   on the individual heights (pick two indices and move a tiny δ).
   Each trial re‑evaluates the full sequence; the loop stops when the
   time budget is exhausted.

All generated numbers are non‑negative; the routine never returns
`np.inf` unless an internal error occurs.  The returned list can be
used directly by the evaluation harness.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: generate a normalised non‑negative sequence of length n
# --------------------------------------------------------------
def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return the power‑law‑difference sequence normalised to L1 = 1."""
    i = np.arange(n, dtype=float)
    # a_i = (i + gamma + 1)^p - (i + gamma)^p
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    # the telescoping sum is (n + gamma)^p - gamma^p
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # normalise
    a /= total
    return a


# --------------------------------------------------------------
# Main entry point required by the framework
# --------------------------------------------------------------
def propose_candidate(budget_s: float = 1000, seed: int = 123) -> list[float]:
    """
    Search for a step‑function with the smallest possible value of
    `evaluate_sequence`.  The function respects the 1000 s time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Coarse grid search (fast, short vectors)
    # ------------------------------------------------------------------
    n_small = 2500                 # short enough for many evaluations
    best_score = float("inf")
    best_p = 0.3
    best_gamma = 10.0

    # p in [0.05, 0.5] – 46 points (step ≈0.01)
    p_grid = np.linspace(0.05, 0.5, 46)
    # gamma from 0 to 300 (step 5)
    gamma_grid = list(range(0, 301, 5))

    print("[phase 1] starting coarse grid search")
    for p in p_grid:
        for gamma in gamma_grid:
            if time.time() > deadline:
                break
            seq = _generate_sequence(n_small, float(p), float(gamma))
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_p, best_gamma = float(p), float(gamma)
                print(f"[grid] new best: score={score:.6f}, p={p:.4f}, γ={gamma}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Local stochastic refinement of (p, γ)
    # ------------------------------------------------------------------
    print("[phase 2] refining parameters")
    while time.time() < deadline:
        # propose a neighbour in the (p,γ) space
        p_cand = best_p + (np.random.rand() * 2 - 1) * 0.015   # ±0.015
        g_cand = best_gamma + (np.random.rand() * 2 - 1) * 3.0  # ±3
        if p_cand <= 0.0 or p_cand >= 1.0:
            continue
        if g_cand < 0.0:
            continue
        seq = _generate_sequence(n_small, p_cand, g_cand)
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_p, best_gamma = p_cand, g_cand
            print(f"[refine‑params] improved: score={score:.6f}, p={p_cand:.4f}, γ={g_cand:.2f}")

    # ------------------------------------------------------------------
    # 3️⃣ Build the final (long) sequence and perform a tiny hill‑climb
    # ------------------------------------------------------------------
    n_big = 8000                # longer vector – still fast enough
    print(f"[final] building length‑{n_big} sequence (p={best_p:.4f}, γ={best_gamma:.2f})")
    seq = _generate_sequence(n_big, best_p, best_gamma)
    try:
        best_score = evaluate_sequence(seq.tolist())
    except Exception:
        best_score = float("inf")
    print(f"[final] initial score = {best_score:.6f}")

    # Light coefficient‑wise refinement (mass transfer)
    # ---------------------------------------------------
    # Move a tiny amount δ from index j to i while keeping the L1 norm unchanged.
    # The step size is deliberately tiny to stay close to the current optimum.
    while time.time() < deadline:
        i, j = np.random.choice(len(seq), 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-5   # signed δ ∈ [‑5e‑6, 5e‑6]
        if seq[i] + delta < 0 or seq[j] - delta < 0:
            continue
        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            seq = cand
            print(f"[refine‑coeff] new best score = {best_score:.6f}")

    print(f"[done] best found score = {best_score:.6f} (length = {len(seq)})")
    return seq.tolist()
