def _optimize_local(a, deadline, rng):
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    conv, cur_max, idx = _conv_max(a)
    best_score = cur_max

    eta = 0.5
    decay = 0.9995
    move_prob = 0.2
    epsilon_move = 0.05
    jitter_prob = 0.01

    while time.time() < deadline:
        if rng.random() < move_prob:
            # random move
            partner = idx - np.arange(n)
            valid = (partner >= 0) & (partner < n)
            partner_idxs = np.where(valid)[0]
            non_partner_idxs = np.where(~valid)[0]
            if partner_idxs.size > 0 and non_partner_idxs.size > 0:
                i = rng.choice(partner_idxs)
                j = rng.choice(non_partner_idxs)
                delta = epsilon_move * a[i]  # small portion
                if delta > a[i]:
                    delta = a[i]
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # compute new max
                conv_new, new_max, new_idx = _conv_max(a_new)
                if new_max <= cur_max:
                    a = a_new
                    cur_max = new_max
                    idx = new_idx
                    if new_max < best_score:
                        best_score = new_max
                        best_a = a.copy()
                # else reject
        else:
            # gradient step
            a = _gradient_step(a, eta)
            conv, cur_max, idx = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
        # jitter occasionally
        if rng.random() < jitter_prob:
            noise = rng.normal(scale= a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0, None)
            a = a / a.sum() * S
        # decay eta
        eta *= decay
    return best_a, best_score

import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the linear convolution a*a using an FFT, returning the full
    convolution array, its maximum value and the index of that maximum.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One multiplicative sub‑gradient step that tries to lower the maximal
    entry of the self‑convolution.
    """
    n = a.shape[0]
    # get index of the maximal convolution entry
    _, _, k = _conv_max(a)

    # compute the sub‑gradient of that entry
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    # non‑diagonal contributions (appear twice in the convolution sum)
    g[valid] = 2.0 * a[partner[valid]]

    # diagonal contribution (only once)
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]

    # multiplicative update – guarantees non‑negativity
    a_new = a * np.exp(-eta * g)

    # re‑normalise to the required L1‑norm (√(2·n))
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:                     # safeguard – should never happen
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new


def _initial_sequence(base_seq: np.ndarray,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial non‑negative sequence of length n and L1‑norm
    √(2·n).  If a base_seq is supplied and long enough we use its leading
    part (with a little Gaussian jitter), otherwise we draw from a
    Dirichlet (uniform) distribution.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # a tiny jitter helps diversity across restarts
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


def _optimize_local(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimiser that mixes multiplicative sub‑gradient steps with
    simulated‑annealing random swaps.  Returns the best sequence seen
    (as a NumPy array) together with its maximal convolution value.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best
    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    # hyper‑parameters
    eta = 0.6                     # initial step‑size for the gradient step
    decay = 0.9999                # exponential decay of η
    move_prob = 0.15              # probability of attempting a random swap
    epsilon_move = 0.05           # fraction of mass moved in a swap
    jitter_prob = 0.01            # occasional Gaussian jitter
    temp = cur_max * 0.1          # temperature for simulated annealing
    temp_decay = 0.9997

    while time.time() < deadline:
        if rng.random() < move_prob:
            # ---- simulated‑annealing random move ----
            partner = cur_idx - np.arange(n)
            valid = (partner >= 0) & (partner < n)
            partner_idx = np.where(valid)[0]
            non_partner_idx = np.where(~valid)[0]

            if partner_idx.size and non_partner_idx.size:
                i = rng.choice(partner_idx)
                j = rng.choice(non_partner_idx)

                # amount to transfer – small fraction of the source entry
                delta = epsilon_move * a[i]
                delta = min(delta, a[i])               # keep non‑negative

                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta

                # evaluate candidate
                _, cand_max, cand_idx = _conv_max(a_candidate)

                # accept if better or with a probability depending on temperature
                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    a = a_candidate
                    cur_max = cand_max
                    cur_idx = cand_idx
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                # temperature cooling
                temp *= temp_decay
        else:
            # ---- multiplicative gradient step ----
            a = _gradient_step(a, eta)
            _, cur_max, cur_idx = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        # occasional jitter to shake the system
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum() * S

        # decay the gradient step size
        eta *= decay

    return best_a, best_score


def propose_candidate(seed: int = 42,

    """

    deadline = time.time() + budget_s - 5.0           # small safety margin

    base_seq = None
    if 'height_sequence_1' in globals():

    while time.time() < deadline:
        attempt += 1

        # choose a length – we vary it a bit to explore the size trade‑off
        if base_seq is not None:

        else:
            n = rng.integers(800, 4000)

        slice_deadline = min(time.time() + 30.0, deadline)

        # keep the best overall result

            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} "
