def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # choose n
    default_n = 5641
    max_n_allowed = 50000
    # if we have enough time left after initial random search, use large n
    n = min(max_n_allowed, default_n*3) # e.g., 15000
    # optionally adjust if time less

    target_sum = np.sqrt(2.0 * n)

    # helper to create sequence from parameters
    def seq_from_params(p, beta):
        arr = (np.arange(n, dtype=np.float64) + beta) ** (-p)
        # ensure nonnegative
        arr = np.maximum(arr, 0.0)
        # normalise
        s = arr.sum()
        if s == 0:
            return np.full_like(arr, target_sum / n)
        arr *= target_sum / s
        return arr

    # random search
    best_a = None
    best_score = np.inf
    num_trials = 200
    for i in range(num_trials):
        # sample p near 0.5
        p = np.random.uniform(0.4, 0.7)
        beta = np.random.uniform(0.0, 3.0)
        a = seq_from_params(p, beta)
        score = _compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a
            print(f"rand trial {i}: p={p:.4f} beta={beta:.4f} score={score:.6f}")

        if time.time() > deadline:
            break

    # if no best found (unlikely), fallback to existing height_sequence_1
    if best_a is None:
        try:
            best_a = np.array(height_sequence_1, dtype=np.float64)
            # Resize to length n by truncating or padding zeros
            if len(best_a) > n:
                best_a = best_a[:n]
            elif len(best_a) < n:
                best_a = np.pad(best_a, (0, n - len(best_a)), constant_values=0.0)
        except Exception:
            # fallback uniform
            best_a = np.full(n, target_sum / n, dtype=np.float64)

        best_score = _compute_score(best_a)

    # gradient descent subgradient
    eta = 0.15
    eta_min = 0.001
    decay = 0.9995
    L = 3  # top positions weight
    iteration = 0
    momentum = np.zeros_like(best_a)
    mu = 0.9

    # pre-allocated arrays for conv? compute each iteration anyway.
    while time.time() < deadline:
        iteration += 1

        # compute convolution
        af = np.fft.rfft(best_a, n=_next_pow2(2*n-1))
        conv = np.fft.irfft(af * af, n=_next_pow2(2*n-1))[:2*n-1]

        # find top L indices
        # Use argpartition for speed.
        if L >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -L)[-L:]
        # compute gradient as average of gradients from each top index
        grad = np.zeros_like(best_a)
        for idx in top_idx:
            start_i = max(0, idx - n + 1)
            end_i = min(idx, n - 1)
            i_vec = np.arange(start_i, end_i + 1, dtype=np.int64)
            grad[i_vec] += 2.0 * best_a[idx - i_vec]
        grad /= len(top_idx)

        # momentum update
        update = -eta * grad + mu * momentum
        momentum = update

        a_new = best_a + update
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / s

        # Evaluate
        score_new = _compute_score(a_new)

        # Accept if better
        if score_new < best_score:
            best_score = score_new
            best_a = a_new
            print(f"Iter {iteration:6d}: new best {best_score:.6f} (eta {eta:.5f})")
        else:
            # occasional random jitter to possibly escape plateau
            if iteration % 200 == 0:
                jitter = 1e-4 * np.random.randn(n)
                a_new = np.maximum(best_a + jitter, 0.0)
                a_new *= target_sum / a_new.sum()
                s_new = _compute_score(a_new)
                if s_new < best_score:
                    best_score = s_new
                    best_a = a_new
                    print(f"Iter {iteration:6d}: jitter improved to {best_score:.6f}")

        # decay step size
        eta = max(eta * decay, eta_min)

    # final trim: maybe convert to list of floats
    return best_a.tolist()

import numpy as np
import time

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _compute_score(a):
    n = a.shape[0]
    nfft = _next_pow2(2*n-1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[:2*n-1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_c / (s * s))

def _make_sequence(p, beta, n, target_sum):
    # compute a_i = (i+beta)^(-p)
    idx = np.arange(n, dtype=np.float64)
    a = (idx + beta) ** (-p)
    # keep nonneg
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, target_sum / n)
    else:
        a *= target_sum / total
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a non‑negative step function minimizing
    2*n*max(conv(a,a))/sum(a)^2.  The algorithm combines:
    1) a parametric random search over power‑law shapes;
    2) smooth‑max gradient descent with FFT‑based gradients;
    3) a final local pairwise transfer (hill‑climbing) phase.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # safety margin

    # Decide length of the sequence.
    base_seq = globals().get('height_sequence_1')
    base_n = len(base_seq) if isinstance(base_seq, (list, tuple, np.ndarray)) else 5000
    # Aim for 2‑3× longer sequence, but cap at ~20000 for speed.
    max_n = 20000
    n = int(min(max_n, base_n * 3))
    if n < 5000:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # -----------------------------------------------------------------
    # 1) Random search in the (p, beta) family to obtain a good start.
    # -----------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # Include the classic 1/sqrt(i+0.5) shape as a candidate.
    a0 = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
    a0 *= target_sum / a0.sum()
    best_a = a0.copy()
    best_score = _compute_score(best_a)
    print(f"Initial classic shape score {best_score:.6f}")

    # Random sampling
    num_trials = 200
    for t in range(num_trials):
        # p around 0.5, beta moderate
        p = np.random.uniform(0.35, 0.65)
        beta = np.random.uniform(0.0, 3.0)
        a = _make_sequence(p, beta, n, target_sum)
        score = _compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Random trial {t:3d}: p={p:.4f} beta={beta:.3f} score={score:.6f}")
        if time.time() > deadline:
            break

    # -----------------------------------------------------------------
    # 2) Smooth‑max gradient descent (soft‑max approximation).
    # -----------------------------------------------------------------
    a = best_a.copy()
    best_seq = a.copy()
    best_score = _compute_score(a)

    # Pre‑compute FFT sizes.
    nfft_conv = _next_pow2(2 * n - 1)          # for a*a
    nfft_corr = _next_pow2(3 * n - 2)          # for weight ⊛ a_rev

    # Hyper‑parameters.
    gamma = 4.0           # start temperature for soft‑max
    gamma_max = 120.0
    eta = 0.15            # step size
    eta_min = 0.001
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # Compute convolution a * a.
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # Soft‑max weights.
        # Subtract max for numerical stability.
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
            w_sum = 1.0
        w /= w_sum

        # Gradient = 2 * ( w ⊛ a_rev )
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # Gradient step with projection.
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)

        # Renormalise to keep sum fixed.
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = target_sum / n
        else:
            a_new *= target_sum / s

        # Small jitter occasionally.
        if iteration % 67 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new *= target_sum / a_new.sum()

        score_new = _compute_score(a_new)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            a = a_new
            elapsed = time.time() - start

            if np.random.rand() < np.exp(-(score_new - best_score) / (eta + 1e-12)):

        # Cool / focus.
        gamma = min(gamma * 1.0025, gamma_max)
        eta = max(eta * 0.9985, eta_min)

        # If we are close to the deadline, break to start local search.
        if time.time() > deadline - 30.0:
            break

    # -----------------------------------------------------------------
    # 3) Final local search: random pairwise mass transfers.
    # -----------------------------------------------------------------
    a = best_seq.copy()
    best_score = _compute_score(a)
    # Ensure the sum stays exactly target_sum to keep evaluation stable.
    # (All moves preserve total mass.)
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Transfer a small fraction of a[i] to a[j]
        delta = 0.005 * a[i]  # 0.5 % of source value
        if delta <= 0.0:
            continue
        if a[i] < delta:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        score_new = _compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            a = a_new
            elapsed = time.time() - start
            print(f"Local move: improved to {best_score:.6f} time {elapsed:.1f}s")
        # occasionally try a larger random perturbation
        if np.random.rand() < 0.001:
            jitter = 2e-4 * np.random.randn(n)
            a_jit = np.maximum(a + jitter, 0.0)
            a_jit *= target_sum / a_jit.sum()

            if s_new < best_score:
                best_score = s_new
                a = a_jit
                print(f"Jitter move: new best {best_score:.6f}")

    # Return best seen sequence.
    print(f"Finished. Best score {best_score:.6f}")
    return a.tolist()

"""
Heuristic search for a non‑negative step function minimising

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The method consists of three stages:

1. **Parametric random search** – sample power‑law shapes
   a_i = (i + β)^–p, normalise them so that
   sum(a) = sqrt(2 n), and keep the best one.

2. **Smooth‑max gradient descent** – minimise a
   differentiable approximation of the maximum of the convolution.
   The gradient is obtained via FFT‑based cross‑correlation of the
   soft‑max weights with the reversed sequence.  The step size (η) and
   the soft‑max temperature (γ) are slowly annealed.

3. **Local pairwise mass transfers** – with the remaining time perform
   cheap hill‑climbing moves that transfer a tiny amount of mass from
   one index to another while preserving the total sum.  Improvements
   are kept instantly.

All operations respect the non‑negativity constraint and keep the
normalisation fixed, therefore the evaluation function is invariant to
scaling.  The algorithm runs for the whole time budget (≈ 1000 s) and
returns the best sequence discovered.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _make_power_law(p: float, beta: float, n: int, target_sum: float) -> np.ndarray:
    """
    Return a sequence a_i = (i + beta)^{-p}, normalised to have the given sum.
    """
    idx = np.arange(n, dtype=np.float64)
    a = (idx + beta) ** (-p)
    a = np.maximum(a, 0.0)                # safety
    total = a.sum()
    if total == 0.0:
        a[:] = target_sum / n
    else:
        a *= target_sum / total
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    The function respects the 1000 s time limit and returns a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Choose the length of the sequence.
    # ------------------------------------------------------------------
    existing = globals().get("height_sequence_1")
    base_n = len(existing) if isinstance(existing, (list, tuple, np.ndarray)) else 5000
    max_n = 20000                                   # keep FFT sizes reasonable
    n = int(min(max_n, base_n * 3))
    if n < 5000:
        n = 5000

    target_sum = np.sqrt(2.0 * n)                    # normalisation used by the original code

    # ------------------------------------------------------------------
    # 1) Random search over power‑law families.
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # Classic 1/sqrt(i+0.5) shape as a baseline.
    a0 = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
    a0 *= target_sum / a0.sum()
    best_a = a0.copy()
    best_score = _compute_score(best_a)
    print(f"[Init] classic shape score = {best_score:.6f}")

    trials = 200
    for t in range(trials):
        p = np.random.uniform(0.35, 0.65)          # around the known optimal exponent 0.5
        beta = np.random.uniform(0.0, 3.0)         # shift of the denominator
        a = _make_power_law(p, beta, n, target_sum)
        sc = _compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[Random] trial {t:3d}: p={p:.4f} beta={beta:.3f} score={sc:.6f}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) Smooth‑max gradient descent (soft‑max approximation of max).
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_seq = a.copy()
    best_score = _compute_score(a)

    # FFT lengths that stay constant throughout the optimisation.
    nfft_conv = _next_pow2(2 * n - 1)          # for a*a
    nfft_corr = _next_pow2(3 * n - 2)          # for weight ⊛ a_rev

    # Hyper‑parameters (will be annealed).
    gamma = 4.0            # soft‑max temperature (low → smoother)
    gamma_max = 120.0
    eta = 0.15             # learning rate
    eta_min = 0.001
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ---- convolution a * a -------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ---- soft‑max weights ---------------------------------------------------
        w = np.exp(gamma * (conv - max_conv))   # max weight = 1
        w_sum = w.sum()
        if w_sum == 0.0:                        # numerical guard
            w.fill(1.0 / w.size)
            w_sum = 1.0
        w /= w_sum

        # ---- gradient of the smooth maximum (2 * w ⊛ a_rev) --------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ---- gradient step with projection --------------------------------------
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)

        # renormalise to keep the prescribed sum (scaling does not affect the score)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = target_sum / n
        else:
            a_new *= target_sum / s

        # occasional tiny jitter to jump off plateaus
        if iteration % 67 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new *= target_sum / a_new.sum()

        score_new = _compute_score(a_new)

        # ---- acceptance ---------------------------------------------------------
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            a = a_new
            elapsed = time.time() - start_time
            print(f"[Grad] iter {iteration:6d}: new best {best_score:.6f}  η={eta:.5f}  time={elapsed:.1f}s")
        else:
            # simple simulated‑annealing style uphill move
            if np.random.rand() < np.exp(-(score_new - best_score) / (eta + 1e-12)):
                a = a_new

        # ---- schedule update ----------------------------------------------------
        gamma = min(gamma * 1.0025, gamma_max)
        eta = max(eta * 0.9985, eta_min)

        # Reserve a few seconds for the final local search.
        if time.time() > deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # 3) Local hill‑climbing: random transfers of a small amount of mass.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    best_score = _compute_score(a)

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Move a tiny fraction from i to j (preserves total sum).
        delta = 0.005 * a[i]                     # 0.5 % of the source entry
        if delta <= 0.0 or a[i] < delta:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        sc = _compute_score(a_new)
        if sc < best_score:
            best_score = sc
            a = a_new
            elapsed = time.time() - start_time
            print(f"[Local] improved to {best_score:.6f} at {elapsed:.1f}s")

        # Rare bigger perturbation to diversify search
        if np.random.rand() < 0.001:
            jitter = 2e-4 * np.random.randn(n)
            a_jit = np.maximum(a + jitter, 0.0)
            a_jit *= target_sum / a_jit.sum()
            sc_jit = _compute_score(a_jit)
            if sc_jit < best_score:
                best_score = sc_jit
                a = a_jit
                print(f"[Jitter] new best {best_score:.6f}")

    print(f"[Done] best score achieved: {best_score:.6f}")
    return a.tolist()
