def _convolution(a):
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n - 1]
    return conv

def _convolution_cross(a,b):
    n = a.size

    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:2*n - 1]

def _solve_lp_max_sum(f, M):
    n = f.size
    # Build constraint matrix A: shape (2n-1, n)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill:
    for j in range(n):
        k_start = j
        k_end = min(j + n, rows)
        length = k_end - k_start
        A[k_start:k_end, j] = f[:length]
    # Solve using scipy.optimize.linprog
    c = -np.ones(n, dtype=np.float64)  # minimize -sum = maximize sum
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n
    # Using scipy's high-level
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        print("LP failed; returning None")
        return None

"""
Propose a step‑function sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The scale of a does not affect the objective, therefore each candidate is
normalised to ‖a‖₁ = √(2·n).  The optimisation proceeds in two stages:

1.  Initialise a power‑law profile a_i ∝ (i+1)^{-p}.
2.  Repeatedly improve the shape by solving a linear programme

       max   Σ b_j
       s.t.  (a * b)_k ≤ M   for all k,
             b_j ≥ 0,

    where M = max(conv(a,a)).  The LP finds a direction b that can carry as
    much mass as possible without increasing the current maximal convolution.
    b is then rescaled to satisfy the ℓ₁‑norm constraint and mixed with a.
    The mixing weight t is chosen by a cheap line‑search on the exact
    expression for the mixed convolution, guaranteeing a monotone decrease
    of the objective.

The loop runs until the time budget expires (default 1000 s).  If SciPy is
unavailable a simple stochastic hill‑climber is used as a fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _convolution_cross(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return float(_convolution(a).max())


# ----------------------------------------------------------------------
# Linear‑program helper (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:                       # pragma: no cover
    SCIPY_AVAILABLE = False


def _solve_lp_max_sum(f: np.ndarray, M: float, L1_target: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M (∀k),   g ≥ 0,   g_j ≤ L1_target.
    Returns the optimal g as a dense NumPy vector, or None on failure.
    """
    if not SCIPY_AVAILABLE:               # pragma: no cover
        return None

    n = f.size
    rows = 2 * n - 1

    # Build the constraint matrix A where A[k, j] = f_{k-j} if 0≤k-j<n else 0.
    # This is a Toeplitz‑like matrix; we fill it column‑wise.
    A = np.empty((rows, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = min(j + n, rows)
        length = end - start
        A[start:end, j] = f[:length]

    # Linear programme data
    c = -np.ones(n, dtype=np.float64)               # minimise –Σg → maximise Σg
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, L1_target) for _ in range(n)]

    # Solve with HiGHS (fast interior‑point)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    # On failure (e.g. unbounded) we simply give up this iteration.
    return None


# ----------------------------------------------------------------------
# Stochastic hill‑climber fallback (used when SciPy is missing)
# ----------------------------------------------------------------------


def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Very simple mass‑transfer hill‑climber (used as a fallback)."""
    n = a.size
    best = a.copy()
    best_val = _max_convolution(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.12
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta
        val = _max_convolution(a)
        if val < best_val:
            best_val = val
            best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta
    return best, best_val


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) minimise the
    supplied `evaluate_sequence` function.  The routine spends up to
    `budget_s` seconds performing a deterministic initialisation followed
    by a Frank‑Wolfe‑style LP‑based refinement (or a stochastic hill‑climb
    if SciPy is unavailable).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ---------------------  configuration  ----------------------------
    n = 2000                       # main problem size (≈ √(2·n) ≈ 63)
    L1_target = float(np.sqrt(2.0 * n))

    # power‑law exponents to try as start points
    exponents = [0.45, 0.48, 0.5, 0.52, 0.55]

    # -----------------------------------------------------------------
    # Helper to normalise a vector to the required ℓ₁‑norm
    # -----------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        s = v.sum()
        if s == 0.0:
            return v
        return v * (L1_target / s)

    # -----------------------------------------------------------------
    # Main loop over different initial exponents
    # -----------------------------------------------------------------
    for p in exponents:
        if time.time() >= deadline:
            break

        # initial power‑law profile a_i ∝ (i+1)^{-p}
        a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a = _normalise(a0)

        # evaluate initial candidate
        cur_score = _max_convolution(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] p={p:.3f}  score={cur_score:.6f}")

        # -----------------------------------------------------------------
        # Iterative improvement: LP + optimal mixing
        # -----------------------------------------------------------------
        iteration = 0
        while time.time() < deadline:
            iteration += 1
            # ----- compute current max‑convolution -----------------------
            conv_ff = _convolution(a)
            M = float(conv_ff.max())

            # ----- solve the LP (direction g) ---------------------------
            g = _solve_lp_max_sum(a, M, L1_target)

            # If the LP could not be solved (SciPy missing) fall back.
            if g is None:
                # -------------------------------------------------------
                # fallback: simple stochastic hill‑climb for the rest of time
                # -------------------------------------------------------
                sub_deadline = deadline
                a, cur_score = _hill_climb(a, sub_deadline)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[hill‑climb] new best={cur_score:.6f}")
                break

            # ----- rescale g to have the required ℓ₁‑norm ---------------
            sum_g = float(g.sum())
            if sum_g <= 1e-12:
                break
            scale = L1_target / sum_g
            g_scaled = g * scale

            # ----- compute the three convolutions needed for line search -
            conv_fg = _convolution_cross(a, g_scaled)
            conv_gg = _convolution(g_scaled)

            # ----- line search on mixing weight t ∈ [0,1] ----------------
            t_grid = np.linspace(0.0, 1.0, 51)   # step = 0.02
            t_arr = t_grid[:, None]             # shape (T,1)
            # convex combination formula for the paired convolution
            conv_t = ((1 - t_arr) ** 2) * conv_ff \
                     + (2 * t_arr * (1 - t_arr)) * conv_fg \
                     + (t_arr ** 2) * conv_gg
            M_vals = conv_t.max(axis=1)
            idx_best = int(np.argmin(M_vals))
            t_opt = float(t_grid[idx_best])
            M_opt = float(M_vals[idx_best])

            # ----- accept the improvement if it really helps -------------
            if M_opt < best_score - 1e-12:
                a = (1.0 - t_opt) * a + t_opt * g_scaled
                # re‑normalise (numerical drift)
                a = _normalise(a)
                best_score = M_opt
                best_seq = a.copy()
                print(f"[iter {iteration:4d}] p={p:.3f}  t={t_opt:.3f}  score={best_score:.6f}")
            else:
                # No progress – stop this exponent and move to the next one.
                break

    # -----------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        # simple uniform vector (guaranteed to be valid)
        n_fallback = 500
        fallback = np.ones(n_fallback, dtype=np.float64)
        best_seq = (fallback / fallback.sum()) * np.sqrt(2.0 * n_fallback)

    # Trim trailing zeros for neatness (evaluate_sequence would ignore them anyway)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
