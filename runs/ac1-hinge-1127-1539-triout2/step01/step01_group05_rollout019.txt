```
def project_onto_simplex(v, s):
    """Projection of vector v onto the simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("s must be nonnegative")
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho = max {j | u_j - cssv_j / j > 0}
    # which is: find first index where u_j - cssv_j / j <= 0, then rho = that - 1
    # but easier: compute condition and find rho
    cond = u - cssv / ind > 0
    if not cond.any():
        # All negative, return zero vector?
        # But must sum = s; if s>0 cannot. fallback uniform
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # due to numeric issues, ensure sum approx s
    # Adjust using scaling
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    else:
        w = np.full_like(v, s / n)
    return w
```

    # Keep best
    best_a = a.copy()
    best_score = _compute_score(best_a)

    # hyperparameters
    gamma = 1.0
    gamma_max = 200.0
    gamma_inc = 1.002
    lr = 0.2
    lr_min = 0.001
    lr_decay = 0.999

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    nfft_conv = _next_pow2(2*n-1)
    nfft_corr = _next_pow2(3*n-2)

    while time.time() < deadline:
        t_iter += 1
        # Compute convolution
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[:2*n-1]

        max_conv = conv.max()
        # if gamma < gamma_max use soft gradient else subgradient
        if gamma < gamma_max:
            # softmax
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
                w_sum = 1.0
            w /= w_sum

            # cross-correlation of w with a_rev
            wf = np.fft.rfft(w, n=nfft_corr)
            af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
            corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
            grad = 2.0 * corr_full[n-1:n-1+n]
        else:
            # subgradient from max indices
            max_indices = np.flatnonzero(conv == max_conv)
            idx = max_indices[np.random.randint(len(max_indices))]
            # compute gradient vector 2*a[idx-i] for valid i
            grad = np.zeros_like(a)
            # compute i range
            i_low = max(0, idx - (n-1))
            i_high = min(idx, n-1)
            i_range = np.arange(i_low, i_high+1)
            j = idx - i_range
            grad[i_range] = 2.0 * a[j]

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t_iter)

            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # Evaluate

            best_a = a_new.copy()

    return best_a.tolist()

"""
Propose a step‐function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation is exactly the maximal entry of the self‑convolution when the
L1‑norm is fixed to √(2 n).  The algorithm consists of three stages:

1. **Coarse parameter search** –  the family   a_i = C·(i+β)^(-p)   (p≈½,
   β>0) is examined on a fine grid; the best parameters give a good
   starting point.

2. **Smoothed‑max / sub‑gradient descent** – a standard Adam optimiser is
   applied to the (soft) maximum of the convolution.  The temperature
   γ of the soft‑max is slowly increased; once it reaches a large value the
   true sub‑gradient (coming from the current maximal index) is used.
   After each update the vector is projected onto the simplex
   {a ≥ 0, Σa = √(2 n)}.

3. **Random pairwise swaps** – when the time budget is almost exhausted a
   cheap hill‑climbing phase moves a tiny amount of mass from one index to
   another and keeps the change if it improves the score.

All steps are performed with FFT‑based convolutions (O(n log n)) and
respect the 1000 s time limit.  The best sequence found is returned as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # With the normalisation Σa = √(2 n) the factor 2 n / Σa² = 1,
    # therefore the score equals the maximal convolution entry.
    return float(max_c)


def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All components would be zero – fallback to uniform vector
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: re‑scale to exact sum if necessary
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    else:
        w = np.full_like(v, s / n)
    return w


def build_shape(n: int, target_sum: float, p: float, beta: float) -> np.ndarray:
    """Return a decreasing shape a_i = C·(i+β)^{-p} with Σa = target_sum."""
    a = (np.arange(n, dtype=np.float64) + beta) ** (-p)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def _optimise_one_seed(
    n: int,
    target_sum: float,
    deadline: float,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """Run the Adam‑based (soft‑max → sub‑gradient) optimisation."""
    np.random.seed(seed)

    # ---- 1. coarse grid for (p, beta) ---------------------------------
    best_a = None
    best_score = np.inf
    # reasonable ranges discovered experimentally
    p_vals = np.linspace(0.44, 0.58, 15)        # around 0.5
    beta_vals = np.linspace(0.01, 5.0, 25)     # shift of the singularity
    for p in p_vals:
        for beta in beta_vals:
            a = build_shape(n, target_sum, p, beta)
            sc = _compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # -------------------------------------------------------------------
    # 2. Adam optimisation with soft‑max temperature scheduling
    # -------------------------------------------------------------------
    a = best_a.copy()
    best_a = a.copy()
    best_score = _compute_score(a)

    # FFT sizes (pre‑computed for speed)
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)  # needed for soft‑max gradient

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # schedule parameters
    gamma = 5.0          # start temperature for soft‑max
    gamma_max = 150.0    # very high – makes the soft‑max almost a hard max
    gamma_inc = 1.003
    lr = 0.25            # base learning rate (Adam multiplies this)
    lr_min = 5e-4
    lr_decay = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- compute convolution (real FFT) -----
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- gradient computation -----
        if gamma < gamma_max:          # soft‑max (smooth) gradient
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
                w_sum = 1.0
            w /= w_sum

            # cross‑correlation w ⊛ a_rev
            wf = np.fft.rfft(w, n=nfft_corr)
            af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
            corr = np.fft.irfft(wf * af_rev, n=nfft_corr)
            grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n
        else:                           # true sub‑gradient from the active index
            # pick one maximal index (if several, randomise)
            max_idxs = np.flatnonzero(conv == max_conv)
            k = max_idxs[np.random.randint(len(max_idxs))]
            # vectorised construction of 2·a_{k‑i}
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[j[mask]]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # projection onto the simplex (non‑negative + fixed L1 norm)
        a_new = project_onto_simplex(a_new, target_sum)

        # occasional tiny random jitter (helps escape plateaus)
        if it % 97 == 0:
            jitter = 1e-5 * np.random.randn(n)
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        score_new = _compute_score(a_new)

        # keep the globally best sequence
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - (deadline - (deadline - (deadline - time.time())))
            print(f"Iter {it:6d}: new best score {best_score:.6f}")

        # accept the candidate for the next iteration if it is not much worse
        # (simple “optimistic” acceptance)
        if score_new <= best_score * 1.001:
            a = a_new

        # ----- schedule updates -----
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

    # -------------------------------------------------------------------
    # 3. Final cheap hill‑climbing with pairwise swaps (very short)
    # -------------------------------------------------------------------
    a = best_a.copy()
    while time.time() < deadline - 0.1:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a tiny amount δ from i to j (δ chosen as a small fraction of a_i)
        delta = 0.001 * a[i] * (np.random.rand() - 0.5)   # can be positive or negative
        if a[i] - delta < 0 or a[j] + delta < 0:
            continue
        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        # re‑project only to fix possible tiny rounding errors
        a_trial = project_onto_simplex(a_trial, target_sum)
        sc = _compute_score(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            print(f"Hill climb: improved to {best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    phase.  The best sequence discovered within the time budget is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # --------------------------------------------------------------
    # Choose length n (use the provided construction if present)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
    except Exception:
        n = 5600                                     # a reasonable default
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Run a few independent optimisation runs (different seeds)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    run = 0
    while time.time() < deadline and run < 3:
        run += 1
        # give each run a different random seed
        run_seed = seed + run * 12345
        seq, val = _optimise_one_seed(
            n=n,

    # If, for any reason, optimisation produced nothing, fall back to a simple shape
    if best_seq is None:
        best_seq = build_shape(n, target_sum, p=0.5, beta=0.5)

    # Return as plain Python list (as required by the external harness)
    return best_seq.tolist()
